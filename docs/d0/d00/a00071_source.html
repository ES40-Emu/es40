373void CAliM1543C::init()
374{
375 add_function(0, ali_cfg_data, ali_cfg_mask);
376
377 int i;
378 char* filename;
379
380 add_legacy_io(1, 0x61, 1);
381
382 state.reg_61 = 0;
383
384 add_legacy_io(2, 0x70, 4);
385 cSystem->RegisterMemory(this, 2, U64(0x00000801fc000070), 4);
386 for (i = 0; i < 4; i++)
387 state.toy_access_ports[i] = 0;
388 for (i = 0; i < 256; i++)
389 state.toy_stored_data[i] = 0;
390 state.toy_offset = 0;
391
392 // restore cmos/mc146818 state from TOY backing file.
393 restore_toy_nvram();
394
395 // The MC146818 is battery-backed: register A keeps its divider/rate across power
396 // cycles, so on real hardware the 1024Hz periodic tick runs from power-on. SRM's
397 // boot-CPU speed calibration (at "lowering IPL", BEFORE it programs the RTC at
398 // "create timer") depends on that: with no ticks it times out after ~0.7s and
399 // falls back to a hardcoded 2800ps cycle time = the mystery 357 MHz readout.
400 state.toy_stored_data[0x0a] = 0x26; // divider on, periodic rate 1024Hz
401
402 arc_year_compat = myCfg->get_myParent()->get_bool_value("arc_year_compat", false);
403
404 // Optional absolute time override from sys0:
405 // time = "YYYY-MM-DD" or "YYYY-MM-DD HH:MM:SS"
406 // Interpreted as UTC to match the runtime ARC/SRM TOY edit path
407 // (see toy_write SET_TIME below). Sets the initial toy_offset; subsequent
408 // guest TOY writes still take precedence and overwrite this offset.
409 char* faketime = myCfg->get_myParent()->get_text_value("time");
410 if (faketime)
411 {
412 struct tm ft = {};
413 int n = sscanf(faketime, "%d-%d-%d %d:%d:%d",
414 &ft.tm_year, &ft.tm_mon, &ft.tm_mday,
415 &ft.tm_hour, &ft.tm_min, &ft.tm_sec);
416 if (n < 3)
417 FAILURE_1(Configuration,
418 "Invalid time format: %s (use YYYY-MM-DD or "
419 "YYYY-MM-DD HH:MM:SS)",
420 faketime);
421 ft.tm_year -= 1900;
422 ft.tm_mon -= 1;
423#ifdef _WIN32
424 time_t set_time = _mkgmtime(&ft);
425#else
426 time_t set_time = timegm(&ft);
427#endif
428 if (set_time == (time_t)-1)
429 FAILURE_1(Configuration, "Invalid time value: %s", faketime);
430 time_t host_now;
431 time(&host_now);
432 state.toy_offset = (long)(set_time - host_now);
433 }
434
435 state.toy_stored_data[0x17] = myCfg->get_bool_value("vga_console") ? 1 : 0;
436
437 if (state.toy_stored_data[0x17] && !cSystem->has_vga_device())
438 {
439 printf("! CONFIGURATION WARNING ! vga_console set to true, but no VGA card installed.\n");
440 state.toy_stored_data[0x17] = 0;
441 }
442
443 ResetPCI();
444
445 // PIT Setup
446 add_legacy_io(6, 0x40, 4);
447 for (i = 0; i < 3; i++)
448 state.pit_status[i] = 0x40; // invalid/null counter
449 for (i = 0; i < 9; i++)
450 state.pit_counter[i] = 0;
451 m_pit_last = std::chrono::steady_clock::now();
452 for (i = 0; i < 3; i++)
453 m_pit_epoch[i] = m_pit_last;
454 m_pit_acc = 0;
455
456 add_legacy_io(7, 0x20, 2);
457 add_legacy_io(8, 0xa0, 2);
458 add_legacy_io(30, 0x4d0, 2);
459 add_legacy_io(9, 0x22, 2); // PIC backdoor control
460 state.pic_control_index = 0;
461
462 // odd one, byte read in PCI IACK (interrupt acknowledge) cycle. Interrupt vector.
463 cSystem->RegisterMemory(this, 20, U64(0x00000801f8000000), 1);
464
465 for (i = 0; i < 2; i++)
466 {
467 state.pic_elcr[i] = 0;
468 state.pic_ltim[i] = 0;
469 pic_init_reset(i);
470 }
471
472 // Initialize parallel port - IO registration is SuperIO driven, so not here, just output handling
473 filename = myCfg->get_text_value("lpt.outfile");
474 if (filename)
475 {
476 lpt = fopen(filename, "ab");
477 }
478 else
479 {
480 lpt = NULL;
481 }
482
483 lpt_reset();
484
485 // SuperIO
486 add_legacy_io(40, 0x370, 2);
487
488 // Populate the SuperIO chip-level / LDN register banks
489 // power on defaults
490 superio_reset();
491
492 // Docking windows may forward to ISA even when no modeled dev exists.
493 add_legacy_io(60, 0, 0x10000);
494
495 // ISA Plug-and-Play address (0x279) and write-data (0xA79) ports.
496 // The OS-selectable READ_DATA port (id 52) is registered dynamically
497 // when the OS programs it via PnP register 0x00.
498 add_legacy_io(50, 0x279, 1);
499 add_legacy_io(51, 0xa79, 1);
500 state.isapnp_state = PNP_WAIT_FOR_KEY;
501 state.isapnp_key_pos = 0;
502 state.isapnp_reg = 0;
503 state.isapnp_rd_port = 0;
504 state.isapnp_wake_csn = 0;
505
506 myRegLock = new CMutex("ali-reg");
507
508 myThread = 0;
509
510 printf("%s: $Id$\n",
511 devid_string);
512}

548bool CAliM1543C::uses_subtractive_decode(int index, u64, int, bool write) const noexcept
549{
550 // M1543C 3.5(g/h): NMI/RTC and external ISA are normally subtractive.
551 switch (index)
552 {
553 case 1: case 2:
554 case 9: case 27: case 40: case 50: case 51: case 52:
555 return !is_docking_mode();
556 }
557
558 // M1543C 3.5(b/e/h): 44h<6> controls PIT/PIC and IACK reads.
559 if (index == 6 || index == 7 || index == 8 || index == 30 ||
560 (index == 20 && !write))
561 return programmable_io_is_subtractive();
562 return false;
563}

575u32 CAliM1543C::config_read_custom(int func, u32 address, int dsize, u32 data)
576{
577 // M1543C 5Fh<2>, p.71: hidden M7101 registers 7Ch-FFh alias here.
578 if (func != 0 || !is_pmu_hidden() || !docking_pmu ||
579 (dsize != 8 && dsize != 16 && dsize != 32))
580 return data;
581 for (u32 lane = 0; lane < (u32)dsize / 8; ++lane)
582 if (address + lane >= 0x7c && address + lane < 0x100)
583 {
584 const u32 value = docking_pmu->config_read(0, address + lane, 8);
585 data = (data & ~(0xffU << (lane * 8))) | (value << (lane * 8));
586 }
587 return data;
588}

590void CAliM1543C::config_write_custom(int func, u32 address, int dsize,
591 u32, u32, u32 data)
592{
593 if (func != 0 || !is_pmu_hidden() || !docking_pmu ||
594 (dsize != 8 && dsize != 16 && dsize != 32))
595 return;
596 // The generic write hook receives data in stored byte order.
597 if (dsize == 16)
598 data = endian_16((u16)data);
599 else if (dsize == 32)
600 data = endian_32(data);
601 for (u32 lane = 0; lane < (u32)dsize / 8; ++lane)
602 if (address + lane >= 0x7c && address + lane < 0x100)
603 // Use the PMU's own mask and register handling for every byte lane.
604 docking_pmu->config_write(0, address + lane, 8,
605 (data >> (lane * 8)) & 0xff);
606}

614CAliM1543C::IsaIoDecode CAliM1543C::isa_io_decode(u32 port) const noexcept
615{
616 if (!is_docking_mode())
617 return IsaIoDecode::Subtractive;
618
619 // PCI positive I/O decode compares every address bit; no high-bit alias.
620 if (port > 0xffff)
621 return IsaIoDecode::None;
622
623 // DMDC 5D:5C (M1543C p.69) enables the UNION of address windows.
624 // Selectors belong to M7101.
625 const u32 dmdc = endian_32(pci_state.config_data[0][0x5c / 4]);
626 if ((dmdc & 0x04) && port >= 0x3b0 && port <= 0x3df)
627 return IsaIoDecode::Positive;
628 if ((dmdc & 0x20) && (port == 0x60 || port == 0x64))
629 return IsaIoDecode::Positive;
630
631 if (dmdc & 0x08)
632 {
633 const u32 selector = docking_pmu ? docking_pmu->docking_config(0x68) : 0;
634 const u32 base = (selector & 1) ? 0x370 : 0x3f0;
635 if (port >= base && port < base + 8)
636 return IsaIoDecode::Positive;
637 }
638
639 if (dmdc & 0x02)
640 {
641 const u32 selector = docking_pmu ? docking_pmu->docking_config(0x6c) : 0;
642 // M7101 6D:6C<15:2>, p.140. The book explicitly prints 338h for ADLIB.
643 static const u16 bases[] = { 0x200, 0x338, 0x300, 0x310, 0x320, 0x330,
644 0x220, 0x240, 0x260, 0x280, 0x530, 0x604, 0xe80, 0xf40 };
645 static const u8 lengths[] = { 8, 4, 4, 4, 4, 4, 20, 20, 20, 20, 8, 8, 8, 8 };
646 for (unsigned i = 0; i < sizeof(bases) / sizeof(bases[0]); ++i)
647 if ((selector & (1U << (i + 2))) &&
648 port >= bases[i] && port < (u32)bases[i] + lengths[i])
649 return IsaIoDecode::Positive;
650 }
651
652 if (dmdc & 0x50)
653 {
654 const u32 selector = docking_pmu ? docking_pmu->docking_config(0x70) : 0;
655 if (dmdc & 0x10)
656 {
657 static const u16 bases[] = { 0x3f8, 0x2f8, 0x3e8, 0x2e8,
658 0x220, 0x228, 0x238, 0x338 };
659 for (unsigned i = 0; i < sizeof(bases) / sizeof(bases[0]); ++i)
660 if ((selector & (1U << i)) && port >= bases[i] && port < (u32)bases[i] + 8)
661 return IsaIoDecode::Positive;
662 }
663 if (dmdc & 0x40)
664 {
665 // 71h<2:0> are address selectors; DRQ/event enables are not ranges.
666 if (((selector & 0x100) && port >= 0x378 && port <= 0x37f) ||
667 ((selector & 0x200) && port >= 0x278 && port <= 0x27f) ||
668 ((selector & 0x400) && port >= 0x3bc && port <= 0x3be))
669 return IsaIoDecode::Positive;
670 }
671 }
672
673 if (dmdc & 0x200)
674 {
675 // A5:A4<1:0> mask address bits 3:2; A1:0 are always ignored (p.156).
676 const u32 selector = docking_pmu ? docking_pmu->docking_config(0xa4) : 0;
677 const u32 mask = 0xfffcU & ~((selector & 3) << 2);
678 if (((port ^ selector) & mask) == 0)
679 return IsaIoDecode::Positive;
680 }
681 return IsaIoDecode::None;
682}

692bool CAliM1543C::decodes_memory_access(int index, u64 address, int dsize,
693 bool write) const noexcept
694{
695 u32 base;
696 u32 length;
697 switch (index)
698 {
699 case 9: base = 0x22; length = 2; break;
700 case 27:
701 base = (state.superio_ldn_regs[3][0x60] << 8) | state.superio_ldn_regs[3][0x61];
702 if (!(state.superio_ldn_regs[3][0x30] & 1) || base == 0)
703 return false;
704 base &= 0xfffc;
705 length = 4;
706 break;
707 case 40:
708 return address < 2 && superio_claims_port(0x370 + (u32)address, write);
709 case 50: base = 0x279; length = 1; break;
710 case 51: base = 0xa79; length = 1; break;
711 case 52:
712 base = state.isapnp_rd_port;
713 if (base == 0)
714 return false;
715 length = 1;
716 break;
717 case 60:
718 return is_docking_mode() && address <= 0xffff &&
719 isa_io_decode((u32)address) == IsaIoDecode::Positive;
720 default:
721 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
722 }
723 return address < length && isa_io_decode(base + (u32)address) != IsaIoDecode::None;
724}

890{

905void CAliM1543C::reg_61_write(u8 data)
906{
907 const bool gate_was_high = (state.reg_61 & 0x01) != 0;
908 const bool gate_is_high = (data & 0x01) != 0;
909 state.reg_61 = (state.reg_61 & 0xf0) | (((u8)data) & 0x0f);
910 const int mode = (state.pit_status[2] & 0x0e) >> 1;
911 if (gate_was_high && !gate_is_high && mode != 0)
912 state.pit_status[2] |= 0x80;
913 if (!gate_was_high && gate_is_high)
914 {

980 state.superio_ldn_regs[4][0xf0] = 0x00;

1134void CAliM1543C::superio_apply_ldn(int ldn)
1135{
1136 const u8 activate = state.superio_ldn_regs[ldn][0x30] & 0x01;
1137 const u16 base = (u16)((state.superio_ldn_regs[ldn][0x60] << 8) |
1138 state.superio_ldn_regs[ldn][0x61]);
1139 const u8 irq1 = state.superio_ldn_regs[ldn][0x70];
1140 const u8 irq2 = state.superio_ldn_regs[ldn][0x72];
1141
1142 switch (ldn) {
1143 case 0: // FDC. CFloppyController binds 0x3f0-0x3f7 statically;
1144 // honoring an OS-driven relocation here would require a hook into
1145 // that class. For now just acknowledge the activation so PnP
1146 // enumeration sees a coherent picture.
1147#ifdef DEBUG_SUPERIO
1148 printf("%s: FDC %s (base=%#06x irq=%u)\n", devid_string,
1149 activate ? "activated" : "deactivated", base, irq1);
1150#endif
1151 break;
1152
1153 case 3: // LPT — only LDN with dynamic binding implemented.
1154 add_legacy_io(27, base & 0xfffc, activate && base != 0 ? 4 : 0);
1155 if (activate && base != 0) {
1156#ifdef DEBUG_SUPERIO
1157 printf("%s: LPT activated at %#06x\n", devid_string, base & 0xfffc);
1158#endif
1159 }
1160#ifdef DEBUG_SUPERIO
1161 else {
1162 printf("%s: LPT deactivated (activate=%u base=%#06x)\n",
1163 devid_string, activate, base);
1164 }
1165#endif
1166 break;
1167
1168 case 4: // UART1 (COM1). CSerial binds its ports per-instance from
1169 // the configurator.
1170 case 5: // UART2 (COM2).
1171#ifdef DEBUG_SUPERIO
1172 printf("%s: COM%d %s (base=%#06x irq=%u)\n", devid_string, ldn - 3,
1173 activate ? "activated" : "deactivated", base, irq1);
1174#endif
1175 break;
1176
1177 case 7: // Keyboard controller. CKeyboard binds 0x60/0x64 statically;
1178 // the LDN advertises the same so the OS sees a self-consistent
1179 // SuperIO view.
1180#ifdef DEBUG_SUPERIO
1181 printf("%s: KBC %s (base=%#06x kbd_irq=%u aux_irq=%u)\n",
1182 devid_string, activate ? "activated" : "deactivated",
1183 base, irq1, irq2);
1184#endif
1185 (void)irq2;
1186 break;
1187
1188 case 8: // AUX (mouse) — IRQ wire lives on the KBC; nothing to bind.
1189#ifdef DEBUG_SUPERIO
1190 printf("%s: AUX %s (irq=%u)\n", devid_string,
1191 activate ? "activated" : "deactivated", irq1);
1192#endif
1193 break;
1194
1195 default:
1196#ifdef DEBUG_SUPERIO
1197 printf("%s: LDN %#x apply (activate=%u base=%#06x)\n",
1198 devid_string, ldn, activate, base);
1199#endif
1200 break;
1201 }

1235void CAliM1543C::isapnp_addr_write(u8 data)
1236{
1237#ifdef DEBUG_ISAPNP
1238 printf("%%PNP-I-ADDR: write 0x%02x to 0x279 (state=%d, key_pos=%d)\n",
1239 data, state.isapnp_state, state.isapnp_key_pos);
1240#endif
1241
1242 if (state.isapnp_state == PNP_WAIT_FOR_KEY)
1243 {
1244 // Per spec 4.3.2: the LFSR is reset by writing 0x00 to ADDRESS
1245 // twice in succession, after which the 32-byte initiation key
1246 // must arrive byte-by-byte without interruption. Any byte that
1247 // fails to match the expected position resets our match counter.
1248 if (data == 0x00 && state.isapnp_key_pos < 2)
1249 {
1250 state.isapnp_key_pos = 0;
1251 return;
1252 }
1253
1254 if (data == isapnp_init_key[state.isapnp_key_pos])
1255 {
1256 state.isapnp_key_pos++;
1257 if (state.isapnp_key_pos >= 32)
1258 {
1259#ifdef DEBUG_ISAPNP
1260 printf("%%PNP-I-KEY: initiation key accepted, entering SLEEP\n");
1261#endif
1262 state.isapnp_state = PNP_SLEEP;
1263 state.isapnp_key_pos = 0;
1264 }
1265 }
1266 else
1267 {
1268 state.isapnp_key_pos = 0;
1269 }
1270 return;
1271 }
1272
1273 // Past the key — every ADDRESS write selects the register that the
1274 // next 0xA79 / READ_DATA access will operate on.
1275 state.isapnp_reg = data;
1276}

1278void CAliM1543C::isapnp_data_write(u8 data)
1279{
1280#ifdef DEBUG_ISAPNP
1281 printf("%%PNP-I-DATA: write 0x%02x to reg 0x%02x (state=%d)\n",
1282 data, state.isapnp_reg, state.isapnp_state);
1283#endif
1284
1285 if (state.isapnp_state == PNP_WAIT_FOR_KEY)
1286 return;
1287
1288 switch (state.isapnp_reg)
1289 {
1290 case 0x00: // Set RD_DATA Port — actual port = (data << 2) | 0x0003
1291 {
1292 u16 new_port = ((u16)data << 2) | 0x0003;
1293 state.isapnp_rd_port = new_port;
1294 // Re-register our READ_DATA hook at the OS-chosen port. The
1295 // system memory map updates id 52 in place, so successive
1296 // programmings just relocate the same registration.
1297 add_legacy_io(52, new_port, 1);
1298#ifdef DEBUG_ISAPNP
1299 printf("%%PNP-I-RDPORT: READ_DATA port set to 0x%03x\n", new_port);
1300#endif
1301 break;
1302 }
1303
1304 case 0x02: // Config Control
1305 // bit 0: Reset (drives RESET_DRV — all cards back to Wait-For-Key)
1306 // bit 1: Wait For Key (return to Wait-For-Key)
1307 // bit 2: Reset CSN (clear all CSNs to zero)
1308 if (data & 0x03)
1309 {
1310 state.isapnp_state = PNP_WAIT_FOR_KEY;
1311 state.isapnp_key_pos = 0;
1312 }
1313 break;
1314
1315 case 0x03: // Wake[CSN]
1316 state.isapnp_wake_csn = data;
1317 // CSN==0 with a card in Sleep starts isolation; non-zero wakes
1318 // the card with that CSN into Config. We have no cards either
1319 // way, but tracking the state lets future reads of register 0x05
1320 // (Status) return something coherent if asked.
1321 if (state.isapnp_state == PNP_SLEEP)
1322 state.isapnp_state = (data == 0) ? PNP_ISOLATION : PNP_CONFIG;
1323 break;
1324
1325 case 0x06: // Set CSN — assign CSN to the card that won isolation.
1326 // No card won, no-op.
1327 break;
1328
1329 default:
1330 // Logical-device select (0x07), card-level resource registers
1331 // (0x20-0x2F) and per-LDN resource registers (0x30-0xFF) are
1332 // per-card. No cards, nothing to record.
1333 break;
1334 }
1335}

1729void CAliM1543C::pit_write(u32 address, u8 data)
1730{
1731 if (address == 3)
1732 {
1733 const int counter = (data >> 6) & 3;
1734 if (counter == 3)
1735 {
1736 state.pit_status[3] = data;
1737 return;
1738 }
1739
1740 const int access = (data >> 4) & 3;
1741 if (access == 0)
1742 {
1743 /* Do not replace an unread latch. */
1744 u32& latch = state.pit_counter[counter + PIT_OFFSET_LATCH];
1745 if ((latch & PIT_LATCH_VALID) == 0)
1746 latch = (state.pit_counter[counter] & 0xFFFFU) | PIT_LATCH_VALID;
1747 return;
1748 }
1749
1750 int mode = (data >> 1) & 7;
1751 if (mode >= 6)
1752 mode -= 4;
1753 state.pit_status[counter] = (u8)((data & 0x31) | (mode << 1) | 0x40);
1754 if (mode != 0)
1755 state.pit_status[counter] |= 0x80;
1756 state.pit_mode[counter] = (u8)access;
1757 state.pit_counter[counter + PIT_OFFSET_LATCH] = 0;
1758 return;
1759 }
1760
1761 if (address < 3)
1762 {
1763 const int access = (state.pit_status[address] & 0x30) >> 4;
1764 u32 count = 0;

1766 {
1767 case 1: // low byte only
1768 count = data;
1769 break;
1770
1771 case 2: // high byte only
1772 count = (u32)data << 8;
1773 break;
1774
1775 case 3: // low byte followed by high byte
1776 if (state.pit_mode[address] == 3)
1777 {
1778 state.pit_counter[address + PIT_OFFSET_LATCH] = data;
1779 state.pit_mode[address] = 2;
1780 return;
1781 }
1782 count = (state.pit_counter[address + PIT_OFFSET_LATCH] & 0xFFU) |
1783 ((u32)data << 8);
1784 state.pit_mode[address] = 3;
1785 break;
1786
1787 default:
1788 return;
1789 }
1790
1791 if (count == 0)
1792 count = 65536;
1793 state.pit_counter[address] = count;
1794 state.pit_counter[address + PIT_OFFSET_MAX] = count;
1795 state.pit_counter[address + PIT_OFFSET_LATCH] = 0;
1796 state.pit_status[address] &= ~0x40;

1798 state.pit_status[address] &= ~0x80;

1810bool CAliM1543C::pit_out(int c)
1811{
1812 const int mode = (state.pit_status[c] & 0x0e) >> 1;
1813 if (state.pit_status[c] & 0x40)
1814 return mode != 0; // after a control word, mode 0 is low; others are high
1815 if (c == 2 && (state.reg_61 & 0x01) == 0)
1816 {
1817 // GATE2 low inhibits counter 2.
1818 return mode == 0 ? (state.pit_status[c] & 0x80) != 0 : true;
1819 }
1820 const u32 n = state.pit_counter[c + PIT_OFFSET_MAX];
1821 if (!n)
1822 return true;
1823 // microsecond granularity: ns * PIT_CLOCK_HZ would overflow u64 after ~4h uptime
1824 const u64 clocks = (u64)std::chrono::duration_cast<std::chrono::microseconds>(
1825 std::chrono::steady_clock::now() - m_pit_epoch[c]).count() * PIT_CLOCK_HZ / 1000000ull;
1826 switch (mode)
1827 {
1828 case 0: // interrupt on terminal count: low while counting, high at terminal
1829 return clocks >= n;
1830
1831 case 3: // square wave: high for the first half of each period
1832 return ((clocks % n) * 2) < n;
1833
1834 default: // unmodelled mode: fall back to the decrement machinery's view
1835 return (state.pit_status[c] & 0x80) != 0;
1836 }

1847void CAliM1543C::pit_clock()
1848{
1849 // Convert elapsed host time to 1.193182 MHz input clocks. The pacing must
1850 // be wall-clock: SRM calibrates the boot CPU's speed against this timer,
1851 // and the guest RPCC it compares against is wall-clock true.
1852 const auto now = std::chrono::steady_clock::now();
1853 u64 elapsed_ns = (u64)std::chrono::duration_cast<std::chrono::nanoseconds>(now - m_pit_last).count();
1854 m_pit_last = now;
1855 if (elapsed_ns > 30000000ull) // stall/restore catch-up cap: under one ch0 half-period,
1856 elapsed_ns = 30000000ull; // so toggles can never bunch into one call
1857 m_pit_acc += elapsed_ns * PIT_CLOCK_HZ;
1858 const u64 clocks = m_pit_acc / 1000000000ull;
1859 m_pit_acc %= 1000000000ull;
1860 if (!clocks)
1861 return;
1862
1863 for (int i = 0; i < 3; i++)
1864 {
1865 if (state.pit_status[i] & 0x40)
1866 continue;
1867 if (i == 2 && (state.reg_61 & 0x01) == 0)
1868 continue;
1869 switch ((state.pit_status[i] & 0x0e) >> 1)
1870 {
1871 case 0: // interrupt at terminal
1872 if (state.pit_counter[i] <= clocks)
1873 {
1874 state.pit_counter[i] = 0;
1875 // OUT high only: NULL COUNT (0x40) would make pit_out() read
1876 // mode 0 as low, hanging port-61h pollers that miss the edge.
1877 state.pit_status[i] |= 0x80;
1878 }
1879 else
1880 state.pit_counter[i] -= (u32)clocks;
1881 break;
1882
1883 case 3: // square wave generator: counts down by two per input clock,
1884 // output toggles at terminal, so one output period = one reload value
1885 {
1886 u64 dec = clocks * 2;
1887 const u32 reload = state.pit_counter[i + PIT_OFFSET_MAX];
1888 while (dec >= state.pit_counter[i])
1889 {
1890 if (!reload)
1891 { // half-programmed counter: don't spin on a zero reload
1892 state.pit_counter[i] = 0;
1893 dec = 0;
1894 break;
1895 }
1896 dec -= state.pit_counter[i];
1897 state.pit_counter[i] = reload;
1898 if (state.pit_status[i] & 0x80)
1899 {
1900 state.pit_status[i] &= ~0x80; // lower output;
1901 }
1902 else
1903 {
1904 state.pit_status[i] |= 0x80; // raise output
1905 if (i == 0)
1906 pic_interrupt(0, 0); // counter 0 is tied to irq 0.
1907 }
1908 }
1909 state.pit_counter[i] -= (u32)dec;
1910 }
1911 break;
1912
1913 default:
1914 break; // we don't care to handle it.
1915 }
1916 }
1917}

2035/* Highest-priority bit in mask, considering rotation. Returns 8 if mask==0. */

2037{
2038 if (mask == 0)
2039 return 8;

2042 priority++;

2051int CAliM1543C::pic_get_irq(int index)
2052{
2053 int mask = state.pic_irr[index] & ~state.pic_imr[index];
2054 int priority = pic_priority_of((u8)mask, state.pic_priority_add[index]);
2055 if (priority == 8)
2056 return -1;
2057
2058 // Compute current in-service priority. In SFNM on the master, slave-
2059 // originated IRQs (cascade IRQ2) do not block other slave IRQs.
2060 int isr_mask = state.pic_isr[index];
2061 if (state.pic_special_mask[index])
2062 isr_mask &= ~state.pic_imr[index];
2063 if (state.pic_special_fnm[index] && index == 0)
2064 isr_mask &= ~(1 << 2);
2065
2066 int cur_priority = pic_priority_of((u8)isr_mask, state.pic_priority_add[index]);
2067 if (priority < cur_priority)
2068 return (priority + state.pic_priority_add[index]) & 7;
2069 return -1;
2070}

2080{

2086 // triggered; honor ELCR/LTIM in case the OS programs it as level.
2087 const u8 cascade_mask = (1 << 2);
2088 const bool is_level =
2089 (state.pic_elcr[0] & cascade_mask) || state.pic_ltim[0];
2090

2094 {
2095 state.pic_irr[0] |= cascade_mask;
2096 }
2097 else

2100 state.pic_irr[0] |= cascade_mask;
