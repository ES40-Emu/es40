35// FNV-1a/64 over a block's source words -- record() uses it to revalidate kept code after a flush
36// instead of recompiling, and to catch self-modifying code (changed bytes -> different hash).
37static inline uint64_t src_hash(const uint8_t* p, uint32_t n_instr)
38{
39 uint64_t h = 1469598103934665603ULL;
40 for (uint32_t i = 0; i < n_instr; ++i) {
41 const auto* w = p + static_cast<std::size_t>(i) * 4;
42 const uint32_t word = uint32_t(w[0]) | (uint32_t(w[1]) << 8)
43 | (uint32_t(w[2]) << 16) | (uint32_t(w[3]) << 24);
44 h ^= word;
45 h *= 1099511628211ULL;
46 }
47 return h;
48}
