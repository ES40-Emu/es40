106static unsigned win32_key_scan(unsigned key, unsigned scan, bool extended)
107{
108 // Some virtual keyboards send break codes or omit the scan entirely.
109 extended = extended || (scan & 0xff00) == 0xe000;
110 if (!(scan & 0x7f)) scan = MapVirtualKeyW(key, MAPVK_VK_TO_VSC_EX);
111 const unsigned prefix = scan & 0xff00;
112 const unsigned make = scan & 0x7f;
113 extended = extended || prefix == 0xe000;
114 // Windows appears to make right shift extended even if the native is
115 // plain 0x36 during capture, both must identify the same to the guest
116 if (key == VK_RSHIFT || (key == VK_SHIFT && make == 0x36)) return 0x36;
117 // Keep Pause/Break's E1 sequence distinct from Control and NumLock.
118 if (key == VK_SNAPSHOT) return win32_print_screen_scan;
119 if (key == VK_NUMLOCK) return 0x145;
120 if (key == VK_PAUSE || key == VK_CANCEL ||
121 (prefix == 0xe100 && make == 0x1d) || (!extended && make == 0x45))
122 return win32_pause_scan;
123 return make | (extended ? 0x100 : 0);
124}

173static win32_linear_sample win32_linear_axis(unsigned position,
174 unsigned source_size, unsigned target_size)
175{
176 // Sample pixel centers and clamp at the image edge, as SDL textures do.
177 const double source = (std::max)(0.0, (std::min)((double)source_size - 1,
178 (position + 0.5) * source_size / target_size - 0.5));
179 const unsigned first = (unsigned)source;
180 return { first, (std::min)(first + 1, source_size - 1),
181 (unsigned)((source - first) * 256 + 0.5) };
182}

184static u32 win32_linear_mix(u32 first, u32 second, unsigned fraction)
185{
186 // Interpolate red/blue together and green separately, with 8-bit weights.
187 const unsigned inverse = 256 - fraction;
188 const u32 rb = (((first & 0x00ff00ff) * inverse +
189 (second & 0x00ff00ff) * fraction + 0x00800080) >> 8) & 0x00ff00ff;
190 const u32 g = (((first & 0x0000ff00) * inverse +
191 (second & 0x0000ff00) * fraction + 0x00008000) >> 8) & 0x0000ff00;
192 return rb | g;
193}

252static std::string trim_win32_hotkey(const std::string& text)
253{
254 size_t first = 0, last = text.size();
255 while (first < last && std::isspace((unsigned char)text[first])) ++first;
256 while (last > first && std::isspace((unsigned char)text[last - 1])) --last;
257 return text.substr(first, last - first);
258}

260static bool win32_hotkey_text_equals(const std::string& text, const char* name)
261{
262 if (text.size() != strlen(name)) return false;
263 for (size_t i = 0; i < text.size(); ++i)
264 if (std::tolower((unsigned char)text[i]) !=
265 std::tolower((unsigned char)name[i])) return false;
266 return true;
267}

269static unsigned parse_win32_hotkey_modifier(const std::string& text)
270{
271 if (win32_hotkey_text_equals(text, "Ctrl") ||
272 win32_hotkey_text_equals(text, "Control")) return WIN32_HOTKEY_CTRL;
273 if (win32_hotkey_text_equals(text, "Alt") ||
274 win32_hotkey_text_equals(text, "Option")) return WIN32_HOTKEY_ALT;
275 if (win32_hotkey_text_equals(text, "Shift")) return WIN32_HOTKEY_SHIFT;
276 for (const char* name : { "GUI", "Super", "Win", "Windows", "Cmd", "Command", "Meta" })
277 if (win32_hotkey_text_equals(text, name)) return WIN32_HOTKEY_GUI;
278 return 0;
279}

281static win32_hotkey_binding parse_win32_hotkey(CConfigurator* cfg,
282 const char* config_name, const char* default_value)
283{
284 win32_hotkey_binding binding;
285 binding.config_name = config_name;
286 const char* configured = cfg->get_text_value(config_name, default_value);
287 const std::string value = trim_win32_hotkey(configured ? configured : "");
288 if (win32_hotkey_text_equals(value, "none")) return binding;
289 std::string key_name = value;
290 bool invalid = value.empty();
291 for (;;)
292 {
293 const size_t separator = key_name.find('+');
294 if (separator == std::string::npos) break;
295 const unsigned modifier = parse_win32_hotkey_modifier(
296 trim_win32_hotkey(key_name.substr(0, separator)));
297 if (!modifier) break;
298 if (binding.modifiers & modifier) { invalid = true; break; }
299 binding.modifiers |= modifier;
300 key_name = trim_win32_hotkey(key_name.substr(separator + 1));
301 }
302 const struct { const char* alias; const char* name; } aliases[] = {
303 { "PgUp", "PageUp" }, { "PgDn", "PageDown" }, { "PageDn", "PageDown" },
304 { "Esc", "Escape" }, { "Del", "Delete" }, { "Ins", "Insert" },
305 { "KeypadPlus", "Keypad +" }, { "KpPlus", "Keypad +" },
306 { "KeypadMinus", "Keypad -" }, { "KpMinus", "Keypad -" }
307 };
308 for (const auto& alias : aliases)
309 if (win32_hotkey_text_equals(key_name, alias.alias)) key_name = alias.name;
310 for (const auto& named : win32_hotkey_names)
311 if (win32_hotkey_text_equals(key_name, named.name))
312 { binding.key = named.key; key_name = named.name; break; }
313 for (unsigned i = 0; i < 24; ++i)
314 {
315 const std::string name = "F" + std::to_string(i + 1);
316 if (win32_hotkey_text_equals(key_name, name.c_str()))
317 { binding.key = win32_named_key | (VK_F1 + i); key_name = name; break; }
318 }
319 if (!binding.key && key_name.size() == 1 &&
320 (unsigned char)key_name[0] >= 33 && (unsigned char)key_name[0] <= 126)
321 {
322 // SDL resolves shifted key names (e.g. ! on a US keyboard) to the
323 // underlying unshifted key, without adding an implicit Shift modifier.
324 const SHORT mapped = VkKeyScanW((wchar_t)(unsigned char)key_name[0]);
325 if (mapped != -1 && (HIBYTE(mapped) & 1))
326 {
327 const unsigned unshifted = MapVirtualKeyW(LOBYTE(mapped), MAPVK_VK_TO_CHAR) & 0xffff;
328 if (unshifted >= 33 && unshifted <= 126) key_name[0] = (char)unshifted;
329 }
330 key_name[0] = (char)std::toupper((unsigned char)key_name[0]);
331 binding.key = (unsigned char)key_name[0];
332 }
333 if (invalid || !binding.key)
334 {
335 printf("%%WIN32-W-HOTKEY: %s has invalid binding \"%s\"; "
336 "action disabled.\n", config_name, value.c_str());
337 return binding;
338 }
339 binding.enabled = true;
340 if (binding.modifiers & WIN32_HOTKEY_CTRL) binding.display += "Ctrl+";
341 if (binding.modifiers & WIN32_HOTKEY_ALT) binding.display += "Alt+";
342 if (binding.modifiers & WIN32_HOTKEY_SHIFT) binding.display += "Shift+";
343 if (binding.modifiers & WIN32_HOTKEY_GUI) binding.display += "GUI+";
344 binding.display += key_name;
345 return binding;
346}

348static unsigned win32_host_hotkey_key(WPARAM key, unsigned scan)
349{
350 // Ctrl+Break reports VK_CANCEL, but SDL exposes it as the Pause key.
351 if (scan == win32_pause_scan) return win32_named_key | VK_PAUSE;
352 // NumLock changes virtual-key codes; SDL keeps keypad keys distinct.
353 if (scan >= 0x47 && scan <= 0x53)
354 {
355 static const unsigned keypad[] = { VK_NUMPAD7, VK_NUMPAD8, VK_NUMPAD9,
356 VK_SUBTRACT, VK_NUMPAD4, VK_NUMPAD5, VK_NUMPAD6, VK_ADD,
357 VK_NUMPAD1, VK_NUMPAD2, VK_NUMPAD3, VK_NUMPAD0, VK_DECIMAL };
358 return win32_keypad_key | keypad[scan - 0x47];
359 }
360 if (key == VK_MULTIPLY || key == VK_DIVIDE ||
361 (key == VK_RETURN && (scan & 0x100))) return win32_keypad_key | (unsigned)key;
362 if (key >= VK_F1 && key <= VK_F24) return win32_named_key | (unsigned)key;
363 for (const auto& named : win32_hotkey_names)
364 if (named.key == (win32_named_key | (unsigned)key)) return named.key;
365 // This does not consume text or change the host's dead-key composition state.
366 return MapVirtualKeyW((UINT)key, MAPVK_VK_TO_CHAR) & 0xffff;
367}

397static u32 win32_keymap_symbol(const char* name)
398{
399 if (strncmp(name, "SDLK_", 5)) return BX_KEYMAP_UNKNOWN;
400 const std::string key = name + 5;
401 if (key.size() == 1 && std::isalnum((unsigned char)key[0]) &&
402 (unsigned char)key[0] < 128)
403 return (unsigned)std::toupper((unsigned char)key[0]);
404 if (key.size() == 4 && key.compare(0, 3, "KP_") == 0 &&
405 key[3] >= '0' && key[3] <= '9')
406 return win32_keypad_key | (VK_NUMPAD0 + key[3] - '0');
407 for (unsigned i = 0; i < 24; ++i)
408 if (key == "F" + std::to_string(i + 1)) return win32_named_key | (VK_F1 + i);
409 for (const auto& named : win32_hotkey_names)
410 if (win32_hotkey_text_equals(key, named.name)) return named.key;
411 for (const auto& named : win32_keymap_names)
412 if (key == named.name) return named.key;
413 return BX_KEYMAP_UNKNOWN;
414}

416static unsigned win32_host_mapping_key(WPARAM key, unsigned scan)
417{
418 // WM_KEY* uses generic modifier VKs; the scan identifies each side.
419 switch (scan)
420 {
421 case win32_pause_scan: return win32_named_key | VK_PAUSE;
422 case 0x2a: return win32_named_key | VK_LSHIFT;
423 case 0x36: return win32_named_key | VK_RSHIFT;
424 case 0x1d: return win32_named_key | VK_LCONTROL;
425 case 0x11d: return win32_named_key | VK_RCONTROL;
426 case 0x38: return win32_named_key | VK_LMENU;
427 case 0x138: return win32_named_key | VK_RMENU;
428 }
429 if (scan >= 0x47 && scan <= 0x53) return win32_host_hotkey_key(key, scan);
430 // SDL's default key events normalize symbol-based number rows to digits.
431 if (key >= '0' && key <= '9') return (unsigned)key;
432 if (key == VK_OEM_NEC_EQUAL) return win32_keypad_key | VK_OEM_NEC_EQUAL;
433 for (const auto& named : win32_keymap_names)
434 if (named.key == (win32_named_key | (unsigned)key)) return named.key;
435 return win32_host_hotkey_key(key, scan);
436}

445static unsigned win32_key_state_key(unsigned key, unsigned scan)
446{
447 // WM_KEY* reports generic modifier VKs, but each side can be held alone.
448 if (key == VK_SHIFT) return scan == 0x36 ? VK_RSHIFT : VK_LSHIFT;
449 if (key == VK_CONTROL) return scan & 0x100 ? VK_RCONTROL : VK_LCONTROL;
450 if (key == VK_MENU) return scan & 0x100 ? VK_RMENU : VK_LMENU;
451 return key < 256 ? key : 0;
452}

1029void win32_application::initialize()
1030{
1031 if (GetCurrentThreadId() != owner_thread || shutting_down)
1032 FAILURE(Win32, "Invalid display application initialization");
1033 if (dispatcher) return;
1034 keyboard_use_mapping = cfg->get_bool_value("keyboard.use_mapping", false);
1035 if (keyboard_use_mapping)
1036 {
1037 // Publish only a complete map. No dispatcher or input can observe a
1038 // partial load, and a failed initialization can retry from scratch.
1039 auto loaded = std::make_unique<bx_keymap_c>(cfg);
1040 loaded->loadKeymap(win32_keymap_symbol);
1041 const bool published = bx_keymap == shared_keymap.get();
1042 shared_keymap.swap(loaded);
1043 if (published) bx_keymap = shared_keymap.get();
1044 }
1045 const char* speed = cfg->get_text_value("mouse.speed", "1.0");
1046 mouse_speed = std::atof(speed);
1047 if (!std::isfinite(mouse_speed) || mouse_speed <= 0.0 ||
1048 mouse_speed > 10.0)
1049 {
1050 printf("%%WIN32-W-MOUSESPEED: invalid mouse.speed \"%s\" "
1051 "(valid: 0.0 < speed <= 10.0); using 1.0.\n", speed);
1052 mouse_speed = 1.0;
1053 }
1054 mouse_invert_x = cfg->get_bool_value("mouse.invert_x", false);
1055 mouse_invert_y = cfg->get_bool_value("mouse.invert_y", false);
1056 mouse_absolute = cfg->get_bool_value("mouse.absolute", false);
1057 const u64 scale = cfg->get_num_value("video.scale_ratio", true, 0);
1058 if (scale > 10)
1059 printf("%%WIN32-W-SCALE: invalid video.scale_ratio %llu (valid: 0 - 10); "
1060 "using 1x.\n", (unsigned long long)scale);
1061 configured_scale = scale >= 1 && scale <= 10 ? (int)scale : 1;
1062 scale_change_enable = cfg->get_bool_value("video.scale_change_enable", false);
1063 video_linear = cfg->get_bool_value("video.linear", true);
1064 hotkey_mouse_capture = parse_win32_hotkey(cfg, "hotkey.mouse_capture", "Ctrl+F10");
1065 hotkey_media = parse_win32_hotkey(cfg, "hotkey.media", "Ctrl+F11");
1066 hotkey_ctrl_alt_delete = parse_win32_hotkey(cfg, "hotkey.ctrl_alt_delete", "Ctrl+Alt+End");
1067 hotkey_reset_window = parse_win32_hotkey(cfg, "hotkey.reset_window", "Ctrl+Alt+Home");
1068 hotkey_scale_up = parse_win32_hotkey(cfg, "hotkey.scale_up", "Ctrl+PageUp");
1069 hotkey_scale_down = parse_win32_hotkey(cfg, "hotkey.scale_down", "Ctrl+PageDown");
1070 win32_hotkey_binding* bindings[] = {
1071 &hotkey_mouse_capture, &hotkey_media, &hotkey_ctrl_alt_delete, &hotkey_reset_window,
1072 &hotkey_scale_up, &hotkey_scale_down
1073 };
1074 const bool active[] = {
1075 hotkey_mouse_capture.enabled, hotkey_media.enabled, hotkey_ctrl_alt_delete.enabled, hotkey_reset_window.enabled,
1076 hotkey_scale_up.enabled && scale_change_enable, hotkey_scale_down.enabled && scale_change_enable
1077 };
1078 constexpr size_t binding_count = sizeof(bindings) / sizeof(bindings[0]);
1079 bool duplicate[binding_count] = {};
1080 for (size_t i = 0; i < binding_count; ++i)
1081 {
1082 for (size_t j = i + 1; j < binding_count; ++j)
1083 {
1084 if (active[i] && active[j] &&
1085 bindings[i]->key == bindings[j]->key &&
1086 bindings[i]->modifiers == bindings[j]->modifiers)
1087 {
1088 printf("%%WIN32-W-HOTKEY: %s and %s both use \"%s\"; both actions disabled.\n",
1089 bindings[i]->config_name, bindings[j]->config_name, bindings[i]->display.c_str());
1090 duplicate[i] = duplicate[j] = true;
1091 }
1092 }
1093 }
1094 for (size_t i = 0; i < binding_count; ++i)
1095 if (duplicate[i]) bindings[i]->enabled = false;
1096 register_win32_class(win32_window_class, win32_window_proc);
1097 register_win32_class(win32_dispatch_class, win32_dispatch_proc);
1098 dispatcher = CreateWindowEx(0, win32_dispatch_class, "", 0, 0, 0,
1099 0, 0, HWND_MESSAGE, nullptr, GetModuleHandle(nullptr), this);
1100 if (!dispatcher)
1101 FAILURE(Win32, "Unable to create display dispatcher");
1102 keyboard = theKeyboard;
1103 system = theSystem;
1104 if (keyboard) keyboard->set_mouse_capture(false);
1105 // Match SDL's default: guest video keeps the host display on even when
1106 // there is no host input. Restore the caller's thread state at shutdown.
1107 previous_execution_state = SetThreadExecutionState(ES_CONTINUOUS | ES_DISPLAY_REQUIRED);
1108 if (!previous_execution_state)
1109 printf("%%WIN32-W-DISPLAYPOWER: Unable to prevent host display blanking.\n");
1110 running.store(true, std::memory_order_release);
1111 const char* host_value = getenv("ES40_MANAGER_HOST");
1112 if (host_value && *host_value)
1113 {
1114 char* end = nullptr;
1115 const unsigned long long host_number = _strtoui64(host_value, &end, 10);
1116 if (!end || *end || !host_number || host_number > UINTPTR_MAX)
1117 FAILURE(Win32, "Invalid machine manager host window");
1118 managed_host = reinterpret_cast<HWND>(static_cast<UINT_PTR>(host_number));
1119 const char* pid_value = getenv("ES40_MANAGER_PID");
1120 const unsigned long long pid_number = pid_value ? _strtoui64(pid_value, &end, 10) : 0;
1121 if (!pid_number || !end || *end || pid_number > MAXDWORD)
1122 FAILURE(Win32, "Invalid machine manager process");
1123 manager_pid = static_cast<DWORD>(pid_number);
1124 wchar_t host_class[64] = {};
1125 DWORD host_pid = 0;
1126 GetWindowThreadProcessId(managed_host, &host_pid);
1127 if (host_pid != manager_pid || manager_pid == GetCurrentProcessId() ||
1128 !GetClassNameW(managed_host, host_class, 64) || wcscmp(host_class, win32_manager_host_class))
1129 FAILURE(Win32, "Machine manager host is no longer available");
1130 manager_process = OpenProcess(SYNCHRONIZE, FALSE, manager_pid);
1131 if (!manager_process || WaitForSingleObject(manager_process, 0) != WAIT_TIMEOUT)
1132 FAILURE(Win32, "Machine manager process is no longer available");
1133 // Allow stop and media commands without a GPU.
1134 if (!displays.empty()) displays.front()->initialize_window();
1135 }

1138void win32_application::invoke(std::function<void()> fn)
1139{
1140 if (GetCurrentThreadId() == owner_thread) { fn(); return; }
1141 if (!running.load(std::memory_order_acquire) || !dispatcher)
1142 FAILURE(Win32, "Display application thread is not running");

1144 if (SendMessage(dispatcher, win32_invoke_message, 0, (LPARAM)&call) != 1)
1145 FAILURE(Win32, "Unable to reach the display application thread");
1146 if (call.error) std::rethrow_exception(call.error);
1147}

1153 SetWindowLongPtr(hwnd, GWLP_USERDATA,

1155 if (msg == win32_invoke_message)
1156 {
1157 auto& call = *reinterpret_cast<win32_call*>(lparam);
1158 try { call.fn(); } catch (...) { call.error = std::current_exception(); }
1159 return 1;
1160 }
1161 if (msg == win32_wake_message)
1162 {
1163 auto* app = reinterpret_cast<win32_application*>(GetWindowLongPtr(hwnd, GWLP_USERDATA));
1164 if (app)
1165 {
1166 try { app->dispatch_captured_keys(); app->media.close(); }
1167 catch (...) { app->record_callback_error(); }
1168 }
1169 return 0;
1170 }
1171 if (msg == win32_keyboard_message)
1172 {
1173 auto* app = reinterpret_cast<win32_application*>(GetWindowLongPtr(hwnd, GWLP_USERDATA));
1174 if (app)
1175 {
1176 try { app->dispatch_captured_keys(); }
1177 catch (...) { app->record_callback_error(); }
1178 }
1179 return 0;
1180 }

1192void win32_application::set_keyboard_capture(bool enabled)
1193{
1194 if (enabled)
1195 {
1196 if (keyboard_hook || keyboard_capture_starting || !keyboard || input_paused || shutting_down || callback_error || media.active() ||
1197 !running.load(std::memory_order_acquire)) return;
1198 // Finish keys already queued before installing the hook.
1199 const unsigned transition = mouse_capture_transition;
1200 MSG pending;
1201 while (PeekMessage(&pending, nullptr, WM_KEYFIRST, WM_KEYLAST, PM_REMOVE | PM_QS_INPUT))
1202 {
1203 if (pending.message == WM_QUIT) { request_shutdown(); return; }
1204 TranslateMessage(&pending); DispatchMessage(&pending);
1205 if (transition != mouse_capture_transition || keyboard_hook || input_paused || shutting_down || media.active()) return;
1206 }
1207 if (transition != mouse_capture_transition || keyboard_hook || !keyboard || input_paused ||
1208 shutting_down || media.active() || !running.load(std::memory_order_acquire) ||
1209 !capture_owner || !win32_display_active(capture_owner->window)) return;
1210 BYTE state[256];
1211 HMODULE module = nullptr;
1212 if (!GetKeyboardState(state) ||
1213 !GetModuleHandleExA(GET_MODULE_HANDLE_EX_FLAG_FROM_ADDRESS | GET_MODULE_HANDLE_EX_FLAG_UNCHANGED_REFCOUNT,
1214 reinterpret_cast<LPCSTR>(win32_keyboard_proc), &module))
1215 {
1216 printf("%%WIN32-W-KEYBOARD: Unable to prepare keyboard capture.\n");
1217 return;
1218 }
1219 std::fill_n(hook_scan_down, 512, false);
1220 for (unsigned key = 0; key < 256; ++key)
1221 {
1222 captured_key_down[key] = pre_hook_key_down[key] = (state[key] & 0x80) != 0;
1223 if (key == VK_LCONTROL && native_altgr_control)
1224 captured_key_down[key] = native_left_control_down;
1225 // Shared VKs cannot identify which physical key was already held.
1226 if (!captured_key_down[key] || key == VK_SHIFT || key == VK_CONTROL || key == VK_MENU ||
1227 key == VK_RETURN || key == VK_CLEAR || (key >= VK_PRIOR && key <= VK_DOWN) ||
1228 key == VK_INSERT || key == VK_DELETE) continue;
1229 const unsigned scan = win32_key_scan(key, 0, false);
1230 if (scan) hook_scan_down[scan] = true;
1231 }
1232 for (unsigned scan = 0; scan < 512; ++scan)
1233 if (host_virtual_key_by_scan[scan] && pre_hook_key_down[host_virtual_key_by_scan[scan]]) hook_scan_down[scan] = true;
1234 keyboard_hook_failed = false;
1235 struct starting_guard
1236 {
1237 bool& starting;
1238 starting_guard(bool& value) : starting(value) { starting = true; }
1239 ~starting_guard() { starting = false; }
1240 } guard(keyboard_capture_starting);
1241 keyboard_hook = SetWindowsHookExA(WH_KEYBOARD_LL, win32_keyboard_proc, module, 0);
1242 if (keyboard_hook) win32_keyboard_application = this;
1243 else printf("%%WIN32-W-KEYBOARD: Unable to capture system shortcuts (error %lu).\n", GetLastError());
1244 // Close the gap between the first drain and installation.
1245 while (keyboard_hook && PeekMessage(&pending, nullptr, WM_KEYFIRST, WM_KEYLAST, PM_REMOVE | PM_QS_INPUT))
1246 {
1247 if (pending.message == WM_QUIT) { request_shutdown(); set_keyboard_capture(false); return; }
1248 TranslateMessage(&pending); DispatchMessage(&pending);
1249 if (transition != mouse_capture_transition || input_paused || shutting_down || media.active()) break;
1250 }
1251 if (transition != mouse_capture_transition || !keyboard || input_paused || shutting_down || media.active() ||
1252 !running.load(std::memory_order_acquire) || !capture_owner ||
1253 !win32_display_active(capture_owner->window))
1254 set_keyboard_capture(false);
1255 return;
1256 }
1257 if (win32_keyboard_application == this) win32_keyboard_application = nullptr;
1258 if (!keyboard_hook) return;
1259 UnhookWindowsHookEx(keyboard_hook);
1260 keyboard_hook = nullptr;
1261 // Retired input must not reach the guest.
1262 for (unsigned i = 0; i < captured_key_count; ++i)
1263 {
1264 const auto& event = captured_keys[(captured_key_first + i) % keyboard_queue_capacity];
1265 observe_key(event.scan, event.key, event.released, event.repeat, false);
1266 }
1267 captured_key_first = captured_key_count = 0;
1268 // Windows did not see intercepted presses.
1269 if (keyboard)
1270 for (unsigned scan = 0; scan < 512; ++scan)
1271 if (guest_keys[scan] && win32_grabbed_key(guest_virtual_key_by_scan[scan]))
1272 pending_keyboard_releases[scan] = true;
1273 release_grabbed_keys();
1274}

1289bool win32_application::capture_keyboard_event(WPARAM msg, const KBDLLHOOKSTRUCT& event) noexcept
1290{
1291 if (!keyboard_hook || keyboard_hook_failed || input_paused || !keyboard || shutting_down || callback_error || media.active() ||
1292 !running.load(std::memory_order_acquire)) return false;
1293 HWND focus = GetFocus();
1294 if (!capture_owner || !window_owner(focus) || !win32_display_active(focus))
1295 {
1296 PostMessage(dispatcher, win32_keyboard_message, 0, 0);
1297 return false;
1298 }
1299 if (msg != WM_KEYDOWN && msg != WM_SYSKEYDOWN && msg != WM_KEYUP && msg != WM_SYSKEYUP) return false;
1300 const bool released = msg == WM_KEYUP || msg == WM_SYSKEYUP;
1301 // Windows adds a synthetic left Control to AltGr; SDL excludes it too.
1302 if ((event.scanCode & ~0x80u) == 0x21d)
1303 {
1304 // Balance a synthetic press Windows saw before capture began.
1305 if (released && native_altgr_control)
1306 {
1307 native_altgr_release_pending = true;
1308 native_altgr_release_time = event.time;
1309 return false;
1310 }
1311 return true;
1312 }
1313 const unsigned scan = win32_key_scan(event.vkCode, event.scanCode,
1314 (event.flags & LLKHF_EXTENDED) != 0);
1315 const unsigned key = win32_key_state_key(event.vkCode, scan);
1316 if (!key) return false;
1317 const bool pass_native = !win32_grabbed_key(key) ||
1318 (released && (keyboard_capture_starting || pre_hook_key_down[key] || !hook_scan_down[scan]));
1319 unsigned native_slot = next_native_key;
1320 if (pass_native)
1321 for (unsigned count = 0; count < keyboard_queue_capacity && native_keys[native_slot].pending; ++count)
1322 native_slot = (native_slot + 1) % keyboard_queue_capacity;
1323 if (captured_key_count == keyboard_queue_capacity || (pass_native && native_keys[native_slot].pending))
1324 {
1325 keyboard_hook_failed = true;
1326 PostMessage(dispatcher, win32_keyboard_message, 0, 0);
1327 return false;
1328 }
1329 const unsigned slot = (captured_key_first + captured_key_count) % keyboard_queue_capacity;
1330 captured_keys[slot] = { focus, key, scan, released, !released && hook_scan_down[scan] };
1331 ++captured_key_count;
1332 if (!PostMessage(dispatcher, win32_keyboard_message, 0, 0))
1333 {
1334 --captured_key_count;
1335 keyboard_hook_failed = true;
1336 return false;
1337 }
1338 if (pass_native)
1339 {
1340 native_keys[native_slot] = { event.time, key, released, true };
1341 next_native_key = (native_slot + 1) % keyboard_queue_capacity;
1342 }
1343 hook_scan_down[scan] = !released;
1344 if (released) pre_hook_key_down[key] = false;
1345 return !pass_native;
1346}

1348bool win32_application::filter_native_altgr(UINT msg, WPARAM key, LPARAM flags)
1349{
1350 const bool released = msg == WM_KEYUP || msg == WM_SYSKEYUP;
1351 const DWORD time = (DWORD)GetMessageTime();
1352 if (native_altgr_release_pending && (LONG)(time - native_altgr_release_time) > 0)
1353 native_altgr_release_pending = false;
1354 const unsigned scan = HIWORD(flags) & 0x1ff;
1355 if ((key == VK_MENU || key == VK_RMENU) && scan == 0x138)
1356 {
1357 native_altgr_press_pending = false;
1358 if (released && native_altgr_control)
1359 {
1360 // The synthetic Control release can follow the RightAlt release.
1361 native_altgr_release_pending = true;
1362 native_altgr_release_time = time;
1363 }
1364 }
1365 if ((key != VK_CONTROL && key != VK_LCONTROL) || (scan & 0x100)) return false;
1366 if (released && !native_altgr_control && !native_altgr_release_pending) return false;
1367 bool synthetic = released && native_altgr_release_pending && native_altgr_release_time == time;
1368 MSG next;
1369 if (!synthetic && PeekMessage(&next, nullptr, 0, 0, PM_NOREMOVE))
1370 {
1371 // Like SDL, recognize AltGr's fake Control by its adjacent RightAlt.
1372 const bool matching_message = released ? next.message == WM_KEYUP || next.message == WM_SYSKEYUP :
1373 next.message == WM_KEYDOWN || next.message == WM_SYSKEYDOWN;
1374 synthetic = matching_message && next.time == time &&
1375 (next.wParam == VK_MENU || next.wParam == VK_RMENU) &&
1376 (HIWORD(win32_key_flags(next.wParam, next.lParam)) & 0x1ff) == 0x138;
1377 }
1378 if (!synthetic) return false;
1379 if (!released && !native_altgr_control && (flags & (LPARAM(1) << 30)))
1380 native_left_control_down = true; // Genuine Control was already held before this fake press.
1381 // A paired fake release must not release a separately held genuine Control.
1382 native_altgr_control = !released;
1383 native_altgr_press_pending = !released;
1384 if (released) native_altgr_release_pending = false;
1385 return true;
1386}

1388bool win32_application::consume_native_key(unsigned key, unsigned scan, bool released, DWORD time)
1389{
1390 key = win32_key_state_key(key, scan);
1391 for (auto& event : native_keys)
1392 {
1393 // Some host keys (ex Print Screen) omit a native message.
1394 if (event.pending && (LONG)(time - event.time) > 0) event.pending = false;
1395 if (event.pending && event.key == key && event.released == released && event.time == time)
1396 {
1397 event.pending = false;
1398 return true;
1399 }
1400 }
1401 // Input already queued when capture began has no hook duplicate.
1402 if (keyboard_hook && key)
1403 {
1404 captured_key_down[key] = !released;
1405 if (keyboard_capture_starting)
1406 {
1407 bool newer_event = false;
1408 bool queued_release = false;
1409 for (unsigned i = 0; i < captured_key_count; ++i)
1410 {
1411 auto& queued = captured_keys[(captured_key_first + i) % keyboard_queue_capacity];
1412 if (queued.scan != scan) continue;
1413 if (!newer_event) queued.repeat = !queued.released && !released;
1414 newer_event = true;
1415 queued_release = queued_release || queued.released;
1416 }
1417 pre_hook_key_down[key] = !released && !queued_release;
1418 if (!newer_event)
1419 {
1420 hook_scan_down[scan] = !released;
1421 }
1422 }
1423 }
1424 return false;
1425}

1451void win32_application::dispatch_captured_keys()
1452{
1453 if (dispatching_captured_key || keyboard_capture_starting) return;
1454 if (!running.load(std::memory_order_acquire)) { set_keyboard_capture(false); return; }
1455 if (keyboard_hook_failed)
1456 {
1457 set_keyboard_capture(false);
1458 keyboard_hook_failed = false;
1459 printf("%%WIN32-W-KEYBOARD: Keyboard capture queue failed; system shortcuts released.\n");
1460 return;
1461 }
1462 if (keyboard_hook && (!window_owner(GetFocus()) || !win32_display_active(GetFocus())))
1463 { release_input_session(); return; }
1464 while (keyboard_hook && captured_key_count)
1465 {
1466 const auto event = captured_keys[captured_key_first];
1467 captured_key_first = (captured_key_first + 1) % keyboard_queue_capacity;
1468 --captured_key_count;
1469 auto* target = window_owner(event.window);
1470 if (!target) continue;
1471 captured_key_down[event.key] = !event.released;
1472 LPARAM flags = 1 | (LPARAM(event.scan) << 16);
1473 if (keyboard_modifiers() & WIN32_HOTKEY_ALT) flags |= LPARAM(1) << 29;
1474 if (event.repeat || event.released) flags |= LPARAM(1) << 30;
1475 if (event.released) flags |= LPARAM(1) << 31;
1476 dispatching_captured_key = true;
1477 try { target->window_event(event.window, event.released ? WM_KEYUP : WM_KEYDOWN, event.key, flags); }
1478 catch (...) { dispatching_captured_key = false; set_keyboard_capture(false); throw; }
1479 dispatching_captured_key = false;
1480 // SDL lets Alt+Tab escape a captured fullscreen window by minimizing it.
1481 if (!event.released && event.key == VK_TAB && target->fullscreen &&
1482 (keyboard_modifiers() & WIN32_HOTKEY_ALT))
1483 {
1484 ShowWindow(event.window, SW_MINIMIZE);
1485 release_input_session();
1486 }
1487 }

1490void win32_application::pump()
1491{
1492 while (running.load(std::memory_order_acquire))
1493 {
1494 if (managed_host)
1495 {
1496 DWORD host_pid = 0;
1497 GetWindowThreadProcessId(managed_host, &host_pid);
1498 if (host_pid != manager_pid || WaitForSingleObject(manager_process, 0) != WAIT_TIMEOUT)
1499 request_shutdown();
1500 }
1501 MSG msg;
1502 while (PeekMessage(&msg, nullptr, 0, 0, PM_REMOVE))
1503 {
1504 if (msg.message == WM_QUIT) request_shutdown();
1505 else { TranslateMessage(&msg); DispatchMessage(&msg); }
1506 }
1507 try { dispatch_captured_keys(); }
1508 catch (...) { record_callback_error(); }
1509 MsgWaitForMultipleObjects(0, nullptr, FALSE, 10, QS_ALLINPUT);
1510 }
1511 set_keyboard_capture(false);
1512}

1540void win32_application::poll_input(unsigned current_key_scan)
1541{
1542 if (callback_error) std::rethrow_exception(callback_error);
1543 if (!dispatching_captured_key && current_key_scan == 512) dispatch_captured_keys();
1544 if (input_paused) return;
1545 if (!keyboard_hook && !native_altgr_release_pending)
1546 {
1547 if (!(GetKeyState(VK_LCONTROL) & 0x8000))
1548 native_altgr_control = native_left_control_down = native_altgr_press_pending = false;
1549 // AltGr may end outside our windows before a genuine Control is pressed.
1550 // Wait for the paired RightAlt down before using its state to retire the mask.
1551 else if (!native_altgr_press_pending && !(GetKeyState(VK_RMENU) & 0x8000))
1552 native_altgr_control = false;
1553 }
1554 release_grabbed_keys();
1555 media.reconcile_input();
1556 sync_input();
1557 if (media.active()) { release_input_session(); return; }
1558 auto* focus = window_owner(GetFocus());
1559 if (!focus || !win32_display_active(focus->window)) release_input_session();
1560 else if (capture_owner && capture_owner != focus) set_mouse_capture(focus);
1561 else if (capture_owner && !mouse_absolute && GetCapture() != capture_owner->window)
1562 set_mouse_capture(nullptr);
1563 reconcile_keyboard_releases(current_key_scan);
1564 reconcile_mouse_releases();
1565}

1567void win32_application::reconcile_keyboard_releases(unsigned current_key_scan)
1568{
1569 if (!keyboard || input_paused) return;
1570 for (unsigned scan = 0; scan < 512; ++scan)
1571 {
1572 const unsigned key = guest_virtual_key_by_scan[scan];
1573 // GetKeyState follows the message queue. Let the current key event finish normally
1574 // and then only repair releases missing from the guest stream.
1575 if (scan == current_key_scan || !guest_keys[scan] || !key ||
1576 guest_key_state_ambiguous[scan] || (keyboard_key_state(key) & 0x8000)) continue;
1577 send_key(scan, true);
1578 host_virtual_key_by_scan[scan] = 0;
1579 reconciled_key_releases[scan] = true;
1580 }
1581}

1610{

1631 }

1633 const unsigned modifiers = keyboard_modifiers();
1634 if (hotkey_media.matches(key, modifiers))
1635 {
1636 if (!(lparam & (1L << 30)))
1637 {

1666void win32_application::observe_key(unsigned scan, unsigned virtual_key, bool released, bool repeat, bool release_previous_press)
1667{
1668 if (scan >= 512) return;
1669 if (scan == 0x1d && (virtual_key == VK_CONTROL || virtual_key == VK_LCONTROL))
1670 native_left_control_down = !released;
1671 if (virtual_key == VK_SNAPSHOT && (released || !repeat || !reconciled_key_releases[scan]))
1672 print_screen_down_seen = !released;
1673 if (released) { host_virtual_key_by_scan[scan] = 0; return; }
1674 if (repeat && reconciled_key_releases[scan]) return;
1675 // A fresh press proves that an earlier press ended, even if no poll ran between actions
1676 if (release_previous_press && !repeat && guest_keys[scan]) send_key(scan, true);
1677 if (!repeat || !host_virtual_key_by_scan[scan])
1678 host_virtual_key_by_scan[scan] = win32_key_state_key(virtual_key, scan);
1679 const unsigned key = host_virtual_key_by_scan[scan];
1680 // Observe unmapped keys and host shortcuts too: their shared VK state could
1681 // make a still valid for the guest held key look released
1682 for (unsigned other = 0; key && other < 512; ++other)
1683 if (other != scan && guest_keys[other] && guest_virtual_key_by_scan[other] == key)
1684 guest_key_state_ambiguous[other] = true;
1685}

1687void win32_application::send_key(unsigned scan, bool released, unsigned host_key, unsigned virtual_key)
1688{
1689 if (!keyboard || input_paused || scan >= 512) return;
1690 sync_input();
1691 if (released && !guest_keys[scan]) return;
1692 u32 key;
1693 if (guest_keys[scan]) key = guest_key_by_scan[scan];
1694 else if (keyboard_use_mapping)
1695 {
1696 const BXKeyEntry* entry = shared_keymap->findHostKey(host_key);
1697 if (!entry)
1698 {
1699 BX_ERROR(("host key 0x%x not mapped!", host_key));
1700 return;
1701 }
1702 key = entry->baseKey;
1703 }
1704 else key = translate_win32_key(scan);
1705 if (key == BX_KEY_UNHANDLED) return;
1706 if (!released && guest_keys[scan] && (key == BX_KEY_CTRL_L ||
1707 key == BX_KEY_CTRL_R || key == BX_KEY_ALT_L || key == BX_KEY_ALT_R ||
1708 key == BX_KEY_SHIFT_L || key == BX_KEY_SHIFT_R)) return;
1709 keyboard->gen_scancode(key | (released ? BX_KEY_RELEASED : 0));
1710 if (!released && !guest_keys[scan])
1711 {
1712 const unsigned state_key = win32_key_state_key(virtual_key, scan);
1713 guest_virtual_key_by_scan[scan] = state_key;
1714 guest_key_state_ambiguous[scan] = false;
1715 // Main/keypad Enter and NumLock-off navigation can share a VK.
1716 for (unsigned other = 0; state_key && other < 512; ++other)
1717 if (other != scan && host_virtual_key_by_scan[other] == state_key)
1718 guest_key_state_ambiguous[scan] = true;
1719 }
1720 if (released) guest_virtual_key_by_scan[scan] = 0;
1721 // Repeats, focus-loss cleanup and releases from any output retain the
1722 // original guest key even if the host keyboard layout changes meanwhile.
1723 guest_key_by_scan[scan] = key;
1724 guest_keys[scan] = !released;
1725}

1734bool win32_application::consume_hotkey_key(unsigned scan, bool released, bool repeat,
1735 bool allow_repeat)
1736{
1737 if (scan >= 512) return false;
1738 // A fresh press also recovers when the previous release went to another app.
1739 if (!released && !repeat)
1740 {
1741 swallowed_hotkey_keys[scan] = false;
1742 fullscreen_hotkey_pressed[scan] = false;
1743 reconciled_key_releases[scan] = false;
1744 }
1745 // Fullscreen owns the whole press, even if releasing Alt makes a scale
1746 // shortcut match. Keep this state across focus changes and output windows.
1747 if (fullscreen_hotkey_pressed[scan])
1748 {
1749 if (released) fullscreen_hotkey_pressed[scan] = false;
1750 return true;
1751 }
1752 // A delayed key-up or repeat must not revive a repaired press.
1753 if (reconciled_key_releases[scan]) return true;
1754 if (!swallowed_hotkey_keys[scan]) return false;
1755 if (!released && repeat && allow_repeat) return false;
1756 if (released) swallowed_hotkey_keys[scan] = false;
1757 return true;
1758}

1827 mouse_remainder_y -= scaled_y;

1868void win32_application::set_mouse_capture(bx_win32_gui_c* target)
1869{
1870 if (target && (input_paused || media.active())) return;
1871 if (target && (!target->window || target->retired ||
1872 !win32_display_active(target->window))) return;
1873 if (capture_owner == target) return;
1874 const unsigned transition = ++mouse_capture_transition;
1875 if (!mouse_absolute)
1876 {
1877 HWND previous_target = raw_mouse_target;
1878 set_raw_mouse_target(nullptr);
1879 discard_raw_mouse_input(previous_target);
1880 if (target && target->window != previous_target) discard_raw_mouse_input(target->window);
1881 // PeekMessage can run a nested main-thread callback. A newer capture
1882 // request or display retirement takes precedence over this request.
1883 if (transition != mouse_capture_transition ||
1884 (target && (shutting_down || !target->window || target->retired ||
1885 !win32_display_active(target->window))))
1886 {
1887 if (capture_owner && !raw_mouse_target &&
1888 !set_raw_mouse_target(capture_owner->window)) set_mouse_capture(nullptr);
1889 return;
1890 }
1891 }
1892 auto* previous = capture_owner;
1893 const bool transferring = previous && target;
1894 // Absolute coordinates belong to one window, while fractional motion
1895 // belongs to the shared guest stream. Only handoffs retain that carry.
1896 if (mouse_absolute && (!previous || !target))
1897 mouse_remainder_x = mouse_remainder_y = 0.0;
1898 capture_owner = nullptr;
1899 if (previous)
1900 {
1901 previous->reset_absolute_mouse_position();
1902 if (!mouse_absolute)
1903 {
1904 ClipCursor(nullptr);
1905 if (GetCapture() == previous->window) ReleaseCapture();
1906 }
1907 previous->update_title();
1908 }
1909 if (target)
1910 {
1911 target->reset_absolute_mouse_position();
1912 target->wheel_remainder = 0;
1913 // Continue guest-held buttons during a handoff.
1914 const unsigned ignored = win32_host_mouse_buttons() &
1915 ~(transferring ? mouse_buttons : 0);
1916 target->ignored_mouse_buttons = win32_mouse_button_flags(ignored);
1917 // Publish the new owner before SetCapture notifies the previous window.
1918 capture_owner = target;
1919 // Like SDL with relative mode disabled, absolute input leaves the
1920 // host pointer free to leave and reenter the client area.
1921 if (!mouse_absolute)
1922 {
1923 SetCapture(target->window);
1924 if (GetCapture() != target->window || !target->confine_mouse() ||
1925 !set_raw_mouse_target(target->window))
1926 {
1927 capture_owner = nullptr;
1928 mouse_remainder_x = mouse_remainder_y = 0.0;
1929 ClipCursor(nullptr);
1930 if (GetCapture() == target->window) ReleaseCapture();
1931 }
1932 }
1933 target->update_title();
1934 }
1935 if (transferring && capture_owner)
1936 reconcile_mouse_releases();
1937 else
1938 {
1939 // A release or failed handoff ends the mouse session.
1940 // Send the/ release before clearing capture.
1941 if (guest_mouse_captured) send_mouse(0, 0, 0, 0);
1942 if (input_paused) input_reset_pending = true;
1943 mouse_buttons = 0;
1944 }
1945 if (!capture_owner) set_raw_mouse_target(nullptr);
1946 set_keyboard_capture(capture_owner != nullptr);
1947 SetCursor(capture_owner ? nullptr : LoadCursor(nullptr, IDC_ARROW));
1948 if (keyboard && !input_paused)
1949 {
1950 guest_mouse_captured = capture_owner != nullptr;
1951 keyboard->set_mouse_capture(guest_mouse_captured);
1952 }
1953}

2271void bx_win32_gui_c::raw_mouse_event(HRAWINPUT handle)
2272{
2273 if (application.mouse_absolute || application.input_paused || application.capture_owner != this) return;
2274 RAWINPUT input = {};
2275 UINT size = sizeof(input);
2276 if (GetRawInputData(handle, RID_INPUT, &input, &size, sizeof(RAWINPUTHEADER)) != sizeof(input) ||
2277 input.header.dwSize != sizeof(input) || input.header.dwType != RIM_TYPEMOUSE) return;
2278 const RAWMOUSE& mouse = input.data.mouse;
2279 if (!(mouse.usFlags & MOUSE_MOVE_ABSOLUTE))
2280 {
2281 // A source can switch modes; relative counts cannot extend its last
2282 // absolute position. Other devices keep their independent baselines.
2283 raw_mouse_positions.erase(input.header.hDevice);
2284 if (mouse.lLastX || mouse.lLastY)
2285 application.send_host_mouse((double)mouse.lLastX, -(double)mouse.lLastY,
2286 0, application.mouse_buttons);
2287 return;
2288 }
2289
2290 // Like SDL, accept local absolute-reporting mice/tablets in relative mode.
2291 // Some sources use pixel coordinates (0x40), otherwise Windows normalizes
2292 // them to 0..65535 over the primary display or the whole virtual desktop.
2293 static constexpr USHORT raw_coordinates = 0x40;
2294 const USHORT mode = mouse.usFlags & (MOUSE_VIRTUAL_DESKTOP | raw_coordinates);
2295 const bool virtual_desktop = (mode & MOUSE_VIRTUAL_DESKTOP) != 0;
2296 const int screen_width = GetSystemMetrics(virtual_desktop ? SM_CXVIRTUALSCREEN : SM_CXSCREEN);
2297 const int screen_height = GetSystemMetrics(virtual_desktop ? SM_CYVIRTUALSCREEN : SM_CYSCREEN);
2298 const int screen_left = virtual_desktop ? GetSystemMetrics(SM_XVIRTUALSCREEN) : 0;
2299 const int screen_top = virtual_desktop ? GetSystemMetrics(SM_YVIRTUALSCREEN) : 0;
2300 if (screen_width <= 0 || screen_height <= 0 ||
2301 (!(mode & raw_coordinates) && (mouse.lLastX < 0 || mouse.lLastX > 65535 ||
2302 mouse.lLastY < 0 || mouse.lLastY > 65535)))
2303 {
2304 raw_mouse_positions.erase(input.header.hDevice);
2305 return;
2306 }
2307 const long long x = (mode & raw_coordinates) ? mouse.lLastX :
2308 (long long)mouse.lLastX * screen_width / 65535;
2309 const long long y = (mode & raw_coordinates) ? mouse.lLastY :
2310 (long long)mouse.lLastY * screen_height / 65535;
2311 auto& position = raw_mouse_positions[input.header.hDevice];
2312 const bool continue_motion = position.valid && !(mouse.usFlags & MOUSE_ATTRIBUTES_CHANGED) &&
2313 position.mode == mode && position.screen_width == screen_width &&
2314 position.screen_height == screen_height && position.screen_left == screen_left &&
2315 position.screen_top == screen_top;
2316 const long long dx = x - position.x, dy = y - position.y;
2317 position = { x, y, screen_width, screen_height, screen_left, screen_top, mode, true };
2318 // Seed a new source/coordinate space, and ignore tablet lift/drop jumps.
2319 // Retain the new position after a rejected jump so normal motion resumes.
2320 if (!continue_motion || dx < -32 || dx > 32 || dy < -32 || dy > 32) return;
2321 if (dx || dy)
2322 application.send_host_mouse((double)dx, -(double)dy, 0, application.mouse_buttons);

2726static LRESULT CALLBACK win32_window_proc(HWND hwnd, UINT msg,
2727 WPARAM wparam, LPARAM lparam)
2728{
2729 auto* owner = reinterpret_cast<bx_win32_gui_c*>(
2730 GetWindowLongPtr(hwnd, GWLP_USERDATA));
2731 if (msg == WM_NCCREATE)
2732 {
2733 owner = static_cast<bx_win32_gui_c*>(
2734 reinterpret_cast<CREATESTRUCT*>(lparam)->lpCreateParams);
2735 SetWindowLongPtr(hwnd, GWLP_USERDATA, (LONG_PTR)owner);
2736 }
2737 if (!owner) return DefWindowProc(hwnd, msg, wparam, lparam);
2738 try
2739 {
2740 if (msg == WM_NCCREATE)
2741 {
2742 owner->window = hwnd;
2743 owner->application.windows.emplace(hwnd, owner);
2744 return DefWindowProc(hwnd, msg, wparam, lparam);
2745 }
2746 if (msg == WM_NCDESTROY)
2747 {
2748 owner->application.windows.erase(hwnd);
2749 SetWindowLongPtr(hwnd, GWLP_USERDATA, 0);
2750 owner->window = nullptr;
2751 return DefWindowProc(hwnd, msg, wparam, lparam);
2752 }
2753 const LRESULT result = owner->window_event(hwnd, msg, wparam, lparam);
2754 if (msg == WM_INPUT) DefWindowProc(hwnd, msg, wparam, lparam);
2755 return result;
2756 }
2757 catch (...)
2758 {
2759 // Never unwind through a native callback. A display worker rethrows the
2760 // first error through handle_events(), and main keeps serving shutdown.
2761 owner->application.record_callback_error();

2763 return msg == WM_NCCREATE ? FALSE : 0;
