234static void append_text(char* out, int out_len, int* pos, const char* text)
235{
236 if (*pos >= out_len - 1)
237 return;
238 int left = out_len - *pos;
239 int wrote = snprintf(out + *pos, left, "%s", text);
240 if (wrote < 0)
241 return;

243 *pos = out_len - 1;
244 else
245 *pos += wrote;

259static void describe_dhcp(const u8* bootp, int len, char* out, int out_len)
260{
261 int pos = 0;
262 int msg_type = -1;
263 const u8* subnet = NULL;
264 const u8* router = NULL;
265 const u8* server = NULL;
266 const u8* requested = NULL;
267 char text[48];
268 char yiaddr[16];
269
270 out[0] = '\0';
271 if (len < 240)
272 return;
273 if (bootp[236] != 0x63 || bootp[237] != 0x82 ||
274 bootp[238] != 0x53 || bootp[239] != 0x63)
275 return;
276
277 for (int off = 240; off < len;)
278 {
279 int opt = bootp[off++];
280 if (opt == 0)
281 continue;
282 if (opt == 255)
283 break;
284 if (off >= len)
285 break;
286
287 int opt_len = bootp[off++];
288 if (opt_len < 0 || off + opt_len > len)
289 break;
290
291 switch (opt)
292 {
293 case 1:
294 if (opt_len >= 4)
295 subnet = &bootp[off];
296 break;
297 case 3:
298 if (opt_len >= 4)
299 router = &bootp[off];
300 break;
301 case 50:
302 if (opt_len >= 4)
303 requested = &bootp[off];
304 break;
305 case 53:
306 if (opt_len >= 1)
307 msg_type = bootp[off];
308 break;
309 case 54:
310 if (opt_len >= 4)
311 server = &bootp[off];
312 break;
313 default:
314 break;
315 }
316 off += opt_len;
317 }
318
319 if (msg_type >= 0)
320 {
321 snprintf(text, sizeof(text), " DHCP=%s(%d)",
322 dhcp_type_name(msg_type), msg_type);
323 append_text(out, out_len, &pos, text);
324 }
325 else
326 append_text(out, out_len, &pos, " DHCP");
327
328 if (bootp[16] || bootp[17] || bootp[18] || bootp[19])
329 {
330 pkt_format_ip(yiaddr, &bootp[16]);
331 snprintf(text, sizeof(text), " yiaddr=%s", yiaddr);
332 append_text(out, out_len, &pos, text);
333 }
334 if (subnet)
335 append_dhcp_ip(out, out_len, &pos, "subnet", subnet);
336 if (router)
337 append_dhcp_ip(out, out_len, &pos, "router", router);
338 if (server)
339 append_dhcp_ip(out, out_len, &pos, "server", server);
340 if (requested)
341 append_dhcp_ip(out, out_len, &pos, "request", requested);
342}

356void CDEC21143::run()
357{
358 try
359 {
360 for (;;)
361 {
362 if (StopThread)
363 return;
364
365 /* Deferred SIA autoneg: complete outside the guest's CSR write. */
366 if (autoneg_pending.load(std::memory_order_acquire) &&
367 std::chrono::steady_clock::now() >= autoneg_complete_at)
368 {
369 autoneg_pending = false;
370 complete_sia_autoneg();
371 }
372
373 receive_process();
374
375 bool asserted;
376
377 if ((state.reg[CSR_OPMODE / 8] & OPMODE_ST))
378 while (dec21143_tx());
379
380 /* Normal and Abnormal interrupt summary (align with 21143/QEMU). */
381 {
382 u32 ie = state.reg[CSR_STATUS / 8] & state.reg[CSR_INTEN / 8];
383 state.reg[CSR_STATUS / 8] &= ~(STATUS_NIS | STATUS_AIS);
384 /* NIS if any normal event is enabled + set */
385 if (ie & (STATUS_TI | STATUS_TU | STATUS_RI | STATUS_TM | STATUS_ER))
386 state.reg[CSR_STATUS / 8] |= STATUS_NIS;
387 /* AIS if any abnormal event is enabled + set */
388 if (ie & (STATUS_LC | STATUS_GPPI | STATUS_SE | STATUS_LNF | STATUS_ETI |
389 STATUS_RWT | STATUS_RPS | STATUS_RU | STATUS_UNF | STATUS_LNPANC |
390 STATUS_TJT | STATUS_TPS))
391 state.reg[CSR_STATUS / 8] |= STATUS_AIS;
392 asserted = (state.reg[CSR_STATUS / 8] & state.reg[CSR_INTEN / 8] &
393 (STATUS_AIS | STATUS_NIS)) != 0;
394 }
395
396 if (asserted != state.irq_was_asserted)
397 {
398#if defined(DEBUG_NIC_IRQ)
399 printf("21143: IRQ %s CSR5=%08x CSR7=%08x (bg)\n",
400 asserted ? "ASSERT" : "DEASSERT",
401 state.reg[CSR_STATUS / 8], state.reg[CSR_INTEN / 8]);
402#endif
403 if (do_pci_interrupt(0, asserted))
404 state.irq_was_asserted = asserted;
405 }
406
407 mySemaphore.tryWait(10);
408 }
409 }
410
411 catch (CException& e)
412 {
413 printf("Exception in NIC thread: %s.\n", e.displayText().c_str());
414
415 // Let the thread die...
416 }
417}

476void CDEC21143::init()
477{
478 char* cfg;
479
480 add_function(0, dec21143_cfg_data, dec21143_cfg_mask);
481
482 net_backend = create_network_backend(devid_string, myCfg);
483 if (!net_backend)
484 FAILURE(Runtime, "Could not create a network backend (see messages above)");
485
486 if (!net_backend->init(devid_string, myCfg))
487 {
488 delete net_backend;
489 net_backend = nullptr;
490 FAILURE(Runtime, "Could not initialize the network backend (see messages above)");
491 }
492
493 // set default mac = Digital ethernet prefix: 08-00-2B + hexified "ES40" + nic number
494 state.mac[0] = 0x08;
495 state.mac[1] = 0x00;
496 state.mac[2] = 0x2B;
497 state.mac[3] = 0xE5;
498 state.mac[4] = 0x40;
499 state.mac[5] = nic_num++;
500
501 // set assigned mac
502 cfg = myCfg->get_text_value("mac");
503 if (cfg)
504 {
505 const char* mac_chars = "0123456789abcdefABCDEF-.:";
506 const char* hex_chars = "0123456789abcdefABCDEF";
507 const char* hex_scanf = "%hx";
508 bool mac_replaced = false;
509 if ((strlen(cfg) == 17) && (strspn(cfg, mac_chars) == 17))
510 {
511 char newmac[18];
512 strcpy(newmac, cfg);
513 newmac[2] = newmac[5] = newmac[8] = newmac[11] = newmac[14] = 0;
514 if ((strspn(&newmac[0], hex_chars) == 2)
515 && (strspn(&newmac[3], hex_chars) == 2)
516 && (strspn(&newmac[6], hex_chars) == 2)
517 && (strspn(&newmac[9], hex_chars) == 2)
518 && (strspn(&newmac[12], hex_chars) == 2)
519 && (strspn(&newmac[15], hex_chars) == 2))
520 {
521 short unsigned int num;
522 sscanf(&newmac[0], hex_scanf, &num);
523 state.mac[0] = num & 0xff;
524 sscanf(&newmac[3], hex_scanf, &num);
525 state.mac[1] = num & 0xff;
526 sscanf(&newmac[6], hex_scanf, &num);
527 state.mac[2] = num & 0xff;
528 sscanf(&newmac[9], hex_scanf, &num);
529 state.mac[3] = num & 0xff;
530 sscanf(&newmac[12], hex_scanf, &num);
531 state.mac[4] = num & 0xff;
532 sscanf(&newmac[15], hex_scanf, &num);
533 state.mac[5] = num & 0xff;
534 mac_replaced = true;
535 }
536 }
537
538 if (mac_replaced)
539 {
540 printf("%s: MAC set to %s\n", devid_string, cfg);
541 }
542 else
543 {
544 FAILURE_1(Configuration,
545 "MAC address (%s) should have xx-xx-xx-xx-xx-xx format", cfg);
546 }
547 }
548 else
549 {
550 char mac[18];
551 sprintf(mac, "%02X-%02X-%02X-%02X-%02X-%02X", state.mac[0], state.mac[1],
552 state.mac[2], state.mac[3], state.mac[4], state.mac[5]);
553 printf("%s: MAC defaulted to %s\n", devid_string, mac);
554 }
555
556 rx_queue = new CPacketQueue("rx_queue",
557 (int)myCfg->get_num_value("queue", false, 1024));
558 calc_crc = myCfg->get_bool_value("crc", false);
559 trace_packets = myCfg->get_bool_value("trace_packets", false);
560 /* Hidden option: defer SIA autoneg completion ~50ms. */
561 autoneg_delay_enabled = myCfg->get_bool_value("autonegotiate_delay", false);
562
563 state.rx.cur_buf = NULL;
564 /* Use a 2KB TX scratch buffer like QEMU's tulip (tx_frame[2048]) to avoid overflows. */
565 state.tx.cur_buf = (unsigned char*)malloc(nic_tx_capacity);
566 state.irq_was_asserted = false;
567
568 ResetPCI();
569 ResetNIC(); // explicit one-shot internal reset; previously implicit via ResetPCI
570
571 myThread = 0;
572
573 printf("%s: $Id$\n",
574 devid_string);
575}

599void CDEC21143::rx_wake_loop()
600{
601 struct pollfd pfd;
602 pfd.fd = net_backend->get_wait_fd();
603 pfd.events = POLLIN;
604
605 while (!StopThread)
606 {
607 int r = poll(&pfd, 1, 250);
608 if (StopThread)
609 break;
610 if (r > 0 && (pfd.revents & (POLLIN | POLLERR | POLLHUP)))
611 {
612 mySemaphore.tryWait(0);
613 mySemaphore.set();
614
615 /* Let the service thread drain the fd; avoids a
616 * level-triggered spin. */
617 usleep(500);
618 }
619 }

696void CDEC21143::receive_process()
697{
698 const u8* packet_data = NULL;
699 int packet_len = 0;
700
701 // if receive process active
702 if (state.reg[CSR_OPMODE / 8] & OPMODE_SR)
703 {
704
705 // get packets from host nic if not in internal loopback mode
706 if (!(state.reg[CSR_OPMODE / 8] & OPMODE_OM_INTLOOP))
707 {
708 while (net_backend->receive(&packet_data, &packet_len) > 0)
709 {
710 if (trace_packets)
711 trace_packet("RX", packet_data, packet_len);
712 bool resl = rx_queue->add_tail(packet_data, packet_len,
713 calc_crc, true);
714 state.reg[CSR_SIASTAT / 8] |= SIASTAT_TRA; //set 10bT activity
715 }
716 }
717
718 // process a receive descriptor until we run out of
719 // descriptors or packets to process
720 while (dec21143_rx());
721 }

724void CDEC21143::trace_packet(const char* dir, const u8* frame, int len)
725{
726 char dst[18];
727 char src[18];
728 u16 ether_type;
729
730 if (len < 14)
731 {
732 printf("%s: %s len=%d short ethernet frame\n", devid_string, dir, len);
733 return;
734 }
735
736 pkt_format_mac(dst, &frame[0]);
737 pkt_format_mac(src, &frame[6]);
738 ether_type = pkt_be16(&frame[12]);
739
740 if (ether_type == 0x0806 && len >= 42)
741 {
742 char sha[18];
743 char tha[18];
744 char spa[16];
745 char tpa[16];
746 u16 op = pkt_be16(&frame[20]);
747
748 pkt_format_mac(sha, &frame[22]);
749 pkt_format_ip(spa, &frame[28]);
750 pkt_format_mac(tha, &frame[32]);
751 pkt_format_ip(tpa, &frame[38]);
752 printf("%s: %s len=%d %s -> %s ARP op=%u sha=%s spa=%s tha=%s tpa=%s\n",
753 devid_string, dir, len, src, dst, op, sha, spa, tha, tpa);
754 return;
755 }
756
757 if (ether_type == 0x0800 && len >= 34)
758 {
759 const u8* ip = &frame[14];
760 int ihl = (ip[0] & 0x0f) * 4;
761 char src_ip[16];
762 char dst_ip[16];
763 u8 proto;
764
765 if (ihl < 20 || len < 14 + ihl)
766 {
767 printf("%s: %s len=%d %s -> %s IPv4 malformed\n",
768 devid_string, dir, len, src, dst);
769 return;
770 }
771
772 proto = ip[9];
773 pkt_format_ip(src_ip, &ip[12]);
774 pkt_format_ip(dst_ip, &ip[16]);
775
776 if (proto == 17 && len >= 14 + ihl + 8)
777 {
778 const u8* udp = ip + ihl;
779 int udp_payload_len = len - 14 - ihl - 8;
780 u16 sport = pkt_be16(&udp[0]);
781 u16 dport = pkt_be16(&udp[2]);
782 char dhcp[220];
783
784 dhcp[0] = '\0';
785 if ((sport == 67 || sport == 68 || dport == 67 || dport == 68) &&
786 udp_payload_len > 0)
787 describe_dhcp(udp + 8, udp_payload_len, dhcp, sizeof(dhcp));
788
789 printf("%s: %s len=%d %s -> %s IPv4 %s:%u -> %s:%u UDP%s\n",
790 devid_string, dir, len, src, dst, src_ip, sport,
791 dst_ip, dport, dhcp);
792 return;
793 }
794
795 if (proto == 1 && len >= 14 + ihl + 2)
796 {
797 const u8* icmp = ip + ihl;
798 printf("%s: %s len=%d %s -> %s IPv4 %s -> %s ICMP type=%u code=%u\n",
799 devid_string, dir, len, src, dst, src_ip, dst_ip,
800 icmp[0], icmp[1]);
801 return;
802 }
803
804 printf("%s: %s len=%d %s -> %s IPv4 %s -> %s proto=%u\n",
805 devid_string, dir, len, src, dst, src_ip, dst_ip, proto);
806 return;
807 }
808
809 printf("%s: %s len=%d %s -> %s ethertype=0x%04x\n",
810 devid_string, dir, len, src, dst, ether_type);
811}

816u32 CDEC21143::nic_read(u32 address, int dsize)
817{
818 u32 data = 0;
819
820 u32 oldreg = 0;
821 int regnr = (int)(address >> 3);
822
823 if ((address & 7) == 0 && regnr < 32)
824 {

1504 // (HRM: "a software or hardware reset is required")
1505 // frames arriving meanwhile are discarded and counted.
1506 // Re-enabling BME doesn't revive the engines.

1508 {

1510 u32 missed = state.reg[CSR_MISSED / 8];
1511 if ((missed & 0xffff) == 0xffff) missed |= 0x10000; else missed++;
1512 state.reg[CSR_MISSED / 8] = missed;
1513 return 0;
1514 }

1517 // latch SE, stop both processes.

1521 set_rx_state(STATUS_RS_STOPPED);
1522 set_tx_state(STATUS_TS_STOPPED);
1523 state.tx.suspend = true;
1524 state.rx.current.used = state.rx.current.len; // discard
1525 u32 missed = state.reg[CSR_MISSED / 8];
1526 if ((missed & 0xffff) == 0xffff) missed |= 0x10000; else missed++;
1527 state.reg[CSR_MISSED / 8] = missed;
1528 return 0;

1534 if (state.reg[CSR_BUSMODE / 8] & BUSMODE_DBO) {
1535 for (int i = 0; i < 4; i++) descr[i] = bswap32_local(descr[i]);

1540 rdes2 = descr[2];
1541 rdes3 = descr[3];

1590 if (buf1_size > 0)

1594 to_xfer = buf1_size;

1597 do_pci_write(rdes2, &state.rx.current.frame[state.rx.current.used], 1,
1598 to_xfer);

1601 state.rx.current.used += to_xfer;

1644 }
1645 if (!perfect && memcmp(dst, state.mac, 6) == 0) {
1646 perfect = true;

1651 rdes0 |= TDSTAT_Rx_FF;
1652
1653 }
1654 }

1658 rdes0 |= TDSTAT_Rx_DT_IL;
1659 }
1660 else if (state.reg[CSR_OPMODE / 8] & OPMODE_OM_EXTLOOP) {
1661 rdes0 |= TDSTAT_Rx_DT_EL;
1662 }
1663 else {
1664 rdes0 |= TDSTAT_Rx_DT_SR;
1665 }
1666

1668 if (state.rx.current.len > 1518)
1669 rdes0 |= TDSTAT_Rx_TL;

1672 // set receive interrupt and receive state to waiting-for-packet
1673 state.reg[CSR_STATUS / 8] = (state.reg[CSR_STATUS / 8] & ~STATUS_RS) |
1674 STATUS_RI |
1675 STATUS_RS_WAIT;
1676 }
1677
1678 // Writeback rdes0, others are read-only
1679 state.reg[CSR_STATUS / 8] = (state.reg[CSR_STATUS / 8] & ~STATUS_RS) | STATUS_RS_CLOSE;
1680 // respect DBO bit - swap descriptor byte order if needed
1681 {
1682 u32 w0 = rdes0;
1683 if (state.reg[CSR_BUSMODE / 8] & BUSMODE_DBO) w0 = bswap32_local(w0);
1684 do_pci_write(state.rx.cur_addr, &w0, 4, 1);
1685 }
1686
1687 // move to next descriptor
1688 if (rdes1 & TDCTL_ER) // end-of-ring, return to base
1689 state.rx.cur_addr = state.reg[CSR_RXLIST / 8];
1690 else
1691 {
1692 if (rdes1 & TDCTL_CH) // explicit chain, use chain address
1693 state.rx.cur_addr = rdes3;
1694 else
1695 // implicit chain
1696 state.rx.cur_addr += (4 * sizeof(uint32_t)) + state.descr_skip;
1697 }
1698
1699 return 1; // indicate processing has occurred
