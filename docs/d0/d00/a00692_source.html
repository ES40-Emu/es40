554void CSym53C810::run()
555{
556 try
557 {
558 for (;;)
559 {
560 mySemaphore.wait();
561 if (StopThread)
562 return;
563 bool fresh_wake = true;
564 for (;;)
565 {
566 // Peer DMA can enter another device: take the bus before the register lock.
567 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
568 CScopedLock<CMutex> regLock(myRegLock);
569 if (fresh_wake)
570 {
571 state.insn_processed = 0;
572 fresh_wake = false;
573 if (state.executing)
574 SYMDIAG('K', 0, 0);
575 }
576 if (!state.executing)
577 break;
578 scripts_running = true;
579 try
580 {
581 execute();
582 }
583 catch (...)
584 {
585 scripts_running = false;
586 throw;
587 }
588 scripts_running = false;
589 }
590 }
591 }
592
593 catch (CException& e)
594 {
595 printf("Exception in SYM thread: %s.\n", e.displayText().c_str());
596
597 // Let the thread die...
598 }
599}

829 void* p;
830
831 switch (bar)
832 {
833 case 0:
834 case 1:
835 address &= 0x7f;
836 switch (dsize)

838 case 8:

844#endif

1888 int scsi_phase = (R8(DCMD) >> 0) & 7;
1889
1890#if defined(DEBUG_SYM_SCRIPTS)

1892 table_indirect, opcode, scsi_phase);

1899#endif

1901 return;

1909 // Timeout or disconnect — check_phase already raised the
1910 // appropriate interrupt

1918 printf("SYM: Phase mismatch! Expected %d, got different.\n", scsi_phase);

1929#if defined(DEBUG_SYM_SCRIPTS)
1930 printf("SYM: Ready for transfer.\n");

1934 u32 count;
1935
1936 if (table_indirect)
1937 {
1938 u32 add = R32(DSA) + sext_u32_24(R32(DSPS));
1939
1940 add &= ~0x03; // 810
1941#if defined(DEBUG_SYM_SCRIPTS)
1942 printf("SYM: Reading table at DSA(%08x)+DSPS(%08x) = %08x.\n",
1943 R32(DSA), R32(DSPS), add);
1944#endif

1946 count &= 0x00ffffff;
1947 do_pci_read(add + 4, &start, 4, 1);
1948 }
1949 else if (indirect)
1950 {
1951 do_pci_read(R32(DSPS), &start, 4, 1);
1952 count = GET_DBC();
1953 }
1954 else

1957 count = GET_DBC();

1960#if defined(DEBUG_SYM_SCRIPTS)
1961 printf("SYM: %08x: MOVE Start/count %x, %x\n", R32(DSP) - 8, start,
1962 count);
1963#endif

2009

2011 case SCSI_PHASE_DATA_IN:
2012 case SCSI_PHASE_MSG_IN:

2044 // residual via DBC.

2048 }

2053/* Execute one SCRIPTS I/O instruction
2054 *
2055 * The I/O instructions perform common SCSI hardware sequences, like
2056 * Selection and reselection. The instruction format in
2057 * the DCMD and DBC register is as follows:
2058 *
2059 * \code
2060 * +---+-----+-+-+-+
2061 * |7 6|5 4 3|2|1|0| DCMD Register
2062 * +---+-----+-+-+-+
2063 * | | | | +- 0: Select with ATN/:
2064 * | | | | Valid only for Select instruction. Assert ATN/ during
2065 * | | | | selection
2066 * | | | +- 1: Table Indirect Mode:
2067 * | | | 0: All information is taken from the instruction,
2068 * | | | and the contents of the SCNTL3 and SXFER registers.
2069 * | | | 1: The DSPS register contains a 24-bit signed offset
2070 * | | | that is added to the DSA register to get a pointer
2071 * | | | to a data structure that contains the destination
2072 * | | | ID, SCNTL3 bits, and SXFER bits. This structure
2073 * | | | is 32 bits long and looks as follows:
2074 * | | | +--------+--------+--------+--------+
2075 * | | | DSA+DSPS: | SCNTL3 | ID | SXFER | |
2076 * | | | +--------+--------+--------+--------+
2077 * | | +- 2: Relative Addrressing:
2078 * | | 0: The value in the DNAD register is an absolute address.
2079 * | | 1: The value in the DNAD register is a 24-bit signed
2080 * | | displacement from the current DSP address.
2081 * | +- 3..5: Op Code
2082 * +- 6..7 Instruction Type: 01 = I/O
2083 *
2084 * +-------+-------++--------+--+-+-++-+-+---+-+-----+
2085 * | |19 16|| |10|9| || |6| |3| | DBC Register
2086 * +-------+-------++--------+--+-+-++-+-+---+-+-----+
2087 * | | | | +- 3: Set/Clear ATN
2088 * | | | +- 6: Set/Clear ACK
2089 * | | +- 9: Set/Clear Target Mode
2090 * | +- 10: Set/Clear Carry
2091 * +- 16..19: Destination ID
2092 * \endcode
2093 *
2094 * The Opcode determines the actual instruction:
2095 * \code
2096 * +-----+-----------------+-------------+
2097 * | OPC | Initiator mode | Target Mode |
2098 * +-----+-----------------+-------------+
2099 * | 000 | Select | Reselect |
2100 * | 001 | Wait Disconnect | Disconnect |
2101 * | 010 | Wait Reselect | Wait Select |
2102 * | 011 | Set | Set |
2103 * | 100 | Clear | Clear |
2104 * +-----+-----------------+-------------+
2105 * \endcode
2106 *
2107 * Select:
2108 * - Arbitrate for the SCSI bus until arbitration is won.
2109 * - If arbitration is won, try to select the destination ID
2110 * - If the controller is selected or reselected before winning
2111 * arbitration, jump to the address in the DNAD register.
2112 * .
2113 *
2114 * Wait Disconnect:
2115 * - Wait for the target to disconnect from the SCSI bus.
2116 * .
2117 *
2118 * Wait Reselect:
2119 * - If the controller is reselected, go to the next instruction.
2120 * - If the controller is selected before being reselected, or if
2121 * the CPU sets the SIGP flag, jump to the address in the DNAD
2122 * register
2123 * .
2124 *
2125 * Set/Clear:
2126 * - Set or Clear the flags whose Set/Clear bits are set in the
2127 * instruction.
2128 * .
2129 **/
