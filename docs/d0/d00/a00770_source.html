175inline bool qmp_key_code(const std::string& name, u32& key)
176{
177 for (const auto& entry : es40_qmp_keymap::keys)
178 if (entry.name && name == entry.name)
179 {
180 key = entry.key;
181 return true;
182 }
183 return false;
184}
185
186inline bool qmp_key_number(unsigned number, u32& key)
187{
188 for (const auto& entry : es40_qmp_keymap::keys)
189 if (number == entry.number)
190 {
191 key = entry.key;
192 return true;
193 }
194 return false;
195}
