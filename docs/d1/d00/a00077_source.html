328{

330
331 add_legacy_io(PRI_COMMAND, 0x1f0, 8);
332 // In compatibility mode the control block is the alternate-status/device-control
333 // port at 3F6h; 3F7h remains available to the legacy floppy path.
334 add_legacy_io(PRI_CONTROL, 0x3f6, 1);
335 add_legacy_io(SEC_COMMAND, 0x170, 8);
336 add_legacy_io(SEC_CONTROL, 0x376, 1);

338 add_legacy_io(SEC_BUSMASTER, 0xf008, 8);

340 usedma = myCfg->get_bool_value("dma", true);
341 if (!usedma)
342 {
343 printf("IDE: DMA Transfers turned off.\n");
344 }
345
346 ResetPCI();
347
348 // start controller threads
349 StopThread = false;
350 mtRegisters[0] = new CRWLock("ide0-registers");
351 mtRegisters[1] = new CRWLock("ide1-registers");
352 mtBusMaster[0] = new CRWLock("ide0-busmaster");
353 mtBusMaster[1] = new CRWLock("ide1-busmaster");
354
355 for (int i = 0; i < 2; i++)
356 {
357 semController[i] = new CSemaphore(0, 1); // disk controller
358 semBusMaster[i] = new CSemaphore(0, 1); // bus master
359 thrController[i] = 0;
360 }

362 printf("%%IDE-I-INIT: New IDE emulator initialized.\n");

466int CAliM1543C_ide::RestoreState(FILE* f)
467{
468 long ss;
469 u32 m1;
470 u32 m2;
471 int res;
472 size_t r;
473
474 if ((res = CPCIDevice::RestoreState(f)))
475 return res;
476
477 r = fread(&m1, sizeof(u32), 1, f);
478 if (r != 1)
479 {
480 printf("%s: unexpected end of file!\n", devid_string);
481 return -1;
482 }
483
484 if (m1 != ide_magic1)
485 {
486 printf("%s: MAGIC 1 does not match!\n", devid_string);
487 return -1;
488 }
489
490 r = fread(&ss, sizeof(long), 1, f);
491 if (r != 1)
492 {
493 printf("%s: unexpected end of file!\n", devid_string);
494 return -1;
495 }
496
497 if (ss != sizeof(state))
498 {
499 printf("%s: STRUCT SIZE does not match!\n", devid_string);
500 return -1;
501 }
502
503 r = fread(&state, sizeof(state), 1, f);
504 if (r != 1)
505 {
506 printf("%s: unexpected end of file!\n", devid_string);
507 return -1;
508 }
509
510 r = fread(&m2, sizeof(u32), 1, f);
511 if (r != 1)
512 {
513 printf("%s: unexpected end of file!\n", devid_string);
514 return -1;
515 }
516
517 if (m2 != ide_magic2)
518 {
519 printf("%s: MAGIC 1 does not match!\n", devid_string);
520 return -1;
521 }
522
523 // Older snapshots saved this configuration bit as read-only. Keep the
524 // restored register value, but apply the current hardware write mask.
525 pci_state.config_mask[0][0x50 / 4] |= endian_32(0x08000000);
526
527 printf("%s: %d bytes restored.\n", devid_string, (int)ss);
528 return 0;
529}

534bool CAliM1543C_ide::decodes_memory_access(int index, u64 address, int dsize,
535 bool write) const noexcept
536{
537 if (!CPCIDevice::decodes_memory_access(index, address, dsize, write))
538 return false;
539
540 // M1543C section 4.2.3.4: compatibility channels use fixed legacy
541 // ports; native channels use the command/control BARs. A programmed
542 // BAR must not create an alias while its channel is in compatibility
543 // mode: the default control BARs otherwise steal the floppy ports.
544 const bool io_enabled = (endian_32(pci_state.config_data[0][1]) & 1U) != 0;
545 switch (index)
546 {
547 case PRI_COMMAND:
548 case PRI_CONTROL:
549 return io_enabled && !channel_is_native(0);
550 case SEC_COMMAND:
551 case SEC_CONTROL:
552 return io_enabled && !channel_is_native(1);
553 }
554
555 if (index >= PCI_RANGE_BASE + BAR_PRI_COMMAND &&
556 index <= PCI_RANGE_BASE + BAR_SEC_CONTROL)
557 {
558 const int bar = index - PCI_RANGE_BASE;
559 if (!io_enabled || !channel_is_native(bar / 2))
560 return false;
561 // Register 50h.5 narrows native control BARs to their third byte.
562 const bool third_byte_only =
563 (endian_32(pci_state.config_data[0][0x50 / 4]) & 0x20) != 0;
564 return !(bar == BAR_PRI_CONTROL || bar == BAR_SEC_CONTROL) ||
565 !third_byte_only || address == 2;
566 }
567
568 return true;
569}

571u32 CAliM1543C_ide::config_read_custom(int func, u32 address, int dsize, u32 data)
572{
573 // M1543C 53h.3 masks the visible BAR contents in compatibility mode;
574 // it does not clear their programmed addresses or change I/O decoding.
575 if (func == 0 &&
576 (endian_32(pci_state.config_data[0][0x50 / 4]) & 0x08000000))
577 {
578 for (u32 byte = 0; byte < (u32)dsize / 8 && byte < 4; ++byte)
579 {
580 const u32 offset = address + byte;
581 if (offset >= 0x10 && offset < 0x20 &&
582 !channel_is_native((offset - 0x10) / 8))
583 data &= ~(0xffU << (byte * 8));
584 }
585 }
586 return data;
587}

703 * Register read/write handlers
704 */
705u32 CAliM1543C_ide::ide_command_read(int index, u32 address, int dsize)
706{
707 u32 data = 0;
708 if (!get_disk(index, 0) && !get_disk(index, 1))
709 {
710
711 // no disks are present, so the data lines actually float to
712 // a high state, which is logical 1.
713 return 0xffffffff;
714 }
715
716 switch (address)
717 {
718 case REG_COMMAND_DATA:
719 if (SEL_STATUS(index).drq && abort_if_offline(index))
720 break;
721 if (!SEL_STATUS(index).drq)
722 {
723#ifdef DEBUG_IDE_REG_COMMAND
724 printf("Reading from data buffer when data is not ready.\n");
725 ide_status(index);
726 PAUSE("WTF");
727#endif
728 break;
729 }
730
731 data = 0;
732
733 switch (dsize)
734 {
735 case 32:
736 data = CONTROLLER(index).data[CONTROLLER(index).data_ptr++];
737 data |= CONTROLLER(index).data[CONTROLLER(index).data_ptr++] << 16;
738 break;
739
740 case 16:
741 data = CONTROLLER(index).data[CONTROLLER(index).data_ptr++];
742 }
743
744 if (CONTROLLER(index).data_ptr >= CONTROLLER(index).data_size)
745 {
746
747 // there's no more to take.
748 SEL_STATUS(index).drq = false;
749 if (SEL_COMMAND(index).command_in_progress)
750 {
751 SEL_STATUS(index).busy = true;
752 SEL_STATUS(index).drive_ready = false;
753 UPDATE_ALT_STATUS(index);
754 semController[index]->set(); // wake up the controller.
755#if defined(DEBUG_IDE_MULTIPLE) || defined(DEBUG_IDE_PACKET)
756 printf("Command still in progress, waking up controller.\n");
757 printf("-- Packet Phase: %d\n", SEL_COMMAND(index).packet_phase);
758 ide_status(index);
759#endif
760 }
761 }
762
763 if (CONTROLLER(index).data_ptr > IDE_BUFFER_SIZE)
764 {
765 printf("%%IDE-W-OVERFLOW: data pointer past end of buffer, setting to 0.\n");
766 CONTROLLER(index).data_ptr = 0;
767 SEL_STATUS(index).drq = false;
768 UPDATE_ALT_STATUS(index);
769 }
770 break;
771
772 case REG_COMMAND_ERROR:
773 data = SEL_REGISTERS(index).error;
774 break;
775
776 case REG_COMMAND_SECTOR_COUNT:
777 data = SEL_REGISTERS(index).sector_count;
778 break;
779
780 case REG_COMMAND_SECTOR_NO:
781 data = SEL_REGISTERS(index).sector_no;
782 break;
783
784 case REG_COMMAND_CYL_LOW:
785 data = SEL_REGISTERS(index).cylinder_no & 0xff;
786 break;
787
788 case REG_COMMAND_CYL_HI:
789 data = (SEL_REGISTERS(index).cylinder_no >> 8) & 0xff;
790 break;
791
792 case REG_COMMAND_DRIVE:
793 data = 0x80 | (SEL_REGISTERS(index).lba_mode ? 0x40 : 0x00) | 0x20 // 512 byte sector size
794 | (CONTROLLER(index).selected ? 0x10 : 0x00) | (SEL_REGISTERS(index).head_no & 0x0f);
795 break;
796
797 case REG_COMMAND_STATUS:
798
799 // get the status and clear the interrupt.
800 data = get_status(index);
801 deassert_interrupt(index);
802#ifdef DEBUG_IDE_INTERRUPT
803 printf("%%IDE-I-INTERRUPT: Interrupt Acknowledged on %d.\n", index);
804#endif
805 break;
806 }
807
808#ifdef DEBUG_IDE_REG_COMMAND
809 if (address != 0)
810 {
811 printf("%%IDE-I-REGCMD: Read from command register %d (%s) on controller %d, value: %x\n",
812 address, register_names[address], index, data);
813 }
814#endif
815 return data;
816}

818void CAliM1543C_ide::ide_command_write(int index, u32 address, int dsize,
819 u32 data)
820{
821#ifdef DEBUG_IDE_REG_COMMAND
822 if (address != 0)
823 {
824 printf("%%IDE-I-REGCMD: Write to command register %d (%s) on controller %d, value: %x\n",
825 address, register_names[address], index, data);
826 }
827
828 SEL_STATUS(index).debug_status_update = true;
829#endif
830 switch (address)
831 {
832 case REG_COMMAND_DATA:
833 if (SEL_STATUS(index).drq && abort_if_offline(index))
834 break;
835 if (!SEL_STATUS(index).drq)
836 {
837#ifdef DEBUG_IDE_REG_COMMAND
838 printf("%%IDE-I-DATA: Unrequested data written to data port: %x\n", data);
839 ide_status(index);
840#endif
841 break;
842 }
843
844 switch (dsize)
845 {
846 case 32:
847 CONTROLLER(index).data[CONTROLLER(index).data_ptr++] = data & 0xffff;
848 CONTROLLER(index).data[CONTROLLER(index).data_ptr++] =
849 (
850 data >>
851 16
852 ) & 0xffff;
853 break;
854
855 case 16:
856 CONTROLLER(index).data[CONTROLLER(index).data_ptr++] = data & 0xffff;
857 }
858
859 if (CONTROLLER(index).data_ptr >= CONTROLLER(index).data_size)
860 {
861
862 // we don't want any more.
863 SEL_STATUS(index).drq = false;
864 SEL_STATUS(index).busy = true;
865 UPDATE_ALT_STATUS(index);
866 semController[index]->set(); // wake the controller up.
867 }
868
869 if (CONTROLLER(index).data_ptr > IDE_BUFFER_SIZE)
870 {
871 printf("%%IDE-W-OVERFLOW: data pointer overflow, setting to 0.\n");
872 CONTROLLER(index).data_ptr = 0;
873 SEL_STATUS(index).drq = false;
874 UPDATE_ALT_STATUS(index);
875 }
876 break;
877
878 case REG_COMMAND_FEATURES:
879 REGISTERS(index, 0).features = data;
880 REGISTERS(index, 1).features = data;
881 break;
882
883 case REG_COMMAND_SECTOR_COUNT:
884 REGISTERS(index, 0).sector_count = REGISTERS(index, 1).sector_count = data & 0xff;
885 break;
886
887 case REG_COMMAND_SECTOR_NO:
888 REGISTERS(index, 0).sector_no = REGISTERS(index, 1).sector_no = data & 0xff;
889 break;
890
891 case REG_COMMAND_CYL_LOW:
892 REGISTERS(index, 0).cylinder_no = REGISTERS(index, 1).cylinder_no =
893 (REGISTERS(index, 1).cylinder_no & 0xff00) |
894 (data & 0xff);
895 break;
896
897 case REG_COMMAND_CYL_HI:
898 REGISTERS(index, 0).cylinder_no = REGISTERS(index, 1).cylinder_no =
899 (REGISTERS(index, 1).cylinder_no & 0xff) |
900 ((data << 8) & 0xff00);
901 break;
902
903 case REG_COMMAND_DRIVE:
904 if (((data >> 4) & 1) != CONTROLLER(index).selected)
905#ifdef DEBUG_IDE
906 printf("Setting selected device on %d to: %d. val=%02x\n", index,
907 (data >> 4) & 1, data);
908#endif
909 CONTROLLER(index).selected = (data >> 4) & 1;
910 REGISTERS(index, 0).head_no = REGISTERS(index, 1).head_no = data & 0x0f;
911 REGISTERS(index, 0).lba_mode = REGISTERS(index, 1).lba_mode =
912 (
913 data >>
914 6
915 ) & 1;
916
917 break;
918
919 case REG_COMMAND_COMMAND:
920 deassert_interrupt(index); // interrupt is cleared on write.
921 if (!SEL_DISK(index))
922 {
923#ifdef DEBUG_IDE
924 printf("%%IDE-I-NODEV: Command to non-existing device %d.%d. cmd=%x\n",
925 index, CONTROLLER(index).selected, data);
926#endif
927 }
928
929 if (SEL_COMMAND(index).command_in_progress == true)
930 {
931
932 // we're already working, why is another command being issued?
933#ifdef DEBUG_IDE
934 printf("%%IDE-W-CIP: Command is already in progress.\n");
935 PAUSE("dang it!\n");
936#endif
937 }
938
939 if ((data & 0xf0) == 0x10)
940 data = 0x10;
941
942 SEL_COMMAND(index).command_in_progress = false;
943 SEL_COMMAND(index).current_command = data;
944#ifdef DEBUG_IDE_CMD
945 printf("%%IDE-I-CMD: Command %02x issued on controller %d, disk %d.\n", data, index, CONTROLLER(index).selected);
946#endif
947 SEL_COMMAND(index).command_cycle = 0;
948 SEL_STATUS(index).drq = false;
949 UPDATE_ALT_STATUS(index);
950 CONTROLLER(index).data_ptr = 0;
951
952 if (data != 0x00)
953 {
954 SEL_STATUS(index).busy = true;
955 UPDATE_ALT_STATUS(index);
956 SEL_COMMAND(index).command_in_progress = true;
957 SEL_COMMAND(index).packet_phase = PACKET_NONE;
958 semController[index]->set(); // wake up the controller.
959 }
960 else
961 {
962
963 // this is a nop, so we cancel everything that's pending and
964 // pretend that this operation got done super fast!
965 if (SEL_DISK(index))
966 command_aborted(index, data);
967 }
968 break;
969 }

972u32 CAliM1543C_ide::ide_control_read(int index, u32 address)
973{
974 u32 data = 0;
975 switch (address)
976 {
977 case 0:
978 {
979 SCOPED_READ_LOCK(mtRegisters[index]);
980 // Compute live from current status flags rather than reading the
981 // cached alt_status, which only gets refreshed at UPDATE_ALT_STATUS
982 // call sites and can lag the controller thread (causing SRM read
983 // timeouts when busy goes false → drq true between updates).
984 if (SEL_DISK(index))
985 {

987 | (SEL_STATUS(index).drive_ready ? 0x40 : 0x00)
988 | (SEL_STATUS(index).fault ? 0x20 : 0x00)
989 | (SEL_STATUS(index).seek_complete ? 0x10 : 0x00)
990 | (SEL_STATUS(index).drq ? 0x08 : 0x00)
991 | (SEL_STATUS(index).index_pulse ? 0x02 : 0x00)
992 | (SEL_STATUS(index).err ? 0x01 : 0x00);
993 data = apply_offline_status(index, (u8)data);
994 }
995#ifdef DEBUG_IDE_REG_CONTROL
996 static u32 last_data = 0;
997 if (last_data != data)
998 {
999 printf("%%IDE-I-READCTRL: alternate status on IDE control %d: 0x%02x\n",
1000 index, data);
1001 }
1002
1003 last_data = data;
1004#endif
1005 }
1006
1007 break;
1008
1009 case 1:
1010
1011 // 3x7h drive address register. (floppy?)
1012 data |= (CONTROLLER(index).selected == 0) ? 1 : 2;
1013 data |= (SEL_REGISTERS(index).head_no) << 2;
1014 data = (~data) & 0xff; // negate everything
1015#ifdef DEBUG_IDE_REG_CONTROL
1016 printf("%%IDE-I-READCTRL: drive address port on IDE control %d: 0x%02x\n",
1017 index, data);
1018#endif
1019 break;
1020 }
1021
1022 return data;

1031#ifdef DEBUG_IDE_REG_CONTROL

1033 (u32)(address), index, data);
1034#endif

1037 case 0:

1039 CONTROLLER(index).reset = (data >> 2) & 1;

1081 set_signature(index, 1);
1082 CONTROLLER(index).selected = 0; // ATA: drive 0 is selected after reset
1083 }
1084 break;
1085
1086 case 1:
1087
1088 // floppy?
1089 break;
1090 }

1148void CAliM1543C_ide::ide_busmaster_write(int index, u32 address, u32 data,
1149 int dsize)
1150{
1151#ifdef DEBUG_IDE_BUSMASTER
1152 if (!(dsize == 8 && (address >= 4 && address <= 7)))
1153 {
1154 printf("%%IDE-I-WRITBUSM: write port %d on IDE bus master %d: 0x%02x, %d bytes\n",
1155 (u32)(address), index, data, dsize / 8);
1156 }
1157#endif
1158
1159 u32 prd_address;
1160
1161 // u32 base, control;
1162 switch (dsize)
1163 {
1164 case 32:
1165 ide_busmaster_write(index, address, data & 0xff, 8);
1166 ide_busmaster_write(index, address + 1, (data >> 8) & 0xff, 8);
1167 ide_busmaster_write(index, address + 2, (data >> 16) & 0xff, 8);
1168 ide_busmaster_write(index, address + 3, (data >> 24) & 0xff, 8);
1169 return;
1170
1171 case 16:
1172 ide_busmaster_write(index, address, data & 0xff, 8);
1173 ide_busmaster_write(index, address + 1, (data >> 8) & 0xff, 8);
1174 return;
1175 }
1176
1177 switch (address)
1178 {
1179 case 0: // command register
1180#ifdef DEBUG_IDE_BUSMASTER
1181 printf("%%IDE-I-BUSM: Bus master command got data: %x (%s,%s)\n", data,
1182 (data & 0x08 ? "write" : "read"), (data & 0x01 ? "start" : "stop"));
1183#endif
1184
1185 // bits 7:4 & 2:1 are reserved and must return zero on read.
1186 CONTROLLER(index).busmaster[0] = data & 0x09;
1187 if (data & 0x01)
1188 {
1189
1190 // set the status register
1191 CONTROLLER(index).busmaster[2] |= 0x01;
1192 semBusMaster[index]->set(); // wake up the controller for busmastering
1193 }
1194 else
1195 {
1196
1197 // clear the status register
1198 CONTROLLER(index).busmaster[2] &= 0xfe;
1199 }
1200 break;
1201
1202 case 2: // status
1203 // bit 7 = simplex only (0=both channels are independent)
1204 // bit 6 = drive 1 dma capable.
1205 // bit 5 = drive 0 dma capable.
1206 // bit 4,3 = reserved
1207 // bit 2 = interrupt (write 1 to reset)
1208 // bit 1 = error (write 1 to reset)
1209 // bit 0 = busmaster active.
1210 {
1211 u8 status = CONTROLLER(index).busmaster[2];
1212 if (data & 0x04) // interrupt
1213 status &= ~0x04;
1214 if (data & 0x02) // error
1215 status &= ~0x02;
1216 if (data & 0x01) // active; several ALi drivers write this to work around sticky active
1217 status &= ~0x01;
1218 CONTROLLER(index).busmaster[2] = (status & 0x07) | (ide_busmaster_status(index) & 0x60);
1219 if ((data & 0x04) || (data & 0x01))
1220 deassert_interrupt(index);
1221 }
1222 break;
1223
1224 case 4: // descriptor table pointer register(s)
1225 case 5:
1226 case 6:
1227 CONTROLLER(index).busmaster[address] = data;
1228 break;
1229
1230 case 7:
1231 CONTROLLER(index).busmaster[address] = data;
1232 prd_address = endian_32(*(u32*)(&CONTROLLER(index).busmaster[4]));
1233#ifdef DEBUG_IDE_BUSMASTER
1234 printf("%%IDE-I-PRD: Virtual address: %" PRIx64 " \n",
1235 endian_32(*(u32*)(&CONTROLLER(index).busmaster[4])));
1236 printf("-IDE-I-PRD: Physical address: %" PRIx64 " \n", prd_address);
1237 u32 base, control;
1238 do
1239 {
1240 do_pci_read(prd_address, &base, 4, 1);
1241 do_pci_read(prd_address + 4, &control, 4, 1);
1242 printf("-IDE-I-PRD: base: %x, control: %x \n", base, control);
1243 prd_address += 8;
1244 } while ((control & 0x80000000) == 0);
1245#endif
1246 break;
1247
1248 default:
1249 break;
1250 }
1251}

1390void CAliM1543C_ide::identify_drive(int index, bool packet)
1391{
1392 char serial_number[21];
1393 char model_number[41];
1394 char rev_number[9];
1395 size_t i;
1396
1397 // clear the block.
1398 for (i = 0; i < 256; i++)
1399 CONTROLLER(index).data[i] = 0;
1400
1401 CONTROLLER(index).data_ptr = 0;
1402 CONTROLLER(index).data_size = 256;
1403
1404 // The data here was taken from T13/1153D revision 18
1405 if (!packet)
1406 {
1407
1408 // flags: 0x0080 = removable, 0x0040 = fixed.
1409 CONTROLLER(index).data[0] = SEL_DISK(index)->cdrom() ? 0x0080 : 0x0040;
1410 }
1411 else
1412 {
1413
1414 // flags: 15-14: 10=atapi, 11=reserved; 12-8: packet set; 7:
1415 // removable 6-5: timing info, 4-2: command, 1-0: 00= 12 byte packet
1416 CONTROLLER(index).data[0] = 0x8580;
1417 }
1418
1419 // logical cylinders
1420 if (SEL_DISK(index)->get_cylinders() > 16383)
1421 CONTROLLER(index).data[1] = 16383;
1422 else
1423 CONTROLLER(index).data[1] = (u16)(SEL_DISK(index)->get_cylinders());
1424
1425 // logical heads
1426 CONTROLLER(index).data[3] = (u16)(SEL_DISK(index)->get_heads());
1427
1428 // logical sectors per logical track
1429 CONTROLLER(index).data[6] = (u16)(SEL_DISK(index)->get_sectors());
1430
1431 // serial number
1432 strcpy(serial_number, " ");
1433 i = strlen(SEL_DISK(index)->get_serial());
1434 i = (i > 20) ? 20 : i;
1435 memcpy(serial_number, SEL_DISK(index)->get_serial(), i);
1436 for (i = 0; i < 10; i++)
1437 CONTROLLER(index).data[10 + i] =
1438 (
1439 serial_number[i * 2] <<
1440 8
1441 ) |
1442 serial_number[i * 2 + 1];
1443
1444 // firmware revision
1445 strcpy(rev_number, " ");
1446 i = strlen(SEL_DISK(index)->get_rev());
1447 i = (i > 8) ? 8 : i;
1448 memcpy(rev_number, SEL_DISK(index)->get_rev(), i);
1449 for (i = 0; i < 4; i++)
1450 CONTROLLER(index).data[23 + i] =
1451 (
1452 rev_number[i * 2] <<
1453 8
1454 ) |
1455 rev_number[i * 2 + 1];
1456
1457 // model number
1458 strcpy(model_number, " ");
1459 i = strlen(SEL_DISK(index)->get_model());
1460 i = (i > 40) ? 40 : i;
1461 memcpy(model_number, SEL_DISK(index)->get_model(), i);
1462 for (i = 0; i < 20; i++)
1463 CONTROLLER(index).data[i + 27] =
1464 (
1465 model_number[i * 2] <<
1466 8
1467 ) |
1468 model_number[i * 2 + 1];
1469
1470 // read/write multiple (15-8 = 0x80, 7-0 = # sectors)
1471 CONTROLLER(index).data[47] = 0x8000 | MAX_MULTIPLE_SECTORS;
1472
1473 // capabilities
1474 if (!packet)
1475 {
1476 CONTROLLER(index).data[49] = 0x0300;
1477 }
1478 else
1479 {
1480 CONTROLLER(index).data[49] = 0x0a00; // iordy; packet devices are PIO-only
1481 }
1482
1483 // capabilities (2)
1484 CONTROLLER(index).data[50] = 0x4000;
1485
1486 // pio data transfer number (bits 15-8)
1487 CONTROLLER(index).data[51] = 0x0300;
1488
1489 // validity: bit 2 = #88 valid, 1 = 64-70 valid, 0 = 54-58 valid
1490 CONTROLLER(index).data[53] = 7;
1491
1492 // geometry: word 54 is the "current" CHS cylinder count. ATA/ATAPI-4
1493 // caps it at 16383 just like word 1 — without the cap, a >=16384-cyl
1494 // disk reports e.g. 0x4000 here while word 1 reports 0x3fff and NT
1495 // logs "current CHS differs from default CHS".
1496 {
1497 u32 cyl = (u32)SEL_DISK(index)->get_cylinders();
1498 CONTROLLER(index).data[54] = (u16)(cyl > 16383 ? 16383 : cyl);
1499 }
1500 CONTROLLER(index).data[55] = (u16)(SEL_DISK(index)->get_heads());
1501 CONTROLLER(index).data[56] = (u16)(SEL_DISK(index)->get_sectors());
1502 CONTROLLER(index).data[57] = (u16)
1503 (SEL_DISK(index)->get_chs_size() >> 0) & 0xFFFF;
1504 CONTROLLER(index).data[58] = (u16)
1505 (SEL_DISK(index)->get_chs_size() >> 16) & 0xFFFF;
1506
1507 // multiple sector setting (valid, 1 sector per interrupt)
1508 if (SEL_PER_DRIVE(index).multiple_size != 0)
1509 {
1510 CONTROLLER(index).data[59] = 0x0100 | SEL_PER_DRIVE(index).multiple_size;
1511 }
1512 else
1513 {
1514 CONTROLLER(index).data[59] = 0x0000;
1515 }
1516
1517 // lba capacity
1518 CONTROLLER(index).data[60] = (u16)
1519 (SEL_DISK(index)->get_lba_size() >> 0) & 0xFFFF;
1520 CONTROLLER(index).data[61] = (u16)
1521 (SEL_DISK(index)->get_lba_size() >> 16) & 0xFFFF;
1522
1523 // multiword dma capability (10-8: modes selected, 2-0, modes
1524 // supported)
1525 if (usedma && !packet)
1526 CONTROLLER(index).data[63] = CONTROLLER(index).dma_mode << 8 | 0x01; // dma 0 supported
1527 else
1528 CONTROLLER(index).data[63] = CONTROLLER(index).dma_mode << 8 | 0x00; // dma not supported
1529
1530 // pio modes supported (bit 0 = mode 3, bit 1 = mode 4)
1531 CONTROLLER(index).data[64] = 0x0002;
1532
1533 // minimum cycle times
1534 CONTROLLER(index).data[65] = 480; // mode 0
1535 CONTROLLER(index).data[66] = 480; // mode 0
1536 CONTROLLER(index).data[67] = 120; // pio4
1537 CONTROLLER(index).data[68] = 120; // pio4
1538 if (packet)
1539 {
1540
1541 // packet to bus release time
1542 CONTROLLER(index).data[71] = 120;
1543
1544 // service to bus release time
1545 CONTROLLER(index).data[72] = 120;
1546 }
1547
1548 // queue depth (we don't do queing)
1549 CONTROLLER(index).data[75] = 0;
1550
1551 // ata version supported (bits/version: 1,2,3,4)
1552 CONTROLLER(index).data[80] = 0x001e;
1553
1554 // atapi revision supported (ata/atapi-4 T13 1153D revision 17)
1555 CONTROLLER(index).data[81] = 0x0017;
1556
1557 // command set supported (cdrom = nop,packet,removable; disk=nop)

1601void CAliM1543C_ide::advance_disk_address(int index, int sectors)
1602{
1603 if (sectors <= 0)
1604 return;
1605
1606 if (SEL_REGISTERS(index).lba_mode)
1607 {
1608 u32 lba = get_disk_lba(index) + sectors;
1609 SEL_REGISTERS(index).sector_no = lba & 0xff;
1610 SEL_REGISTERS(index).cylinder_no = (lba >> 8) & 0xffff;
1611 SEL_REGISTERS(index).head_no = (lba >> 24) & 0x0f;
1612 return;
1613 }
1614
1615 long heads = SEL_DISK(index)->get_heads();
1616 long sectors_per_track = SEL_DISK(index)->get_sectors();
1617 if (heads <= 0 || sectors_per_track <= 0)
1618 {
1619 FAILURE(InvalidArgument, "Invalid CHS disk geometry");
1620 }
1621
1622 int sector = (SEL_REGISTERS(index).sector_no - 1) + sectors;
1623 SEL_REGISTERS(index).sector_no = (sector % sectors_per_track) + 1;
1624
1625 int head = SEL_REGISTERS(index).head_no + (sector / sectors_per_track);
1626 SEL_REGISTERS(index).head_no = head % heads;
1627 SEL_REGISTERS(index).cylinder_no += head / heads;
1628}

1655bool CAliM1543C_ide::abort_if_offline(int index)
1656{
1657 CDisk* disk = SEL_DISK(index);
1658 if (!disk || !disk->is_offline())
1659 return false;
1660
1661 // Reset, identification and interface setup still work. Packet commands
1662 // reach CDisk's SCSI engine, which supplies NOT READY sense as appropriate.
1663 switch (SEL_COMMAND(index).current_command) {
1664 case 0x08: case 0x90: case 0x91:
1665 case 0xa0: case 0xa1: case 0xc6: case 0xec: case 0xef:
1666 return false;
1667 }
1668
1669 SEL_STATUS(index).busy = false;
1670 SEL_STATUS(index).drive_ready = true; // Offline is overlaid in status reads.
1671 SEL_STATUS(index).fault = false;
1672 SEL_STATUS(index).seek_complete = false;
1673 SEL_STATUS(index).drq = false;
1674 SEL_STATUS(index).err = true;
1675 SEL_REGISTERS(index).error = 0x04; // ABRT
1676 CONTROLLER(index).data_ptr = 0;
1677 CONTROLLER(index).data_size = 0;
1678 SEL_COMMAND(index).command_in_progress = false;
1679 if (SEL_COMMAND(index).current_command >= 0xc8 &&
1680 SEL_COMMAND(index).current_command <= 0xcb) {
1681 SCOPED_WRITE_LOCK(mtBusMaster[index]);
1682 // End DMA with error and completion pending, including a fault before
1683 // the guest sets the bus-master start bit.
1684 CONTROLLER(index).busmaster[2] =
1685 (CONTROLLER(index).busmaster[2] & ~0x01) | 0x06;
1686 }
1687 UPDATE_ALT_STATUS(index);
1688 raise_interrupt(index);
1689 return true;
1690}

1692bool CAliM1543C_ide::transfer_sectors(int index, u32 lba, void* buffer,
1693 size_t sectors, bool write)
1694{
1695 if (abort_if_offline(index))
1696 return false;
1697 CDisk* disk = SEL_DISK(index);
1698 bool ok = disk->seek_block(lba);
1699 if (ok)
1700 ok = (write ? disk->write_blocks(buffer, sectors) :
1701 disk->read_blocks(buffer, sectors)) == sectors;
1702 // The state may change between command dispatch and the backing I/O.
1703 if (abort_if_offline(index))
1704 return false;
1705 if (!ok)
1706 command_aborted(index, SEL_COMMAND(index).current_command);
1707 return ok;
1708}

1746void CAliM1543C_ide::execute(int index)
1747{
1748 if (abort_if_offline(index))
1749 return;
1750 if (SEL_DISK(index) == NULL && SEL_COMMAND(index).current_command != 0x90)
1751 {
1752 // Command issued to a nonexistent device. Clear BSY/DRQ that
1753 // REG_COMMAND_COMMAND set so the host's next status read returns
1754 // 0xff (via get_status()'s absent-drive path).
1755 SEL_STATUS(index).busy = false;
1756 SEL_STATUS(index).drq = false;
1757 SEL_COMMAND(index).command_in_progress = false;
1758 }
1759 else
1760 {
1761#ifdef DEBUG_IDE_COMMAND
1762 printf("%%IDE-I-COMMAND: Processing command on controller %d.\n", index);
1763 ide_status(index);
1764#endif
1765 switch (SEL_COMMAND(index).current_command)
1766 {
1767 case 0x00: // nop
1768 SEL_REGISTERS(index).error = 0x04;
1769 SEL_STATUS(index).busy = false;
1770 SEL_STATUS(index).drive_ready = true;
1771 SEL_STATUS(index).fault = true;
1772 SEL_STATUS(index).drq = false;
1773 SEL_STATUS(index).err = true;
1774 SEL_COMMAND(index).command_in_progress = false;
1775 raise_interrupt(index);
1776 printf("got nop on %d.%d\n", index, CONTROLLER(index).selected);
1777
1778 // FAILURE("This isn't possible, but you're seeing it.");
1779 break;
1780
1781 case 0x08: // device reset
1782 if (SEL_DISK(index)->cdrom() || 1)
1783 {
1784
1785 // the spec says that non-packet devices must not respond
1786 // to device reset. However, by allowing it, Tru64
1787 // recognizes the device properly.
1788 SEL_COMMAND(index).command_in_progress = false;
1789 if (CONTROLLER(index).selected == 0)
1790 {
1791 REGISTERS(index, 0).error = 0x01; // device passed.
1792 REGISTERS(index, 1).error = 0x01; // slave not present or passed
1793 }
1794 else
1795 {
1796 REGISTERS(index, 1).error = 0x01; // slave passed
1797 }
1798
1799 set_signature(index, CONTROLLER(index).selected);
1800
1801 // step "k", page 216 (232)
1802 SEL_STATUS(index).drq = false; // bit 3
1803 SEL_STATUS(index).bit_2 = false; // bit 2
1804 SEL_STATUS(index).err = false; // bit 0
1805 if (SEL_DISK(index)->cdrom())
1806 {
1807 SEL_STATUS(index).fault = false; // bit 5
1808 SEL_STATUS(index).drive_ready = false; // per step "m2"
1809 }
1810 else
1811 {
1812 SEL_STATUS(index).drive_ready = true; // step "m1"
1813 }
1814
1815 SEL_STATUS(index).busy = false;
1816
1817 // some sources say there's no reset on device reset.
1818 // raise_interrupt(index);
1819 }
1820 else
1821 {
1822 command_aborted(index, SEL_COMMAND(index).current_command);
1823 }
1824 break;
1825
1826 case 0x10: // calibrate drive
1827 SEL_STATUS(index).busy = false;
1828 SEL_STATUS(index).drive_ready = true;
1829 SEL_STATUS(index).seek_complete = true;
1830 SEL_STATUS(index).fault = false;
1831 SEL_STATUS(index).drq = false;
1832 SEL_STATUS(index).err = false;
1833 SEL_REGISTERS(index).cylinder_no = 0;
1834 SEL_COMMAND(index).command_in_progress = false;
1835 raise_interrupt(index);
1836 break;
1837
1838 case 0x20: // read with retries
1839 case 0x21: // read without retries
1840 if (SEL_COMMAND(index).command_cycle == 0)
1841 {
1842
1843 // fixup the 0=256 case.
1844 if (SEL_REGISTERS(index).sector_count == 0)
1845 SEL_REGISTERS(index).sector_count = 256;
1846 }
1847
1848 if (!SEL_STATUS(index).drq)
1849 {
1850
1851 // buffer is empty, so lets fill it.
1852 {
1853 u32 lba = get_disk_lba(index);
1854
1855 if (!transfer_sectors(index, lba, CONTROLLER(index).data, 1, false))
1856 break;
1857#if defined(ES40_BIG_ENDIAN)
1858 for (int i = 0; i < SEL_DISK(index)->get_block_size() / sizeof(u16);
1859 i++)
1860 CONTROLLER(index).data[i] = endian_16(CONTROLLER(index).data[i]);
1861#endif
1862 SEL_STATUS(index).busy = false;
1863 SEL_STATUS(index).drive_ready = true;
1864 SEL_STATUS(index).fault = false;
1865 SEL_STATUS(index).drq = true;
1866 SEL_STATUS(index).err = false;
1867 CONTROLLER(index).data_ptr = 0;
1868 CONTROLLER(index).data_size = 256;
1869
1870 // prepare for next sector
1871 SEL_REGISTERS(index).sector_count--;
1872 if (SEL_REGISTERS(index).sector_count == 0)
1873 {
1874 SEL_COMMAND(index).command_in_progress = false;
1875 if (SEL_DISK(index)->cdrom())
1876 set_signature(index, CONTROLLER(index).selected); // per 9.1
1877 }
1878 else
1879 {
1880
1881 // set the next block to read.
1882 advance_disk_address(index, 1);
1883 }
1884 }
1885
1886 raise_interrupt(index);
1887 }
1888 break;
1889
1890 case 0x30: // write with retries
1891 case 0x31: // write without retries
1892 if (SEL_COMMAND(index).command_cycle == 0)
1893 {
1894
1895 // this is our first time through
1896 if (SEL_DISK(index)->cdrom() || SEL_DISK(index)->ro())
1897 {
1898 printf("%%IDE-W-RO: Write attempt to read-only disk %d.%d.\n", index,
1899 CONTROLLER(index).selected);
1900 command_aborted(index, SEL_COMMAND(index).current_command);
1901 }
1902 else
1903 {
1904 SEL_STATUS(index).drq = true;
1905 SEL_STATUS(index).busy = false;
1906 CONTROLLER(index).data_size = 256;
1907 if (SEL_REGISTERS(index).sector_count == 0)
1908 SEL_REGISTERS(index).sector_count = 256;
1909 }
1910 }
1911 else
1912 {
1913
1914 // now we should be getting data.
1915 if (!SEL_STATUS(index).drq)
1916 {
1917
1918 // the buffer is full. Do something with the data.
1919 {
1920 u32 lba = get_disk_lba(index);
1921
1922#if defined(ES40_BIG_ENDIAN)
1923 {
1924 u16 data[IDE_BUFFER_SIZE];
1925
1926 for (int i = 0; i < SEL_DISK(index)->get_block_size() / sizeof(u16);
1927 i++)
1928 data[i] = endian_16(CONTROLLER(index).data[i]);
1929 if (!transfer_sectors(index, lba, data, 1, true))
1930 break;
1931 }
1932
1933#else
1934 if (!transfer_sectors(index, lba, CONTROLLER(index).data, 1, true))
1935 break;
1936#endif
1937 SEL_STATUS(index).busy = false;
1938 SEL_STATUS(index).drive_ready = true;
1939 SEL_STATUS(index).fault = false;
1940 SEL_STATUS(index).drq = true;
1941 SEL_STATUS(index).err = false;
1942 CONTROLLER(index).data_ptr = 0;
1943
1944 // prepare for next sector
1945 SEL_REGISTERS(index).sector_count--;
1946 if (SEL_REGISTERS(index).sector_count == 0)
1947 {
1948
1949 // we're done
1950 SEL_STATUS(index).drq = false;
1951 SEL_COMMAND(index).command_in_progress = false;
1952 }
1953 else
1954 {
1955
1956 // set the next block to read.
1957 advance_disk_address(index, 1);
1958 }
1959 }
1960
1961 raise_interrupt(index);
1962 }
1963 }
1964 break;
1965
1966 case 0x40: // read verify sectors with retries
1967 case 0x41: // read verify sectors without retries
1968 if (SEL_DISK(index)->cdrom())
1969 {
1970 command_aborted(index, SEL_COMMAND(index).current_command);
1971 }
1972 else
1973 {
1974 if (SEL_REGISTERS(index).sector_count == 0)
1975 SEL_REGISTERS(index).sector_count = 256;
1976
1977 u32 lba = get_disk_lba(index);
1978 int sectors = SEL_REGISTERS(index).sector_count;
1979 if ((off_t_large)lba + sectors > SEL_DISK(index)->get_lba_size())
1980 {
1981 command_aborted(index, SEL_COMMAND(index).current_command);
1982 }
1983 else
1984 {
1985 advance_disk_address(index, sectors);
1986 SEL_REGISTERS(index).sector_count = 0;
1987 SEL_STATUS(index).busy = false;
1988 SEL_STATUS(index).drive_ready = true;
1989 SEL_STATUS(index).seek_complete = true;
1990 SEL_STATUS(index).fault = false;
1991 SEL_STATUS(index).drq = false;
1992 SEL_STATUS(index).err = false;
1993 SEL_COMMAND(index).command_in_progress = false;
1994 raise_interrupt(index);
1995 }
1996 }
1997 break;
1998
1999 case 0x70: // seek
2000 if (SEL_DISK(index)->cdrom())
2001 {
2002 command_aborted(index, SEL_COMMAND(index).current_command);
2003 }
2004 else
2005 {
2006 SEL_STATUS(index).busy = false;
2007 SEL_STATUS(index).drive_ready = true;
2008 SEL_STATUS(index).seek_complete = true;
2009 SEL_STATUS(index).fault = false;
2010 SEL_STATUS(index).drq = false;
2011 SEL_STATUS(index).err = false;
2012 SEL_COMMAND(index).command_in_progress = false;
2013 raise_interrupt(index);
2014 }
2015 break;
2016
2017 /*
2018 * 0x90: execute device diagnostic: mandatory
2019 */
2020 case 0x91: // initialize device parameters
2021 SEL_COMMAND(index).command_in_progress = false;
2022 if (SEL_DISK(index)->cdrom())
2023 {
2024 command_aborted(index, SEL_COMMAND(index).current_command);
2025 }
2026 else
2027 {
2028#ifdef DEBUG_IDE
2029 printf("Original c: %d, h: %d, s: %d\n",
2030 SEL_DISK(index)->get_cylinders(), SEL_DISK(index)->get_heads(),
2031 SEL_DISK(index)->get_sectors());
2032 printf("Requested c: %d, h: %d, s: %d\n",
2033 SEL_REGISTERS(index).cylinder_no, SEL_REGISTERS(index).head_no + 1,
2034 SEL_REGISTERS(index).sector_count);
2035#endif
2036 if (SEL_DISK(index)->get_heads() == (SEL_REGISTERS(index).head_no + 1)
2037 && SEL_DISK(index)->get_sectors() == SEL_REGISTERS(index).sector_count)
2038 {
2039
2040 // use the default translation -- ok!
2041 SEL_STATUS(index).busy = false;
2042 SEL_STATUS(index).drive_ready = true;
2043 SEL_STATUS(index).fault = false;
2044 SEL_STATUS(index).drq = false;
2045 SEL_STATUS(index).err = false;
2046 raise_interrupt(index);
2047 }
2048 else
2049 {
2050#ifdef DEBUG_IDE
2051 PAUSE("INIT DEV PARAMS -- geometry not supported!");
2052#endif
2053 SEL_STATUS(index).busy = false;
2054 SEL_STATUS(index).drive_ready = true;
2055 SEL_STATUS(index).fault = false;
2056 SEL_STATUS(index).drq = false;
2057 SEL_STATUS(index).err = true;
2058 SEL_REGISTERS(index).error = 0x04; // ABORT.
2059 raise_interrupt(index);
2060 }
2061 }
2062 break;
2063
2064 case 0xa0: // packet send
2065 /*
2066 * The state machine and protocol used here was actually
2067 * derived from ATA/ATAPI-5 (D1321R3) instead of the -4
2068 * documenation. State names were taken from that document.
2069 */
2070 if (!SEL_DISK(index)->cdrom())
2071 {
2072 command_aborted(index, SEL_COMMAND(index).current_command);
2073 }
2074 else
2075 {
2076 if (SEL_REGISTERS(index).features & 0x02)
2077 {
2078
2079 // overlap not supported
2080 PAUSE("overlapping not supported");
2081 command_aborted(index, SEL_COMMAND(index).current_command);
2082 }
2083 else
2084 {
2085 if (SEL_COMMAND(index).packet_phase == PACKET_NONE)
2086 {
2087
2088 // this must be the first time through.
2089 if (!scsi_arbitrate(index))
2090 FAILURE(IllegalState, "ATAPI SCSI bus busy");
2091 if (!scsi_select(index, CONTROLLER(index).selected))
2092 FAILURE(IllegalState, "ATAPI device not responding to selection");
2093 SEL_REGISTERS(index).REASON = IR_CD;
2094 SEL_STATUS(index).busy = false;
2095 SEL_STATUS(index).drq = true;
2096 SEL_STATUS(index).DMRD = false;
2097 SEL_STATUS(index).SERV = false;
2098 CONTROLLER(index).data_ptr = 0;
2099 CONTROLLER(index).data_size = 6;
2100 SEL_COMMAND(index).packet_sense = 0;
2101 SEL_COMMAND(index).packet_asc = 0;
2102 SEL_COMMAND(index).packet_ascq = 0;
2103 SEL_COMMAND(index).packet_dma = false;
2104 SEL_COMMAND(index).packet_phase = PACKET_DP1;
2105
2106 // we drop out of the thread and shut down the
2107 // controller. we will be awakened when the drq flag
2108 // is false, and the process will start in the state machine.
2109 break;
2110 }
2111
2112 /*
2113 * This is the Packet I/O state machine. The gist of
2114 * it is this: we loop until yield==true, so we can
2115 * move from state to state in the same DoClock().
2116 * By the time we get here, we're in DP1 (Receive
2117 * Packet) and we're waiting for an actual packet to
2118 * arrive.
2119 */
2120#ifdef DEBUG_IDE_PACKET
2121 printf("Entering Packet state machine %d\n", index);
2122 ide_status(index);
2123#endif
2124
2125 bool yield = false;
2126 do
2127 {
2128#ifdef DEBUG_IDE_PACKET
2129 printf("PACKET STATE: %s (%d)\n",
2130 packet_states[SEL_COMMAND(index).packet_phase],
2131 SEL_COMMAND(index).packet_phase);
2132#endif
2133 switch (SEL_COMMAND(index).packet_phase)
2134 {
2135 case PACKET_DP1: // receive packet
2136 // we now have a full command packet.
2137 if (scsi_get_phase(index) != SCSI_PHASE_COMMAND)
2138 FAILURE(IllegalState, "SCSI command phase expected");
2139
2140 memcpy(scsi_xfer_ptr(index, 12), CONTROLLER(index).data, 12);
2141 memcpy(SEL_COMMAND(index).packet_command, CONTROLLER(index).data,
2142 12);
2143 scsi_xfer_done(index);
2144
2145 SEL_COMMAND(index).packet_phase = PACKET_DP2;
2146 SEL_COMMAND(index).packet_buffersize = SEL_REGISTERS(index).cylinder_no;
2147 SEL_STATUS(index).busy = true;
2148 break;
2149
2150 case PACKET_DP2: // prepare b
2151 SEL_STATUS(index).busy = true;
2152 SEL_STATUS(index).drq = false;
2153
2154 if (SEL_COMMAND(index).command_in_progress)
2155 {
2156 switch (scsi_get_phase(index))
2157 {
2158 case SCSI_PHASE_DATA_IN:
2159 {
2160 size_t num_bytes = scsi_expected_xfer(index);
2161 // ATAPI BYTE_COUNT register is 16 bits and *some* drivers
2162 // don't honor the spec's BYTE_COUNT==0 means 65536 convention.
2163 // Cap each chunk at 65534 (and to host-provided
2164 // packet_buffersize if smaller, even-aligned for 16-bit PIO).
2165 // If the SCSI engine has more data, scsi_xfer_done leaves us
2166 // in the DATA_IN phase and DP34 below loops back to DP2 for the

2868 bool direction)
2869{

2871 size_t xfersize = 0;
2872 u8 status = 0;
2873 u8 count = 0;
2874 u32 prd;
2875 // A fault can be injected while the guest has not yet started DMA.
2876 // Check between bounded waits so the command completes with an error.
2877 while (!semBusMaster[index]->tryWait(20)) {
2878 if (abort_if_offline(index))
2879 return 3;
2880 }
2881 if (abort_if_offline(index))
2882 return 3;

2884 SCOPED_READ_LOCK(mtBusMaster[index]);
2885 prd = endian_32(*(u32*)(&CONTROLLER(index).busmaster[4]));
2886 }

2888 {
2889 if (abort_if_offline(index))
2890 return 3;
2891 u32 base;
2892 do_pci_read(prd, &base, 4, 1);
2893
2894 u16 size_16;
2895 do_pci_read(prd + 4, &size_16, 2, 1);
2896
2897 size_t size = size_16 ? size_16 : 65536;
2898 do_pci_read(prd + 7, &xfer, 1, 1);
2899
2900#ifdef DEBUG_IDE_DMA
2901 printf("-IDE-I-DMA: Transfer %d bytes to/from %lx (%x)\n", size, base, xfer);
2902#endif
2903 if (xfersize + size > buffersize)
2904 {
2905
2906 // only copy as much data as we have from the disk.
2907 size = buffersize - xfersize;
2908 status = 2;
2909#ifdef DEBUG_IDE_DMA
2910 printf("-IDE-I-DMA: Actual transfer size: %d bytes\n", size);
2911#endif
2912 }
2913
2914 // copy it to/from ram.
2915 if (!direction)
2916 {
2917 do_pci_write(base, buffer, 1, size);
2918 buffer += size;
2919 }
2920 else
2921 {
2922 do_pci_read(base, buffer, 1, size);
2923 buffer += size;
2924 }
2925
2926 xfersize += size;
2927 prd += 8; // go to next entry.
2928 if (xfer == 0x80 && xfersize < buffersize)
2929 {
2930
2931 // we still have disk data left over!
2932 status = 1;
2933 }
2934
2935 if (count++ > 32)
2936 {
2937 FAILURE(InvalidArgument, "Too many PRD nodes?");
2938 }
2939
2940 if (buffersize == xfersize && xfer != 0x80)
2941 {
2942
2943 // we're done, but there's more prd nodes.
2944 status = 2;
2945 }
2946 } while (xfer != 0x80 && status == 0);
2947
2948 switch (status)
2949 {
2950 case 0: // normal completion.
2951 {
2952 SCOPED_WRITE_LOCK(mtBusMaster[index]);
2953 CONTROLLER(index).busmaster[2] &= 0xfe; // clear active.
2954 }
2955
2956 raise_interrupt(index);
2957 break;
2958
2959 case 1: // PRD is smaller than the data we have.
2960 {
2961 SCOPED_WRITE_LOCK(mtBusMaster[index]);
2962 CONTROLLER(index).busmaster[2] &= 0xfe; // clear active.
2963 }
2964
2965 // do not raise an interrupt
2966 break;
2967
2968 case 2: // PRD is larger than the data we have.
2969 // leave active set.
2970 raise_interrupt(index);
2971 break;
2972 }
2973
2974 return status;
2975}
