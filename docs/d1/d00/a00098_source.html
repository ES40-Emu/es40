93 struct SDMA_result
94 {
95 size_t transferred; // Bytes moved; zero for verify or standalone EOP.
96 bool blocked; // No DMA service; registers and buffers are unchanged.
97 bool terminal_count; // This call exhausted the count, even with auto-init.
98 bool external_eop; // This call honored the device's EOP indication.
99 };
100
101 // Buffers and lengths are in bytes.
102 // length 0 uses the current count.
103 // Transfers on channels 5-7 must cover whole words.
104 // send: device to memory; recv: memory to device.
105 // Calls remain device-paced and do not require an explicit set_drq().
106 // A software request or retained block bypasses only that channel's mask;
107 // incoming hardware DRQ does not. Enable and cascade requirements still apply.
108 // Channels 0-3 also require controller 1 enabled and channel 4 unmasked
109 // and programmed for cascade. Channel 4's address/count are not serviced.
110 // Cascade mode, a wrong direction or an illegal transfer type is blocked.
111 // Verify services the count without accessing memory or the buffer:
112 // blocked is false, transferred is zero, and terminal_count reports exhaustion.
113 // eop ends the transfer after the last unit serviced by this call, including
114 // when length is capped by the current count. It is not a latched pin level.
115 // Blocked calls do not apply EOP, including calls on cascade-mode channels.
116 // EOP sets completion status and masks or auto-initializes the channel;
117 // otherwise the remaining address/count are retained. DRQ is unchanged.
118 // terminal_count and external_eop can both be true; completion occurs once.
119 // Results do not clear the guest-visible terminal-count status.
120 SDMA_result send_data(int channel, void* data, size_t length = 0,
121 bool eop = false);
122 SDMA_result recv_data(int channel, void* data, size_t length = 0,
123 bool eop = false);
124 // One byte on channels 0-3, one little-endian word on channels 5-7.
125 // Byte sends use the low 8 bits; byte receives are zero-extended.
126 // A blocked or verify receive leaves data unchanged.
127 SDMA_result send_unit(int channel, u16 data, bool eop = false);
128 SDMA_result recv_unit(int channel, u16& data, bool eop = false);
129 // Explicit-request service, at most one byte/word per call.
130 // Demand requires unmasked hardware DRQ; single also accepts a software request.
131 // Block starts from either request and, after successful service, retains it
132 // until TC/EOP even if DRQ or the software request drops. Auto-init releases it.
133 // Between calls, own mask-set/disable/reset or channel reprogramming drops
134 // retention without generating TC/EOP. Pending requests can start a new service.
135 // A held DRQ can service successive calls; these calls never lower DRQ.
136 // Cascade returns blocked. Existing enable/cascade and completion rules apply.
137 // Blocked or verify receives leave data unchanged; blocked calls ignore EOP.
138 // Command bit 4 selects rotating priority; otherwise local order is 0-3.
139 // Rotation makes the released channel lowest priority, including channel 4
140 // for lower-controller service. Ongoing demand/block service is not preempted.
141 // Demand releases on DRQ withdrawal; single re-arbitrates for every unit.
142 // Failed calls and idle register writes do not rotate. Bus timing is not modeled.
143 SDMA_result service_send_unit(int channel, u16 data, bool eop = false);
144 SDMA_result service_recv_unit(int channel, u16& data, bool eop = false);
145 // End an accepted demand/block service without consuming another unit.
146 SDMA_result service_eop(int channel);
147 // Device queries use global channels 0-3 or 5-7; invalid channels throw.
148 // Raw programmed 8237 mode byte, including the local channel-select bits.
149 u8 get_mode(int channel);
150 // Readiness for service_send_unit/service_recv_unit, including explicit
151 // requests and arbitration.
152 bool can_service_send(int channel);
153 bool can_service_recv(int channel);
154 // Raw byte/word count register (number of DMA units minus one).
155 int get_count(int channel);
156 // Current count plus one in bytes; independent of mask/enable state.
157 size_t get_transfer_size(int channel);
