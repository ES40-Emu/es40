187static string trim_answer(const string& value)
188{
189 size_t first = value.find_first_not_of(" \t\r\n");
190 if (first == string::npos)
191 return "";
192 size_t last = value.find_last_not_of(" \t\r\n");
193 return value.substr(first, last - first + 1);
194}

227void add_disks(ShrinkingChoiceQuestion* disk_q, ostream* os, bool floppy = false)
228{
229 /* Loop until there are no more disks to be added.
230 */
231 for (;;)
232 {
233 /* If disk_q has only one answer left, this is the
234 * "none" answer. This controller can contain no
235 * more disks.
236 */
237 if (disk_q->countAnswers() == 1)
238 break;
239
240 /* If the answer value is "", the answer "none"
241 * was given. Stop adding disks to this controller.
242 */
243 if (disk_q->ask() == "")
244 break;
245
246 /* Find out if this should be using:
247 * - a disk image file
248 * - a raw device
249 * - a RAM DISK
250 */
251 MultipleChoiceQuestion type_q;
252 type_q.setQuestion("How should " + disk_q->getAnswer() + " be emulated?");
253 type_q.setDefault("file");
254 type_q.setExplanation("Disks can be emulated in several ways.");
255 type_q.addAnswer("file", "file", "The disk uses a disk image on the host system's disk.");
256 type_q.addAnswer("device", "device", "The disk uses one of the host system's raw disks.");
257 type_q.addAnswer("ramdisk", "ramdisk", "The disk stores it's data in RAM. Volatile.");
258
259 *os << " " << disk_q->getAnswer() << " = " << type_q.ask() << "\n";
260 *os << " {\n";
261
262 MultipleChoiceQuestion cdrom_q;
263 if (type_q.getAnswer() == "file" || type_q.getAnswer() == "device")
264 {
265 /* For a file or device, we need to know what
266 * file or device to use.
267 */
268 FreeTextQuestion img_q;
269 img_q.setQuestion("What " + type_q.getAnswer() + " should " + disk_q->getAnswer() + " use?");
270 img_q.setExplanation("Enter the path to the " + type_q.getAnswer() + " to use for this disk.");
271 *os << " " << type_q.getAnswer() << " = \"" << img_q.ask() << "\";\n";
272
273 /* We need to know whether to emulate this as
274 * a cd-rom, or as a hard-disk.
275 */
276 cdrom_q.setQuestion("Should " + disk_q->getAnswer() + " be a disk or a cd-rom device?");
277 cdrom_q.setExplanation("Do you want the OS to see this " + type_q.getAnswer() + " as a hard-disk, or as a cd-rom?");
278 cdrom_q.addAnswer("disk", "false", "Hard-disk");
279 cdrom_q.addAnswer("cd-rom", "true", "CD-ROM drive");
280 cdrom_q.setDefault("disk");
281 if (floppy)
282 {
283 cdrom_q.setAnswer("false");
284 *os << " cdrom = " << "false" << ";\n";
285 }
286 else
287 *os << " cdrom = " << cdrom_q.ask() << ";\n";
288 }
289
290 std::string disk_image_format = "auto";
291 if (type_q.getAnswer() == "file" && cdrom_q.getAnswer() != "true" && !floppy)
292 {
293 MultipleChoiceQuestion format_q;
294 format_q.setQuestion("What format is this disk image?");
295 format_q.setExplanation("Detect an existing image from its header, or choose an explicit format.");
296 format_q.addAnswer("auto", "auto", "Detect from the image header; unrecognized files are raw (the default)");
297 format_q.addAnswer("raw", "raw", "Raw disk image");
298 format_q.addAnswer("qcow2", "qcow2", "QCOW2 disk image");
299 format_q.setDefault("auto");
300 format_q.explain();
301 disk_image_format = format_q.ask();
302 *os << " format = \"" << disk_image_format << "\";\n";
303 }
304
305 if (type_q.getAnswer() == "file" && cdrom_q.getAnswer() != "true")
306 {
307 /* For a file, we need to know whether to create
308 * it when it doesn't exist or not.
309 */
310 MultipleChoiceQuestion create_q;
311 create_q.setQuestion("If the file doesn't exist, do you want us to create it?");
312 create_q.setExplanation("The file will be created the first time the emulator runs.");
313 create_q.addAnswer("no", "no", "Don't create this file");
314 create_q.addAnswer("yes", "yes", "Create this file");
315 create_q.setDefault("yes");
316 create_q.ask();
317 if (create_q.getAnswer() == "yes")
318 {
319 if (!floppy)
320 {
321 MultipleChoiceQuestion create_format_q;
322 create_format_q.setQuestion("What format should a new disk image use?");
323 create_format_q.setExplanation("This applies only when the file is missing. It must match an explicit image format override; existing files are unchanged.");
324 create_format_q.addAnswer("raw", "raw", "Raw disk image");
325 create_format_q.addAnswer("qcow2", "qcow2", "QCOW2 disk image");
326 create_format_q.setDefault(disk_image_format == "auto" ? "raw" : disk_image_format);
327 create_format_q.explain();
328 *os << " autocreate_format = \"" << create_format_q.ask() << "\";\n";
329 }
330 /* If we should create the file, we need to
331 * know it's size.
332 */
333 MultipleChoiceQuestion unit_q;
334 unit_q.setQuestion("What unit do you want to use to specify the disk size?");
335 unit_q.setExplanation("This is needed to create the file if it doesn't exist.");
336 unit_q.addAnswer("KB", "K", "Kilobytes");
337 unit_q.addAnswer("MB", "M", "Megabytes");
338 unit_q.addAnswer("GB", "G", "Gigabytes");
339 unit_q.setDefault("MB");
340 unit_q.ask();
341 NumberQuestion size_q;
342 size_q.setQuestion("How many " + unit_q.getAnswer() + "Bytes should the disk be?");
343 size_q.setExplanation("This is needed to create the file if it doesn't exist.");
344 size_q.setRange(1, 1024);
345 size_q.setDefault("10");
346 size_q.ask();
347 *os << " autocreate_size = \"" << size_q.getAnswer() << unit_q.getAnswer() << "\";\n";
348 }
349 }
350
351 if (type_q.getAnswer() == "ramdisk")
352 {
353 cdrom_q.setQuestion("Should " + disk_q->getAnswer() + " be a disk or a read-only in-memory cd-rom device?");
354 cdrom_q.setExplanation("Do you want the OS to see this " + type_q.getAnswer() + " as a hard-disk, or as a read-only in-memory cd-rom?");
355 cdrom_q.addAnswer("disk", "false", "Hard-disk");
356 cdrom_q.addAnswer("cd-rom", "true", "CD-ROM drive");
357 cdrom_q.setDefault("disk");
358 if (floppy)
359 {
360 cdrom_q.setAnswer("false");
361 *os << " cdrom = " << "false" << ";\n";
362 }
363 else
364 *os << " cdrom = " << cdrom_q.ask() << ";\n";
365 if (cdrom_q.getAnswer() == "cd-rom" || cdrom_q.getAnswer() == "true")
366 {
367 FreeTextQuestion img_q;
368 img_q.setQuestion("What file should " + disk_q->getAnswer() + " use?");
369 img_q.setExplanation("Enter the path to the file to use for this disk.");
370 *os << " " << "file = \"" << img_q.ask() << "\";\n";
371 }
372 else
373 {
374 /* For a RAM DISK, we need to know what
375 * size it should be.
376 */
377 MultipleChoiceQuestion unit_q;
378 unit_q.setQuestion("What unit do you want to use to specify the disk size?");
379 unit_q.setExplanation("This is needed to create the RAMDISK.");
380 unit_q.addAnswer("KB", "K", "Kilobytes");
381 unit_q.addAnswer("MB", "M", "Megabytes");
382 unit_q.addAnswer("GB", "G", "Gigabytes");
383 unit_q.setDefault("MB");
384 unit_q.ask();
385 NumberQuestion size_q;
386 size_q.setQuestion("How many " + unit_q.getAnswer() + "Bytes should the disk be?");
387 size_q.setExplanation("This is needed to create the RAMDISK.");
388 size_q.setRange(1, 1024);
389 size_q.setDefault("10");
390 size_q.ask();
391 *os << " size = \"" << size_q.getAnswer() << unit_q.getAnswer() << "\";\n";
392 }
393 }
394
395 /* We also need to know whether this is a
396 * writeable or a read-only device.
397 */
398 MultipleChoiceQuestion ro_q;
399 ro_q.setQuestion("Should " + disk_q->getAnswer() + " be a read-only disk?");
400 ro_q.setExplanation("You might want to write-protect this disk.");
401 ro_q.addAnswer("no", "false", "writeable");
402 ro_q.addAnswer("yes", "true", "read-only");
403 ro_q.setDefault("no");
404
405 if (cdrom_q.getAnswer() == "true")
406 {
407 /* CD-ROMs are always read-only.
408 */
409 ro_q.setAnswer("true");
410 }
411 else if (type_q.getAnswer() == "ramdisk")
412 {
413 /* Read-only RAM DISKs don't make any sense.
414 */
415 ro_q.setAnswer("false");
416 }
417 else
418 {
419 /* Otherwise, ask.
420 */
421 ro_q.ask();
422 }
423 *os << " read_only = " << ro_q.getAnswer() << ";\n";
424
425 MultipleChoiceQuestion offline_q;
426 offline_q.setQuestion("Should " + disk_q->getAnswer() + " start offline (failed)?");
427 offline_q.setExplanation("An offline drive remains attached and identifiable, but reports not ready and rejects disk I/O.");
428 offline_q.addAnswer("no", "false", "Normal disk operation");
429 offline_q.addAnswer("yes", "true", "Inject a disk failure");
430 offline_q.setDefault("no");
431 if (offline_q.ask() == "true")
432 *os << " offline = true;\n";
433
434 /* The user can define a custom model
435 * number for the device.
436 */
437 FreeTextQuestion ft_q;
438 ft_q.setQuestion("Would you like to set a disk model number?");
439 ft_q.setExplanation("Leave blank to choose the default.");
440 if (ft_q.ask() != "")
441 *os << " model_number = \"" << ft_q.getAnswer() << "\";\n";
442
443 /* The user can define a custom revision
444 * number for the device.
445 */
446 ft_q.setQuestion("Would you like to set a revision number?");
447 if (ft_q.ask() != "")
448 *os << " rev_number = \"" << ft_q.getAnswer() << "\";\n";
449
450 /* The user can define a custom serial
451 * number for the device.
452 */
453 ft_q.setQuestion("Would you like to set a serial number?");
454 if (ft_q.ask() != "")
455 *os << " serial_number = \"" << ft_q.getAnswer() << "\";\n";
456
457 *os << " }\n\n";
458 }
459}
