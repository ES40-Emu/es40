127// WMB: HRM 4.11.1 requires WMB to order all prior stores from this CPU
128// before all subsequent stores, as observed by every other CPU/DMA agent.
129// A standalone atomic_thread_fence(release) only synchronizes with a matching
130// acquire fence/load on another thread; it does NOT impose store-store
131// ordering visible to unsynchronized observers.
132#define DO_WMB std::atomic_thread_fence(std::memory_order_seq_cst);
