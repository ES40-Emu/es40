94// Definition of the per-thread diagnostic-stall accumulator declared in diag_rpcc.h.
95thread_local int64_t g_diag_excluded_ns = 0;
96
97// printf() that times its console-I/O stall into g_diag_excluded_ns so jit_run keeps it out of the
98// guest RPCC -- the PCI decode-off diagnostics below can spam during VGA init and jump the counter.
99static void diag_printf(const char* fmt, ...)
100{
101 const auto t0 = std::chrono::steady_clock::now();
102 va_list ap; va_start(ap, fmt); vprintf(fmt, ap); va_end(ap);
103 g_diag_excluded_ns += (int64_t) std::chrono::duration_cast<std::chrono::nanoseconds>(
104 std::chrono::steady_clock::now() - t0).count();
105}

112static bool pci_bar_is_upper_64bit(const u32* config, int bar)
113{
114 // Walk BAR boundaries: an upper address dword has no type attributes.
115 for (int low = 0; low < bar; ++low)
116 {
117 const u32 attributes = endian_32(config[4 + low]);
118 if ((attributes & 1U) == 0 &&
119 (attributes & PCI_MEM_TYPE_MASK) == PCI_MEM_TYPE_64)
120 {
121 if (low + 1 == bar)
122 return true;
123 ++low;
124 }
125 }
126 return false;
127}

129static size_t pci_dma_chunk_limit(u64 phys_addr, size_t remaining)
130{
131 const size_t dma_page = 8192;
132 size_t page_remaining = dma_page - (size_t)(phys_addr & (dma_page - 1));
133 return (remaining < page_remaining) ? remaining : page_remaining;
134}

196{

217void CPCIDevice::config_write(int func, u32 address, int dsize, u32 data)
218{
219
220 // printf("%s(%s).%d config write %d bytes @ %x = %x\n",myCfg->get_myName(), myCfg->get_myValue(), func,dsize/8,address, data);
221 u8* x;
222 u8* y;
223
224 u32 mask = 0;
225 u32 old_data = 0;
226 u32 new_data = 0;
227
228 x = (u8*)pci_state.config_data[func];
229 x += address;
230 y = (u8*)pci_state.config_mask[func];
231 y += address;
232
233#if defined(DEBUG_PCI)
234 if (address == 0x3c && (data & 0xff) != 0xff)
235 printf("%s.%d PCI Interrupt set to %02x.\n", devid_string, func, data & 0xff);
236#endif
237 switch (dsize)
238 {
239 case 8:
240 data = endian_8(data);
241 old_data = (*x) & 0xff;
242 mask = (*y) & 0xff;
243 new_data = (old_data & ~mask) | (data & mask);
244 *x = (u8)new_data;
245 break;
246
247 case 16:
248 data = endian_16(data);
249 old_data = (*((u16*)x)) & 0xffff;
250 mask = (*((u16*)y)) & 0xffff;
251 new_data = (old_data & ~mask) | (data & mask);
252 *((u16*)x) = (u16)new_data;
253 break;
254
255 case 32:
256 data = endian_32(data);
257 old_data = (*((u32*)x));
258 mask = (*((u32*)y));
259 new_data = (old_data & ~mask) | (data & mask);
260 *((u32*)x) = new_data;
261 break;
262 }
263
264 bool upper_64bit_bar = false;
265 if (dsize == 32 && address >= 0x14 && address <= 0x24)
266 {
267 const int bar = (address - 0x10) / 4;
268 upper_64bit_bar = pci_bar_is_upper_64bit(std_config_data[func], bar);
269 }
270 const u32 access_bytes = (u32)dsize / 8U;
271 const bool rom_bar_write = address < 0x34U &&
272 address + access_bytes > 0x30U;
273 const u32 write_ones = dsize == 8 ? 0xFFU :
274 (dsize == 16 ? 0xFFFFU : 0xFFFFFFFFU);
275
276 /* Preserve the existing ROM sizing-probe handling. */
277 if (rom_bar_write && data != write_ones)
278 {
279 const u32 rom_data = endian_32(pci_state.config_data[func][0x30 / 4]);
280 const u32 rom_mask = endian_32(pci_state.config_mask[func][0x30 / 4]);
281 if (rom_mask != 0)
282 register_bar(func, 6, rom_data, rom_mask);
283 }
284 else if ((dsize == 8 || (dsize == 16 && (address & 1U) == 0)) &&
285 address >= 0x10U && address + access_bytes <= 0x28U && mask != 0)
286 {
287 // Byte enables change the complete 32-bit BAR's decoded address.
288 // Skip both halves of 64-bit BARs; their partial-write handling is separate.
289 for (int bar = 0; bar < 6; ++bar)
290 {
291 const u32 attributes = endian_32(std_config_data[func][4 + bar]);
292 if ((attributes & 1U) == 0 &&
293 (attributes & PCI_MEM_TYPE_MASK) == PCI_MEM_TYPE_64)
294 {
295 ++bar;
296 continue;
297 }
298 if ((address - 0x10U) / 4U == (u32)bar)
299 {
300 // An all-ones byte/word is still an address write, not a dword probe.
301 register_bar(func, bar, endian_32(pci_state.config_data[func][4 + bar]),
302 endian_32(pci_state.config_mask[func][4 + bar]));
303 break;
304 }
305 }
306 }
307 else if (dsize == 32 && upper_64bit_bar && ((data & mask) != mask))
308 {
309 register_bar(func, (address - 0x10) / 4,
310 endian_32(new_data), endian_32(mask));
311 }
312 // A masked maximum address is a valid assignment (including on restore).
313 // Only the unmasked all-ones dword retains the existing size-probe policy.
314 else if (dsize == 32 && mask != 0 && data != write_ones)
315 {
316 switch (address)
317 {
318 case 0x10:
319 case 0x14:
320 case 0x18:
321 case 0x1c:
322 case 0x20:
323 case 0x24:
324 register_bar(func, (address - 0x10) / 4, endian_32(new_data),
325 endian_32(mask));
326 break;
327 }
328 }
329
330 config_write_custom(func, address, dsize, old_data, new_data, data);
331}

333void CPCIDevice::register_bar(int func, int bar, u32 data, u32 mask)
334{
335 /* A 64-bit BAR's upper dword belongs to the preceding BAR. */
336 if (bar > 0 && bar < 6 && pci_bar_is_upper_64bit(std_config_data[func], bar))
337 {
338 const u32 low =
339 endian_32(pci_state.config_data[func][4 + bar - 1]);
340 register_bar(func, bar - 1, low,
341 endian_32(pci_state.config_mask[func][4 + bar - 1]));
342 return;
343 }
344
345 int id = PCI_RANGE_BASE + (func * 8) + bar;
346
347 // IO BAR here, never BAR 6 though
348 if ((data & 1u) && bar != 6) {
349 pci_range_is_io[func][bar] = true;
350
351 u32 length = (~(mask & PCI_IO_ADDRESS_MASK)) + 1u; // Size probe from mask
352 if (length < 4u) length = 4u;
353
354 u32 base = (data & PCI_IO_ADDRESS_MASK) & ~(length - 1u); // IO BAR alignment
355 if (base == 0)
356 {
357 cSystem->RegisterMemory(this, id, 0, 0);
358 return;
359 }
360
361 const u64 t = U64(0x00000801fc000000) + (U64(0x0000000200000000) * myPCIBus) + base;
362
363 cSystem->RegisterMemory(this, id, t, length);
364 printf("%s(%s).%d PCI BAR %d set to IO %" PRIx64 ", len %x.\n",
365 myCfg->get_myName(), myCfg->get_myValue(), func, bar, t, length);
366 return;
367 }
368
369 // Everything else is memory BAR ......
370 pci_range_is_io[func][bar] = false;
371
372 // PCI Option ROM BAR, bit 0 = enable, address 31:11, 10:1 reserve by spec
373 if (bar == 6) {
374 const bool rom_enable = (data & PCI_ROM_ADDRESS_ENABLE) != 0;
375
376 if (!rom_enable) {
377 cSystem->RegisterMemory(this, id, 0, 0);
378 printf("%s(%s).%d PCI BAR 6 ROM disabled.\n", myCfg->get_myName(), myCfg->get_myValue(), func);
379 return;
380 }
381
382 // Size from probe mask; ignore enable & reserved bits via ROM mask
383 u32 length = (~(mask & PCI_ROM_ADDRESS_MASK)) + 1u;
384 if (length < 0x800u) length = 0x800u; // spec minimum 2 KiB
385
386 // Base: drop enable/reserved via mask, then align to size
387 u32 base = (data & PCI_ROM_ADDRESS_MASK) & ~(length - 1u);
388
389 const u64 t = U64(0x0000080000000000) + (U64(0x0000000200000000) * myPCIBus) + base;
390
391 cSystem->RegisterMemory(this, id, t, length);
392 printf("%s(%s).%d PCI BAR 6 set to MEM %" PRIx64 " (ROM), len %x.\n",
393 myCfg->get_myName(), myCfg->get_myValue(), func, t, length);
394 return;
395 }
396
397 // Normal memory BAR (0..5 that are memory)
398 {
399 // Size from probe mask; clear attr bits via MEM mask
400 u32 length = (~(mask & PCI_MEM_ADDRESS_MASK)) + 1u;
401 if (length < 0x10u) length = 0x10u; // spec minimum 16 bytes
402
403 const bool is_64bit = bar < 5 &&
404 (data & PCI_MEM_TYPE_MASK) == PCI_MEM_TYPE_64;
405 u32 high = 0;
406 if (is_64bit)
407 {
408 high = endian_32(pci_state.config_data[func][4 + bar + 1]);
409 pci_range_is_io[func][bar + 1] = false;
410
411 /* Ignore all-ones size probes on either half. */
412 if (high == 0xFFFFFFFFU ||
413 (data & PCI_MEM_ADDRESS_MASK) ==
414 (mask & PCI_MEM_ADDRESS_MASK))
415 return;
416 }
417
418 // Base: clear attribute bits, combine the pair, and align to size.
419 u64 base = ((u64)high << 32) | (data & PCI_MEM_ADDRESS_MASK);
420 base &= ~((u64)length - 1U);
421 if (base == 0)
422 {
423 cSystem->RegisterMemory(this, id, 0, 0);
424 return;
425 }
426
427 const u64 t = U64(0x0000080000000000) + (U64(0x0000000200000000) * myPCIBus) + base;
428
429 cSystem->RegisterMemory(this, id, t, length);
430 printf("%s(%s).%d PCI BAR %d set to MEM %" PRIx64 ", len %x.\n",
431 myCfg->get_myName(), myCfg->get_myValue(), func, bar, t, length);
432 }
433}

435void CPCIDevice::ResetPCI()
436{
437 int i;
438
439 for (i = 0; i < 8; i++)
440 {
441 if (device_at[i])
442 {
443 cSystem->RegisterMemory(this, PCI_RANGE_BASE + (i * 8) + 7,
444 U64(0x00000801fe000000) + (U64(0x0000000200000000) * myPCIBus) +
445 (U64(0x0000000000000800) * myPCIDev) +
446 (U64(0x0000000000000100) * i), 0x100);
447 memcpy(pci_state.config_data[i], std_config_data[i], 64 * sizeof(u32));
448 memcpy(pci_state.config_mask[i], std_config_mask[i], 64 * sizeof(u32));
449
450 config_write(i, 0x10, 32, endian_32(pci_state.config_data[i][4]));
451 config_write(i, 0x14, 32, endian_32(pci_state.config_data[i][5]));
452 config_write(i, 0x18, 32, endian_32(pci_state.config_data[i][6]));
453 config_write(i, 0x1c, 32, endian_32(pci_state.config_data[i][7]));
454 config_write(i, 0x20, 32, endian_32(pci_state.config_data[i][8]));
455 config_write(i, 0x24, 32, endian_32(pci_state.config_data[i][9]));
456 config_write(i, 0x30, 32, endian_32(pci_state.config_data[i][12]));
457 }
458 }

461bool CPCIDevice::decodes_memory_access(int index, u64, int, bool) const noexcept
462{
463 if (index < 0)
464 return false;
465 // Legacy registrations can include special cycles so their decode rules can belong to the device.
466 if (index < PCI_RANGE_BASE)
467 return true;
468
469 const int range = index - PCI_RANGE_BASE;
470 if (range >= 8 * 8)
471 return false;
472 const int func = range / 8;
473 const int bar = range & 7;
474 if (!device_at[func])
475 return false;
476 // PCI configuration cycles remain available when I/O and memory are disabled.
477 if (bar == 7)
478 return true;
479
480 const u32 command = endian_32(pci_state.config_data[func][1]);
481 const u32 enable = bar != 6 && pci_range_is_io[func][bar] ? 1U : 2U;
482 if (!(command & enable))
483 return false;
484 // Expansion ROM decoding requires both Memory Space Enable and ROM Enable.
485 return bar != 6 ||
486 (pci_state.config_data[func][0x30 / 4] & endian_32(PCI_ROM_ADDRESS_ENABLE));
487}

489u64 CPCIDevice::ReadMem(int index, u64 address, int dsize)
490{
491 int func;
492 int bar;
493
494 if (dsize == 64)
495 return ReadMem(index, address, 32) | (((u64)ReadMem(index, address + 4, 32)) << 32);
496
497 if (dsize != 8 && dsize != 16 && dsize != 32)
498 {
499 FAILURE_5(InvalidArgument,
500 "ReadMem: %s(%s) Unsupported dsize %d. (%d, %" PRIx64 ")\n",
501 myCfg->get_myName(), myCfg->get_myValue(), dsize, index, address);
502 }
503
504 if (index < PCI_RANGE_BASE)
505 {
506 if (dev_range_is_io[index] && !(pci_state.config_data[0][1] & endian_32(1)))
507 {
508 diag_printf("%s(%s) Legacy IO access with IO disabled from PCI config.\n",
509 myCfg->get_myName(), myCfg->get_myValue());
510 return 0;
511 }
512
513 if (!dev_range_is_io[index] && !(pci_state.config_data[0][1] & endian_32(2)))
514 {
515 diag_printf("%s(%s) Legacy memory access with memory disabled from PCI config.\n",
516 myCfg->get_myName(), myCfg->get_myValue());
517 return 0;
518 }
519
520 // printf("%s(%s) Calling ReadMem_Legacy(%d).\n",myCfg->get_myName(), myCfg->get_myValue(), index);
521 return ReadMem_Legacy(index, (u32)address, dsize);
522 }
523
524 index -= PCI_RANGE_BASE;
525
526 bar = index & 7;
527 func = (index / 8) & 7;
528
529 if (bar == 7)
530 return config_read(func, (u32)address, dsize);
531
532 if (pci_range_is_io[func][bar] && !(pci_state.config_data[func][1] & endian_32(1)))
533 {
534 diag_printf("%s(%s).%d PCI IO access with IO disabled from PCI config.\n",
535 myCfg->get_myName(), myCfg->get_myValue(), func);
536 return 0;
537 }
538
539 if (!pci_range_is_io[func][bar]
540 && !(pci_state.config_data[func][1] & endian_32(2)))
541 {
542 diag_printf("%s(%s).%d PCI memory access with memory disabled from PCI config.\n",
543 myCfg->get_myName(), myCfg->get_myValue(), func);
544 return 0;
545 }
546
547 if (bar == 6)
548 cSystem->PrepareX86OptionROMMemory();
549
550 // printf("%s(%s).%d Calling ReadMem_Bar(%d,%d).\n",myCfg->get_myName(), myCfg->get_myValue(), func,func,bar);
551 return ReadMem_Bar(func, bar, (u32)address, dsize);
552}

622#endif

626bool CPCIDevice::do_pci_interrupt(int func, bool asserted)
627{
628 // Per Tsunami HRM section 6.3, the 64 TIGbus interrupt inputs (DRIR
629 // bits 0-63) are board-wired to PCI INTx pins. ES40 uses the slot-
630 // based pattern SRM programs at first pass:
631 // DRIR_bit = ((slot + 1) * 4 + bus * 0x10 + (pin - 1)) & 0x3f.
632 //
633 // Two gates the cfg-space CFIT longword tells us:
634 // - Pin (cfg+0x3D) selects which INTx (1=A, 2=B, 3=C, 4=D); 0 means
635 // the device declares no INTx capability and we early-return.
636 // - Line (cfg+0x3C) == 0xFF is PCI spec's "no IRQ assigned" — that's
637 // also the reset value, so it doubles as "SRM has not configured
638 // this device's IRQ yet". Honouring it prevents firing INTx into
639 // the OS during the pre-bootstrap window where exception handlers
640 // aren't installed yet (OpenVMS in particular crashes hard with
641 // ACV through vector 0x80 if INTx delivers in that window).
642 //
643 // Once a real value lands in Line (any 8-bit value < 0xFF), we ignore
644 // it and use the slot formula instead
645 const u32 cfit = endian_32(pci_state.config_data[func][0x0f]);
646 const u8 line = cfit & 0xff;
647 const u8 pin = (cfit >> 8) & 0xff;
648 if (pin == 0)
649 return false;
650
651 const int intx = (pin - 1) & 0x3;
652 const int slot = myPCIDev & 0x1f;
653 const int bus_offset = (myPCIBus & 0x3) * 0x10;
654 const int drir_bit = ((slot + 1) * 4 + bus_offset + intx) & 0x3f;
655
656#ifdef DEBUG_PCI_IRQ
657 printf("PCI-IRQ: %s.%d %s line=0x%02x pin=%d slot=%d bus=%d -> DRIR bit %d\n",
658 devid_string, func, asserted ? "ASSERT " : "DEASSERT",
659 line, pin, myPCIDev, myPCIBus, drir_bit);
660#endif
661 cSystem->interrupt(drir_bit, asserted);
662 return true;
663}

671int CPCIDevice::SaveState(FILE* f)
672{
673 long ss = sizeof(pci_state);
674
675 fwrite(&pci_magic1, sizeof(u32), 1, f);
676 fwrite(&ss, sizeof(long), 1, f);
677 fwrite(&pci_state, sizeof(pci_state), 1, f);

686int CPCIDevice::RestoreState(FILE* f)
687{
688 long ss;
689 u32 m1;
690 u32 m2;
691 size_t r;
692 SPCI_state restored_state;
693
694 r = fread(&m1, sizeof(u32), 1, f);
695 if (r != 1)
696 {
697 printf("%s: unexpected end of file!\n", devid_string);
698 return -1;
699 }
700
701 if (m1 != pci_magic1)
702 {
703 printf("%s: PCI MAGIC 1 does not match!\n", devid_string);
704 return -1;
705 }
706
707 r = fread(&ss, sizeof(long), 1, f);
708 if (r != 1)
709 {
710 printf("%s: unexpected end of file!\n", devid_string);
711 return -1;
712 }
713
714 if (ss != sizeof(pci_state))
715 {
716 printf("%s: PCI STRUCT SIZE does not match!\n", devid_string);
717 return -1;
718 }
719
720 r = fread(&restored_state, sizeof(restored_state), 1, f);
721 if (r != 1)
722 {
723 printf("%s: unexpected end of file!\n", devid_string);
724 return -1;
725 }
726
727 r = fread(&m2, sizeof(u32), 1, f);
728 if (r != 1)
729 {
730 printf("%s: unexpected end of file!\n", devid_string);
731 return -1;
732 }
733
734 if (m2 != pci_magic2)
735 {
736 printf("%s: PCI MAGIC 2 does not match!\n", devid_string);
737 return -1;
738 }
739
740 pci_state = restored_state;
741 // Config-space bytes alone do not move the active I/O/MMIO apertures.
742 // Rebuild them through the normal write path, including its 64-bit BAR,
743 for (int func = 0; func < 8; ++func)
744 {
745 if (!device_at[func])
746 continue;
747
748 for (int bar = 0; bar < 7; ++bar)
749 {
750 cSystem->RegisterMemory(this, PCI_RANGE_BASE + func * 8 + bar, 0, 0);
751 pci_range_is_io[func][bar] = false;
752 }
753 for (int bar = 0; bar < 7; ++bar)
754 {
755 const u32 address = bar == 6 ? 0x30U : 0x10U + bar * 4U;
756 if (pci_state.config_mask[func][address / 4] != 0)
757 config_write(func, address, 32,
758 endian_32(pci_state.config_data[func][address / 4]));
759 }
760 }
761
762 printf("%s: %d PCI bytes restored.\n", devid_string, (int)ss);
763 return 0;
764}

797void CPCIDevice::do_pci_read(u32 address, void* dest, size_t element_size,
798 size_t element_count)
799{
800 size_t el;
801 char* dst = (char*)dest;
802
803 if (element_count == 0)
804 return;
805
806 // PCI Command.BME gates all device-initiated DMA.
807 // SRM early probe writes Command=0 to every device
808 if (!(config_read(0, 0x04, 16) & 0x4))
809 {
810 memset(dest, 0, element_size * element_count);
811 return;
812 }
813
814 // get the 64-bit system wide address
815 u64 phys_addr = cSystem->PCI_Phys(myPCIBus, address);
816
817 // if there is only one element to read, this is a simple ReadMem operation.
818 if (element_count == 1)
819 {
820 switch (element_size)
821 {
822 case 1:
823 *(u8*)dest = (u8)cSystem->ReadMem(phys_addr, 8, this);
824 break;
825
826 case 2:
827 *(u16*)dest = (u16)cSystem->ReadMem(phys_addr, 16, this);
828 break;
829
830 case 4:
831 *(u32*)dest = (u32)cSystem->ReadMem(phys_addr, 32, this);
832 break;
833
834 default:
835 FAILURE(InvalidArgument, "Strange element size");
836 }
837
838 return;
839 }
840
841#if defined(ES40_BIG_ENDIAN)
842
843 // if this is a big-endian host machine, the memcpy method is only valid
844 // if we're transferring bytes. Otherwise, endian-conversions need to be done.
845 if (element_size == 1)
846 {
847#endif
848 size_t remaining = element_size * element_count;
849 u32 cur_address = address;
850
851 while (remaining != 0)
852 {
853 u64 cur_phys = cSystem->PCI_Phys(myPCIBus, cur_address);
854 size_t chunk = pci_dma_chunk_limit(cur_phys, remaining);
855
856 // get a pointer to system memory if the address is inside main memory
857 char* memptr = cSystem->PtrToMem(cur_phys);
858 char* memptr2 = cSystem->PtrToMem(cur_phys + chunk - 1);
859
860 // Copy only within a single translated DMA page. Scatter-gather DMA does
861 // not guarantee that adjacent PCI bus addresses map to contiguous host
862 // physical memory beyond the current page.
863 if (memptr && memptr2)
864 {
865 memcpy(dst, memptr, chunk);
866 }
867 else
868 {
869 for (el = 0; el < chunk; el++)
870 dst[el] = (u8)cSystem->ReadMem(cur_phys + el, 8, this);
871 }
872
873 dst += chunk;
874 cur_address += (u32)chunk;
875 remaining -= chunk;
876 }
877 return;
878
879#if defined(ES40_BIG_ENDIAN)
880 }
881
882 // outside main memory, or inside main memory with endian-conversion
883 // required we need to do the transfer element-by-element.
884 switch (element_size)
885 {
886 case 1:
887 {
888 for (el = 0; el < element_count; el++)
889 {
890 *(u8*)dst = (u8)cSystem->ReadMem(phys_addr, 8, this);
891 dst++;
892 phys_addr++;
893 }
894 break;
895 }
896
897 case 2:
898 {
899 *(u16*)dst = endian_16((u16)cSystem->ReadMem(phys_addr, 16, this));
900 dst += 2;
901 phys_addr += 2;
902 break;
903 }
904
905 case 4:
906 {
907 *(u32*)dst = endian_32((u32)cSystem->ReadMem(phys_addr, 32, this));
908 dst += 4;
909 phys_addr += 4;
910 break;
911 }
912
913 default:
914 FAILURE(InvalidArgument, "Strange element size");
915 break;
916 }
917#endif
918}
