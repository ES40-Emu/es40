30// EV4 PAL entry offsets; reservations remain with PALcode and system logic.
31constexpr AlphaPALExceptionEntry ev4_pal_exception_entry(AlphaPALException reason,
32 std::uint64_t current_pc)
33{
34 const auto saved_pc = current_pc & ~std::uint64_t(2);
35 switch (reason)
36 {
37 case AlphaPALException::Reset: return {0x0000, saved_pc, false};
38 // EXC_ADDR[1] marks the restart PC as correct for MCHK/ARITH.
39 case AlphaPALException::MachineCheck: return {0x0020, saved_pc | 2, false};
40 // Arithmetic completes with no trap shadow.
41 case AlphaPALException::Arithmetic: return {0x0060, (saved_pc + 4) | 2, false};
42 case AlphaPALException::Interrupt: return {0x00e0, saved_pc, false};
43 case AlphaPALException::DataFault: return {0x01e0, saved_pc, false};
44 case AlphaPALException::InstructionTlbMiss: return {0x03e0, saved_pc, false};
45 case AlphaPALException::InstructionAccessViolation: return {0x07e0, saved_pc, false};
46 case AlphaPALException::DataTlbMiss:
47 return {(current_pc & 1) ? 0x09e0u : 0x08e0u, saved_pc, false};
48 case AlphaPALException::Unaligned: return {0x11e0, saved_pc, false};
49 case AlphaPALException::OpcodeDecode: return {0x13e0, saved_pc, false};
50 case AlphaPALException::FloatingPointDisabled: return {0x17e0, saved_pc, false};
51 default: return {0, saved_pc, false, false};
52 }
53}

55constexpr bool ev4_call_pal_valid(std::uint32_t function, std::uint32_t mode)
56{
57 return function <= 0x3f ? mode == 0 : function >= 0x80 && function <= 0xbf;
58}

60// CALL_PAL saves the next PC in EXC_ADDR without a GPR link.
61constexpr AlphaPALCallEntry ev4_call_pal_entry(std::uint32_t function,
62 std::uint64_t,
63 std::uint64_t next_pc)
64{
65 const std::uint64_t offset = std::uint64_t(0x2000)
66 | (std::uint64_t(function & 0x80) << 5)
67 | (std::uint64_t(function & 0x3f) << 6) | 1;
68 return {offset, next_pc & ~std::uint64_t(2), 0, 0, 0, false};
69}

71// HW_REI uses EXC_ADDR; PALcode handles reservation clearing.
72constexpr AlphaPALReturnDescriptor ev4_hardware_return(std::uint32_t instruction)
73{
74 if ((instruction >> 26) != 0x1e)
75 return {};
76 AlphaPALReturnDescriptor entry{true, ~std::uint64_t(2), 0, false, true,
77 AlphaPALReturnSource::ExceptionAddress};
78#if defined(IDB)
79 entry.mnemonic = "HW_REI";
80#endif
81 return entry;
82}

84constexpr AlphaPALInstructionAccess ev4_hardware_instruction_access(std::uint32_t instruction)
85{
86 switch (instruction >> 26)
87 {
88 case 0x19: // HW_MFPR.
89 case 0x1b: // HW_LD.
90 case 0x1d: // HW_MTPR.
91 case 0x1e: // HW_REI.
92 case 0x1f: // HW_ST.
93 return AlphaPALInstructionAccess::PalOrKernelHwe;
94 default:
95 return AlphaPALInstructionAccess::Denied;
96 }
97}
