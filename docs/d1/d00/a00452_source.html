153static void init_parse_line(char* line_to_parse)
154{
155
156 // chop off newline
157 lineptr = (unsigned char*)line_to_parse;
158
159 char* nl;
160 if ((nl = strchr(line_to_parse, '\n')) != NULL)
161 {
162 *nl = 0;
163 }

166static s32 get_next_word(char* output)
167{
168 char* copyp = output;
169
170 // find first nonspace
171 while (*lineptr && isspace(*lineptr))
172 lineptr++;
173 if (!*lineptr)
174 return -1; // nothing but spaces until end of line
175 if (*lineptr == '#')
176 return -1; // nothing but a comment
177
178 // copy nonspaces into the output
179 while (*lineptr && !isspace(*lineptr))
180 *copyp++ = *lineptr++;
181 *copyp = 0; // null terminate the copy
182
183 // there must be at least one nonspace, since that's why we stopped the
184 // first loop!
185 // BX_ASSERT (copyp != output);
186 return 0;
187}

189static s32 get_next_keymap_line(FILE* fp, char* bxsym, char* modsym,
190 s32* ascii, char* hostsym)
191{
192 char line[256];
193 char buf[256];
194 line[0] = 0;
195 while (1)
196 {
197 lineCount++;
198 if (!fgets(line, sizeof(line) - 1, fp))
199 return -1; // EOF
200 init_parse_line(line);
201 if (get_next_word(bxsym) >= 0)
202 {
203 modsym[0] = 0;
204
205 char* p;
206 if ((p = strchr(bxsym, '+')) != NULL)
207 {
208 *p = 0; // truncate bxsym.
209 p++; // move one char beyond the +
210 strcpy(modsym, p); // copy the rest to modsym
211 }
212
213 if (get_next_word(buf) < 0)
214 {
215 BX_PANIC(("keymap line %d: expected 3 columns", lineCount));
216 return -1;
217 }
218
219 if (buf[0] == '\'' && buf[2] == '\'' && buf[3] == 0)
220 {
221 *ascii = (u8)buf[1];
222 }
223 else if (!strcmp(buf, "space"))
224 {
225 *ascii = ' ';
226 }
227 else if (!strcmp(buf, "return"))
228 {
229 *ascii = '\n';
230 }
231 else if (!strcmp(buf, "tab"))
232 {
233 *ascii = '\t';
234 }
235 else if (!strcmp(buf, "backslash"))
236 {
237 *ascii = '\\';
238 }
239 else if (!strcmp(buf, "apostrophe"))
240 {
241 *ascii = '\'';
242 }
243 else if (!strcmp(buf, "none"))
244 {
245 *ascii = -1;
246 }
247 else
248 {
249 BX_PANIC((
250 "keymap line %d: ascii equivalent is \"%s\" but it must be char constant like 'x', or one of space,tab,return,none",
251 lineCount, buf));
252 }
253
254 if (get_next_word(hostsym) < 0)
255 {
256 BX_PANIC(("keymap line %d: expected 3 columns", lineCount));
257 return -1;
258 }
259
260 return 0;
261 }
262
263 // no words on this line, keep reading.
264 }

270void bx_keymap_c::loadKeymap(u32 stringToSymbol(const char*),
271 const char* filename)
272{
273 FILE* keymapFile;
274 char baseSym[256];
275 char modSym[256];
276 char hostSym[256];
277 s32 ascii = 0;
278 u32 baseKey;
279 u32 modKey;
280 u32 hostKey;
281
282 if ((keymapFile = fopen(filename, "r")) == NULL)
283 {
284 BX_PANIC(("Can not open keymap file '%s'.", filename));
285 return;
286 }
287
288 BX_INFO(("Loading keymap from '%s'", filename));
289 init_parse();
290
291 // Read keymap file one line at a time
292 while (1)
293 {
294 if (get_next_keymap_line(keymapFile, baseSym, modSym, &ascii, hostSym) < 0)
295 {
296 break;
297 }
298
299 // convert X_KEY_* symbols to values
300 baseKey = convertStringToBXKey(baseSym);
301 modKey = convertStringToBXKey(modSym);
302 hostKey = 0;
303 if (stringToSymbol != NULL)
304 hostKey = stringToSymbol(hostSym);
305
306 BX_DEBUG((
307 "baseKey='%s' (%d), modSym='%s' (%d), ascii=%d, guisym='%s' (%d)",
308 baseSym, baseKey, modSym, modKey, ascii, hostSym, hostKey));
309
310 // Check if data is valid
311 if (baseKey == BX_KEYMAP_UNKNOWN)
312 {
313 BX_PANIC(("line %d: unknown BX_KEY constant '%s'", lineCount, baseSym));
314 continue;
315 }
316
317 if (hostKey == BX_KEYMAP_UNKNOWN)
318 {
319 BX_PANIC((
320 "line %d: unknown host key name '%s' (wrong keymap ?)", lineCount,
321 hostSym));
322 continue;
323 }
324
325 CHECK_REALLOCATION(keymapTable,
326 realloc(keymapTable, (keymapCount + 1) * sizeof(BXKeyEntry)),
327 BXKeyEntry);
328
329 if (keymapTable == NULL)
330 BX_PANIC(("Can not allocate memory for keymap table."));
331
332 keymapTable[keymapCount].baseKey = baseKey;
333 keymapTable[keymapCount].modKey = modKey;
334 keymapTable[keymapCount].ascii = ascii;
335 keymapTable[keymapCount].hostKey = hostKey;
336
337 keymapCount++;
338 }
339
340 BX_INFO(("Loaded %d symbols", keymapCount));
341
342 fclose(keymapFile);
343}

348u32 bx_keymap_c::convertStringToBXKey(const char* string)
349{
350
351 // We look through the bx_key_symbol table to find the searched string
352 for (u16 i = 0; i < BX_KEY_NBKEYS; i++)
353 {
354 if (strcmp(string, bx_key_symbol[i]) == 0)
355 {
356 return i;
357 }
358 }
359
360 // Key is not known
361 return BX_KEYMAP_UNKNOWN;
362}

368BXKeyEntry* bx_keymap_c::findHostKey(u32 key)
369{
370
371 // We look through the keymap table to find the searched key
372 for (u16 i = 0; i < keymapCount; i++)
373 {
374 if (keymapTable[i].hostKey == key)
375 {
376 BX_DEBUG(("key 0x%02x matches hostKey for entry #%d", key, i));
377 return &keymapTable[i];
378 }
379 }
380
381 BX_DEBUG(("key %02x matches no entries", key));
382
383 // Return default
384 return NULL;
385}
