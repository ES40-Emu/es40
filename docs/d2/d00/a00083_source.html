105CAliM1543C_pmu::CAliM1543C_pmu(CConfigurator* cfg, CSystem* c, int pcibus, int pcidev)
106 : CPCIDevice(cfg, c, pcibus, pcidev)
107{
108 add_function(0, pmu_cfg_data, pmu_cfg_mask);
109
110 for (int i = 0; i < (int)sizeof(state.pm_block); i++)
111 state.pm_block[i] = 0;
112 for (int i = 0; i < (int)sizeof(state.smb_block); i++)
113 state.smb_block[i] = 0;
114
115 // SMBus host status: bit 0 = HOST_BUSY, leave clear; bit 1 = INTR done,
116 // leave clear so the first probe sees an idle controller.
117 state.smb_block[0x00] = 0x00;
118
119 state.pm_timer_anchor_us = 0;
120
121 ResetPCI();
122
123 printf("%s: $Id$\n", devid_string);
124}

184// Free-running 32-bit timer at 3.579545 MHz, the ACPI-fixed PM-timer rate.
185// Returned as ticks since first read of the device, so the count starts at
186// 0 the first time anyone looks and advances monotonically thereafter.
187u32 CAliM1543C_pmu::pm_timer_value()
188{
189 using clk = std::chrono::steady_clock;
190 auto now_us = std::chrono::duration_cast<std::chrono::microseconds>(
191 clk::now().time_since_epoch()).count();
192 if (state.pm_timer_anchor_us == 0)
193 state.pm_timer_anchor_us = (u64)now_us;
194 u64 elapsed_us = (u64)now_us - state.pm_timer_anchor_us;
195 // 3.579545 MHz → 3.579545 ticks per us. Use a /1000 factor that's close
196 // enough for delay-calibration loops without needing floating-point.
197 return (u32)((elapsed_us * 3579545ull) / 1000000ull);
198}

200// PM I/O block, 64 bytes wide. Layout the OS expects:
201// 0x00-0x01 PM1_STS (RW1C in real silicon, plain RW here)
202// 0x02-0x03 PM1_EN
203// 0x04-0x05 PM1_CNT
204// 0x08-0x0B PM1_TMR (32-bit, the only register with live behaviour)
205// 0x18-0x19 GPE0_STS
206// 0x1A-0x1B GPE0_EN
207u32 CAliM1543C_pmu::pm_io_read(u32 address, int dsize)
208{
209 const u32 off = address & 0x3f;
210
211 if (off == 0x08 && dsize == 32)
212 {
213 u32 v = pm_timer_value();
214#ifdef DEBUG_PMU
215 printf("%%PMU-I-PMTMR: read %08x\n", v);
216#endif
217 return v;
218 }
219
220 u32 v = 0;
221 const int bytes = dsize / 8;
222 for (int i = 0; i < bytes && (off + i) < (int)sizeof(state.pm_block); i++)
223 v |= ((u32)state.pm_block[off + i]) << (8 * i);
224
225#ifdef DEBUG_PMU
226 printf("%%PMU-I-PMRD: off=%02x dsize=%d -> %08x\n", off, dsize, v);
227#endif
228 return v;
229}

244// SMBus I/O block, 32 bytes wide. Stub: stores writes, returns last
245// written value on read. No transactions are actually performed.
246u32 CAliM1543C_pmu::smb_io_read(u32 address, int dsize)
247{

257#endif

261void CAliM1543C_pmu::smb_io_write(u32 address, int dsize, u32 data)
262{
263 const u32 off = address & 0x1f;
264
265#ifdef DEBUG_PMU
266 printf("%%PMU-I-SMBWR: off=%02x dsize=%d data=%08x\n", off, dsize, data);
267#endif
268
269 const int bytes = dsize / 8;
270 for (int i = 0; i < bytes && (off + i) < (int)sizeof(state.smb_block); i++)
271 state.smb_block[off + i] = (u8)(data >> (8 * i));
272
273 // SMBus host status (offset 0). Bit 0 (HOST_BUSY) auto-clears so the
274 // next status read shows the (stub) transaction completed; bit 1
275 // (INTR/done) gets set so polled drivers see success.
276 if (off == 0x00)
277 {
278 state.smb_block[0] &= ~0x01;
279 state.smb_block[0] |= 0x02;
280 }
281}

302int CAliM1543C_pmu::RestoreState(FILE* f)
303{
304 long ss;
305 u32 m1, m2;
306 int res;
307 size_t r;
308
309 if ((res = CPCIDevice::RestoreState(f)))
310 return res;
311
312 r = fread(&m1, sizeof(u32), 1, f);
313 if (r != 1) { printf("%s: unexpected end of file!\n", devid_string); return -1; }
314 if (m1 != pmu_magic1) { printf("%s: MAGIC 1 does not match!\n", devid_string); return -1; }
315
316 r = fread(&ss, sizeof(long), 1, f);
317 if (r != 1) { printf("%s: unexpected end of file!\n", devid_string); return -1; }
318 if (ss != sizeof(state)) { printf("%s: STRUCT SIZE does not match!\n", devid_string); return -1; }
319
320 r = fread(&state, sizeof(state), 1, f);
321 if (r != 1) { printf("%s: unexpected end of file!\n", devid_string); return -1; }
322
323 r = fread(&m2, sizeof(u32), 1, f);
324 if (r != 1) { printf("%s: unexpected end of file!\n", devid_string); return -1; }
325 if (m2 != pmu_magic2) { printf("%s: MAGIC 2 does not match!\n", devid_string); return -1; }
326
327 printf("%s: %d bytes restored.\n", devid_string, (int)ss);
328 return 0;
