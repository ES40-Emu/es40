32std::uint64_t Ev68IPR::read(CAlphaCPU& cpu, std::uint32_t instruction)
33{
34 const auto function = selector(instruction);
35 const auto& state = cpu.state;
36 if ((function & 0xc0) == 0x40) // PCTX
37 return ((u64)state.asn << 39) | ((u64)state.astrr << 9) | ((u64)state.aster << 5)
38 | (state.fpen ? U64(0x1) << 2 : 0) | (state.ppcen ? U64(0x1) << 1 : 0);
39
40 switch (function)
41 {
42 case 0x05: return state.pmpc; // PMPC
43 case 0x06: return state.exc_addr; // EXC_ADDR
44 case 0x07: return cpu.va_form(state.exc_addr, true); // IVA_FORM
45 case 0x08: case 0x09: case 0x0a: case 0x0b: // IER_CM / CM / IER
46 return (((u64)state.eien) << 33) | (((u64)state.slen) << 32)
47 | (((u64)state.cren) << 31) | (((u64)state.pcen) << 29)
48 | (((u64)state.sien) << 13) | (((u64)state.asten) << 13)
49 | (((u64)state.cm) << 3);
50 case 0x0c: return ((u64)state.sir) << 13; // SIRR
51 case 0x0d: // ISUM
52 return (((u64)(state.eir & state.eien)) << 33)
53 | (((u64)(state.slr & state.slen)) << 32)
54 | (((u64)(state.crr & state.cren)) << 31)
55 | (((u64)(state.pcr & state.pcen)) << 29)
56 | (((u64)(state.sir & state.sien)) << 13)
57 | (((u64)(((U64(0x1) << (state.cm + 1)) - 1) & state.aster & state.astrr & (state.asten * 0x3))) << 3)
58 | (((u64)(((U64(0x1) << (state.cm + 1)) - 1) & state.aster & state.astrr & (state.asten * 0xc))) << 7);
59 case 0x0f: return state.exc_sum; // EXC_SUM
60 case 0x10: return state.pal_base; // PAL_BASE
61 case 0x11: // I_CTL
62 return state.i_ctl_other | (((u64)cpu.get_profile().chip_id) << 24) | (u64)state.i_ctl_vptb
63 | (((u64)state.i_ctl_va_mode) << 15) | (state.hwe ? U64(0x1) << 12 : 0)
64 | (state.sde ? U64(0x1) << 7 : 0) | (((u64)state.i_ctl_spe) << 3);
65 case 0x14: return state.pctr_ctl; // PCTR_CTL
66 case 0x16: return state.i_stat; // I_STAT
67 case 0x27: return state.mm_stat; // MM_STAT
68 case 0x2a: return state.dc_stat; // DC_STAT
69 case 0x2b: return 0; // C_DATA
70 case 0xc0: return (((u64)state.cc_offset) << 32) | (state.cc & U64(0xffffffff)); // CC
71 case 0xc2: return state.fault_va; // VA
72 case 0xc3: return cpu.va_form(state.va_form_va, false); // VA_FORM
73 default:
74 // Unimplemented EV68 indices read zero and warn once per index.
75 static bool warned_ipr[256];
76 if (!warned_ipr[function & 0xff])
77 {
78 warned_ipr[function & 0xff] = true;
79 printf("%%CPU-W-IPR: unimplemented HW_MFPR index %02x at pc=%016" PRIx64 " (ignored)\n",
80 (unsigned)function, state.current_pc);
81 }
82 return 0;
83 }

86void Ev68IPR::write(CAlphaCPU& cpu, std::uint32_t instruction, std::uint64_t value,
87 AlphaIPRWriteContext context)
88{
89 const auto function = selector(instruction);
90 // Keep writes that may change dispatch context out of a compiled frame.
91 if (context == AlphaIPRWriteContext::Compiled
92 && describe_write(instruction) == AlphaIPRWriteKind::Interpreter)
93 return;
94
95 auto& state = cpu.state;
96 if ((function & 0xc0) == 0x40)
97 {
98 if (function & 1)
99 {
100 state.asn = (int) (value >> 39) & 0xff;
101 cpu.flush_data_page_cache();
102 cpu.jit_note_asn_change();
103 }
104 if (function & 2)
105 {
106 state.aster = (int) (value >> 5) & 0xf;
107 }
108 if (function & 4)
109 {
110 state.astrr = (int) (value >> 9) & 0xf;
111 }
112 if (function & 8)
113 state.ppcen = (int) (value >> 1) & 1;
114 if (function & 16)
115 state.fpen = (int) (value >> 2) & 1;
116 if (function & 6)
117 cpu.kick_int_if_pending();
118 }
119 else
120 {
121 switch (function)
122 {
123 case 0x00: /* ITB_TAG */
124 state.last_tb_virt = value;
125 break;
126
127 case 0x01: /* ITB_PTE */
128 cpu.add_tb_i(state.last_tb_virt, value);
129 break;
130
131 case 0x02: /* ITB_IAP */
132 cpu.tbiap(ACCESS_EXEC);
133 break;
134
135 case 0x03: /* ITB_IA */
136 cpu.tbia(ACCESS_EXEC);
137 break;
138
139 case 0x04: /* ITB_IS */
140 cpu.tbis(value, ACCESS_EXEC);
141 break;
142
143 case 0x09: /* CM */
144 {
145 const int old_cm = state.cm;
146 state.cm = (int) (value >> 3) & 3;
147#ifdef ES40_JIT
148 if (state.cm != old_cm)
149 cpu.jit_note_mode_change();
150#else
151 (void)old_cm;
152#endif
153 if (context == AlphaIPRWriteContext::Compiled)
154 cpu.dpc_sync_bank();
155 else
156 cpu.flush_data_page_cache();
157 cpu.kick_int_if_pending();
158 break;
159 }
160
161 case 0x0b: /* IER_CM */
162 {
163 const int old_cm = state.cm;
164 state.cm = (int) (value >> 3) & 3;
165#ifdef ES40_JIT
166 if (state.cm != old_cm)
167 cpu.jit_note_mode_change();
168#else
169 (void)old_cm;
170#endif
171 if (context == AlphaIPRWriteContext::Compiled)
172 cpu.dpc_sync_bank();
173 else
174 cpu.flush_data_page_cache();
175 }
176 [[fallthrough]];
177 case 0x0a: /* IER */
178 state.asten = (int) (value >> 13) & 1;
179 state.sien = (int) (value >> 13) & 0xfffe;
180 state.pcen = (int) (value >> 29) & 3;
181 state.cren = (int) (value >> 31) & 1;
182 state.slen = (int) (value >> 32) & 1;
183 state.eien = (int) (value >> 33) & 0x3f;
184 cpu.kick_int_if_pending();
185 break;
186
187 case 0x0c: /* SIRR */
188 state.sir = (int) (value >> 13) & 0xfffe;
189 cpu.kick_int_if_pending();
190 break;
191
192 case 0x0e: /* HW_INT_CLR */
193 state.pcr &= ~((value >> 29) & U64(0x3));
194 state.crr &= ~((value >> 31) & U64(0x1));
195 state.slr &= ~((value >> 32) & U64(0x1));
196 break;
197
198 case 0x10: /* PAL_BASE */
199 cpu.set_PAL_BASE(value & U64(0x00000fffffff8000));
200 break;
201
202 case 0x11: /* i_ctl */
203 /* Bit 20 (CALL_PAL_R23 / ST_WAIT_64K) is hardwired-on for EV6/EV68 */
204 state.i_ctl_other = (value & U64(0x00000000006e2f67))
205 | U64(0x0000000000100000);
206 state.i_ctl_vptb = sext_u64_48(value & U64(0x0000ffffc0000000));
207 state.i_ctl_spe = (int) ((value >> 3) & 7);
208 state.sde = (value >> 7) & 1;
209 state.hwe = (value >> 12) & 1;
210 state.i_ctl_va_mode = (int) (value >> 15) & 3;
211 break;
212
213 case 0x12: /* ic_flush_asm */
214 cpu.flush_icache_asm();
215 break;
216
217 case 0x13: /* IC_FLUSH */
218 cpu.flush_icache();
219 break;
220
221 case 0x14: /* PCTR_CTL */
222 state.pctr_ctl = value & U64(0xffffffffffffffdf);
223 break;
224
225 case 0x08: /* IER_CM, neither field selected (HRM index 0000 10xx) */
226 case 0x0f: /* EXC_SUM: RO, implicitly written at trap delivery -- the W2K
227 build 1855 PALcode writes it and EV6 ignores it (issue #169) */
228 case 0x15: /* CLR_MAP */
229 case 0x17: /* SLEEP */
230 case 0x27: /* MM_STAT */
231 case 0x2b: /* C_DATA */
232 case 0x2c: /* C_SHIFT */
233 case 0x2d: /* M_FIX */
234 break;
235
236 case 0x16: /* I_STAT */
237 state.i_stat &= ~value; /* W1C */
238 break;
239
240 case 0x20: /* DTB_TAG0 */
241 cpu.last_dtb_virt[0] = value;
242 break;
243
244 case 0x21: /* DTB_PTE0 */
245 cpu.add_tb_d(cpu.last_dtb_virt[0], value, 0);
246 break;
247
248 case 0x24: /* DTB_IS0 */
249 cpu.tbis_d(value, state.asn0);
250 break;
251
252 case 0x25: /* DTB_ASN0 */
253 state.asn0 = (int) (value >> 56);
254 cpu.flush_data_page_cache();
255 break;
256
257 case 0x26: /* DTB_ALTMODE */
258 state.alt_cm = (int) (value & 3);
259 break;
260
261 case 0x28: /* M_CTL */
262 state.smc = (int) (value >> 4) & 3;
263 state.m_ctl_spe = (int) (value >> 1) & 7;
264 cpu.flush_data_page_cache();
265 break;
266
267 case 0x29: /* DC_CTL */
268 state.dc_ctl = value;
269 break;
270
271 case 0x2a: /* DC_STAT */
272 state.dc_stat &= ~value;
273 break;
274
275 case 0xa0: /* DTB_TAG1 */
276 cpu.last_dtb_virt[1] = value;
277 break;
278
279 case 0xa1: /* DTB_PTE1 */
280 cpu.add_tb_d(cpu.last_dtb_virt[1], value, 1);
281 break;
282
283 case 0xa2: /* DTB_IAP */
284 cpu.tbiap(ACCESS_READ);
285 break;
286
287 case 0xa3: /* DTB_IA */
288 cpu.tbia(ACCESS_READ);
289 break;
290
291 case 0xa4: /* DTB_IS1 */
292 cpu.tbis_d(value, state.asn1);
293 break;
294
295 case 0xa5: /* DTB_ASN1 */
296 state.asn1 = (int) (value >> 56);
297 cpu.flush_data_page_cache();
298 break;
299
300 case 0xc0: /* CC */
301 state.cc_offset = (u32) (value >> 32);
302 break;
303
304 case 0xc1: /* CC_CTL */
305 state.cc_ena = (value >> 32) & 1;
306 state.cc = (u32) (value & U64(0xfffffff0));
307 cpu.cc_last_read = state.cc; /* counter rewritten: rebase the rpcc_read floor
308 epoch or the next read would jump/false-borrow
309 against the old counter value */
310 cpu.cc_borrow = 0;
311 cpu.cc_wall_remainder = 0;
312 break;
313
314 case 0xc4: /* VA_CTL */
315 state.va_ctl_vptb = sext_u64_48(value & U64(0x0000ffffc0000000));
316 state.va_ctl_va_mode = (int) (value >> 1) & 3;
317 cpu.flush_data_page_cache();
318 cpu.tbia(ACCESS_READ);
319 break;
320
321 default:
322 // Unimplemented EV68 indices ignore writes and warn once per index.
323 static bool warned_ipr[256];
324 if (!warned_ipr[function & 0xff])
325 {
326 warned_ipr[function & 0xff] = true;
327 printf("%%CPU-W-IPR: unimplemented HW_MTPR index %02x at pc=%016" PRIx64 " (ignored)\n",
328 (unsigned)function, state.current_pc);
329 }
330 break;
331 }
332 }
