52const char *strvmnetstatus (vmnet_return_t status) {
53 switch (status) {
54 case VMNET_SUCCESS:
55 return "Successfully completed.";
56 case VMNET_FAILURE:
57 return "General failure.";
58 case VMNET_MEM_FAILURE:
59 return "Memory allocation failure.";
60 case VMNET_INVALID_ARGUMENT:
61 return "Invalid argument specified.";
62 case VMNET_SETUP_INCOMPLETE:
63 return "Interface setup is not complete.";
64 case VMNET_INVALID_ACCESS:
65 return "Permission denied.";
66 case VMNET_PACKET_TOO_BIG:
67 return "Packet size larger than MTU.";
68 case VMNET_BUFFER_EXHAUSTED:
69 return "Buffers exhausted in kernel.";
70 case VMNET_TOO_MANY_PACKETS:
71 return "Packet count exceeds limit.";
72 case VMNET_SHARING_SERVICE_BUSY:
73 return "vmnet interface cannot be started as conflicting sharing service is in use.";
74 case VMNET_NOT_AUTHORIZED:
75 return "The operation could not be completed due to missing authorization.";
76 default:
77 return "Unknown status code";
78 }

81bool CNetworkVmnet::init(const char *devid_string, CConfigurator *cfg)
82{
83 bool success;
84 xpc_object_t interface_names = vmnet_copy_shared_interface_list ();
85 int n_interfaces, i;
86 char *adapter = cfg->get_text_value ("adapter");
87 char *interface;
88 char selected_adapter[IFNAMSIZ];
89 struct es40_vmnet_open_reply reply;
90
91 devid_for_log = devid_string;
92 dead = false;
93
94 if (interface_names == nullptr) {
95 printf ("%s: Cannot get list of vmnet capable interfaces.\n", devid_for_log);
96 return false;
97 } else
98 n_interfaces = xpc_array_get_count (interface_names);
99
100 if (adapter && adapter[0]) {
101 // Verify that vmnet can use this adapter
102 for (i = 0; i < n_interfaces; i++) {
103 interface = (char *) xpc_array_get_string (interface_names, i);
104 if (strncmp (adapter, interface, IFNAMSIZ) == 0)
105 break;
106 }
107 if (n_interfaces == i) {
108 printf ("%s: Adapter %s cannot serve as a bridge.\n", devid_for_log, adapter);
109 xpc_release (interface_names);
110 return false;
111 }
112 }
113
114 else {
115 // Use the first vmnet capable interface
116 if (n_interfaces == 0) {
117 xpc_release (interface_names);
118 printf ("%s: No network interfaces available to serve as a bridge.\n", devid_for_log);
119 return false;
120 }
121 adapter = (char *) xpc_array_get_string (interface_names, 0);
122 }
123
124 strlcpy (selected_adapter, adapter, sizeof(selected_adapter));
125 xpc_release (interface_names);
126
127 ipc = new CNetworkVmnetIPC();
128 memset(&reply, 0, sizeof(reply));
129 success = ipc->start(devid_string, selected_adapter, &reply);
130
131 // Drop privileges if allowed
132 if (cfg->get_bool_value ("drop_privileges", true)) {
133 seteuid (getuid ());
134 setegid (getgid ());
135 }
136
137 if (!success) {
138 close();
139 return false;
140 }
141 if (reply.msg.type != ES40_VMNET_MSG_OPEN_REPLY ||
142 reply.magic != ES40_VMNET_PROTO_MAGIC ||
143 reply.version != ES40_VMNET_PROTO_VERSION) {
144 printf("%s: %s uses an incompatible vmnet protocol.\n",
145 devid_for_log, ipc->get_helper_path());
146 close();
147 return false;
148 }
149 if (reply.status != ES40_VMNET_STATUS_OK) {
150 if (reply.status == ES40_VMNET_STATUS_HELPER_ERROR) {
151 printf("%s: vmnet helper failed: %s\n", devid_for_log,
152 strerror(reply.system_error));
153 } else {
154 printf("%s: Unable to start vmnet on %s: %s\n",
155 devid_for_log, selected_adapter,
156 strvmnetstatus((vmnet_return_t)reply.status));
157 }
158 if (reply.helper_euid != 0) {
159 printf("%s: %s must be installed setuid root.\n",
160 devid_for_log, ipc->get_helper_path());
161 }
162 close();
163 return false;
164 }
165 if (reply.max_packet_size == 0 ||
166 reply.max_packet_size > ES40_VMNET_MAX_FRAME) {
167 printf("%s: vmnet reported an invalid maximum packet size (%u).\n",
168 devid_for_log, reply.max_packet_size);
169 close();
170 return false;
171 }
172
173 max_packet_size = reply.max_packet_size;
174 rx_buf = malloc(max_packet_size);
175 if (rx_buf == nullptr) {
176 printf("%s: Cannot allocate vmnet receive buffer.\n",
177 devid_for_log);
178 close();
179 return false;
180 }
181
182 return true;
183}

193int CNetworkVmnet::send(const u8 *data, int len)
194{
195 if (ipc == nullptr || dead)
196 return -1;
197 if (len <= 0 || (uint64_t)len > max_packet_size)
198 return -1;
199 if (ipc->send_packet(data, (size_t)len) == 0)
200 return 0;
201 report_dead(strerror(errno));
202 return -1;
203}

205int CNetworkVmnet::receive(const u8 **data, int *len)
206{
207 if (ipc == nullptr || dead)
208 return -1;
209
210 for (;;) {
211 struct es40_vmnet_status status;
212 int result = ipc->receive(rx_buf, max_packet_size, len, &status);
213 if (result == 1) {
214 *data = (const u8 *)rx_buf;
215 return 1;
216 }
217 if (result == 0)
218 return 0;
219 if (result < 0) {
220 report_dead(strerror(errno));
221 return -1;
222 }
223
224 const char *operation = status.operation == ES40_VMNET_OP_READ ?
225 "receiving" : "sending";
226 printf("%s: Error %s packet: %s\n", devid_for_log, operation,
227 strvmnetstatus((vmnet_return_t)status.status));
228 }
229}

248void CNetworkVmnet::close()
249{
250 if (ipc != nullptr) {
251 delete ipc;
252 ipc = nullptr;
253 }
254 if (rx_buf != nullptr) {
255 free(rx_buf);
256 rx_buf = nullptr;
257 }
258 max_packet_size = 0;
259 dead = false;
260}
