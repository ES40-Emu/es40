53void I2CBus::recompute_lines() {
54 bool dev_scl_pull = false, dev_sda_pull = false;
55 for (const auto& d : devs_) {
56 dev_scl_pull = dev_scl_pull || d->pull_scl_low();
57 dev_sda_pull = dev_sda_pull || d->pull_sda_low();
58 }
59 line_scl_ = host_scl_ && !dev_scl_pull; // wired-AND (open-drain)
60 line_sda_ = host_sda_ && !dev_sda_pull;
61}

68void I2CBus::drive_from_host(bool scl_release, bool sda_release) {
69 // Save old line levels before change
70 bool old_scl = line_scl_;
71 bool old_sda = line_sda_;
72
73 host_scl_ = scl_release;
74 host_sda_ = sda_release;
75
76 // First recompute lines with new host drives (devices might already be pulling)
77 recompute_lines();
78
79 // START/STOP detection (based on *line* edges while SCL is high)
80 if (old_sda && !line_sda_ && line_scl_) {
81 notify_start();
82 }
83 else if (!old_sda && line_sda_ && line_scl_) {
84 notify_stop();
85 }
86
87 // SCL edges
88 if (!old_scl && line_scl_) {
89 notify_scl_rise();
90 }
91 else if (old_scl && !line_scl_) {
92 notify_scl_fall();
93 }
94
95 // Devices may drive ACK/data while SCL low; recompute one more time.
96 recompute_lines();
97}

128void Eeprom24C02::enter_xmit() {
129 st_ = XMIT; bitpos_ = 0; ack_phase_ = ACK_DATA;
130 // Prepare first data bit while SCL is low (caller ensures this is called on SCL low path)
131 shreg_ = mem_[wordptr_++];
132 // MSB first
133 ack_pull_ = ((shreg_ & 0x80) == 0); // '0' bit -> pull low
134}

145void Eeprom24C02::on_scl_rise(bool sda_line) {
146 switch (st_) {
147 case ADDR:
148 // Shift in 7-bit address + R/W on rising edges
149 shreg_ = static_cast<uint8_t>((shreg_ << 1) | (sda_line ? 1 : 0));
150 if (++bitpos_ == 8) {
151 rw_ = (shreg_ & 1) != 0;
152 addr_match_ = (static_cast<uint8_t>(shreg_ >> 1) == addr7_);
153 // Drive ACK during the 9th clock if we match
154 ack_pull_ = addr_match_; // ACK low
155 st_ = RECV_ACK;
156 bitpos_ = 0;
157 }
158 break;
159
160 case WORD:
161 // Shift in 8-bit memory pointer
162 shreg_ = static_cast<uint8_t>((shreg_ << 1) | (sda_line ? 1 : 0));
163 if (++bitpos_ == 8) {
164 if (addr_match_) wordptr_ = shreg_;
165 ack_pull_ = addr_match_; // ACK low
166 st_ = RECV_ACK;
167 bitpos_ = 0;
168 }
169 break;
170
171 case XMIT:
172 // Host samples our data on rising edges; bitpos_ counts 0..7
173 if (++bitpos_ == 8) {
174 // Release SDA for host ACK/NACK on the 9th clock
175 ack_pull_ = false;
176 st_ = RECV_ACK;
177 bitpos_ = 0;
178 }
179 break;
180
181 case RECV_ACK:
182 // Host drives ACK/NACK (low/high) on this 9th rising edge.
183 if (ack_phase_ == ACK_DATA) {
184 // If NACK, stop transmitting; if ACK, continue reading
185 bool host_ack = (sda_line == false);
186 if (!host_ack) {
187 enter_idle();
188 }
189 }
190 // For ADDR/WORD ACKs we don't need to read host; we already drove ACK.
191 break;
192
193 default: break;
194 }
195}

197void Eeprom24C02::prepare_next_tx_bit() {
198 // bitpos_ in 0..7 for data bits; prepare SDA while SCL is low
199 if (bitpos_ < 8) {
200 uint8_t mask = static_cast<uint8_t>(0x80 >> bitpos_);
201 ack_pull_ = ((shreg_ & mask) == 0);
202 }
203}

205void Eeprom24C02::on_scl_fall(bool /*sda_line*/) {
206 switch (st_) {
207 case RECV_ACK:
208 // Transition to next state after the ACK bit low-to-high phase.
209 if (!addr_match_) { enter_idle(); break; }
210
211 if (ack_phase_ == ACK_ADDR) {
212 // After address ACK: if WRITE -> expect word address; if READ -> begin transmit
213 if (!rw_) {
214 enter_word();
215 }
216 else {
217 enter_xmit(); // prepare first data byte
218 }
219 }
220 else if (ack_phase_ == ACK_WORD) {
221 // After word pointer ACK: expect repeated START + READ address
222 // We simply wait in ADDR for the next START (already handled in on_start()).
223 enter_addr();
224 }
225 else if (ack_phase_ == ACK_DATA) {
226 // After ACKing a transmitted byte: continue sending next byte
227 enter_xmit();
228 }
229 break;
230
231 case XMIT:
232 // Prepare subsequent data bits while SCL is low.
233 prepare_next_tx_bit();
234 break;
235
236 default:
237 break;
238 }

242std::vector<uint8_t> build_sdram_spd(uint32_t mb, bool registered_ecc)
243{
244 // ES40-typical: Registered ECC PC100 SDRAM (168-pin), CL=2/3 supported.
245 // Geometry chosen to make capacity math coherent for 64/128/256/512/1024 MB.
246 struct Geo { uint8_t rows, cols, ranks; } g{};
247 switch (mb) {
248 case 8: g = { 10, 8, 1 }; break; // tiny legacy parts
249 case 16: g = { 11, 8, 1 }; break;
250 case 32: g = { 11, 9, 1 }; break;
251 case 64: g = { 12, 9, 1 }; break; // 8Mx8 devices, 1 rank
252 case 128: g = { 13, 9, 1 }; break; // 16Mx8, 1 rank
253 case 256: g = { 13, 10, 2 }; break; // 16Mx8, 2 ranks
254 case 512: g = { 14, 10, 2 }; break; // 32Mx8, 2 ranks
255 case 1024: g = { 14, 10, 2 }; break; // 32Mx8, 2 ranks (denser parts)
256 default: g = { 13, 10, 2 }; break;
257 }
258 std::vector<uint8_t> b(256, 0x00);
259 b[0] = 0x80; // bytes used
260 b[1] = 0x08; // SPD rev 1.3 (0x08 is commonly used)
261 b[2] = 0x04; // SDR SDRAM
262 b[3] = g.rows; // Row address bits
263 b[4] = g.cols; // Column address bits
264 // Byte 5: module attributes - bit1 Registered, bit5 ECC
265 b[5] = (registered_ecc ? 0x20 : 0x00) | 0x02; // ECC + Registered
266 b[6] = 0x04; // SDRAM device banks (4)
267 // Data width 64, ECC width 8 -> 72-bit module (ES40 expects ECC)
268 b[7] = 64; b[8] = 0;
269 b[11] = 8; b[12] = 0;
270 b[17] = g.ranks; // module ranks
271 // Conservative PC100 timings (CL=2/3). Units are ns.
272 b[9] = 20; // tAA (CL=2) 20 ns
273 b[10] = 2; // tWR (~2ns; not used by SRM)
274 b[18] = 20; // tRCD 20 ns
275 b[19] = 20; // tRP 20 ns
276 b[20] = 10; // tCK min at highest supported CL (10 ns => 100 MHz)
277 b[21] = 10; // tCK at CL=2 also 10 ns (safe)
278 b[22] = 45; // tRAS 45 ns (common PC100 value)
279 // Byte 23: supported CAS latencies bitmask: bit1=CL2, bit2=CL3
280 b[23] = 0x06; // CL=2 and CL=3 supported
281 // Compute checksum over bytes 0..62
282 uint8_t sum = 0; for (int i = 0; i <= 62; i++) sum = (uint8_t)(sum + b[i]);
283 b[63] = (uint8_t)(0x100 - sum);
284 return b;
285}
