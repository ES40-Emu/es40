94 void readLock()
95 {
96#if defined(DEBUG_LOCKS)
97 printf(" READ LOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
98#endif
99 _mutex.lock_shared();
100 _exclusiveHeld.store(false, std::memory_order_release);
101#if defined(DEBUG_LOCKS)
102 printf(" READ LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
103#endif
104 }

106 void writeLock()
107 {
108#if defined(DEBUG_LOCKS)
109 printf(" WRITE LOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
110#endif
111 _mutex.lock();
112 _exclusiveHeld.store(true, std::memory_order_release);
113#if defined(DEBUG_LOCKS)
114 printf("WRITE LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
115#endif
116 }

118 void readLock(long milliseconds)
119 {
120#if defined(DEBUG_LOCKS)
121 printf(" READ LOCK mutex %s from thread %s (%ld ms).\n", lockName, CURRENT_THREAD_NAME, milliseconds);
122#endif
123 if (!tryReadLock(milliseconds))
124 {
125 printf("TIMEOUT read-locking %s from thread %s after %ld ms.\n",
126 lockName, CURRENT_THREAD_NAME, milliseconds);
127 throw std::runtime_error("CRWLock::readLock: timeout");
128 }

131#endif
132 }

136#if defined(DEBUG_LOCKS)
137 printf(" WRITE LOCK mutex %s from thread %s (%ld ms).\n", lockName, CURRENT_THREAD_NAME, milliseconds);
138#endif
139 if (!tryWriteLock(milliseconds))
140 {
141 printf("TIMEOUT write-locking %s from thread %s after %ld ms.\n",
142 lockName, CURRENT_THREAD_NAME, milliseconds);
143 throw std::runtime_error("CRWLock::writeLock: timeout");
144 }
145#if defined(DEBUG_LOCKS)
146 printf("WRITE LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
147#endif
148 }

150 bool tryReadLock()
151 {
152 bool res = _mutex.try_lock_shared();
153 if (res)
154 _exclusiveHeld.store(false, std::memory_order_release);
155#if defined(DEBUG_LOCKS)
156 printf(" %s mutex %s from thread %s.\n",
157 res ? " RD LOCKED" : "CAN'T LOCK", lockName, CURRENT_THREAD_NAME);
158#endif
159 return res;
160 }
161
162 bool tryWriteLock()
163 {
164 bool res = _mutex.try_lock();
165 if (res)
166 _exclusiveHeld.store(true, std::memory_order_release);
167#if defined(DEBUG_LOCKS)
168 printf(" %s mutex %s from thread %s.\n",
169 res ? " WR LOCKED" : "CAN'T LOCK", lockName, CURRENT_THREAD_NAME);
170#endif
171 return res;
172 }
173
174 bool tryReadLock(long milliseconds)
175 {
176 auto deadline = std::chrono::steady_clock::now()
177 + std::chrono::milliseconds(milliseconds);
178 do
179 {
180 if (_mutex.try_lock_shared())
181 {
182 _exclusiveHeld.store(false, std::memory_order_release);
183#if defined(DEBUG_LOCKS)
184 printf(" RD LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
185#endif
186 return true;
187 }
188 std::this_thread::sleep_for(std::chrono::milliseconds(5));
189 } while (std::chrono::steady_clock::now() < deadline);
190#if defined(DEBUG_LOCKS)
191 printf("CAN'T LOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
192#endif
193 return false;
194 }
195
196 bool tryWriteLock(long milliseconds)
197 {
198 auto deadline = std::chrono::steady_clock::now()
199 + std::chrono::milliseconds(milliseconds);
200 do
201 {
202 if (_mutex.try_lock())
203 {
204 _exclusiveHeld.store(true, std::memory_order_release);
205#if defined(DEBUG_LOCKS)
206 printf(" WR LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
207#endif
208 return true;
209 }
210 std::this_thread::sleep_for(std::chrono::milliseconds(5));
211 } while (std::chrono::steady_clock::now() < deadline);
212#if defined(DEBUG_LOCKS)
213 printf("CAN'T LOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
214#endif
215 return false;
216 }
217
218 void unlockRead()
219 {
220#if defined(DEBUG_LOCKS)
221 printf(" UNLOCKRD mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
222#endif
223 _mutex.unlock_shared();
224 }
225
226 void unlockWrite()
227 {
228#if defined(DEBUG_LOCKS)
229 printf(" UNLOCKWR mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
230#endif
231 _mutex.unlock();
232 }
233
234 void unlock()
235 {
236 if (_exclusiveHeld.load(std::memory_order_acquire))
237 unlockWrite();
238 else
239 unlockRead();
240 }
241
242 char* lockName;
243
244private:
245 std::shared_mutex _mutex;
246 std::atomic<bool> _exclusiveHeld{ false };
247
248 CRWLock(const CRWLock&) = delete;
249 CRWLock& operator=(const CRWLock&) = delete;
250};

252class CScopedRWLock
253{
254public:
255 CScopedRWLock(CRWLock* rwl, bool write = false)
256 : _rwl(rwl), _write(write)
257 {
258 if (_write)
259 _rwl->writeLock();
260 else
261 _rwl->readLock();
262 }
263
264 ~CScopedRWLock()
265 {
266 if (_write)
267 _rwl->unlockWrite();
268 else
269 _rwl->unlockRead();
270 }
271
272 CScopedRWLock(const CScopedRWLock&) = delete;
273 CScopedRWLock& operator=(const CScopedRWLock&) = delete;
274
275private:
276 CRWLock* _rwl;
277 bool _write;
278};
