183CConfigurator::CConfigurator(class CConfigurator* parent, char* name,
184 char* value, char* text, size_t textlen, bool initialize_root)
185{
186 if (parent == 0)
187 {
188
189 /* Phase 1: Basic Syntax Check & Strip [first pass only]
190 * - Make sure quotes and comments are closed.
191 * - Make sure braces are balanced
192 * - remove everything except configuration data.
193 */
194 char* dst = (char*)malloc(textlen + 1);
195 char* p = dst;
196 char* q = text;
197
198 int cbrace = 0;
199 enum
200 {
201 STATE_NONE,
202 STATE_C_COMMENT,
203 STATE_CC_COMMENT,
204 STATE_STRING
205 }
206
207 state = STATE_NONE;
208
209 int state_start = 0;
210 int line = 1;
211 int col = 1;
212 bool line_has_content = false;
213 for (unsigned i = 0; i < textlen; i++, q++, col++)
214 {
215 if (*q == 0x0a)
216 {
217 line++;
218 col = 1;
219 line_has_content = false;
220 }
221
222 switch (state)
223 {
224 case STATE_NONE:
225 switch (*q)
226 {
227 case '"':
228 state = STATE_STRING;
229 state_start = line;
230 line_has_content = true;
231 *p++ = *q;
232 break;
233
234 case '/':
235 if (i == textlen - 1)
236 FAILURE_2(Configuration,
237 "Configuration file ends in mid-token at line %d, col %d",
238 line, col);
239
240 if (*(q + 1) == '/')
241 {
242 state = STATE_CC_COMMENT;
243 state_start = line;
244 }
245 else if (*(q + 1) == '*')
246 {
247 state = STATE_C_COMMENT;
248 state_start = line;
249 }
250 break;
251
252 case '{':
253 cbrace++;
254 line_has_content = true;
255 *p++ = *q;
256 break;
257
258 case '}':
259 if (cbrace == 0)
260 FAILURE_2(Configuration,
261 "Too many closed braces at line %d, col %d", line, col);
262 cbrace--;
263 line_has_content = true;
264 *p++ = *q;
265 break;
266
267 case ';':
268 // A ';' as the first non-whitespace character of a line is
269 // an INI-style comment running to end of line. Elsewhere it
270 // keeps its meaning as the value terminator.
271 if (!line_has_content)
272 {
273 state = STATE_CC_COMMENT;
274 state_start = line;
275 }
276 else
277 *p++ = *q;
278 break;
279
280 default:
281 if (!isspace(*q))
282 {
283 line_has_content = true;
284 if (isalnum(*q) || *q == '_' || *q == '.' || *q == '=')
285 {
286 *p++ = *q;
287 }
288 else
289 FAILURE_3(Configuration,
290 "Illegal character %c at line %d, col %d", *q, line, col);
291 }
292 }
293 break;
294
295 case STATE_CC_COMMENT: // c++ comment
296 if (*q == 0x0d || *q == 0x0a)
297 {
298 state = STATE_NONE;
299 state_start = line;
300 }
301 break;
302
303 case STATE_C_COMMENT: // c comment
304 if (*q == '*')
305 {
306 if (i == textlen - 1)
307 FAILURE_2(Configuration,
308 "Configuration file ends in mid-comment at line %d, col %d",
309 line, col);
310 if (*(q + 1) == '/')
311 {
312 state = STATE_NONE;
313 state_start = line;
314 }
315 }
316 break;
317
318 case STATE_STRING: // string
319 if (*q == '"')
320 {
321 if (i == textlen - 1)
322 FAILURE_2(Configuration,
323 "Configuration file ends in mid-string at line %d, col %d",
324 line, col);
325 if (*(q + 1) != '"')
326 {
327 state_start = line;
328 state = STATE_NONE;
329 }
330 else
331 {
332 *p++ = *q;
333 i++;
334 q++;
335 }
336 }
337 else if (*q == 0x0a || *q == 0x0d)
338 FAILURE_2(Configuration,
339 "Multi-line strings are forbidden at line %d, col %d", line,
340 col);
341 *p++ = *q;
342 break;
343 }
344 }
345
346 *p++ = 0;
347
348 if (state != 0 && state != 1)
349 {
350 printf("%%SYS-E-PARSE: unclosed %s. Started on line %d.\n",
351 state == STATE_C_COMMENT ? "comment" : "string", state_start);
352 }
353
354 if (cbrace != 0)
355 {
356 printf("%%SYS-E-PARSE: unclosed brace in file.\n");
357 }
358
359 textlen = strlen(dst);
360 memcpy(text, dst, textlen + 1);
361 free(dst);
362 }
363
364 /* Phase 2: Parse the file [every time]
365 * The data is in a very compressed form at this point. We also
366 * know some important things about the data:
367 * - There is no whitespace (except in strings)
368 * - the braces are all closed
369 * - comments have been removed.
370 * - strings are valid.
371 */
372 enum
373 {
374 STATE_NONE,
375 STATE_NAME,
376 STATE_IS,
377 STATE_VALUE,
378 STATE_QUOTE,
379 STATE_CHILD
380 }
381
382 state = STATE_NONE;
383
384 char* cur_name;
385 char* cur_value;
386
387 int child_depth = 0;
388
389 size_t name_start = 0;
390 size_t name_len = 0;
391
392 size_t value_start = 0;
393 size_t value_len = 0;
394
395 size_t child_start = 0;
396 size_t child_len = 0;
397
398 pParent = parent;
399 iNumChildren = 0;
400 iNumValues = 0;
401 myName = name;
402 myValue = value;
403 myDevice = nullptr;
404 myClassId = c_none;
405 myFlags = 0;
406
407 for (size_t curtext = 0; curtext < textlen; curtext++)
408 {
409 switch (state)
410 {
411 case STATE_NONE:
412 if (isalnum((unsigned char)text[curtext]) || text[curtext] == '.'
413 || text[curtext] == '_')
414 {
415 name_start = curtext;
416 state = STATE_NAME;
417 }
418 break;
419
420 case STATE_NAME:
421 if (text[curtext] == '=')
422 {
423 state = STATE_IS;
424 name_len = curtext - name_start;
425 }
426 break;
427
428 case STATE_IS:
429 if (isalnum((unsigned char)text[curtext]) || text[curtext] == '.'
430 || text[curtext] == '_')
431 {
432 value_start = curtext;
433 value_len = 1;
434 state = STATE_VALUE;
435 }
436
437 if (text[curtext] == '\"')
438 {
439 value_start = curtext;
440 state = STATE_QUOTE;
441
442 //curtext--;
443 }
444 break;
445
446 case STATE_VALUE:
447 if (text[curtext] == ';')
448 {
449 value_len = curtext - value_start;
450 cur_name = (char*)malloc(name_len + 1);
451 memcpy(cur_name, &text[name_start], name_len);
452 cur_name[name_len] = '\0';
453 cur_value = (char*)malloc(value_len + 1);
454 memcpy(cur_value, &text[value_start], value_len);
455 cur_value[value_len] = '\0';
456
457 // printf("Calling strip_string for <%s>. \n",cur_value);
458 strip_string(cur_value);
459 add_value(cur_name, cur_value);
460
461 state = STATE_NONE;
462 }
463 else if (text[curtext] == '{')
464 {
465 value_len = curtext - value_start;
466 state = STATE_CHILD;
467 child_start = curtext + 1;
468 child_depth = 1;
469 }
470 break;
471
472 case STATE_QUOTE:
473 if ((text[curtext] == '\"') && (text[curtext + 1] == '\"'))
474 {
475 curtext++;
476 }
477 else if (text[curtext] == '\"')
478 {
479 state = STATE_VALUE;
480 }
481 break;
482
483 case STATE_CHILD:
484 if (text[curtext] == '{')
485 child_depth++;
486 else if (text[curtext] == '}')
487 {
488 child_depth--;
489 if (!child_depth)
490 {
491 cur_name = (char*)malloc(name_len + 1);
492 memcpy(cur_name, &text[name_start], name_len);
493 cur_name[name_len] = '\0';
494 cur_value = (char*)malloc(value_len + 1);
495 memcpy(cur_value, &text[value_start], value_len);
496 cur_value[value_len] = '\0';
497 child_len = curtext - child_start;
498 state = STATE_NONE;
499
500 strip_string(cur_value);
501
502 pChildren[iNumChildren++] = new CConfigurator(this, cur_name,
503 cur_value,
504 &text[child_start],
505 child_len);
506 }
507 }
508 }
509 }
510
511 if (!parent && initialize_root) initialize();
512}

551char* CConfigurator::strip_string(char* c)
552{
553 char* pos = c;
554 char* org = c + 1;
555 bool end_it = false;
556
557 if (c[0] == '\"')
558 {
559 while (!end_it)
560 {
561 if (*org == '\"')
562 {
563 org++;
564 if (*org == '\"')
565 *pos++ = *org++;
566 else
567 end_it = true;
568 }
569 else
570 *pos++ = *org++;
571 }
572
573 *pos = '\0';
574 }
575
576 return c;
577}

626 {

629 case 'y':

655u64 CConfigurator::get_num_value(const char* n, bool decimal, u64 def)
656{
657 int i;
658 u64 multiplier = decimal ? 1000 : 1024;
659 for (i = 0; i < iNumValues; i++)
660 {
661 if (!strcasecmp(pValues[i].name, n))
662 {
663 u64 retval = 0;
664 u64 partval = 0;
665 char* val = pValues[i].value;
666 int j = 0;
667 for (;;)
668 {
669 while (val[j] && strchr("0123456789", val[j]))
670 {
671 partval *= 10;
672 partval += val[j] - '0';
673 j++;
674 }
675
676 switch (val[j])
677 {
678 case 't':
679 case 'T':
680 partval *= multiplier;
681 [[fallthrough]];
682 case 'g':
683 case 'G':
684 partval *= multiplier;
685 [[fallthrough]];
686 case 'm':
687 case 'M':
688 partval *= multiplier;
689 [[fallthrough]];
690 case 'k':
691 case 'K':
692 retval += partval * multiplier;
693 partval = 0;
694 j++;
695 break;
696
697 case '\0':
698 retval += partval;
699 return retval;
700
701 default:
702 FAILURE_2(Configuration, "Illegal numeric value (%s) for %s",
703 pValues[i].value, n);
704 }
705 }
706 }
707 }
708
709 return def;
710}

839const char* CConfigurator::get_gui_name() const
840{
841#if defined(_WIN32) && !defined(CONFIGURATION_ONLY)
842 // Audio follows the active GUI.
843 const char* requested_gui = getenv("ES40_GUI");
844 if (requested_gui && !strcasecmp(requested_gui, "win32")) return "win32";
845#endif
846 const CConfigurator* root = this;
847 while (root->pParent) root = root->pParent;
848 for (int i = 0; i < root->iNumChildren; ++i) {
849 const CConfigurator* child = root->pChildren[i];
850 for (const auto& entry : classes)
851 if ((entry.flags & IS_GUI) && child->myValue &&
852 !strcasecmp(child->myValue, entry.name)) return entry.name;
853 }
854 return nullptr;
855}

861void CConfigurator::initialize()
862{
863 if (!pParent)
864 {
865#if defined(_WIN32) && !defined(CONFIGURATION_ONLY)
866 // Serial-only children still need a shutdown endpoint.
867 const char* managed_gui = getenv("ES40_GUI");
868 if (managed_gui && !strcasecmp(managed_gui, "win32"))
869 {
870 bool has_gui = false;
871 for (int i = 0; i < iNumChildren; ++i)
872 {
873 const char* value = pChildren[i]->get_myValue();
874 if (!strcasecmp(value, "sdl") || !strcasecmp(value, "win32") || !strcasecmp(value, "X11"))
875 has_gui = true;
876 }
877 if (!has_gui)
878 {
879 if (iNumChildren >= CFG_MAX_CHILDREN)
