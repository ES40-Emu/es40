39// Decoded hardware load, outside saved CPU state. Virtual accesses translate the
40// original address before rounding down the physical read; longwords sign-extend.
41struct AlphaHWLoadDescriptor
42{
43 AlphaHWLoadAddress address = AlphaHWLoadAddress::Invalid;
44 std::uint32_t size_bits = 0;
45 std::int32_t displacement = 0;
46 bool vpte = false;
47 bool alternate = false;
48 bool write_check = false;
49 bool locked = false;
50 bool compilable = false;
51 bool discard_safe = false; // Compiled physical loads may omit a discarded read.
52};
53
54constexpr bool alpha_hw_load_valid(const AlphaHWLoadDescriptor& load)
55{
56 return (load.address == AlphaHWLoadAddress::Physical
57 || load.address == AlphaHWLoadAddress::Virtual)
58 && (load.size_bits == 32 || load.size_bits == 64);
59}
60
61constexpr bool alpha_hw_load_compilable(const AlphaHWLoadDescriptor& load)
62{
63 return alpha_hw_load_valid(load) && load.compilable && !load.locked;
64}
65
66// Shared virtual-load helper ABI; these bits are independent of guest encoding.
67constexpr std::uint32_t alpha_hw_load_helper_descriptor(const AlphaHWLoadDescriptor& load)
68{
69 return load.size_bits | (load.vpte ? 0x100u : 0)
70 | (load.alternate ? 0x200u : 0) | (load.write_check ? 0x400u : 0);
71}

73struct AlphaHWLoadPolicy

78constexpr AlphaHWLoadDescriptor alpha_decode_hw_load(const AlphaHWLoadPolicy& policy,
79 std::uint32_t instruction)
80{
81 return policy.decode ? policy.decode(instruction) : AlphaHWLoadDescriptor{};
