32// One contiguous family-approved instruction span, beginning at the queried PC.
33// Bytes are little endian and need not be host-aligned.
34struct AlphaInstructionSourceView
35{
36 const std::uint8_t* bytes = nullptr;
37 std::uint64_t identity = 0; // Stable start-source key; EV68 uses its physical PC.
38 std::uint32_t word_count = 0;
39 bool asm_global = false;
40
41 bool contains(std::uint32_t count) const
42 {
43 return bytes && count != 0 && count <= word_count;
44 }
45
46 std::uint32_t word(std::uint32_t index) const
47 {
48 assert(bytes && index < word_count);
49 const auto* p = bytes + static_cast<std::size_t>(index) * 4;
50 return std::uint32_t(p[0]) | (std::uint32_t(p[1]) << 8)
51 | (std::uint32_t(p[2]) << 16) | (std::uint32_t(p[3]) << 24);
52 }
53
54 bool copy_words(std::uint32_t* out, std::uint32_t count) const
55 {
56 if (!out || !contains(count)) return false;
57 for (std::uint32_t i = 0; i < count; ++i) out[i] = word(i);
58 return true;
59 }
60};

62// A bound reader may be retained; its context must outlive the JIT using it.
63struct AlphaInstructionSource
64{
65 void* context = nullptr;
66 AlphaInstructionSourceView (*query)(void*, std::uint64_t guest_pc) = nullptr;
67
68 AlphaInstructionSourceView view(std::uint64_t pc) const
69 {
70 return query ? query(context, pc) : AlphaInstructionSourceView{};
71 }
72
73 // Copy a candidate word before another query or instruction can change the view.
74 // A failed observation leaves the caller's word unchanged.
75 bool read_word(std::uint64_t pc, std::uint64_t identity, bool asm_global,
76 std::uint32_t index, std::uint32_t& word) const
77 {
78 const auto current = view(pc);
79 if (current.identity != identity || current.asm_global != asm_global
80 || !current.contains(1) || index >= current.word_count) return false;
81 word = current.word(index);
82 return true;
83 }
84
85 bool matches(std::uint64_t pc, std::uint64_t identity, bool asm_global,
86 const std::uint32_t* words, std::uint32_t count) const
87 {
88 const auto current = view(pc);
89 if (!words || current.identity != identity || current.asm_global != asm_global
90 || !current.contains(count)) return false;
91 for (std::uint32_t i = 0; i < count; ++i)
92 if (current.word(i) != words[i]) return false;
93 return true;
94 }
95};
