33void Ev68InstructionFetch::prepare_dispatch(CAlphaCPU& cpu)
34{
35 // Preserve the ES40 firmware entry behavior after an in-place image rewrite.
36 if (cpu.state.pc == (cpu.state.pal_base | 1))
37 cpu.flush_icache();
38}

40AlphaInstructionMapping Ev68InstructionFetch::probe(CAlphaCPU& cpu, u64 pc)
41{
42 AlphaInstructionMapping mapping;
43 if (pc & 1)
44 {
45 // PAL instruction addresses are physical; retain the dispatcher's alignment.
46 mapping.physical = pc & ~U64(1);
47 mapping.asm_global = true;
48 }
49 else
50 {
51 const int line = (int)(((pc & ~U64(3)) >> 11) & (ICACHE_ENTRIES - 1));
52 if (cpu.icache_hit(line, pc))
53 {
54 mapping.physical = cpu.state.icache[line].p_address + (pc & ICACHE_BYTE_MASK);
55 mapping.asm_global = cpu.state.icache[line].asm_bit;
56 }
57 else if (cpu.virt2phys(pc, &mapping.physical, ACCESS_EXEC | FAKE,
58 &mapping.asm_global, 0) != 0)
59 return mapping;
60 }
61 mapping.valid = true;
62 return mapping;
63}

65bool Ev68InstructionFetch::peek_cached(const CAlphaCPU& cpu, u32* instruction)
66{
67 if (cpu.state.pc == cpu.seq_next_pc && cpu.seq_remaining > 0 && cpu.seq_cm == cpu.state.cm)
68 {
69 *instruction = endian_32(cpu.seq_line_ptr[cpu.seq_offset]);
70 return true;
71 }
72 if (cpu.icache_enabled)
73 {
74 const int line = (int)((cpu.state.pc >> 11) & (ICACHE_ENTRIES - 1));
75 if (cpu.icache_hit(line, cpu.state.pc))
76 {
77 *instruction = endian_32(cpu.state.icache[line].data[(cpu.state.pc >> 2) & ICACHE_INDEX_MASK]);
78 return true;
79 }
80 }
81 return false;
82}

84bool Ev68InstructionFetch::matches_interpreted(const CAlphaCPU& cpu, u64 start_pc, u32 count)
85{
86 if (cpu.icache_enabled)
87 {
88 const u64 vs = start_pc & ~U64(3);
89 const u64 ve = vs + 4 * (u64)count;
90 for (u64 v = vs & ~U64(0x7ff); v < ve; v += 0x800)
91 {
92 const int li = (int)((v >> 11) & (ICACHE_ENTRIES - 1));
93 const auto& line = cpu.state.icache[li];
94 // Compare cached bytes even if the current mode has no cached permission.
95 // This deliberately differs from the execution/peek hit predicate.
96 if (line.valid && (line.asn == cpu.state.asn || line.asm_bit)
97 && line.address == ((v | (start_pc & 1)) & ICACHE_MATCH_MASK))
98 {
99 const u64 lo = (v > vs) ? v : vs;
100 const u64 hi = (v + 0x800 < ve) ? v + 0x800 : ve;
101 const u64 poff = line.p_address + (lo - v);
102 if (poff + (hi - lo) <= cpu.dram_size
103 && memcmp((const uint8_t*)cpu.dram_ptr + poff,
104 (const uint8_t*)line.data + (lo - v), (size_t)(hi - lo)))
105 return false;
106 }
107 }
108 }
109 return true;
110}

118AlphaInstructionSourceView Ev68InstructionFetch::source(CAlphaCPU& cpu, u64 pc)
119{
120 const auto mapping = probe(cpu, pc);
121 if (!mapping.valid || !cpu.dram_ptr || (mapping.physical & 3)
122 || mapping.physical >= cpu.dram_size)
123 return {};
124
125 // The EV68 JIT uses backing RAM so its interpreted-span check separately rejects
126 // cached insn that differ. Never cross the current 8 KiB translation.
127 const u64 ram_words = (cpu.dram_size - mapping.physical) / sizeof(u32);
128 const u64 page_words = (U64(0x2000) - (mapping.physical & U64(0x1fff))) / sizeof(u32);
129 const u32 count = (u32)((ram_words < page_words) ? ram_words : page_words);
130 if (!count) return {};
131 return {reinterpret_cast<const u8*>(cpu.dram_ptr) + mapping.physical,
132 mapping.physical, count, mapping.asm_global};
133}

135int Ev68InstructionFetch::fetch(CAlphaCPU& cpu, u32* instruction)
136{
137 if (cpu.state.pc == cpu.seq_next_pc && cpu.seq_remaining > 0 && cpu.seq_cm == cpu.state.cm)
138 {
139 *instruction = endian_32(cpu.seq_line_ptr[cpu.seq_offset]);
140#if defined(DEBUG_ARC) && !defined(ES40_JIT)
141 if (cpu.state.current_pc < 0x10000)
142 printf("PC=%016" PRIx64 " ins=%08x\n", cpu.state.current_pc, *instruction);
143#endif
144 cpu.seq_offset++;
145 cpu.seq_remaining--;
146 cpu.seq_next_pc += 4;
147 }
148 else
149 {
150 prepare_dispatch(cpu);
151 const int result = read_cache(cpu, cpu.state.pc, instruction);
152 if (result)
153 return result;
154
155 if (!(cpu.state.pc & 1) && cpu.icache_hit(cpu.state.last_found_icache, cpu.state.pc))
156 {
157 const int line = cpu.state.last_found_icache;
158 cpu.seq_line_ptr = cpu.state.icache[line].data;
159 cpu.seq_offset = (int)((cpu.state.pc >> 2) & ICACHE_INDEX_MASK) + 1;
160 cpu.seq_remaining = ICACHE_LINE_SIZE - cpu.seq_offset;
161 cpu.seq_next_pc = (cpu.state.pc & ~U64(0x3)) + 4;
162 cpu.seq_cm = cpu.state.cm;
163 }
164 else
165 cpu.seq_remaining = 0;
166 }
167
168#if defined(IDB)
169 cpu.current_pc_physical = cpu.state.pc_phys;
170#endif
171 return 0;
172}

194int Ev68InstructionFetch::read_cache(CAlphaCPU& cpu, u64 address, u32* data)
195{
196 // Direct-map the icache: 2 KiB lines (ICACHE_LINE_SIZE * 4 == 2048 bytes).
197 // Use VA[...:11] as the set index. The PAL bit (VA<0>) remains part of the tag.
198 const u64 kLineShift = 11;
199 const u64 v_aligned = address & ~U64(0x3);
200 const int i = (int)((v_aligned >> kLineShift) & (ICACHE_ENTRIES - 1));
201 u64 v_a;
202 u64 p_a;
203 int result;
204 bool asm_bit;
205
206 if (cpu.icache_enabled)
207 {
208 bool hit = cpu.icache_line_matches(i, address);
209 if (hit && !(address & 1) && !(cpu.icache_valid_modes[i] & (1u << cpu.state.cm))) {
210 // A cached instruction is still subject to the current mode's access
211 // checks, including for ASM lines. Keep its bytes until IMB when the
212 // translation is unchanged; SRM can depend on the cached instruction.
213 result = cpu.virt2phys(address, &p_a, ACCESS_EXEC, &asm_bit, 0);
214 if (result) return result;
215 hit = p_a == cpu.state.icache[i].p_address + (address & ICACHE_BYTE_MASK);
216 if (hit) {
217 if (asm_bit != cpu.state.icache[i].asm_bit) {
218 // A newly checked process-specific mapping must not grant access
219 // to other ASNs through an older global cache line (or vice versa).
220 cpu.icache_valid_modes[i] = 0;
221 cpu.state.icache[i].asn = cpu.state.asn;
222 cpu.state.icache[i].asm_bit = asm_bit;
223 }
224 cpu.icache_valid_modes[i] |= (u8)(1u << cpu.state.cm);
225 }
226 }
227
228 // ---- Fast hit probe
229 if (hit) {
230
231 *data = endian_32(cpu.state.icache[i].data[(address >> 2) & ICACHE_INDEX_MASK]);
232
233 // keep debug/pc_phys coherent even when icache is enabled
234 cpu.state.pc_phys = cpu.state.icache[i].p_address + (address & ICACHE_BYTE_MASK);
235
236#ifdef IDB
237 cpu.current_pc_physical = cpu.state.pc_phys;
238#endif
239 cpu.state.last_found_icache = i;
240 return 0;
241 }
242
243 // ---- Miss: translate + fill
244 v_a = address & ICACHE_MATCH_MASK;
245 if (address & 1) {
246 // PALmode: VA<0> is PAL marker; physical is VA with bit0 cleared
247 p_a = v_a & ~U64(0x1);
248 asm_bit = true;
249 }
250 else {
251 result = cpu.virt2phys(v_a, &p_a, ACCESS_EXEC, &asm_bit, 0);
252 if (result) return result;
253 }
254
255 // Attempt to get a pointer into DRAM. If this is PIO (e.g., TIG flash),
256 // PtrToMem returns null and we must *not* try to memcpy from it.
257 char* mem = cpu.cSystem->PtrToMem(p_a);
258 if (mem && cpu.cSystem->PtrToMem(p_a + ((ICACHE_LINE_SIZE * 4) - 1))) {
259 // DRAM-backed: fill the direct-mapped icache line.
260 memcpy(cpu.state.icache[i].data, mem, ICACHE_LINE_SIZE * 4);
261 cpu.state.icache[i].valid = true;
262 cpu.state.icache[i].asn = cpu.state.asn;
263 cpu.state.icache[i].asm_bit = asm_bit;
264 cpu.state.icache[i].address = address & ICACHE_MATCH_MASK;
265 cpu.state.icache[i].p_address = p_a;
266 cpu.icache_valid_modes[i] = (u8)(1u << cpu.state.cm);
267
268 *data = endian_32(cpu.state.icache[i].data[(address >> 2) & ICACHE_INDEX_MASK]);
269
270 // same pc_phys update on fill
271 cpu.state.pc_phys = p_a + (address & ICACHE_BYTE_MASK);
272#ifdef IDB
273 cpu.current_pc_physical = cpu.state.pc_phys;
274#endif
275 cpu.state.last_found_icache = i;
276 return 0;
277 }
278 else {
279 // PIO/TIG-backed: cannot fill icache lines.
280 // Read exactly the requested instruction as 4 byte reads via the system bus.
281 const u64 p_instr = p_a + (address & ICACHE_BYTE_MASK);
282 u32 ins = 0;
283 ins |= (u8)cpu.cSystem->ReadMem(p_instr + 0, 8, &cpu);
284 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 1, 8, &cpu)) << 8;
285 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 2, 8, &cpu)) << 16;
286 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 3, 8, &cpu)) << 24;
287 *data = ins; // already in target little-endian form
288
289 cpu.state.pc_phys = p_instr;
290#ifdef IDB
291 cpu.current_pc_physical = cpu.state.pc_phys;
292#endif
293 return 0;
294 }
295 }
296
297 // ---- Icache disabled (unchanged)
298 if (address & 1) {
299 cpu.state.pc_phys = address & ~U64(0x3);
300 cpu.state.rem_ins_in_page = 1;
301 }
302 else {
303 if (!cpu.state.rem_ins_in_page || cpu.ins_page_cm != cpu.state.cm) {
304 result = cpu.virt2phys(address, &cpu.state.pc_phys, ACCESS_EXEC, &asm_bit, 0);
305 if (result) return result;
306 cpu.state.rem_ins_in_page = 2048 - ((((u32)address) >> 2) & 2047);
307 cpu.ins_page_cm = cpu.state.cm;
308 }
309 }
310
311 *data = (u32)cpu.cSystem->ReadMem(cpu.state.pc_phys, 32, &cpu);
312 return 0;
313}
