32AlphaMMUAccess Ev68MMU::access(const CAlphaCPU& cpu, int flags)
33{
34 const auto& state = cpu.state;
35 // HRM 6.4.1: VPTE fetches check kernel permissions; /alt uses DTB_ALT_MODE.
36 // VPTE takes precedence over ALT. Neither changes the selected lookup ASN.
37 return {
38 (flags & VPTE) ? 0 : (flags & ALT) ? state.alt_cm : state.cm,
39 (flags & ACCESS_EXEC) ? state.asn : state.asn0
40 };
41}

43AlphaMMUAddress Ev68MMU::map_address(const CAlphaCPU& cpu, std::uint64_t virt,
44 int flags, int mode)
45{
46 const auto& state = cpu.state;
47 const bool exec = (flags & ACCESS_EXEC) != 0;
48 const bool va48 = (exec ? state.i_ctl_va_mode : state.va_ctl_va_mode) & 1;
49 const u64 canonical = va48 ? sext_u64_48(virt)
50 : (virt & U64(0x0000040000000000)) ? (virt | U64(0xfffff80000000000))
51 : (virt & U64(0x000007ffffffffff));
52 if (canonical != virt)
53 return {AlphaMMUAddressKind::Invalid, 0};
54
55 const int spe = exec ? state.i_ctl_spe : state.m_ctl_spe;
56 u64 phys;
57 // HRM 5.3.9: SPE[2] maps VA[43:0], ignoring VA[45:44].
58 if ((spe & 4) && (virt & U64(0x0000c00000000000)) == U64(0x0000800000000000))
59 phys = virt & U64(0x00000fffffffffff);
60 // SPE[1] maps VA[40:0], sign-extending bit 40 through PA[43:41].
61 else if ((spe & 2) && (virt & U64(0x0000fe0000000000)) == U64(0x0000fc0000000000))
62 phys = (virt & U64(0x000001ffffffffff))
63 | ((virt & U64(0x0000010000000000)) ? U64(0x00000e0000000000) : 0);
64 // SPE[0] maps VA[29:0], clearing PA[43:30].
65 else if ((spe & 1) && (virt & U64(0x0000ffffc0000000)) == U64(0x0000ffff80000000))
66 phys = virt & U64(0x000000003fffffff);
67 else
68 return {AlphaMMUAddressKind::TranslationBuffer, 0};
69
70 // Superpages are kernel-only even for accesses carrying NO_CHECK.
71 return mode ? AlphaMMUAddress{AlphaMMUAddressKind::AccessViolation, 0}
72 : AlphaMMUAddress{AlphaMMUAddressKind::Direct, phys};
73}

79std::uint64_t Ev68MMU::va_form(const CAlphaCPU& cpu, std::uint64_t address, bool bIBOX)
80{
81 const auto& state = cpu.state;
82 switch (bIBOX ? state.i_ctl_va_mode : state.va_ctl_va_mode)
83 {
84 case 0:
85 return((bIBOX ? state.i_ctl_vptb : state.va_ctl_vptb) & U64(0xfffffffe00000000)) | ((address >> 10) & U64(0x00000001fffffff8));
86
87 case 1:
88 return
89 ((bIBOX ? state.i_ctl_vptb : state.va_ctl_vptb) & U64(0xfffff80000000000)) | ((address >> 10) & U64(0x0000003ffffffff8)) |
90 (((address >> 10) & U64(0x0000002000000000)) * U64(0x3e));
91
92 case 2:
93 return((bIBOX ? state.i_ctl_vptb : state.va_ctl_vptb) & U64(0xffffffffc0000000)) | ((address >> 10) & U64(0x00000000003ffff8));
94 }
95
96 return 0;
97}

99AlphaTBFill Ev68MMU::decode_fill(const CAlphaCPU&, std::uint64_t virt,
100 std::uint64_t pte_phys, std::uint64_t pte_flags,
101 int)
102{
103 // EV68CB/EV68DC HRM 5.3.1/5.3.2: VA[47:13], PA[43:13], GH[1:0].
104 // GH widens the granule by three bits per step (13/16/19/22 offset bits).
105 static constexpr std::uint64_t match_masks[4] = {
106 kEv68TBFillPolicy.base_page_match_mask, U64(0x0000ffffffff0000),
107 U64(0x0000fffffff80000), U64(0x0000ffffffc00000)
108 };
109 static constexpr std::uint64_t phys_masks[4] = {
110 U64(0x00000fffffffe000), U64(0x00000fffffff0000),
111 U64(0x00000ffffff80000), U64(0x00000fffffc00000)
112 };
113 static constexpr std::uint64_t keep_masks[4] = {
114 U64(0x0000000000001fff), U64(0x000000000000ffff),
115 U64(0x000000000007ffff), U64(0x00000000003fffff)
116 };
117 const unsigned gh = (unsigned)((pte_flags >> 5) & 3);
118
119 AlphaTBFill fill{};
120 fill.virt = virt;
121 fill.phys = pte_phys & phys_masks[gh];
122 fill.match_mask = match_masks[gh];
123 fill.keep_mask = keep_masks[gh];
124 fill.fault[0] = (int)pte_flags & 2;
125 fill.fault[1] = (int)pte_flags & 4;
126 fill.fault[2] = (int)pte_flags & 8;
127 fill.access[0][0] = (int)pte_flags & 0x100;
128 fill.access[1][0] = (int)pte_flags & 0x1000;
129 fill.access[0][1] = (int)pte_flags & 0x200;
130 fill.access[1][1] = (int)pte_flags & 0x2000;
131 fill.access[0][2] = (int)pte_flags & 0x400;
132 fill.access[1][2] = (int)pte_flags & 0x4000;
133 fill.access[0][3] = (int)pte_flags & 0x800;
134 fill.access[1][3] = (int)pte_flags & 0x8000;
135 fill.asm_bit = (int)pte_flags & 0x10;
136 return fill;
137}

203int Ev68MMU::tb_miss(CAlphaCPU& cpu, std::uint64_t virt, int flags, std::uint32_t ins)
204{
205 auto& state = cpu.state;
206 if (!state.pal_vms) // unknown PALcode
207 {
208
209 // transfer execution to PALcode
210 state.exc_addr = state.current_pc;
211 if (flags & VPTE)
212 {
213 // HRM 5.1.3: VA is NOT written for LD_VPTE misses
214 state.va_form_va = virt;
215 state.exc_sum = (u64)I_GETRA(ins) << 8; /* HRM 5.2.13: EXC_SUM REG[12:8] is 5 bits */
216 /*
217 * I_CTL[VA_48] selects the DTB double-miss PAL entry.
218 * state.i_ctl_va_mode packs bits [16:15] of I_CTL, so bit 0
219 * here is the architectural VA_48 bit.
220 */
221#ifdef JIT_STATS
222 cpu.m_stat_tb[1]++;
223#endif
224 const auto reason = (state.i_ctl_va_mode & 1)
225 ? AlphaPALException::DataTlbDoubleMiss48 : AlphaPALException::DataTlbDoubleMiss43;
226 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(reason) + 1);
227 }
228 else if (flags & ACCESS_EXEC)
229 {
230#ifdef JIT_STATS
231 cpu.m_stat_tb[2]++;
232#endif
233 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(AlphaPALException::InstructionTlbMiss) + 1);
234 }
235 else
236 {
237 state.fault_va = virt;
238 state.va_form_va = virt;
239 state.exc_sum = (u64)I_GETRA(ins) << 8; /* HRM 5.2.13: EXC_SUM REG[12:8] is 5 bits */
240
241 const u32 opcode = I_GETOP(ins);
242 state.mm_stat = ((opcode == 0x1b || opcode == 0x1f) ? opcode - 0x18 : opcode) << 4
243 | (flags & ACCESS_WRITE);
244#ifdef JIT_STATS
245 cpu.m_stat_tb[0]++;
246 {
247 const int t = TB_INDEX_DATA;
248 const int h = (int)cpu.m_tb_idx[t][cpu.tb_idx_slot(virt)] - 1;
249 if (h >= 0) cpu.m_stat_tb[state.tb[t][h].valid ? 9 : 8]++;
250 const u64 pg = virt >> 13; const u32 sl = (u32)(pg * 0x9E3779B1u) >> 25;
251 if (cpu.m_stat_va[sl][0] != pg) { cpu.m_stat_va[sl][0] = pg; cpu.m_stat_va[sl][1] = 0; }
252 cpu.m_stat_va[sl][1]++;
253 }
254#endif
255 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(AlphaPALException::DataTlbMiss) + 1);
256 }
257
258 return -1;
259 }
260 else // VMS PALcode
261 {
262 if (flags & RECUR) // we already tried this
263 {
264 printf("Translationbuffer RECUR lookup failed!\n");
265 return -1;
266 }
267
268 state.exc_addr = state.current_pc;
269 if (flags & VPTE)
270 {
271
272 // try to handle the double miss. If this needs to transfer control
273 // to the OS, it will return non-zero value.
274 return cpu.vmspal_ent_dtbm_double_3(flags);
275 }
276 else if (flags & ACCESS_EXEC)
277 {
278
279 // try to handle the ITB miss. If this needs to transfer control
280 // to the OS, it will return non-zero value.
281 return cpu.vmspal_ent_itbm(flags);
282 }
283 else
284 {
285 state.fault_va = virt;
286 state.exc_sum = (u64)I_GETRA(ins) << 8; /* HRM 5.2.13: EXC_SUM REG[12:8] is 5 bits */
287
288 const u32 opcode = I_GETOP(ins);
289 state.mm_stat = ((opcode == 0x1b || opcode == 0x1f) ? opcode - 0x18 : opcode) << 4
290 | (flags & ACCESS_WRITE);
291
292 // try to handle the single miss. If this needs to transfer control
293 // to the OS, it will return non-zero value.
294 return cpu.vmspal_ent_dtbm_single(flags);
295 }
296 }
297}

299int Ev68MMU::instruction_fault(CAlphaCPU& cpu, std::uint64_t, int flags,
300 std::uint32_t, AlphaInstructionFaultSource)
301{
302 auto& state = cpu.state;
303 state.exc_addr = state.current_pc;
304 state.exc_sum = 0;
305 if (state.pal_vms)
306 return cpu.vmspal_ent_iacv(flags);
307
308 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(AlphaPALException::InstructionAccessViolation) + 1);
309 return -1;
310}

312int Ev68MMU::access_violation(CAlphaCPU& cpu, std::uint64_t virt, int flags,
313 std::uint32_t ins, [[maybe_unused]] AlphaDataFaultSource source)
314{
315 auto& state = cpu.state;
316 state.exc_addr = state.current_pc;
317 state.fault_va = virt;
318 /* HRM 5.1.5/D.24: VA_FORM is derived from VA, which is written
319 on every D-stream fault (incl. DFAULT), not just TB miss.
320 Keep the VA_FORM snapshot current so the OS fault handler
321 computes the right PTE. */
322 state.va_form_va = virt;
323 state.exc_sum = (u64)I_GETRA(ins) << 8; /* HRM 5.2.13: EXC_SUM REG[12:8] is 5 bits */
324
325 const u32 opcode = I_GETOP(ins);
326 state.mm_stat = ((opcode == 0x1b || opcode == 0x1f) ? opcode - 0x18 : opcode) << 4
327 | (flags & ACCESS_WRITE) | 2;
328 if (state.pal_vms)
329 return cpu.vmspal_ent_dfault(flags);
330
331#ifdef JIT_STATS
332 if (source == AlphaDataFaultSource::TranslationBuffer)
333 cpu.m_stat_tb[3]++;
334#endif
335 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(AlphaPALException::DataFault) + 1);
336 return -1;
337}

339int Ev68MMU::access_fault(CAlphaCPU& cpu, std::uint64_t virt, int flags,
340 std::uint32_t ins, bool fault_on_read, bool fault_on_write)
341{
342 auto& state = cpu.state;
343 state.exc_addr = state.current_pc;
344 state.fault_va = virt;
345 /* HRM 5.1.5/D.24: VA_FORM is derived from VA, which is written
346 on every D-stream fault (including FOR/FOW DFAULT), not just TB
347 miss. Keep the VA_FORM snapshot current so the OS fault
348 handler computes the right PTE address. */
349 state.va_form_va = virt;
350 state.exc_sum = (u64)I_GETRA(ins) << 8; /* HRM 5.2.13: EXC_SUM REG[12:8] is 5 bits */
351
352 /* HRM 5.3.8 MM_STAT: FOR [bit 2] is set when a fault-on-read
353 * error occurs during a read transaction with PTE[FOR] set.
354 * FOW [bit 3] is set when a fault-on-write error occurs.
355 * For HW_LD WrChk variants the chip is performing both a
356 * read AND a write-protection check, so PTE[FOW] (if set)
357 * is reportable too
358 */
359 const u32 opcode = I_GETOP(ins);
360 const int for_bit = !(flags & ACCESS_WRITE) && fault_on_read ? 4 : 0;
361 const int fow_bit = ((flags & ACCESS_WRITE) || (flags & WRCHK)) && fault_on_write ? 8 : 0;
362 state.mm_stat = ((opcode == 0x1b || opcode == 0x1f) ? opcode - 0x18 : opcode) << 4
363 | (flags & ACCESS_WRITE) | for_bit | fow_bit;
364 if (state.pal_vms)
365 return cpu.vmspal_ent_dfault(flags);
366
367#ifdef JIT_STATS
368 cpu.m_stat_tb[3]++;
369#endif
370 cpu.set_pc(state.pal_base + cpu.pal_exception_offset(AlphaPALException::DataFault) + 1);
371 return -1;
372}

388void Ev68MMU::unaligned(CAlphaCPU& cpu, std::uint64_t virt, int flags,
389 std::uint32_t ins, [[maybe_unused]] int align)
390{
391 auto& state = cpu.state;
392 state.fault_va = virt;
393 state.va_form_va = virt;
394 state.exc_sum = (u64)I_GETRA(ins) << 8;
395 /* HRM 5.3.8: pure misalignment is not an access violation (ACV). */
396 state.mm_stat = (I_GETOP(ins) << 4) | ((flags & ACCESS_WRITE) ? 1 : 0);
397 TRACE_UNALIGN(flags, align);
398 cpu.enter_unaligned_exception();
399}
