31// Shared control contains no disk pointer, so a selector remains safe during
32// emulator teardown. The emulation thread observes the flag at I/O boundaries.
33class CDiskFaultControl
34{
35public:
36 CDiskFaultControl(const std::string& label, bool initial_offline)
37 : device_label(label), offline(initial_offline), active(true) {}
38
39 const std::string& label() const { return device_label; }
40 bool is_offline() const noexcept { return offline.load(std::memory_order_acquire); }
41 bool is_active() const noexcept { return active.load(std::memory_order_acquire); }
42 bool set_offline(bool value) noexcept
43 {
44 if (!is_active()) return false;
45 offline.store(value, std::memory_order_release);
46 return true;
47 }
48 void deactivate() noexcept { active.store(false, std::memory_order_release); }
49
50private:
51 const std::string device_label;
52 std::atomic<bool> offline;
53 std::atomic<bool> active;
54};

68inline void register_disk_fault_control(const std::shared_ptr<CDiskFaultControl>& control)
69{
70 auto& registry = get_disk_fault_registry();
71 std::lock_guard<std::mutex> lock(registry.mutex);
72 registry.controls.push_back(control);

75inline std::vector<std::shared_ptr<CDiskFaultControl>> disk_fault_controls()
76{
77 auto& registry = get_disk_fault_registry();
78 std::lock_guard<std::mutex> lock(registry.mutex);
79 std::vector<std::shared_ptr<CDiskFaultControl>> result;
80 for (auto it = registry.controls.begin(); it != registry.controls.end();)
81 {
82 auto control = it->lock();
83 if (control && control->is_active())
84 {
85 result.push_back(std::move(control));
86 ++it;
87 }
88 else
89 it = registry.controls.erase(it);
90 }
91 return result;
92}
