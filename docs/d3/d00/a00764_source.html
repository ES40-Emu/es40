102{
103 const std::string& text = string_value(value);

113void unicode_escape(std::string& out, unsigned cp)
114{
115 if (cp > 0xffff) {
116 unicode_escape(out, 0xd800 + ((cp - 0x10000) >> 10));
117 unicode_escape(out, 0xdc00 + ((cp - 0x10000) & 0x3ff));
118 return;
119 }
120 char escaped[7];
121 std::snprintf(escaped, sizeof escaped, "\\u%04x", cp);
122 out += escaped;
123}

125// Accept QMP quoting and normalize UTF-8 to ASCII without changing numeric IDs.
126std::string normalize(const std::string& input)
127{
128 std::string out;
129 char quote = 0;
130 for (size_t i = 0; i < input.size(); ++i) {
131 unsigned char c = (unsigned char)input[i];
132 if (quote) {
133 if (c == '\\') {
134 if (++i == input.size()) invalid("Invalid JSON escape");
135 if (input[i] == '\'') out += '\'';
136 else { out += '\\'; out += input[i]; }
137 } else if (c == (unsigned char)quote) {
138 quote = 0;
139 out += '"';
140 } else if (c == '"') {
141 out += "\\\"";
142 } else if (c < 0x20) {
143 invalid("Invalid JSON string");
144 } else if (c < 0x80) {
145 out += (char)c;
146 } else {
147 unsigned cp, minimum, remaining;
148 if (c >= 0xc2 && c <= 0xdf) { cp = c & 0x1f; minimum = 0x80; remaining = 1; }
149 else if (c >= 0xe0 && c <= 0xef) { cp = c & 0x0f; minimum = 0x800; remaining = 2; }
150 else if (c >= 0xf0 && c <= 0xf4) { cp = c & 0x07; minimum = 0x10000; remaining = 3; }
151 else { invalid("Invalid UTF-8"); return {}; }
152 while (remaining--) {
153 if (++i == input.size() || ((unsigned char)input[i] & 0xc0) != 0x80)
154 invalid("Invalid UTF-8");
155 cp = (cp << 6) | ((unsigned char)input[i] & 0x3f);
156 }
157 if (cp < minimum || cp > 0x10ffff || (cp >= 0xd800 && cp <= 0xdfff))
158 invalid("Invalid UTF-8");
159 unicode_escape(out, cp);
160 }
161 } else if (c == '"' || c == '\'') {
162 quote = (char)c;
163 out += '"';
164 } else if (c == '-' || digit((char)c)) {
165 const size_t begin = i;
166 if (c == '-') ++i;
167 if (i == input.size() || !digit(input[i])) invalid("Invalid JSON number");
168 if (input[i] == '0') ++i;
169 else while (i < input.size() && digit(input[i])) ++i;
170 if (i < input.size() && input[i] == '.') {
171 ++i;
172 if (i == input.size() || !digit(input[i])) invalid("Invalid JSON number");
173 while (i < input.size() && digit(input[i])) ++i;
174 }
175 if (i < input.size() && (input[i] == 'e' || input[i] == 'E')) {
176 ++i;
177 if (i < input.size() && (input[i] == '+' || input[i] == '-')) ++i;
178 if (i == input.size() || !digit(input[i])) invalid("Invalid JSON number");
179 while (i < input.size() && digit(input[i])) ++i;
180 }
181 if (i < input.size() && !whitespace(input[i]) && input[i] != ','
182 && input[i] != '}' && input[i] != ']') invalid("Invalid JSON number");
183 out.append(input, begin, i - begin);
184 --i;
185 } else {
186 if (c >= 0x80) invalid("Invalid JSON token");
187 out += (char)c;
188 }
189 }
190 if (quote) invalid("Incomplete JSON string");
191 return out;
192}

194size_t value_end(const std::string& text, size_t pos)
195{
196 unsigned depth = 0;
197 bool quoted = false, escape = false;
198 for (; pos < text.size(); ++pos) {
199 const char c = text[pos];
200 if (quoted) {
201 if (escape) escape = false;
202 else if (c == '\\') escape = true;
203 else if (c == '"') { quoted = false; if (!depth) return pos + 1; }
204 } else if (c == '"') quoted = true;
205 else if (c == '{' || c == '[') ++depth;
206 else if (c == '}' || c == ']') {
207 if (!depth) return pos;
208 if (!--depth) return pos + 1;
209 } else if (!depth && (c == ',' || whitespace(c))) return pos;
210 }
211 return pos;
212}

214std::string request_id(const std::string& text)
215{
216 size_t pos = 1;
217 while (pos < text.size()) {
218 while (whitespace(text[pos])) ++pos;
219 if (text[pos] == '}') break;
220 const size_t end = value_end(text, pos);
221 Json key;
222 picojson::parse(key, text.substr(pos, end - pos));
223 pos = end;
224 while (whitespace(text[pos])) ++pos;
225 ++pos;
226 while (whitespace(text[pos])) ++pos;
227 const size_t begin = pos;
228 pos = value_end(text, pos);
229 if (key.get<std::string>() == "id") return text.substr(begin, pos - begin);
230 while (whitespace(text[pos])) ++pos;
231 if (text[pos] != ',') break;
232 ++pos;
233 }
234 return {};
235}

252{
253 const int flags = fcntl(s, F_GETFL);

255 || fcntl(s, F_SETFD, FD_CLOEXEC) < 0) return false;
256#ifdef SO_NOSIGPIPE

307 std::string input(const Json& args, bool tracked = false)
308 {
309 fields(args, {"events"});
310 const Json& events = args.get("events");
311 if (!events.is<picojson::array>()) invalid("Expected an events array");
312 if (events.get<picojson::array>().size() > CKeyboard::INPUT_CAPACITY)
313 invalid("Too many input events");
314 std::vector<CKeyboard::InputEvent> batch;
315 const unsigned initial_buttons = keyboard.input_buttons(source);
316 unsigned buttons = initial_buttons;
317 for (const Json& item : events.get<picojson::array>()) {
318 fields(item, {"type", "data"});
319 const std::string& type = string_value(item.get("type"));
320 const Json& data = item.get("data");
321 CKeyboard::InputEvent event;
322 if (type == "key") {
323 fields(data, {"down", "key"});
324 const bool down = json_boolean(data.get("down"));
325 const Json& key = data.get("key");
326 fields(key, {"type", "data"});
327 const std::string& kind = string_value(key.get("type"));
328 u32 code = 0;
329 bool mapped = false;
330 if (kind == "qcode") mapped = qmp_key_code(string_value(key.get("data")), code);
331 else if (kind == "number") mapped = qmp_key_number((unsigned)integer(key.get("data"), 0, 255), code);
332 if (!mapped) invalid("Unsupported key");
333 event.key = code | (down ? 0 : BX_KEY_RELEASED);
334 } else {
335 event.type = CKeyboard::InputEvent::Type::Mouse;
336 if (type == "rel") {
337 fields(data, {"axis", "value"});
338 const std::string& axis = string_value(data.get("axis"));
339 const int value = integer(data.get("value"), -32767, 32767);
340 if (axis == "x") event.dx = value;
341 else if (axis == "y") event.dy = -value;
342 else invalid("Unsupported relative axis");
343 } else if (type == "btn") {
344 fields(data, {"down", "button"});
345 const bool down = json_boolean(data.get("down"));
346 const std::string& name = string_value(data.get("button"));
347 unsigned mask = 0;
348 if (name == "left") mask = 1;
349 else if (name == "right") mask = 2;
350 else if (name == "middle") mask = 4;
351 else if (name == "wheel-up" || name == "wheel-down") {
352 if (!down) continue;
353 event.dz = name == "wheel-up" ? 1 : -1;
354 } else invalid("Unsupported mouse button");
355 if (down) buttons |= mask;
356 else buttons &= ~mask;
357 } else invalid("Unsupported input event type");
358 event.buttons = buttons;
359 }
360 if (event.type == CKeyboard::InputEvent::Type::Mouse && !batch.empty()) {
361 auto& previous = batch.back();
362 if (previous.type == event.type && previous.buttons == event.buttons
363 && std::abs(previous.dx + event.dx) <= 32767
364 && std::abs(previous.dy + event.dy) <= 32767
365 && std::abs(previous.dz + event.dz) <= 32767) {
366 previous.dx += event.dx; previous.dy += event.dy; previous.dz += event.dz;
367 continue;
368 }
369 }
370 batch.push_back(event);
371 }
372 u64 batch_id = 0;
373 switch (keyboard.queue_input(batch.data(), batch.size(), tracked ? &batch_id : nullptr,
374 &initial_buttons, source)) {
375 case CKeyboard::InputResult::Accepted:
376 return tracked ? "{\"__es40-batch\":\"" + std::to_string(batch_id) + "\"}" : "{}";
377 case CKeyboard::InputResult::Full:
378 invalid(tracked ? "Input queue or completion history is full" : "Input queue is full"); break;
379 case CKeyboard::InputResult::Unavailable: invalid("Input device or connection cleanup is not ready"); break;
380 default: invalid("Unsupported input"); break;
381 }
382 return "{}";
383 }

400 void request(const std::string& raw)
401 {
402 std::string id;
403 try {
404 const std::string text = normalize(raw);
405 Json root;
406 std::string parse_error;
407 const auto end = picojson::parse(root, text.begin(), text.end(), &parse_error);
408 if (!parse_error.empty() || end != text.end() || !root.is<picojson::object>())
409 invalid("Invalid JSON request");
410 id = request_id(text);
411 fields(root, {"execute", "arguments", "id"});
412 const std::string& command = string_value(root.get("execute"));
413 const Json empty{picojson::object{}};
414 const Json& args = root.contains("arguments") ? root.get("arguments") : empty;
415 if (!args.is<picojson::object>()) invalid("Expected arguments object");
416 if (command == "qmp_capabilities" && !negotiated) {
417 fields(args, {"enable"});
418 if (args.contains("enable") && (!args.get("enable").is<picojson::array>()
419 || !args.get("enable").get<picojson::array>().empty()))
420 invalid("Unsupported capability");
421 negotiated = true;
422 reply("\"return\":{}", id);
423 } else if (!negotiated || command == "qmp_capabilities") {
424 error("CommandNotFound", "Command is not available in this negotiation state", id);
425 } else if (command == "input-send-event") {
426 reply("\"return\":" + input(args), id);
427 } else if (command == "__es40-input-send-event") {
428 reply("\"return\":" + input(args, true), id);
429 } else if (command == "__es40-query-input") {
430 reply("\"return\":" + input_status(args), id);
431 } else if (command == "query-version") {
432 fields(args, {});
433 reply(std::string("\"return\":") + version, id);
434 } else if (command == "query-status") {
435 fields(args, {});
436 const auto state = system.get_run_state();
437 const bool running = state == CSystem::RunState::Running;
438 const char* status = state == CSystem::RunState::Prelaunch ? "prelaunch"
439 : running ? "running" : "paused";
440 reply(std::string("\"return\":{\"running\":") + (running ? "true" : "false")
441 + ",\"status\":\"" + status + "\"}", id);
442 } else if (command == "query-mice") {
443 fields(args, {});
444 reply(R"("return":[{"name":"ES40 PS/2 mouse","index":0,"current":true,"absolute":false}])", id);
445 } else if (command == "query-commands") {
446 fields(args, {});
447 reply(R"("return":[{"name":"qmp_capabilities"},{"name":"query-version"},{"name":"query-status"},{"name":"query-commands"},{"name":"query-mice"},{"name":"input-send-event"},{"name":"__es40-input-send-event"},{"name":"__es40-query-input"}])", id);
448 } else error("CommandNotFound", "Command is not supported", id);
449 } catch (const QmpError& e) {
450 error(e.type, e.text, id);
451 } catch (const std::exception&) {
452 error("GenericError", "Invalid JSON request", id);
453 }
454 }
455
456 bool consume(char c)
457 {
458 if (frame.empty() && whitespace(c)) return false;
459 if ((unsigned char)c < 0x20 && (!whitespace(c) || quote)) {
460 frame.clear(); nesting.clear(); quote = 0; escape = false;
461 error("GenericError", "Invalid JSON character");
462 return true;
463 }
464 if (frame.empty() && c != '{') {
465 error("GenericError", "Expected a JSON object");
466 return true;
467 }
468 frame += c;
469 if (frame.size() > message_limit) { disconnect(); return true; }
470 if (quote) {
471 if (escape) escape = false;
472 else if (c == '\\') escape = true;
473 else if (c == quote) quote = 0;
474 } else if (c == '"' || c == '\'') quote = c;
475 else if (c == '{' || c == '[') {
476 nesting += c;
477 if (nesting.size() > 32) { disconnect(); return true; }
478 } else if (c == '}' || c == ']') {
479 if (nesting.empty() || (c == '}' ? nesting.back() != '{' : nesting.back() != '[')) {
480 frame.clear(); nesting.clear();
481 error("GenericError", "Mismatched JSON delimiters");
482 return true;
483 }
484 nesting.pop_back();
485 if (nesting.empty()) {
486 std::string complete;
487 complete.swap(frame);
488 request(complete);
489 return true;
490 }
491 }
492 return false;
493 }

495#ifdef ES40_QMP_SOCKETS
496 void flush()
497 {
498 if (client == no_socket || output.empty()) return;
499 int flags = 0;
500#ifdef MSG_NOSIGNAL
501 flags = MSG_NOSIGNAL;
502#endif
503 const int n = (int)send(client, output.data(), (int)std::min<size_t>(output.size(), 16384), flags);
504 if (n > 0) output.erase(0, (size_t)n);
505 else if (n == 0 || !would_block()) disconnect();
506 }
507
508 void poll()
509 {
510 flush();
511 if (client == no_socket) return;
512 if (pending.size() < 4096 && output.size() < output_limit / 2) {
513 char buffer[4096];
514 const int n = (int)recv(client, buffer, sizeof buffer, 0);
515 if (n > 0) pending.append(buffer, (size_t)n);
516 else if (n == 0 || !would_block()) { disconnect(); return; }
517 }
518 size_t consumed = 0;
519 unsigned messages = 0;
520 while (consumed < pending.size() && messages < 32
521 && client != no_socket && output.size() < output_limit / 2) {
522 const char c = pending[consumed++];
523 if (consume(c)) ++messages;
524 }
525 if (client != no_socket) pending.erase(0, consumed);
526 flush();
527 }
528#endif
529};
530}

532struct QmpInput::Impl : CRunnable {
533 QmpInput& owner;
534 CThread thread{"qmp"};
535 std::atomic<bool> stop_requested{true};
536 bool started = false;
537 unsigned port = 0;
538 std::array<std::unique_ptr<QmpClient>, CKeyboard::INPUT_SOURCES> clients;
539 size_t next_client = 0;
540#ifdef ES40_QMP_SOCKETS
541 Socket listener = no_socket;
542#endif

545#endif
546
547 Impl(QmpInput& owner, CSystem& system, CKeyboard& keyboard, const char* setting)
548 : owner(owner)
549 {
550 if (!setting || !*setting) FAILURE(Configuration, "Invalid qmp.port");
551 for (const char* p = setting; *p; ++p) {
552 if (!digit(*p) || port > 6553) FAILURE(Configuration, "Invalid qmp.port");
553 port = port * 10 + (*p - '0');
554 }
555 if (port > 65535) FAILURE(Configuration, "qmp.port must be 0..65535");
556 for (size_t i = 0; i < clients.size(); ++i)
557 clients[i] = std::make_unique<QmpClient>(system, keyboard, i);
558 }
559
560 void run() override
561 {
562 while (!stop_requested) {
563 owner.poll();
564 CThread::sleep(20);
565 }
566 }
567};

573void QmpInput::start()
574{
575 if (!impl->port) return;
576 if (impl->started) { check_state(); return; }
577#ifdef ES40_QMP_SOCKETS
578#ifdef _WIN32
579 WSADATA wsa;
580 const int result = WSAStartup(MAKEWORD(2, 2), &wsa);
581 if (result) FAILURE(Configuration, "QMP Winsock startup failed");
582 impl->winsock = true;
583#endif
584 impl->listener = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
585 bool ready = impl->listener != no_socket && nonblocking(impl->listener);
586 int one = 1;
587#ifdef _WIN32
588 if (ready) ready = setsockopt(impl->listener, SOL_SOCKET, SO_EXCLUSIVEADDRUSE,
589 (const char*)&one, sizeof one) == 0;
590#else
591 if (ready) ready = setsockopt(impl->listener, SOL_SOCKET, SO_REUSEADDR, &one, sizeof one) == 0;
592#endif
593 sockaddr_in address{};
594 address.sin_family = AF_INET;
595 address.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
596 address.sin_port = htons((unsigned short)impl->port);
597 if (ready) ready = bind(impl->listener, (sockaddr*)&address, sizeof address) == 0;
598 if (ready) ready = listen(impl->listener, 4) == 0;
599 if (!ready) {
600 const int code = socket_error();
601 stop();
602 FAILURE(Configuration, "QMP listener failed (socket error " + std::to_string(code) + ")");
603 }
604 try {

643void QmpInput::poll()
644{
645#ifdef ES40_QMP_SOCKETS
646 if (impl->listener == no_socket) return;
647 for (unsigned i = 0; i < 8; ++i) {
648 Socket socket = accept(impl->listener, nullptr, nullptr);
649 if (socket == no_socket) break;
650 const auto slot = std::find_if(impl->clients.begin(), impl->clients.end(),
651 [](const auto& client) { return client->client == no_socket; });
652 if (slot == impl->clients.end() || !nonblocking(socket)) { close_socket(socket); continue; }
653 int one = 1;
654 setsockopt(socket, IPPROTO_TCP, TCP_NODELAY, (const char*)&one, sizeof one);
655 (*slot)->client = socket;
656 (*slot)->reply(std::string("\"QMP\":{\"version\":") + version + ",\"capabilities\":[]}");
657 }
658 // Rotate first service as clients also share the bounded guest input queue.
659 for (size_t i = 0; i < impl->clients.size(); ++i)
660 impl->clients[(impl->next_client + i) % impl->clients.size()]->poll();
661 impl->next_client = (impl->next_client + 1) % impl->clients.size();
662#endif
663}
