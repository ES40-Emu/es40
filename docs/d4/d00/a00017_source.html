32const detail::Driver* driver(Backend backend)
33{
34 switch (backend)
35 {
36#ifdef HAVE_SDL
37 case Backend::sdl: return &detail::sdl_driver();
38#endif
39#ifdef _WIN32
40 case Backend::dsound: return &detail::dsound_driver();
41#endif
42 case Backend::null: return &detail::null_driver();
43 default: return nullptr;
44 }
45}
46}

48SetupGuard::SetupGuard(Backend backend, std::initializer_list<DeviceId*> devices,
49 std::initializer_list<Stream*> streams)
50 : backend_(backend), active_(backend != Backend::sdl)
51{
52 if (active_) {
53 devices_.assign(devices);
54 streams_.assign(streams);
55 }
56}
57
58SetupGuard::~SetupGuard()
59{
60 if (!active_) return;
61 for (auto stream : streams_)
62 if (*stream) destroy_stream(*stream);
63 for (auto device : devices_)
64 if (*device) close_device(*device);
65}
66
67void SetupGuard::complete()
68{
69 if (active_)
70 for (auto stream : streams_)
71 if (!*stream) HOST_AUDIO_FAILURE(backend_, "Failed to initialize audio streams");
72 active_ = false;
73}

130[[noreturn]] void fail(Backend backend, const char* context, const char* file,
131 int line, const char* function)
132{
133 std::string message(context);
134 const auto position = message.find("%s");
135 if (position != std::string::npos)
136 message.replace(position, 2, backend == Backend::sdl ? "SDL" :
137 backend == Backend::dsound ? "DirectSound" : "null");
138 message += ": ";
139 message += error(backend);
140 std::string where = std::string(file) + ", line " + std::to_string(line);
141 if (function) where += std::string(", function '") + function + "'";
142 if (backend == Backend::sdl) throw CSDLException(message, where);
143 if (backend == Backend::dsound) throw CWin32Exception(message, where);
144 throw CRuntimeException(message, where);
145}
