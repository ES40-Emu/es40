57static bool bare(const std::string& value)
58{
59 return !value.empty() && std::all_of(value.begin(), value.end(), [](unsigned char c)
60 { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
61 (c >= '0' && c <= '9') || c == '_' || c == '.'; });
62}
63static std::string encode_value(const std::string& value)
64{
65 std::string result = "\"";
66 for (char c : value)
67 {
68 if (c == '\r' || c == '\n' || c == '\0')
69 throw std::runtime_error("Configuration values cannot contain line breaks or NUL bytes");
70 result += c;
71 if (c == '"') result += c;
72 }
73 return result + "\"";
74}
75static std::string replacement(const ConfigDocumentToken& old, const std::string& value, const std::string& source)
76{
77 return source[old.begin] != '"' && bare(value) ? value : encode_value(value);
78}
79class ConfigDocumentParser
80{
81 const std::string& source;
82 size_t pos = 0;
83 bool line_has_content = false;
84 ConfigDocumentToken token;
85 [[noreturn]] void fail() const
86 { throw std::runtime_error("Invalid configuration syntax near byte " + std::to_string(pos)); }
87 ConfigDocumentToken next()
88 {
89 for (;;)
90 {
91 if (pos == source.size()) return {pos, pos, 0, {}};
92 char c = source[pos];
93 if (c == '\r' || c == '\n') { line_has_content = false; ++pos; continue; }
94 if (c == ' ' || c == '\t' || c == '\f' || c == '\v') { ++pos; continue; }
95 if ((c == ';' && !line_has_content) || source.compare(pos, 2, "//") == 0)
96 {
97 pos = source.find_first_of("\r\n", pos);
98 if (pos == std::string::npos) pos = source.size();
99 continue;
100 }
101 if (source.compare(pos, 2, "/*") == 0)
102 {
103 size_t end = source.find("*/", pos + 2);
104 if (end == std::string::npos) fail();
105 if (source.find_first_of("\r\n", pos) < end) line_has_content = false;
106 pos = end + 2; continue;
107 }
108 break;
109 }
110 ConfigDocumentToken t; t.begin = pos;
111 const char c = source[pos++];
112 line_has_content = true;
113 if (c == '=' || c == ';' || c == '{' || c == '}') t.kind = c;
114 else if (c == '"')
115 {
116 t.kind = 'v';
117 for (;;)
118 {
119 if (pos == source.size()) fail();
120 char ch = source[pos++];
121 if (ch == '\r' || ch == '\n' || ch == '\0') fail();
122 if (ch != '"') { t.value += ch; continue; }
123 if (pos < source.size() && source[pos] == '"') { t.value += '"'; ++pos; }
124 else break;
125 }
126 }
127 else
128 {
129 t.kind = 'v'; t.value += c;
130 while (pos < source.size() && bare(std::string(1, source[pos]))) t.value += source[pos++];
131 if (!bare(t.value)) fail();
132 }
133 t.end = pos;
134 return t;
135 }
136 void advance() { token = next(); }
137 void entries(ConfigDocumentEntry& parent, int depth)
138 {
139 if (depth > 64) fail();
140 int values = 0, children = 0;
141 while (token.kind && token.kind != '}')
142 {
143 ConfigDocumentEntry entry; entry.name = token;
144 if (token.kind != 'v' || !bare(token.value) || source[token.begin] == '"') fail();
145 advance(); if (token.kind != '=') fail();
146 advance(); if (token.kind != 'v') fail();
147 entry.value = token; advance();
148 if (token.kind == '{')
149 {
150 if (++children > CFG_MAX_CHILDREN) fail();
151 entry.block = true; advance(); entries(entry, depth + 1);
152 if (token.kind != '}') fail();
153 entry.close = token.begin; entry.end = token.end; advance();
154 if (token.kind == ';') { entry.end = token.end; advance(); }
155 }
156 else
157 {
158 if (++values > CFG_MAX_VALUES || token.kind != ';') fail();
159 entry.end = token.end; advance();
160 }
161 parent.entries.push_back(std::move(entry));
162 }
163 }
164public:
165 explicit ConfigDocumentParser(const std::string& text) : source(text)
166 {
167 if (source.compare(0, 3, "\xef\xbb\xbf") == 0) pos = 3;
168 }
169 ConfigDocumentEntry parse()
170 {
171 ConfigDocumentEntry root; root.block = true; root.close = root.end = source.size();
172 advance(); entries(root, 0); if (token.kind) fail(); return root;
173 }

176static void populate(const ConfigDocumentEntry& entry, CConfigurator *config)
177{
178 for (const auto& child : entry.entries)

187static void bind_entries(ConfigDocumentEntry& entry, CConfigurator *config)
188{
189 entry.original = config;
190 std::set<int> used;
191 for (auto& child : entry.entries)
192 if (child.block)
193 {
194 CConfigurator *match = nullptr;
195 for (int i = 0; i < config->editor_child_count(); ++i)
196 if (!used.count(i) && child.name.value == config->editor_child(i)->get_myName())
197 { used.insert(i); match = config->editor_child(i); break; }
198 if (!match) throw std::runtime_error("Configuration source and device model disagree");
199 bind_entries(child, match);
200 }

212struct ConfigDocumentTemporaryFile
213{
214 fs::path path;
215 FILE *file = nullptr;
216 explicit ConfigDocumentTemporaryFile(const fs::path& target)
217 {
218 std::random_device random;
219 for (int i = 0; i < 100; ++i)
220 {
221 path = target; path += ".tmp-" + std::to_string(random());
222#ifdef _WIN32
223 file = open_file(path, "wbx");
224#else
225 const int descriptor = open(path.c_str(), O_WRONLY | O_CREAT | O_EXCL, S_IRUSR | S_IWUSR);
226 if (descriptor >= 0)
227 {
228 file = fdopen(descriptor, "wb");
229 if (!file) { close(descriptor); std::error_code ignored; fs::remove(path, ignored); }
230 }
231#endif
232 if (file) return;
233 if (errno != EEXIST) break;
234 }
235 throw std::runtime_error("Could not create a temporary file beside " + target.string());
236 }
237 ~ConfigDocumentTemporaryFile()
238 {
239 if (file) fclose(file);
240 std::error_code ignored; fs::remove(path, ignored);
241 }
242 void write(const std::string& text)
243 {
244 if (fwrite(text.data(), 1, text.size(), file) != text.size() || fflush(file))
245 throw std::runtime_error("Failed writing temporary configuration file");
246#ifdef _WIN32
247 if (_commit(_fileno(file)))
248#else
249 if (fsync(fileno(file)))
250#endif
251 throw std::runtime_error("Failed flushing temporary configuration file");
252 FILE *closing = file; file = nullptr;
253 if (fclose(closing)) throw std::runtime_error("Failed closing temporary configuration file");
254 }
255};
256static void replace_file(const fs::path& from, const fs::path& to)
257{
258#ifdef _WIN32
259 if (!MoveFileExW(from.c_str(), to.c_str(), MOVEFILE_REPLACE_EXISTING | MOVEFILE_WRITE_THROUGH))
260 throw std::system_error(GetLastError(), std::system_category(), "Could not replace " + to.string());
261#else
262 fs::rename(from, to);
263#endif
264}

266struct ConfigDocument::Impl
267{
268 std::string text, newline;
269 ConfigDocumentEntry root;
270 explicit Impl(std::string source) : text(std::move(source)), root(ConfigDocumentParser(text).parse())
271 {
272 size_t pos = text.find_first_of("\r\n");
273 newline = pos == std::string::npos ? "\n" :
274 (text[pos] == '\r' ? (text.compare(pos, 2, "\r\n") == 0 ? "\r\n" : "\r") : "\n");
275 }
276 std::string indent(size_t position) const
277 {
278 size_t start = text.find_last_of("\r\n", position ? position - 1 : 0);
279 start = start == std::string::npos ? 0 : start + 1;
280 size_t end = start; while (end < position && (text[end] == ' ' || text[end] == '\t')) ++end;
281 return text.substr(start, end - start);
282 }
283 std::string child_indent(const ConfigDocumentEntry& entry) const
284 {
285 for (const auto& child : entry.entries)
286 {
287 auto value = indent(child.name.begin);
288 if (value.size() > indent(entry.name.begin).size() || &entry == &root) return value;
289 }
290 return &entry == &root ? "" : indent(entry.name.begin) + " ";
291 }
292 std::string serialize(CConfigurator *node, const std::string& padding) const
293 {
294 std::string value = node->get_myValue();
295 std::string result = padding + node->get_myName() + " = " + (bare(value) ? value : encode_value(value)) + newline + padding + "{" + newline;
296 for (int i = 0; i < node->editor_value_count(); ++i)
297 result += padding + " " + node->editor_value_name(i) + " = " + encode_value(node->editor_value(i)) + ";" + newline;
298 for (int i = 0; i < node->editor_child_count(); ++i) result += serialize(node->editor_child(i), padding + " ");
299 return result + padding + "}" + newline;
300 }
301 void insert(const ConfigDocumentEntry& entry, const std::string& content, std::vector<ConfigDocumentPatch>& patches) const
302 {
303 if (content.empty()) return;
304 size_t position = entry.close;
305 size_t line = text.find_last_of("\r\n", position ? position - 1 : 0);
306 line = line == std::string::npos ? 0 : line + 1;
307 bool own_line = std::all_of(text.begin() + line, text.begin() + position, [](char c) { return c == ' ' || c == '\t'; });
308 if (&entry != &root && own_line) position = line;
309 std::string added = content;
310 if (position && text[position - 1] != '\r' && text[position - 1] != '\n') added = newline + added;
311 if (&entry != &root && !own_line) added += indent(entry.name.begin);
312 patches.push_back({position, position, added});
313 }
314 void render(const ConfigDocumentEntry& entry, CConfigurator *node, std::vector<ConfigDocumentPatch>& patches) const
315 {
316 if (&entry != &root)
317 {
318 if (entry.name.value != node->get_myName()) patches.push_back({entry.name.begin, entry.name.end, node->get_myName()});
319 if (entry.value.value != node->get_myValue()) patches.push_back({entry.value.begin, entry.value.end, replacement(entry.value, node->get_myValue(), text)});
320 }
321 std::set<int> values, children;
322 std::set<CConfigurator *> original_children;
323 for (const auto& child : entry.entries)
324 if (child.block && child.original) original_children.insert(child.original);
325 for (const auto& child : entry.entries)
326 {
327 if (!child.block)
328 {
329 int match = -1;
330 for (int i = 0; i < node->editor_value_count(); ++i)
331 if (!values.count(i) && child.name.value == node->editor_value_name(i)) { match = i; break; }
332 if (match < 0) patches.push_back({child.name.begin, child.end, ""});
333 else
334 {
335 values.insert(match);
336 if (child.value.value != node->editor_value(match))
337 patches.push_back({child.value.begin, child.value.end, replacement(child.value, node->editor_value(match), text)});
338 }
339 }
340 else
341 {
342 int match = -1;
343 for (int i = 0; i < node->editor_child_count(); ++i)
344 if (!children.count(i) && child.original == node->editor_child(i)) { match = i; break; }
345 if (match < 0)
346 for (int i = 0; i < node->editor_child_count(); ++i)
347 if (!children.count(i) && !original_children.count(node->editor_child(i)) && child.name.value == node->editor_child(i)->get_myName()) { match = i; break; }
348 if (match < 0) patches.push_back({child.name.begin, child.end, ""});
349 else { children.insert(match); render(child, node->editor_child(match), patches); }
350 }
351 }
352 std::string added;
353 const auto padding = child_indent(entry);
354 for (int i = 0; i < node->editor_value_count(); ++i)
355 if (!values.count(i)) added += padding + node->editor_value_name(i) + " = " + encode_value(node->editor_value(i)) + ";" + newline;
356 std::vector<int> order;
357 for (int i = 0; i < node->editor_child_count(); ++i) order.push_back(i);
358 if (&entry == &root)
359 std::stable_sort(order.begin(), order.end(), [&](int a, int b)
360 { return !strcasecmp(node->editor_child(a)->get_myName(), "gui") && strcasecmp(node->editor_child(b)->get_myName(), "gui"); });
361 for (int i : order)
362 if (!children.count(i))
363 {
364 auto child = node->editor_child(i);
365 if (&entry == &root && !strcasecmp(child->get_myName(), "gui"))
366 {
367 // The emulator initializes top-level devices in file order.
368 auto system = std::find_if(entry.entries.begin(), entry.entries.end(), [](const ConfigDocumentEntry& e) { return e.block && !strcasecmp(e.name.value.c_str(), "sys0"); });
369 if (system != entry.entries.end())
370 {
371 const size_t position = system->name.begin - indent(system->name.begin).size();
372 patches.push_back({position, position, serialize(child, padding)}); continue;
373 }
374 }
375 added += serialize(child, padding);
376 }
377 insert(entry, added, patches);
378 }

393std::string ConfigDocument::render(CConfigurator *configuration) const
394{
395 std::vector<ConfigDocumentPatch> patches; impl->render(impl->root, configuration, patches);
396 std::stable_sort(patches.begin(), patches.end(), [](const ConfigDocumentPatch& a, const ConfigDocumentPatch& b) { return a.begin < b.begin || (a.begin == b.begin && a.end < b.end); });
397 std::string result; size_t cursor = 0;
398 for (const auto& patch : patches)
399 {
400 if (patch.begin < cursor) throw std::runtime_error("Overlapping configuration edits");
401 result.append(impl->text, cursor, patch.begin - cursor); result += patch.text; cursor = patch.end;
402 }
403 return result + impl->text.substr(cursor);
404}

406ConfigFileSnapshot read_config_file(const fs::path& filename, bool allow_missing)
407{
408 ConfigFileSnapshot result;
409 result.path = fs::absolute(filename).lexically_normal();
410 std::error_code error;
411 const auto status = fs::status(result.path, error);
412 if (error && error != std::errc::no_such_file_or_directory) throw fs::filesystem_error("Cannot inspect configuration", result.path, error);
413 if (!fs::exists(status))
414 {
415 if (fs::is_symlink(fs::symlink_status(result.path))) throw std::runtime_error("Configuration symlink has no target");
416 if (allow_missing) return result;
417 throw std::runtime_error("Configuration file does not exist: " + filename.string());
418 }
419 if (!fs::is_regular_file(status)) throw std::runtime_error("Configuration is not a regular file: " + filename.string());
420 result.path = fs::canonical(result.path);
421 FILE *file = open_file(result.path, "rb");
422 if (!file) throw std::runtime_error("Cannot read configuration: " + filename.string());
423 char buffer[8192]; size_t count;
424 while ((count = fread(buffer, 1, sizeof(buffer), file))) result.text.append(buffer, count);
425 const bool failed = ferror(file) != 0; fclose(file);
426 if (failed) throw std::runtime_error("Failed reading configuration: " + filename.string());
427 result.exists = true;
428 return result;
429}

431void save_config_file(const ConfigFileSnapshot& expected, const std::string& text)
432{
433 auto current = read_config_file(expected.path, true);
434 auto unchanged = [&]()
435 {
436 auto state = read_config_file(expected.path, true);
437 if (state.exists != expected.exists || state.text != expected.text)
438 throw std::runtime_error("Configuration changed on disk; reload it before saving: " + expected.path.string());
439 };
440 unchanged();
441 if (current.exists && current.text == text) return;
442 if (current.exists && (fs::status(current.path).permissions() & (fs::perms::owner_write | fs::perms::group_write | fs::perms::others_write)) == fs::perms::none)
443 throw std::runtime_error("Configuration is read-only: " + current.path.string());
444 ConfigDocumentTemporaryFile output(expected.path); output.write(text);
445 fs::path backup = expected.path; backup += ".bak";
446 if (current.exists)
447 {
448 fs::permissions(output.path, fs::status(current.path).permissions());
449 ConfigDocumentTemporaryFile copy(backup); copy.write(current.text);
450 fs::permissions(copy.path, fs::status(current.path).permissions());
451 unchanged();
452 replace_file(copy.path, backup);
453 }
454 unchanged();
455 replace_file(output.path, expected.path);
456#ifndef _WIN32
457 // Flush the directory entry after the atomic rename where supported.
458 int directory = open(expected.path.parent_path().c_str(), O_RDONLY);
459 if (directory >= 0) { fsync(directory); close(directory); }
460#endif
461}
