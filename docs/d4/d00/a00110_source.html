96 // State accesses use the system's shared device-bus lock.
97 // Device callers must take that lock before their own locks.
98 // Even across multi-call sequences such as reading the count and transferring a buffer.
99
100 // Software request register: controller 0-1, local channel 0-3.
101 void set_request(int index, int channel, int data);
102 // Device request line: global channel 0-3 or 5-7.
103 // asserted is the logical request state, not the electrical pin level.
104 // Outside system reset, only the device lowers this line;
105 // TC, EOP and master clear do not lower it.
106 void set_drq(int channel, bool asserted);

108 struct SDMA_result
109 {
110 size_t transferred; // Bytes moved; zero for verify or standalone EOP.
111 bool blocked; // No DMA service; registers and buffers are unchanged.
112 bool terminal_count; // This call exhausted the count, even with auto-init.
113 bool external_eop; // This call honored the device's EOP indication.
114 };
115
116 // Buffers and lengths are in bytes.
117 // length 0 uses the current count.
118 // Transfers on channels 5-7 must cover whole words.
119 // send: device to memory; recv: memory to device.
120 // Calls remain device-paced and do not require an explicit set_drq().
121 // A software request or retained block bypasses only that channel's mask;
122 // incoming hardware DRQ does not. Enable and cascade requirements still apply.
123 // Channels 0-3 also require controller 1 enabled and channel 4 unmasked
124 // and programmed for cascade. Channel 4's address/count are not serviced.
125 // Cascade mode, a wrong direction or an illegal transfer type is blocked.
126 // Verify services the count without accessing memory or the buffer:
127 // blocked is false, transferred is zero, and terminal_count reports exhaustion.
128 // eop ends the transfer after the last unit serviced by this call, including
129 // when length is capped by the current count. It is not a latched pin level.
130 // Blocked calls do not apply EOP, including calls on cascade-mode channels.
131 // EOP sets completion status and masks or auto-initializes the channel;
132 // otherwise the remaining address/count are retained. DRQ is unchanged.
133 // terminal_count and external_eop can both be true; completion occurs once.
134 // Results do not clear the guest-visible terminal-count status.
135 SDMA_result send_data(int channel, void* data, size_t length = 0,
136 bool eop = false);
137 SDMA_result recv_data(int channel, void* data, size_t length = 0,
138 bool eop = false);
139 // One byte on channels 0-3, one little-endian word on channels 5-7.
140 // Byte sends use the low 8 bits; byte receives are zero-extended.
141 // A blocked or verify receive leaves data unchanged.
142 SDMA_result send_unit(int channel, u16 data, bool eop = false);
143 SDMA_result recv_unit(int channel, u16& data, bool eop = false);
144 // Explicit-request service, at most one byte/word per call.
145 // Demand requires unmasked hardware DRQ; single also accepts a software request.
146 // Block starts from either request and, after successful service, retains it
147 // until TC/EOP even if DRQ or the software request drops. Auto-init releases it.
148 // Between calls, own mask-set/disable/reset or channel reprogramming drops
149 // retention without generating TC/EOP. Pending requests can start a new service.
150 // A held DRQ can service successive calls; these calls never lower DRQ.
151 // Cascade returns blocked. Existing enable/cascade and completion rules apply.
152 // Blocked or verify receives leave data unchanged; blocked calls ignore EOP.
153 // Command bit 4 selects rotating priority; otherwise local order is 0-3.
154 // Rotation makes the released channel lowest priority, including channel 4
155 // for lower-controller service. Ongoing demand/block service is not preempted.
156 // Demand releases on DRQ withdrawal; single re-arbitrates for every unit.
157 // Failed calls and idle register writes do not rotate. Bus timing is not modeled.
158 SDMA_result service_send_unit(int channel, u16 data, bool eop = false);
159 SDMA_result service_recv_unit(int channel, u16& data, bool eop = false);
160 // End an accepted demand/block service without consuming another unit.
161 SDMA_result service_eop(int channel);
162 // Device queries use global channels 0-3 or 5-7; invalid channels throw.
163 // Raw programmed 8237 mode byte, including the local channel-select bits.
164 u8 get_mode(int channel);
165 // Readiness for service_send_unit/service_recv_unit, including explicit
166 // requests and arbitration.
167 bool can_service_send(int channel);
168 bool can_service_recv(int channel);
169 // Raw byte/word count register (number of DMA units minus one).
170 int get_count(int channel);
171 // Current count plus one in bytes; independent of mask/enable state.
172 size_t get_transfer_size(int channel);
