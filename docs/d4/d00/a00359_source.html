37void Ev68CPUState::capture(const CAlphaCPU& cpu, std::uint64_t* d)
38{
39 const auto& state = cpu.state;
40 d[0] = state.last_tb_virt; d[1] = cpu.last_dtb_virt[0]; d[2] = cpu.last_dtb_virt[1];
41 d[3] = state.pctr_ctl; d[4] = state.dc_ctl; d[5] = (u64)state.cc_offset; d[6] = (u64)(u32)state.alt_cm;
42 // IER enables; check_int not snapshotted (rolling it back could suppress a poll)
43 d[7] = (u64)(u32)state.asten; d[8] = (u64)(u32)state.sien; d[9] = (u64)(u32)state.pcen;
44 d[10] = (u64)(u32)state.cren; d[11] = (u64)(u32)state.slen; d[12] = (u64)(u32)state.eien;
45 d[13] = state.exc_sum; // FPSTART may clear it (compiled ITOFx/FTOIx/FLTL)
46 d[14] = state.fpcr; // MT_FPCR (compiled FLTL) writes it
47 d[15] = (u64)state.sde; d[16] = (u64)state.hwe; // I_CTL (compiled as terminator)
48 d[17] = (u64)(u32)state.i_ctl_spe; d[18] = (u64)(u32)state.i_ctl_va_mode;
49 d[19] = state.i_ctl_vptb; d[20] = state.i_ctl_other;
50 d[21] = (u64)(u32)state.cm; d[22] = (u64)(u32)state.sir; // CM, SIRR
51 d[23] = (u64)(u32)state.aster; d[24] = (u64)(u32)state.astrr; // 0x40-group ASTER/ASTRR
52 d[25] = (u64)(u32)state.fpen; d[26] = (u64)(u32)state.ppcen; // 0x40-group FPEN/PPCEN
53 d[27] = state.exc_addr; // PAL entry writes it; family returns may read it.
54}

56void Ev68CPUState::prepare_replay(CAlphaCPU& cpu, const std::uint64_t* s)
57{
58 auto& state = cpu.state;
59 state.sde = (bool)s[15]; state.hwe = (bool)s[16]; // restore I_CTL too: a compiled HW_MFPR must read the pre-interp value
60 state.i_ctl_spe = (int)s[17]; state.i_ctl_va_mode = (int)s[18];
61 state.i_ctl_vptb = s[19]; state.i_ctl_other = s[20];
62 state.cm = (int)s[21]; state.sir = (int)s[22]; // ...and CM/SIRR
63 cpu.flush_data_page_cache();
64 state.aster = (int)s[23]; state.astrr = (int)s[24]; // ...and the PCTX read-backs (a
65 state.fpen = (int)s[25]; state.ppcen = (int)s[26]; // HW_MFPR PCTX reads these live)
66 state.exc_addr = s[27];
67 state.exc_sum = s[13];
68 state.fpcr = s[14];
69}

90bool Ev68CPUState::differs(int ii, std::uint64_t x, std::uint64_t y, bool side_exit)
91{
92 // Slots 0-2 (last TB fills) are replay-invisible (verify loads skip translation),
93 // so they're only sound at full-span compares (by inheritance) -- skip here.
94 if (side_exit && ii < 3) return false;
95 // EXC_SUM REG[12:8] (HRM 5.2.13) reports the trapping op's dest register -- report state,
96 // not execution state, and which pass recorded it differs benignly. Compare the exception
97 // bits strictly, mask the REG field.
98 if (ii == 13) { x &= ~U64(0x1f00); y &= ~U64(0x1f00); }
99 return x != y;
100}
