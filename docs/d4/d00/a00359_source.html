39// Usage:
40// XTAL(25'174'800).value() -> 25174800 (u32)
41// XTAL(28'636'363).dvalue() -> 28636363.0 (double)
42// (XTAL(14'318'181) * 2).value() -> 28636362
43// XTAL(14'318'181) / 3 -> XTAL with current_clock ~ 4772727

51class XTAL
52{
53public:
54 constexpr explicit XTAL(double base_clock) noexcept : m_base_clock(base_clock), m_current_clock(base_clock) {}
55
56 constexpr double dvalue() const noexcept { return m_current_clock; }
57 constexpr uint32_t value() const noexcept { return uint32_t(m_current_clock + 1e-3); }
58 constexpr double base() const noexcept { return m_base_clock; }
59
60 template <typename T> constexpr XTAL operator *(T mult) const noexcept { return XTAL(m_base_clock, m_current_clock * mult); }
61 template <typename T> constexpr XTAL operator /(T div) const noexcept { return XTAL(m_base_clock, m_current_clock / div); }
62
63 friend constexpr XTAL operator *(int mult, const XTAL& xtal);
64 friend constexpr XTAL operator *(unsigned int mult, const XTAL& xtal);
65 friend constexpr XTAL operator *(double mult, const XTAL& xtal);
66
67private:
68 double m_base_clock, m_current_clock;
69
70 constexpr XTAL(double base_clock, double current_clock) noexcept : m_base_clock(base_clock), m_current_clock(current_clock) {}
71};
