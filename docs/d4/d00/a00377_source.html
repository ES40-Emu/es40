60static uint32_t gloria_setting(
61 CConfigurator* cfg, const char* key, uint32_t def)
62{
63 const char* text = cfg->get_text_value(key, nullptr);
64 if (!text || !*text)
65 return def;
66 char* end = nullptr;
67 errno = 0;
68 const unsigned long long n = std::strtoull(text, &end, 0);
69 if (errno || *end || text[0] == '-' || n > 0xffffffffULL)
70 throw std::runtime_error(
71 std::string("Invalid gloria integer setting: ") + key);
72 return static_cast<uint32_t>(n);
73}

100static void gloria_write_u32(FILE* f, uint32_t v)
101{
102 uint8_t b[4];
103 for (unsigned i = 0; i < 4; ++i)
104 b[i] = uint8_t(v >> (8 * i));
105 if (fwrite(b, 1, 4, f) != 4)
106 throw std::runtime_error("Snapshot write failed");
107}

109static uint32_t gloria_read_u32(FILE* f)
110{
111 uint8_t b[4];
112 if (fread(b, 1, 4, f) != 4)
113 throw std::runtime_error("Snapshot read failed");
114 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
115 (uint32_t(b[3]) << 24);
116}

118static bool gloria_vga_window_offset(u32& address, unsigned selection)
119{
120 switch (selection & 3)
121 {
122 case 1:
123 return address < 0x10000;
124 case 2:
125 if (address < 0x10000 || address >= 0x18000)
126 return false;
127 address -= 0x10000;
128 break;
129 case 3:
130 if (address < 0x18000 || address >= 0x20000)
131 return false;
132 address -= 0x18000;
133 break;
134 default:
135 return address < 0x20000;
136 }
137 return true;
138}

150void CGloriaSynergy::sequencer_map(address_map& map)
151{
152 unimplemented_map(map, "VGA sequencer register");
153 // These frontends mask stored bytes, including direct snapshot writes.
154 // Limit the canonical map to SR00-04 so unknown indices keep diagnostics.
155 auto standard = std::make_shared<address_map>(5);
156 vga_device::sequencer_map(*standard);
157 map(0x00, 0x04).lrw8(
158 NAME([standard](offs_t index) {
159 return standard->read_byte(index);
160 }),
161 NAME(([this, standard](offs_t index, u8 data) {
162 static constexpr u8 masks[] = {0x03, 0x3f, 0x0f, 0x3f, 0x0e};
163 vga.sequencer.data[index] = data & masks[index];
164 standard->write_byte(index, data);
165 })));
166 map(0x05, 0x05).lr8(NAME([this](offs_t) {
167 return vga.sequencer.data[5];
168 }));
169 // TVP4020 Hardware Reference Manual, issue 6, section 3.5.3, p.61.
170 // Host memory/DAC gating, display selection and extended DAC addressing
171 // are consumed by the corresponding card access paths. Native output is
172 // enabled separately through VideoControl, not the VGA VTG control bit.
173 map(0x05, 0x05).lw8(NAME([this](offs_t, u8 data) {
174 vga.sequencer.data[5] = data & 0x7f;
175 }));
176}

178void CGloriaSynergy::gc_map(address_map& map)
179{
180 unimplemented_map(map, "VGA graphics controller register");
181 vga_device::gc_map(map);
182 // TVP4020 HRM section 3.5.4, p.62. The display-start extension is write-only.
183 map(0x09, 0x09)
184 .lrw8(
185 NAME([this](offs_t) {
186 return u8(m_mode640 & 0xbf);
187 }),
188 NAME([this](offs_t, u8 data) {
189 m_mode640 = data;
190 vga.crtc.start_addr_latch =
191 (vga.crtc.start_addr_latch & ~0x10000u) |
192 (u32(data & 0x40) << 10);
193 }));
194}

196void CGloriaSynergy::attribute_map(address_map& map)
197{
198 unimplemented_map(map, "VGA attribute register");
199 vga_device::attribute_map(map);
200 // Palette reads remain available when PAS disables CPU writes.
201 map(0x20, 0x2f).lr8(NAME([this](offs_t offset) {
202 return vga.attribute.data[offset];
203 }));
204 // These frontends retain mode-control bits 6/7 and mask reserved bit 4.
205 map(0x10, 0x10).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
206 vga.attribute.data[0x10] = data & 0xef;
207 }));
208 // Color select does not change the pending horizontal panning value.
209 map(0x14, 0x14).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
210 vga.attribute.data[0x14] = data & 0xf;
211 }));
212}

231u8 CGloriaSynergy::mem_r(offs_t address)
232{
233 const u32 bank = (m_mode640 & 0x80)
234 ? ((m_mode640 >> ((address & 0x10000) ? 3 : 0)) & 7) * 0x10000u
235 : 0;
236 if (!gloria_vga_window_offset(address, vga.gc.memory_map_sel))
237 return 0xff;
238 if (m_mode640 & 0x80)
239 address &= 0xffff;
240 unsigned selected = vga.gc.read_map_sel & 3;
241 if (vga.sequencer.data[4] & 8)
242 {
243 // Keep packed chain-four bytes in the layout consumed by vga_device scanout.
244 for (unsigned plane = 0; plane < 4; ++plane)
245 vga.gc.latch[plane] = vga.memory[bank + (address & ~3u) + plane];
246 selected = address & 3;
247 }
248 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
249 {
250 // vga_device text stores character/attribute pairs together; fonts stay planar.
251 vga.gc.latch[0] = vga.memory[bank + (address & ~1u)];
252 vga.gc.latch[1] = vga.memory[bank + (address & ~1u) + 1];
253 for (unsigned plane = 2; plane < 4; ++plane)
254 vga.gc.latch[plane] =
255 vga.memory[bank + (address >> 1) + plane * 0x10000];
256 selected = (vga.gc.read_map_sel & 2) | (address & 1);
257 }
258 else
259 {
260 const u32 offset = bank + (address & 0xffff);
261 for (unsigned plane = 0; plane < 4; ++plane)
262 vga.gc.latch[plane] = vga.memory[offset + plane * 0x10000];
263 }
264 if (!vga.gc.read_mode)
265 return vga.gc.latch[selected];
266 u8 result = 0xff;
267 for (unsigned plane = 0; plane < 4; ++plane)
268 if (vga.gc.color_dont_care & (1 << plane))
269 result &=
270 u8(vga.gc.latch[plane] ^
271 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
272 return result;
273}

275void CGloriaSynergy::mem_w(offs_t address, u8 data)
276{
277 const u32 bank = (m_mode640 & 0x80)
278 ? ((m_mode640 >> ((address & 0x10000) ? 3 : 0)) & 7) * 0x10000u
279 : 0;
280 if (!gloria_vga_window_offset(address, vga.gc.memory_map_sel))
281 return;
282 if (m_mode640 & 0x80)
283 address &= 0xffff;
284 if (vga.sequencer.data[4] & 8)
285 {
286 if (vga.sequencer.map_mask & (1 << (address & 3)))
287 vga.memory[bank + address] = vga_latch_write(address & 3, data);
288 return;
289 }
290 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
291 {
292 const unsigned parity = address & 1;
293 if (vga.sequencer.map_mask & (1 << parity))
294 vga.memory[bank + address] = vga_latch_write(parity, data);
295 if (vga.sequencer.map_mask & (1 << (parity + 2)))
296 vga.memory[bank + (address >> 1) + (parity + 2) * 0x10000] =
297 vga_latch_write(parity + 2, data);
298 return;
299 }
300 const u32 offset = bank + (address & 0xffff);
301 for (unsigned plane = 0; plane < 4; ++plane)
302 if (vga.sequencer.map_mask & (1 << plane))
303 vga.memory[offset + plane * 0x10000] = vga_latch_write(plane, data);
304}

330CPermedia2::Frame CGloriaSynergy::render_frame()
331{
332 std::lock_guard<std::mutex> frame_guard(m_frame_mutex);
333 std::unique_lock<std::recursive_mutex> bus_guard(
334 cSystem->get_device_bus_mutex());
335 if (m_pause.load())
336 return {};
337 // ChipConfig can disable SVGA entirely; otherwise SR5 bit 3 selects the
338 // VGA or graphics-processor display (SLAU011A, section 5.2.3). Native timing
339 // and blanking follow VideoControl in scanout(), not SR5's VGA VTG control.
340 if (!(m_permedia2.peek(CPermedia2::ChipConfig) & 2) ||
341 !(vga.sequencer.data[5] & 8))
342 {
343 m_permedia2.capture_scanout(m_scanout);
344 bus_guard.unlock();
345 return CPermedia2::scanout(m_scanout);
346 }
347 // vga_device renders live CRTC state and updates text-cursor/palette state.
348 // Keep that legacy path serialized until it has its own captured renderer.
349 if (!vga.crtc.sync_en || (vga.sequencer.data[0] & 3) != 3 ||
350 (vga.sequencer.data[1] & 0x20) || !(vga.attribute.index & 0x20) ||
351 !vga.crtc.maximum_scan_line)
352 return {};
353 CPermedia2::Frame frame;
354 frame.width = (vga.crtc.horz_disp_end + 1) *
355 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
356 frame.height = vga.crtc.vert_disp_end + 1;
357 if (!frame.width || frame.width > 4096 || frame.height > 2048)
358 return {};
359 // vga_device can write the final repeated scanline past the visible-area bottom.
360 // Reserve one full character row; only the visible rectangle is submitted.
361 m_render_bitmap.resize(frame.width, frame.height + 64);
362 screen().set_visible_area(frame.width, frame.height);
363 screen().tick_frame();
364 vga.crtc.start_addr = latch_start_addr();
365 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
366 palette_update();
367 svga_device::screen_update(
368 screen(), m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
369 frame.argb.assign(
370 m_render_bitmap.raw(),
371 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
372 return frame;
373}

375void CGloriaSynergy::update()
376{
377 const CPermedia2::Frame frame = render_frame();
378 // Never hold the device-bus mutex while acquiring a GUI mutex.
379 {
380 CGloriaDisplayLock guard(m_output.display());
381 if (!frame.argb.empty())
382 {
383 if (frame.width != m_host_width || frame.height != m_host_height)
384 {
385 m_output.display().dimension_update(
386 frame.width, frame.height, 0, 0, 32);
387 m_host_width = frame.width;
388 m_host_height = frame.height;
389 }
390 m_output.display().graphics_frame_update(
391 frame.argb.data(), frame.width, frame.height);
392 }
393 else
394 {
395 m_output.display().clear_screen();
396 }
397 m_output.display().flush();
398 }

401void CGloriaSynergy::run()
402{
403 try
404 {
405 if (!m_gui_initialized)
406 {
407 m_output.display().init(X_TILESIZE, Y_TILESIZE);
408 m_gui_initialized = true;
409 }
410 while (!m_stop.load())
411 {
412 {
413 CGloriaDisplayLock guard(m_output.display());
414 m_output.display().handle_events();
415 }
416 if (m_pause.load())
417 {
418 m_pause_ack.store(true);
419 CThread::sleep(10);
420 continue;
421 }
422 m_pause_ack.store(false);
423 update();
424 CThread::sleep(20);
425 }
426 }
427 catch (const CException& e)
428 {
429 printf(
430 "%s: Exception in display thread: %s.\n",
431 devid_string,
432 e.displayText().c_str());
433 m_worker_failed.store(true);
434 }
435 catch (const std::exception& e)
436 {
437 printf(
438 "%s: Exception in display thread: %s.\n", devid_string, e.what());
439 m_worker_failed.store(true);
440 }
441}

469CGloriaSynergy::OptionROM CGloriaSynergy::OptionROM::parse(
470 const std::vector<uint8_t>& raw, const std::string& mapping)
471{
472 auto fail = [](const char* message) {
473 throw std::invalid_argument(message);
474 };
475 if (raw.size() != 32768 && raw.size() != 65536)
476 fail("ROM must be an unmodified 32 or 64 KiB image");
477 auto word = [&](size_t p) -> uint16_t {
478 if (p > raw.size() - 2)
479 fail("ROM field out of bounds");
480 return uint16_t(uint16_t(raw[p]) | (uint16_t(raw[p + 1]) << 8));
481 };
482 if (word(0) != 0xaa55)
483 fail("ROM missing 55 AA signature");
484 const size_t p = word(0x18);
485 if (p > raw.size() - 24 || raw[p] != 'P' || raw[p + 1] != 'C' ||
486 raw[p + 2] != 'I' || raw[p + 3] != 'R')
487 fail("ROM missing/bounded PCIR structure");
488 const uint16_t pcir_size = word(p + 10);
489 if (pcir_size < 24 || size_t(pcir_size) > raw.size() - p)
490 fail("Invalid PCIR structure length");
491 const size_t image = size_t(word(p + 16)) * 512,
492 header = size_t(raw[2]) * 512;
493 if (!image || image > raw.size() || header != image ||
494 p + pcir_size > image)
495 fail("Invalid/inconsistent option-ROM image lengths");
496 if (raw[p + 20] != 0 || !(raw[p + 21] & 0x80))
497 fail("Only a single final x86 option image is supported");
498 unsigned sum = 0;
499 for (size_t i = 0; i < image; ++i)
500 sum += raw[i];
501 if (sum & 255)
502 fail("Option-ROM checksum failed");
503 OptionROM rom;
504 rom.file_size = static_cast<uint32_t>(raw.size());
505 rom.image_size = static_cast<uint32_t>(image);
506 rom.vendor = word(p + 4);
507 rom.device = word(p + 6);
508 rom.layout = mapping;
509 rom.bytes.assign(65536, 0xff);
510 if (raw.size() == 65536)
511 {
512 if (mapping != "exact")
513 fail("64 KiB ROM requires rom_layout=exact");
514 rom.bytes = raw;
515 }
516 else
517 {
518 if (mapping != "mirror32" && mapping != "pad_ff")
519 fail(
520 "32 KiB ROM requires explicit rom_layout=mirror32 or pad_ff "
521 "(upper decode unmeasured)");
522 std::copy(raw.begin(), raw.end(), rom.bytes.begin());
523 if (mapping == "mirror32")
524 std::copy(raw.begin(), raw.end(), rom.bytes.begin() + 32768);
525 }
526 const std::string content(raw.begin(), raw.end());
527 rom.agp_banner = content.find("/AGP/") != std::string::npos ||
528 content.find("/AGP]") != std::string::npos;
529 rom.subsystem_vendor = uint16_t(
530 uint16_t(rom.bytes[0xfffc]) | (uint16_t(rom.bytes[0xfffd]) << 8));
531 rom.subsystem_device = uint16_t(
532 uint16_t(rom.bytes[0xfffe]) | (uint16_t(rom.bytes[0xffff]) << 8));
533 return rom;
534}

564void CGloriaSynergy::init()
565try
566{
567 if (m_initialized)
568 return;
569 CConfigurator* cfg = myCfg;
570 init_maps();
571 add_legacy_io(LegacyIO, 0x3b0, 0x30);
572 add_legacy_mem(LegacyMemory, 0xa0000, 0x20000);
573 m_profile.vendor = gloria_setting16(cfg, "vendor", m_profile.vendor);
574 m_profile.device = gloria_setting16(cfg, "device", m_profile.device);
575 m_profile.subsystem_vendor = gloria_setting16(cfg, "subsystem_vendor", 0);
576 m_profile.subsystem_device = gloria_setting16(cfg, "subsystem_device", 0);
577 const auto revision = gloria_setting(cfg, "revision", m_profile.revision);
578 if (revision > 255)
579 throw std::runtime_error("gloria revision exceeds 8 bits");
580 m_profile.revision = static_cast<uint8_t>(revision);
581 m_service_budget = gloria_setting(cfg, "service_budget", 4096);
582 m_scanline_access_divisor =
583 gloria_setting(cfg, "scanline_access_divisor", 1024);
584 if (m_service_budget == 0 || m_service_budget > 1000000 ||
585 m_scanline_access_divisor == 0)
586 throw std::runtime_error("Invalid gloria scheduling budget/divisor");
587 m_allow_dma = cfg->get_bool_value("allow_dma", true);
588 m_trace_apertures = cfg->get_bool_value("trace_apertures", false);
589 m_frame_file = cfg->get_text_value("frame_file", "");
590 const std::string romname = cfg->get_text_value("rom", "");
591 m_rom_layout = cfg->get_text_value("rom_layout", "exact");
592 const bool from_rom = cfg->get_bool_value("subsystem_from_rom", false);
593 if (!romname.empty())
594 {
595 std::ifstream f(romname, std::ios::binary | std::ios::ate);
596 if (!f)
597 throw std::runtime_error("Cannot open gloria ROM: " + romname);
598 const auto size = f.tellg();
599 if (size != 32768 && size != 65536)
600 throw std::runtime_error("gloria ROM must be 32 or 64 KiB");
601 f.seekg(0);
602 std::vector<uint8_t> raw(static_cast<size_t>(size));
603 f.read(

818
819void CGloriaSynergy::sync_pci_config()
820{
821 const u32 chip_config = m_permedia2.peek(CPermedia2::ChipConfig);
822 const bool fixed = (chip_config & 4) != 0;
823 // HRM 2.14.4: non-fixed VGA is an "other" display controller. Keep the
824 // board VGA class when fixed decode is enabled.
825 const u32 class_code = (chip_config & 1)
826 ? (fixed ? 0x000100u : 0x000000u)
827 : (fixed ? 0x030000u : 0x038000u);
828 pci_state.config_data[0][2] = endian_32(
829 (class_code << 8) | (endian_32(pci_state.config_data[0][2]) & 0xff));
830
831 // HRM 2.14.6: without fixed VGA decode, I/O and palette snoop enables
832 // are read-only zero. Re-enabling decode does not restore their old values.
833 const u32 vga_command_bits = 0x21;
834 u32 command = endian_32(pci_state.config_data[0][1]);
835 u32 mask = endian_32(pci_state.config_mask[0][1]);
836 mask = (mask & ~vga_command_bits) |
837 (fixed ? endian_32(std_config_mask[0][1]) & vga_command_bits : 0);
838 if (!fixed)
839 command &= ~vga_command_bits;
840 pci_state.config_data[0][1] = endian_32(command);
841 pci_state.config_mask[0][1] = endian_32(mask);
842}

844bool CGloriaSynergy::legacy_enabled(bool memory) const noexcept
845{
846 const u32 command = endian_32(pci_state.config_data[0][1]);
847 if ((m_permedia2.peek(CPermedia2::ChipConfig) & 6) != 6)
848 return false;

850 (!memory || (vga.sequencer.data[5] & 1));

853bool CGloriaSynergy::decodes_memory_access(
854 int index, u64 address, int dsize, bool write) const noexcept
855{
856 if (index >= PCI_RANGE_BASE)
857 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
858 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
859 return false;
860 if (index == LegacyIO)
861 {
862 if (!legacy_enabled(false) || address + dsize / 8 > 0x30)
863 return false;
864 const u32 port = u32(address) + 0x3b0;
865 if ((port >= 0x3b0 && port < 0x3c0 && m_ioas) ||
866 (port >= 0x3d0 && !m_ioas))
867 return false;
868 if (port >= 0x3c6 && port <= 0x3c9 && !(vga.sequencer.data[5] & 2))
869 return false;
870 return true;
871 }
872 if (index != LegacyMemory || !legacy_enabled(true) ||
873 address + dsize / 8 > 0x20000)
874 return false;
875 switch (vga.gc.memory_map_sel & 3)
876 {
877 case 1:
878 return address + dsize / 8 <= 0x10000;
879 case 2:
880 return address >= 0x10000 && address + dsize / 8 <= 0x18000;
881 case 3:
882 return address >= 0x18000;
883 default:
884 return true;
885 }
886}

888u32 CGloriaSynergy::ReadMem_Legacy(int index, u32 address, int dsize)
889{
890 std::lock_guard<std::recursive_mutex> guard(
891 cSystem->get_device_bus_mutex());
892 if (!decodes_memory_access(index, address, dsize, false))
893 return 0xffffffff;
894 if (dsize != 8 && dsize != 16 && dsize != 32)
895 return 0xffffffff;

900 : mem_r(address + i))

924u32 CGloriaSynergy::ReadMem_Bar(int func, int bar, u32 address, int dsize)
925{
926 std::lock_guard<std::recursive_mutex> guard(
927 cSystem->get_device_bus_mutex());
928 if (func != 0)
929 return 0xffffffff;
930 tick();
931 u32 data = 0xffffffff;
932 switch (bar)
933 {
934 case 0:
935 if (address < 0x20000 && (address & 0xf000) == 0x6000)
936 {
937 if (dsize != 8 && dsize != 16 && dsize != 32)
938 return 0xffffffff;
939 data = 0;
940 for (int i = 0; i < dsize / 8; ++i)
941 {
942 const u32 lane =
943 (address + i) ^ ((address & 0x10000) ? 3u : 0u);
944 data |= u32(io_read_b(lane & 0x3ff)) << (8 * i);
945 }
946 }
947 else
948 data = m_permedia2.ReadMem(address, dsize);
949 break;
950 case 1:
951 case 2:
952 data = m_permedia2.mem_read(
953 static_cast<unsigned>(bar - 1), address, dsize);
954 break;
955 case 6:
956 if ((dsize == 8 || dsize == 16 || dsize == 32) &&
957 address <= m_rom.size() - static_cast<unsigned>(dsize) / 8)
958 {
959 data = 0;
960 for (int i = 0; i < dsize / 8; ++i)
961 data |= u32(m_rom[address + static_cast<unsigned>(i)])
962 << (8 * i);
963 }
964 break;
965 }
966 trace('R', bar, address, dsize, data);
967 return data;
968}

970void CGloriaSynergy::WriteMem_Bar(
971 int func, int bar, u32 address, int dsize, u32 data)
972{
973 std::lock_guard<std::recursive_mutex> guard(
974 cSystem->get_device_bus_mutex());
975 if (func != 0)
976 return;
977 tick();
978 trace('W', bar, address, dsize, data);
979 switch (bar)
980 {
981 case 0:
982 if (address < 0x20000 && (address & 0xf000) == 0x6000)
983 {
984 if (dsize != 8 && dsize != 16 && dsize != 32)
985 return;
986 for (int i = 0; i < dsize / 8; ++i)
987 {
988 const u32 lane =
989 (address + i) ^ ((address & 0x10000) ? 3u : 0u);
990 io_write_b(lane & 0x3ff, u8(data >> (8 * i)));
991 }
992 }
993 else if (m_permedia2.WriteMem(address, dsize, data))
994 {
995 if ((address & 0xffff) == CPermedia2::ChipConfig ||
996 (address & 0xffff) == CPermedia2::ResetStatus)
997 sync_pci_config();
998 if ((address & 0xffff) == CPermedia2::PaletteRead)
999 vga.dac.read = 1;
1000 if ((address & 0xffff) == CPermedia2::PaletteWrite)
1001 vga.dac.read = 0;
1002 }
1003 break;
1004 case 1:
1005 case 2:
1006 m_permedia2.mem_write(
1007 static_cast<unsigned>(bar - 1), address, dsize, data);
1008 break;
1009 }
1010}

1038u32 CGloriaSynergy::config_read_custom(
1039 int func, u32 address, int dsize, u32 data)
1040{
1041 std::lock_guard<std::recursive_mutex> guard(
1042 cSystem->get_device_bus_mutex());
1043 if (func == 0 && !m_replaying_pci && (address & ~3u) == 0xfc)
1044 {
1045 // Narrow initial contract: DWORD port accesses only. Never service a
1046 // partial port read by destructively reading/discarding a whole DWORD.
1047 if (address != 0xfc || dsize != 32)
1048 {
1049 printf(
1050 "%s: Unsupported indirect-data read (address=%02x, "
1051 "dsize=%d).\n",
1052 devid_string,
1053 address,
1054 dsize);
1055 data = 0xffffffff;
1056 }
1057 else
1058 {
1059 const u32 indirect = config_read(0, 0xf8, 32),
1060 region = indirect >> 28, offset = indirect & 0x7fffff;
1061 if ((region == 0 && offset < 0x20000) || region == 1 || region == 2)
1062 data = ReadMem_Bar(0, int(region), offset, 32);
1063 else if (region == 7 && offset <= 0xfffc)
1064 data = ReadMem_Bar(0, 6, offset, 32);
1065 else
1066 data = 0xffffffff;
1067 }
1068 }
1069 if (m_trace)
1070 m_trace << "# CFG R " << std::hex << func << ' ' << address << ' '
1071 << dsize << ' ' << data << std::dec << '\n';
1072 return data;

1111void CGloriaSynergy::check_state()
1112{
1113 if (m_worker_failed.load())
1114 FAILURE(Thread, "GLoria display thread failed");
1115 // Retain the periodic work allowance, but release the shared bus between
1116 // bounded batches so audio DMA and other devices can make progress.
1117 for (size_t remaining = 1000000; remaining;)
1118 {
1119 const size_t budget = std::min(remaining, size_t(4096));
1120 size_t used;
1121 {
1122 std::lock_guard<std::recursive_mutex> guard(
1123 cSystem->get_device_bus_mutex());
1124 if (m_trace)
1125 m_trace << "STEP " << std::hex << budget << std::dec << '\n';
1126 used = m_permedia2.service(budget);
1127 }
1128 if (used < budget)
1129 break;
1130 remaining -= budget;
1131 if (remaining)
1132 std::this_thread::yield();
1133 }
1134 // ES40 currently calls this roughly every 100ms. This compatibility pump
1135 // advances ONE virtual frame per call, not a real 60 Hz monitor model.
1136 {
1137 std::lock_guard<std::recursive_mutex> guard(
1138 cSystem->get_device_bus_mutex());
1139 const uint32_t lines = (m_permedia2.peek(CPermedia2::VTotal) & 0x7ff) + 1;
1140 m_permedia2.advance_scanlines(lines);
1141 if (m_trace)
1142 {
1143 m_trace << "LINES " << std::hex << lines << std::dec << '\n';
1144 m_trace.flush();
1145 }
1146 }
1147 try
1148 {
1149 publish_frame();
1150 }
1151 catch (const std::exception& e)
1152 {
1153 printf(
1154 "%s: Could not write graphics frame: %s.\n",
1155 devid_string,
1156 e.what());
1157 }
1158}

1206{

1241int CGloriaSynergy::RestoreState(FILE* f)
1242{
1243 if (!f)
1244 return -1;
1245 std::lock_guard<std::recursive_mutex> guard(
1246 cSystem->get_device_bus_mutex());
1247 try
1248 {
