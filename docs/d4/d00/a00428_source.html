39// EV6/EV68 PAL exception vectors.
40constexpr AlphaPALExceptionEntry ev68_pal_exception_entry(AlphaPALException reason,
41 std::uint64_t current_pc)
42{
43 switch (reason)
44 {
45 case AlphaPALException::DataTlbDoubleMiss43: return {0x100, current_pc, false};
46 case AlphaPALException::DataTlbDoubleMiss48: return {0x180, current_pc, false};
47 case AlphaPALException::FloatingPointDisabled: return {0x200, current_pc, true};
48 case AlphaPALException::Unaligned: return {0x280, current_pc, true};
49 case AlphaPALException::DataTlbMiss: return {0x300, current_pc, false};
50 case AlphaPALException::DataFault: return {0x380, current_pc, true};
51 case AlphaPALException::OpcodeDecode: return {0x400, current_pc, true};
52 case AlphaPALException::InstructionAccessViolation: return {0x480, current_pc, true};
53 case AlphaPALException::MachineCheck: return {0x500, current_pc, true};
54 case AlphaPALException::InstructionTlbMiss: return {0x580, current_pc, false};
55 case AlphaPALException::Arithmetic: return {0x600, current_pc, true};
56 case AlphaPALException::Interrupt: return {0x680, current_pc, true};
57 case AlphaPALException::MtFpcr: return {0x700, current_pc, true};
58 case AlphaPALException::Reset: return {0x780, current_pc, true};
59 }
60 return {0, current_pc, true};
61}

63constexpr bool ev68_call_pal_valid(std::uint32_t function, std::uint32_t mode)
64{
65 return function <= 0x3f ? mode == 0 : function >= 0x80 && function <= 0xbf;
66}

68// EV68 records the CALL_PAL instruction address and writes next PC to R23,
69// or R55 when the PAL shadow bank is enabled.
70constexpr AlphaPALCallEntry ev68_call_pal_entry(std::uint32_t function,
71 std::uint64_t current_pc,
72 std::uint64_t next_pc)
73{
74 const std::uint64_t offset = std::uint64_t(0x2000)
75 | (std::uint64_t(function & 0x80) << 5)
76 | (std::uint64_t(function & 0x3f) << 6) | 1;
77 return {offset, current_pc, next_pc & ~std::uint64_t(2), 23, 55, true};
78}

80// EV68 HW_RET reads Rb, ignores hints/qualifiers, clears bit 1 and preserves the lock.
81constexpr AlphaPALReturnDescriptor ev68_hardware_return(std::uint32_t instruction)
82{
83 return (instruction >> 26) == 0x1e
84 ? AlphaPALReturnDescriptor{true, ~std::uint64_t(2), 0, false, true,
85 AlphaPALReturnSource::RegisterB}
86 : AlphaPALReturnDescriptor{};
87}
88
89constexpr AlphaPALInstructionAccess ev68_hardware_instruction_access(std::uint32_t instruction)
90{
91 switch (instruction >> 26)
92 {
93 case 0x19: // HW_MFPR.
94 case 0x1b: // HW_LD.
95 case 0x1d: // HW_MTPR.
96 case 0x1e: // HW_RET.
97 case 0x1f: // HW_ST.
98 return AlphaPALInstructionAccess::PalOrKernelHwe;
99 default:
100 return AlphaPALInstructionAccess::Denied;
101 }
102}
