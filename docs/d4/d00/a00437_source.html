146bx_gui_c& bx_gui_c::display_for_output(const std::string& device_path,
147 unsigned output_id)
148{
149 if (device_path.empty())
150 FAILURE(Configuration, "Display output requires a device identity");
151 if (output_bound)
152 {
153 if (output_device_path != device_path || bound_output_id != output_id)
154 FAILURE(Configuration, "GUI backend supports only one display output");
155 return *this;
156 }
157 output_device_path = device_path;
158 bound_output_id = output_id;
159 output_bound = true;
160 return *this;
161}

262bx_svga_tileinfo_t* bx_gui_c::graphics_tile_info(bx_svga_tileinfo_t* info)
263{
264 if (!info)
265 {
266 info = (bx_svga_tileinfo_t*)malloc(sizeof(bx_svga_tileinfo_t));
267 if (!info)
268 {
269 return NULL;
270 }
271 }
272
273 host_pitch = host_xres * ((host_bpp + 1) >> 3);
274
275 info->bpp = host_bpp;
276 info->pitch = host_pitch;
277 switch (info->bpp)
278 {
279 case 15:
280 info->red_shift = 15;
281 info->green_shift = 10;
282 info->blue_shift = 5;
283 info->red_mask = 0x7c00;
284 info->green_mask = 0x03e0;
285 info->blue_mask = 0x001f;
286 break;
287
288 case 16:
289 info->red_shift = 16;
290 info->green_shift = 11;
291 info->blue_shift = 5;
292 info->red_mask = 0xf800;
293 info->green_mask = 0x07e0;
294 info->blue_mask = 0x001f;
295 break;
296
297 case 24:
298 case 32:
299 info->red_shift = 24;
300 info->green_shift = 16;
301 info->blue_shift = 8;
302 info->red_mask = 0xff0000;
303 info->green_mask = 0x00ff00;
304 info->blue_mask = 0x0000ff;
305 break;
306 }
307
308 info->is_indexed = (host_bpp == 8);
309#ifdef BX_LITTLE_ENDIAN
310 info->is_little_endian = 1;
311#else
312 info->is_little_endian = 0;
313#endif
314 return info;
315}

317u8* bx_gui_c::graphics_tile_get(unsigned x0, unsigned y0, unsigned* w,
318 unsigned* h)
319{
320 if (x0 + X_TILESIZE > host_xres)
321 {
322 *w = host_xres - x0;
323 }
324 else
325 {
326 *w = X_TILESIZE;
327 }
328
329 if (y0 + Y_TILESIZE > host_yres)
330 {
331 *h = host_yres - y0;
332 }
333 else
334 {
335 *h = Y_TILESIZE;
336 }
337
338 return (u8*)framebuffer + y0 * host_pitch + x0 * ((host_bpp + 1) >> 3);
339}

341void bx_gui_c::graphics_tile_update_in_place(unsigned x0, unsigned y0,
342 unsigned w, unsigned h)
343{
344 u8 tile[X_TILESIZE * Y_TILESIZE * 4];
345 u8* tile_ptr;
346 u8* fb_ptr;
347 u16 xc;
348 u16 yc;
349 u16 fb_pitch;
350 u16 tile_pitch;
351 u8 r;
352 u8 diffx;
353 u8 diffy;
354
355 diffx = (x0 % X_TILESIZE);
356 diffy = (y0 % Y_TILESIZE);
357 if (diffx > 0)
358 {
359 x0 -= diffx;
360 w += diffx;
361 }
362
363 if (diffy > 0)
364 {
365 y0 -= diffy;
366 h += diffy;
367 }
368
369 fb_pitch = host_pitch;
370 tile_pitch = X_TILESIZE * ((host_bpp + 1) >> 3);
371 for (yc = y0; yc < (y0 + h); yc += Y_TILESIZE)
372 {
373 for (xc = x0; xc < (x0 + w); xc += X_TILESIZE)
374 {
375 fb_ptr = framebuffer + (yc * fb_pitch + xc * ((host_bpp + 1) >> 3));
376 tile_ptr = &tile[0];
377 for (r = 0; r < h; r++)
378 {
379 memcpy(tile_ptr, fb_ptr, tile_pitch);
380 fb_ptr += fb_pitch;
381 tile_ptr += tile_pitch;
382 }
383
384 graphics_tile_update(tile, xc, yc);
385 }
386 }
387}

389void bx_gui_c::graphics_frame_update(const u32* pixels,
390 unsigned width, unsigned height)
391{
392 u8 tile[X_TILESIZE * Y_TILESIZE * 4];
393
394 for (unsigned y = 0; y < height; y += Y_TILESIZE)
395 {
396 for (unsigned x = 0; x < width; x += X_TILESIZE)
397 {
398 const unsigned tw = (x + X_TILESIZE <= width)
399 ? (unsigned)X_TILESIZE : (width - x);
400 const unsigned th = (y + Y_TILESIZE <= height)
401 ? (unsigned)Y_TILESIZE : (height - y);
402
403 u8* dst = tile;
404 for (unsigned row = 0; row < th; row++)
405 {
406 const u32* src_row = pixels + (size_t)(y + row) * width + x;
407 memcpy(dst, src_row, tw * sizeof(u32));
408 // Pad partial tile columns with black
409 if (tw < (unsigned)X_TILESIZE)
410 memset(dst + tw * 4, 0, ((unsigned)X_TILESIZE - tw) * 4);
411 dst += X_TILESIZE * 4;
412 }
413 // Pad partial tile rows with black
414 for (unsigned row = th; row < (unsigned)Y_TILESIZE; row++)
415 {
416 memset(dst, 0, X_TILESIZE * 4);
417 dst += X_TILESIZE * 4;
418 }
419
420 graphics_tile_update(tile, x, y);
421 }
422 }
