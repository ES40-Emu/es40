109void CSystemComponent::add_state_field(StateField field)
110{
111 if (m_state_registration_closed)
112 throw std::logic_error("State registration is closed");
113 if (field.name.empty() || field.name.size() > 65536 ||
114 m_state_fields.size() == (std::numeric_limits<u32>::max)())
115 throw std::invalid_argument("Invalid saved field name or count");
116 for (const auto& existing : m_state_fields)
117 if (existing.name == field.name && existing.index == field.index)
118 throw std::invalid_argument("Duplicate saved field: " + field.name);
119 m_state_fields.push_back(std::move(field));
120}

124bool CSystemComponent::save_registered_state(FILE* f) const
125{
126 if (!f)
127 return false;
128 m_state_registration_closed = true;
129 const auto write = [f](const void* data, size_t size) {
130 return !size || fwrite(data, size, 1, f) == 1;
131 };
132 const u32 count = u32(m_state_fields.size());
133 if (!write(&registered_state_magic, sizeof(u32)) || !write(&count, sizeof(count)))
134 return false;
135 for (const auto& field : m_state_fields)
136 {
137 const auto* source = static_cast<const u8*>(field.address());
138 if (field.count && !source)
139 return false;
140 const u32 name_size = u32(field.name.size());
141 if (!write(&name_size, sizeof(name_size)) || !write(field.name.data(), name_size) ||
142 !write(&field.index, sizeof(field.index)) || !write(&field.type, sizeof(field.type)) ||
143 !write(&field.element_size, sizeof(field.element_size)) || !write(&field.count, sizeof(field.count)))
144 return false;
145 for (u32 block = 0; block < field.blocks; ++block)
146 {
147 if (field.type == 1)
148 {
149 const auto* values = reinterpret_cast<const bool*>(source);
150 for (size_t i = 0; i < field.block_size / sizeof(bool); ++i)
151 {
152 const u8 value = values[i] ? 1 : 0;
153 if (!write(&value, sizeof(value)))
154 return false;
155 }
156 }
157 else if (!write(source, field.block_size))
158 return false;
159 if (block + 1 < field.blocks)
160 source += field.stride;
161 }
162 }
163 return true;
164}

166bool CSystemComponent::read_registered_state(FILE* f, RegisteredState& saved) const
167{
168 saved.clear();
169 if (!f)
170 return false;
171 m_state_registration_closed = true;
172 const auto read = [f](void* data, size_t size) {
173 return !size || fread(data, size, 1, f) == 1;
174 };
175 u32 magic = 0, count = 0;
176 if (!read(&magic, sizeof(magic)) || magic != registered_state_magic ||
177 !read(&count, sizeof(count)) || count != m_state_fields.size())
178 return false;
179 RegisteredState pending;
180 pending.reserve(count);
181 for (const auto& field : m_state_fields)
182 {
183 u32 name_size = 0, type = 0, element_size = 0, elements = 0;
184 s32 index = 0;
185 if (!read(&name_size, sizeof(name_size)) || name_size != field.name.size())
186 return false;
187 std::string name(name_size, '\0');
188 if (!read(name.data(), name.size()) || name != field.name ||
189 !read(&index, sizeof(index)) || index != field.index ||
190 !read(&type, sizeof(type)) || type != field.type ||
191 !read(&element_size, sizeof(element_size)) || element_size != field.element_size ||
192 !read(&elements, sizeof(elements)) || elements != field.count)
193 return false;
194 pending.emplace_back(field.size());
195 auto& bytes = pending.back();
196 if (!read(bytes.data(), bytes.size()))
197 return false;
198 if (field.type == 1)
199 for (u8 value : bytes)
200 if (value > 1)
201 return false;
202 }
203 saved = std::move(pending);
204 return true;
205}

207bool CSystemComponent::commit_registered_state(const RegisteredState& saved) const
208{
209 if (saved.size() != m_state_fields.size())
210 return false;
211 // Validate all destinations before modifying state.
212 std::vector<u8*> destinations;
213 destinations.reserve(m_state_fields.size());
214 for (size_t i = 0; i < m_state_fields.size(); ++i)
215 {
216 const auto& field = m_state_fields[i];
217 auto* address = static_cast<u8*>(field.address());
218 if ((field.count && !address) || saved[i].size() != field.size())
219 return false;
220 if (field.type == 1)
221 for (u8 value : saved[i])
222 if (value > 1)
223 return false;
224 destinations.push_back(address);
225 }
226 for (size_t i = 0; i < m_state_fields.size(); ++i)
227 {
228 const auto& field = m_state_fields[i];
229 auto* destination = destinations[i];
230 const auto* source = saved[i].data();
231 for (u32 block = 0; block < field.blocks; ++block)
232 {
233 if (field.type == 1)
234 {
235 auto* values = reinterpret_cast<bool*>(destination);
236 for (size_t j = 0; j < field.block_size / sizeof(bool); ++j)
237 values[j] = *source++ != 0;
238 }
239 else
240 {
241 memcpy(destination, source, field.block_size);
242 source += field.block_size;
243 }
244 if (block + 1 < field.blocks)
245 destination += field.stride;
246 }
247 }
248 return true;
249}
