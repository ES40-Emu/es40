141 void layout()
142 {
143 RECT bounds = { 0, 0, 640, 480 };
144 GetClientRect(parent, &bounds);
145 width = (std::max)(260, (std::min)(560, (int)bounds.right - 24));
146 row_height = width >= 480 ? 58 : 44;
147 header_height = width >= 480 ? 72 : 56;
148 footer_height = width >= 480 ? 38 : 28;
149 int available = (std::max)(1, ((int)bounds.bottom - 24 - header_height - footer_height) / row_height);
150 int visible = (std::max)(1, (std::min)(item_count(), (std::min)(available, 8)));
151 height = header_height + footer_height + row_height * visible;
152 if (selected < item_count())
153 {
154 if (selected < first_visible) first_visible = selected;
155 if (selected >= first_visible + visible) first_visible = selected - visible + 1;
156 }
157 first_visible = (std::max)(0, (std::min)(first_visible, (std::max)(0, item_count() - visible)));
158 rows.clear();
159 for (int i = first_visible; i < (std::min)(item_count(), first_visible + visible); ++i)
160 {
161 int y = header_height + (i - first_visible) * row_height;
162 rows.push_back({ 14, y, width - 14, y + row_height - 6 });
163 }
164 reset_button = { width - 110, height - footer_height + 3, width - 14, height - 4 };
165 }

179 void create_popup()
180 {
181 if (stopping || !IsWindow(parent)) { close_popup(false); return; }
182 WNDCLASSA wc = {};
183 wc.lpfnWndProc = window_proc; wc.hInstance = GetModuleHandle(nullptr);
184 wc.hCursor = LoadCursor(nullptr, IDC_ARROW); wc.lpszClassName = media_class;
185 if (!RegisterClassA(&wc))
186 {
187 const DWORD error = GetLastError();
188 if (error != ERROR_CLASS_ALREADY_EXISTS)
189 FAILURE_1(Win32, "Unable to register media selector window class (error %lu)", error);
190 }
191 layout();
192 window = CreateWindowExA(WS_EX_TOOLWINDOW, media_class, "ES40 Media", WS_POPUP,
193 0, 0, width, height, parent, nullptr, wc.hInstance, this);
194 if (!window) FAILURE_1(Win32, "Unable to create media selector window (error %lu)", GetLastError());
195 update();
196 // Native callbacks can already have dismissed a failed popup.
197 if (window) ShowWindow(window, SW_SHOW);
198 if (window) SetFocus(window);
199 }

221 static UINT_PTR CALLBACK dialog_hook(HWND hwnd, UINT msg, WPARAM, LPARAM lparam)
222 {
223 auto* self = reinterpret_cast<state*>(GetWindowLongPtr(hwnd, GWLP_USERDATA));
224 if (msg == WM_INITDIALOG)
225 {
226 auto* options = reinterpret_cast<OPENFILENAMEA*>(lparam);
227 self = reinterpret_cast<state*>(options->lCustData);
228 SetWindowLongPtr(hwnd, GWLP_USERDATA, (LONG_PTR)self);
229 self->dialog = GetParent(hwnd);
230 if (self->stopping || self->cancel_dialog)
231 PostMessage(self->dialog, WM_COMMAND, IDCANCEL, 0);
232 }
233 else if (msg == WM_DESTROY && self) self->dialog = nullptr;
234 return 0;

237 void file_dialog(std::shared_ptr<CDiskFileMediaMailbox> mailbox,
238 bool force = false, size_t blank_size = 0)
239 {
240 if (!mailbox || dialog_open || stopping || !IsWindow(parent)) return;
241 // Nested native modal loops can dispatch emulator teardown. Retain this
242 // state and an independent hidden owner until the common dialog returns.
243 auto keep_alive = shared_from_this();
244 std::vector<char> path(32768, 0);
245 if (blank_size) lstrcpyA(path.data(), "blank.img");
246 RECT bounds = {}; GetWindowRect(parent, &bounds);
247 HWND owner = CreateWindowExA(WS_EX_TOOLWINDOW, "STATIC", "ES40 media selector", WS_POPUP,
248 bounds.left, bounds.top, bounds.right - bounds.left, bounds.bottom - bounds.top,
249 nullptr, nullptr, GetModuleHandle(nullptr), nullptr);
250 if (!owner) { show_message("Could not prepare the file dialog."); return; }
251 struct dialog_owner
252 {
253 state& selector;
254 HWND window;
255 void finish()
256 {
257 selector.dialog = nullptr;
258 if (window) { DestroyWindow(window); window = nullptr; }
259 selector.dialog_open = false;
260 }
261 ~dialog_owner() { finish(); }
262 } owner_guard = { *this, owner };
263 dialog_open = true; cancel_dialog = false;
264 close_popup(false);
265 OPENFILENAMEA options = {};
266 options.lStructSize = sizeof(options); options.hwndOwner = owner;
267 options.lpstrFilter = mailbox->is_floppy() ? floppy_filters : cd_filters;
268 options.nFilterIndex = 1; options.lpstrFile = path.data(); options.nMaxFile = (DWORD)path.size();
269 options.lpstrTitle = blank_size ? "Create blank floppy image" : "Change removable media";
270 options.lpstrDefExt = blank_size ? "img" : nullptr;
271 options.Flags = OFN_EXPLORER | OFN_ENABLEHOOK | OFN_ENABLESIZING | OFN_HIDEREADONLY |
272 OFN_NOCHANGEDIR | OFN_PATHMUSTEXIST | (blank_size ? OFN_OVERWRITEPROMPT : OFN_FILEMUSTEXIST);
273 options.lpfnHook = dialog_hook; options.lCustData = (LPARAM)this;
274 BOOL accepted = blank_size ? GetSaveFileNameA(&options) : GetOpenFileNameA(&options);
275 DWORD error = accepted ? 0 : CommDlgExtendedError();
276 owner_guard.finish();
277 swallow_pointer = swallow_pointer || pointer_down();
278 if (release_input) release_input();
279 if (stopping || cancel_dialog || !IsWindow(parent)) return;
280 SetActiveWindow(parent); SetFocus(parent);
281 if (error) { show_message("Could not open file selector (error " + std::to_string(error) + ")."); return; }
282 if (!accepted || !path[0]) return;
283 if (blank_size) queued(mailbox->request_blank_floppy(path.data(), blank_size));
284 else if (!mailbox->is_floppy() && mailbox->media_locked() && !force)
285 {
286 devices = disk_file_removable_media(); device = mailbox; device_selection = 0;
287 auto found = std::find(devices.begin(), devices.end(), mailbox);
288 if (found != devices.end()) device_selection = (int)(found - devices.begin());
289 locked(false, path.data());
290 }
291 else queued(mailbox->request_image_change(path.data(), force));
292 }

294 void back()
295 {
296 if (page == mode::faults)
297 {
298 page = mode::devices; disks.clear(); selected = (int)devices.size();
299 }
300 else if (page == mode::size) { page = mode::floppy; selected = 1; }
301 else if (page == mode::locked)
302 {
303 page = mode::cdrom; selected = locked_selection;
304 pending_path.clear(); pending_eject = false;
305 }
306 else if (page == mode::floppy || page == mode::cdrom)
307 {
308 page = mode::devices; device.reset(); pending_path.clear();
309 selected = (std::min)(device_selection, item_count() - 1);
310 }
311 else { close_popup(); return; }
312 first_visible = 0; update();
313 }
314
315 void activate()
316 {
317 if (page == mode::devices)
318 {
319 if (selected < (int)devices.size())
320 {
321 device_selection = selected; device = devices[selected];
322 if (!device) { close_popup(); return; }
323 page = device->is_floppy() ? mode::floppy : mode::cdrom;
324 selected = first_visible = 0; update();
325 }
326 else if (selected == (int)devices.size())
327 {
328 disks = disk_fault_controls(); page = mode::faults;
329 selected = first_visible = 0; update();
330 }
331 else
332 {
333 const bool reset_selected = selected == item_count();
334 auto request_reset = reset;
335 close_popup();
336 if (reset_selected && request_reset) request_reset();
337 }
338 }
339 else if (page == mode::faults)
340 {
341 if (selected < (int)disks.size())
342 {
343 auto disk = disks[selected];
344 if (disk->set_offline(!disk->is_offline())) update();
345 else show_message("The selected disk is no longer available.");
346 }
347 else back();
348 }
349 else if (page == mode::floppy && device)
350 {
351 if (selected == 0) file_dialog(device);
352 else if (selected == 1) { page = mode::size; selected = 3; first_visible = 0; update(); }
353 else if (selected == 2) queued(device->request_eject());
354 else if (selected == 3) queued(device->request_read_only_toggle());
355 else back();
356 }
357 else if (page == mode::size && device)
358 {
359 if (selected < 5) file_dialog(device, false, blank_formats[selected].bytes);
360 else back();
361 }
362 else if (page == mode::cdrom && device)
363 {
364 if (selected < 2)
365 {
366 if (device->media_locked()) locked(selected == 1);
367 else if (selected == 0) file_dialog(device);
368 else queued(device->request_eject());
369 }
370 else back();
371 }
372 else if (page == mode::locked && device)
373 {
374 if (selected == 0)
375 {
376 if (pending_eject) queued(device->request_eject(true));
377 else if (pending_path.empty()) file_dialog(device, true);
378 else queued(device->request_image_change(pending_path.c_str(), true));
379 }
380 else back();
381 }
382 else close_popup();
383 }

409 void draw(HDC dc)
410 {
411 if (!title_font) title_font = CreateFontA(-20, 0, 0, 0, FW_SEMIBOLD, FALSE, FALSE, FALSE,
412 DEFAULT_CHARSET, OUT_DEFAULT_PRECIS, CLIP_DEFAULT_PRECIS, DEFAULT_QUALITY, DEFAULT_PITCH, "Segoe UI");
413 if (!body_font) body_font = CreateFontA(-16, 0, 0, 0, FW_NORMAL, FALSE, FALSE, FALSE,
414 DEFAULT_CHARSET, OUT_DEFAULT_PRECIS, CLIP_DEFAULT_PRECIS, DEFAULT_QUALITY, DEFAULT_PITCH, "Segoe UI");
415 if (!detail_font) detail_font = CreateFontA(-13, 0, 0, 0, FW_NORMAL, FALSE, FALSE, FALSE,
416 DEFAULT_CHARSET, OUT_DEFAULT_PRECIS, CLIP_DEFAULT_PRECIS, DEFAULT_QUALITY, DEFAULT_PITCH, "Segoe UI");
417 RECT all = { 0, 0, width, height };
418 FillRect(dc, &all, GetSysColorBrush(COLOR_WINDOW));
419 FrameRect(dc, &all, GetSysColorBrush(COLOR_WINDOWFRAME));
420 std::string title = "Disk and media controls", subtitle;
421 switch (page)
422 {
423 case mode::devices: subtitle = "Choose a drive or manage disk failures"; break;
424 case mode::faults:
425 title = "Disk failures";
426 subtitle = disks.empty() ? "No disks are configured." : "Select a disk to toggle offline / online";
427 break;
428 case mode::floppy: title = "Floppy media"; subtitle = device->label(); break;
429 case mode::size: title = "Create blank floppy image"; subtitle = "Choose an unformatted raw-image capacity"; break;
430 case mode::cdrom: title = "CD-ROM media"; subtitle = device->label(); break;
431 case mode::locked:
432 title = "CD-ROM media locked";
433 subtitle = pending_eject ? "The guest locked this CD-ROM; eject requires an override." :
434 (!pending_path.empty() ? "The guest locked this CD-ROM while the file selector was open." :
435 (device ? device->label() + " is locked by the guest." : "The selected CD-ROM is locked by the guest."));
436 break;
437 default: subtitle = message; break;
438 }
439 text(dc, { 20, 9, width - 20, header_height - 26 }, title, title_font, GetSysColor(COLOR_WINDOWTEXT));
440 text(dc, { 20, header_height - 26, width - 20, header_height - 5 }, subtitle, detail_font, GetSysColor(COLOR_WINDOWTEXT));
441 for (size_t i = 0; i < rows.size(); ++i)
442 {
443 int index = first_visible + (int)i;
444 RECT row = rows[i];
445 bool active = index == selected;
446 FillRect(dc, &row, GetSysColorBrush(active ? COLOR_HIGHLIGHT : COLOR_BTNFACE));
447 COLORREF color = GetSysColor(active ? COLOR_HIGHLIGHTTEXT : COLOR_BTNTEXT);
448 std::string label, image;
449 if (page == mode::devices)
450 {
451 if (index < (int)devices.size())
452 {
453 label = devices[index]->label();
454 image = "Image: " + devices[index]->mounted_image_name();
455 }
456 else label = index == (int)devices.size() ? "Disk failures..." : "Cancel";
457 }
458 else if (page == mode::faults)
459 {
460 if (index < (int)disks.size())
461 {
462 label = disks[index]->label();
463 image = disks[index]->is_offline() ?
464 "Offline (failed) - select to bring online" :
465 "Online - select to take offline";
466 }
467 else label = "Back";
468 }
469 else if (page == mode::floppy)
470 {
471 const char* labels[] = { "Change image...", "Create blank image...", "Eject", "", "Back" };
472 label = index == 3 ? (device->displayed_read_only() ? "Make writable" : "Make read-only") : labels[index];
473 }
474 else if (page == mode::size) label = index < 5 ? blank_formats[index].label : "Back";
475 else if (page == mode::cdrom)
476 {
477 const char* labels[] = { "Change image...", "Eject", "Back" }; label = labels[index];
478 }
479 else if (page == mode::locked) label = index == 0 ? (pending_eject ? "Force eject anyway" : "Force change anyway...") : "Back";
480 else label = "Close";
481 row.left += 12; row.right -= 12;
482 if (image.empty()) text(dc, row, label, body_font, color);
483 else
484 {
485 RECT line = row; line.bottom = line.top + (row.bottom - row.top) / 2;
486 text(dc, line, label, body_font, color);
487 line.top = line.bottom; line.bottom = row.bottom;
488 text(dc, line, image, detail_font, color);
489 }
490 }
491 RECT footer = { 20, height - footer_height, width - 20, height };
492 if (page == mode::devices)
493 {
494 bool active = selected == item_count();
495 FillRect(dc, &reset_button, GetSysColorBrush(active ? COLOR_HIGHLIGHT : COLOR_BTNFACE));
496 text(dc, reset_button, "Reset VM", detail_font,
497 GetSysColor(active ? COLOR_HIGHLIGHTTEXT : COLOR_BTNTEXT), DT_CENTER | DT_VCENTER);
498 footer.right = reset_button.left - 8;
499 }
500 text(dc, footer, width >= 480 ? "Arrows/Tab Enter Esc" : "Tab Enter Esc", detail_font, GetSysColor(COLOR_WINDOWTEXT));
501 }

503 LRESULT event(HWND hwnd, UINT msg, WPARAM wparam, LPARAM lparam)
504 {
505 if ((msg == WM_KEYDOWN || msg == WM_KEYUP || msg == WM_SYSKEYDOWN || msg == WM_SYSKEYUP) &&
506 key_handler && key_handler(msg, wparam, lparam)) return 0;
507 switch (msg)
508 {
509 case WM_KEYDOWN: case WM_SYSKEYDOWN:
510 switch (wparam)
511 {
512 case VK_UP: move(-1); break;
513 case VK_DOWN: move(1); break;
514 case VK_TAB: move(GetKeyState(VK_SHIFT) & 0x8000 ? -1 : 1); break;
515 case VK_HOME: selected = 0; update(); break;
516 case VK_END: selected = selection_count() - 1; update(); break;
517 case VK_RETURN: case VK_SPACE: if (!(lparam & (1L << 30))) activate(); break;
518 case VK_ESCAPE: if (!(lparam & (1L << 30))) back(); break;
519 case VK_F4: if (GetKeyState(VK_MENU) & 0x8000) close_popup(); break;
520 }
521 return 0;
522 case WM_KEYUP: case WM_SYSKEYUP: case WM_CHAR: case WM_SYSCHAR: return 0;
523 case WM_MOUSEWHEEL: if (GET_WHEEL_DELTA_WPARAM(wparam)) move(GET_WHEEL_DELTA_WPARAM(wparam) > 0 ? -1 : 1); return 0;
524 case WM_MOUSEMOVE: case WM_LBUTTONUP:
525 {
526 int index = hit(GET_X_LPARAM(lparam), GET_Y_LPARAM(lparam));
527 if (index >= 0)
528 {
529 if (selected != index) { selected = index; update(); }
530 if (msg == WM_LBUTTONUP) activate();
531 }
532 return 0;
533 }
534 case WM_SYSCOMMAND:
535 // The focused media popup needs the same display-power policy as
536 // guest windows; the application owns the execution-state request.
537 if ((wparam & 0xfff0) == SC_SCREENSAVE || (wparam & 0xfff0) == SC_MONITORPOWER)
538 return 0;
539 break;
540 case WM_CLOSE: close_popup(); return 0;
541 case WM_KILLFOCUS:
542 swallow_pointer = swallow_pointer || pointer_down();
543 close_popup(false); return 0;
544 case WM_SHOWWINDOW: if (!wparam && window == hwnd) close_popup(false); return 0;
545 case WM_ERASEBKGND: return 1;
546 case WM_PAINT:
547 {
548 PAINTSTRUCT paint; HDC dc = BeginPaint(hwnd, &paint);
549 HDC buffer = CreateCompatibleDC(dc);
550 HBITMAP bitmap = buffer ? CreateCompatibleBitmap(dc, width, height) : nullptr;
551 HGDIOBJ old = bitmap ? SelectObject(buffer, bitmap) : nullptr;
552 try { draw(bitmap ? buffer : dc); }
553 catch (...)
554 {
555 if (bitmap) { SelectObject(buffer, old); DeleteObject(bitmap); }
556 if (buffer) DeleteDC(buffer);
557 EndPaint(hwnd, &paint); throw;
558 }
559 if (bitmap)
560 {
561 BitBlt(dc, 0, 0, width, height, buffer, 0, 0, SRCCOPY);
562 SelectObject(buffer, old); DeleteObject(bitmap);
563 }
564 if (buffer) DeleteDC(buffer);
565 EndPaint(hwnd, &paint); return 0;
566 }
567 }
568 return DefWindowProcA(hwnd, msg, wparam, lparam);
569 }

571 static LRESULT CALLBACK window_proc(HWND hwnd, UINT msg, WPARAM wparam, LPARAM lparam)
572 {
