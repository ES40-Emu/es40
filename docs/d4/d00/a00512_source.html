37inline bool x11_hotkey_equal(const std::string& text, const char* name)
38{
39 if(!name || text.size() != std::char_traits<char>::length(name)) return false;
40 for(std::size_t i = 0; i < text.size(); ++i)
41 if(std::tolower(static_cast<unsigned char>(text[i])) !=
42 std::tolower(static_cast<unsigned char>(name[i]))) return false;
43 return true;

48inline std::string trim(const std::string& text)
49{
50 std::size_t first = 0, last = text.size();
51 while(first < last && std::isspace(static_cast<unsigned char>(text[first]))) ++first;
52 while(last > first && std::isspace(static_cast<unsigned char>(text[last - 1]))) --last;
53 return text.substr(first, last - first);
54}

56inline unsigned modifier(const std::string& text)
57{
58 if(x11_hotkey_equal(text, "Ctrl") || x11_hotkey_equal(text, "Control")) return X11_HOTKEY_CTRL;
59 if(x11_hotkey_equal(text, "Alt") || x11_hotkey_equal(text, "Option")) return X11_HOTKEY_ALT;
60 if(x11_hotkey_equal(text, "Shift")) return X11_HOTKEY_SHIFT;
61 const char* names[] = { "GUI", "Super", "Win", "Windows", "Cmd", "Command", "Meta" };
62 for(const char* name : names)
63 if(x11_hotkey_equal(text, name)) return X11_HOTKEY_GUI;
64 return 0;
65}

68template<class Resolve>
69x11_hotkey_binding parse_x11_hotkey(const char* configured, Resolve resolver)
70{
71 x11_hotkey_binding binding;
72 binding.display = "none";
73 std::string key_name = x11_hotkey_detail::trim(configured ? configured : "");
74 if(x11_hotkey_equal(key_name, "none")) return binding;
75 binding.invalid = true;
76 unsigned modifiers = 0;
77 for(;;)
78 {
79 const std::size_t separator = key_name.find('+');
80 if(separator == std::string::npos) break;
81 const unsigned modifier = x11_hotkey_detail::modifier(
82 x11_hotkey_detail::trim(key_name.substr(0, separator)));
83 if(!modifier) break;
84 if(modifiers & modifier) return binding;
85 modifiers |= modifier;
86 key_name = x11_hotkey_detail::trim(key_name.substr(separator + 1));
87 }
88 if(key_name.empty() || x11_hotkey_detail::modifier(key_name)) return binding;
89 const auto resolved = resolver(key_name);
90 if(!resolved.key) return binding;
91 binding.key = resolved.key;
92 binding.modifiers = modifiers;
93 binding.enabled = true;
94 binding.invalid = false;
95 binding.display.clear();
96 if(modifiers & X11_HOTKEY_CTRL) binding.display += "Ctrl+";
97 if(modifiers & X11_HOTKEY_ALT) binding.display += "Alt+";
98 if(modifiers & X11_HOTKEY_SHIFT) binding.display += "Shift+";
99 if(modifiers & X11_HOTKEY_GUI) binding.display += "GUI+";
100 binding.display += resolved.name;
101 return binding;
102}
