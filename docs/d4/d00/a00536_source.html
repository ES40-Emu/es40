53static inline bool serial_parse_listen_addresses(const std::string &value,
54 std::vector<std::string> &addresses, std::string &error)
55{
56 addresses.clear();
57 error.clear();
58 if (value.find_first_not_of(" \t\r\n") == std::string::npos) {
59 addresses.push_back("0.0.0.0");
60 addresses.push_back("::");
61 return true;
62 }
63 size_t start = 0;
64 for (;;) {
65 const size_t end = value.find(',', start);
66 std::string address = value.substr(start,
67 end == std::string::npos ? std::string::npos : end - start);
68 const size_t first = address.find_first_not_of(" \t\r\n");
69 if (first == std::string::npos) {
70 error = "Empty address in listen_address";
71 addresses.clear();
72 return false;
73 }
74 address = address.substr(first,
75 address.find_last_not_of(" \t\r\n") - first + 1);
76 addrinfo hints = {};
77 hints.ai_family = AF_UNSPEC;
78 hints.ai_socktype = SOCK_STREAM;
79 hints.ai_flags = AI_NUMERICHOST;
80 addrinfo *resolved = nullptr;
81 char canonical[NI_MAXHOST];
82#if defined(_WIN32)
83 WSADATA wsa;
84 const int startup_error = WSAStartup(MAKEWORD(2, 2), &wsa);
85 if (startup_error != 0) {
86 error = "Winsock startup failed (error " + std::to_string(startup_error) + ")";
87 addresses.clear();
88 return false;
89 }
90#endif
91 int result = address.find("]:") != std::string::npos
92 ? EAI_NONAME : getaddrinfo(address.c_str(), nullptr, &hints, &resolved);
93 if (result == 0) {
94 if (resolved->ai_family == AF_INET &&
95 ((sockaddr_in *)resolved->ai_addr)->sin_addr.s_addr == INADDR_BROADCAST)
96 result = EAI_NONAME;
97 else
98 result = getnameinfo(resolved->ai_addr, (socklen_t)resolved->ai_addrlen,
99 canonical, sizeof(canonical), nullptr, 0, NI_NUMERICHOST);
100 }
101 if (resolved)
102 freeaddrinfo(resolved);
103#if defined(_WIN32)
104 WSACleanup();
105#endif
106 if (result != 0) {
107 error = "Invalid IPv4/IPv6 address: \"" + address + "\"";
108 addresses.clear();
109 return false;
110 }
111 const std::string canonical_address = canonical;
112 if (std::find(addresses.begin(), addresses.end(), canonical_address) ==
113 addresses.end())
114 addresses.push_back(canonical_address);
115 if (end == std::string::npos)
116 return true;
117 start = end + 1;
118 }
119}
