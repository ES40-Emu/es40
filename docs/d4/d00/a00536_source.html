56static uint32_t radeon_setting(
57 CConfigurator* cfg, const char* key, uint32_t def)
58{
59 const char* text = cfg->get_text_value(key, nullptr);
60 if (!text || !*text)
61 return def;
62 char* end = nullptr;
63 errno = 0;
64 const unsigned long long n = std::strtoull(text, &end, 0);
65 if (errno || *end || text[0] == '-' || n > 0xffffffffULL)
66 throw std::runtime_error(
67 std::string("Invalid radeon integer setting: ") + key);
68 return static_cast<uint32_t>(n);
69}

92static void radeon_write_u32(FILE* f, uint32_t v)
93{
94 uint8_t b[4];
95 for (unsigned i = 0; i < 4; ++i)
96 b[i] = uint8_t(v >> (8 * i));
97 if (fwrite(b, 1, 4, f) != 4)
98 throw std::runtime_error("Snapshot write failed");
99}

101static uint32_t radeon_read_u32(FILE* f)
102{
103 uint8_t b[4];
104 if (fread(b, 1, 4, f) != 4)
105 throw std::runtime_error("Snapshot read failed");
106 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
107 (uint32_t(b[3]) << 24);
108}

110static bool radeon_vga_window_offset(u32& address, unsigned selection)
111{
112 switch (selection & 3)
113 {
114 case 1:
115 return address < 0x10000;
116 case 2:
117 if (address < 0x10000 || address >= 0x18000)
118 return false;
119 address -= 0x10000;
120 break;
121 case 3:
122 if (address < 0x18000 || address >= 0x20000)
123 return false;
124 address -= 0x18000;
125 break;
126 default:
127 return address < 0x20000;
128 }
129 return true;
130}

142void CRadeon7500::sequencer_map(address_map& map)
143{
144 unimplemented_map(map, "VGA sequencer register");
145 // These frontends mask stored bytes, including direct snapshot writes.
146 // Limit the canonical map to SR00-04 so unknown indices keep diagnostics.
147 auto standard = std::make_shared<address_map>(5);
148 vga_device::sequencer_map(*standard);
149 map(0x00, 0x04).lrw8(
150 NAME([standard](offs_t index) {
151 return standard->read_byte(index);
152 }),
153 NAME(([this, standard](offs_t index, u8 data) {
154 static constexpr u8 masks[] = {0x03, 0x3f, 0x0f, 0x3f, 0x0e};
155 vga.sequencer.data[index] = data & masks[index];
156 standard->write_byte(index, data);
157 })));
158}

166void CRadeon7500::attribute_map(address_map& map)
167{
168 unimplemented_map(map, "VGA attribute register");
169 vga_device::attribute_map(map);
170 // Palette reads remain available when PAS disables CPU writes.
171 map(0x20, 0x2f).lr8(NAME([this](offs_t offset) {
172 return vga.attribute.data[offset];
173 }));
174 // These frontends retain mode-control bits 6/7 and mask reserved bit 4.
175 map(0x10, 0x10).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
176 vga.attribute.data[0x10] = data & 0xef;
177 }));
178 // Color select does not change the pending horizontal panning value.
179 map(0x14, 0x14).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
180 vga.attribute.data[0x14] = data & 0xf;
181 }));
182}

201u8 CRadeon7500::mem_r(offs_t address)
202{
203 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
204 return 0xff;
205 unsigned selected = vga.gc.read_map_sel & 3;
206 if (vga.sequencer.data[4] & 8)
207 {
208 // Keep packed chain-four bytes in the layout consumed by vga_device scanout.
209 for (unsigned plane = 0; plane < 4; ++plane)
210 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
211 selected = address & 3;
212 }
213 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
214 {
215 // vga_device text stores character/attribute pairs together; fonts stay planar.
216 vga.gc.latch[0] = vga.memory[address & ~1u];
217 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
218 for (unsigned plane = 2; plane < 4; ++plane)
219 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
220 selected = (vga.gc.read_map_sel & 2) | (address & 1);
221 }
222 else
223 {
224 for (unsigned plane = 0; plane < 4; ++plane)
225 vga.gc.latch[plane] =
226 vga.memory[(address & 0xffff) + plane * 0x10000];
227 }
228 if (!vga.gc.read_mode)
229 return vga.gc.latch[selected];
230 u8 result = 0xff;
231 for (unsigned plane = 0; plane < 4; ++plane)
232 if (vga.gc.color_dont_care & (1 << plane))
233 result &=
234 u8(vga.gc.latch[plane] ^
235 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
236 return result;
237}

239void CRadeon7500::mem_w(offs_t address, u8 data)
240{
241 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
242 return;
243 if (vga.sequencer.data[4] & 8)
244 {
245 if (vga.sequencer.map_mask & (1 << (address & 3)))
246 vga.memory[address] = vga_latch_write(address & 3, data);
247 return;
248 }
249 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
250 {
251 const unsigned parity = address & 1;
252 if (vga.sequencer.map_mask & (1 << parity))
253 vga.memory[address] = vga_latch_write(parity, data);
254 if (vga.sequencer.map_mask & (1 << (parity + 2)))
255 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
256 vga_latch_write(parity + 2, data);
257 return;
258 }
259 for (unsigned plane = 0; plane < 4; ++plane)
260 if (vga.sequencer.map_mask & (1 << plane))
261 vga.memory[(address & 0xffff) + plane * 0x10000] =
262 vga_latch_write(plane, data);
263}

279CRV200::Frame CRadeon7500::render_frame()
280{
281 std::lock_guard<std::recursive_mutex> guard(
282 cSystem->get_device_bus_mutex());
283 if (m_pause.load() || !m_rv200.display_output_enabled())
284 return {};
285 // The primary DAC may select CRTC2 even while CRTC1 is in VGA mode.
286 if (m_rv200.native_display())
287 return m_rv200.scanout();
288 if (!m_vga_enable || !vga.crtc.sync_en ||
289 (vga.sequencer.data[0] & 3) != 3 || (vga.sequencer.data[1] & 0x20) ||
290 !(vga.attribute.index & 0x20) || !vga.crtc.maximum_scan_line ||
291 (m_rv200.peek(CRV200::CRTC_EXT_CNTL) & CRV200::DISPLAY_DIS))
292 return {};
293 CRV200::Frame frame;
294 frame.width = (vga.crtc.horz_disp_end + 1) *
295 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
296 frame.height = vga.crtc.vert_disp_end + 1;
297 if (!frame.width || frame.width > 4096 || frame.height > 1024)
298 return {};
299 // vga_device can write repeated scanlines past the visible-area bottom. Cover CGA
300 // double-scan plus one character row; only the visible rectangle is used.
301 m_render_bitmap.resize(frame.width, frame.height * 2 + 64);
302 screen().set_visible_area(frame.width, frame.height);
303 screen().tick_frame();
304 vga.crtc.start_addr = latch_start_addr();
305 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
306 palette_update();
307 svga_device::screen_update(
308 screen(), m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
309 frame.argb.assign(
310 m_render_bitmap.raw(),
311 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
312 return frame;
313}

315void CRadeon7500::update()
316{
317 const CRV200::Frame frame = render_frame();
318 // Never hold the device-bus mutex while acquiring a GUI mutex.
319 CRadeonDisplayLock guard(m_output.display());
320 if (!frame.argb.empty())
321 {
322 m_output.present_argb32(frame.argb.data(), frame.width, frame.height);

324 else
325 {
326 m_output.clear_screen();
327 }
328 m_output.display().flush();

332{
333 try

335 m_output.initialize(X_TILESIZE, Y_TILESIZE);

371std::array<u32, 64> CRadeon7500::PCIConfig::config_data() const
372{
373 std::array<u32, 64> data{};
374 data[0] = u32(vendor) | (u32(device) << 16);
375 data[2] = (class_code << 8) | revision;
376 data[4] = 0x00000008; // prefetchable framebuffer
377 data[5] = 0x00000001; // I/O register window
378 data[11] = u32(subsystem_vendor) | (u32(subsystem_device) << 16);
379 data[15] = 0x000001ff; // INTA#, interrupt line unassigned
380 // No AGP or power-management capability chain is modeled.
381 return data;
382}

384std::array<u32, 64> CRadeon7500::PCIConfig::config_mask(u32 rom_size)
385{
386 std::array<u32, 64> mask{};
387 mask[1] = 0x00000147; // I/O, memory, bus master, parity response, SERR
388 mask[3] = 0x0000ffff; // cache-line size and latency timer
389 mask[4] = ~(FramebufferBARSize - 1);
390 mask[5] = ~(IOBARSize - 1);
391 mask[6] = ~(RegisterBARSize - 1);
392 if (rom_size)
393 mask[12] = (~(rom_size - 1) & 0xfffff800u) | 1; // ROM decode and enable
394 mask[15] = 0xff; // interrupt line only; pin fixed
395 return mask;
396}

398CRadeon7500::OptionROM CRadeon7500::OptionROM::parse(
399 const std::vector<uint8_t>& raw, const std::string& layout)
400{
401 auto fail = [](const char* message) {
402 throw std::invalid_argument(message);
403 };
404 const size_t n = raw.size();
405 if (n < 2048 || n > 1024 * 1024)
406 fail("ROM must be 2 KiB to 1 MiB");
407 if (layout != "exact" && layout != "pad_ff")
408 fail("rom_layout must be exact or pad_ff");
409 if (layout == "exact" && (n & (n - 1)))
410 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
411 auto word = [&](size_t p) -> uint16_t {
412 if (p > n - 2)
413 fail("ROM field out of bounds");
414 return uint16_t(uint16_t(raw[p]) | (uint16_t(raw[p + 1]) << 8));
415 };
416 size_t at = 0;
417 bool last = false;
418 for (unsigned images = 0; images < 32 && !last; ++images)
419 {
420 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
421 fail("ROM missing 55 AA signature");
422 const size_t p = at + word(at + 0x18);
423 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
424 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
425 fail("ROM missing/bounded PCIR structure");
426 if (word(p + 4) != 0x1002 || word(p + 6) != 0x5157)
427 fail("ROM PCIR identity is not ATI 1002:5157 (RV200)");
428 const size_t length = size_t(word(p + 0x10)) * 512;
429 if (!length || length > n - at || p + 0x18 > at + length)
430 fail("Invalid option-ROM image length");
431 const size_t pcir_length = word(p + 0x0a);
432 if (pcir_length < 0x18 || pcir_length > at + length - p)
433 fail("Invalid PCIR structure length");
434 if (raw[p + 0x14] == 0 &&
435 (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length))
436 fail("Invalid x86 initialization image length");
437 unsigned sum = 0;
438 for (size_t i = at; i < at + length; ++i)
439 sum += raw[i];
440 if (sum & 255)
441 fail("Option-ROM checksum failed");
442 if (raw[p + 0x14] != 0 && raw[p + 0x14] != 3)
443 fail("Only x86 and EFI option-ROM images are supported");
444 last = (raw[p + 0x15] & 0x80) != 0;
445 at += length;
446 }
447 if (!last)
448 fail("Unterminated option-ROM image chain");
449 OptionROM rom;
450 rom.file_size = static_cast<uint32_t>(n);
451 const std::string agp = "V200AGP";
452 if (std::search(raw.begin(), raw.end(), agp.begin(), agp.end()) !=
453 raw.end())
454 fail("AGP ROM rejected for PCI 3X-PBXGG-AA target; choose a PCI image");
455 rom.bytes = raw;
456 if (layout == "pad_ff")
457 {
458 // Erased padding to the next power of two; the upper decode is unmeasured.
459 size_t decoded = 2048;
460 while (decoded < n)
461 decoded *= 2;
462 rom.bytes.resize(decoded, 0xff);
463 }
464 return rom;
465}

495void CRadeon7500::init()
496try
497{
498 if (m_initialized)
499 return;
500 CConfigurator* cfg = myCfg;
501 init_maps();
502 add_legacy_io(LegacyIO, 0x3b0, 0x30);
503 add_legacy_mem(LegacyMemory, 0xa0000, 0x20000);
504 m_profile.revision =
505 static_cast<uint8_t>(radeon_bounded_setting(cfg, "revision", 0, 255));
506 m_profile.subsystem_vendor = static_cast<uint16_t>(
507 radeon_bounded_setting(cfg, "subsystem_vendor", 0, 65535));
508 m_profile.subsystem_device = static_cast<uint16_t>(
509 radeon_bounded_setting(cfg, "subsystem_device", 0, 65535));
510 m_scanline_access_divisor =
511 radeon_setting(cfg, "scanline_access_divisor", 1024);
512 if (m_scanline_access_divisor == 0)
513 throw std::runtime_error("Invalid radeon scanline_access_divisor");
514 m_trace_apertures = cfg->get_bool_value("trace_apertures", false);
515 m_frame_file = cfg->get_text_value("frame_file", "");
516 const std::string romname = cfg->get_text_value("rom", "");
517 m_rom_layout = cfg->get_text_value("rom_layout", "exact");
518 if (!romname.empty())
519 {
520 std::ifstream f(romname, std::ios::binary | std::ios::ate);
521 if (!f)
522 throw std::runtime_error("Cannot open radeon ROM: " + romname);
523 const auto size = f.tellg();
524 if (size < 2048 || size > 1024 * 1024)
525 throw std::runtime_error("radeon ROM must be 2 KiB to 1 MiB");
526 f.seekg(0);
527 std::vector<uint8_t> raw(static_cast<size_t>(size));
528 f.read(
529 reinterpret_cast<char*>(raw.data()),
530 static_cast<std::streamsize>(raw.size()));
531 if (!f)
532 throw std::runtime_error("Cannot read full radeon ROM");
533 const auto parsed = OptionROM::parse(raw, m_rom_layout);
534 m_rom = parsed.bytes;
535#ifdef DEBUG_VGA
536 printf(
537 "%s: ROM %u bytes, layout=%s, decoded %u bytes.\n",
538 devid_string,
539 parsed.file_size,
540 m_rom_layout.c_str(),
541 unsigned(m_rom.size()));
542#endif
543 }
544 const std::string tracename = cfg->get_text_value("trace", "");
545 if (!tracename.empty())
546 {
547 if (std::filesystem::exists(tracename))
548 throw std::runtime_error(
549 "Refusing to overwrite existing radeon trace: " + tracename);
550 m_trace.open(tracename);
551 if (!m_trace)
552 throw std::runtime_error("Cannot create radeon trace");
553 m_trace << "# RV200 ES40 PCI/BAR traffic; values hex, sizes bits\n";
554 }
555 const auto profile_data = m_profile.config_data(),
556 profile_mask = PCIConfig::config_mask(u32(m_rom.size()));
557 u32 data[64]{}, mask[64]{};
558 for (unsigned i = 0; i < 64; ++i)
559 {
560 data[i] = profile_data[i];
561 mask[i] = profile_mask[i];
562 }
563 add_function(0, data, mask);
564 m_rv200.set_vga_callbacks(
565 [this](uint32_t port) {
566 return io_read_b(port);
567 },
568 [this](uint32_t port, uint8_t value) {
569 io_write_b(port, value);
570 });
571 m_rv200.set_palette_index_callback([this]() {
572 vga.dac.state = 0;
573 });
574 m_rv200.set_config_reader([this](uint32_t address) {
575 return endian_32(pci_state.config_data[0][(address & 0xfc) / 4]);
576 });
577 m_rv200.set_dma_reader(
578 [this](uint32_t address, uint8_t* destination, size_t count) {
579 if (!(config_read(0, 4, 16) & 4))
580 return false;
581 // Use the shared PCI/IOMMU path: OpenVMS allocates its ring and
582 // indirect buffers in scatter-gather mapped system memory.
583 do_pci_read(address, destination, 1, count);
584 if (m_trace && count == 4)
585 {
586 const u32 word = u32(destination[0]) | (u32(destination[1]) << 8) |
587 (u32(destination[2]) << 16) | (u32(destination[3]) << 24);
588 trace("DR", 0, address, 32, word);
589 }
590 return true;
591 });
592 m_rv200.set_dma_writer(
593 [this](uint32_t address, const uint8_t* source, size_t count) {
594 if (!(config_read(0, 4, 16) & 4))
595 return false;
596 do_pci_write(address, const_cast<uint8_t*>(source), 1, count);
597 if (m_trace && count == 4)
598 {
599 const u32 word = u32(source[0]) | (u32(source[1]) << 8) |
600 (u32(source[2]) << 16) | (u32(source[3]) << 24);
601 trace("DW", 0, address, 32, word);
602 }
603 return true;
604 });
605 m_rv200.set_irq_callback([this](bool level) {
606 if (!m_replaying_pci)
607 do_pci_interrupt(0, level);
608 });

760bool CRadeon7500::decodes_memory_access(
761 int index, u64 address, int dsize, bool write) const noexcept
762{
763 if (index >= PCI_RANGE_BASE)
764 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
765 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
766 return false;
767 const unsigned bytes = unsigned(dsize) / 8;
768 if (index == LegacyIO)
769 {
770 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
771 return false;
772 // Claim an access only when every byte belongs to this device; reserved
773 // port bytes must not make us a competing positive decoder.
774 for (unsigned i = 0; i < bytes; ++i)
775 if (!vga_port_enabled(0x3b0 + u32(address) + i))
776 return false;
777 return true;
778 }
779 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
780 bytes > 0x20000 - address)
781 return false;
782 switch (vga.gc.memory_map_sel & 3)
783 {
784 case 1:
785 return address + bytes <= 0x10000;
786 case 2:
787 return address >= 0x10000 && address + bytes <= 0x18000;
788 case 3:
789 return address >= 0x18000;
790 default:
791 return true;
792 }
793}

839 cSystem->get_device_bus_mutex());

1029std::vector<u8> CRadeon7500::save_vga_state() const
1030{
1031 std::vector<u8> bytes{1};
1032 for (unsigned i = 0; i < 25; ++i)
1033 bytes.push_back(m_crtc_map.read_byte(i));
1034 for (unsigned i = 0; i < 5; ++i)
1035 bytes.push_back(m_seq_map.read_byte(i));
1036 for (unsigned i = 0; i < 9; ++i)
1037 bytes.push_back(m_gc_map.read_byte(i));
1038 for (unsigned i = 0; i < 21; ++i)
1039 bytes.push_back(m_atc_map.read_byte(i));
1040 bytes.insert(
1041 bytes.end(),
1042 {vga.crtc.index,
1043 vga.sequencer.index,
1044 vga.gc.index,
1045 vga.attribute.index,
1046 vga.miscellaneous_output,
1047 vga.feature_control,
1048 m_vga_enable,
1049 vga.dac.mask,
1050 u8(vga.attribute.state),
1051 u8(vga.dac.read),
1052 u8(vga.dac.state)});
1053 for (const u8 value : vga.dac.loading)
1054 bytes.push_back(value);
1055 for (const u8 value : vga.gc.latch)
1056 bytes.push_back(value);
1057 for (unsigned i = 0; i < 4; ++i)
1058 bytes.push_back(u8(vga.crtc.start_addr >> (8 * i)));
1059 bytes.push_back(vga.attribute.pel_shift);
1060 bytes.push_back(vga.cursor.visible);
1061 bytes.push_back(vga.crtc.irq_latch);
1062 for (unsigned i = 0; i < 4; ++i)
1063 bytes.push_back(u8(m_vga_scanline >> (8 * i)));
1064 for (unsigned i = 0; i < 4; ++i)
1065 bytes.push_back(u8(m_vga_status_phase >> (8 * i)));
1066 return bytes;
1067}

1069void CRadeon7500::restore_vga_state(const std::vector<u8>& bytes)
1070{
1071 auto dword = [&](size_t offset) {
1072 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1073 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1074 };
1075 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1076 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1077 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1078 dword(90) > 31)
1079 throw std::runtime_error("Invalid VGA snapshot fields");
1080 reset_vga();
1081 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1082 for (unsigned i = 0; i < 25; ++i)
1083 if (i != 0x11)
1084 m_crtc_map.write_byte(i, bytes[1 + i]);
1085 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1086 for (unsigned i = 0; i < 5; ++i)
1087 m_seq_map.write_byte(i, bytes[26 + i]);
1088 for (unsigned i = 0; i < 9; ++i)
1089 m_gc_map.write_byte(i, bytes[31 + i]);
1090 for (unsigned i = 0; i < 21; ++i)
1091 m_atc_map.write_byte(i, bytes[40 + i]);
1092 // Every register must read back exactly as saved.
1093 for (unsigned i = 0; i < 25; ++i)
1094 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1095 throw std::runtime_error("Invalid CRTC snapshot encoding");
1096 for (unsigned i = 0; i < 5; ++i)
1097 if (m_seq_map.read_byte(i) != bytes[26 + i])
1098 throw std::runtime_error("Invalid sequencer snapshot encoding");
1099 for (unsigned i = 0; i < 9; ++i)
1100 if (m_gc_map.read_byte(i) != bytes[31 + i])
1101 throw std::runtime_error(
1102 "Invalid graphics-controller snapshot encoding");
1103 for (unsigned i = 0; i < 21; ++i)
1104 if (m_atc_map.read_byte(i) != bytes[40 + i])
1105 throw std::runtime_error("Invalid attribute snapshot encoding");
1106 vga.crtc.index = bytes[61];
1107 vga.sequencer.index = bytes[62];
1108 vga.gc.index = bytes[63];
1109 vga.attribute.index = bytes[64];
1110 vga.miscellaneous_output = bytes[65];
1111 m_ioas = (bytes[65] & 1) != 0;
1112 vga.feature_control = bytes[66];
1113 m_vga_enable = bytes[67];
1114 vga.dac.mask = bytes[68];
1115 vga.attribute.state = bytes[69];
1116 vga.dac.read = bytes[70];
1117 vga.dac.state = bytes[71];
1118 for (unsigned i = 0; i < 3; ++i)
1119 vga.dac.loading[i] = bytes[72 + i];
1120 for (unsigned i = 0; i < 4; ++i)
1121 vga.gc.latch[i] = bytes[75 + i];
1122 vga.crtc.start_addr = dword(79);
1123 vga.attribute.pel_shift = bytes[83];
1124 vga.cursor.visible = bytes[84];
1125 vga.crtc.irq_latch = bytes[85];
1126 m_vga_scanline = dword(86);
1127 m_vga_status_phase = dword(90);
1128 vga.dac.dirty = 1;
1129}

1165
1166int CRadeon7500::RestoreState(FILE* f)
1167{
1168 if (!f)
1169 return -1;
1170 std::lock_guard<std::recursive_mutex> guard(
1171 cSystem->get_device_bus_mutex());
1172 try
1173 {
1174 // PCI state is restored first, following the other ES40 devices.
1175 // A failure aborts the whole system restore; it must not resume the VM.
1176 struct ReplayGuard
1177 {
1178 bool& flag;
1179
1180 ReplayGuard(bool& value) : flag(value) { flag = true; }
1181
1182 ~ReplayGuard() { flag = false; }
1183 } replay(m_replaying_pci);
1184
1185 if (CPCIDevice::RestoreState(f))
1186 return -1;
1187 if (radeon_read_u32(f) != 0x31563752 ||
1188 radeon_read_u32(f) != VGAStateSize)
1189 return -1;
1190 std::vector<u8> vga_bytes(VGAStateSize);
1191 if (fread(vga_bytes.data(), 1, vga_bytes.size(), f) != vga_bytes.size())
1192 return -1;
1193 const u32 lo = radeon_read_u32(f), hi = radeon_read_u32(f),
1194 n = radeon_read_u32(f);
1195 const u32 vram = m_rv200.options().vram_bytes;
1196 if (n < vram || n > vram + 65536 + 16)
1197 return -1;
1198 std::string bytes(n, '\0');
1199 if (fread(bytes.data(), 1, n, f) != n)
1200 return -1;
1201 std::istringstream stream(bytes, std::ios::binary);
1202 m_rv200.RestoreState(stream);
1203 // The core's transactional loader owns its VRAM; refresh our borrowed view.
1204 vga.memory = m_rv200.vram_data();
1205 if (stream.peek() != std::char_traits<char>::eof())
1206 return -1;
1207 restore_vga_state(vga_bytes);
1208 m_access_count = u64(lo) | (u64(hi) << 32);
1209 sync_apertures();
1210 // IRQ output is committed by finalize_restore after all devices restore.
1211 return 0;
1212 }
1213 catch (const std::exception& e)
1214 {
1215 vga.memory = m_rv200.vram_data();
1216 printf(
1217 "%s: Could not restore graphics state: %s.\n",
1218 devid_string,
1219 e.what());
1220 return -1;
1221 }

1232u8 CRadeon7500::input_status()
1233{
1234 vga.attribute.state = 0;
1235 // Deterministic compatibility timing: 32 status reads per synthetic
1236 // scanline. Decode probes and host frame delivery do not clock it.
1237 const bool native = m_rv200.native_display();
