103bool CSCSIBus::arbitrate(int initiator)
104{
105 if (state.phase != SCSI_PHASE_FREE && state.initiator != initiator)
106 {
107 printf("Could not arbitrate for the SCSI bus.\n");
108 return false;
109 }
110
111 state.initiator = initiator;
112 state.phase = SCSI_PHASE_ARBITRATION;
113 return true;
114}

124bool CSCSIBus::select(int initiator, int target)
125{
126 if (state.phase != SCSI_PHASE_ARBITRATION || state.initiator != initiator)
127 FAILURE(IllegalState,
128 "Attempt to select while the device has not won SCSI arbitration");
129
130 if (!targets[target])
131 return false;
132
133 state.target = target;
134 targets[target]->scsi_select_me(target_bus_no[target]);
135 return(state.phase >= 0);
136}

158void CSCSIBus::free_bus(int initiator)
159{
160
161 // nothing to do
162 if (state.phase == SCSI_PHASE_FREE)
163 return;
164
165 if (state.phase == SCSI_PHASE_ARBITRATION)
166 {
167 if (initiator != state.initiator)
168 FAILURE(IllegalState, "Attempt to free the scsi bus");
169 }
170 else
171 {
172 if (targets[initiator] != targets[state.target])
173 FAILURE(IllegalState, "Attempt to free the scsi bus");
174 }
175
176 state.phase = SCSI_PHASE_FREE;
177}

200int CSCSIBus::RestoreState(FILE* f)
201{
202 long ss;
203 u32 m1;
204 u32 m2;
205 size_t r;
206
207 r = fread(&m1, sizeof(u32), 1, f);
208 if (r != 1)
209 {
210 printf("%s: unexpected end of file!\n", devid_string);
211 return -1;
212 }
213
214 if (m1 != scsi_magic1)
215 {
216 printf("%s: MAGIC 1 does not match!\n", devid_string);
217 return -1;
218 }
219
220 r = fread(&ss, sizeof(long), 1, f);
221 if (r != 1)
222 {
223 printf("%s: unexpected end of file!\n", devid_string);
224 return -1;
225 }
226
227 if (ss != sizeof(state))
228 {
229 printf("%s: STRUCT SIZE does not match!\n", devid_string);
230 return -1;
231 }
232
233 r = fread(&state, sizeof(state), 1, f);
234 if (r != 1)
235 {
236 printf("%s: unexpected end of file!\n", devid_string);
237 return -1;
238 }
239
240 r = fread(&m2, sizeof(u32), 1, f);
241 if (r != 1)
242 {
243 printf("%s: unexpected end of file!\n", devid_string);
244 return -1;
245 }
246
247 if (m2 != scsi_magic2)
248 {
249 printf("%s: MAGIC 1 does not match!\n", devid_string);
250 return -1;
251 }
252
253 printf("%s: %d bytes restored.\n", devid_string, (int)ss);
254 return 0;
255}
