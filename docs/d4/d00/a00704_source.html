131void CDPR::init()
132{
133 int i;
134
135 memset(&state, 0, sizeof(state));
136 RestoreStateF();
137
138 for (i = 0; i < cSystem->get_cpu_num(); i++)
139 {
140 state.ram[i * 0x20 + 0x00] = 1; // EV6 BIST
141 state.ram[i * 0x20 + 0x01] = (i == 0) ? 0x80 : i; // SROM status
142 state.ram[i * 0x20 + 0x02] = 1; // STR status
143 state.ram[i * 0x20 + 0x03] = 1; // CSC status
144 state.ram[i * 0x20 + 0x04] = 1; // Pchip0 status
145 state.ram[i * 0x20 + 0x05] = 1; // Pchip1 status
146 state.ram[i * 0x20 + 0x06] = 1; // DIMx status
147 state.ram[i * 0x20 + 0x07] = 1; // TIG bus status
148 state.ram[i * 0x20 + 0x08] = 0xdd; // DPR test started
149 state.ram[i * 0x20 + 0x09] = 1; // DPR status
150 state.ram[i * 0x20 + 0x0a] = 0xff; // CPU speed status
151 state.ram[i * 0x20 + 0x0b] = (cSystem->get_cpu(i)->get_speed() / 1000000) % 256; //speed
152 state.ram[i * 0x20 + 0x0c] = (cSystem->get_cpu(i)->get_speed() / 1000000) / 256; //speed
153
154 // powerup time BCD:
155 time_t now = time(NULL);
156 bool faked = false;
157
158 // Optional absolute time override (sys0 "time" config; same UTC parser
159 // as AliM1543C). DPR::init runs before AliM1543C::init, so we stay
160 // silent on parse failure — bad values trip AliM1543C's FAILURE_1
161 // before the guest starts.
162 char* faketime = myCfg->get_text_value("time");
163 if (faketime)
164 {
165 struct tm ft = {};
166 if (sscanf(faketime, "%d-%d-%d %d:%d:%d",
167 &ft.tm_year, &ft.tm_mon, &ft.tm_mday,
168 &ft.tm_hour, &ft.tm_min, &ft.tm_sec) >= 3)
169 {
170 ft.tm_year -= 1900;
171 ft.tm_mon -= 1;
172#ifdef _WIN32
173 time_t set_time = _mkgmtime(&ft);
174#else
175 time_t set_time = timegm(&ft);
176#endif
177 if (set_time != (time_t)-1)
178 {
179 now = set_time;
180 faked = true;
181 }
182 }
183 }
184
185 // Display in UTC when faketime is in effect (matches the TOY clock);
186 // host local otherwise (preserves existing behavior).
187 struct tm* t = faked ? gmtime(&now) : localtime(&now);
188 state.ram[i * 0x20 + 0x10] = ToBCD(t->tm_hour);
189 state.ram[i * 0x20 + 0x11] = ToBCD(t->tm_min);
190 state.ram[i * 0x20 + 0x12] = ToBCD(t->tm_sec);
191 state.ram[i * 0x20 + 0x13] = ToBCD(t->tm_mday);
192 state.ram[i * 0x20 + 0x14] = ToBCD(t->tm_mon + 1);
193 state.ram[i * 0x20 + 0x15] = ToBCD(t->tm_year - 100); // tm_year is based on 1900
194#if defined(DEBUG_DPR)
195 printf("%%DPR-I-BOOTDATE: %02x-%02x-%02x, %02x:%02x:%02x\n",
196 state.ram[i * 0x20 + 21], state.ram[i * 0x20 + 20],
197 state.ram[i * 0x20 + 19], state.ram[i * 0x20 + 16],
198 state.ram[i * 0x20 + 17], state.ram[i * 0x20 + 18]);
199#endif
200 state.ram[i * 0x20 + 0x16] = 0; // no error
201 state.ram[i * 0x20 + 0x1e] = 0x80; // CPU SROM sync moet 0x80 zijn; anders --> cpu0 startup failure
202 state.ram[i * 0x20 + 0x1f] = 8; // cach size in MB
203 }
204
205 state.ram[0xda] = 0xaa; // TIG load
206
207 // DIMM config: mirror the modeled topology (CSystem::init_spd_from_config_mb):
208 // array a = MMB a, DIMMs J1..J4 (and J5..J8 when twice-split).
209 const CSystem::SDimmLayout& lay = cSystem->get_dimm_layout();
210 const std::vector<uint8_t>& spd = cSystem->get_dimm_spd();
211 for (int a = 0; a < lay.n_arrays; a++)
212 {
213 // Service guide Table C-1, byte 0x80: <7:4> F = twice-split 8 DIMMs,
214 // 4 = non-split lower set only; <3:0> = configured array position.
215 state.ram[0x80 + 2 * a] = (u8)(((lay.dimms_per_array == 8) ? 0xf0 : 0x40) | a);
216 u8 sz = (u8)(lay.dimm_mb / 64); // DIMM size in 64MB units
217 state.ram[0x81 + 2 * a] = sz ? sz : 1;
218 }
219
220 // RMC-cached SPD reads, one 256-byte region per installed DIMM.
221 // Region for MMB m DIMM Jd is at 0x100 * (m*8 + d), d = 1..8
222 // (service guide Table C-1: 100 = MMB0 J1 ... 2000 = MMB3 J8).
223 for (int a = 0; a < lay.n_arrays; a++)
224 for (int d = 1; d <= lay.dimms_per_array; d++)
225 memcpy(&state.ram[0x100 * (a * 8 + d)], spd.data(), spd.size());
226
227 // powerup failure bits
228 state.ram[0x88] = 0; // each bit is one DIMM on MMB0
229 state.ram[0x89] = 0x00; // MMB1
230 state.ram[0x8a] = 0x00; // MMB2
231 state.ram[0x8b] = 0x00; // MMB3
232
233 // misconfigured DIMM bits
234 state.ram[0x8c] = 0; // each bit is one DIMM on MMB0
235 state.ram[0x8d] = 0; // MMB1
236 state.ram[0x8e] = 0; // MMB2
237 state.ram[0x8f] = 0; // MMB3
238 state.ram[0x90] = 0xff; // psu / vterm present
239 state.ram[0x91] = 0x00; // psu ok bits
240 state.ram[0x92] = 0x07; // ac inputs valid
241 state.ram[0x93] = 0x25; // cpu 0 temp in C
242 state.ram[0x94] = 0x25; // cpu 1 temp in C
243 state.ram[0x95] = 0x25; // cpu 2 temp in C
244 state.ram[0x96] = 0x25; // cpu 3 temp in C
245 state.ram[0x97] = 0x25; // pci 0 temp in C
246 state.ram[0x98] = 0x25; // pci 1 temp in C
247 state.ram[0x99] = 0x25; // pci 2 temp in C
248 state.ram[0x9a] = 0x8b; // fan 0 speed
249 state.ram[0x9b] = 0x8b; // fan 1 speed
250 state.ram[0x9c] = 0x8b; // fan 2 speed
251 state.ram[0x9d] = 0x8b; // fan 3 speed
252 state.ram[0x9e] = 0x8b; // fan 4 speed
253 state.ram[0x9f] = 0x8b; // fan 5 speed
254
255 // vector 680 info (various faults)
256 for (i = 0xa0; i < 0xaa; i++)
257 state.ram[i] = 0;
258
259 state.ram[0xaa] = 0x00; // fans good
260
261 // RMC read failure DIMM bits (bit d-1 = Jd, set = no SPD to read):
262 // MMB m carries array m's DIMMs J1..J4 (J1..J8 when twice-split).
263 {
264 u8 in_set = (lay.dimms_per_array == 8) ? 0xff : 0x0f;
265 for (int m = 0; m < 4; m++)
266 state.ram[0xab + m] = (u8)((m < lay.n_arrays) ? ~in_set : 0xff);
267 }
268 switch (cSystem->get_cpu_num())
269 {
270 case 1: state.ram[0xaf] = 0x0e; // all MMB I2C's read + CPU 0
271 break;
272 case 2: state.ram[0xaf] = 0x0c; // all MMB I2C's read + CPU 0
273 break;
274 case 3: state.ram[0xaf] = 0x08; // all MMB I2C's read + CPU 0
275 break;
276 case 4: state.ram[0xaf] = 0x00; // all MMB I2C's read + CPU 0
277 break;
278 }
279
280 state.ram[0xb0] = 0x00; // PCI i2c read
281 state.ram[0xb1] = 0x00; // mainboard i2c read
282 state.ram[0xb2] = 0x00; // psu's and scsi backplanes i2c read
283 state.ram[0xba] = 0xba; // i2c finished
284 state.ram[0xbb] = 0x00; // rmc error
285 state.ram[0xbc] = 0x00; //rmc flash update error status
286
287 // 680 fatal registers
288 state.ram[0xbd] = 0x07; // ac inputs valid
289 state.ram[0xbe] = 0; // faults
290 state.ram[0xbf] = 0; // faults
291 state.ram[0xda] = 0xaa; // tig load success
292
293 // Power-supplies
294 state.ram[0xdb] = 0xf4; // PS0 id
295 state.ram[0xdc] = 0x45; // 3.3v current
296 state.ram[0xdd] = 0x51; // 5.0v current
297 state.ram[0xde] = 0x37; // 12v current
298 state.ram[0xdf] = 0x8b; // fan speed
299 state.ram[0xe0] = 0xd6; // ac voltage (230v)
300 state.ram[0xe1] = 0x49; // internal temp. (56 C)
301 state.ram[0xe2] = 0x4b; // inlet temp. (20 C)
302 state.ram[0xe4] = 0xf5; // PS1 id
303 state.ram[0xe5] = 0x45; // 3.3v current
304 state.ram[0xe6] = 0x51; // 5.0v current
305 state.ram[0xe7] = 0x37; // 12v current
306 state.ram[0xe8] = 0x8b; // fan speed
307 state.ram[0xe9] = 0xd6; // ac voltage (230v)
308 state.ram[0xea] = 0x49; // internal temp. (56 C)
309 state.ram[0xeb] = 0x4b; // inlet temp. (20 C)
310 state.ram[0xed] = 0xf6; // PS2 id
311 state.ram[0xee] = 0x45; // 3.3v current
312 state.ram[0xef] = 0x51; // 5.0v current
313 state.ram[0xf0] = 0x37; // 12v current
314 state.ram[0xf1] = 0x8b; // fan speed
315 state.ram[0xf2] = 0xd6; // ac voltage (230v)
316 state.ram[0xf3] = 0x49; // internal temp. (56 C)
317 state.ram[0xf4] = 0x4b; // inlet temp. (20 C)
318
319 // EEROMs
320
321 /* Service guide (EK-ES240-SV) Table C-1:
322 100: MMB0 J1 DIMM 1
323 200: MMB0 J2 DIMM 2
324 ...
325 800: MMB0 J8 DIMM 8
326 900: MMB1 J1 DIMM 1
327 ...
328 1000: MMB1 J8 DIMM 8
329 1100: MMB2 J1 DIMM 1
330 ...
331 1800: MMB2 J8 DIMM 8
332 1900: MMB3 J1 DIMM 1
333 ...
334 2000: MMB3 J8 DIMM 8
335 2100: CPU0
336 2200: CPU1
337 2300: CPU2
338 2400: CPU3
339 2500: MMB0
340 2600: MMB1
341 2700: MMB2
342 2800: MMB3
343 2900: CPB (PCI backplane)
344 2a00: CSB (motherboard)
345 3100: PSU0 cont @ 3d00
346 3200: PSU1 cont @ 3e00
347 3300: PSU2 cont @ 3f00
348 3b00: SCSI0 (backplane)
349 3c00: SCSI1
350
351 2B00:2BFF RMC Last EV6 Correctable Error
352 ASCII character string that indicates correctable error occurred, type, FRU, and so on.
353 2C00:2CFF RMC Last Redundant Failure
354 ASCII character string that indicates redundant failure occurred, type, FRU, and so on.
355 2D00:2DFF RMC Last System Failure
356 ASCII character string that indicates system failure occurred, type, FRU, and so on.
357 2E00:2FFF RMC Uncorrectable machine logout frame (512 bytes)
358 */
359
360 // 3000:3008 SROM Version (ASCII string)
361 state.ram[0x3000] = 'V';
362 state.ram[0x3001] = '2';
363 state.ram[0x3002] = '.';
364 state.ram[0x3003] = '2';
365 state.ram[0x3004] = '2';
366 state.ram[0x3005] = 'G';
367 state.ram[0x3006] = 0;
368 state.ram[0x3007] = 0;
369 state.ram[0x3008] = 0;
370
371 // 3009:300B RMC Rev Level of RMC first byte is letter Rev [x/t/v] second 2 bytes are major/minor.
372 // This is the rev level of the RMC on-chip code.
373 state.ram[0x3009] = 'V';
374 state.ram[0x300a] = 0x31;
375 state.ram[0x300b] = 0x30;
376
377 // 300C:300E RMC Rev Level of RMC first byte is letter Rev [x/t/v] second 2 bytes are major/minor.
378 // This is the rev level of the RMC flash code.
379 state.ram[0x300c] = 'V';
380 state.ram[0x300d] = 0x31;
381 state.ram[0x300e] = 0x30;
382
383 // 300F:3010 300F RMC Revision Field of the DPR Structure
384 // 3400 SROM Size of Bcache in MB
385 state.ram[0x3400] = 8;
386
387 //3401 SROM Flash SROM is valid flag; 8 = valid,0 = invalid
388 state.ram[0x3401] = 8;
389
390 //3402 SROM System's errors determined by SROM
391 state.ram[0x3402] = 0;
392
393 for (i = 0; i < cSystem->get_cpu_num(); i++)
394 {
395 state.ram[0x3418 + 0x10 * i] = 0xff;
396 //3410:3417 SROM/SRM Jump to address for CPU0
397 //3418 SROM/SRM Waiting to jump to flag for CPU0
398 //3419 SROM Shadow of value written to EV6 DC_CTL register.
399 //341A:341E SROM Shadow of most recent writes to EV6 CBOX "Write-many" chain.
400 }
401
402 //34A0:34A7 SROM Array 0 to DIMM ID translation
403 // Bits<4:0>
404 // Bits<7:5>
405 // 0 = Exists, No Error Bits <2:0> =
406 // 1 = Expected Missing DIMM + 1 (1-8)
407 // 2 = Error - Missing DIMM(s) Bits <4:3> =
408 // 4 = Error - Illegal MMB (0-3) DIMM(s)
409 // 6 = Error - Incompatible DIMM(s)
410 // 34A8:34AF SROM Repeat for Array 1 of Array 0 34A0:34A7
411 // 34B0:34B7 SROM Repeat for Array 2 of Array 0 34A0:34A7
412 // 34B8:34CF SROM Repeat for Array 3 of Array 0 34A0:34A7
413 // Entry j of array a: DIMM J(j+1) on MMB a.
414 for (int a = 0; a < 4; a++)
415 for (int j = 0; j < 8; j++)
416 {
417 u8 status = (a < lay.n_arrays &&
418 j < lay.dimms_per_array) ? 0 : 1; // 1 = expected missing
419 state.ram[0x34a0 + a * 8 + j] = (u8)((status << 5) | (a << 3) | j);
420 }
421
422 // 34C0:34FF Used as scratch area for SROM
423 // 3500:35FF Used as the dedicated buffer in which SRM writes OCP or FRU EEROM data.
424 // Firmware will write this data, RMC will only read this data.
425 // 3600:36FF 3600 SRM Reserved
426 // 3700:37FF SRM Reserved
427 // 3800:3AFF RMC RMC scratch space
428 printf("%s: $Id$\n",
429 devid_string);
430}

451{

457u64 CDPR::ReadMem(int index, u64 address, int dsize)
458{
459 u64 data = 0;
460 int a = (int)(address >> 6);
461
462 data = state.ram[a];
463
464#if defined(DEBUG_DPR)
465 printf("%%DPR-I-READ: Dual-Port RAM read @ 0x%08x: 0x%02x\n", a,
466 (u32)(data & 0xff));
467#endif
468 return data;
469}

471void CDPR::WriteMem(int index, u64 address, int dsize, u64 data)
472{
473 int i;
474 int a = (int)(address >> 6);
475#if defined(DEBUG_DPR)
476 printf("%%DPR-I-WRITE: Dual-Port RAM write 0x%08x 0x%02x:\n", a,
477 (u32)(data & 0xff));
478#endif
479
480 // FOR COMMANDS:
481 //
482 // 0xf9: buffer size
483 // 0xfb:fa qualifier / address
484 // 0xfc: completion code (0 = ok, 80 = error, 81 = invalid code, 82 = invalid qualifier)
485 // 0xfd: rmc command id for response
486 // 0xfe: command code
487 // 0xff: rmc command id for command
488 // COMMANDS:
489 // 01: update EEPROM
490 // 02: update baud rate
491 // 03: write to OCP
492 // F0: update RMC flash
493 state.ram[a] = (char)data;
494 dirty = true;
495 last_dirty = time(nullptr);
496 switch (a)
497 {
498 case 0xff:
499
500 // command
501 state.ram[0xfd] = state.ram[0xff];
502 switch (state.ram[0xfe])
503 {
504 case 1:
505
506 /* Service guide (EK-ES240-SV) Table C-1:
507 100-800: MMB0 J1 DIMM 1 ... J8 DIMM 8
508 900-1000: MMB1 J1 DIMM 1 ... J8 DIMM 8
509 1100-1800: MMB2 J1 DIMM 1 ... J8 DIMM 8
510 1900-2000: MMB3 J1 DIMM 1 ... J8 DIMM 8
511 2100: CPU0
512 2200: CPU1
513 2300: CPU2
514 2400: CPU3
515 2500: MMB0
516 2600: MMB1
517 2700: MMB2
518 2800: MMB3
519 2900: CPB (PCI backplane)
520 2a00: CSB (motherboard)
521 3100: PSU0 cont @ 3d00
522 3200: PSU1 cont @ 3e00
523 3300: PSU2 cont @ 3f00
524 3b00: SCSI0 (backplane)
525 3c00: SCSI1 */
526
527 // FRU-Write
528 switch (state.ram[0xfb])
529 {
530 case 0x21:
531 case 0x22:
532 case 0x23:
533 case 0x24:
534 if ((state.ram[0xfb] - 0x20) > cSystem->get_cpu_num())
535 {
536 state.ram[0xfc] = 0x80;
537 break;
538 }
539
540 case 1:
541 case 2:
542 case 3:
543 case 4:
544 case 5:
545 case 6:
546 case 7:
547 case 8:
548 case 0x09: // MMB1..MMB3 DIMM regions (see map above)
549 case 0x0a:
550 case 0x0b:
551 case 0x0c:
552 case 0x0d:
553 case 0x0e:
554 case 0x0f:
555 case 0x10:
556 case 0x11:
557 case 0x12:
558 case 0x13:
559 case 0x14:
560 case 0x15:
561 case 0x16:
562 case 0x17:
563 case 0x18:
564 case 0x19:
565 case 0x1a:
566 case 0x1b:
567 case 0x1c:
568 case 0x1d:
569 case 0x1e:
570 case 0x1f:
571 case 0x20:
572 case 0x25:
573 case 0x26:
574 case 0x27:
575 case 0x28:
576 case 0x29:
577 case 0x2a:
578 case 0x31:
579 case 0x32:
580 case 0x33:
581 case 0x3b:
582 case 0x3c:
583 case 0x3d:
584 case 0x3e:
585 case 0x3f:
586 for (i = 0; i < state.ram[0xf9] + 1; i++)
587 {
588 state.ram[state.ram[0xfb] * 0x100 + state.ram[0xfa] + i] = state.ram[0x3500 + state.ram[0xfa] + i];
589#if defined(DEBUG_DPR)
590 printf("%%DPR-I-FRU: FRU data %02x @ FRU %02x set to %02x\n",
591 state.ram[0xfa] + i, state.ram[0xfb],
592 state.ram[0x3500 + state.ram[0xfa] + i]);
593#endif
594 }
595
596 state.ram[0xfc] = 0;
597 break;
598
599 default:
600#if defined(DEBUG_DPR)
601 printf("%%DPR-I-RMC: RMC Command given: %02x\r\n", state.ram[0xfe]);
602 printf("%%DPR-I-RMC: f9:%02x fb-fa:%02x%02x\r\n", state.ram[0xf9],
603 state.ram[0xfb], state.ram[0xfa]);
604#endif
605 state.ram[0xfc] = 0x80;
606 }
607 break;
608
609 case 2:
610 state.ram[0xfc] = 0;
611 break;
612
613 case 3:
614
615 // OCP-Write
616//#if defined(DEBUG_DPR)
617 {
618 char buf[17];
619 memcpy(buf, &(state.ram[0x3500]), 16);
620 buf[16] = 0;
621 fprintf(stderr, "%%%%DPR-I-OCP: OCP message: [%s]\n", buf);
622 }
623//#endif
624 state.ram[0xfc] = 0;
625 break;
626
627 case 0xf0:
628 state.ram[0xfc] = 0;
629
630 default:
631#if defined(DEBUG_DPR)
632 printf("%%DPR-I-RMC: RMC Command given: %02x\r\n", state.ram[0xfe]);
633 printf("%%DPR-I-RMC: f9:%02x fb-fa:%02x%02x\r\n", state.ram[0xf9],
634 state.ram[0xfb], state.ram[0xfa]);
635#endif
636 state.ram[0xfc] = 0x81;
637 }
638 break;
639
640 case 0xfd:
641
642 // end of command
643 state.ram[0xff] = state.ram[0xfd];
644 break;
645
646 case 0x3428:
647 case 0x3438:
648 case 0x3448:
649 {
650 // start cpu 1..3 (waiting-to-jump flag at 0x3418+0x10*n)
651 int n = ((a - 0x3418) >> 4);
652 if (cSystem->get_cpu_num() > n)
653 {
654 CAlphaCPU* c = cSystem->get_cpu(n);
655
656 // Only launch a parked (cold-boot) CPU.
657 // A running CPU is running, a cross-thread set_pc would corrupt it.
658 if (c->get_waiting())
659 {
660 printf("*** DPR *** Starting CPU %d ***\n", n);
661 c->set_pc(0x8001); // should come from dpr...
662 c->stop_waiting();
663 }
664 else
665 printf("*** DPR *** CPU %d is running, not redirecting ***\n", n);
666 }
667 break;
668 }
669 }
670
671 return;
672}

677void CDPR::SaveStateF(char* fn, bool verbose)
678{
679 FILE* ff;
