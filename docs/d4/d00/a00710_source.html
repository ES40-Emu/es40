164bool CFlash::HasBootFirmware() const
165{
166 // A real ES40 flash carries a partitioned layout (TIG/SRM/ABIOS/SROM) with
167 // a CPQ header at the start of the SRM partition. Treat that signature as
168 // the sole marker for "this flash holds bootable firmware".
169 const u8* const p = state.Flash + srm_partition_offset + cpq_sig_offset;
170 return p[0] == 'C' && p[1] == 'P' && p[2] == 'Q' && p[3] == 0;
171}

201u64 CFlash::ReadMem(int index, u64 address, int dsize)
202{
203 u64 data = 0;
204 int a = (int)(address >> 6);
205
206 // TIGbus flash is an 8-bit device on a wider bus. Only the low 32-bit lane is wired.
207 // Ignore reads from the upper lane (addr & 0x4) to match real ES40 behavior and QEMU.
208 if (address & 0x4)
209 return 0;
210
211 // Out-of-range reads behave like open bus / erased flash.
212 if ((unsigned)a >= (unsigned)sizeof(state.Flash))
213 return 0xFF;
214
215 switch (state.mode)
216 {
217 case MODE_AUTOSEL:
218 // AM29F016 autoselect IDs alias within each 64 KiB sector.
219 // ARC probes sector-local offsets, not only absolute 0/1.
220 switch (a & 0xff)
221 {
222 case 0:
223 data = 1; // manufacturer
224 break;
225
226 case 1:
227 data = 0xad; // device
228 break;
229
230 case 2:
231 data = 0; // sector is not protected
232 break;
233
234 default:
235 data = 0;
236 }
237 break;
238
239 case MODE_CONFIRM_0:
240 data = 0x80;
241 state.mode = MODE_READ;
242 break;
243
244 case MODE_CONFIRM_1:
245 data = 0x80;
246 state.mode = MODE_CONFIRM_0;
247 break;
248
249 default:
250 data = state.Flash[a];
251 break;
252 }
253
254 return data;
255}

300void CFlash::WriteMem(int index, u64 address, int dsize, u64 data)
301{
302 // Flash is mapped with 64-byte spacing, so byte index is address >> 6.
303 const int a = (int)(address >> 6);
304 const int ad = a & 0xffff;
305 const u8 byte = (u8)data;
306
307 // Upper 32-bit TIGbus lane is not connected to the flash.
308 if (address & 0x4)
309 return;
310
311 // sanity check... are we supposed to be here?
312 if ((unsigned)a >= (unsigned)sizeof(state.Flash))
313 {
314 state.mode = MODE_READ;
315 return;
316 }
317
318
319 switch (state.mode)
320 {
321 case MODE_PROGRAM:
322 {
323 // More realistic flash behavior: programming can only change 1 -> 0.
324 // Any attempt to set bits back to 1 requires an erase.
325 const u8 oldv = state.Flash[a];
326 const u8 newv = (u8)(oldv & byte);
327
328 if (newv != oldv)
329 {
330 state.Flash[a] = newv;
331 dirty = true;
332 last_dirty = time(nullptr);
333 printf("%%SRM-I-FLASH: Wrote data: 0x%02X to sector address: 0x%04X\n", byte, a);
334 }
335
336 state.mode = MODE_READ;
337 return;
338 }
339
340 case MODE_READ:
341 case MODE_AUTOSEL:
342 // formerly we bailed as if confirm_0 or confirm_1 here, now
343 // explicitly handle those modes for future-proofing.
344 case MODE_CONFIRM_0:
345 case MODE_CONFIRM_1:
346 // AMD-style flashes support "reset/read-array" with 0xF0 or 0xFF.
347 // For firmware/software to abort a command sequence.
348 if (byte == 0xF0 || byte == 0xFF)
349 {
350 state.mode = MODE_READ;
351 return;
352 }
353
354 if ((ad == 0x5555) && (byte == 0xaa))
355 {
356 state.mode = MODE_STEP1;
357 return;
358 }
359
360 state.mode = MODE_READ;
361 return;
362
363 case MODE_STEP1:
364 if (byte == 0xF0 || byte == 0xFF)
365 {
366 state.mode = MODE_READ;
367 return;
368 }
369
370 if ((ad == 0x2aaa) && (byte == 0x55))
371 {
372 state.mode = MODE_STEP2;
373 return;
374 }
375
376 state.mode = MODE_READ;
377 return;
378
379 case MODE_STEP2:
380 if (byte == 0xF0 || byte == 0xFF)
381 {
382 state.mode = MODE_READ;
383 return;
384 }
385
386 if (ad != 0x5555)
387 {
388 state.mode = MODE_READ;
389 return;
390 }
391
392 switch (byte)
393 {
394 case 0x90:
395 state.mode = MODE_AUTOSEL;
396 return;
397
398 case 0xa0:
399 state.mode = MODE_PROGRAM;
400 return;
401
402 case 0x80:
403 state.mode = MODE_ERASE_STEP3;
404 return;
405
406 default:
407 state.mode = MODE_READ;
408 return;
409 }
410
411 case MODE_ERASE_STEP3:
412 if (byte == 0xF0 || byte == 0xFF)
413 {
414 state.mode = MODE_READ;
415 return;
416 }
417
418 if ((ad == 0x5555) && (byte == 0xaa))
419 {
420 state.mode = MODE_ERASE_STEP4;
421 return;
422 }
423
424 state.mode = MODE_READ;
425 return;
426
427 case MODE_ERASE_STEP4:
428 if (byte == 0xF0 || byte == 0xFF)
429 {
430 state.mode = MODE_READ;
431 return;
432 }
433
434 if ((ad == 0x2aaa) && (byte == 0x55))
435 {
436 state.mode = MODE_ERASE_STEP5;
437 return;
438 }
439
440 state.mode = MODE_READ;
441 return;
442
443 case MODE_ERASE_STEP5:
444 if (byte == 0xF0 || byte == 0xFF)
445 {
446 state.mode = MODE_READ;
447 return;
448 }
449
450 // Chip erase: AA/55/80/AA/55/10 at 5555
451 if ((ad == 0x5555) && (byte == 0x10))
452 {
453 printf("%%SRM-I-FLASH: Erasing flash chip\n");
454 memset(state.Flash, 0xff, sizeof(state.Flash));
455 dirty = true;
456 last_dirty = time(nullptr);
457 state.mode = MODE_CONFIRM_1;
458 return;
459 }
460
461 // Sector erase: AA/55/80/AA/55 then 0x30 written "anywhere" in sector
462 if (byte == 0x30)
463 {
464 printf("%%SRM-I-FLASH: Erasing flash sector\n");
465 const int sector_start = a & ~0xFFFF;
466 memset(&state.Flash[sector_start], 0xFF, 0x10000);
467 dirty = true;
468 last_dirty = time(nullptr);
469 state.mode = MODE_CONFIRM_1;
470 return;
471 }
472
473 state.mode = MODE_READ;
474 return;
475
476 default:
477 // unknown mode, reset to read
478 state.mode = MODE_READ;
479 return;
480 }
481}

486void CFlash::save_raw_image(const char* fn)
487{
488 FILE* ff = fopen(fn, "wb");
489 if (!ff)
490 FAILURE_1(Runtime, "Flash could not be saved to %s", fn);
491
492 const size_t n = fwrite(state.Flash, 1, sizeof(state.Flash), ff);
493 const int close_result = fclose(ff);
494
495 if (n != sizeof(state.Flash))
496 FAILURE_3(Runtime, "Short flash write (%zu of %zu bytes) to %s",
497 n, sizeof(state.Flash), fn);
498 if (close_result != 0)
499 FAILURE_1(Runtime, "Could not finish writing flash image %s", fn);
500 printf("%%FLS-I-SAVEST: Flash saved to %s\n", fn);
501}

522void CFlash::RestoreStateF(char* fn)
523{
524 FILE* ff = fopen(fn, "rb");
525 if (!ff)
526 {
527 // First boot: create a blank 2 MiB flash so firmware writes have
528 // somewhere to land.
529 FILE* wf = fopen(fn, "wb");
530 if (!wf)
531 {
532 printf("%%FLS-F-NOCREATE: Flash could not be created at %s\n", fn);
533 return;
534 }
535 const size_t n = fwrite(state.Flash, 1, sizeof(state.Flash), wf);
536 const int close_result = fclose(wf);
537 if (n != sizeof(state.Flash))
538 printf("%%FLS-F-NOCREATE: Short write (%zu of %zu bytes) creating %s\n",
539 n, sizeof(state.Flash), fn);
540 else if (close_result != 0)
541 printf("%%FLS-F-NOCREATE: Could not finish creating flash image %s\n", fn);
542 else
543 printf("%%FLS-I-CREATE: Blank 2 MiB flash image created at %s\n", fn);
544 return;
545 }
546
547 fseek(ff, 0, SEEK_END);
548 const long file_size = ftell(ff);
549 fseek(ff, 0, SEEK_SET);
550
551 if (file_size == (long)sizeof(state.Flash))
552 {
553 const size_t n = fread(state.Flash, 1, sizeof(state.Flash), ff);
554 fclose(ff);
555 if (n != sizeof(state.Flash))
556 {
557 printf("%%FLS-F-SHORTRD: Short read (%zu of %zu bytes) from %s\n",
558 n, sizeof(state.Flash), fn);
559 FAILURE(Runtime, "Short read loading flash image");
560 }
561 printf("%%FLS-I-RESTST: Flash restored from %s\n", fn);
562 return;
563 }
564
565 u32 header = 0;
566 auto r = fread(&header, sizeof(header), 1, ff);
567 if (r != 1)
568 FAILURE_1(IO, "%s: unexpected end of file!\n", fn);
569 fclose(ff);
570
571 if (header == flash_magic1)
572 {
573 printf("%%FLS-F-LEGACYFMT: %s uses the obsolete wrapped ES40 flash format; "
574 "rename or delete it, then restart.\n", fn);
575 FAILURE(Runtime, "Flash file uses obsolete wrapped format");
576 }
577
578 printf("%%FLS-F-BADSIZE: %s is %ld bytes, expected %zu; "
579 "rename or delete it, then restart.\n",
580 fn, file_size, sizeof(state.Flash));
581 FAILURE(Runtime, "Flash file has invalid size");
582}

592int CFlash::SaveState(FILE* f)
593{
594 if (state.mode < MODE_READ || state.mode > MODE_CONFIRM_1)
595 return -1;
596 const u32 size = (u32)sizeof(state.Flash);
597 const u32 mode = (u32)state.mode;
598 if (fwrite(&flash_snapshot_magic1, sizeof(flash_snapshot_magic1), 1, f) != 1 ||
599 fwrite(&size, sizeof(size), 1, f) != 1 ||
600 fwrite(&mode, sizeof(mode), 1, f) != 1 ||
601 fwrite(state.Flash, 1, sizeof(state.Flash), f) != sizeof(state.Flash) ||
602 fwrite(&flash_snapshot_magic2, sizeof(flash_snapshot_magic2), 1, f) != 1)
603 return -1;
604 return 0;
605}

610int CFlash::RestoreState(FILE* f)
611{
612 u32 magic = 0, size = 0, mode = 0;
613 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != flash_snapshot_magic1 ||
614 fread(&size, sizeof(size), 1, f) != 1 || size != sizeof(state.Flash) ||
615 fread(&mode, sizeof(mode), 1, f) != 1 || mode > MODE_CONFIRM_1)
616 return -1;
617 std::unique_ptr<void, decltype(&free)> restored(malloc(sizeof(state.Flash)), &free);
618 if (!restored || fread(restored.get(), 1, sizeof(state.Flash), f) != sizeof(state.Flash) ||
619 fread(&magic, sizeof(magic), 1, f) != 1 || magic != flash_snapshot_magic2)
620 return -1;
621 memcpy(state.Flash, restored.get(), sizeof(state.Flash));
622 state.mode = (int)mode;
623 // It may still fail a restore at a later component, so let's not write, even if save state was dirty.
624 dirty = false;
625 return 0;
626}
