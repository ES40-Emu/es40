573 // Wall-clock RPCC: state.cc advances by real elapsed time * cpu_hz so it tracks the configured
574 // CPU frequency regardless of how fast/bursty the JIT runs. This is the last sync timestamp;
575 // the delta since it (when enabled) is added to state.cc at each sync point, then it's reset to now.
576 std::chrono::steady_clock::time_point cc_last_sync;
577 u64 cc_wall_remainder = 0; // fractional cycle numerator, / 1e9
578
579 // Advance the wall-clock cc to now. Called at batch boundaries AND on every guest RPCC read.
580 void sync_cc_wallclock()
581 {
582 const auto now = std::chrono::steady_clock::now();
583 if (cc_last_sync > now)
584 cc_last_sync = now;
585 auto cc_delta = now - cc_last_sync;
586 cc_last_sync = now;
587 if (cycle_counter_enabled())
588 {
589 if (cc_delta > std::chrono::seconds(1)) // cap (not drop) odd deltas, as at batch top
590 cc_delta = std::chrono::seconds(1);
591 const u64 ns = (u64)std::chrono::duration_cast<std::chrono::nanoseconds>(cc_delta).count();
592 // Carry sub-cycle precision across frequent syncs.
593 const u64 scaled = ns * cpu_hz + cc_wall_remainder;
594 u64 add = scaled / 1000000000ULL;
595 cc_wall_remainder = scaled % 1000000000ULL;
596 // Repay any rpcc_read floor-borrow by WITHHOLDING wall progress, never by subtracting
597 if (cc_borrow)
598 {
599 const u64 repay = cc_borrow < add ? cc_borrow : add;
600 add -= repay;
601 cc_borrow -= repay;
602 }
603 state.cc += add;
604 }
605 }

607 // RPCC read with a forward-progress guarantee. Real EV6 RPCC advances every cycle
608 static constexpr u64 kRpccCollisionStep = 1;
609 u64 cc_last_read = 0;
610 u64 cc_borrow = 0;
611 u64 rpcc_read()
612 {
613 sync_cc_wallclock();
614 if (cycle_counter_enabled() && state.cc <= cc_last_read)
615 {
616 cc_borrow += cc_last_read + kRpccCollisionStep - state.cc;
617 state.cc = cc_last_read + kRpccCollisionStep;
618 }
619 cc_last_read = state.cc;
620 const auto read = m_profile.cycle_counter.read;
621 return read ? read(*this) : 0;
622 }

641 // Data page translation cache: direct-mapped by virtual page (kDpcEntries slots/dir) so a
642 // multi-page access pattern doesn't thrash a single slot. The inline load checks one slot.
643 // Cached normal accesses use the family-selected mode/ASN; control writes that
644 // affect address rules must invalidate it. Privileged VPTE/ALT accesses bypass it.
645 static constexpr int kDpcBits = 7; // 128 slots per mode bank (8KB pages -> 1 MB)
646 // Page-indexed hint into state.tb[]: FindTBEntry and add_tb probe it before anything else.
647 // Entries are validated on use, so a stale hint is harmless and nothing needs clearing.
648 // Hashed by page number: Tru64 text (0x120000000) and data (0x140000000) pages share their
649 // low page bits, so a low-bits index made each alternation fall through to the linear scan.
650 static constexpr int kTbIdxBits = 10;
651 static inline u32 tb_idx_slot(u64 virt) { return ((u32)(virt >> 13) * 0x9E3779B1u) >> (32 - kTbIdxBits); }

697#if defined(JIT_REGPROF) && defined(_WIN32)
698 m_dpc_flushes++;
699#endif
700 // The generated and helper hit paths compare virt_page before using the slot. An all-ones
701 // tag cannot equal an 8 KB-aligned virtual page, so a bulk fill invalidates both rows
702 // and avoids excessive stores on native-PAL translation flushes.
703 memset(data_page_cache, 0xff, sizeof(data_page_cache));

705 inline void invalidate_data_page_cache(u64 virt, u64 match_mask) {
706 // If all cache index bits participate in the match, every alias shares one slot per bank.
707 if (dpc_index(match_mask) == kDpcMask) {
708 const u64 i = dpc_index(virt);
709 for (int bank = 0; bank < 4; ++bank) data_page_cache[0][(bank << kDpcBits) | i].virt_page = ~U64(0);
710 for (int bank = 0; bank < 4; ++bank) data_page_cache[1][(bank << kDpcBits) | i].virt_page = ~U64(0);
711 return;
712 }

1051inline void CAlphaCPU::irq_h(int number, bool assert, int delay)
1052{
1053 std::unique_lock<std::mutex> irq_lock(m_irq_lock);
1054 const u64 bit = U64(0x1) << number;
1055 bool active = (state.eir & bit) || state.irq_h_timer[number];
1056 if (assert && !active)
1057 {
1058 if (delay)
1059 {
1060 state.irq_h_timer[number] = delay;
1061 state.check_timers = true;
1062 }
1063 else
1064 {
1065 state.eir |= bit;
1066#ifdef JIT_STATS
1067 m_stat_kick[3]++;
1068#endif
1069 state.check_int = true;
1070 }
1071
1072 irq_lock.unlock();
1073 wake_idle();
1074 return;
1075 }
1076
1077 if (assert && (state.eir & bit))
1078 {
1079 // New request on an already-high line - a second device asserting while
1080 // the first is still being serviced.
1081 // The level in eir is already correct, but check_int was consumed when the
1082 // interrupt was dispatched, and after its REI nothing re-polls.
1083 // Re-kick so it is picked up at the next dispatch boundary, matching real hardware.
1084 state.check_int = true;
1085 irq_lock.unlock();
1086 wake_idle();
1087 return;
1088 }
1089
1090 if (!assert && active)
1091 {
1092 state.eir &= ~bit;
1093 state.irq_h_timer[number] = 0;
1094 bool timers_pending = false;
1095 for (int i = 0; i < 6; i++)
1096 {
1097 if (state.irq_h_timer[i])
1098 timers_pending = true;
1099 }
1100 state.check_timers = timers_pending;
1101 }
