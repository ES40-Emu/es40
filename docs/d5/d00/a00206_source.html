44// Family fetch behavior, copied with the immutable CPU profile.
45struct AlphaInstructionFetchPolicy
46{
47 // Fetch one decoded word at the current PC, including cache/cursor handling.
48 // Return zero on success; nonzero means the attempt could not fetch a word.
49 // The family delivers any fetch fault. PC advancement and retirement stay shared.
50 // An absent callback cannot fetch and leaves architectural state unchanged.
51 int (*fetch)(CAlphaCPU& cpu, std::uint32_t* instruction) = nullptr;
52
53 // Optional entry preparation before the JIT probes or dispatches a block.
54 // fetch() owns the equivalent preparation for interpreted execution.
55 void (*prepare_dispatch)(CAlphaCPU& cpu) = nullptr;
56
57 // Observational JIT hooks: no instruction fetch, cache/TB fill, fault or bus read.
58 // Missing hooks cannot resolve a source, override a word, approve a span or chain.
59 // probe consumes the supplied PC without imposing shared alignment rules.
60 // It may update translation lookup hints, but cannot deliver a guest fault.
61 AlphaInstructionMapping (*probe)(CAlphaCPU& cpu, std::uint64_t pc) = nullptr;
62
63 // Peek the current instruction's cached word without consuming its cursor.
64 // False means no cached override: leave the caller's legacy source word intact.
65 bool (*peek_cached)(const CAlphaCPU& cpu, std::uint32_t* instruction) = nullptr;
66
67 // Can the legacy JIT record this interpreted span from its backing source?
68 // Detect cached instructions that differ from the current backing bytes.
69 bool (*matches_interpreted)(const CAlphaCPU& cpu, std::uint64_t start_pc,
70 std::uint32_t count) = nullptr;
71
72 // Reject targets that need dispatcher preparation before entering compiled code.
73 // Called before lookup; a PAL target need not subsequently probe a translation.
74 bool (*can_chain)(const CAlphaCPU& cpu, std::uint64_t target) = nullptr;
75
76 // Bounded backing words for JIT snapshots and revalidation.
77 AlphaInstructionSourceView (*source)(CAlphaCPU& cpu, std::uint64_t pc) = nullptr;
78};
