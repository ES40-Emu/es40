34constexpr AlphaHWLoadDescriptor decode(std::uint32_t instruction)
35{
36 AlphaHWLoadDescriptor load{};
37 if ((instruction >> 26) != 0x1b)
38 return load;
39
40 // EV68 HW_LD TYPE[15:12]: bit 0 selects quadword; 6/7 are unallocated.
41 const unsigned form = (instruction >> 12) & 0xf;
42 load.size_bits = (form & 1) ? 64 : 32;
43 load.displacement = static_cast<std::int32_t>(instruction & 0xfff);
44 if (load.displacement & 0x800)
45 load.displacement -= 0x1000;
46 if (form == 6 || form == 7)
47 return load;
48
49 load.address = form < 4 ? AlphaHWLoadAddress::Physical : AlphaHWLoadAddress::Virtual;
50 load.locked = form == 2 || form == 3;
51 load.vpte = form == 4 || form == 5;
52 load.alternate = form >= 12;
53 load.write_check = form == 10 || form == 11 || form == 14 || form == 15;
54 // Keep physical load-locked forms interpreted; virtual forms retain fault probes.
55 load.compilable = !load.locked;
56 load.discard_safe = load.address == AlphaHWLoadAddress::Physical && !load.locked;
57 return load;
58}
59}
