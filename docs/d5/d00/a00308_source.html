77// Open-addressed key -> int32 index with no per-insert allocation. Links are wiped on every
78// exec and cross-ASN eviction, and std::unordered_set's node mallocs were ~30% of host time
79// in process-creation loads. clear() resets only the used slots, so capacity survives a wipe.
80// No erase: callers mark their own entries dead instead.
81template <class Key>
82class JitFlatIndex
83{
84public:
85 explicit JitFlatIndex(uint32_t capacity = 1024) { rebuild(capacity); }
86
87 // The value for key; a new key gets -1 and sets inserted.
88 int32_t& get(const Key& key, bool& inserted)
89 {
90 if ((m_used.size() + 1) * 2 > m_slots.size()) rebuild((uint32_t) m_slots.size() * 2);
91 uint32_t i = jit_flat_hash(key) & m_mask;
92 for (; m_slots[i].full; i = (i + 1) & m_mask)
93 if (m_slots[i].key == key) { inserted = false; return m_slots[i].val; }
94 m_slots[i].key = key; m_slots[i].val = -1; m_slots[i].full = true;
95 m_used.push_back(i);
96 inserted = true;
97 return m_slots[i].val;
98 }
99 void add(const Key& key) { bool inserted; (void) get(key, inserted); }
100 int32_t* find(const Key& key)
101 {
102 for (uint32_t i = jit_flat_hash(key) & m_mask; m_slots[i].full; i = (i + 1) & m_mask)
103 if (m_slots[i].key == key) return &m_slots[i].val;
104 return nullptr;
105 }
106 template <class F> void for_each(F f) const { for (uint32_t i : m_used) f(m_slots[i].key); }
107 size_t size() const { return m_used.size(); }
108 void clear()
109 {
110 for (uint32_t i : m_used) m_slots[i].full = false;
111 m_used.clear();
112 }
113
114private:
115 struct Slot { Key key; int32_t val; bool full; };
116 std::vector<Slot> m_slots;
117 std::vector<uint32_t> m_used; // occupied slot indices, in insertion order
118 uint32_t m_mask = 0;
119
120 void rebuild(uint32_t capacity) // capacity: a power of two
121 {
122 std::vector<Slot> old;
123 old.swap(m_slots);
124 std::vector<uint32_t> used;
125 used.swap(m_used);
126 m_slots.assign(capacity, Slot{});
127 m_used.reserve(capacity / 2);
128 m_mask = capacity - 1;
129 for (uint32_t o : used) {
130 uint32_t i = jit_flat_hash(old[o].key) & m_mask;
131 while (m_slots[i].full) i = (i + 1) & m_mask;
132 m_slots[i] = old[o];
133 m_used.push_back(i);
134 }
135 }
136};

138class CJitEngine
139{
140public:
141 // 256K slots: the OS-active CPU's block working set (50K+) thrashed the old 16K direct-mapped cache
142 // (~190K recompiles/100M); 64K cut that to ~14K/100M, but a 50K set in 64K slots still conflict-evicts
143 // (load ~0.8). 256K drops the load to ~0.2. JitBlock ~110 B -> ~28 MB/CPU of metadata.
144 //
145 // SET-ASSOCIATIVE: more slots could never fix the churn. Every Unix process has
146 // its text at the SAME virtual addresses, so a virt_pc-only index puts every
147 // process's copy of a/ block in one slot with only the ASN telling them apart
148 // so they evit each other on every context switch, at any table size.
149 static constexpr int kCacheBits = 18;
150 static constexpr int kCacheEntries = 1 << kCacheBits; // TOTAL entries (kSets * kWays)
151 static constexpr int kWays = 8; // keys per set; raise if fresh-cause asn stays high
152 static constexpr int kSets = kCacheEntries / kWays;
153 static constexpr uint64_t kSetMask = (uint64_t) kSets - 1;
154 static_assert(kWays > 0 && (kWays & (kWays - 1)) == 0, "kWays must be a power of 2 (victim cursor masks with kWays-1)");
155 static_assert(kSets * kWays == kCacheEntries, "kSets * kWays must cover the whole table (flush scans it linearly)");
156
157 // Trace tier: a small SECOND cache, beside m_blocks, for hot superblock heads.
158 // Only the hottest loop heads are promoted, so it stays small.
159 static constexpr int kTraceBits = 12; // 4K trace heads
160 static constexpr uint64_t kTraceEntries = 1 << kTraceBits;
161 static constexpr uint64_t kTraceIndexMask = kTraceEntries - 1;
162 static constexpr uint32_t kMaxTraceSegs = 16; // fused blocks per trace (multi-block coherence)
163 static constexpr uint32_t kMaxTraceExits = 16; // guards / side-exits per trace
164
165 // Reclaim executable memory once compiled code passes this many bytes, rather
166 // than tearing down the asmjit runtime on every flush (see flush()).
167 static constexpr uint64_t kReclaimBytes = 128 * 1024 * 1024; // a VMS boot alone passes 32 MB six times

181 // Packed successor snapshot for the poly-link chain guard: everything the emitted guard reads
182 // lives in the SOURCE block's cache line meaning no dereference into the successor's JitBlock on the
183 // hot path.
184 struct LinkSlot
185 {
186 uint64_t tag; // this exit's target virtual PC (entries are keyed, so a hit is correct for that PC)
187 uint64_t vgen; // low 63 bits: validation epoch; bit 63: target PAL shadow variant
188 void* body; // successor's chained entry point at patch time; null = empty slot
189 };

191 // Target-keyed computed-jump cache, probed on a per-site PIC miss before the resolver.
192 struct JumpCacheEntry
193 {
194 uint64_t tag; // target virtual PC (bit 0 = PALmode)
195 uint64_t gen; // low 63 bits: m_jc_gen at fill; bit 63: target PAL shadow variant
196 void* body; // target's chained entry point
197 uint64_t asn; // I-stream ASN the body was resolved under
198 };
199 static_assert(sizeof(JumpCacheEntry) == 32, "jump-cache entry stride");
200 static constexpr int kJumpCacheEntries = 4096;
201 static inline uint32_t jcache_index(uint64_t target) { return (uint32_t) (target >> 2) & (kJumpCacheEntries - 1); }
202 inline void jcache_fill(uint64_t target, bool shadow, void* body, uint32_t asn)
203 {
204 JumpCacheEntry& e = m_jcache[jcache_index(target)];
205 e.tag = target; e.gen = m_jc_gen | (shadow ? (uint64_t) 1 << 63 : 0); e.body = body; e.asn = asn;
206 }

219 // Permission bits stay with the emitted code, which can outlive its
220 // JitBlock cache slot.
221 struct NativeExecProof
222 {
223 uint64_t epoch; // epoch of the last mode_mask change (JitBlock::exec_auth is current)
224 uint32_t mode_mask; // modes with execute access
225 uint32_t guard_imm_back; // x64: bytes back from this proof to TEST's imm8; A64: zero
226 };
227 static_assert(offsetof(NativeExecProof, epoch) == 0, "native proof epoch offset");
228 static_assert(offsetof(NativeExecProof, mode_mask) == 8, "native proof mode offset");
229 static_assert(offsetof(NativeExecProof, guard_imm_back) == 12, "native guard immediate offset");
230 static_assert(sizeof(NativeExecProof) == 16, "native execution proof layout");

232 // FIELD ORDER IS LOAD-BEARING: everything lookup() reads -- tag, asn, asm_global,
233 // pal_shadow, valid, flush_gen -- is packed into the first 40 bytes so a way costs ONE
234 // cache line to test.
235 struct JitBlock
236 {
237 uint64_t tag; // start VIRTUAL PC (validity tag / key)
238 uint64_t phys; // start physical PC (source bytes for compilation)
239 uint32_t asn; // address space number (key; ignored when asm_global)
240 bool asm_global; // global (ASM) page: matches any ASN, like the icache
241 bool pal_shadow; // PALmode variant: true maps R4-7/R20-23 to the shadow bank
242 uint32_t n_instr; // instructions in the straight-line block
243 bool valid;
244 uint64_t flush_gen; // global or process-specific flush generation at last source validation
245 JitFn code; // compiled safe-prefix, or null (prologue entry, for C calls)
246 void* jit_body; // chained re-entry point (after the prologue); null when not compiled
247 void* jit_body_same_page; // after native mode guard; only static same-page edges, null for PAL
248 NativeExecProof* exec_proof; // embedded data for this code allocation; null for PAL
249 uint64_t exec_auth; // copy of *exec_proof: epoch << 4 | mode_mask; set only via set_exec_proof
250 NativeExecProof* exec_proof_rw; // x64: writable alias of exec_proof, null until first authorized
251 LinkSlot link[kLinkSlots]; // cached direct successors (poly-link, round-robin back-patched)
252#ifdef JIT_STATS
253 uint32_t link_misses; // instrumentation: per-source link-miss count, cumulative (poly-link sizing)
254 uint8_t link_fanout; // distinct re-link targets seen (saturates at 4; >4 => a small successor cache won't help)
255 uint64_t link_seen[4];
256#endif
257 uint32_t prefix_len; // # safe ALU ops in code
258 bool compiled; // compile has been attempted
259 uint8_t compile_encounters; // eligible cold passes, saturated at 2; separate from trace hotness
260 uint32_t body_off; // jit_body's offset within code -- restores the chained entry on revalidate
261 uint64_t src_sum; // hash of the source words at compile time (revalidate vs self-mod)
262 uint32_t hash_len; // word count src_sum covers -- frozen at compile time; n_instr drifts
263 // (interrupt-truncated cold passes shrink it), so it must NOT key the hash
264 uint64_t vgen; // m_vgen_cur at last full validation (phys + code bytes). The counter is
265 // monotonic, so one compare detects any invalidating event since then
266 // -- the single chain guard (see emit_chain / jit_indirect).
267 uint32_t hot; // dispatches since record; at the promote threshold -> form a trace
268#ifdef JIT_STATS
269 uint64_t st_exit1; // STATS: dispatcher returns through this block's gate/guard exit (kind 1)
270#endif
271#ifdef JIT_REGPROF
272 uint64_t rp_hits; // REGPROF: block executions since record (body-entry inc -- counts chained runs)
273 uint32_t rp_mask; // REGPROF: Alpha GPRs touched by this block (bit r); compile-time, exec-weighted at report
274 uint32_t rp_csz; // REGPROF: emitted host bytes for this block -- rp_hits x rp_csz = exec-weighted expansion
275#endif
276 };

362 // mirroring the icache's hit rule.

364 bool pal_shadow)
365 {
366 return (b.valid || b.code) && b.tag == virt_pc && (b.asm_global || b.asn == asn)
367 && (!(virt_pc & 1) || b.pal_shadow == pal_shadow);

375 // Virtual+ASN keyed: no translation on the dispatch hot path. flush_gen-stale blocks
376 // miss here; revalidate_flushed() resurrects them after a source-hash check.
377 inline JitBlock* lookup(uint64_t virt_pc, uint32_t asn, bool pal_shadow)
378 {
379 JitBlock* const set = set_base(virt_pc);
380 for (int w = 0; w < kWays; ++w) {
381 JitBlock& b = set[w];
382 if (b.valid && b.flush_gen == flush_generation(b.asm_global) && b.tag == virt_pc
383 && (b.asm_global || b.asn == asn)
384 && (!(virt_pc & 1) || b.pal_shadow == pal_shadow))
385 return &b;
386 }
387 return nullptr;
388 }
