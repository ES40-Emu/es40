57static std::string helper_next_to_executable()
58{
59 char path[PATH_MAX];
60 char resolved[PATH_MAX];
61 uint32_t size = sizeof(path);
62
63 if (_NSGetExecutablePath(path, &size) != 0 ||
64 realpath(path, resolved) == nullptr)
65 return std::string();
66 char *slash = strrchr(resolved, '/');
67 if (slash == nullptr)
68 return std::string();
69 *(slash + 1) = '\0';
70 return std::string(resolved) + ES40_VMNET_HELPER_NAME;
71}

73static std::string find_helper()
74{
75 std::string adjacent = helper_next_to_executable();
76 if (!adjacent.empty() && access(adjacent.c_str(), X_OK) == 0)
77 return adjacent;
78 if (geteuid() == 0)
79 return std::string();
80 return ES40_VMNET_HELPER_NAME;
81}

83static bool set_cloexec(int socket_fd)
84{
85 int flags = fcntl(socket_fd, F_GETFD, 0);
86 return flags >= 0 &&
87 fcntl(socket_fd, F_SETFD, flags | FD_CLOEXEC) == 0;
88}

90static bool prepare_socket(int socket_fd)
91{
92 int one = 1;
93 int size = socket_buffer_size;
94 struct timeval timeout;
95
96 timeout.tv_sec = helper_timeout_seconds;
97 timeout.tv_usec = 0;
98 if (!set_cloexec(socket_fd) ||
99 setsockopt(socket_fd, SOL_SOCKET, SO_NOSIGPIPE, &one,
100 sizeof(one)) != 0 ||
101 setsockopt(socket_fd, SOL_SOCKET, SO_RCVTIMEO, &timeout,
102 sizeof(timeout)) != 0)
103 return false;
104
105 /* Larger queues reduce drops, but the platform default still works. */
106 setsockopt(socket_fd, SOL_SOCKET, SO_SNDBUF, &size, sizeof(size));
107 setsockopt(socket_fd, SOL_SOCKET, SO_RCVBUF, &size, sizeof(size));
108 return true;
109}

111bool CNetworkVmnetIPC::start(const char *devid_string, const char *adapter,
112 struct es40_vmnet_open_reply *reply)
113{
114 int sockets[2] = { -1, -1 };
115
116 if (socketpair(AF_UNIX, SOCK_DGRAM, 0, sockets) != 0 ||
117 !prepare_socket(sockets[0]) || !prepare_socket(sockets[1])) {
118 printf("%s: Cannot create vmnet helper socket: %s\n",
119 devid_string, strerror(errno));
120 if (sockets[0] >= 0)
121 ::close(sockets[0]);
122 if (sockets[1] >= 0)
123 ::close(sockets[1]);
124 return false;
125 }
126
127 helper_path = find_helper();
128 if (helper_path.empty()) {
129 printf("%s: Cannot find %s next to the ES40 executable.\n",
130 devid_string, ES40_VMNET_HELPER_NAME);
131 ::close(sockets[0]);
132 ::close(sockets[1]);
133 return false;
134 }
135
136 int child_fd = fcntl(sockets[1], F_DUPFD_CLOEXEC, 10);
137 if (child_fd < 0) {
138 printf("%s: Cannot prepare vmnet helper socket: %s\n",
139 devid_string, strerror(errno));
140 ::close(sockets[0]);
141 ::close(sockets[1]);
142 return false;
143 }
144
145 posix_spawn_file_actions_t actions;
146 int rc = posix_spawn_file_actions_init(&actions);
147 bool actions_initialized = (rc == 0);
148 if (rc == 0)
149 rc = posix_spawn_file_actions_adddup2(&actions, child_fd,
150 ES40_VMNET_HELPER_FD);
151 if (rc == 0)
152 rc = posix_spawn_file_actions_addclose(&actions, child_fd);
153
154 pid_t pid = 0;
155 char *const argv[] = { (char *)ES40_VMNET_HELPER_NAME, nullptr };
156 if (rc == 0) {
157 rc = posix_spawnp(&pid, helper_path.c_str(), &actions, nullptr,
158 argv, environ);
159 }
160 if (actions_initialized)
161 posix_spawn_file_actions_destroy(&actions);
162 ::close(child_fd);
163 ::close(sockets[1]);
164
165 if (rc != 0) {
166 printf("%s: Cannot start %s: %s\n", devid_string,
167 helper_path.c_str(), strerror(rc));
168 ::close(sockets[0]);
169 return false;
170 }
171
172 fd = sockets[0];
173 helper_pid = pid;
174 struct es40_vmnet_open_req request;
175 memset(&request, 0, sizeof(request));
176 request.msg.type = ES40_VMNET_MSG_OPEN;
177 request.magic = ES40_VMNET_PROTO_MAGIC;
178 request.version = ES40_VMNET_PROTO_VERSION;
179 strlcpy(request.adapter, adapter, sizeof(request.adapter));
180
181 ssize_t sent;
182 do {
183 sent = ::send(fd, &request, sizeof(request), 0);
184 } while (sent < 0 && errno == EINTR);
185 ssize_t received = -1;
186 if (sent == (ssize_t)sizeof(request)) {
187 struct iovec iov = { reply, sizeof(*reply) };
188 struct msghdr message;
189 memset(&message, 0, sizeof(message));
190 message.msg_iov = &iov;
191 message.msg_iovlen = 1;
192 do {
193 received = recvmsg(fd, &message, 0);
194 } while (received < 0 && errno == EINTR);
195 if (message.msg_flags & MSG_TRUNC)
196 received = -1;
197 }
198 if (sent != (ssize_t)sizeof(request) ||
199 received != (ssize_t)sizeof(*reply)) {
200 printf("%s: %s did not answer the vmnet open request.\n",
201 devid_string, helper_path.c_str());
202 close();
203 return false;
204 }
205 return true;
206}

208int CNetworkVmnetIPC::send_packet(const void *data, size_t len)
209{
210 struct es40_vmnet_msg header = { ES40_VMNET_MSG_PACKET };
211 struct iovec iov[2] = {
212 { &header, sizeof(header) },
213 { (void *)data, len }
214 };
215 struct msghdr message;
216 memset(&message, 0, sizeof(message));
217 message.msg_iov = iov;
218 message.msg_iovlen = 2;
219
220 ssize_t sent;
221 do {
222 sent = sendmsg(fd, &message, MSG_DONTWAIT);
223 } while (sent < 0 && errno == EINTR);
224 if (sent == (ssize_t)(sizeof(header) + len))
225 return 0;
226 if (sent < 0 && (errno == EAGAIN || errno == EWOULDBLOCK ||
227 errno == ENOBUFS))
228 return 0;
229 if (sent >= 0)
230 errno = EIO;
231 return -1;
232}

234int CNetworkVmnetIPC::receive(void *data, size_t capacity, int *len,
235 struct es40_vmnet_status *status)
236{
237 struct es40_vmnet_msg header;
238 struct iovec iov[2] = {
239 { &header, sizeof(header) },
240 { data, capacity }
241 };
242 struct msghdr message;
243 memset(&message, 0, sizeof(message));
244 message.msg_iov = iov;
245 message.msg_iovlen = 2;
246
247 ssize_t received;
248 do {
249 received = recvmsg(fd, &message, MSG_DONTWAIT);
250 } while (received < 0 && errno == EINTR);
251 if (received < 0) {
252 if (errno == EAGAIN || errno == EWOULDBLOCK)
253 return 0;
254 return -1;
255 }
256 if ((message.msg_flags & MSG_TRUNC) ||
257 received < (ssize_t)sizeof(header)) {
258 errno = EPROTO;
259 return -1;
260 }
261
262 size_t payload_len = (size_t)received - sizeof(header);
263 if (header.type == ES40_VMNET_MSG_PACKET) {
264 if (payload_len == 0) {
265 errno = EPROTO;
266 return -1;
267 }
268 *len = (int)payload_len;
269 return 1;
270 }
271 if (header.type == ES40_VMNET_MSG_STATUS &&
272 payload_len == sizeof(*status) - sizeof(header)) {
273 status->msg = header;
274 memcpy((char *)status + sizeof(header), data, payload_len);
275 return 2;
276 }
277
278 errno = EPROTO;
279 return -1;
280}

282static bool reap(pid_t pid, int milliseconds)
283{
284 int status;
285 int attempts = milliseconds / 50;
286
287 for (int i = 0; i < attempts; i++) {
288 pid_t result = waitpid(pid, &status, WNOHANG);
289 if (result == pid || (result < 0 && errno == ECHILD))
290 return true;
291 if (result < 0 && errno != EINTR)
292 return false;
293 usleep(50000);
294 }
295 return false;
296}

298void CNetworkVmnetIPC::close()
299{
300 if (fd >= 0) {
301 struct es40_vmnet_msg shutdown = { ES40_VMNET_MSG_SHUTDOWN };
302 ::send(fd, &shutdown, sizeof(shutdown), MSG_DONTWAIT);
303 ::close(fd);
304 fd = -1;
305 }
306 if (helper_pid > 0) {
307 if (!reap(helper_pid, 2000)) {
308 kill(helper_pid, SIGTERM);
309 if (!reap(helper_pid, 1000)) {
310 kill(helper_pid, SIGKILL);
311 reap(helper_pid, 1000);
312 }
313 }
314 helper_pid = 0;
315 }
316}
