68bool valid_format(const Format& format)
69{
70 return format.freq > 0 && format.freq <= 768000 &&
71 (format.channels == 1 || format.channels == 2) &&
72 (format.format == SampleFormat::u8 || format.format == SampleFormat::s16_le);
73}
74
75bool same_format(const Format& a, const Format& b)
76{
77 return a.freq == b.freq && a.channels == b.channels && a.format == b.format;
78}
79
80size_t frame_bytes(const Format& format)
81{
82 return format.channels * (format.format == SampleFormat::u8 ? 1 : 2);
83}
84
85WAVEFORMATEX wave_format(const Format& format)
86{
87 WAVEFORMATEX wave = {};
88 wave.wFormatTag = WAVE_FORMAT_PCM;
89 wave.nChannels = static_cast<WORD>(format.channels);
90 wave.nSamplesPerSec = format.freq;
91 wave.wBitsPerSample = format.format == SampleFormat::u8 ? 8 : 16;
92 wave.nBlockAlign = static_cast<WORD>(frame_bytes(format));
93 wave.nAvgBytesPerSec = wave.nSamplesPerSec * wave.nBlockAlign;
94 return wave;
95}

97struct ByteQueue
98{
99 std::vector<unsigned char> data;
100 size_t first = 0;
101
102 size_t size() const { return data.size() - first; }
103 const unsigned char* front() const { return data.data() + first; }
104 void clear() { data.clear(); first = 0; }
105 void consume(size_t bytes)
106 {
107 first += bytes;
108 if (first == data.size()) clear();
109 }
110 void append(const void* source, size_t bytes)
111 {
112 if (!bytes) return;
113 if (first && (first >= data.size() / 2 || data.size() + bytes > queue_limit)) {
114 data.erase(data.begin(), data.begin() + first);
115 first = 0;
116 }
117 const auto* begin = static_cast<const unsigned char*>(source);
118 data.insert(data.end(), begin, begin + bytes);
119 }
120};

122int sample(const unsigned char* frame, const Format& format, int channel)
123{
124 if (format.channels == 1) channel = 0;
125 if (format.format == SampleFormat::u8) return (int(frame[channel]) - 128) * 256;
126 frame += channel * 2;
127 const int value = int(frame[0]) | (int(frame[1]) << 8);
128 return value >= 32768 ? value - 65536 : value;
129}
130
131void put_sample(unsigned char* destination, SampleFormat format, int value)
132{
133 value = (std::max)(-32768, (std::min)(32767, value));
134 if (format == SampleFormat::u8) {
135 *destination = static_cast<unsigned char>((value + 32768) >> 8);
136 } else {
137 const unsigned bits = static_cast<unsigned>(value);
138 destination[0] = static_cast<unsigned char>(bits);
139 destination[1] = static_cast<unsigned char>(bits >> 8);
140 }
141}

143// Rational phase preserves rate and interpolation across arbitrarily split writes.
144size_t convert(ByteQueue& input, const Format& source, const Format& target,
145 std::uint64_t& phase, unsigned char* output, size_t frames)
146{
147 const size_t source_bytes = frame_bytes(source);
148 const size_t target_bytes = frame_bytes(target);
149 const size_t sample_bytes = target.format == SampleFormat::u8 ? 1 : 2;
150 size_t produced = 0;
151 while (produced < frames) {
152 const size_t index = static_cast<size_t>(phase / target.freq);
153 const int fraction = static_cast<int>(phase % target.freq);
154 if (index + 1 + (fraction != 0) > input.size() / source_bytes) break;
155 const unsigned char* a = input.front() + index * source_bytes;
156 const unsigned char* b = fraction ? a + source_bytes : a;
157 int stereo[2];
158 for (int channel = 0; channel < 2; ++channel) {
159 const int first = sample(a, source, channel);
160 const int second = sample(b, source, channel);
161 stereo[channel] = first + static_cast<int>(
162 (std::int64_t(second - first) * fraction) / target.freq);
163 }
164 if (target.channels == 1) stereo[0] = (stereo[0] + stereo[1]) / 2;
165 for (int channel = 0; channel < target.channels; ++channel)
166 put_sample(output + channel * sample_bytes, target.format, stereo[channel]);
167 output += target_bytes;
168 ++produced;
169 phase += source.freq;
170 const size_t consumed = (std::min)(static_cast<size_t>(phase / target.freq),
171 input.size() / source_bytes);
172 input.consume(consumed * source_bytes);
173 phase -= std::uint64_t(consumed) * target.freq;
174 }
175 return produced;
176}

189struct NativeStream
190{
191 Stream& owner;
192 std::recursive_mutex mutex;
193 std::condition_variable_any wake;
194 std::thread worker;
195 bool quitting = false;
196 bool capture;
197 bool restart = false;
198 bool failed = false;
199 char failure[256] = {};
200 std::uintptr_t device = 0;
201 Format guest;
202 Format host{host_rate, 2, SampleFormat::s16_le};
203 IDirectSoundBuffer* output = nullptr;
204 IDirectSoundCaptureBuffer* input = nullptr;
205 DWORD ring_bytes = 0;
206 DWORD previous_cursor = 0;
207 std::uint64_t play_position = 0;
208 std::uint64_t producer = 0;
209 std::uint64_t phase = 0;
210 Clock::time_point last_service;
211 Clock::time_point retry_after;
212 ByteQueue source;
213 ByteQueue captured;
214 std::vector<unsigned char> scratch;
215
216 NativeStream(Stream& stream, bool recording, const Format& format)
217 : owner(stream), capture(recording), guest(format) {}
218 ~NativeStream() { release_buffers(); }
219
220 bool check()
221 {
222 if (!failed) return true;
223 std::memcpy(last_error, failure, sizeof(last_error));
224 return false;
225 }
226 bool remember(const char* operation, HRESULT result)
227 {
228 fail(operation, result);
229 std::memcpy(failure, last_error, sizeof(failure));
230 failed = true;
231 if (output) output->Stop();
232 if (input) input->Stop();
233 return false;
234 }
235 void release_buffers()
236 {
237 if (output) { output->Stop(); output->Release(); output = nullptr; }
238 if (input) { input->Stop(); input->Release(); input = nullptr; }
239 device = 0;
240 restart = false;
241 }
242 bool retry_restore(HRESULT result)
243 {
244 if (result != DSERR_BUFFERLOST && result != DSERR_PRIOLEVELNEEDED &&
245 result != DSERR_OTHERAPPHASPRIO) return false;
246 restart = true;
247 retry_after = Clock::now() + std::chrono::milliseconds(20);
248 return true;
249 }
250 bool silence(DWORD offset, DWORD bytes)
251 {
252 if (!bytes) return true;
253 void* first = nullptr;
254 void* second = nullptr;
255 DWORD first_bytes = 0, second_bytes = 0;
256 const HRESULT result = output->Lock(offset, bytes, &first, &first_bytes,
257 &second, &second_bytes, 0);
258 if (result == DSERR_BUFFERLOST) { restart = true; return true; }
259 if (FAILED(result)) return remember("DirectSound buffer lock failed", result);
260 std::memset(first, 0, first_bytes);
261 if (second_bytes) std::memset(second, 0, second_bytes);
262 const HRESULT unlocked = output->Unlock(first, first_bytes, second, second_bytes);
263 return SUCCEEDED(unlocked) || remember("DirectSound buffer unlock failed", unlocked);
264 }
265 bool start_output()
266 {
267 HRESULT result = output->Stop();
268 if (FAILED(result) && result != DSERR_BUFFERLOST)
269 return remember("DirectSound playback stop failed", result);
270 DWORD status = 0;
271 result = output->GetStatus(&status);
272 if (FAILED(result)) return retry_restore(result) ||
273 remember("DirectSound playback status failed", result);
274 if (status & DSBSTATUS_BUFFERLOST) {
275 result = output->Restore();
276 if (FAILED(result)) return retry_restore(result) ||
277 remember("DirectSound buffer restore failed", result);
278 }
279 restart = false;
280 if (!silence(0, ring_bytes) || restart) return !failed;
281 result = output->SetCurrentPosition(0);
282 if (FAILED(result)) return remember("DirectSound playback position failed", result);
283 previous_cursor = 0;
284 play_position = producer = 0;
285 last_service = Clock::now();
286 result = output->Play(0, 0, DSBPLAY_LOOPING);
287 if (retry_restore(result)) return true;
288 return SUCCEEDED(result) || remember("DirectSound playback start failed", result);
289 }
290 bool pump_output()
291 {
292 const auto now = Clock::now();
293 if (restart && now < retry_after) return true;
294 if (now - last_service >= std::chrono::milliseconds(200)) restart = true;
295 DWORD status = 0;
296 HRESULT result = output->GetStatus(&status);
297 if (FAILED(result)) return remember("DirectSound playback status failed", result);
298 if (status & DSBSTATUS_BUFFERLOST) restart = true;
299 if (restart && (!start_output() || restart)) return !failed;
300 DWORD play = 0, safe = 0;
301 result = output->GetCurrentPosition(&play, &safe);
302 if (result == DSERR_BUFFERLOST) { restart = true; return true; }
303 if (FAILED(result)) return remember("DirectSound playback cursor failed", result);
304 const DWORD frame = static_cast<DWORD>(frame_bytes(host));
305 play -= play % frame;
306 safe = ((safe + frame - 1) / frame * frame) % ring_bytes;
307 play_position += (play + ring_bytes - previous_cursor) % ring_bytes;
308 previous_cursor = play;
309 last_service = now;
310 const std::uint64_t safe_position = play_position + (safe + ring_bytes - play) % ring_bytes;
311 producer = (std::max)(producer, safe_position);
312 const std::uint64_t end = (std::min)(safe_position + period_frames * frame,
313 play_position + ring_bytes - frame);
314 if (producer < end) {
315 const size_t bytes = static_cast<size_t>(end - producer);
316 scratch.assign(bytes, 0);
317 convert(source, guest, host, phase, scratch.data(), bytes / frame);
318 void* first = nullptr;
319 void* second = nullptr;
320 DWORD first_bytes = 0, second_bytes = 0;
321 result = output->Lock(static_cast<DWORD>(producer % ring_bytes),
322 static_cast<DWORD>(bytes), &first, &first_bytes, &second, &second_bytes, 0);
323 if (result == DSERR_BUFFERLOST) { restart = true; return true; }
324 if (FAILED(result)) return remember("DirectSound playback lock failed", result);
325 std::memcpy(first, scratch.data(), first_bytes);
326 if (second_bytes) std::memcpy(second, scratch.data() + first_bytes, second_bytes);
327 result = output->Unlock(first, first_bytes, second, second_bytes);
328 if (FAILED(result)) return remember("DirectSound playback unlock failed", result);
329 producer = end;
330 }
331 // Clear the remaining safe region so underruns cannot replay old samples.
332 const DWORD clear_bytes = static_cast<DWORD>(play_position + ring_bytes - producer);
333 if (!silence(static_cast<DWORD>(producer % ring_bytes), clear_bytes)) return false;
334 if (owner.callback) owner.callback(owner.userdata);
335 return true;
336 }
337 bool start_input()
338 {
339 HRESULT result = input->Stop();
340 if (FAILED(result)) return remember("DirectSound capture stop failed", result);
341 result = input->Start(DSCBSTART_LOOPING);
342 if (FAILED(result)) return remember("DirectSound capture start failed", result);
343 result = input->GetCurrentPosition(nullptr, &previous_cursor);
344 if (FAILED(result)) return remember("DirectSound capture cursor failed", result);
345 previous_cursor -= previous_cursor % frame_bytes(host);
346 last_service = Clock::now();
347 restart = false;
348 return true;
349 }
350 bool pump_input()
351 {
352 if (restart && !start_input()) return false;
353 DWORD readable = 0;
354 HRESULT result = input->GetCurrentPosition(nullptr, &readable);
355 if (FAILED(result)) return remember("DirectSound capture cursor failed", result);
356 const DWORD frame = static_cast<DWORD>(frame_bytes(host));
357 readable -= readable % frame;
358 const auto now = Clock::now();
359 if (now - last_service >= std::chrono::milliseconds(200)) {
360 previous_cursor = readable;
361 source.clear();
362 phase = 0;
363 }
364 last_service = now;
365 const DWORD bytes = (readable + ring_bytes - previous_cursor) % ring_bytes;
366 if (!bytes) return true;
367 scratch.resize(bytes);
368 void* first = nullptr;
369 void* second = nullptr;
370 DWORD first_bytes = 0, second_bytes = 0;
371 result = input->Lock(previous_cursor, bytes, &first, &first_bytes,
372 &second, &second_bytes, 0);
373 if (FAILED(result)) return remember("DirectSound capture lock failed", result);
374 std::memcpy(scratch.data(), first, first_bytes);
375 if (second_bytes) std::memcpy(scratch.data() + first_bytes, second, second_bytes);
376 result = input->Unlock(first, first_bytes, second, second_bytes);
377 if (FAILED(result)) return remember("DirectSound capture unlock failed", result);
378 previous_cursor = readable;
379 source.append(scratch.data(), bytes);
380 const size_t guest_frame = frame_bytes(guest);
381 const size_t capacity = static_cast<size_t>(
382 (std::uint64_t(source.size() / frame) + 1) * guest.freq / host.freq + 2);
383 scratch.resize(capacity * guest_frame);
384 const size_t produced = convert(source, host, guest, phase, scratch.data(), capacity) * guest_frame;
385 if (captured.size() + produced > queue_limit) {
386 const size_t excess = captured.size() + produced - queue_limit;
387 captured.consume((std::min)(captured.size(),
388 (excess + guest_frame - 1) / guest_frame * guest_frame));
389 }
390 captured.append(scratch.data(), produced);
391 if (produced && owner.callback) owner.callback(owner.userdata);
392 return true;
393 }
394 void run()
395 {
396 std::unique_lock<std::recursive_mutex> lock(mutex);
397 while (!quitting) {
398 try {
399 if (device && !failed) {
400 if (capture) pump_input();
401 else pump_output();
402 }
403 } catch (...) {
404 remember("DirectSound audio worker failed", E_OUTOFMEMORY);
405 }
406 wake.wait_for(lock, std::chrono::milliseconds(2));
407 }
408 }
409};

454void create_stream(Stream& stream, const Format* source, const Format* target)
455{
456 stream.native = nullptr;
457 const Format* guest = source ? source : target;
458 if (!guest || !valid_format(*guest)) { fail("Invalid DirectSound stream format", E_INVALIDARG); return; }
459 try {
460 std::unique_ptr<NativeStream> instance(new NativeStream(stream, source == nullptr, *guest));
461 instance->worker = std::thread(&NativeStream::run, instance.get());
462 stream.native = instance.release();
463 } catch (...) {

514bool bind_stream(std::uintptr_t handle, Stream& stream)
515{
516 auto* instance = native(stream);
517 auto* device = reinterpret_cast<Device*>(handle);
518 if (!instance) return false;
519 if (!device) return fail("Invalid DirectSound audio device", E_POINTER);
520 std::lock_guard<std::recursive_mutex> lock(instance->mutex);
521 if (!instance->check()) return false;
522 if (instance->device == handle) return true;
523 if (instance->device) return fail("DirectSound stream already bound", E_INVALIDARG);
524 HRESULT result = E_FAIL;
525 if (instance->capture) {
526 if (!device->input) return fail("Expected a DirectSound capture device", E_INVALIDARG);
527 const Format choices[] = {{48000, 2, SampleFormat::s16_le}, {44100, 2, SampleFormat::s16_le},
528 {48000, 1, SampleFormat::s16_le}, {44100, 1, SampleFormat::s16_le}};
529 for (const auto& format : choices) {
530 WAVEFORMATEX wave = wave_format(format);
531 DSCBUFFERDESC descriptor = {};
532 descriptor.dwSize = sizeof(descriptor);
533 descriptor.dwBufferBytes = ring_frames * wave.nBlockAlign;
534 descriptor.lpwfxFormat = &wave;
535 result = device->input->CreateCaptureBuffer(&descriptor, &instance->input, nullptr);
536 if (SUCCEEDED(result)) {
537 instance->host = format;
538 instance->ring_bytes = descriptor.dwBufferBytes;
539 break;
540 }
541 }
542 if (FAILED(result)) return instance->remember("DirectSound capture buffer creation failed", result);
543 instance->source.clear();
544 instance->phase = 0;
545 } else {
546 if (!device->output) return fail("Expected a DirectSound playback device", E_INVALIDARG);
547 WAVEFORMATEX wave = wave_format(instance->host);
548 DSBUFFERDESC descriptor = {};
549 descriptor.dwSize = sizeof(descriptor);
550 descriptor.dwFlags = DSBCAPS_GLOBALFOCUS | DSBCAPS_GETCURRENTPOSITION2;
551 descriptor.dwBufferBytes = ring_frames * wave.nBlockAlign;
552 descriptor.lpwfxFormat = &wave;
553 result = device->output->CreateSoundBuffer(&descriptor, &instance->output, nullptr);
554 if (FAILED(result)) return instance->remember("DirectSound playback buffer creation failed", result);
555 instance->ring_bytes = descriptor.dwBufferBytes;
556 }
557 instance->device = handle;
558 instance->restart = true;
559 instance->wake.notify_one();
560 return true;
561}

621int read(Stream& stream, void* destination, int bytes)
622{
623 auto* instance = native(stream);
624 if (!instance) return -1;
625 std::lock_guard<std::recursive_mutex> lock(instance->mutex);
626 if (!instance->check()) return -1;
627 if (!instance->capture || bytes < 0 || (!destination && bytes)) {
628 fail("Invalid DirectSound capture read", E_INVALIDARG);
629 return -1;
630 }
631 const size_t count = (std::min)(static_cast<size_t>(bytes), instance->captured.size());
632 if (count) std::memcpy(destination, instance->captured.front(), count);
633 instance->captured.consume(count);
634 return static_cast<int>(count);
635}

637bool write(Stream& stream, const void* source, int bytes)
638{
639 auto* instance = native(stream);
640 if (!instance) return false;
641 std::lock_guard<std::recursive_mutex> lock(instance->mutex);
642 if (!instance->check()) return false;
643 if (instance->capture || bytes < 0 || (!source && bytes))
644 return fail("Invalid DirectSound playback write", E_INVALIDARG);
645 if (static_cast<size_t>(bytes) > queue_limit - instance->source.size())
646 return fail("DirectSound playback queue is full", E_OUTOFMEMORY);
647 try {
648 instance->source.append(source, bytes);
649 } catch (...) {
