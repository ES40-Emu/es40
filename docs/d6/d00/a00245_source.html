41// Read-only PAL calls with no GPR inputs and at most one GPR output.
42struct AlphaIdlePALCall
43{
44 bool allowed = false;
45 int destination = -1; // -1 means no GPR output.
46};

48struct AlphaIdlePolicy
49{
50 AlphaIdleContext (*observe)(const CAlphaCPU& cpu) = nullptr; // Null disables detection.
51 // Pure, stable classification; null rejects PAL-containing loops.
52 AlphaIdlePALCall (*describe_pal_call)(std::uint32_t function) = nullptr;
53};

55constexpr AlphaIdlePALCall alpha_idle_pal_call(const AlphaIdlePolicy& policy,
56 std::uint32_t function)
57{
58 const auto call = policy.describe_pal_call
59 ? policy.describe_pal_call(function) : AlphaIdlePALCall{};
60 return call.allowed && call.destination >= -1 && call.destination < 32
61 ? call : AlphaIdlePALCall{};
62}
