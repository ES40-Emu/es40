32AlphaMMUAccess Ev68MMU::access(const CAlphaCPU& cpu, int flags)
33{
34 const auto& state = cpu.state;
35 // HRM 6.4.1: VPTE fetches check kernel permissions; /alt uses DTB_ALT_MODE.
36 // VPTE takes precedence over ALT. Neither changes the selected lookup ASN.
37 return {
38 (flags & VPTE) ? 0 : (flags & ALT) ? state.alt_cm : state.cm,
39 (flags & ACCESS_EXEC) ? state.asn : state.asn0
40 };
41}

43AlphaMMUAddress Ev68MMU::map_address(const CAlphaCPU& cpu, std::uint64_t virt,
44 int flags, int mode)
45{
46 const auto& state = cpu.state;
47 const bool exec = (flags & ACCESS_EXEC) != 0;
48 const bool va48 = (exec ? state.i_ctl_va_mode : state.va_ctl_va_mode) & 1;
49 const u64 canonical = va48 ? sext_u64_48(virt)
50 : (virt & U64(0x0000040000000000)) ? (virt | U64(0xfffff80000000000))
51 : (virt & U64(0x000007ffffffffff));
52 if (canonical != virt)
53 return {AlphaMMUAddressKind::Invalid, 0};
54
55 const int spe = exec ? state.i_ctl_spe : state.m_ctl_spe;
56 u64 phys;
57 // HRM 5.3.9: SPE[2] maps VA[43:0], ignoring VA[45:44].
58 if ((spe & 4) && (virt & U64(0x0000c00000000000)) == U64(0x0000800000000000))
59 phys = virt & U64(0x00000fffffffffff);
60 // SPE[1] maps VA[40:0], sign-extending bit 40 through PA[43:41].
61 else if ((spe & 2) && (virt & U64(0x0000fe0000000000)) == U64(0x0000fc0000000000))
62 phys = (virt & U64(0x000001ffffffffff))
63 | ((virt & U64(0x0000010000000000)) ? U64(0x00000e0000000000) : 0);
64 // SPE[0] maps VA[29:0], clearing PA[43:30].
65 else if ((spe & 1) && (virt & U64(0x0000ffffc0000000)) == U64(0x0000ffff80000000))
66 phys = virt & U64(0x000000003fffffff);
67 else
68 return {AlphaMMUAddressKind::TranslationBuffer, 0};
69
70 // Superpages are kernel-only even for accesses carrying NO_CHECK.
71 return mode ? AlphaMMUAddress{AlphaMMUAddressKind::AccessViolation, 0}
72 : AlphaMMUAddress{AlphaMMUAddressKind::Direct, phys};
73}

75AlphaTBFill Ev68MMU::decode_fill(const CAlphaCPU&, std::uint64_t virt,
76 std::uint64_t pte_phys, std::uint64_t pte_flags,
77 int)
78{
79 // EV68CB/EV68DC HRM 5.3.1/5.3.2: VA[47:13], PA[43:13], GH[1:0].
80 // GH widens the granule by three bits per step (13/16/19/22 offset bits).
81 static constexpr std::uint64_t match_masks[4] = {
82 U64(0x0000ffffffffe000), U64(0x0000ffffffff0000),
83 U64(0x0000fffffff80000), U64(0x0000ffffffc00000)
84 };
85 static constexpr std::uint64_t phys_masks[4] = {
86 U64(0x00000fffffffe000), U64(0x00000fffffff0000),
87 U64(0x00000ffffff80000), U64(0x00000fffffc00000)
88 };
89 static constexpr std::uint64_t keep_masks[4] = {
90 U64(0x0000000000001fff), U64(0x000000000000ffff),
91 U64(0x000000000007ffff), U64(0x00000000003fffff)
92 };
93 const unsigned gh = (unsigned)((pte_flags >> 5) & 3);
94
95 AlphaTBFill fill{};
96 fill.virt = virt;
97 fill.phys = pte_phys & phys_masks[gh];
98 fill.match_mask = match_masks[gh];
99 fill.keep_mask = keep_masks[gh];
100 fill.fault[0] = (int)pte_flags & 2;
101 fill.fault[1] = (int)pte_flags & 4;
102 fill.fault[2] = (int)pte_flags & 8;
103 fill.access[0][0] = (int)pte_flags & 0x100;
104 fill.access[1][0] = (int)pte_flags & 0x1000;
105 fill.access[0][1] = (int)pte_flags & 0x200;
106 fill.access[1][1] = (int)pte_flags & 0x2000;
107 fill.access[0][2] = (int)pte_flags & 0x400;
108 fill.access[1][2] = (int)pte_flags & 0x4000;
109 fill.access[0][3] = (int)pte_flags & 0x800;
110 fill.access[1][3] = (int)pte_flags & 0x8000;
111 fill.asm_bit = (int)pte_flags & 0x10;
112 return fill;
113}
