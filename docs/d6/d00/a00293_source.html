34// EV6/EV68 PAL exception vectors.
35constexpr AlphaPALExceptionEntry exception_entry(AlphaPALException reason,
36 std::uint64_t current_pc)
37{
38 switch (reason)
39 {
40 case AlphaPALException::DataTlbDoubleMiss43: return {0x100, current_pc, false};
41 case AlphaPALException::DataTlbDoubleMiss48: return {0x180, current_pc, false};
42 case AlphaPALException::FloatingPointDisabled: return {0x200, current_pc, true};
43 case AlphaPALException::Unaligned: return {0x280, current_pc, true};
44 case AlphaPALException::DataTlbMiss: return {0x300, current_pc, false};
45 case AlphaPALException::DataFault: return {0x380, current_pc, true};
46 case AlphaPALException::OpcodeDecode: return {0x400, current_pc, true};
47 case AlphaPALException::InstructionAccessViolation: return {0x480, current_pc, true};
48 case AlphaPALException::MachineCheck: return {0x500, current_pc, true};
49 case AlphaPALException::InstructionTlbMiss: return {0x580, current_pc, false};
50 case AlphaPALException::Arithmetic: return {0x600, current_pc, true};
51 case AlphaPALException::Interrupt: return {0x680, current_pc, true};
52 case AlphaPALException::MtFpcr: return {0x700, current_pc, true};
53 case AlphaPALException::Reset: return {0x780, current_pc, true};
54 }
55 return {0, current_pc, true};
56}

58constexpr bool call_pal_valid(std::uint32_t function, std::uint32_t mode)
59{
60 return function <= 0x3f ? mode == 0 : function >= 0x80 && function <= 0xbf;
61}

63// EV68 records the CALL_PAL instruction address and writes next PC to R23,
64// or R55 when the PAL shadow bank is enabled.
65constexpr AlphaPALCallEntry call_pal_entry(std::uint32_t function,
66 std::uint64_t current_pc,
67 std::uint64_t next_pc)
68{
69 const std::uint64_t offset = std::uint64_t(0x2000)
70 | (std::uint64_t(function & 0x80) << 5)
71 | (std::uint64_t(function & 0x3f) << 6) | 1;
72 return {offset, current_pc, next_pc & ~std::uint64_t(2), 23, 55, true};
73}
74
75// EV68 HW_RET reads Rb, ignores hints/qualifiers, clears bit 1 and preserves the lock.
76constexpr AlphaPALReturnDescriptor hardware_return(std::uint32_t instruction)
77{
78 return (instruction >> 26) == 0x1e
79 ? AlphaPALReturnDescriptor{true, ~std::uint64_t(2), 0, false, true,
80 AlphaPALReturnSource::RegisterB}
81 : AlphaPALReturnDescriptor{};
82}
83
84constexpr AlphaPALInstructionAccess hardware_instruction_access(std::uint32_t instruction)
85{
86 switch (instruction >> 26)
87 {
88 case 0x19: // HW_MFPR.
89 case 0x1b: // HW_LD.
90 case 0x1d: // HW_MTPR.
91 case 0x1e: // HW_RET.
92 case 0x1f: // HW_ST.
93 return AlphaPALInstructionAccess::PalOrKernelHwe;
94 default:
95 return AlphaPALInstructionAccess::Denied;
96 }
97}
