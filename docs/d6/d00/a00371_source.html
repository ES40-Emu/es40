112void cirrus_gd5401_vga_device::recompute_params()
113{
114 u8 xtal_select = (vga.miscellaneous_output & 0x0c) >> 2;
115 int xtal;
116
117 switch(xtal_select & 3)
118 {
119 case 0: xtal = XTAL(25'174'800).value(); break;
120 case 1: xtal = XTAL(28'636'363).value(); break;
121 case 2: xtal = XTAL(41'164'000).value(); break;
122 case 3: xtal = XTAL(35'795'000).value(); break;
123 }
124
125 recompute_params_clock(1, xtal);
126}

219u8 cirrus_gd5428_vga_device::ramdac_overlay_r(offs_t offset)
220{
221 if(!cirrus.m_ext_palette_enabled)
222 return vga_device::ramdac_data_r(0);
223
224 u8 res = 0xff;
225 if (vga.dac.read && !machine().side_effects_disabled())
226 {
227 switch (vga.dac.state++)
228 {
229 case 0:
230 res = cirrus.m_ext_palette[vga.dac.read_index & 0x0f].red;
231 break;
232 case 1:
233 res = cirrus.m_ext_palette[vga.dac.read_index & 0x0f].green;
234 break;
235 case 2:
236 res = cirrus.m_ext_palette[vga.dac.read_index & 0x0f].blue;
237 break;
238 }
239
240 if (vga.dac.state == 3)
241 {
242 vga.dac.state = 0;
243 vga.dac.read_index++;
244 }
245 }
246
247 return res;
248}

250void cirrus_gd5428_vga_device::ramdac_overlay_w(offs_t offset, u8 data)
251{
252 if(!cirrus.m_ext_palette_enabled)
253 {
254 vga_device::ramdac_data_w(0, data);
255 return;
256 }
257
258 if (!vga.dac.read)
259 {
260 switch (vga.dac.state++) {
261 case 0:
262 cirrus.m_ext_palette[vga.dac.write_index & 0x0f].red=data;
263 break;
264 case 1:
265 cirrus.m_ext_palette[vga.dac.write_index & 0x0f].green=data;
266 break;
267 case 2:
268 cirrus.m_ext_palette[vga.dac.write_index & 0x0f].blue=data;
269 break;
270 }
271 vga.dac.dirty=1;
272 if (vga.dac.state==3)
273 {
274 vga.dac.state=0;
275 vga.dac.write_index++;
276 }
277 }

280void cirrus_gd5428_vga_device::crtc_map(address_map &map)
281{
282 svga_device::crtc_map(map);
283
284 // VGA Vertical Blank end
285 // some SVGA chipsets use all 8 bits, and this is one of them (according to MFGTST CRTC tests)
286 map(0x16, 0x16).lrw8(
287 NAME([this] (offs_t offset) {
288 return vga.crtc.vert_blank_end & 0x00ff;
289 }),
290 NAME([this] (offs_t offset, u8 data) {
291 vga.crtc.vert_blank_end &= ~0x00ff;
292 vga.crtc.vert_blank_end |= data;
293 recompute_params();
294 })
295 );
296 map(0x19, 0x19).lrw8(
297 NAME([this] (offs_t offset) {
298 return cirrus.m_cr19;
299 }),
300 NAME([this] (offs_t offset, u8 data) {
301 LOGMASKED(LOG_REGS, "CR19: Interlace End %02x\n", data);
302 cirrus.m_cr19 = data;
303 })
304 );
305 map(0x1a, 0x1a).lrw8(
306 NAME([this] (offs_t offset) {
307 return cirrus.m_cr1a;
308 }),
309 NAME([this] (offs_t offset, u8 data) {
310 LOGMASKED(LOG_REGS, "CR1A: Interlace Control %02x\n", data);
311 cirrus.m_cr1a = data;
312 vga.crtc.horz_blank_end = (vga.crtc.horz_blank_end & 0xff3f) | ((data & 0x30) << 2);
313 vga.crtc.vert_blank_end = (vga.crtc.vert_blank_end & 0xfcff) | ((data & 0xc0) << 2);
314 recompute_params();
315 })
316 );
317 map(0x1b, 0x1b).lrw8(
318 NAME([this] (offs_t offset) {
319 return cirrus.m_cr1b;
320 }),
321 NAME([this] (offs_t offset, u8 data) {
322 LOGMASKED(LOG_REGS, "CR1B: Extended Display Controls %02x\n", data);
323 cirrus.m_cr1b = data;
324 vga.crtc.start_addr_latch &= ~0x070000;
325 vga.crtc.start_addr_latch |= ((data & 0x01) << 16);
326 vga.crtc.start_addr_latch |= ((data & 0x0c) << 15);
327 vga.crtc.offset = (vga.crtc.offset & 0x00ff) | ((data & 0x10) << 4);
328 recompute_params();
329 })
330 );
331// map(0x25, 0x25) PSR Part Status (r/o, "factory testing and internal tracking only")
332 map(0x27, 0x27).lr8(
333 NAME([this] (offs_t offset) {
334 // NOTE: verbose
335 //LOGMASKED(LOG_REGS, "CR27: Read ID\n");
336 return cirrus.m_chip_id;
337 })
338 );
339}

341void cirrus_gd5428_vga_device::gc_map(address_map &map)
342{
343 svga_device::gc_map(map);
344
345 map(0x00, 0x00).lrw8(
346 NAME([this](offs_t offset) {
347 return vga.gc.set_reset & ((cirrus.gc_mode_ext & 0x04) ? 0xff : 0x0f);
348 }),
349 NAME([this](offs_t offset, u8 data) {
350 // if extended writes are enabled (bit 2 of index 0bh), then index 0 and 1 are extended to 8 bits,
351 // however XFree86 does not appear to do this...
352 vga.gc.set_reset = data & 0xff;
353 })
354 );
355 map(0x01, 0x01).lrw8(
356 NAME([this](offs_t offset) {
357 return vga.gc.enable_set_reset & ((cirrus.gc_mode_ext & 0x04) ? 0xff : 0x0f);
358 }),
359 NAME([this](offs_t offset, u8 data) {
360 vga.gc.enable_set_reset = data & 0xff;
361 })
362 );
363 map(0x05, 0x05).lrw8(
364 NAME([this](offs_t offset) {
365 u8 res = (vga.gc.shift256 & 1) << 6;
366 res |= (vga.gc.shift_reg & 1) << 5;
367 res |= (vga.gc.host_oe & 1) << 4;
368 res |= (vga.gc.read_mode & 1) << 3;
369 if(cirrus.gc_mode_ext & 0x04)
370 res |= (vga.gc.write_mode & 7);
371 else
372 res |= (vga.gc.write_mode & 3);
373 return res;
374 }),
375 NAME([this](offs_t offset, u8 data) {
376 vga.gc.shift256 = BIT(data, 6);
377 vga.gc.shift_reg = BIT(data, 5);
378 vga.gc.host_oe = BIT(data, 4);
379 vga.gc.read_mode = BIT(data, 3);
380 if(cirrus.gc_mode_ext & 0x04)
381 vga.gc.write_mode = data & 7;
382 else
383 vga.gc.write_mode = data & 3;
384 })
385 );
386 // Offset register 0/1
387 map(0x09, 0x0a).lrw8(
388 NAME([this](offs_t offset) {
389 return cirrus.gc_bank[offset];
390 }),
391 NAME([this](offs_t offset, u8 data) {
392 cirrus.gc_bank[offset] = data;
393 LOGMASKED(LOG_BANK, "GR%d: Offset register %d set to %02x\n", offset + 9, offset, data);
394 })
395 );
396 // Graphics controller mode extensions
397 map(0x0b, 0x0b).lrw8(
398 NAME([this](offs_t offset) {
399 return cirrus.gc_mode_ext;
400 }),
401 NAME([this](offs_t offset, u8 data) {
402 LOGMASKED(LOG_REGS, "GRB: Graphics Controller Mode Extensions %02x\n", data);
403 cirrus.gc_mode_ext = data;
404 if(!(data & 0x04))
405 {
406 vga.gc.set_reset &= 0x0f;
407 vga.gc.enable_set_reset &= 0x0f;
408 }
409 if(!(data & 0x08))
410 vga.sequencer.map_mask &= 0x0f;
411 })
412 );

414 // map(0x0c, 0x0c)
415 // Colour Key Mask
416 // map(0x0d, 0x0d)
417 // Miscellaneous Control
418 // map(0x0e, 0x0e)

838 // draw hardware graphics cursor

904void cirrus_gd5428_vga_device::recompute_params()
905{
906 uint8_t divisor = 1;
907 float clock;
908 // TODO: coming from OSC, expose as this->clock()
909 const XTAL xtal = XTAL(14'318'181);
910 uint8_t clocksel = (vga.miscellaneous_output & 0xc) >> 2;
911
912 if(cirrus.gc_locked || cirrus.m_vclk_num[clocksel] == 0 || cirrus.m_vclk_denom[clocksel] == 0)
913 clock = ((vga.miscellaneous_output & 0xc) ? xtal*2: xtal*1.75).dvalue();
914 else
915 {
916 int numerator = cirrus.m_vclk_num[clocksel] & 0x7f;
917 int denominator = (cirrus.m_vclk_denom[clocksel] & 0x3e) >> 1;
918 int mul = (cirrus.m_vclk_denom[clocksel] & 0x01) ? 2 : 1;
919 clock = (xtal * numerator / denominator / mul).dvalue();
920 LOGMASKED(LOG_PLL, "CL: PLL setting %d num %d denom %d mul %d -> %f\n", clocksel, numerator, denominator, mul, clock);
921 }
922
923 svga.rgb8_en = svga.rgb15_en = svga.rgb16_en = svga.rgb24_en = svga.rgb32_en = 0;
924
925 if (!cirrus.gc_locked)
926 {
927 // gambl186 relies on this, don't setup any hidden DAC but only this
928 if (vga.sequencer.data[0x07] & 0x01)
929 svga.rgb8_en = 1;
930
931 if (BIT(cirrus.m_hidden_dac_mode, 7))
932 {
933 // TODO: needs subclassing, earlier chips don't have all of these modes
934 if (BIT(cirrus.m_hidden_dac_mode, 4))
935 popmessage("pc_vga_cirrus: Unsupported mixed 5-5-5 / 8bpp mode selected");
936 switch(cirrus.m_hidden_dac_mode & 0x4f)
937 {
938 case 0x00:
939 case 0x40:
940 // 5-5-5 Sierra
941 svga.rgb15_en = 1;
942 break;
943 case 0x41:
944 svga.rgb16_en = 1;
945 break;
946 case 0x43: // CCIR601 YUV422 16-bit
947 case 0x44: // YUV411 8-bit
948 case 0x4a: // 16bpp + YUV422 overlay
949 case 0x4b: // 16bpp + YUV411 overlay
950 popmessage("pc_vga_cirrus: CL-GD545 YUV mode selected %02x", cirrus.m_hidden_dac_mode);
951 break;
952 case 0x45:
953 svga.rgb24_en = 1;
954 break;
955 case 0x46:
956 case 0x47:
957 popmessage("pc_vga_cirrus: CL-GD545+ DAC power down selected %02x", cirrus.m_hidden_dac_mode);
958 break;
959 case 0x48:
960 popmessage("pc_vga_cirrus: CL-GD545+ 8-bit grayscale selected");
961 break;
962 case 0x49:
963 svga.rgb8_en = 1;
964 break;
965 default:
966 // TODO: 0xff in pciagp (alias for a DAC power down?)
967 if (cirrus.m_hidden_dac_mode != 0xff)
968 popmessage("pc_vga_cirrus: reserved mode selected %02x", cirrus.m_hidden_dac_mode);
969 break;
970 }
971 }
972
973 switch(vga.sequencer.data[0x07] & 0x06) // bit 3 is reserved on GD542x
974 {
975 case 0x00: break;
976 case 0x02: clock /= 2; break; // Clock / 2 for 16-bit data
977 case 0x04: clock /= 3; break; // Clock / 3 for 24-bit data
978 case 0x06:
979 // Clock rate for 16-bit data = VCLK
980 // TODO: verify clock, may just be clock / 2 again?
981 //divisor = 2;
982 break;
983 }
984 }
985 recompute_params_clock(divisor, (int)clock);
986}

1010void cirrus_gd5428_vga_device::start_bitblt()
1011{
1012 uint32_t x,y;
1013
1014 // Bit 4: 16-bit Color / Transparency
1015 // Bit 6: 8x8 Pattern Copy
1016 // Bit 7: Color Expand
1017 const unsigned bytes_per_pixel = BIT(cirrus.m_blt_mode, 4) ? 2 : 1;
1018 const unsigned pattern_pitch = BIT(cirrus.m_blt_mode, 7) ? 1 : (8 * bytes_per_pixel);
1019 const u32 pattern_base = cirrus.m_blt_source & ~7U;
1020
1021 if(cirrus.m_blt_mode & 0x01)
1022 {
1023 start_reverse_bitblt();
1024 return;
1025 }
1026
1027 LOGMASKED(LOG_BLIT, "CL: BitBLT started: Src: %06x Dst: %06x Width: %i Height %i ROP: %02x Mode: %02x\n",cirrus.m_blt_source,cirrus.m_blt_dest,cirrus.m_blt_width,cirrus.m_blt_height,cirrus.m_blt_rop,cirrus.m_blt_mode);
1028
1029 cirrus.m_blt_source_current = cirrus.m_blt_source;
1030 cirrus.m_blt_dest_current = cirrus.m_blt_dest;
1031
1032 for(y=0;y<=cirrus.m_blt_height;y++)
1033 {
1034 const u32 pattern_row_base = BIT(cirrus.m_blt_mode, 6) ? (pattern_base + (((cirrus.m_blt_source + y) & 7) * pattern_pitch)) : 0;
1035 for(x=0;x<=cirrus.m_blt_width;x++)
1036 {
1037 if (BIT(cirrus.m_blt_mode, 6))
1038 cirrus.m_blt_source_current = pattern_row_base + (BIT(cirrus.m_blt_mode, 7) ? 0 : (x % pattern_pitch));
1039 if(cirrus.m_blt_mode & 0x80) // colour expand
1040 {
1041 if(cirrus.m_blt_mode & 0x10) // 16-bit colour expansion / transparency width
1042 {
1043 // use GR0/1/10/11 background/foreground regs
1044 const bool foreground = BIT(vga.memory[cirrus.m_blt_source_current % vga.svga_intf.vram_size], ~(x/2) & 7);
1045 uint16_t pixel = foreground ? ((cirrus.m_gr11 << 8) | vga.gc.enable_set_reset) : ((cirrus.m_gr10 << 8) | vga.gc.set_reset);
1046
1047 if(cirrus.m_blt_dest_current & 1)
1048 copy_pixel(pixel >> 8, vga.memory[cirrus.m_blt_dest_current % vga.svga_intf.vram_size], foreground, x);
1049 else
1050 copy_pixel(pixel & 0xff, vga.memory[cirrus.m_blt_dest_current % vga.svga_intf.vram_size], foreground, x);
1051 if((x % 8) == 7 && !(cirrus.m_blt_mode & 0x40)) // don't increment if a pattern (it's only 8 bits)
1052 cirrus.m_blt_source_current++;
1053 }
1054 else
1055 {
1056 const bool foreground = BIT(vga.memory[cirrus.m_blt_source_current % vga.svga_intf.vram_size], ~x & 7);
1057 uint8_t pixel = foreground ? vga.gc.enable_set_reset : vga.gc.set_reset; // use GR0/1/10/11 background/foreground regs
1058

1195// colour-expanded BitBLTs from system memory are on a byte boundary, unused bits are ignored
1196void cirrus_gd5428_vga_device::blit_byte()
1197{
1198 // Width is stored as a byte count minus one. Each source bit expands
1199 // to one pixel; GR10/11 supply the second byte only in 16-bit mode.
1200 const unsigned bytes_per_pixel = BIT(cirrus.m_blt_mode, 4) ? 2 : 1;
1201 for (unsigned bit = 0; bit < 8; bit++)
1202 {
1203 const bool foreground = BIT(cirrus.m_blt_system_buffer, 7 - bit);
1204 for (unsigned byte = 0; byte < bytes_per_pixel; byte++)
1205 {
1206 const u8 pixel = byte ? (foreground ? cirrus.m_gr11 : cirrus.m_gr10) :
1207 (foreground ? vga.gc.enable_set_reset : vga.gc.set_reset);
1208 if (cirrus.m_blt_pixel_count <= cirrus.m_blt_width)
1209 copy_pixel(pixel, vga.memory[cirrus.m_blt_dest_current % vga.svga_intf.vram_size], foreground, cirrus.m_blt_pixel_count);
1210 cirrus.m_blt_dest_current++;
1211 cirrus.m_blt_pixel_count++;
1212 }
1213 }
1214 if(cirrus.m_blt_pixel_count > cirrus.m_blt_width)
1215 {
1216 cirrus.m_blt_pixel_count = 0;
1217 cirrus.m_blt_scan_count++;
1218 cirrus.m_blt_dest_current = cirrus.m_blt_dest + (cirrus.m_blt_dest_pitch*cirrus.m_blt_scan_count);
1219 }
1220 if(cirrus.m_blt_scan_count > cirrus.m_blt_height)
1221 {
1222 cirrus.m_blt_system_transfer = false; // BitBLT complete
1223 cirrus.m_blt_status &= ~0x0b;
1224 }

1227void cirrus_gd5428_vga_device::copy_pixel(uint8_t src, uint8_t dst, bool foreground, unsigned dst_byte_offset)
1228{
1229 // GR2F suppresses the first n pixels on each expanded or pattern scan line.
1230 if ((cirrus.m_blt_mode & 0xc0) && dst_byte_offset < cirrus.m_blt_write_mask * (BIT(cirrus.m_blt_mode, 4) ? 2 : 1))
1231 return;
1232
1233 uint8_t res = src;
1234
1235 switch(cirrus.m_blt_rop)
1236 {
1237 case 0x00: // BLACK
1238 res = 0x00;
1239 break;
1240 case 0x05: // SRCAND
1241 res = src & dst;
1242 break;
1243 case 0x06: // DST
1244 res = dst;
1245 break;
1246 case 0x09: // SDna
1247 res = src & ~dst;
1248 break;
1249 case 0x0b: // DSTINVERT
1250 res = ~dst;
1251 break;
1252 case 0x0d: // SRC
1253 res = src;
1254 break;
1255 case 0x0e: // WHITE
1256 res = 0xff;
1257 break;
1258 case 0x50: // DSna / DPna
1259 // used by zorro2:picasso2p, unknown purpose
1260 res = (dst & (~src));
1261 break;
1262 case 0x59: // SRCINVERT
1263 res = src ^ dst;
1264 break;
1265 case 0x6d: // SRCPAINT / DSo
1266 // zorro2:picasso2p on VGA Workbench (upper right icon)
1267 res = src | dst;
1268 break;
1269 case 0x90: // NOTSRCERASE
1270 res = ~(src | dst);
1271 break;
1272 case 0x95: // NOTSRCINVERT
1273 res = ~(src ^ dst);
1274 break;
1275 case 0xad: // SDno
1276 res = src | ~dst;
1277 break;
1278 case 0xd0: // NOTSRCCOPY
1279 res = ~src;
1280 break;
1281 case 0xd6: // MERGEPAINT
1282 res = ~src | dst;
1283 break;
1284 case 0xda: // NOTSRCAND
1285 res = ~(src & dst);
1286 break;
1287 default:
1288 popmessage("pc_vga_cirrus: Unsupported BitBLT ROP mode %02x",cirrus.m_blt_rop);
1289 }
1290
1291 // handle transparency compare
1292 if (cirrus.m_chip_id >= 0xa0 && BIT(cirrus.m_blt_mode, 7) && BIT(cirrus.m_blt_mode, 3))
1293 {
1294 if (!foreground)
1295 return;
1296 }
1297 else if(cirrus.m_blt_mode & 0x08) // TODO: 16-bit compare
1298 {
1299 // if ROP result matches the transparency colour, don't change the pixel
1300 if((res & (~cirrus.m_blt_trans_colour_mask & 0xff)) == ((cirrus.m_blt_trans_colour & 0xff) & (~cirrus.m_blt_trans_colour_mask & 0xff)))
1301 return;
1302 }
1303
1304 vga.memory[cirrus.m_blt_dest_current % vga.svga_intf.vram_size] = res;
1305}

1342uint8_t cirrus_gd5428_vga_device::mem_r(offs_t offset)
1343{
1344 uint32_t addr;
1345 uint8_t cur_mode = pc_vga_choosevideomode();
1346
1347 const uint8_t bank = offset_select(offset);
1348
1349 // TODO: incomplete, just enough for zorro2:picasso2p to not outright crash at display init
1350 // value should be a base to apply, also GRB[5] remaps banking granularity
1351 // NOTE: bebox also wants this
1352 if(vga.sequencer.data[0x07] & 0xf0)
1353 {
1354 return svga_device::mem_linear_r((offset & 0xffff) + bank * 0x10000);
1355 }
1356
1357 if(cirrus.gc_locked || offset >= 0x10000 || cur_mode == TEXT_MODE || cur_mode == SCREEN_OFF)
1358 {
1359 return vga_device::mem_r(offset & 0x1ffff);
1360 }
1361
1362 if(cirrus.gc_mode_ext & 0x20) // 16kB bank granularity
1363 addr = bank * 0x4000;
1364 else // 4kB bank granularity
1365 addr = bank * 0x1000;
1366
1367 // Is the display address adjusted automatically when not using Chain-4 addressing?
1368 // The GD542x BIOS doesn't do it, but Virtual Pool expects it.
1369 if(!(vga.sequencer.data[4] & 0x8))
1370 addr <<= 2;
1371
1372 if(svga.rgb8_en || svga.rgb15_en || svga.rgb16_en || svga.rgb24_en)
1373 {
1374 uint8_t data = 0;
1375 if(cirrus.gc_mode_ext & 0x01)
1376 {
1377 if(offset & 0x10000)
1378 return 0;
1379 if(offset < 0x8000)
1380 offset &= 0x7fff;
1381 else
1382 {
1383 offset -= 0x8000;
1384 offset &= 0x7fff;
1385 }
1386 }
1387 else
1388 offset &= 0xffff;
1389
1390 if(vga.sequencer.data[4] & 0x8)
1391 data = vga.memory[(offset+addr) % vga.svga_intf.vram_size];
1392 else
1393 {
1394 // Unchained packed pixels still use the VGA data path. Each
1395 // plane/latch holds one of four adjacent eight-bit pixels.
1396 if (!machine().side_effects_disabled())
1397 {
1398 for (unsigned plane = 0; plane < 4; plane++)
1399 vga.gc.latch[plane] = vga.memory[(offset * 4 + plane + addr) % vga.svga_intf.vram_size];
1400 }
1401 if (vga.gc.read_mode)
1402 {
1403 for (unsigned bit = 0; bit < 8; bit++)
1404 {
1405 u8 colour = 0;
1406 for (unsigned plane = 0; plane < 4; plane++)
1407 colour |= BIT(vga.gc.latch[plane], bit) << plane;
1408 if ((colour & vga.gc.color_dont_care) == (vga.gc.color_compare & vga.gc.color_dont_care))
1409 data |= 1 << bit;
1410 }
1411 }
1412 else
1413 data = vga.gc.latch[vga.gc.read_map_sel];
1414 }
1415 return data;
1416 }
1417
1418 switch(vga.gc.memory_map_sel & 0x03)
1419 {
1420 case 0: break;
1421 case 1: if(cirrus.gc_mode_ext & 0x01) offset &= 0x7fff; else offset &= 0x0ffff; break;
1422 case 2: offset -= 0x10000; offset &= 0x07fff; break;
1423 case 3: offset -= 0x18000; offset &= 0x07fff; break;
1424 }
1425
1426 if(vga.sequencer.data[4] & 4)
1427 {
1428 int data;
1429 if (!machine().side_effects_disabled())
1430 {
1431 vga.gc.latch[0]=vga.memory[(offset+addr) % vga.svga_intf.vram_size];
1432 vga.gc.latch[1]=vga.memory[((offset+addr)+0x10000) % vga.svga_intf.vram_size];
1433 vga.gc.latch[2]=vga.memory[((offset+addr)+0x20000) % vga.svga_intf.vram_size];
1434 vga.gc.latch[3]=vga.memory[((offset+addr)+0x30000) % vga.svga_intf.vram_size];
1435 }
1436
1437 if (vga.gc.read_mode)
1438 {
1439 uint8_t byte,layer;
1440 uint8_t fill_latch;
1441 data=0;
1442
1443 for(byte=0;byte<8;byte++)
1444 {
1445 fill_latch = 0;
1446 for(layer=0;layer<4;layer++)
1447 {
1448 if(vga.gc.latch[layer] & 1 << byte)
1449 fill_latch |= 1 << layer;
1450 }
1451 fill_latch &= vga.gc.color_dont_care;
1452 if(fill_latch == vga.gc.color_compare)
1453 data |= 1 << byte;
1454 }
1455 }
1456 else
1457 data=vga.gc.latch[vga.gc.read_map_sel];
1458
1459 return data;
1460 }
1461 else
1462 {
1463 // TODO: Lines up in 16-colour mode, likely different for 256-colour modes (docs say video addresses are shifted right 3 places)
1464 //uint8_t bits = ((gc_mode_ext & 0x08) && (vga.gc.write_mode == 1)) ? 8 : 4;
1465
1466 uint8_t data = 0;
1467 //printf("%08x\n",offset);
1468
1469 if(cirrus.gc_mode_ext & 0x02)
1470 {
1471 for(uint8_t i = 0; i < 8; i++)
1472 {
1473 if(BIT(vga.sequencer.map_mask, i))
1474 data |= vga.memory[(((offset+addr))+i*0x10000) % vga.svga_intf.vram_size];
1475 }
1476 }
1477 else
1478 {
1479 for(uint8_t i = 0; i < 4; i++)
1480 {
1481 if(BIT(vga.sequencer.map_mask, i))
1482 data |= vga.memory[(((offset+addr))+i*0x10000) % vga.svga_intf.vram_size];
1483 }
1484 }
1485
1486 return data;
1487 }

1490void cirrus_gd5428_vga_device::mem_w(offs_t offset, uint8_t data)
1491{
1492 uint32_t addr;
1493 uint8_t cur_mode = pc_vga_choosevideomode();
1494
1495 if(cirrus.m_blt_system_transfer)
1496 {
1497 if(cirrus.m_blt_mode & 0x80) // colour expand
1498 {
1499 cirrus.m_blt_system_buffer &= ~(0x000000ff);
1500 cirrus.m_blt_system_buffer |= data;
1501 blit_byte();
1502 cirrus.m_blt_system_count = 0;
1503 }
1504 else
1505 {
1506 cirrus.m_blt_system_buffer &= ~(0x000000ff << (cirrus.m_blt_system_count * 8));
1507 cirrus.m_blt_system_buffer |= (data << (cirrus.m_blt_system_count * 8));
1508 cirrus.m_blt_system_count++;
1509 if(cirrus.m_blt_system_count >= 4)
1510 {
1511 blit_dword();
1512 cirrus.m_blt_system_count = 0;
1513 }
1514 }
1515 return;
1516 }
1517
1518 const uint8_t bank = offset_select(offset);
1519
1520 // TODO: as above
1521 if(vga.sequencer.data[0x07] & 0xf0)
1522 {
1523 svga_device::mem_linear_w((offset + bank * 0x10000), data);
1524 return;
1525 }
1526
1527 if(cirrus.gc_locked || offset >= 0x10000 || cur_mode == TEXT_MODE || cur_mode == SCREEN_OFF)
1528 {
1529 vga_device::mem_w(offset & 0x1ffff,data);
1530 return;
1531 }
1532
1533 if(cirrus.gc_mode_ext & 0x20) // 16kB bank granularity
1534 addr = bank * 0x4000;
1535 else // 4kB bank granularity
1536 addr = bank * 0x1000;
1537
1538 // Is the display address adjusted automatically when using Chain-4 addressing? The GD542x BIOS doesn't do it, but Virtual Pool expects it.
1539 if(!(vga.sequencer.data[4] & 0x8))
1540 addr <<= 2;
1541
1542 if(svga.rgb8_en || svga.rgb15_en || svga.rgb16_en || svga.rgb24_en)
1543 {
1544 if(offset & 0x10000)
1545 return;
1546 if(cirrus.gc_mode_ext & 0x01)
1547 {
1548 if(offset < 0x8000)
1549 offset &= 0x7fff;
1550 else
1551 {
1552 offset -= 0x8000;
1553 offset &= 0x7fff;
1554 }
1555 }
1556 else
1557 offset &= 0xffff;
1558
1559 // GR0 (and GR10 for 16-bit) = background colour in write mode 5
1560 // GR1 (and GR11 for 16-bit) = foreground colour in write modes 4 or 5
1561 // GRB = colour expansion width
1562 // SR2 = pixel mask for the eight expanded pixels, MSB first
1563 if (vga.gc.write_mode == 4)
1564 {
1565 const unsigned bytes_per_pixel = BIT(cirrus.gc_mode_ext, 4) ? 2 : 1;
1566 const u32 destination = (addr + offset) * 8 * bytes_per_pixel;
1567 for (unsigned pixel = 0; pixel < 8; pixel++)
1568 {
1569 if (BIT(vga.sequencer.map_mask, 7 - pixel) && BIT(data, 7 - pixel))
1570 {
1571 const u32 pixel_address = destination + pixel * bytes_per_pixel;
1572 vga.memory[pixel_address % vga.svga_intf.vram_size] = vga.gc.enable_set_reset;
1573 if (bytes_per_pixel == 2)
1574 vga.memory[(pixel_address + 1) % vga.svga_intf.vram_size] = cirrus.m_gr11;
1575 }
1576 }
1577 return;
1578 }
1579
1580 if (vga.gc.write_mode == 5)
1581 {
1582 const unsigned bytes_per_pixel = BIT(cirrus.gc_mode_ext, 4) ? 2 : 1;
1583 const u32 destination = (addr + offset) * 8 * bytes_per_pixel;
1584 for (unsigned pixel = 0; pixel < 8; pixel++)
1585 {
1586 if (BIT(vga.sequencer.map_mask, 7 - pixel))
1587 {
1588 const bool foreground = BIT(data, 7 - pixel);
1589 const u32 pixel_address = destination + pixel * bytes_per_pixel;
1590 vga.memory[pixel_address % vga.svga_intf.vram_size] = foreground ? vga.gc.enable_set_reset : vga.gc.set_reset;
1591 if (bytes_per_pixel == 2)
1592 vga.memory[(pixel_address + 1) % vga.svga_intf.vram_size] = foreground ? cirrus.m_gr11 : cirrus.m_gr10;
1593 }
1594 }
1595 return;
1596 }
1597
1598 if(vga.sequencer.data[4] & 0x8)
1599 vga.memory[(offset+addr) % vga.svga_intf.vram_size] = data;
1600 else
1601 {
1602 for(int i = 0; i < 4; i++)
1603 {
1604 if(BIT(vga.sequencer.map_mask, i))
1605 vga.memory[((offset*4+i)+addr) % vga.svga_intf.vram_size] =
1606 BIT(vga.sequencer.data[4], 2) ? vga_latch_write(i, data) : data;
1607 }
1608 }
1609 }
1610 else
1611 {
1612 //Inside each case must prevent writes to non-mapped VGA memory regions, not only mask the offset.
1613 switch(vga.gc.memory_map_sel & 0x03)
1614 {
1615 case 0: break;
1616 case 1:
1617 if(offset & 0x10000)
1618 return;
1619
1620 if(cirrus.gc_mode_ext & 0x01)
1621 offset &= 0x7fff;
1622 else
1623 offset &= 0xffff;
1624 break;
1625 case 2:
1626 if((offset & 0x18000) != 0x10000)
1627 return;
1628
1629 offset &= 0x07fff;
1630 break;
1631 case 3:
1632 if((offset & 0x18000) != 0x18000)
1633 return;
1634
1635 offset &= 0x07fff;
1636 break;
1637 }
1638
1639 {
1640 // TODO: Lines up in 16-colour mode, likely different for 256-colour modes (docs say video addresses are shifted right 3 places)
1641 //uint8_t bits = ((gc_mode_ext & 0x08) && (vga.gc.write_mode == 1)) ? 8 : 4;
1642
1643 for(uint8_t i = 0; i < 4; i++)
1644 {
1645 if(BIT(vga.sequencer.map_mask, i))
1646 {
1647 if(cirrus.gc_mode_ext & 0x02)
1648 {
1649 vga.memory[(((offset+addr) << 1)+i*0x10000) % vga.svga_intf.vram_size] = (vga.sequencer.data[4] & 4) ? vga_latch_write(i,data) : data;
1650 vga.memory[(((offset+addr) << 1)+i*0x10000+1) % vga.svga_intf.vram_size] = (vga.sequencer.data[4] & 4) ? vga_latch_write(i,data) : data;
1651 }
1652 else
1653 vga.memory[(((offset+addr))+i*0x10000) % vga.svga_intf.vram_size] = (vga.sequencer.data[4] & 4) ? vga_latch_write(i,data) : data;
1654 }
1655 }
1656 return;
1657 }

1690/*
1691 * GD5430 BitBLT registers:
1692 * GR0: Background colour byte 0 (00h).
1693 * GR1: Foreground colour byte 0 (04h).
1694 * GR10: Background colour byte 1 (01h).
1695 * GR11: Foreground colour byte 1 (05h).
1696 * GR20/21: BLT width bytes 0/1 (08h/09h).
1697 * GR22/23: BLT height bytes 0/1 (0Ah/0Bh).
1698 * GR24/25: BLT destination pitch bytes 0/1 (0Ch/0Dh).
1699 * GR26/27: BLT source pitch bytes 0/1 (0Eh/0Fh).
1700 * GR28-2A: BLT destination start bytes 0-2 (10h-12h).
1701 * GR2C-2E: BLT source start bytes 0-2 (14h-16h).
1702 * GR2F: BLT destination left-side clipping (17h).
1703 * GR30: BLT mode (18h).
1704 * GR31: BLT start/status (40h), the only register readable through MMIO.
1705 * GR32: BLT raster operation (1Ah).
1706 *
1707 * Notes:
1708 * - This is the GD5430/'40 subset from the Alpine Family TRM. Colour
1709 * bytes are grouped by background/foreground rather than GR index.
1710 *
1711 * TODO:
1712 * - GD5430 reserves 02h/03h/06h/07h (GR12-GR15) and 1Bh-3Fh. GD5446
1713 * also defines those colour bytes, GR33 mode extensions and GR34/35
1714 * colour keys; these are not implemented by inheriting this decoder.
1715 * - Give later chips their own MMIO address map and replace the GR
1716 * lambdas with named handlers shared by indexed and MMIO maps, allowing
1717 * each chip to define its register set and MMIO access permissions.
1718 */
1719u8 cirrus_gd5430_vga_device::mmio_register(offs_t offset)
1720{
1721 if ((offset >= 0x08 && offset <= 0x12) ||
1722 (offset >= 0x14 && offset <= 0x18) || offset == 0x1a)
1723 return offset + 0x18;
1724
1725 switch (offset)
1726 {
1727 case 0x00: return 0x00;
1728 case 0x01: return 0x10;
1729 case 0x04: return 0x01;
1730 case 0x05: return 0x11;
1731 case 0x40: return 0x31;
1732 default: return 0xff;
1733 }
1734}

1739 // the remaining MMIO registers are write-only.

1922{
1923 if (index >= PCI_RANGE_BASE)
1924 return is_pci() && CPCIDevice::decodes_memory_access(index, address, dsize, write);
1925 if (index < 0 || index >= MAX_DEV_RANGES)
1926 return false;
1927 if (is_pci())
1928 {
1929 const u32 command = endian_32(pci_state.config_data[0][1]);
1930 if (!(command & (dev_range_is_io[index] ? 1u : 2u)))
1931 return false;
1932 }
1933 if (index == 5)
1934 return !is_pci() && address < sizeof(option_rom);
1935 if (index == 4)
1936 {
1937 if (address >= 0x20000)
1938 return false;

1940 (!BIT(cirrus.m_sr17, 6) || !(vga.sequencer.data[7] & 0xf0)) && address >= 0x18000)

1950 switch (index)
1951 {
1952 case 1: case 3: return !BIT(vga.miscellaneous_output, 0);

2282void CCirrus::advance_scan(std::chrono::steady_clock::time_point now)
2283{
2284 // A nested bus access can sample a later instant before the outer access
2285 // resumes. Never rewind a running counter's anchor and count time twice.

2290 // Mode programming does not restart the oscillator. Retain the counters
2291 // and reduce out-of-range positions when software changes the totals.
2292 m_scan.character = std::fmod(m_scan.character, double(t.horizontal_total));
2293 m_scan.line %= field_lines;

2297 t.characters_per_second > 0.0) {

2301 // Reduce elapsed time before combining it with the saved position; a
2302 // long host pause must not overflow an integer scan counter.
2303 double position = m_scan.character + double(m_scan.line) * t.horizontal_total;
2304 if (t.interlace)
2305 position += m_scan.field * field_characters;
2306 position = std::fmod(position + std::fmod(seconds * t.characters_per_second,
2307 cycle_characters), cycle_characters);
2308 m_scan.field = t.interlace && position >= field_characters;
2309 position = std::fmod(position, field_characters);
2310 m_scan.line = u32(position / t.horizontal_total);
2311 m_scan.character = std::fmod(position, double(t.horizontal_total));
2312 }
