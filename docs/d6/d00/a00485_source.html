52static filesystem::path library_absolute(const filesystem::path& path)
53{
54 if (path.empty()) throw runtime_error("The machine configuration path is empty.");
55 const wstring text = path.wstring();
56 if (any_of(text.begin(), text.end(), [](wchar_t c) { return c < 32; }))
57 throw runtime_error("The machine configuration path contains a control character.");
58 return filesystem::absolute(path).lexically_normal();
59}

61static wstring library_identity(const filesystem::path& path)
62{
63 const filesystem::path absolute = library_absolute(path);
64 error_code error;
65 const filesystem::path canonical = filesystem::weakly_canonical(absolute, error);
66 return (error ? absolute : canonical).wstring();
67}

69bool win32_library_same_configuration(const filesystem::path& first, const filesystem::path& second)
70{
71 const wstring a = library_identity(first), b = library_identity(second);
72 if (CompareStringOrdinal(a.c_str(), -1, b.c_str(), -1, TRUE) == CSTR_EQUAL) return true;
73 error_code error;
74 return filesystem::equivalent(first, second, error) && !error;
75}

88vector<filesystem::path> win32_library_load(const filesystem::path& store)
89{
90 const filesystem::path path = library_absolute(store) / L"machines.txt";
91 win32_library_handle file(CreateFileW(path.c_str(), GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_DELETE,
92 nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr));
93 if (file.value == INVALID_HANDLE_VALUE)
94 {
95 const DWORD error = GetLastError();
96 if (error == ERROR_FILE_NOT_FOUND || error == ERROR_PATH_NOT_FOUND) return {};
97 library_error("Unable to read the machine library", error);
98 }
99 string bytes;
100 char buffer[8192];
101 for (;;)
102 {
103 DWORD count = 0;
104 if (!ReadFile(file.value, buffer, sizeof(buffer), &count, nullptr))
105 library_error("Unable to read the machine library");
106 if (!count) break;
107 if (bytes.size() > static_cast<size_t>(INT_MAX) - count)
108 throw runtime_error("The machine library is too large.");
109 bytes.append(buffer, count);
110 }
111 if (bytes.empty()) return {};
112 const int count = MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS,
113 bytes.data(), (int)bytes.size(), nullptr, 0);
114 if (!count) throw runtime_error("The machine library is not valid UTF-8.");
115 wstring text(count, L'\0');
116 if (!MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS,
117 bytes.data(), (int)bytes.size(), text.data(), count))
118 library_error("Unable to decode the machine library");
119 vector<filesystem::path> entries;
120 size_t begin = text.front() == 0xfeff ? 1 : 0;
121 while (begin < text.size())
122 {
123 size_t end = text.find(L'\n', begin);
124 if (end == wstring::npos) end = text.size();
125 size_t length = end - begin;
126 if (length && text[begin + length - 1] == L'\r') --length;
127 if (length)
128 {
129 const filesystem::path entry(text.substr(begin, length));
130 if (!entry.is_absolute()) throw runtime_error("The machine library contains a relative configuration path.");
131 const filesystem::path absolute = library_absolute(entry);
132 if (none_of(entries.begin(), entries.end(), [&](const auto& existing)
133 { return win32_library_same_configuration(existing, absolute); }))
134 entries.push_back(absolute);
135 }
136 begin = end == text.size() ? end : end + 1;
137 }
138 return entries;
139}

141void win32_library_update(const filesystem::path& store, const filesystem::path& configuration, bool add)
142{
143 const filesystem::path directory = library_absolute(store);
144 const filesystem::path target = library_absolute(configuration);
145 filesystem::create_directories(directory);
146 const wstring name = library_mutex_name(directory);
147 win32_library_handle mutex(CreateMutexW(nullptr, FALSE, name.c_str()));
148 if (!mutex.value) library_error("Unable to lock the machine library");
149 const DWORD waited = WaitForSingleObject(mutex.value, 5000);
150 if (waited == WAIT_TIMEOUT) throw runtime_error("Another manager is updating the machine library. Try again shortly.");
151 if (waited != WAIT_OBJECT_0 && waited != WAIT_ABANDONED)
152 library_error("Unable to lock the machine library");
153 struct mutex_guard
154 {
155 HANDLE handle;
156 ~mutex_guard() { ReleaseMutex(handle); }
157 } lock{mutex.value};
158
159 vector<filesystem::path> entries = win32_library_load(directory);
160 const auto matches = [&](const auto& entry) { return win32_library_same_configuration(entry, target); };
161 if (!add) entries.erase(remove_if(entries.begin(), entries.end(), matches), entries.end());
162 else if (none_of(entries.begin(), entries.end(), matches)) entries.push_back(target);
163 wstring text;
164 for (const auto& entry : entries) { text += entry.wstring(); text += L'\n'; }
165 if (text.size() > INT_MAX) throw runtime_error("The machine library is too large.");
166 string bytes;
167 if (!text.empty())
168 {
169 const int count = WideCharToMultiByte(CP_UTF8, WC_ERR_INVALID_CHARS,
170 text.data(), (int)text.size(), nullptr, 0, nullptr, nullptr);
171 if (!count) throw runtime_error("The machine library contains an invalid Unicode path.");
172 bytes.resize(count);
173 if (!WideCharToMultiByte(CP_UTF8, WC_ERR_INVALID_CHARS,
174 text.data(), (int)text.size(), bytes.data(), count, nullptr, nullptr))
175 library_error("Unable to encode the machine library");
176 }
