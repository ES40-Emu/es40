185 template <typename ItemType, typename StructType, typename ElementType>
186 void save_item(ItemType& value, ElementType StructType::*element,
187 const char* valname, int index = 0)
188 {
189 using object_type = std::remove_all_extents_t<ItemType>;
190 static_assert(std::is_base_of_v<StructType, object_type>,
191 "Saved member must belong to the registered object type");
192 register_state<ElementType>(valname, index, [&value, element]() -> void* {
193 auto* objects = reinterpret_cast<object_type*>(std::addressof(value));
194 return std::addressof(objects[0].*element);
195 }, sizeof(ItemType) / sizeof(object_type), sizeof(object_type));
196 }

198 template <typename ItemType>
199 void save_pointer(ItemType&& value, const char* valname, u32 count, int index = 0)
200 {
201 using storage_type = std::remove_reference_t<ItemType>;
202 static_assert(std::is_lvalue_reference_v<ItemType>,
203 "Saved storage must outlive its registration");
204 if constexpr (std::is_array_v<storage_type>)
205 {
206 if (count > std::extent_v<storage_type>)
207 throw std::invalid_argument("Saved array count exceeds its extent");
208 }
209 using element_type = std::remove_reference_t<decltype(value[0])>;
210 // Follow buffer reallocations after registration.
211 register_state<element_type>(valname, index, [&value]() -> void* {
212 if constexpr (std::is_array_v<storage_type> || std::is_pointer_v<storage_type>)
213 return value;
214 else
215 return value.get();
216 }, count, sizeof(element_type));
217 }

243 template <typename ItemType>
244 void register_state(const char* name, int index, std::function<void*()> address,
245 size_t blocks, size_t stride)
246 {
247 using atom_type = std::remove_all_extents_t<ItemType>;
248 static_assert(!std::is_const_v<atom_type> &&
249 (std::is_arithmetic_v<atom_type> || std::is_enum_v<atom_type>),
250 "Register scalar values, arrays, or individual structure members");
251 constexpr bool boolean = std::is_same_v<atom_type, bool>;
252 constexpr u32 element_size = boolean ? 1 : sizeof(atom_type);
253 constexpr size_t elements = sizeof(ItemType) / sizeof(atom_type);
254 if (blocks > (std::numeric_limits<u32>::max)() / elements ||
255 blocks > (std::numeric_limits<size_t>::max)() / elements / element_size)
256 throw std::invalid_argument("Saved array is too large");
257 const u32 type = boolean ? 1 : std::is_enum_v<atom_type> ? 5 :
258 std::is_floating_point_v<atom_type> ? 4 : std::is_signed_v<atom_type> ? 2 : 3;
259 const bool contiguous = !boolean && blocks && stride == sizeof(ItemType);
260 add_state_field({ name ? name : "", s32(index), type, element_size,
261 u32(blocks * elements), contiguous ? blocks * sizeof(ItemType) : sizeof(ItemType),
262 stride, contiguous ? 1u : u32(blocks), std::move(address) });
263 }
