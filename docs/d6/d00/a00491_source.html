64static wstring manager_text(const string& text)
65{
66 if (text.empty()) return {};
67 UINT page = CP_UTF8;
68 int count = MultiByteToWideChar(page, MB_ERR_INVALID_CHARS, text.data(), (int)text.size(), nullptr, 0);
69 if (!count) { page = CP_ACP; count = MultiByteToWideChar(page, 0, text.data(), (int)text.size(), nullptr, 0); }
70 wstring result(count, L'\0');
71 if (count) MultiByteToWideChar(page, 0, text.data(), (int)text.size(), result.data(), count);
72 return result;
73}

102static string manager_config_bytes(const wstring& text)
103{
104 if (text.empty()) return {};
105 if (text.size() > INT_MAX) throw runtime_error("The configuration is too large.");
106 const bool unicode = GetACP() == CP_UTF8;
107 const DWORD flags = unicode ? WC_ERR_INVALID_CHARS : WC_NO_BEST_FIT_CHARS;
108 BOOL substituted = FALSE;
109 BOOL* fallback = unicode ? nullptr : &substituted;
110 const int count = WideCharToMultiByte(CP_ACP, flags, text.data(), (int)text.size(), nullptr, 0, nullptr, fallback);
111 string bytes(count, '\0');
112 if (!count || !WideCharToMultiByte(CP_ACP, flags, text.data(), (int)text.size(),
113 bytes.data(), count, nullptr, fallback) || substituted)
114 throw runtime_error("Choose paths representable in the active Windows code page.");
115 return bytes;
116}

138static string manager_read_configuration(const filesystem::path& path)
139{
140 if (!filesystem::is_regular_file(path)) throw runtime_error("The source configuration file is missing.");
141 struct file_guard
142 {
143 HANDLE value;
144 ~file_guard() { if (value != INVALID_HANDLE_VALUE) CloseHandle(value); }
145 } file{CreateFileW(path.c_str(), GENERIC_READ, FILE_SHARE_READ, nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr)};
146 if (file.value == INVALID_HANDLE_VALUE)
147 throw system_error((int)GetLastError(), system_category(), "Unable to read the source configuration");
148 string bytes;
149 char buffer[8192];
150 for (;;)
151 {
152 DWORD count = 0;
153 if (!ReadFile(file.value, buffer, sizeof(buffer), &count, nullptr))
154 throw system_error((int)GetLastError(), system_category(), "Unable to read the source configuration");
155 if (!count) return bytes;
156 if (bytes.size() > (size_t)INT_MAX - count) throw runtime_error("The source configuration is too large.");
157 bytes.append(buffer, count);
158 }
159}

161static void manager_write_configuration(const filesystem::path& configuration, const string& bytes, bool check_companions)
162{
163 if (check_companions) manager_require_new_paths(configuration);
164 else manager_require_unused_path(configuration);
165 struct temporary_file
166 {
167 filesystem::path path;
168 HANDLE handle = INVALID_HANDLE_VALUE;
169 bool created = false;
170 ~temporary_file()
171 {
172 if (handle != INVALID_HANDLE_VALUE) CloseHandle(handle);
173 if (created) DeleteFileW(path.c_str());
174 }
175 } temporary;
176 const auto fail = [](const char* message, DWORD error)
177 { throw system_error((int)error, system_category(), message); };
178 const wstring stem = configuration.filename().wstring() + L".manager-" +
179 to_wstring(GetCurrentProcessId()) + L"-" + to_wstring(GetTickCount64());
180 for (unsigned attempt = 0; attempt < 100; ++attempt)
181 {
182 temporary.path = configuration.parent_path() / (stem + L"-" + to_wstring(attempt) + L".tmp");
183 temporary.handle = CreateFileW(temporary.path.c_str(), GENERIC_WRITE, 0,
184 nullptr, CREATE_NEW, FILE_ATTRIBUTE_NORMAL, nullptr);
185 if (temporary.handle != INVALID_HANDLE_VALUE) { temporary.created = true; break; }
186 const DWORD error = GetLastError();
187 if (error != ERROR_FILE_EXISTS && error != ERROR_ALREADY_EXISTS)
188 fail("Unable to create the machine configuration", error);
189 }
190 if (!temporary.created) throw runtime_error("Unable to choose a temporary configuration filename.");
191 size_t offset = 0;
192 while (offset < bytes.size())
193 {
194 DWORD written = 0;
195 if (!WriteFile(temporary.handle, bytes.data() + offset, (DWORD)(bytes.size() - offset), &written, nullptr))
196 fail("Unable to write the machine configuration", GetLastError());
197 if (!written) fail("Unable to write the machine configuration", ERROR_WRITE_FAULT);
198 offset += written;
199 }
200 if (!FlushFileBuffers(temporary.handle)) fail("Unable to flush the machine configuration", GetLastError());
201 if (!CloseHandle(temporary.handle)) fail("Unable to close the machine configuration", GetLastError());
202 temporary.handle = INVALID_HANDLE_VALUE;
203 // Recheck paths before publishing the configuration.
204 if (check_companions) manager_require_new_paths(configuration);
205 else manager_require_unused_path(configuration);
206 if (!MoveFileExW(temporary.path.c_str(), configuration.c_str(), MOVEFILE_WRITE_THROUGH))
207 fail("Unable to save the new machine configuration", GetLastError());
208 temporary.created = false;
209}

211static wstring manager_read_log(const filesystem::path& path)
212{
213 if (path.empty()) return {};
214 struct file_guard
215 {
216 HANDLE value;
217 ~file_guard() { if (value != INVALID_HANDLE_VALUE) CloseHandle(value); }
218 } file{CreateFileW(path.c_str(), GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
219 nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr)};
220 if (file.value == INVALID_HANDLE_VALUE) return {};
221 LARGE_INTEGER size = {}, offset = {};
222 if (!GetFileSizeEx(file.value, &size)) return {};
223 const DWORD count = (DWORD)min<LONGLONG>(size.QuadPart, 64 * 1024);
224 offset.QuadPart = size.QuadPart - count;
225 if (!SetFilePointerEx(file.value, offset, nullptr, FILE_BEGIN)) return {};
226 string bytes(count, '\0');
227 DWORD read = 0;
228 if (!ReadFile(file.value, bytes.data(), count, &read, nullptr)) return {};
229 bytes.resize(read);
230 const wstring decoded = manager_text(bytes);
231 wstring text = offset.QuadPart ? L"[Earlier output omitted]\r\n" : L"";
232 for (size_t i = 0; i < decoded.size(); ++i)
233 {
234 if (decoded[i] == L'\n' && (!i || decoded[i - 1] != L'\r')) text += L'\r';
235 text += decoded[i];
236 }
237 return text;
238}

240static wstring manager_configuration_preview(const filesystem::path& path)
241{
242 struct file_guard
243 {
244 HANDLE value;
245 ~file_guard() { if (value != INVALID_HANDLE_VALUE) CloseHandle(value); }
246 } file{CreateFileW(path.c_str(), GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
247 nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr)};
248 if (file.value == INVALID_HANDLE_VALUE) return L"Unable to read the configuration file.";
249 const DWORD limit = 128 * 1024;
250 string bytes(limit + 1, '\0');
251 DWORD read = 0;
252 if (!ReadFile(file.value, bytes.data(), (DWORD)bytes.size(), &read, nullptr))
253 return L"Unable to read the configuration file.";
254 const int length = (int)min(read, limit);
255 const int count = MultiByteToWideChar(CP_ACP, 0, bytes.data(), length, nullptr, 0);
256 wstring decoded(count, L'\0'), text;
257 if (count) MultiByteToWideChar(CP_ACP, 0, bytes.data(), length, decoded.data(), count);
258 for (size_t i = 0; i < decoded.size(); ++i)
259 {
260 if (decoded[i] == L'\n' && (!i || decoded[i - 1] != L'\r')) text += L'\r';
261 text += decoded[i];
262 }
263 if (read > limit) text += L"\r\n[Configuration preview truncated]";
264 return text;
265}

444 void layout()
445 {
446 if (!library || !empty || !status) return;
447 RECT bounds = {};
448 GetClientRect(window, &bounds);
449 const int width = bounds.right, height = bounds.bottom;
450 SendMessageW(status, WM_SIZE, 0, 0);
451 RECT footer = {};
452 GetWindowRect(status, &footer);
453 const int footer_height = footer.bottom - footer.top;
454 const int margin = px(16), sidebar = px(256);
455 MoveWindow(heading, margin, px(14), max(1, width - 2 * margin), px(30), TRUE);
456 MoveWindow(subtitle, margin, px(48), max(1, width - 2 * margin), px(24), TRUE);
457 int x = margin, y = px(82);
458 for (HWND button : buttons)
459 {
460 const int button_width = px(GetDlgCtrlID(button) == manager_fullscreen ? 108 : 86);
461 if (x > margin && x + button_width > width - margin) { x = margin; y += px(38); }
462 MoveWindow(button, x, y, button_width, px(30), TRUE);
463 x += button_width + px(8);
464 }
465 const int top = y + px(44), content_height = max(1, height - top - footer_height - margin);
466 MoveWindow(library, margin, top, sidebar, content_height, TRUE);
467 const int left = sidebar + 2 * margin, content_width = max(1, width - left - margin);
468 SetWindowPos(tabs, HWND_BOTTOM, left, top, content_width, content_height, SWP_NOACTIVATE);
469 RECT content = {};
470 GetClientRect(tabs, &content);
471 TabCtrl_AdjustRect(tabs, FALSE, &content);
472 MapWindowPoints(tabs, window, reinterpret_cast<POINT*>(&content), 2);
473 const int page_width = max(1L, content.right - content.left), page_height = max(1L, content.bottom - content.top);
474 MoveWindow(empty, content.left + px(24), content.top + px(60), max(1, page_width - px(48)),
475 max(1, page_height - px(80)), TRUE);
476 ShowWindow(empty, selected() ? SW_HIDE : SW_SHOWNA);
477 for (const auto& item : machines)
478 {
479 MoveWindow(item->host, content.left, content.top, page_width, page_height, TRUE);
480 ShowWindow(item->host, item.get() == selected() && page == 0 ? SW_SHOWNA : SW_HIDE);
481 fit_display(*item);
482 }
483 MoveWindow(details, content.left, content.top, page_width, page_height, TRUE);
484 MoveWindow(output, content.left, content.top, page_width, page_height, TRUE);
485 ShowWindow(details, selected() && page == 1 ? SW_SHOWNA : SW_HIDE);
486 ShowWindow(output, selected() && page == 2 ? SW_SHOWNA : SW_HIDE);
487 }

529 void refresh()
530 {
531 if (!window) return;
532 const auto* item = selected();
533 const HMENU menu = GetMenu(window);

664 {
665 if (closing) return;
666 const filesystem::path configuration = filesystem::absolute(path).lexically_normal();
667 if (save && !filesystem::is_regular_file(configuration)) throw runtime_error("The configuration file is missing.");
668 for (size_t i = 0; i < machines.size(); ++i)
669 {
670 if (!win32_library_same_configuration(machines[i]->configuration, configuration)) continue;
671 if (save) win32_library_update(store, configuration, true);
672 select((int)i);
673 return;
674 }
675 auto item = make_unique<win32_manager_machine>();
676 item->owner = this; item->configuration = configuration;
677 item->host = CreateWindowExW(0, win32_manager_host_class, L"", WS_CHILD | WS_TABSTOP |
678 WS_CLIPCHILDREN | WS_CLIPSIBLINGS, 0, 0, 1, 1, window, nullptr, GetModuleHandleW(nullptr), item.get());
679 if (!item->host) throw runtime_error("Unable to create the console host.");
680 machines.reserve(machines.size() + 1);
681 if (save) win32_library_update(store, configuration, true);
682 if (auto* old = selected()) old->session.send(win32_manager_release);
683 machines.push_back(move(item));
684 selection = (int)machines.size() - 1;
685 rebuild_library(); select(selection);
686 }

872 LRESULT event(HWND hwnd, UINT message, WPARAM wparam, LPARAM lparam)
873 {
874 switch (message)
875 {
876 case WM_SIZE:
877 if (wparam == SIZE_MINIMIZED) release_input();

886 case WM_SYSCOLORCHANGE: update_colors(); break;
887 case WM_CTLCOLORSTATIC:

894 {

1001 case WM_QUERYENDSESSION:

1004 return TRUE;
1005 case WM_DESTROY:

1028 {
1029 auto* item = reinterpret_cast<win32_manager_machine*>(GetWindowLongPtrW(hwnd, GWLP_USERDATA));
1030 if (message == WM_NCCREATE)
1031 {
1032 item = static_cast<win32_manager_machine*>(reinterpret_cast<CREATESTRUCTW*>(lparam)->lpCreateParams);
1033 item->host = hwnd;
1034 SetWindowLongPtrW(hwnd, GWLP_USERDATA, (LONG_PTR)item);
1035 }
1036 if (!item) return DefWindowProcW(hwnd, message, wparam, lparam);
1037 auto* self = item->owner;
1038 try
1039 {
1040 switch (message)
1041 {
1042 case win32_manager_display_ready:
1043 if (item->session.attach_display((HWND)wparam, (DWORD)lparam))
1044 { self->fit_display(*item); self->refresh(); }
1045 return 0;
1046 case win32_manager_display_size:
1047 if (item->session.running() && item->session.display())
1048 { item->width = (unsigned)wparam; item->height = (unsigned)lparam; self->refresh_status(); }
1049 return 0;
1050 case WM_SIZE: self->fit_display(*item); return 0;
1051 case WM_SETFOCUS:
1052 if (item->session.display() && GetParent(item->session.display()) == hwnd)
1053 SetFocus(item->session.display());
1054 return 0;
1055 case WM_ERASEBKGND: return 1;
1056 case WM_PAINT:
1057 {
1058 PAINTSTRUCT paint;
1059 HDC dc = BeginPaint(hwnd, &paint);
1060 RECT bounds = {}; GetClientRect(hwnd, &bounds);
