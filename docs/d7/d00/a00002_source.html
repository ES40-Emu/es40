248void segv_handler(int signum)
249{
250 int nptrs = backtrace(btbuffer, BTCOUNT);
251 char** strings;
252
253 printf("%%SYS-F-SEGFAULT: The Alpha Simulator has Segfaulted.\n");
254 printf("-SYS-F-SEGFAULT: Backtrace follows.\n");
255
256 printf("backtrace() returned %d addresses.\n", nptrs);
257 strings = backtrace_symbols(btbuffer, nptrs);
258 if (strings == NULL)
259 {
260 perror("backtrace_symbols");
261 exit(1);
262 }
263
264 for (int i = 0; i < nptrs; i++)
265 {
266 printf("%3d %s\n", nptrs - i, strings[i]);
267 }
268
269 free(strings);
270 if (signum == SIGSEGV)
271 _exit(1);
272}

308int main(int argc, char* argv[])
309{
310#if defined(_WIN32)
311 // Strip --run before normal argument handling.
312 const bool direct_run = argc > 1 && strcmp(argv[1], "--run") == 0;
313 if (direct_run)
314 {
315 for (int i = 1; i < argc; ++i) argv[i] = argv[i + 1];
316 --argc;
317 const char* host = getenv("ES40_MANAGER_HOST");
318 if (host && *host)
319 {
320 setvbuf(stdout, nullptr, _IONBF, 0);
321 setvbuf(stderr, nullptr, _IONBF, 0);
322 }
323 }
324#endif
325
326 if (argc == 2 &&
327 (strcmp(argv[1], "--version") == 0 || strcmp(argv[1], "-v") == 0))
328 {
329 int features = 0;
330 printf("ES40 %s (GitHub commit %s)\nFeatures:", VERSION, ES40_GIT_COMMIT);
331#ifdef ES40_JIT
332 printf(" AsmJit"); ++features;
333#endif
334#ifdef HAVE_SDL
335 printf(" SDL"); ++features;
336#endif
337#ifdef HAVE_X11
338 printf(" X11"); ++features;
339#endif
340#ifdef HAVE_PCAP
341 printf(" PCap"); ++features;
342#endif
343#ifdef HAVE_TAP_NET
344 printf(" TAP"); ++features;
345#endif
346#ifdef HAVE_VMNET
347 printf(" vmnet"); ++features;
348#endif
349#ifdef HAVE_SLIRP
350 printf(" slirp"); ++features;
351#endif
352 if (features == 0)
353 printf(" (none)");
354 printf("\n");
355 return 0;
356 }
357
358#if defined(_WIN32)
359 if (!direct_run && argc > 1 && strcmp(argv[1], "--manager") == 0)
360 {
361#if defined(IDB)
362 fprintf(stderr, "The machine manager requires a non-IDB emulator build.\n");
363 return 1;
364#else
365 return win32_manager_main();
366#endif
367 }
368#endif
369
370 print_es40_banner("AlphaServer ES40 emulator");
371
372 const char* filename = 0;
373 FILE* f;
374
375#ifdef HAS_BACKTRACE
376 signal(SIGSEGV, &segv_handler);
377 signal(SIGUSR1, &segv_handler);
378#endif
379 try
380 {
381#if defined(IDB) && (defined(LS_MASTER) || defined(LS_SLAVE))
382 lockstep_init();
383#endif
384#if defined(IDB)
385 if ((argc == 2 || argc == 3) && argv[1][0] != '@')
386#else
387 if (argc == 2)
388#endif
389 {
390 if (argv[1] && argv[1][0] != '\0')
391 filename = argv[1];
392 }
393 else
394 {
395 // Probe defaults; only assign on success
396 for (int i = 0; path[i]; ++i)
397 {
398 FILE* probe = fopen(path[i], "rb");
399 if (probe)
400 {
401 fclose(probe);
402 filename = path[i];
403 break;
404 }
405 }
406 if (!filename) {
407 FAILURE(FileNotFound, "configuration file");
408 }
409 }
410
411 // Open chosen file (print the path on failure)
412 f = fopen(filename, "rb");
413 if (!f) {
414 char buf[1024];
415 snprintf(buf, sizeof(buf), "failed to open configuration file: %s (%s)",
416 filename, strerror(errno));
417 FAILURE(Configuration, buf);
418 }
419
420 // Determine size (Windows: 64-bit length; POSIX: fseek/ftell)
421 size_t ll1 = 0;
422#if defined(_WIN32)
423 if (_fseeki64(f, 0, SEEK_END) != 0) {
424 char buf[1024];
425 fclose(f);
426 snprintf(buf, sizeof(buf), "failed to seek end of configuration file: %s", filename);
427 FAILURE(Configuration, buf);
428 }
429 __int64 pos64 = _ftelli64(f);
430 if (pos64 < 0) {
431 char buf[1024];
432 fclose(f);
433 snprintf(buf, sizeof(buf), "failed to tell size of configuration file: %s", filename);
434 FAILURE(Configuration, buf);
435 }
436 ll1 = (size_t)pos64;
437 if (_fseeki64(f, 0, SEEK_SET) != 0) {
438 char buf[1024];
439 fclose(f);
440 snprintf(buf, sizeof(buf), "failed to rewind configuration file: %s", filename);
441 FAILURE(Configuration, buf);
442 }
443#else
444 if (fseek(f, 0, SEEK_END) != 0) {
445 char buf[1024];
446 fclose(f);
447 snprintf(buf, sizeof(buf), "failed to seek end of configuration file: %s", filename);
448 FAILURE(Configuration, buf);
449 }
450 long pos = ftell(f);
451 if (pos < 0) {
452 char buf[1024];
453 fclose(f);
454 snprintf(buf, sizeof(buf), "failed to tell size of configuration file: %s", filename);
455 FAILURE(Configuration, buf);
456 }
457 ll1 = (size_t)pos;
458 if (fseek(f, 0, SEEK_SET) != 0) {
459 char buf[1024];
460 fclose(f);
461 snprintf(buf, sizeof(buf), "failed to rewind configuration file: %s", filename);
462 FAILURE(Configuration, buf);
463 }
464#endif
465
466 // Allocate buffer (+1 for NUL), read, and NUL-terminate for safety
467 char* ch1 = (char*)calloc(ll1 + 1, 1);
468 if (!ch1) {
469 char buf[1024];
470 fclose(f);
471 snprintf(buf, sizeof(buf), "out of memory reading configuration file: %s", filename);
472 FAILURE(Configuration, buf);
473 }
474
475 size_t read_bytes = fread(ch1, 1, ll1, f);
476 if (read_bytes != ll1) {
477 char buf[1024];
478 fclose(f);
479 free(ch1);
480 snprintf(buf, sizeof(buf), "failed to read entire configuration file: %s", filename);
481 FAILURE(Configuration, buf);
482 }
483 ch1[ll1] = '\0';
484
485#if defined(_WIN32) && !defined(IDB)
486 // Inspect startup options before creating devices.
487 CConfigurator* c = new CConfigurator(0, 0, 0, ch1, ll1, false);
488#else
489 CConfigurator* c = new CConfigurator(0, 0, 0, ch1, ll1);
490#endif
491 fclose(f);
492 free(ch1);
493
494#if defined(_WIN32) && !defined(IDB)
495 if (!direct_run && c->win32_workstation_enabled())
496 {
497 delete c;
498 return win32_manager_main(filename);
499 }
500 c->initialize();
501#endif
502
503 if (!theSystem)
504 FAILURE(Configuration, "no system initialized");
505
506#if defined(IDB)
507 trc = new CTraceEngine(theSystem);
508#endif
509 // Components are initialized and no workers have started.
510 // Restore the whole file before deciding to skip firmware loading.
511 // This way, we restore into a resumable state or not at all.
512 const char* restore_file = getenv("ES40_RESTORE");
513 bool restored = false;
514 if (restore_file && *restore_file)
515 {
516 printf("%%SYS-I-RESTORE: Restoring state from %s before startup.\n",
517 restore_file);
518 restored = theSystem->RestoreState(restore_file);
519 if (restored)
520 printf("%%SYS-I-RESTORED: State restored; firmware loading skipped.\n");
521 else
522 printf("%%SYS-W-NORESTORE: State was not loaded; cold booting.\n");
523 }
524 if (!restored)
525 {
526 theSystem->LoadROM();
527 theDPR->init();
528 }
529
530#if defined(PROFILE)
531 {
532 u64 p_i;
533 for (p_i = PROFILE_FROM; p_i < PROFILE_TO; p_i += (4 * PROFILE_BUCKSIZE))
534 PROFILE_BUCKET(p_i) = 0;
535 profiled_insts = 0;
536 }
537#endif
538 if (bx_gui && bx_gui->requires_main_thread())
539 bx_gui->main_thread_init();
540 }
541 catch (CException& e)
542 {
543 printf("Emulator Failure: %s\n", e.displayText().c_str());
544 if (theSystem)
545 {
546 theSystem->stop_threads();
547 if (detach_gui_guest_input())
548 delete theSystem;
549 }
550 return 1;
551 }
552
553 // A GUI that owns main() pumps its event loop there while the emulator,
554 // including its shutdown, runs on a worker thread. Everything the GUI
555 // thread posts back has to keep being dispatched until that worker has
556 // joined the GUI thread, so run_emulator() ends the pump itself.
557 int result = 0;
558 if (bx_gui && bx_gui->requires_main_thread())
559 {
560 bx_gui_c* const main_thread_gui = bx_gui;
561 std::thread emulator([&] { result = run_emulator(argc, argv); });
562 main_thread_gui->main_thread_pump();
563 emulator.join();
564 // Workers and their synchronous GUI calls have finished. Release host
565 // display resources here, on the same main thread that initialized SDL.
566 try
567 {
568 main_thread_gui->exit_application();
569 }
570 catch (const CException& cleanup_error)
571 {
572 printf("GUI shutdown failure: %s\n", cleanup_error.displayText().c_str());
573 result = 1;
574 }
575 }
576 else
577 result = run_emulator(argc, argv);
578
579 return result;
580}

585static int run_emulator(int argc, char* argv[])
586{
587 int result = 0;
588 try
589 {
590#if defined(IDB)
591 theSystem->start_threads(); // fix from axpbox commit 9ef3473
592
593 if (argc > 1 && argc < 4 && argv[argc - 1][0] == '@')
594 trc->run_script(argv[argc - 1] + 1);
595 else
596 trc->run_script(NULL);
597#else
598 theSystem->Run();
599#endif
600 }
601 catch (CGracefulException& e)
602 {
603 printf("Exiting gracefully: %s\n", e.displayText().c_str());
604
605 try
606 {
607 theSystem->stop_threads();
608 theSystem->flush_storage();
609
610 // save flash and dpr rom only if not terminated with a fatal error
611 theSROM->SaveStateF();
612 theDPR->SaveStateF();
613 if (theAli)
614 theAli->save_toy_nvram(true);
615
616#if defined(PROFILE)
617 {
618 FILE* p_fp;
619 u64 p_max = 0;
620 u64 p_i;
621 int p_j;
622
623 printf("Writing profile to profile.txt");
624
625 p_fp = fopen("profile.txt", "w");
626 for (p_i = PROFILE_FROM; p_i < PROFILE_TO; p_i += (4 * PROFILE_BUCKSIZE))
627 {
628 if (PROFILE_BUCKET(p_i) > p_max)
629 p_max = PROFILE_BUCKET(p_i);
630 }
631 fprintf
632 (
633 p_fp, "p_max = %10" PRId64 "; %10" PRId64 " profiled instructions.\n\n", p_max,
634 profiled_insts
635 );
636 for (p_i = PROFILE_FROM; p_i < PROFILE_TO; p_i += (4 * PROFILE_BUCKSIZE))
637 {
638 if (PROFILE_BUCKET(p_i))
639 {
640 fprintf(p_fp, "%016" PRIx64 ": %10" PRId64 " ", p_i, PROFILE_BUCKET(p_i));
641 for (p_j = 0; p_j < (((float)PROFILE_BUCKET(p_i) / (float)p_max) * 100);
642 p_j++)
643 fprintf(p_fp, "*");
644 fprintf(p_fp, "\n");
645 }
646 }
647 fclose(p_fp);
648 }
649#endif
650 }
651 catch (const CException& save_error)
652 {
653 printf("Shutdown save failure: %s\n", save_error.displayText().c_str());
654 result = 1;
655 }
656 }
657 catch (CException& e)
658 {
659 printf("Emulator Failure: %s\n", e.displayText().c_str());
660 result = 1;
661 }
662
663 if (theSystem)
664 {
665 theSystem->stop_threads();
666 if (detach_gui_guest_input())
667 delete theSystem;
668 else
669 result = 1;
670 }
671 if (bx_gui)
672 bx_gui->main_thread_stop();
673 return result;
674}
