286static inline u64 dram_read(const char* dram_ptr, u64 phys, int dsize)
287{
288 const char* p = dram_ptr + phys;
289 switch (dsize) {
290 case 8: return *(const u8*)p;
291 case 16: return *(const u16*)p;
292 case 32: return *(const u32*)p;
293 case 64: return *(const u64*)p;
294 default: return 0; // unreachable in practice
295 }
296}

298static inline void dram_write(char* dram_ptr, u64 phys, int dsize, u64 data)
299{
300 char* p = dram_ptr + phys;
301 switch (dsize) {
302 case 8: *(u8*)p = (u8)data; break;
303 case 16: *(u16*)p = (u16)data; break;
304 case 32: *(u32*)p = (u32)data; break;
305 case 64: *(u64*)p = data; break;
306 }
307}

396/* 64b * 64b unsigned multiply */
397inline u64 uemul64(u64 a, u64 b, u64* hi)
398{
399#if defined(_MSC_VER) && defined(_M_X64)
400 /* Full 128b product in one MULX; bit-identical to the portable path below. */
401 u64 h;
402 const u64 lo = _umul128(a, b, &h);
403 if (hi)
404 *hi = h;
405 return lo;
406#elif defined(__SIZEOF_INT128__)
407 const unsigned __int128 p = (unsigned __int128)a * (unsigned __int128)b;
408 if (hi)
409 *hi = (u64)(p >> 64);
410 return (u64)p;
411#else
412 u64 ahi;
413
414 u64 alo;
415
416 u64 bhi;
417
418 u64 blo;
419
420 u64 rhi;
421
422 u64 rmid1;
423
424 u64 rmid2;
425
426 u64 rlo;
427
428 ahi = (a >> 32) & X64_LONG;
429 alo = a & X64_LONG;
430 bhi = (b >> 32) & X64_LONG;
431 blo = b & X64_LONG;
432 rhi = ahi * bhi;
433 rmid1 = ahi * blo;
434 rmid2 = alo * bhi;
435 rlo = alo * blo;
436 rhi = rhi + ((rmid1 >> 32) & X64_LONG) + ((rmid2 >> 32) & X64_LONG);
437 rmid1 = (rmid1 << 32) & X64_QUAD;
438 rmid2 = (rmid2 << 32) & X64_QUAD;
439 rlo = (rlo + rmid1) & X64_QUAD;
440 if (rlo < rmid1)
441 rhi = rhi + 1;
442 rlo = (rlo + rmid2) & X64_QUAD;
443 if (rlo < rmid2)
444 rhi = rhi + 1;
445 if (hi)
446 *hi = rhi & X64_QUAD;
447 return rlo;
448#endif
449}

451/* 64b / 64b unsigned fraction divide */
452inline u64 ufdiv64(u64 dvd, u64 dvr, u32 prec, u32* sticky)
453{
454 u64 quo;
455 u32 i;
456
457 quo = 0; /* clear quotient */
458 for (i = 0; (i < prec) && dvd; i++)
459 { /* divide loop */
460 quo = quo << 1; /* shift quo */
461 if (dvd >= dvr)
462 { /* div step ok? */
463 dvd = dvd - dvr; /* subtract */
464 quo = quo + 1;
465 } /* quo bit = 1 */
466
467 dvd = dvd << 1;
468 } /* shift divd */
469
470 quo = quo << (UF_V_NM - i + 1); /* shift quo */
471 if (sticky)
472 *sticky = (dvd ? 1 : 0); /* set sticky bit */
473 return quo; /* return quotient */
474}

476/* SoftFloat estimateDiv128To64: approximate (a0:a1)/b to 64 bits; the
477 result is never greater than the true quotient and within 2 of it.
478 Requires b > a0 (guaranteed by fsqrt64: b has bit 63 set beyond a0). */
479inline u64 udiv128to64(u64 a0, u64 a1, u64 b)
480{
481 u64 b0, b1, rem0, rem1, term0, term1, z;
482 if (b <= a0)
483 return X64_QUAD;
484 b0 = b >> 32;
485 z = ((b0 << 32) <= a0) ? U64(0xffffffff00000000) : (a0 / b0) << 32;
486 term1 = uemul64(b, z, &term0); /* term0:term1 = b*z */
487 rem0 = a0 - term0 - ((a1 < term1) ? 1 : 0); /* a0:a1 - term0:term1 */
488 rem1 = a1 - term1;
489 while (Q_GETSIGN(rem0) != 0)
490 {
491 z -= U64(0x100000000);
492 b1 = b << 32;
493 rem1 = (rem1 + b1) & X64_QUAD;
494 rem0 = (rem0 + b0 + ((rem1 < b1) ? 1 : 0)) & X64_QUAD;
495 }
496 rem0 = (rem0 << 32) | (rem1 >> 32);
497 z |= ((b0 << 32) <= rem0) ? U64(0xffffffff) : rem0 / b0;
498 return z;
499}

501/* Fraction square root routine - code from SoftFloat */
502inline u64 fsqrt64(u64 asig, s32 exp)
503{
504 static const u32 sqrtOdd[] = {
505 0x0004, 0x0022, 0x005D, 0x00B1, 0x011D, 0x019F, 0x0236, 0x02E0,
506 0x039C, 0x0468, 0x0545, 0x0631, 0x072B, 0x0832, 0x0946, 0x0A67 };
507 static const u32 sqrtEven[] = {
508 0x0A2D, 0x08AF, 0x075A, 0x0629, 0x051A, 0x0429, 0x0356, 0x029E,
509 0x0200, 0x0179, 0x0109, 0x00AF, 0x0068, 0x0034, 0x0012, 0x0002 };
510
511 u64 zsig;
512 u64 remh;
513 u64 reml;
514 u64 t;
515 u32 index;
516 u32 z;
517 u32 a;
518 u32 sticky = 0;
519
520 /* Calculate an approximation to the square root of the 32-bit significand given
521 by 'a'. Considered as an integer, 'a' must be at least 2^31. If bit 0 of
522 'exp' (the least significant bit) is 1, the integer returned approximates
523 2^31*sqrt('a'/2^31), where 'a' is considered an integer. If bit 0 of 'exp'
524 is 0, the integer returned approximates 2^31*sqrt('a'/2^30). In either
525 case, the approximation returned lies strictly within +/-2 of the exact
526 value. */
527 a = (u32)(asig >> 32); /* high order frac */
528 index = (a >> 27) & 0xF; /* bits<30:27> */
529 if (exp & 1)
530 { /* odd exp? */
531 z = 0x4000 + (a >> 17) - sqrtOdd[index]; /* initial guess */
532 z = ((a / z) << 14) + (z << 15); /* Newton iteration */
533 a = a >> 1;
534 }
535 else
536 {
537 z = 0x8000 + (a >> 17) - sqrtEven[index]; /* initial guess */
538 z = (a / z) + z; /* Newton iteration */
539 z = (z >= 0x20000) ? 0xFFFF8000 : (z << 15);
540 if (z <= a)
541 z = (a >> 1) | 0x80000000;
542 }
543
544 zsig = (((((u64)a) << 31) / ((u64)z)) + (z >> 1)) & X64_LONG;
545
546 /* Calculate the final answer in two steps. First, do one iteration of
547 Newton's approximation. The divide-by-2 is accomplished by clever
548 positioning of the operands. Then, check the bits just below the
549 (double precision) rounding bit to see if they are close to zero
550 (that is, the rounding bits are close to midpoint). If so, make
551 sure that the result^2 is <below> the input operand */
552 asig = asig >> ((exp & 1) ? 3 : 2); /* leave 2b guard */
553 /* Newton iteration: asig*2^64 / (zsig*2^32), per SoftFloat.
554 ufdiv64 was wrong here: with zsig >= 2^31 (always, on the even-exp path)
555 its divisor has bit 63 set, the shifted dividend wraps past 2^64, and the
556 garbage estimate sends the correction loop below walking ~2^60 ULPs */
557 zsig = udiv128to64(asig, 0, zsig << 32) + (zsig << 30);
558 if ((zsig & 0x1FF) <= 5)
559 { /* close to even? */
560 reml = uemul64(zsig, zsig, &remh); /* result^2 (returns lo, *hi = hi) */
561 remh = (asig - remh - (reml ? 1 : 0)) & X64_QUAD; /* arg - result^2 */
562 reml = NEG_Q(reml);
563 while (Q_GETSIGN(remh) != 0)
564 { /* if arg < result^2 */
565 zsig = (zsig - 1) & X64_QUAD; /* decr result */
566 t = ((zsig << 1) & X64_QUAD) | 1; /* incr result^2 */
567 reml = (reml + t) & X64_QUAD; /* and retest */
568 remh = (remh + (zsig >> 63) + ((reml < t) ? 1 : 0)) & X64_QUAD;
569 }
570
571 if ((remh | reml) != 0)
572 sticky = 1;
573 } /* not exact? */
574
575 zsig = (zsig << 1) | sticky; /* left justify result */
576 return zsig;
577}
