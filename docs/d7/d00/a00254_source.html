337 // Alpha ordering approximations (conservative):
338 // TRAPB: serialize exceptions + memory; use full fence.
339#define DO_TRAPB std::atomic_thread_fence(std::memory_order_seq_cst);
340// EXCB/MB: full fence (data+io ordering).
341#define DO_EXCB std::atomic_thread_fence(std::memory_order_seq_cst);
342#define DO_MB std::atomic_thread_fence(std::memory_order_seq_cst);
343// WMB: HRM 4.11.1 requires WMB to order all prior stores from this CPU
344// before all subsequent stores, as observed by every other CPU/DMA agent.
345// A standalone atomic_thread_fence(release) only synchronizes with a matching
346// acquire fence/load on another thread; it does NOT impose store-store
347// ordering visible to unsynchronized observers.
348#define DO_WMB std::atomic_thread_fence(std::memory_order_seq_cst);

352#define DO_ECB ; /* cache hint: no effect */
