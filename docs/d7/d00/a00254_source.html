30// Architectural AMASK assignments, shared by feature reporting and execution.
31inline constexpr std::uint64_t kAlphaBWX = 0x001;
32inline constexpr std::uint64_t kAlphaFIX = 0x002;
33inline constexpr std::uint64_t kAlphaCIX = 0x004;
34inline constexpr std::uint64_t kAlphaMVI = 0x100;
35
36// Only extension availability is decided here. Reserved encodings, operand
37// restrictions, privilege, and family-specific HW forms remain decoder rules.
38constexpr std::uint64_t alpha_instruction_extensions(std::uint32_t ins)
39{
40 switch (ins >> 26) {
41 case 0x0a: case 0x0c: case 0x0d: case 0x0e: // LDBU/LDWU/STW/STB
42 return kAlphaBWX;
43 case 0x14: // ITFP: ITOF* and all SQRT* forms; absent without FIX.
44 return kAlphaFIX;
45 case 0x1c:
46 switch ((ins >> 5) & 0x7f) {
47 case 0x00: case 0x01: // SEXTB/SEXTW
48 return kAlphaBWX;
49 case 0x30: case 0x32: case 0x33: // CTPOP/CTLZ/CTTZ
50 return kAlphaCIX;
51 case 0x31: // PERR
52 case 0x34: case 0x35: case 0x36: case 0x37: // UNPK*/PK*
53 case 0x38: case 0x39: case 0x3a: case 0x3b: // MIN*
54 case 0x3c: case 0x3d: case 0x3e: case 0x3f: // MAX*
55 return kAlphaMVI;
56 case 0x70: case 0x78: // FTOIT/FTOIS
57 return kAlphaFIX;
58 }
59 }
60 return 0;
61}

63constexpr bool alpha_instruction_supported(std::uint64_t amask, std::uint32_t ins)
64{
65 return (alpha_instruction_extensions(ins) & ~amask) == 0;
66}

68// Pure family decode rules, after extension checks and zero-register cleanup.
69struct AlphaInstructionDecodePolicy
70{
71 // True skips operand checks, execution, and exceptions; null decodes normally.
72 bool (*retire_at_decode)(std::uint32_t ins) = nullptr;
73 // Pure encoding-only FEN rule after retirement; null decodes normally.
74 bool (*fen_at_decode)(std::uint32_t ins) = nullptr;
75 // Pure selector: low four bits mark forbidden FPCR.DYN modes; null permits all.
76 std::uint32_t (*fen_rounding_modes)(std::uint32_t ins) = nullptr;
77};

79constexpr bool alpha_instruction_retires_at_decode(
80 const AlphaInstructionDecodePolicy& policy, std::uint32_t ins)
81{
82 return policy.retire_at_decode && policy.retire_at_decode(ins);
83}

85constexpr bool alpha_instruction_fen_at_decode(
86 const AlphaInstructionDecodePolicy& policy, std::uint32_t ins)
87{
88 return policy.fen_at_decode && policy.fen_at_decode(ins);
89}

91constexpr std::uint32_t alpha_instruction_fen_rounding_modes(
92 const AlphaInstructionDecodePolicy& policy, std::uint32_t ins)
93{
94 return policy.fen_rounding_modes ? policy.fen_rounding_modes(ins) & 0xfu : 0;
95}

97constexpr bool alpha_instruction_fen_for_fpcr(
98 const AlphaInstructionDecodePolicy& policy, std::uint32_t ins, std::uint64_t fpcr)
99{
100 return (alpha_instruction_fen_rounding_modes(policy, ins)
101 & (1u << ((fpcr >> 58) & 3))) != 0;
102}
