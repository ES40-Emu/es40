56static uint32_t radeon_setting(
57 CConfigurator* cfg, const char* key, uint32_t def)
58{
59 const char* text = cfg->get_text_value(key, nullptr);
60 if (!text || !*text)
61 return def;
62 char* end = nullptr;
63 errno = 0;
64 const unsigned long long n = std::strtoull(text, &end, 0);
65 if (errno || *end || text[0] == '-' || n > 0xffffffffULL)
66 throw std::runtime_error(
67 std::string("Invalid radeon integer setting: ") + key);
68 return static_cast<uint32_t>(n);
69}

92static void radeon_write_u32(FILE* f, uint32_t v)
93{
94 uint8_t b[4];
95 for (unsigned i = 0; i < 4; ++i)
96 b[i] = uint8_t(v >> (8 * i));
97 if (fwrite(b, 1, 4, f) != 4)
98 throw std::runtime_error("Snapshot write failed");
99}

102{
103 uint8_t b[4];
104 if (fread(b, 1, 4, f) != 4)
105 throw std::runtime_error("Snapshot read failed");
106 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
107 (uint32_t(b[3]) << 24);

110static bool radeon_vga_window_offset(u32& address, unsigned selection)
111{
112 switch (selection & 3)
113 {
114 case 1:
115 return address < 0x10000;
116 case 2:
117 if (address < 0x10000 || address >= 0x18000)
118 return false;
119 address -= 0x10000;
120 break;
121 case 3:
122 if (address < 0x18000 || address >= 0x20000)
123 return false;
124 address -= 0x18000;
125 break;
126 default:
127 return address < 0x20000;
128 }
129 return true;
130}

142void CRadeon7500::sequencer_map(address_map& map)
143{
144 unimplemented_map(map, "VGA sequencer register");
145 // These frontends mask stored bytes, including direct snapshot writes.
146 // Limit the canonical map to SR00-04 so unknown indices keep diagnostics.
147 auto standard = std::make_shared<address_map>(5);
148 vga_device::sequencer_map(*standard);
149 map(0x00, 0x04).lrw8(
150 NAME([standard](offs_t index) {
151 return standard->read_byte(index);
152 }),
153 NAME(([this, standard](offs_t index, u8 data) {
154 static constexpr u8 masks[] = {0x03, 0x3f, 0x0f, 0x3f, 0x0e};
155 vga.sequencer.data[index] = data & masks[index];
156 standard->write_byte(index, data);
157 })));
158}

166void CRadeon7500::attribute_map(address_map& map)
167{
168 unimplemented_map(map, "VGA attribute register");
169 vga_device::attribute_map(map);
170 // Palette reads remain available when PAS disables CPU writes.
171 map(0x20, 0x2f).lr8(NAME([this](offs_t offset) {
172 return vga.attribute.data[offset];
173 }));
174 // These frontends retain mode-control bits 6/7 and mask reserved bit 4.
175 map(0x10, 0x10).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
176 vga.attribute.data[0x10] = data & 0xef;
177 }));
178 // Color select does not change the pending horizontal panning value.
179 map(0x14, 0x14).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
180 vga.attribute.data[0x14] = data & 0xf;
181 }));
182}

201u8 CRadeon7500::mem_r(offs_t address)
202{
203 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
204 return 0xff;
205 unsigned selected = vga.gc.read_map_sel & 3;
206 if (vga.sequencer.data[4] & 8)
207 {
208 // Keep packed chain-four bytes in the layout consumed by vga_device scanout.
209 for (unsigned plane = 0; plane < 4; ++plane)
210 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
211 selected = address & 3;
212 }
213 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
214 {
215 // vga_device text stores character/attribute pairs together; fonts stay planar.
216 vga.gc.latch[0] = vga.memory[address & ~1u];
217 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
218 for (unsigned plane = 2; plane < 4; ++plane)
219 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
220 selected = (vga.gc.read_map_sel & 2) | (address & 1);
221 }
222 else
223 {
224 for (unsigned plane = 0; plane < 4; ++plane)
225 vga.gc.latch[plane] =
226 vga.memory[(address & 0xffff) + plane * 0x10000];
227 }
228 if (!vga.gc.read_mode)
229 return vga.gc.latch[selected];
230 u8 result = 0xff;
231 for (unsigned plane = 0; plane < 4; ++plane)
232 if (vga.gc.color_dont_care & (1 << plane))
233 result &=
234 u8(vga.gc.latch[plane] ^
235 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
236 return result;
237}

239void CRadeon7500::mem_w(offs_t address, u8 data)
240{
241 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
242 return;
243 if (vga.sequencer.data[4] & 8)
244 {
245 if (vga.sequencer.map_mask & (1 << (address & 3)))
246 vga.memory[address] = vga_latch_write(address & 3, data);
247 return;
248 }
249 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
250 {
251 const unsigned parity = address & 1;
252 if (vga.sequencer.map_mask & (1 << parity))
253 vga.memory[address] = vga_latch_write(parity, data);
254 if (vga.sequencer.map_mask & (1 << (parity + 2)))
255 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
256 vga_latch_write(parity + 2, data);
257 return;
258 }
259 for (unsigned plane = 0; plane < 4; ++plane)
260 if (vga.sequencer.map_mask & (1 << plane))
261 vga.memory[(address & 0xffff) + plane * 0x10000] =
262 vga_latch_write(plane, data);
263}

291CRV200::Frame CRadeon7500::render_frame()
292{
293 std::lock_guard<std::recursive_mutex> guard(
294 cSystem->get_device_bus_mutex());
295 if (m_pause.load() || !m_rv200.display_output_enabled())
296 return {};
297 // The primary DAC may select CRTC2 even while CRTC1 is in VGA mode.
298 if (m_rv200.native_display())
299 return m_rv200.scanout();
300 if (!m_vga_enable || !vga.crtc.sync_en ||
301 (vga.sequencer.data[0] & 3) != 3 || (vga.sequencer.data[1] & 0x20) ||
302 !(vga.attribute.index & 0x20) || !vga.crtc.maximum_scan_line ||
303 (m_rv200.peek(CRV200::CRTC_EXT_CNTL) & CRV200::DISPLAY_DIS))
304 return {};
305 CRV200::Frame frame;
306 frame.width = (vga.crtc.horz_disp_end + 1) *
307 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
308 frame.height = vga.crtc.vert_disp_end + 1;
309 if (!frame.width || frame.width > 4096 || frame.height > 1024)
310 return {};
311 // vga_device can write repeated scanlines past the visible-area bottom. Cover CGA
312 // double-scan plus one character row; only the visible rectangle is used.
313 m_render_bitmap.resize(frame.width, frame.height * 2 + 64);
314 screen().set_visible_area(frame.width, frame.height);
315 screen().tick_frame();
316 vga.crtc.start_addr = latch_start_addr();
317 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
318 palette_update();
319 svga_device::screen_update(
320 screen(), m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
321 frame.argb.assign(
322 m_render_bitmap.raw(),
323 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
324 return frame;
325}

327void CRadeon7500::update()
328{
329 const CRV200::Frame frame = render_frame();
330 // Never hold the device-bus mutex while acquiring a GUI mutex.
331 CRadeonDisplayLock guard(m_output.display());
332 if (!frame.argb.empty())
333 {
334 if (frame.width != m_host_width || frame.height != m_host_height)
335 {
336 m_output.display().dimension_update(
337 frame.width, frame.height, 0, 0, 32);
338 m_host_width = frame.width;
339 m_host_height = frame.height;
340 }
341 m_output.display().graphics_frame_update(
342 frame.argb.data(), frame.width, frame.height);
343 }
344 else
345 {
346 m_output.display().clear_screen();
347 }
348 m_output.display().flush();
349}

354 {
355 if (!m_gui_initialized)

395std::array<u32, 64> CRadeon7500::PCIConfig::config_data() const
396{
397 std::array<u32, 64> data{};
398 data[0] = u32(vendor) | (u32(device) << 16);
399 data[2] = (class_code << 8) | revision;
400 data[4] = 0x00000008; // prefetchable framebuffer
401 data[5] = 0x00000001; // I/O register window
402 data[11] = u32(subsystem_vendor) | (u32(subsystem_device) << 16);
403 data[15] = 0x000001ff; // INTA#, interrupt line unassigned
404 // No AGP or power-management capability chain is modeled.
405 return data;
406}

408std::array<u32, 64> CRadeon7500::PCIConfig::config_mask(u32 rom_size)
409{
410 std::array<u32, 64> mask{};
411 mask[1] = 0x00000147; // I/O, memory, bus master, parity response, SERR
412 mask[3] = 0x0000ffff; // cache-line size and latency timer
413 mask[4] = ~(FramebufferBARSize - 1);
414 mask[5] = ~(IOBARSize - 1);
415 mask[6] = ~(RegisterBARSize - 1);
416 if (rom_size)
417 mask[12] = (~(rom_size - 1) & 0xfffff800u) | 1; // ROM decode and enable
418 mask[15] = 0xff; // interrupt line only; pin fixed
419 return mask;
420}

422CRadeon7500::OptionROM CRadeon7500::OptionROM::parse(
423 const std::vector<uint8_t>& raw, const std::string& layout)
424{
425 auto fail = [](const char* message) {
426 throw std::invalid_argument(message);
427 };
428 const size_t n = raw.size();
429 if (n < 2048 || n > 1024 * 1024)
430 fail("ROM must be 2 KiB to 1 MiB");
431 if (layout != "exact" && layout != "pad_ff")
432 fail("rom_layout must be exact or pad_ff");
433 if (layout == "exact" && (n & (n - 1)))
434 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
435 auto word = [&](size_t p) -> uint16_t {
436 if (p > n - 2)
437 fail("ROM field out of bounds");
438 return uint16_t(uint16_t(raw[p]) | (uint16_t(raw[p + 1]) << 8));
439 };
440 size_t at = 0;
441 bool last = false;
442 for (unsigned images = 0; images < 32 && !last; ++images)
443 {
444 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
445 fail("ROM missing 55 AA signature");
446 const size_t p = at + word(at + 0x18);
447 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
448 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
449 fail("ROM missing/bounded PCIR structure");
450 if (word(p + 4) != 0x1002 || word(p + 6) != 0x5157)
451 fail("ROM PCIR identity is not ATI 1002:5157 (RV200)");
452 const size_t length = size_t(word(p + 0x10)) * 512;
453 if (!length || length > n - at || p + 0x18 > at + length)
454 fail("Invalid option-ROM image length");
455 const size_t pcir_length = word(p + 0x0a);
456 if (pcir_length < 0x18 || pcir_length > at + length - p)
457 fail("Invalid PCIR structure length");
458 if (raw[p + 0x14] == 0 &&
459 (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length))
460 fail("Invalid x86 initialization image length");
461 unsigned sum = 0;
462 for (size_t i = at; i < at + length; ++i)
463 sum += raw[i];
464 if (sum & 255)
465 fail("Option-ROM checksum failed");
466 if (raw[p + 0x14] != 0 && raw[p + 0x14] != 3)
467 fail("Only x86 and EFI option-ROM images are supported");
468 last = (raw[p + 0x15] & 0x80) != 0;
469 at += length;
470 }
471 if (!last)
472 fail("Unterminated option-ROM image chain");
473 OptionROM rom;
474 rom.file_size = static_cast<uint32_t>(n);
475 const std::string agp = "V200AGP";
476 if (std::search(raw.begin(), raw.end(), agp.begin(), agp.end()) !=
477 raw.end())
478 fail("AGP ROM rejected for PCI 3X-PBXGG-AA target; choose a PCI image");
479 rom.bytes = raw;
480 if (layout == "pad_ff")
481 {
482 // Erased padding to the next power of two; the upper decode is unmeasured.
483 size_t decoded = 2048;
484 while (decoded < n)
485 decoded *= 2;
486 rom.bytes.resize(decoded, 0xff);
487 }
488 return rom;
489}

519void CRadeon7500::init()
520try
521{
522 if (m_initialized)
523 return;
524 CConfigurator* cfg = myCfg;
525 init_maps();
526 add_legacy_io(LegacyIO, 0x3b0, 0x30);
527 add_legacy_mem(LegacyMemory, 0xa0000, 0x20000);
528 m_profile.revision =
529 static_cast<uint8_t>(radeon_bounded_setting(cfg, "revision", 0, 255));
530 m_profile.subsystem_vendor = static_cast<uint16_t>(
531 radeon_bounded_setting(cfg, "subsystem_vendor", 0, 65535));
532 m_profile.subsystem_device = static_cast<uint16_t>(
533 radeon_bounded_setting(cfg, "subsystem_device", 0, 65535));
534 m_scanline_access_divisor =
535 radeon_setting(cfg, "scanline_access_divisor", 1024);
536 if (m_scanline_access_divisor == 0)
537 throw std::runtime_error("Invalid radeon scanline_access_divisor");
538 m_trace_apertures = cfg->get_bool_value("trace_apertures", false);
539 m_frame_file = cfg->get_text_value("frame_file", "");
540 const std::string romname = cfg->get_text_value("rom", "");
541 m_rom_layout = cfg->get_text_value("rom_layout", "exact");
542 if (!romname.empty())
543 {
544 std::ifstream f(romname, std::ios::binary | std::ios::ate);
545 if (!f)
546 throw std::runtime_error("Cannot open radeon ROM: " + romname);
547 const auto size = f.tellg();
548 if (size < 2048 || size > 1024 * 1024)
549 throw std::runtime_error("radeon ROM must be 2 KiB to 1 MiB");
550 f.seekg(0);
551 std::vector<uint8_t> raw(static_cast<size_t>(size));
552 f.read(
553 reinterpret_cast<char*>(raw.data()),
554 static_cast<std::streamsize>(raw.size()));
555 if (!f)
556 throw std::runtime_error("Cannot read full radeon ROM");
557 const auto parsed = OptionROM::parse(raw, m_rom_layout);
558 m_rom = parsed.bytes;
559#ifdef DEBUG_VGA
560 printf(
561 "%s: ROM %u bytes, layout=%s, decoded %u bytes.\n",
562 devid_string,
563 parsed.file_size,
564 m_rom_layout.c_str(),
565 unsigned(m_rom.size()));
566#endif
567 }
568 const std::string tracename = cfg->get_text_value("trace", "");
569 if (!tracename.empty())
570 {
571 if (std::filesystem::exists(tracename))
572 throw std::runtime_error(
573 "Refusing to overwrite existing radeon trace: " + tracename);
574 m_trace.open(tracename);
575 if (!m_trace)
576 throw std::runtime_error("Cannot create radeon trace");
577 m_trace << "# RV200 ES40 PCI/BAR traffic; values hex, sizes bits\n";
578 }
579 const auto profile_data = m_profile.config_data(),
580 profile_mask = PCIConfig::config_mask(u32(m_rom.size()));
581 u32 data[64]{}, mask[64]{};
582 for (unsigned i = 0; i < 64; ++i)
583 {
584 data[i] = profile_data[i];
585 mask[i] = profile_mask[i];
586 }
587 add_function(0, data, mask);
588 m_rv200.set_vga_callbacks(
589 [this](uint32_t port) {
590 return io_read_b(port);
591 },
592 [this](uint32_t port, uint8_t value) {
593 io_write_b(port, value);
594 });
595 m_rv200.set_palette_index_callback([this]() {
596 vga.dac.state = 0;
597 });
598 m_rv200.set_config_reader([this](uint32_t address) {
599 return endian_32(pci_state.config_data[0][(address & 0xfc) / 4]);
600 });
601 m_rv200.set_dma_reader(
602 [this](uint32_t address, uint8_t* destination, size_t count) {
603 if (!(config_read(0, 4, 16) & 4))
604 return false;
605 // Use the shared PCI/IOMMU path: OpenVMS allocates its ring and
606 // indirect buffers in scatter-gather mapped system memory.
607 do_pci_read(address, destination, 1, count);
608 if (m_trace && count == 4)
609 {
610 const u32 word = u32(destination[0]) | (u32(destination[1]) << 8) |
611 (u32(destination[2]) << 16) | (u32(destination[3]) << 24);
612 trace("DR", 0, address, 32, word);
613 }
614 return true;

784bool CRadeon7500::decodes_memory_access(
785 int index, u64 address, int dsize, bool write) const noexcept
786{
787 if (index >= PCI_RANGE_BASE)
788 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
789 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
790 return false;
791 const unsigned bytes = unsigned(dsize) / 8;
792 if (index == LegacyIO)
793 {
794 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
795 return false;
796 // Claim an access only when every byte belongs to this device; reserved
797 // port bytes must not make us a competing positive decoder.
798 for (unsigned i = 0; i < bytes; ++i)
799 if (!vga_port_enabled(0x3b0 + u32(address) + i))
800 return false;
801 return true;
802 }
803 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
804 bytes > 0x20000 - address)
805 return false;
806 switch (vga.gc.memory_map_sel & 3)
807 {
808 case 1:
809 return address + bytes <= 0x10000;
810 case 2:
811 return address >= 0x10000 && address + bytes <= 0x18000;
812 case 3:
813 return address >= 0x18000;
814 default:
815 return true;
816 }
817}

860u32 CRadeon7500::ReadMem_Bar(int func, int bar, u32 address, int dsize)
861{
862 std::lock_guard<std::recursive_mutex> guard(
863 cSystem->get_device_bus_mutex());
864 if (func != 0 || (dsize != 8 && dsize != 16 && dsize != 32))
865 return 0xffffffff;
866 tick();
867 u32 data = 0xffffffff;
868 switch (bar)
869 {
870 case 0:
871 data = u32(m_rv200.mem_read(address, dsize));
872 break;
873 case 1:
874 if (address <= IOBARSize - u32(dsize) / 8)
875 data = m_rv200.ReadMem(address, dsize);
876 break;
877 case 2:
878 data = m_rv200.ReadMem(address, dsize);
879 break;
880 case 6:
881 // BUS_CNTL can hide the option ROM from the BAR.
882 if (!(m_rv200.peek(CRV200::BUS_CNTL) & CRV200::BUS_BIOS_DIS_ROM) &&
883 !m_rom.empty() && address <= m_rom.size() - u32(dsize) / 8)
884 {
885 data = 0;
886 for (int i = 0; i < dsize / 8; ++i)
887 data |= u32(m_rom[address + static_cast<unsigned>(i)])
888 << (8 * i);
889 }
890 break;
891 }
892 trace("BR", bar, address, dsize, data);
893 return data;
894}

1053std::vector<u8> CRadeon7500::save_vga_state() const
1054{
1055 std::vector<u8> bytes{1};
1056 for (unsigned i = 0; i < 25; ++i)
1057 bytes.push_back(m_crtc_map.read_byte(i));
1058 for (unsigned i = 0; i < 5; ++i)
1059 bytes.push_back(m_seq_map.read_byte(i));
1060 for (unsigned i = 0; i < 9; ++i)
1061 bytes.push_back(m_gc_map.read_byte(i));
1062 for (unsigned i = 0; i < 21; ++i)
1063 bytes.push_back(m_atc_map.read_byte(i));
1064 bytes.insert(
1065 bytes.end(),
1066 {vga.crtc.index,
1067 vga.sequencer.index,
1068 vga.gc.index,
1069 vga.attribute.index,
1070 vga.miscellaneous_output,
1071 vga.feature_control,
1072 m_vga_enable,
1073 vga.dac.mask,
1074 u8(vga.attribute.state),
1075 u8(vga.dac.read),
1076 u8(vga.dac.state)});
1077 for (const u8 value : vga.dac.loading)
1078 bytes.push_back(value);
1079 for (const u8 value : vga.gc.latch)
1080 bytes.push_back(value);
1081 for (unsigned i = 0; i < 4; ++i)
1082 bytes.push_back(u8(vga.crtc.start_addr >> (8 * i)));
1083 bytes.push_back(vga.attribute.pel_shift);
1084 bytes.push_back(vga.cursor.visible);
1085 bytes.push_back(vga.crtc.irq_latch);
1086 for (unsigned i = 0; i < 4; ++i)
1087 bytes.push_back(u8(m_vga_scanline >> (8 * i)));
1088 for (unsigned i = 0; i < 4; ++i)
1089 bytes.push_back(u8(m_vga_status_phase >> (8 * i)));
1090 return bytes;
1091}

1093void CRadeon7500::restore_vga_state(const std::vector<u8>& bytes)
1094{
1095 auto dword = [&](size_t offset) {
1096 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1097 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1098 };
1099 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1100 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1101 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1102 dword(90) > 31)
1103 throw std::runtime_error("Invalid VGA snapshot fields");
1104 reset_vga();
1105 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1106 for (unsigned i = 0; i < 25; ++i)
1107 if (i != 0x11)
1108 m_crtc_map.write_byte(i, bytes[1 + i]);
1109 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1110 for (unsigned i = 0; i < 5; ++i)
1111 m_seq_map.write_byte(i, bytes[26 + i]);
1112 for (unsigned i = 0; i < 9; ++i)
1113 m_gc_map.write_byte(i, bytes[31 + i]);
1114 for (unsigned i = 0; i < 21; ++i)
1115 m_atc_map.write_byte(i, bytes[40 + i]);
1116 // Every register must read back exactly as saved.
1117 for (unsigned i = 0; i < 25; ++i)
1118 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1119 throw std::runtime_error("Invalid CRTC snapshot encoding");
1120 for (unsigned i = 0; i < 5; ++i)
1121 if (m_seq_map.read_byte(i) != bytes[26 + i])
1122 throw std::runtime_error("Invalid sequencer snapshot encoding");
1123 for (unsigned i = 0; i < 9; ++i)
1124 if (m_gc_map.read_byte(i) != bytes[31 + i])
1125 throw std::runtime_error(
1126 "Invalid graphics-controller snapshot encoding");
1127 for (unsigned i = 0; i < 21; ++i)
1128 if (m_atc_map.read_byte(i) != bytes[40 + i])
1129 throw std::runtime_error("Invalid attribute snapshot encoding");
1130 vga.crtc.index = bytes[61];
1131 vga.sequencer.index = bytes[62];
1132 vga.gc.index = bytes[63];
1133 vga.attribute.index = bytes[64];
1134 vga.miscellaneous_output = bytes[65];
1135 m_ioas = (bytes[65] & 1) != 0;
1136 vga.feature_control = bytes[66];
1137 m_vga_enable = bytes[67];
1138 vga.dac.mask = bytes[68];
1139 vga.attribute.state = bytes[69];
1140 vga.dac.read = bytes[70];
1141 vga.dac.state = bytes[71];
1142 for (unsigned i = 0; i < 3; ++i)
1143 vga.dac.loading[i] = bytes[72 + i];
1144 for (unsigned i = 0; i < 4; ++i)
1145 vga.gc.latch[i] = bytes[75 + i];
1146 vga.crtc.start_addr = dword(79);
1147 vga.attribute.pel_shift = bytes[83];
1148 vga.cursor.visible = bytes[84];
1149 vga.crtc.irq_latch = bytes[85];
1150 m_vga_scanline = dword(86);
1151 m_vga_status_phase = dword(90);
1152 vga.dac.dirty = 1;
1153}

1190int CRadeon7500::RestoreState(FILE* f)
1191{
1192 if (!f)
1193 return -1;
1194 std::lock_guard<std::recursive_mutex> guard(
1195 cSystem->get_device_bus_mutex());
1196 try
1197 {
1198 // PCI state is restored first, following the other ES40 devices.
1199 // A failure aborts the whole system restore; it must not resume the VM.
1200 struct ReplayGuard
1201 {
1202 bool& flag;
1203
1204 ReplayGuard(bool& value) : flag(value) { flag = true; }
1205
1206 ~ReplayGuard() { flag = false; }
1207 } replay(m_replaying_pci);
1208
1209 if (CPCIDevice::RestoreState(f))
1210 return -1;
1211 if (radeon_read_u32(f) != 0x31563752 ||
1212 radeon_read_u32(f) != VGAStateSize)
1213 return -1;
1214 std::vector<u8> vga_bytes(VGAStateSize);
1215 if (fread(vga_bytes.data(), 1, vga_bytes.size(), f) != vga_bytes.size())
1216 return -1;
1217 const u32 lo = radeon_read_u32(f), hi = radeon_read_u32(f),
1218 n = radeon_read_u32(f);
1219 const u32 vram = m_rv200.options().vram_bytes;
1220 if (n < vram || n > vram + 65536 + 16)
1221 return -1;
1222 std::string bytes(n, '\0');
1223 if (fread(bytes.data(), 1, n, f) != n)
1224 return -1;
1225 std::istringstream stream(bytes, std::ios::binary);
1226 m_rv200.RestoreState(stream);
1227 // The core's transactional loader owns its VRAM; refresh our borrowed view.
1228 vga.memory = m_rv200.vram_data();
1229 if (stream.peek() != std::char_traits<char>::eof())
1230 return -1;
1231 restore_vga_state(vga_bytes);
1232 m_access_count = u64(lo) | (u64(hi) << 32);
1233 sync_apertures();
1234 // IRQ output is committed by finalize_restore after all devices restore.
1235 return 0;
1236 }
1237 catch (const std::exception& e)
1238 {

1290{

1292 // monochrome/colour port selection.

1294 return 0xff;

1296 {
1297 case 0x3ba:
1298 case 0x3da:
1299 return input_status();

1301 return m_vga_enable;
1302 case 0x3c6:
1303 return vga.dac.mask;
1304 case 0x3c7:

1306 case 0x3c8:

1308 case 0x3c9:

1317 {
1318 vga.dac.state = 0;
1319 m_rv200.WriteMem(
1320 CRV200::PALETTE_INDEX,
1321 32,
1322 (m_rv200.peek(CRV200::PALETTE_INDEX) & ~0xff0000u) |
1323 (((index + 1) & 255) << 16));
1324 }

1326 }
1327 default:
1328 {
1329 u8 data;
1330 if (vga_io_read(port, data))
1331 return data;
