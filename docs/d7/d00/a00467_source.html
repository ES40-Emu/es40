337template <typename F>
338static void on_main_thread(F fn)
339{
340 if (!sdl_main_thread_owns_sdl.load(std::memory_order_acquire) || SDL_IsMainThread())
341 {
342 fn();
343 return;
344 }
345
346 struct SCall
347 {
348 F* fn;
349 std::exception_ptr error;
350 } call = { &fn, nullptr };
351
352 if (!SDL_RunOnMainThread([](void* p)
353 {
354 SCall* c = static_cast<SCall*>(p);
355 try { (*c->fn)(); }
356 catch (...) { c->error = std::current_exception(); }
357 }, &call, true))
358 {
359 FAILURE_1(SDL, "Unable to reach the SDL main thread: %s", SDL_GetError());
360 }
361
362 if (call.error)
363 std::rethrow_exception(call.error);
364}

400void bx_sdl_gui_c::main_thread_pump()
401{
402 // Collect OS events into SDL's queue and dispatch whatever the GUI thread
403 // posted with on_main_thread(). handle_events() drains the queue itself,
404 // so the wait is passed NULL and leaves the events where they are.
405 //
406 // SDL_PumpEvents() is what runs the queued on_main_thread() callbacks, so
407 // it has to happen every time round: SDL_WaitEventTimeout() returns early
408 // without pumping while the queue is non-empty, and waiting on that alone
409 // would hang any thread blocked in on_main_thread(). With an empty queue
410 // it sleeps until a posted callback or an OS event wakes it, which keeps
411 // the loop off the CPU without adding latency to the GUI thread.
412 while (sdl_main_thread_owns_sdl.load(std::memory_order_acquire))
413 {
414 SDL_PumpEvents();
415 if (SDL_WaitEventTimeout(NULL, sdl_pump_idle_ms))
416 SDL_Delay(1); // queue not drained yet; let the GUI thread catch up
417 }
418}

481static std::string trim_hotkey_text(const std::string& value)
482{
483 size_t first = 0;
484 while (first < value.size() &&
485 std::isspace((unsigned char)value[first]))
486 first++;
487
488 size_t last = value.size();
489 while (last > first && std::isspace((unsigned char)value[last - 1]))
490 last--;
491
492 return value.substr(first, last - first);
493}

495static bool hotkey_text_equals(const std::string& lhs, const char* rhs)
496{
497 if (!rhs || lhs.size() != strlen(rhs))
498 return false;
499
500 for (size_t i = 0; i < lhs.size(); i++)
501 {
502 if (std::tolower((unsigned char)lhs[i]) !=
503 std::tolower((unsigned char)rhs[i]))
504 return false;
505 }
506 return true;
507}

509static bool parse_hotkey_modifier(const std::string& token,
510 unsigned* modifier)
511{
512 if (hotkey_text_equals(token, "ctrl") ||
513 hotkey_text_equals(token, "control"))
514 *modifier = SDL_HOTKEY_MOD_CTRL;
515 else if (hotkey_text_equals(token, "alt") ||
516 hotkey_text_equals(token, "option"))
517 *modifier = SDL_HOTKEY_MOD_ALT;
518 else if (hotkey_text_equals(token, "shift"))
519 *modifier = SDL_HOTKEY_MOD_SHIFT;
520 else if (hotkey_text_equals(token, "gui") ||
521 hotkey_text_equals(token, "super") ||
522 hotkey_text_equals(token, "win") ||
523 hotkey_text_equals(token, "windows") ||
524 hotkey_text_equals(token, "cmd") ||
525 hotkey_text_equals(token, "command") ||
526 hotkey_text_equals(token, "meta"))
527 *modifier = SDL_HOTKEY_MOD_GUI;
528 else
529 return false;
530
531 return true;
532}

557static unsigned sdl_hotkey_modifiers(SDL_Keymod modifiers)
558{
559 unsigned result = 0;
560 if (modifiers & SDL_KMOD_CTRL) result |= SDL_HOTKEY_MOD_CTRL;
561 if (modifiers & SDL_KMOD_ALT) result |= SDL_HOTKEY_MOD_ALT;
562 if (modifiers & SDL_KMOD_SHIFT) result |= SDL_HOTKEY_MOD_SHIFT;
563 if (modifiers & SDL_KMOD_GUI) result |= SDL_HOTKEY_MOD_GUI;
564 if (modifiers & (SDL_KMOD_MODE | SDL_KMOD_LEVEL5))
565 result |= SDL_HOTKEY_MOD_UNSUPPORTED;
566 return result;
567}

575static sdl_hotkey_binding parse_sdl_hotkey(CConfigurator* cfg,
576 const char* config_name, const char* default_value)
577{
578 sdl_hotkey_binding binding;
579 binding.config_name = config_name;
580
581 const char* configured = cfg->get_text_value(config_name, default_value);
582 std::string value = trim_hotkey_text(configured ? configured : "");
583 if (hotkey_text_equals(value, "none"))
584 {
585 binding.display = "none";
586 return binding;
587 }
588 if (value.empty())
589 {
590 printf("%%SDL-W-HOTKEY: %s has an empty binding; action disabled.\n",
591 config_name);
592 binding.display = "none";
593 return binding;
594 }
595
596 std::string key_name = value;
597 unsigned modifiers = 0;
598 for (;;)
599 {
600 size_t separator = key_name.find('+');
601 if (separator == std::string::npos)
602 break;
603
604 std::string token = trim_hotkey_text(key_name.substr(0, separator));
605 unsigned modifier = 0;
606 if (!parse_hotkey_modifier(token, &modifier))
607 break;
608 if (modifiers & modifier)
609 {
610 printf("%%SDL-W-HOTKEY: %s has invalid binding \"%s\"; "
611 "action disabled.\n", config_name, value.c_str());
612 binding.display = "none";
613 return binding;
614 }
615 modifiers |= modifier;
616 key_name = trim_hotkey_text(key_name.substr(separator + 1));
617 }
618
619 key_name = normalize_hotkey_key_name(trim_hotkey_text(key_name));
620 if (key_name.empty())
621 {
622 printf("%%SDL-W-HOTKEY: %s has invalid binding \"%s\"; "
623 "action disabled.\n", config_name, value.c_str());
624 binding.display = "none";
625 return binding;
626 }
627
628 SDL_ClearError();
629 SDL_Keycode key = SDL_GetKeyFromName(key_name.c_str());
630 if (key == SDLK_UNKNOWN || is_hotkey_modifier_key(key))
631 {
632 printf("%%SDL-W-HOTKEY: %s has invalid binding \"%s\"; "
633 "action disabled.\n", config_name, value.c_str());
634 binding.display = "none";
635 return binding;
636 }
637
638 binding.key = key;
639 binding.modifiers = modifiers;
640 binding.enabled = true;
641 if (modifiers & SDL_HOTKEY_MOD_CTRL) binding.display += "Ctrl+";
642 if (modifiers & SDL_HOTKEY_MOD_ALT) binding.display += "Alt+";
643 if (modifiers & SDL_HOTKEY_MOD_SHIFT) binding.display += "Shift+";
644 if (modifiers & SDL_HOTKEY_MOD_GUI) binding.display += "GUI+";
645 const char* display_key = SDL_GetKeyName(key);
646 binding.display += (display_key && *display_key) ? display_key : key_name;
647 return binding;
648}

673bx_sdl_gui_c& sdl_application::display_for_output(const std::string& device_path,
674 unsigned output_id)
675{
676 if (shutdown_started)
677 FAILURE(SDL, "Cannot bind a display after application shutdown");
678 if (device_path.empty())
679 FAILURE(Configuration, "Display output requires a device identity");
680 const auto key = std::make_pair(device_path, output_id);
681 const auto existing = display_outputs.find(key);
682 if (existing != display_outputs.end())
683 return *existing->second;
684
685 // Reserve the identity first; a failed display allocation must not consume it.
686 const auto entry = display_outputs.emplace(key, nullptr).first;
687 try
688 {
689 if (display_outputs.size() == 1 && !displays.empty())
690 entry->second = displays.front().get();
691 else
692 entry->second = &create_display();
693 }

821{

825 const auto entry = sdl_window_owners.find(registered_window_id);
826 if (entry != sdl_window_owners.end() && entry->second == this)
827 sdl_window_owners.erase(entry);
828 // Explicit exit releases native capture first.
829 if (sdl_mouse_input.capture_window_id == registered_window_id)
830 {
831 sdl_mouse_input.capture_window_id = 0;
832 sdl_mouse_input.captured = false;

1213 const SDL_KeyboardEvent& event, const sdl_hotkey_binding& binding,
1214 bool release_guest_modifiers)
1215{

1217 event.scancode < SDL_SCANCODE_COUNT)
1218 sdl_keyboard_input.swallowed_hotkey_releases[event.scancode] = true;
1219
1220 if (!release_guest_modifiers)
1221 return;

1224 auto suppress_modifier = [keys](SDL_Scancode left,
1225 SDL_Scancode right)
1226 {
1227 SDL_Scancode scancodes[2] = { left, right };
1228 for (int i = 0; i < 2; i++)
1229 {
1230 SDL_Scancode scancode = scancodes[i];
1231 if ((keys && keys[scancode]) || sdl_keyboard_input.guest_key_pressed[scancode])
1232 {
1233 sdl_keyboard_input.swallowed_hotkey_releases[scancode] = true;
1234 release_guest_key(scancode);
1235 }
1236 }
1237 };
1238
1239 if (binding.modifiers & SDL_HOTKEY_MOD_CTRL)
1240 suppress_modifier(SDL_SCANCODE_LCTRL, SDL_SCANCODE_RCTRL);
1241 if (binding.modifiers & SDL_HOTKEY_MOD_ALT)
1242 suppress_modifier(SDL_SCANCODE_LALT, SDL_SCANCODE_RALT);
1243 if (binding.modifiers & SDL_HOTKEY_MOD_SHIFT)
1244 suppress_modifier(SDL_SCANCODE_LSHIFT, SDL_SCANCODE_RSHIFT);

1264void sdl_event_dispatcher::poll()
1265{
1266 if (sdl_guest_input_paused || sdl_poll_in_progress)
1267 return;
1268 // PollEvent also runs other workers' callbacks. One drain owns the queue,
1269 // including the single deferred event, until this call returns.
1270 struct PollGuard
1271 {
1272 PollGuard() { sdl_poll_in_progress = true; }
1273 ~PollGuard() { sdl_poll_in_progress = false; }
1274 } guard;
1275 sdl_media_pump();
1276 if (sdl_guest_input_paused)
1277 return;
1278 if (sdl_media_input_active())
1279 reconcile_input_focus();
1280 while (!sdl_guest_input_paused)
1281 {
1282 SDL_Event event;
1283 if (sdl_has_deferred_event)
1284 {
1285 event = sdl_deferred_event;
1286 sdl_has_deferred_event = false;
1287 }
1288 else
1289 {
1290 const bool available = SDL_PollEvent(&event);
1291 // PollEvent can run a nested main-thread pause callback.
1292 if (sdl_guest_input_paused)
1293 {
1294 if (available)
1295 defer_event(event);
1296 return;
1297 }
1298 if (!available)
1299 break;
1300 }
1301 dispatch(event);
1302 }
1303 if (sdl_guest_input_paused)
1304 return;
1305
1306 // Focus queries describe SDL's latest state, not each queued event's state.
1307 reconcile_input_focus();
1308 reconcile_input_releases();
1309
1310 // Native popups can intercept releases. Reconcile against SDL's current
1311 // physical keyboard state so a later normal press is never swallowed.
1312 reconcile_hotkey_release_state();
1313}

1315void sdl_event_dispatcher::defer_event(const SDL_Event& event)
1316{
1317 // SDL owns pointer payloads until a later pump. Retain only the scalar
1318 // event families consumed by this dispatcher and the media UI; other
1319 // event types are ignored by both and need only their common fields.
1320 sdl_deferred_event = {};
1321 sdl_deferred_event.common = event.common;
1322 if (event.type >= SDL_EVENT_WINDOW_FIRST && event.type <= SDL_EVENT_WINDOW_LAST)
1323 sdl_deferred_event.window = event.window;
1324 else switch (event.type)
1325 {
1326 case SDL_EVENT_KEY_DOWN:
1327 case SDL_EVENT_KEY_UP:
1328 sdl_deferred_event.key = event.key;
1329 break;
1330 case SDL_EVENT_MOUSE_MOTION:
1331 sdl_deferred_event.motion = event.motion;
1332 break;
1333 case SDL_EVENT_MOUSE_BUTTON_DOWN:
1334 case SDL_EVENT_MOUSE_BUTTON_UP:
1335 sdl_deferred_event.button = event.button;
1336 break;
1337 case SDL_EVENT_MOUSE_WHEEL:
1338 sdl_deferred_event.wheel = event.wheel;
1339 break;
1340 default:
1341 break;
1342 }
1343 sdl_has_deferred_event = true;
1344}

1346void sdl_event_dispatcher::dispatch(const SDL_Event& event)
1347{
1348 if (sdl_guest_input_paused)
1349 return;
1350 // GUI hotkeys consume their trigger key and, for actions that release
1351 // guest modifiers, the corresponding physical modifier releases. Keep
1352 // this ahead of the media popup so a remapped media hotkey also closes it.
1353 if (event.type == SDL_EVENT_KEY_UP)
1354 {
1355 // The media popup has its own release tracker for dialog keys. Let it
1356 // clear that state before the global tracker consumes a release shared
1357 // with the media-toggle chord.
1358 bool media_release_handled = sdl_media_handle_event(&event);
1359 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1360 event.key.scancode < SDL_SCANCODE_COUNT &&
1361 (sdl_keyboard_input.swallowed_hotkey_releases[event.key.scancode] ||
1362 sdl_keyboard_input.reconciled_key_releases[event.key.scancode]))
1363 {
1364 sdl_keyboard_input.swallowed_hotkey_releases[event.key.scancode] = false;
1365 sdl_keyboard_input.fullscreen_hotkey_pressed[event.key.scancode] = false;
1366 sdl_keyboard_input.reconciled_key_releases[event.key.scancode] = false;
1367 return;
1368 }
1369 if (media_release_handled)
1370 return;
1371 }
1372
1373 if (event.type == SDL_EVENT_KEY_DOWN)
1374 {
1375 bx_sdl_gui_c* owner = bx_sdl_gui_c::find_window_owner(event.key.windowID);
1376 // The popup's toggle binding belongs to the display that opened it.
1377 if (!owner)
1378 owner = bx_sdl_gui_c::find_window_owner(
1379 sdl_media_parent_window_id(event.key.windowID));
1380 if (owner && owner->hotkey_media.matches(event.key))
1381 {
1382 if (!event.key.repeat)
1383 {
1384 suppress_hotkey_releases(event.key, owner->hotkey_media, true);
1385 release_input_session();
1386 sdl_select_media(owner->sdl_window);
1387 }
1388 return;
1389 }
1390 }
1391
1392 if (event.type != SDL_EVENT_KEY_UP &&
1393 sdl_media_handle_event(&event))
1394 return;
1395
1396 // Update the requested window, regardless of which display drained the
1397 // queue. Media sees the event once, before lookup; retired IDs have no owner.
1398 // Delivery and explicit retirement are serialized on this SDL main thread.
1399 switch (event.type)
1400 {
1401 case SDL_EVENT_WINDOW_CLOSE_REQUESTED:
1402 if (event.window.windowID &&
1403 bx_sdl_gui_c::find_window_owner(event.window.windowID) &&
1404 theSystem)
1405 theSystem->RequestShutdown();
1406 return;
1407 case SDL_EVENT_QUIT:
1408 if (theSystem)
1409 theSystem->RequestShutdown();
1410 return;
1411 case SDL_EVENT_KEY_DOWN:
1412 case SDL_EVENT_KEY_UP:
1413 {
1414 if (sdl_media_input_active())
1415 return;
1416 bx_sdl_gui_c* owner = bx_sdl_gui_c::find_window_owner(event.key.windowID);
1417 if (owner)
1418 owner->handle_keyboard_event_impl(event);
1419 return;
1420 }
1421 case SDL_EVENT_WINDOW_FOCUS_GAINED:
1422 case SDL_EVENT_WINDOW_FOCUS_LOST:
1423 {
1424 bx_sdl_gui_c* owner = bx_sdl_gui_c::find_window_owner(event.window.windowID);
1425 if (owner)
1426 owner->handle_focus_event_impl(event);
1427 return;
1428 }
1429 case SDL_EVENT_WINDOW_EXPOSED:
1430 case SDL_EVENT_WINDOW_RESTORED:
1431 case SDL_EVENT_WINDOW_DISPLAY_SCALE_CHANGED:
1432 case SDL_EVENT_WINDOW_MOUSE_ENTER:
1433 case SDL_EVENT_WINDOW_MOUSE_LEAVE:
1434 case SDL_EVENT_WINDOW_MOVED:
1435 case SDL_EVENT_WINDOW_RESIZED:
1436 case SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED:
1437 case SDL_EVENT_WINDOW_ENTER_FULLSCREEN:
1438 case SDL_EVENT_WINDOW_LEAVE_FULLSCREEN:
1439 {
1440 bx_sdl_gui_c* owner = bx_sdl_gui_c::find_window_owner(event.window.windowID);
1441 if (owner)
1442 owner->handle_display_event_impl(event);
1443 return;
1444 }
1445 case SDL_EVENT_MOUSE_MOTION:
1446 case SDL_EVENT_MOUSE_BUTTON_DOWN:
1447 case SDL_EVENT_MOUSE_BUTTON_UP:
1448 case SDL_EVENT_MOUSE_WHEEL:
1449 {
1450 if (sdl_media_input_active())
1451 return;
1452 SDL_WindowID event_window;
1453 if (event.type == SDL_EVENT_MOUSE_MOTION)
1454 event_window = event.motion.windowID;
1455 else if (event.type == SDL_EVENT_MOUSE_WHEEL)
1456 event_window = event.wheel.windowID;
1457 else
1458 event_window = event.button.windowID;
1459 // The host capture source supplies one guest mouse across all outputs.
1460 if (!event_window || (sdl_mouse_input.captured &&
1461 event_window != sdl_mouse_input.capture_window_id))
1462 return;
1463 bx_sdl_gui_c* owner = bx_sdl_gui_c::find_window_owner(event_window);
1464 if (owner)
1465 owner->handle_mouse_event_impl(event);
1466 return;
1467 }
1468 default:
1469 break;
1470 }
1471

1501{

1517void sdl_event_dispatcher::reconcile_input_releases()
1518{
1519 // Repair releases, never new presses.
1520 const bool* keys = SDL_GetKeyboardState(NULL);
1521 if (keys)
1522 {
1523 for (int i = 1; i < SDL_SCANCODE_COUNT; i++)
1524 {
1525 if (sdl_keyboard_input.guest_key_pressed[i] && !keys[i])
1526 {
1527 release_guest_key((SDL_Scancode)i);
1528 // A later queued key-up must not use another display's mapping.
1529 sdl_keyboard_input.reconciled_key_releases[i] = true;
1530 }
1531 }
1532 }
1533
1534 if (!sdl_mouse_input.captured || !sdl_mouse_input.buttons)
1535 return;
1536 const SDL_MouseButtonFlags state = SDL_GetMouseState(NULL, NULL);
1537 int buttons = 0;
1538 if (state & SDL_BUTTON_LMASK) buttons |= 0x01;
1539 if (state & SDL_BUTTON_RMASK) buttons |= 0x02;
1540 if (state & SDL_BUTTON_MMASK) buttons |= 0x04;
1541 buttons &= sdl_mouse_input.buttons;
1542 if (buttons != sdl_mouse_input.buttons)
1543 {
1544 sdl_mouse_input.buttons = buttons;
1545 sdl_event_dispatcher::send_guest_mouse(0, 0, 0, buttons);
1546 }

1557void bx_sdl_gui_c::handle_keyboard_event_impl(const SDL_Event& event)
1558{
1559 u32 key_event;
1560 switch (event.type)
1561 {
1562 case SDL_EVENT_KEY_DOWN:
1563 // A queued repeat cannot restart a reconciled press or turn a held
1564 // fullscreen trigger into guest input/another shortcut after Alt is up.
1565 if (event.key.repeat && event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1566 event.key.scancode < SDL_SCANCODE_COUNT &&
1567 (sdl_keyboard_input.reconciled_key_releases[event.key.scancode] ||
1568 sdl_keyboard_input.fullscreen_hotkey_pressed[event.key.scancode]))
1569 break;
1570 if (hotkey_ctrl_alt_delete.matches(event.key))
1571 {
1572 if (!event.key.repeat)
1573 {
1574 sdl_event_dispatcher::suppress_hotkey_releases(event.key,
1575 hotkey_ctrl_alt_delete, true);
1576 sdl_event_dispatcher::send_guest_ctrl_alt_delete();
1577 }
1578 break;
1579 }
1580
1581 if (hotkey_reset_window.matches(event.key))
1582 {
1583 if (!event.key.repeat)
1584 {
1585 sdl_event_dispatcher::suppress_hotkey_releases(event.key,
1586 hotkey_reset_window, false);
1587 reset_window_size();
1588 }
1589 break;
1590 }
1591
1592 // Runtime scale adjustment remains gated by video.scale_change_enable.
1593 if (vid_scale_change_enable && hotkey_scale_up.matches(event.key))
1594 {
1595 if (!event.key.repeat)
1596 sdl_event_dispatcher::suppress_hotkey_releases(event.key,
1597 hotkey_scale_up, false);
1598 adjust_window_scale(+1);
1599 break;
1600 }
1601 if (vid_scale_change_enable && hotkey_scale_down.matches(event.key))
1602 {
1603 if (!event.key.repeat)
1604 sdl_event_dispatcher::suppress_hotkey_releases(event.key,
1605 hotkey_scale_down, false);
1606 adjust_window_scale(-1);
1607 break;
1608 }
1609
1610 if (hotkey_mouse_capture.matches(event.key))
1611 {
1612 if (!event.key.repeat)
1613 {
1614 sdl_event_dispatcher::suppress_hotkey_releases(event.key,
1615 hotkey_mouse_capture, true);
1616 set_mouse_capture_impl(!sdl_mouse_input.captured);
1617 }
1618 break;
1619 }
1620
1621 // Match Win32's Alt+Enter desktop-fullscreen shortcut, after the
1622 // configurable bindings so an explicit user binding takes precedence.
1623 if ((event.key.key == SDLK_RETURN || event.key.key == SDLK_KP_ENTER) &&
1624 (event.key.mod & SDL_KMOD_ALT))
1625 {
1626 if (!event.key.repeat)
1627 {
1628 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1629 event.key.scancode < SDL_SCANCODE_COUNT)
1630 {
1631 sdl_event_dispatcher::release_guest_key(event.key.scancode);
1632 sdl_keyboard_input.swallowed_hotkey_releases[event.key.scancode] = true;
1633 sdl_keyboard_input.fullscreen_hotkey_pressed[event.key.scancode] = true;
1634 }
1635 const bool fullscreen = fullscreen_transition_pending ? fullscreen_requested :
1636 (SDL_GetWindowFlags(sdl_window) & SDL_WINDOW_FULLSCREEN) != 0;
1637 set_fullscreen(!fullscreen);
1638 }
1639 break;
1640 }
1641
1642 // convert sym -> bochs code
1643 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1644 event.key.scancode < SDL_SCANCODE_COUNT &&
1645 sdl_keyboard_input.guest_key_pressed[event.key.scancode])
1646 {
1647 // A held key keeps its mapping across host focus changes and repeats.
1648 key_event = sdl_keyboard_input.guest_key_by_scancode[event.key.scancode];
1649 }
1650 else if (!myCfg->get_bool_value("keyboard.use_mapping", false))
1651 {
1652 key_event = sdl_scan_to_bx_key(event.key.scancode);
1653 }
1654 else
1655 {
1656 /* use mapping */
1657 BXKeyEntry* entry = bx_keymap->findHostKey(event.key.key);
1658 if (!entry)
1659 {
1660 BX_ERROR(("host key 0x%x not mapped!",
1661 (unsigned)event.key.key));
1662 break;
1663 }
1664 key_event = entry->baseKey;
1665 }
1666
1667 if (key_event == BX_KEY_UNHANDLED)
1668 break;
1669
1670 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1671 event.key.scancode < SDL_SCANCODE_COUNT &&
1672 !sdl_keyboard_input.guest_key_pressed[event.key.scancode])
1673 {
1674 sdl_keyboard_input.guest_key_by_scancode[event.key.scancode] = key_event;
1675 sdl_keyboard_input.guest_key_pressed[event.key.scancode] = true;
1676 sdl_keyboard_input.reconciled_key_releases[event.key.scancode] = false;
1677 }
1678 sdl_event_dispatcher::send_guest_key(key_event);
1679
1680 // Locks: generate immediate press+release pair
1681 if ((key_event == BX_KEY_NUM_LOCK) || (key_event == BX_KEY_CAPS_LOCK))
1682 {
1683 sdl_event_dispatcher::send_guest_key(key_event | BX_KEY_RELEASED);
1684 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1685 event.key.scancode < SDL_SCANCODE_COUNT)
1686 sdl_keyboard_input.guest_key_pressed[event.key.scancode] = false;
1687 }
1688 break;
1689
1690 case SDL_EVENT_KEY_UP:
1691 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1692 event.key.scancode < SDL_SCANCODE_COUNT &&
1693 sdl_keyboard_input.guest_key_pressed[event.key.scancode])
1694 {
1695 key_event = sdl_keyboard_input.guest_key_by_scancode[event.key.scancode];
1696 }
1697 else if (!myCfg->get_bool_value("keyboard.use_mapping", false))
1698 {
1699 key_event = sdl_scan_to_bx_key(event.key.scancode);
1700 }
1701 else
1702 {
1703 BXKeyEntry* entry = bx_keymap->findHostKey(event.key.key);
1704 if (!entry)
1705 {
1706 BX_ERROR(("host key 0x%x not mapped!",
1707 (unsigned)event.key.key));
1708 break;
1709 }
1710 key_event = entry->baseKey;
1711 }
1712
1713 if (key_event == BX_KEY_UNHANDLED)
1714 break;
1715
1716 if ((key_event == BX_KEY_NUM_LOCK) || (key_event == BX_KEY_CAPS_LOCK))
1717 {
1718 sdl_event_dispatcher::send_guest_key(key_event);
1719 }
1720
1721 sdl_event_dispatcher::send_guest_key(key_event | BX_KEY_RELEASED);
1722 if (event.key.scancode > SDL_SCANCODE_UNKNOWN &&
1723 event.key.scancode < SDL_SCANCODE_COUNT)
1724 sdl_keyboard_input.guest_key_pressed[event.key.scancode] = false;
1725 break;
1726 }

1729void bx_sdl_gui_c::handle_mouse_event_impl(const SDL_Event& event)
1730{
1731 if (!sdl_window)
1732 return;
1733 switch (event.type)
1734 {
1735 case SDL_EVENT_MOUSE_MOTION:
1736 if (sdl_mouse_input.captured)
1737 {
1738 double rel_x = (double)event.motion.xrel;
1739 double rel_y = (double)event.motion.yrel;
1740 if (mouse_absolute)
1741 {
1742 const double x = (double)event.motion.x;
1743 const double y = (double)event.motion.y;
1744 if (!mouse_position_valid)
1745 {
1746 mouse_last_x = x;
1747 mouse_last_y = y;
1748 mouse_position_valid = true;
1749 break;
1750 }
1751 rel_x = x - mouse_last_x;
1752 rel_y = y - mouse_last_y;
1753 mouse_last_x = x;
1754 mouse_last_y = y;
1755 }
1756
1757 // PS/2 mouse Y is positive-up, SDL is positive-down; hence the
1758 // baseline Y negation. invert_x/y flip on top of that.
1759 double mx = rel_x * mouse_speed;
1760 double my = -rel_y * mouse_speed;
1761 sdl_mouse_input.remainder_x += mouse_invert_x ? -mx : mx;
1762 sdl_mouse_input.remainder_y += mouse_invert_y ? -my : my;
1763
1764 int dx = (int)sdl_mouse_input.remainder_x;
1765 int dy = (int)sdl_mouse_input.remainder_y;
1766
1767 if (dx != 0 || dy != 0)
1768 {
1769 sdl_mouse_input.remainder_x -= dx;
1770 sdl_mouse_input.remainder_y -= dy;
1771 sdl_event_dispatcher::send_guest_mouse(dx, dy, 0, sdl_mouse_input.buttons);
1772 }
1773 }
1774 break;
1775
1776 case SDL_EVENT_MOUSE_BUTTON_DOWN:
1777 case SDL_EVENT_MOUSE_BUTTON_UP:
1778 {
1779 if (!sdl_mouse_input.captured)
1780 {
1781 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN
1782 && event.button.button == SDL_BUTTON_LEFT)
1783 {
1784 set_mouse_capture_impl(true);
1785 }
1786 break;
1787 }
1788
1789 int bitmask = 0;
1790 switch (event.button.button)
1791 {
1792 case SDL_BUTTON_LEFT: bitmask = 0x01; break;
1793 case SDL_BUTTON_RIGHT: bitmask = 0x02; break;
1794 case SDL_BUTTON_MIDDLE: bitmask = 0x04; break;
1795 default: break;
1796 }
1797
1798 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN)
1799 sdl_mouse_input.buttons |= bitmask;
1800 else
1801 sdl_mouse_input.buttons &= ~bitmask;
1802
1803 sdl_event_dispatcher::send_guest_mouse(0, 0, 0, sdl_mouse_input.buttons);
1804 break;
1805 }
1806
1807 case SDL_EVENT_MOUSE_WHEEL:
1808 if (sdl_mouse_input.captured)
1809 {
1810 float wy = event.wheel.y; // SDL3: float; +y = away from user (scroll up)
1811 if (event.wheel.direction == SDL_MOUSEWHEEL_FLIPPED)
1812 wy = -wy;
1813 int dz = (int)wy;
1814 if (dz != 0)
1815 sdl_event_dispatcher::send_guest_mouse(0, 0, dz, sdl_mouse_input.buttons);
1816 }
1817 break;
1818 }
1819}

1821void bx_sdl_gui_c::handle_display_event_impl(const SDL_Event& event)
1822{
1823 if (!sdl_window)
1824 return;
1825 switch (event.type)
1826 {
1827 case SDL_EVENT_WINDOW_EXPOSED:
1828 redraw_impl();
1829 break;
1830 case SDL_EVENT_WINDOW_RESTORED:
1831 // Preserve manual geometry and the existing texture on an ordinary
1832 // restore. Only a deferred guest/scale change requests a new size.
1833 reset_absolute_mouse_position();
1834 resize_window_to_guest();
1835 redraw_impl();
1836 break;
1837 case SDL_EVENT_WINDOW_DISPLAY_SCALE_CHANGED:
1838 if (res_x > 0 && res_y > 0)
1839 dimension_update(res_x, res_y);
1840 break;
1841 case SDL_EVENT_WINDOW_ENTER_FULLSCREEN:
1842 case SDL_EVENT_WINDOW_LEAVE_FULLSCREEN:
1843 {
1844 // Events can be queued behind a newer fullscreen request. Consult the
1845 // current flags before releasing its resize barrier.
1846 const bool fullscreen = (SDL_GetWindowFlags(sdl_window) & SDL_WINDOW_FULLSCREEN) != 0;
1847 if (!fullscreen_transition_pending)
1848 fullscreen_requested = fullscreen;
1849 else if ((event.type == SDL_EVENT_WINDOW_ENTER_FULLSCREEN) == fullscreen_requested &&
1850 fullscreen == fullscreen_requested)
1851 fullscreen_transition_pending = false;
1852 resize_window_to_guest();
1853 reset_absolute_mouse_position();
1854 break;
1855 }
1856 case SDL_EVENT_WINDOW_MOUSE_ENTER:
1857 case SDL_EVENT_WINDOW_MOUSE_LEAVE:
