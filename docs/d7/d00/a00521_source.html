191 u8 get_status_byte()
192 {
193
194 // top bit is 0 , bit 6 is 1 if remote mode.
195 u8 ret = (u8)((mode == MOUSE_MODE_REMOTE) ? 0x40 : 0);
196 ret |= (enable << 5);
197 ret |= (scaling == 1) ? 0 : (1 << 4);
198 ret |= ((button_status & 0x1) << 2);
199 ret |= ((button_status & 0x2) << 0);
200 return ret;
201 }
202
203 u8 get_resolution_byte()
204 {
205 u8 ret = 0;
206
207 switch (resolution_cpmm)
208 {
209 case 1: ret = 0; break;
210 case 2: ret = 1; break;
211 case 4: ret = 2; break;
212 case 8: ret = 3; break;
213 default: FAILURE(NotImplemented, "mouse: invalid resolution_cpmm");
214 };
215 return ret;
216 }
