152inline u64 sext_u64_8(u64 a)
153{
154#if defined(ES40_BIG_ENDIAN)
155 return(((a)&U64(0x0000000000000080)) ? ((a) | U64(0xffffffffffffff00))
156 : ((a)&U64(0x00000000000000ff)));
157#else // little endian optimization
158 auto aa = static_cast<s64>(*reinterpret_cast<s8*>(&a));
159 return *reinterpret_cast<u64*>(&aa);
160#endif
161}
