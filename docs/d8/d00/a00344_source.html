38// PAL[7], ABX[6], IBX[5], INDEX[4:0] (HRM Table 4-6).
39constexpr Ev4IPRSelector ev4_ipr_selector(std::uint32_t instruction)
40{
41 return {instruction & 0x1f, instruction & 0xe0};
42}
43
44constexpr bool ev4_ipr_ignored(Ev4IPRSelector selector, bool pal_mode)
45{
46 // HRM 4.6(10); mixed-bank accesses stay with their handlers.
47 return selector.banks == 0 || (!pal_mode && selector.banks == 0x20
48 && (selector.index == 0 || selector.index == 1 || selector.index == 3));
49}

51// Unimplemented registers stay with the interpreter.
52constexpr AlphaIPRReadDescriptor ev4_describe_ipr_read(std::uint32_t instruction,
53 bool pal_mode)
54{
55 if ((instruction >> 26) != 0x19)
56 return {AlphaIPRReadKind::Unsupported, false, false};
57 const auto selector = ev4_ipr_selector(instruction);
58 if (ev4_ipr_ignored(selector, pal_mode))
59 return {AlphaIPRReadKind::Noop, false, true};
60 if (selector.banks == 0x80) // PAL_TEMP only.
61 return {AlphaIPRReadKind::Helper, false, true};
62 if (selector.banks == 0x20 && selector.index == 4) // EXC_ADDR.
63 return {AlphaIPRReadKind::ExceptionAddress, false, true};
64 if (selector.banks == 0x20 && selector.index == 9) // PS.
65 return {AlphaIPRReadKind::Helper, false, true};
66 if (selector.banks == 0x20 && selector.index == 10) // EXC_SUM shifts MSK, even to R31.
67 return {AlphaIPRReadKind::Helper, false, false};
68 if (selector.banks == 0x20 && selector.index == 11) // PAL_BASE.
69 return {AlphaIPRReadKind::PalBase, false, true};
70 return {AlphaIPRReadKind::Unsupported, false, false};
71}

73constexpr AlphaIPRWriteKind ev4_describe_ipr_write(std::uint32_t instruction,
74 bool pal_mode)
75{
76 if ((instruction >> 26) != 0x1d)
77 return AlphaIPRWriteKind::Interpreter;
78 const auto selector = ev4_ipr_selector(instruction);
79 if (ev4_ipr_ignored(selector, pal_mode))
80 return AlphaIPRWriteKind::Noop;
81 if (selector.banks == 0x80) // PAL_TEMP only.
82 return AlphaIPRWriteKind::Helper;
83 if (selector.banks == 0x20 && selector.index == 9) // PS.
84 return AlphaIPRWriteKind::Helper;
85 if (selector.banks == 0x20 && selector.index == 10) // EXC_SUM.
86 return AlphaIPRWriteKind::Helper;
87 if (selector.banks == 0x20 && selector.index == 11) // PAL_BASE.
88 return AlphaIPRWriteKind::Helper;
89 if (selector.banks == 0x40 && selector.index == 15) // ALT_MODE.
90 return AlphaIPRWriteKind::Helper;
91 if (selector.banks == 0x40 && selector.index == 16) // CC.
92 return AlphaIPRWriteKind::Helper;
93 return AlphaIPRWriteKind::Interpreter;
94}

96// PS reads use bits 34/1; writes use bits 4:3 (HRM 5.2.11).
97constexpr std::uint64_t ev4_read_processor_status(std::uint32_t mode)
98{
99 return (std::uint64_t(mode & 1) << 1) | (std::uint64_t(mode & 2) << 33);
100}

102constexpr std::uint32_t ev4_decode_processor_status(std::uint64_t value)
103{
104 return static_cast<std::uint32_t>((value >> 3) & 3);
105}
106
107// ALT_MODE selects the alternate mode with bits 4:3 (HRM 5.3.12).
108constexpr std::uint32_t ev4_decode_alternate_mode(std::uint64_t value)
109{
110 return static_cast<std::uint32_t>((value >> 3) & 3);
111}

113// CC writes update only the offset in bits 63:32 (HRM 5.3.13).
114constexpr std::uint32_t ev4_decode_cycle_counter_offset(std::uint64_t value)
115{
116 return static_cast<std::uint32_t>(value >> 32);

125// CC_CTL: counter[31:0], enable[32]; bits 3:0 must be zero (HRM 5.3.14).
126constexpr Ev4CycleCounterControl ev4_decode_cycle_counter_control(std::uint64_t value)
127{
128 return {static_cast<std::uint32_t>(value), ((value >> 32) & 1) != 0};
129}

131// PAL_BASE holds bits 33:14 (HRM 5.2.13).
132constexpr std::uint64_t ev4_normalize_pal_base(std::uint64_t value)
