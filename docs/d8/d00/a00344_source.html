62struct rectangle
63{

66
67 rectangle() = default;
68 rectangle(int x0, int x1, int y0, int y1)
69 : min_x(x0), max_x(x1), min_y(y0), max_y(y1) {
70 }
71
72 int width() const { return max_x - min_x + 1; }
73 int height() const { return max_y - min_y + 1; }

82class bitmap_rgb32
83{
84public:
85 bitmap_rgb32() = default;
86
87 void allocate(int w, int h)
88 {
89 m_width = w;
90 m_height = h;
91 m_pixels.resize((size_t)w * h, 0);
92 }
93
94 void resize(int w, int h) { allocate(w, h); }
95
96 int width() const { return m_width; }
97 int height() const { return m_height; }
98 bool valid() const { return m_width > 0 && m_height > 0; }
99
100 // pix(y, x) — returns a reference to pixel (x, y).
101 uint32_t& pix(int y, int x = 0)
102 {
103 return m_pixels[(size_t)y * m_width + x];
104 }
105
106 uint32_t* rowptr(int y)
107 {
108 return m_pixels.data() + (size_t)y * m_width;
109 }
110 const uint32_t* rowptr(int y) const
111 {
112 return m_pixels.data() + (size_t)y * m_width;
113 }
114
115 void fill(uint32_t color, const rectangle& clip)
116 {
117 int y0 = (std::max)(clip.min_y, 0);
118 int y1 = (std::min)(clip.max_y, m_height - 1);
119 int x0 = (std::max)(clip.min_x, 0);
120 int x1 = (std::min)(clip.max_x, m_width - 1);
121 for (int y = y0; y <= y1; y++)
122 {
123 uint32_t* row = &pix(y);
124 for (int x = x0; x <= x1; x++)
125 row[x] = color;
126 }
127 }
128
129 // MAME text cursor: plot a horizontal box
130 void plot_box(int x, int y, int w, int h, uint32_t color)
131 {
132 for (int dy = 0; dy < h; dy++)
133 {
134 if (y + dy < 0 || y + dy >= m_height) continue;
135 uint32_t* row = &pix(y + dy);
136 for (int dx = 0; dx < w; dx++)
137 {
138 if (x + dx >= 0 && x + dx < m_width)
139 row[x + dx] = color;
140 }
141 }
142 }
143
144 // Direct pixel pointer access (whole buffer)
145 const uint32_t* raw() const { return m_pixels.data(); }
146 uint32_t* raw() { return m_pixels.data(); }
147
148 // Clip rect covering the entire bitmap
149 rectangle cliprect() const { return rectangle(0, m_width - 1, 0, m_height - 1); }
150
151private:
152 int m_width = 0;
153 int m_height = 0;
154 std::vector<uint32_t> m_pixels;
155};
