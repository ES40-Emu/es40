88// Big-endian byte buffers; no host byte-order or alignment assumptions.
89inline u16 get_be16(const u8* p)
90{
91 return u16((u16(p[0]) << 8) | p[1]);
92}
93
94inline u32 get_be32(const u8* p)
95{
96 return (u32(p[0]) << 24) | (u32(p[1]) << 16) |
97 (u32(p[2]) << 8) | p[3];
98}
99
100inline u64 get_be64(const u8* p)
101{
102 return (u64(get_be32(p)) << 32) | get_be32(p + 4);
103}
104
105inline void put_be16(u8* p, u16 value)
106{
107 p[0] = u8(value >> 8);
108 p[1] = u8(value);
109}
110
111inline void put_be32(u8* p, u32 value)
112{
113 p[0] = u8(value >> 24);
114 p[1] = u8(value >> 16);
115 p[2] = u8(value >> 8);
116 p[3] = u8(value);
117}
118
119inline void put_be64(u8* p, u64 value)
120{
121 put_be32(p, u32(value >> 32));
122 put_be32(p + 4, u32(value));
123}

125#define swap_64(x) \
126 ( \
127 (((x) & U64(0x00000000000000ff)) << 56) | \
128 (((x) & U64(0x000000000000ff00)) << 40) | \
129 (((x) & U64(0x0000000000ff0000)) << 24) | \
130 (((x) & U64(0x00000000ff000000)) << 8) | \
131 (((x) & U64(0x000000ff00000000)) >> 8) | \
132 (((x) & U64(0x0000ff0000000000)) >> 24) | \
133 (((x) & U64(0x00ff000000000000)) >> 40) | \
134 (((x) & U64(0xff00000000000000)) >> 56) \
135 )
136#define swap_32(x) \
137 ( \
138 (((x) & 0x000000ff) << 24) | (((x) & 0x0000ff00) << 8) | \
139 (((x) & 0x00ff0000) >> 8) | (((x) & 0xff000000) >> 24) \
140 )
141#define swap_16(x) ((((x) & 0x00ff) << 8) | (((x) & 0xff00) >> 8))
142#define swap_8(x) ((x) & 0xff)
143#if defined(ES40_BIG_ENDIAN)
144#define endian_64(x) swap_64(x)

153#endif // defined(ES40_BIG_ENDIAN)
154inline u64 endian_bits(u64 x, int numbits)
155{
156 switch (numbits)
157 {
158 case 64: return endian_64(x);
159 case 32: return endian_32(x);
160 case 16: return endian_16(x);
161 case 8: return endian_8(x);
162 default: FAILURE(InvalidArgument, "Weird numbits in endian_bits");
163 }
164}
