145static u32 eth_crc32(u32 crc, const void* vbuf, int len)
146{
147 const u32 mask = 0xFFFFFFFF;
148 const unsigned char* buf = (const unsigned char*)vbuf;
149
150 crc ^= mask;
151 while (0 != len--)
152 crc = (crc >> 8) ^ crcTable[(crc ^ (*buf++)) & 0xFF];
153 return(crc ^ mask);
154}

156bool CPacketQueue::add_tail(const u8* packet_data, int packet_len,
157 bool calc_crc, bool need_crc)
158{
159 if ((cnt >= max) || (packet_len < 1) || (packet_len > ETH_MAX_PACKET_RAW))
160 {
161 const char* why = (cnt >= max) ? "queue full" : "illegal size";
162 dropped += 1;
163 printf("CPacketQueue(%s):add() packet lost (%s)! Size = %d, dropped = %d",
164 name, why, packet_len, dropped);
165 if (packet_len >= 12)
166 {
167 printf(".. dst: %02x-%02x-%02x-%02x-%02x-%02x ", packet_data[0],
168 packet_data[1], packet_data[2], packet_data[3], packet_data[4],
169 packet_data[5]);
170 printf(".. src: %02x-%02x-%02x-%02x-%02x-%02x", packet_data[6],
171 packet_data[7], packet_data[8], packet_data[9], packet_data[10],
172 packet_data[11]);
173 }
174 printf("\n");
175 return false;
176 }
177
178 tail += 1;
179 if (tail >= max)
180 {
181 tail = 0;
182 }
183
184 eth_packet* next = &packets[tail];
185 next->len = packet_len;
186 next->used = 0;
187 memcpy(next->frame, packet_data, packet_len); // copy packet data
188 if (need_crc)
189 { // If packet needs CRC
190 u32 crc = calc_crc ? eth_crc32(0, packet_data, packet_len) : 0; // recalculate crc if needed
191 u32 ncrc = htonl(crc); // put crc in network order
192 memcpy(&next->frame[packet_len], &ncrc, 4); // append CRC to packet
193 next->len += 4; // increase packet length
194 }
195
196 cnt += 1;
197 if (cnt > highwater)
198 {
199 highwater = cnt;
200 }
201
202 return true;
203}

205bool CPacketQueue::get_head(eth_packet& packet)
206{
207 if (cnt <= 0)
208 {
209 return false;
210 }
211
212 eth_packet* headp = &packets[head];
213 packet.len = headp->len;
214 packet.used = headp->used;
215 memcpy(packet.frame, headp->frame, sizeof(packet.frame));
216 head += 1;
217 if (head >= max)
218 {
219 head = 0;
220 }
221
222 cnt -= 1;
223 return true;
224}
