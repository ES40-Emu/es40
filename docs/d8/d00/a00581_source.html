69bool CNetworkSlirp::init(const char *devid_string, CConfigurator *cfg)
70{
71 devid_for_log = devid_string;
72
73 if (es40_slirp_available() == 0) {
74 printf("%s: slirp support not available in this build.\n", devid_string);
75 return false;
76 }
77
78 struct Es40SlirpConfig sc;
79 memset(&sc, 0, sizeof(sc));
80
81 /* The guest MAC is assigned by the NIC after the backend is created;
82 * the Rust side does not need it, so leave it zeroed. */
83 sc.network = cfg->get_text_value("network");
84 sc.gateway = cfg->get_text_value("gateway");
85 sc.dhcp_start = cfg->get_text_value("dhcp_start");
86 sc.dhcp_end = cfg->get_text_value("dhcp_end");
87 sc.dns = cfg->get_text_value("dns");
88 sc.lease_secs = (unsigned int)cfg->get_num_value("dhcp_lease", false, 3600);
89
90 ctx = es40_slirp_new(&sc);
91 if (!ctx) {
92 printf("%s: could not start the slirp backend (see messages above).\n",
93 devid_string);
94 return false;
95 }
96
97 const char *network = sc.network ? sc.network : "10.0.2.0/24";
98 const char *gateway = sc.gateway ? sc.gateway : "10.0.2.2";
99 printf("%s: slirp network %s, gateway %s\n", devid_string, network, gateway);
100
101 return true;
102}

111int CNetworkSlirp::receive(const u8 **data, int *len)
112{
113 if (!ctx)
114 return -1;
115 int n = es40_slirp_output(ctx, rx_buf, (int)sizeof(rx_buf));
116 if (n > 0) {
117 *data = rx_buf;
118 *len = n;
119 return 1;
120 }
121 return n == 0 ? 0 : -1;
122}
