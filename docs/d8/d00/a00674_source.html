326/* MPI messages contain little-endian byte fields. */
327static inline u16 read_le16(const u8* p)
328{
329 return (u16)((u16)p[0] | ((u16)p[1] << 8));
330}
331
332static inline u32 read_le32(const u8* p)
333{
334 return (u32)p[0] | ((u32)p[1] << 8) | ((u32)p[2] << 16) | ((u32)p[3] << 24);
335}
336
337static inline void write_le16(u8* p, u16 value)
338{
339 p[0] = (u8)value;
340 p[1] = (u8)(value >> 8);
341}
342
343static inline void write_le32(u8* p, u32 value)
344{
345 p[0] = (u8)value;
346 p[1] = (u8)(value >> 8);
347 p[2] = (u8)(value >> 16);
348 p[3] = (u8)(value >> 24);
349}

351/* Validate an x86 PCI option ROM and return its size. */
352static bool valid_option_rom(const u8* image, size_t available, u32* image_bytes)
353{
354 if (available < 0x34U || image[0] != 0x55U || image[1] != 0xAAU || image[2] == 0)
355 return false;
356
357 const u32 bytes = (u32)image[2] * 512U;
358 const u32 pcir = read_le16(&image[0x18]);
359 if (bytes > available || pcir > bytes - 0x18U ||
360 memcmp(&image[pcir], "PCIR", 4U) != 0 ||
361 read_le16(&image[pcir + 4U]) != LSI_PCI_VENDOR_ID ||
362 read_le16(&image[pcir + 6U]) != LSI_PCI_DEVICE_ID ||
363 read_le16(&image[pcir + 0x0AU]) < 0x18U ||
364 (read_le32(&image[pcir + 0x0CU]) >> 8) != LSI_PCI_CLASS_CODE ||
365 (u32)read_le16(&image[pcir + 0x10U]) * 512U != bytes ||
366 image[pcir + 0x14U] != 0)
367 return false;
368
369 u8 checksum = 0;
370 for (u32 i = 0; i < bytes; i++)
371 checksum = (u8)(checksum + image[i]);
372 if (checksum != 0)
373 return false;
374
375 if (image_bytes)
376 *image_bytes = bytes;
377 return true;
378}

380/*
381 * The supplied Fujitsu image is tagged for the 1030 B0 firmware family. The
382 * tag is accepted for flash seeding only; the emulated PCI device remains a
383 * 53C1020 (device 0030).
384 */
385static bool valid_ioc_firmware(const u8* image, size_t available, u32* image_bytes)
386{
387 const u16 product_id = available >= 0x24U ? read_le16(&image[0x22]) : 0U;
388 if (available < 0x34U || read_le32(&image[0x04]) != 0x5AEAA55AU ||
389 read_le32(&image[0x08]) != 0xA55AEAA5U ||
390 read_le32(&image[0x0C]) != 0x5AA55AEAU ||
391 read_le16(&image[0x20]) != LSI_PCI_VENDOR_ID ||
392 (product_id != LSI_FW_PRODUCT_1020_C0 &&
393 product_id != LSI_FW_PRODUCT_1030_B0_COMPAT))
394 return false;
395
396 const u32 bytes = read_le32(&image[0x2C]);
397 if (bytes < 0x34U || bytes > available || (bytes & 3U) != 0)
398 return false;
399
400 u32 checksum = 0;
401 for (u32 i = 0; i < bytes; i += 4U)
402 checksum += read_le32(&image[i]);
403 if (checksum != 0)
404 return false;
405
406 if (image_bytes)
407 *image_bytes = bytes;
408 return true;
409}

455 chip_reset();
456 printf("%s: LSI53C1020 Fusion-MPT SCSI controller.\n", devid_string);
457}

464void CLSI53C1020::load_flash()
465{
466 flash.clear();
467 download_image.clear();
468 flash_path.clear();
469 option_rom_offset = 0;
470 option_rom_size = 0;
471 ioc_firmware_size = 0;
472 firmware_product_id = LSI_FW_PRODUCT_1020_C0;
473 firmware_version = LSI_FW_VERSION;
474 download_image_type = 0;
475 download_active = false;
476 flash_dirty = false;
477 flash_views_dirty = false;
478 flash_last_dirty = 0;
479
480 const char* backing = myCfg->get_text_value("flash");
481 if (backing && backing[0])
482 {
483 flash_path = backing;
484 if (load_flash_image(flash_path))
485 {
486 rebuild_flash_views();
487 return;
488 }
489 }
490
491 /* An erased flash remains available when no backing or seed image is present. */
492 flash.assign(LSI_FLASH_SIZE, 0xFFU);
493 if (!flash_path.empty())
494 mark_flash_dirty();
495
496 const char* firmware = myCfg->get_text_value("firmware");
497 if (firmware && firmware[0])
498 load_ioc_firmware(firmware);
499
500 const char* rom = myCfg->get_text_value("rom");
501 if (rom && rom[0])
502 load_option_rom(rom);
503
504 rebuild_flash_views();
505 flush_flash();
506}

508bool CLSI53C1020::load_flash_image(const std::string& path)
509{
510 FILE* file = fopen(path.c_str(), "rb");
511 if (!file)
512 {
513 if (errno == ENOENT)
514 {
515 printf("%%LSI-I-NEWFLASH: %s: card flash %s does not exist; "
516 "it will be initialized from any available seed images.\n",
517 devid_string, path.c_str());
518 }
519 else
520 {
521 printf("%%LSI-W-NOFLASH: %s: card flash %s could not be opened; "
522 "continuing with volatile flash.\n", devid_string, path.c_str());
523 flash_path.clear();
524 }
525 return false;
526 }
527
528 if (fseek(file, 0, SEEK_END) != 0)
529 {
530 fclose(file);
531 printf("%%LSI-W-BADFLASH: %s: card flash %s could not be sized; "
532 "continuing with volatile flash.\n", devid_string, path.c_str());
533 flash_path.clear();
534 return false;
535 }
536 const long length = ftell(file);
537 if (length != LSI_FLASH_SIZE || fseek(file, 0, SEEK_SET) != 0)
538 {
539 fclose(file);
540 printf("%%LSI-W-BADFLASH: %s: card flash %s is %ld bytes; expected "
541 "%u. The file will not be overwritten.\n", devid_string,
542 path.c_str(), length, (unsigned)LSI_FLASH_SIZE);
543 flash_path.clear();
544 return false;
545 }
546
547 flash.resize(LSI_FLASH_SIZE);
548 const size_t count = fread(flash.data(), 1, flash.size(), file);
549 const int close_result = fclose(file);
550 if (count != flash.size() || close_result != 0)
551 {
552 flash.clear();
553 printf("%%LSI-W-BADFLASH: %s: card flash %s could not be read; "
554 "continuing with volatile flash.\n", devid_string, path.c_str());
555 flash_path.clear();
556 return false;
557 }
558
559 printf("%s: restored 512 KiB card flash from %s.\n", devid_string,
560 path.c_str());
561 return true;
562}

573
574 file.seekg(0, std::ios::end);
575 const std::streamoff length = file.tellg();
576 if (length <= 0 || length > LSI_FLASH_MAX_SIZE)
577 {
578 printf("%%LSI-W-BADROM: %s: optional PCI BIOS %s has an invalid "
579 "size; continuing without it.\n", devid_string, path.c_str());
580 return false;
581 }
582 file.seekg(0, std::ios::beg);
583 std::vector<u8> image((size_t)length);
584 if (!file.read((char*)image.data(), (std::streamsize)image.size()))
585 {
586 printf("%%LSI-W-BADROM: %s: optional PCI BIOS %s could not be "
587 "read; continuing without it.\n", devid_string, path.c_str());
588 return false;
589 }
590
591 u32 image_size = 0;
592 if (!valid_option_rom(image.data(), image.size(), &image_size) ||
593 image_size != image.size())
594 {
595 printf("%%LSI-W-BADROM: %s: optional PCI BIOS %s is not a valid "
596 "LSI 53C1020 x86 ROM; continuing without it.\n",
597 devid_string, path.c_str());
598 return false;
599 }
600
601 if (flash.empty())
602 flash.assign(LSI_FLASH_SIZE, 0xFFU);
603 if (image_size > flash.size())
604 {
605 printf("%%LSI-W-BADROM: %s: optional PCI BIOS %s does not fit in "
606 "the 512 KiB card flash; continuing without it.\n",
607 devid_string, path.c_str());
608 return false;
609 }
610
611 /* FLASH1030 places the BIOS at the top of the part. BAR6 remaps that
612 * physical tail region to logical option-ROM address zero. */
613 const u32 offset = (u32)flash.size() - image_size;
614 std::copy(image.begin(), image.end(), flash.begin() + offset);
615 mark_flash_dirty();
616 printf("%s: seeded PCI BIOS %s into card flash (%u bytes).\n",
617 devid_string, path.c_str(), image_size);
618 return true;
619}

622bool CLSI53C1020::load_ioc_firmware(const std::string& path)
623{
624 std::ifstream file(path, std::ios::binary);
625 if (!file)
626 {
627 printf("%%LSI-W-NOFW: %s: optional IOC firmware %s was not found; "
628 "continuing without an IOC image.\n", devid_string,
629 path.c_str());
630 return false;
631 }
632
633 file.seekg(0, std::ios::end);
634 const std::streamoff length = file.tellg();
635 if (length <= 0 || length > LSI_FLASH_MAX_SIZE)
636 {
637 printf("%%LSI-W-BADFW: %s: optional IOC firmware %s has an invalid "
638 "size; continuing without an IOC image.\n",
639 devid_string, path.c_str());
640 return false;
641 }
642 file.seekg(0, std::ios::beg);
643 std::vector<u8> image((size_t)length);
644 if (!file.read((char*)image.data(), (std::streamsize)image.size()))
645 {
646 printf("%%LSI-W-BADFW: %s: optional IOC firmware %s could not be "
647 "read; continuing without an IOC image.\n",
648 devid_string, path.c_str());
649 return false;
650 }
651
652 u32 image_size = 0;
653 if (!valid_ioc_firmware(image.data(), image.size(), &image_size) ||
654 image_size != image.size())

656 printf("%%LSI-W-BADFW: %s: optional IOC firmware %s is not valid "

658 devid_string, path.c_str());
659 return false;
660 }
661
662 if (flash.empty())
663 flash.assign(LSI_FLASH_SIZE, 0xFFU);
664 if (image_size > flash.size())

666 printf("%%LSI-W-BADFW: %s: optional IOC firmware %s does not fit "

668 devid_string, path.c_str());
669 return false;
670 }
671
672 std::copy(image.begin(), image.end(), flash.begin());
673 mark_flash_dirty();
674 const u32 version = read_le32(&image[0x24]);

678 "retained in flash while the emulated IOC remains a 53C1020 C0.\n",
679 devid_string);
680 }
681 printf("%s: seeded IOC firmware %s into card flash "
682 "(%u.%02u.%02u.%02u).\n",
683 devid_string, path.c_str(), (version >> 24) & 0xFFU,
684 (version >> 16) & 0xFFU, (version >> 8) & 0xFFU, version & 0xFFU);
685 return true;
686}

688void CLSI53C1020::rebuild_flash_views()
689{
690 option_rom_offset = 0;
691 option_rom_size = 0;
692 ioc_firmware_size = 0;
693 firmware_product_id = LSI_FW_PRODUCT_1020_C0;
694 firmware_version = LSI_FW_VERSION;
695
696 if (flash.empty())
697 {
698 flash_views_dirty = false;
699 return;
700 }
701
702 if (valid_ioc_firmware(flash.data(), flash.size(), &ioc_firmware_size))
703 {
704 firmware_version = read_le32(&flash[0x24]);
705 }
706
707 /* A production BIOS is top-aligned; use the highest valid image. */
708 for (u32 offset = (u32)flash.size() - 512U;; offset -= 512U)
709 {
710 u32 bytes = 0;
711 if (valid_option_rom(&flash[offset], flash.size() - offset, &bytes))
712 {
713 option_rom_offset = offset;
714 option_rom_size = bytes;
715 break;
716 }
717 if (offset == 0U)
718 break;
719 }
720
721 /* The Expansion ROM aperture is a logical BIOS view. The IOC firmware
722 * remains at physical flash offset zero and is never exposed as x86 code. */
723 flash_views_dirty = false;
724}

943int CLSI53C1020::RestoreState(FILE* f)
944{
945 long ss;
946 u32 m1;
947 u32 m2;
948 u32 flash_size;
949 int res;
950 size_t r;
951 SLSI_state saved_state;
952 std::vector<u8> saved_flash;
953 bool flash_in_state = false;
954
955 if ((res = CPCIDevice::RestoreState(f)))
956 return res;
957
958 r = fread(&m1, sizeof(u32), 1, f);
959 if (r != 1)
960 {
961 printf("%s: unexpected end of file!\n", devid_string);
962 return -1;
963 }
964
965 if (m1 != lsi_magic1)
966 {
967 printf("%s: MAGIC 1 does not match!\n", devid_string);
968 return -1;
969 }
970
971 r = fread(&ss, sizeof(long), 1, f);
972 if (r != 1)
973 {
974 printf("%s: unexpected end of file!\n", devid_string);
975 return -1;
976 }
977
978 if (ss != sizeof(state))
979 {
980 printf("%s: STRUCT SIZE does not match!\n", devid_string);
981 return -1;
982 }
983
984 r = fread(&saved_state, sizeof(saved_state), 1, f);
985 if (r != 1)
986 {
987 printf("%s: unexpected end of file!\n", devid_string);
988 return -1;
989 }
990
991 /* Older state files put the trailing magic directly after the controller
992 * structure. New files store the complete raw card flash first. */
993 r = fread(&flash_size, sizeof(u32), 1, f);
994 if (r != 1)
995 {
996 printf("%s: unexpected end of file!\n", devid_string);
997 return -1;
998 }
999 if (flash_size == lsi_magic2)
1000 {
1001 m2 = flash_size;
1002 }
1003 else
1004 {
1005 if (flash_size != 0U && flash_size != LSI_FLASH_SIZE)
1006 {
1007 printf("%s: card flash size does not match!\n", devid_string);
1008 return -1;
1009 }
1010 flash_in_state = true;
1011 saved_flash.resize(flash_size);
1012 if (flash_size != 0U &&
1013 fread(saved_flash.data(), 1, flash_size, f) != flash_size)
1014 {
1015 printf("%s: unexpected end of file!\n", devid_string);
1016 return -1;
1017 }
1018 r = fread(&m2, sizeof(u32), 1, f);
1019 if (r != 1)
1020 {
1021 printf("%s: unexpected end of file!\n", devid_string);
1022 return -1;
1023 }
1024 }
1025
1026 if (m2 != lsi_magic2)
1027 {
1028 printf("%s: MAGIC 2 does not match!\n", devid_string);
1029 return -1;
1030 }
1031
1032 state = saved_state;
1033 if (flash_in_state)
1034 {
1035 flash.swap(saved_flash);
1036 flash_dirty = false;
1037 flash_views_dirty = true;
1038 rebuild_flash_views();
1039 mark_flash_dirty();
1040 flush_flash();
1041 }
1042 else if (flash_views_dirty)
1043 {
1044 rebuild_flash_views();
1045 }
1046 download_image.clear();
1047 download_image_type = 0;
1048 download_active = false;
1049
1050 register_bar(0, 0, endian_32(pci_state.config_data[0][4]),
1051 endian_32(pci_state.config_mask[0][4]));
1052 register_bar(0, 1, endian_32(pci_state.config_data[0][5]),
1053 endian_32(pci_state.config_mask[0][5]));
1054 register_bar(0, 3, endian_32(pci_state.config_data[0][7]),
1055 endian_32(pci_state.config_mask[0][7]));
1056 register_bar(0, 6, endian_32(pci_state.config_data[0][12]),
1057 endian_32(pci_state.config_mask[0][12]));
1058
1059 trace_his_valid = false;
1060 trace_doorbell_valid = false;
1061 release_scsi_bus();
1062 const bool asserted = state.irq_asserted;
1063 state.irq_asserted = !asserted;
1064 eval_interrupts();
1065 printf("%s: %d bytes restored.\n", devid_string,
1066 (int)(ss + (flash_in_state ? sizeof(flash_size) + flash.size() : 0U)));
1067 return 0;
1068}

1073void CLSI53C1020::WriteMem_Bar(int func, int bar, u32 address, int dsize, u32 data)
1074{
1075 if (func != 0)
1076 return;
1077 if (dsize != 8 && dsize != 16 && dsize != 32)
1078 return;
1079 if (bar == 6)
1080 return;
1081 if (bar == 3)
1082 {
1083 address &= 0xFFFFU;
1084 CScopedLock<CMutex> regLock(myRegLock);
1085 if ((state.host_diag_bits & LSI_HOSTDIAG_MEMORY_ENABLE) == 0U)
1086 return;
1087
1088 const u32 base = state.test_base & 0xFFFF0000U;
1089 const u32 bytes = (u32)dsize / 8U;
1090 for (u32 i = 0; i < bytes; i++)
1091 write_b_diag(base | ((address + i) & 0xFFFFU),
1092 (u8)(data >> (i * 8U)));
1093 return;
1094 }
1095 if (bar != 0 && bar != 1)
1096 return;
1097
1098 const bool io_space = bar == 0;
1099 // Memory[0] occupies the 1 KiB aperture described in section 2.3.1.3.
1100 address &= io_space ? 0xFFU : (LSI_MEMORY0_SIZE - 1U);
1101 CScopedLock<CMutex> regLock(myRegLock);
1102
1103 const u8 bytes = (u8)(dsize / 8);
1104 u8 i = 0;
1105 while (i < bytes)
1106 {
1107 const u32 byte_offset = address + i;
1108 const u32 reg = byte_offset & ~3U;
1109 const u32 lane = byte_offset & 3U;
1110 u32 chunk = 4U - lane;
1111 if (chunk > (u32)(bytes - i))
1112 chunk = (u32)(bytes - i);
1113
1114 u32 value;
1115 if (chunk == 4U)
1116 {
1117 value = (u32)(data >> (i * 8U));
1118 }
1119 else
1120 {
1121 value = peek_l_register(reg, io_space);
1122 for (u32 j = 0; j < chunk; j++)
1123 {
1124 const u32 shift = (lane + j) * 8U;
1125 value &= ~(0xFFU << shift);
1126 value |= (u32)((data >> ((i + j) * 8U)) & 0xFFU) << shift;
1127 }
1128 }
1129 write_l_register(reg, value, io_space);
1130 i = (u8)(i + chunk);
1131 }
1132}

1137u32 CLSI53C1020::ReadMem_Bar(int func, int bar, u32 address, int dsize)
1138{
1139 if (func != 0)
1140 return 0xFFFFFFFFU;
1141 if (dsize != 8 && dsize != 16 && dsize != 32)
1142 return 0xFFFFFFFFU;
1143 if (bar == 6)
1144 {
1145 CScopedLock<CMutex> regLock(myRegLock);
1146 if (flash_views_dirty)
1147 rebuild_flash_views();
1148 u32 value = 0;
1149 const u32 bytes = (u32)dsize / 8U;
1150 for (u32 i = 0; i < bytes; i++)
1151 {
1152 const u32 offset = address + i;
1153 const u8 byte = offset < option_rom_size ?
1154 flash[option_rom_offset + offset] : 0xFFU;
1155 value |= (u32)byte << (i * 8U);
1156 }
1157 return value;
1158 }
1159 if (bar == 3)

1167 const u32 base = state.test_base & 0xFFFF0000U;

1169 for (u32 i = 0; i < bytes; i++)
1170 value |= (u32)read_b_diag(base | ((address + i) & 0xFFFFU)) <<
1171 (i * 8U);

1173 }

1180

1186 {

1189 if (reg != cached_reg)
1190 {
1191 cached_value = read_l_register(reg, io_space);
1192 cached_reg = reg;
1193 }
1194 value |= ((cached_value >> ((byte_offset & 3U) * 8U)) & 0xFFU) <<
1195 (i * 8U);
1196 }
1197 return value;

1217
1218void CLSI53C1020::write_l_doorbell(u32 value)
1219{
1220 if (state.hs_receiving)
1221 {
1222 if (state.hs_received_dwords < LSI_HANDSHAKE_MAX_DWORDS)
1223 state.hs_buffer[state.hs_received_dwords] = value;
1224 state.hs_received_dwords++;
1225 if (state.hs_received_dwords >= state.hs_expected_dwords)
1226 {
1227 state.hs_receiving = false;
1228 execute_handshake();
1229 }
1230 return;
1231 }
1232 if (state.hs_replying)
1233 return;
1234
1235 const u8 function = (u8)(value >> 24);
1236 switch (function)
1237 {
1238 case LSI_DOORBELL_IO_UNIT_RESET:
1239 reset_scsi_bus();
1240 message_unit_reset();
1241 break;
1242
1243 case LSI_DOORBELL_MESSAGE_UNIT_RESET:
1244 message_unit_reset();
1245 break;
1246
1247 case LSI_DOORBELL_HANDSHAKE:
1248 state.hs_expected_dwords = (u8)(value >> 16);
1249 state.hs_received_dwords = 0;
1250 memset(state.hs_buffer, 0, sizeof(state.hs_buffer));
1251 state.doorbell_interrupt = true;
1252 if (state.hs_expected_dwords == 0U)
1253 execute_handshake();
1254 else
1255 state.hs_receiving = true;
1256 break;
1257
1258 case LSI_DOORBELL_REPLY_FRAME_REMOVAL:
1259 {
1260 u8 reply[LSI_REPLY_REMOVAL_BYTES];
1261 const u32 count = state.reply_free_count;
1262
1263 memset(reply, 0, sizeof(reply));
1264 write_le32(reply, count);
1265 for (u32 i = 0; i < count; i++)
1266 {
1267 write_le32(&reply[sizeof(u32) * (i + 1U)],
1268 state.reply_free_fifo[state.reply_free_head]);
1269 state.reply_free_head =
1270 (state.reply_free_head + 1U) % LSI_REPLY_FIFO_DEPTH;
1271 }
1272 state.reply_free_count = 0;
1273 post_handshake_reply(reply, sizeof(u32) * (count + 1U));
1274 trace("reply-frame removal returned %u frame%s", (unsigned)count,

1276 break;
1277 }
1278
1279 default:
1280 break;
1281 }

1312u8 CLSI53C1020::read_b_flash(u32 address) const
1313{
1314 if (flash.empty())
1315 return 0xFFU;
1316 address %= (u32)flash.size();
1317
1318 if (state.flash_mode == LSI_FLASH_MODE_AUTOSELECT)
1319 {
1320 switch (address & (LSI_FLASH_SECTOR_SIZE - 1U))
1321 {
1322 case 0:
1323 return LSI_FLASH_MANUFACTURER_ID;
1324 case 1:
1325 return LSI_FLASH_DEVICE_ID;
1326 case 2:
1327 return 0; // sector protection is not fitted
1328 default:
1329 return 0;
1330 }
1331 }
1332
1333 if (state.flash_mode == LSI_FLASH_MODE_CFI)
1334 {
1335 switch (address)
1336 {
1337 case 0x10:
1338 return 'Q';
1339 case 0x11:
1340 return 'R';
1341 case 0x12:
1342 return 'Y';
1343 case 0x13:
1344 return 0x02U; // AMD/Fujitsu command set
1345 case 0x27:
1346 return 19U; // 2^19 bytes
1347 case 0x2C:
1348 return 1U; // one erase-block region
1349 case 0x2D:
1350 return 7U; // eight sectors, encoded as count - 1
1351 case 0x2F:
1352 return 0U;
1353 case 0x30:
1354 return 1U; // 0x0100 * 256 = 64 KiB
1355 default:
1356 return 0;
1357 }
1358 }
1359
1360 return flash[address];
1361}

1364void CLSI53C1020::write_b_flash(u32 address, u8 value)
1365{
1366 if (flash.empty() || (state.flash_csr & LSI_FLASH_CSR_WRITE_ENABLE) == 0U)
1367 return;
1368 address %= (u32)flash.size();
1369
1370 if (value == 0xF0U && state.flash_mode != LSI_FLASH_MODE_PROGRAM &&
1371 state.flash_mode != LSI_FLASH_MODE_BYPASS_PROGRAM)
1372 {
1373 state.flash_mode = LSI_FLASH_MODE_READ;
1374 return;
1375 }
1376
1377 switch (state.flash_mode)
1378 {
1379 case LSI_FLASH_MODE_READ:
1380 if (address == LSI_FLASH_UNLOCK_ADDRESS1 && value == 0xAAU)
1381 state.flash_mode = LSI_FLASH_MODE_UNLOCK1;
1382 else if (address == 0x55U && value == 0x98U)
1383 state.flash_mode = LSI_FLASH_MODE_CFI;
1384 break;
1385
1386 case LSI_FLASH_MODE_UNLOCK1:
1387 state.flash_mode =
1388 address == LSI_FLASH_UNLOCK_ADDRESS2 && value == 0x55U ?
1389 LSI_FLASH_MODE_UNLOCK2 : LSI_FLASH_MODE_READ;
1390 break;
1391
1392 case LSI_FLASH_MODE_UNLOCK2:
1393 if (address != LSI_FLASH_UNLOCK_ADDRESS1)
1394 {
1395 state.flash_mode = LSI_FLASH_MODE_READ;
1396 break;
1397 }
1398 switch (value)
1399 {
1400 case 0x90:
1401 state.flash_mode = LSI_FLASH_MODE_AUTOSELECT;
1402 break;
1403 case 0xA0:
1404 state.flash_mode = LSI_FLASH_MODE_PROGRAM;
1405 break;
1406 case 0x80:
1407 state.flash_mode = LSI_FLASH_MODE_ERASE_UNLOCK1;
1408 break;
1409 case 0x20:
1410 state.flash_mode = LSI_FLASH_MODE_BYPASS;
1411 break;
1412 default:
1413 state.flash_mode = LSI_FLASH_MODE_READ;
1414 break;
1415 }
1416 break;
1417
1418 case LSI_FLASH_MODE_PROGRAM:
1419 {
1420 const u8 programmed = (u8)(flash[address] & value);
1421 if (programmed != flash[address])
1422 {
1423 flash[address] = programmed;
1424 mark_flash_dirty();
1425 }
1426 state.flash_mode = LSI_FLASH_MODE_READ;
1427 break;
1428 }
1429
1430 case LSI_FLASH_MODE_ERASE_UNLOCK1:
1431 state.flash_mode =
1432 address == LSI_FLASH_UNLOCK_ADDRESS1 && value == 0xAAU ?
1433 LSI_FLASH_MODE_ERASE_UNLOCK2 : LSI_FLASH_MODE_READ;
1434 break;
1435
1436 case LSI_FLASH_MODE_ERASE_UNLOCK2:
1437 state.flash_mode =
1438 address == LSI_FLASH_UNLOCK_ADDRESS2 && value == 0x55U ?
1439 LSI_FLASH_MODE_ERASE_SELECT : LSI_FLASH_MODE_READ;
1440 break;
1441
1442 case LSI_FLASH_MODE_ERASE_SELECT:
1443 if (address == LSI_FLASH_UNLOCK_ADDRESS1 && value == 0x10U)
1444 {
1445 if (!std::all_of(flash.begin(), flash.end(), [](u8 byte) { return byte == 0xFFU; }))
1446 {
1447 std::fill(flash.begin(), flash.end(), 0xFFU);
1448 mark_flash_dirty();
1449 }
1450 }
1451 else if (value == 0x30U)
1452 {
1453 const u32 start = address & ~(LSI_FLASH_SECTOR_SIZE - 1U);
1454 const u32 end = std::min<u32>(start + LSI_FLASH_SECTOR_SIZE,
1455 (u32)flash.size());
1456 if (!std::all_of(flash.begin() + start, flash.begin() + end,
1457 [](u8 byte) { return byte == 0xFFU; }))
1458 {
1459 std::fill(flash.begin() + start, flash.begin() + end, 0xFFU);
1460 mark_flash_dirty();
1461 }
1462 }
1463 state.flash_mode = LSI_FLASH_MODE_READ;
1464 break;
1465
1466 case LSI_FLASH_MODE_BYPASS:
1467 if (value == 0xA0U)
1468 state.flash_mode = LSI_FLASH_MODE_BYPASS_PROGRAM;
1469 else if (value == 0x90U)
1470 state.flash_mode = LSI_FLASH_MODE_BYPASS_EXIT;
1471 break;
1472
1473 case LSI_FLASH_MODE_BYPASS_PROGRAM:
1474 {
1475 const u8 programmed = (u8)(flash[address] & value);
1476 if (programmed != flash[address])
1477 {
1478 flash[address] = programmed;
1479 mark_flash_dirty();
1480 }
1481 state.flash_mode = LSI_FLASH_MODE_BYPASS;
1482 break;
1483 }
1484
1485 case LSI_FLASH_MODE_BYPASS_EXIT:
1486 state.flash_mode = value == 0 ? LSI_FLASH_MODE_READ : LSI_FLASH_MODE_BYPASS;
1487 break;
1488
1489 case LSI_FLASH_MODE_AUTOSELECT:
1490 case LSI_FLASH_MODE_CFI:
1491 default:
1492 break;
1493 }
1494}

1553u32 CLSI53C1020::peek_l_register(u32 reg, bool io_space) const
1554{
1555 if (!io_space && reg >= LSI_SHARED_RAM_OFFSET && reg < LSI_MEMORY0_SIZE)
1556 return read_le32(&state.shared_ram[reg - LSI_SHARED_RAM_OFFSET]);
1557
1558 switch (reg)
1559 {
1560 case R_DOORBELL:
1561 return read_l_doorbell();
1562 case R_WRITE_SEQUENCE:
1563 return 0x0000000BU;
1564 case R_HOST_DIAG:
1565 return state.host_diag_bits |
1566 (state.diag_write_enable ? LSI_HOSTDIAG_DIAG_WRITE_ENABLE : 0U);
1567 case R_TEST_BASE:
1568 return state.test_base;
1569 case R_DIAG_RW_DATA:
1570 if (!io_space || (state.host_diag_bits & LSI_HOSTDIAG_DIAG_RW_ENABLE) == 0U)
1571 return 0xFFFFFFFFU;
1572 return read_l_diag(state.diag_rw_addr);
1573 case R_DIAG_RW_ADDR:
1574 return io_space ? state.diag_rw_addr : 0xFFFFFFFFU;
1575 case R_HOST_INT_STATUS:
1576 return read_l_host_int_status();
1577 case R_HOST_INT_MASK:
1578 return state.host_interrupt_mask;
1579 case R_REQUEST_QUEUE:
1580 return 0xFFFFFFFFU;
1581 case R_REPLY_QUEUE:
1582 return state.reply_post_count ? state.reply_post_fifo[state.reply_post_head] : 0xFFFFFFFFU;
1583 default:
1584 return 0U;
1585 }
1586}

1591u32 CLSI53C1020::read_l_register(u32 reg, bool io_space)
1592{
1593 u32 value;
1594 switch (reg)
1595 {
1596 case R_DIAG_RW_DATA:
1597 if (!io_space || (state.host_diag_bits & LSI_HOSTDIAG_DIAG_RW_ENABLE) == 0U)
1598 return 0xFFFFFFFFU;
1599 value = read_l_diag(state.diag_rw_addr);
1600 state.diag_rw_addr += 4U;
1601 return value;
1602
1603 case R_REPLY_QUEUE:
1604 if (state.reply_post_count == 0U && state.bus_reset_event_pending && state.events_enabled)
1605 {
1606 state.bus_reset_event_pending = false;
1607 post_event(LSI_EVENT_IOC_BUS_RESET, 0U);
1608 }
1609 else if (state.reply_post_count == 0U && state.taskmgmt_mirror_pending)
1610 {
1611 state.taskmgmt_mirror_pending = false;
1612 state.hs_receiving = false;
1613 state.hs_expected_dwords = 0;
1614 state.hs_received_dwords = 0;
1615 state.hs_replying = false;
1616 state.hs_reply_len_words = 0;
1617 state.hs_reply_pos = 0;
1618 state.hs_word_consumed = false;
1619 state.hs_int_cleared = false;
1620 trace("task-management reply mirrored to reply-post FIFO");
1621 post_address_reply(state.taskmgmt_reply_frame, sizeof(state.taskmgmt_reply_frame));
1622 eval_interrupts();
1623 }
1624 if (state.reply_post_count != 0U)
1625 {
1626 value = state.reply_post_fifo[state.reply_post_head];
1627 state.reply_post_head = (state.reply_post_head + 1U) % LSI_REPLY_FIFO_DEPTH;
1628 state.reply_post_count--;
1629 /* bit3 stays asserted (sticky) until the empty read below */
1630 eval_interrupts();
1631 return value;
1632 }
1633 /* empty (p.4-42): this read clears the sticky reply interrupt */
1634 if (state.reply_int_sticky)
1635 {
1636 state.reply_int_sticky = false;
1637 eval_interrupts();
1638 }
1639 return 0xFFFFFFFFU;
1640
1641 case R_HOST_INT_STATUS:
1642 value = read_l_host_int_status();
1643 if (trace_enabled && (!trace_his_valid || value != trace_last_his))
1644 {
1645 trace_last_his = value;
1646 trace_his_valid = true;
1647 trace("HIS read -> 0x%08x", value);
1648 }
1649 return value;
1650
1651 case R_DOORBELL:
1652 value = read_l_doorbell();
1653 if (trace_enabled && (!trace_doorbell_valid || value != trace_last_doorbell))
1654 {
1655 trace_last_doorbell = value;
1656 trace_doorbell_valid = true;
1657 trace("doorbell read -> 0x%08x", value);
1658 }
1659 if (state.hs_replying && state.hs_reply_pos < state.hs_reply_len_words)
1660 {
1661 state.hs_word_consumed = true;
1662 advance_handshake_reply();
1663 eval_interrupts();
1664 }
1665 return value;
1666
1667 default:
1668 return peek_l_register(reg, io_space);
1669 }
1670}

1675void CLSI53C1020::write_l_register(u32 reg, u32 value, bool io_space)
1676{
1677 if (!io_space && reg >= LSI_SHARED_RAM_OFFSET && reg < LSI_MEMORY0_SIZE)
1678 {
1679 write_le32(&state.shared_ram[reg - LSI_SHARED_RAM_OFFSET], value);
1680 return;
1681 }
1682
1683 switch (reg)
1684 {
1685 case R_DOORBELL:
1686 write_l_doorbell(value);
1687 break;
1688
1689 case R_WRITE_SEQUENCE:
1690 {
1691 const u8 key = (u8)(value & 0x0FU);
1692 if (state.write_sequence_pos < 5U &&
1693 key == lsi_write_sequence_keys[state.write_sequence_pos])
1694 {
1695 state.write_sequence_pos++;
1696 if (state.write_sequence_pos == 5U)
1697 {
1698 state.diag_write_enable = true;
1699 state.write_sequence_pos = 0;
1700 }
1701 }
1702 else
1703 {
1704 state.diag_write_enable = false;
1705 state.write_sequence_pos = key == lsi_write_sequence_keys[0] ? 1U : 0U;
1706 }
1707 break;
1708 }
1709
1710 case R_HOST_DIAG:
1711 if (!state.diag_write_enable)
1712 break;
1713 if ((value & LSI_HOSTDIAG_RESET_ADAPTER) != 0U)
1714 {
1715 trace("HostDiag RESET_ADAPTER (hard reset)");
1716 chip_reset();
1717 break;
1718 }
1719 state.host_diag_bits = (u8)(value & LSI_HOSTDIAG_STORED_MASK);
1720 break;
1721
1722 case R_TEST_BASE:
1723 state.test_base = value & 0xFFFF0000U;
1724 break;
1725
1726 case R_DIAG_RW_DATA:
1727 if (io_space && state.diag_write_enable &&
1728 (state.host_diag_bits & LSI_HOSTDIAG_DIAG_RW_ENABLE) != 0U)
1729 {
1730 write_l_diag(state.diag_rw_addr, value);
1731 state.diag_rw_addr += 4U;
1732 }
1733 break;
1734
1735 case R_DIAG_RW_ADDR:
1736 if (io_space && state.diag_write_enable &&
1737 (state.host_diag_bits & LSI_HOSTDIAG_DIAG_RW_ENABLE) != 0U)
1738 state.diag_rw_addr = value;
1739 break;
1740
1741 case R_HOST_INT_STATUS:
1742 write_l_host_int_status();
1743 break;
1744
1745 case R_HOST_INT_MASK:
1746 state.host_interrupt_mask = value & LSI_HIM_WRITABLE_MASK;
1747 break;
1748
1749 case R_REQUEST_QUEUE:
1750 execute_request(value);
1751 break;
1752
1753 case R_REPLY_QUEUE:
1754 if (state.reply_free_count < LSI_REPLY_FIFO_DEPTH)
1755 {
1756 const u32 tail =
1757 (state.reply_free_head + state.reply_free_count) % LSI_REPLY_FIFO_DEPTH;
1758 state.reply_free_fifo[tail] = value;
1759 state.reply_free_count++;
1760 }
1761 break;
1762
1763 default:
1764 break;
1765 }
1766 eval_interrupts();
1767}

1891void CLSI53C1020::post_handshake_reply(const u8* frame, u32 frame_bytes)
1892{
1893 u32 words = (frame_bytes + 1U) / 2U;
1894 if (words > LSI_REPLY_MAX_WORDS)
1895 words = LSI_REPLY_MAX_WORDS;
1896
1897 for (u32 i = 0; i < words; i++)
1898 {
1899 const u16 lo = frame[i * 2U];
1900 const u16 hi = (i * 2U + 1U < frame_bytes) ? frame[i * 2U + 1U] : 0U;
1901 state.hs_reply_words[i] = (u16)(lo | (hi << 8));
1902 }
1903 state.hs_reply_len_words = (u16)words;
1904 state.hs_reply_pos = 0;
1905 state.hs_word_consumed = false;
1906 state.hs_int_cleared = false;
1907 state.hs_replying = true;
1908 state.doorbell_interrupt = true;
1909}

1916 state.hs_word_consumed = false;
1917 state.hs_int_cleared = false;
1918 state.hs_reply_pos++;
1919 if (state.hs_reply_pos >= state.hs_reply_len_words)
1920 {
1921 state.hs_replying = false;
1922 state.taskmgmt_mirror_pending = false;
1923 }
1924 state.doorbell_interrupt = true;
1925}

1942bool CLSI53C1020::dma_write(u64 address, const u8* data, u32 bytes)
1943{
1944 if (bytes == 0)
1945 return true;
1946 if ((address >> 32) != 0 || address > 0xFFFFFFFFULL - (u64)(bytes - 1U))
1947 {
1948 trace("unsupported 64-bit DMA write at 0x%016" PRIx64, address);
1949 enter_fault(LSI_FAULT_NO_DMA);

1960void CLSI53C1020::post_address_reply(const u8* frame, u32 frame_bytes)
1961{
1962 if (state.reply_post_count >= LSI_REPLY_FIFO_DEPTH)
1963 {
1964 enter_fault(LSI_FAULT_REPLY_POST_FULL);
1965 return;
1966 }
1967 if (state.reply_free_count == 0)
1968 {
1969 enter_fault(LSI_FAULT_REPLY_FREE_EMPTY);
1970 return;
1971 }
1972
1973 const u32 mfa = state.reply_free_fifo[state.reply_free_head];
1974 u32 limit = state.reply_frame_size;
1975 if (limit > LSI_FRAME_MAX)
1976 limit = LSI_FRAME_MAX;
1977 if (frame_bytes > limit)
1978 frame_bytes = limit;

1980 if (!dma_write(mfa_address, frame, frame_bytes))
1981 return;
1982
1983 state.reply_free_head = (state.reply_free_head + 1U) % LSI_REPLY_FIFO_DEPTH;
1984 state.reply_free_count--;
1985 const u32 reply = (mfa >> 1) | LSI_REPLY_MFA_ADDRESS_BIT;
1986 const u32 tail = (state.reply_post_head + state.reply_post_count) % LSI_REPLY_FIFO_DEPTH;
1987 state.reply_post_fifo[tail] = reply;

1992}

2394int CLSI53C1020::collect_sgl(const u8* frame, u32 sge_offset,
2395 u8 chain_offset_dwords, SLSI_sge* out, u32 max_entries)
2396{
2397 u8 segment[LSI_SGL_SEG_MAX];
2398 const u8* current = frame;
2399 u32 segment_len = LSI_FRAME_MAX;
2400 u32 pos = sge_offset;
2401 u32 chain_pos = (u32)chain_offset_dwords * 4U;
2402 u32 elements = 0;
2403 int count = 0;
2404
2405 for (;;)
2406 {
2407 if (++elements > LSI_SGL_MAX_ELEMENTS || pos + 8U > segment_len)
2408 return -1;
2409
2410 const u32 flags_len = read_le32(&current[pos]);
2411 const u8 flags = (u8)(flags_len >> 24);
2412 const u8 type = (u8)(flags & LSI_SGEF_TYPE_MASK);
2413 u64 address;
2414
2415 if (type == LSI_SGEF_TYPE_CHAIN)
2416 {
2417 const u32 chain_len = flags_len & 0xFFFFU;
2418 const u32 next_offset = (flags_len >> 16) & 0xFFU;
2419 if (flags & LSI_SGEF_ADDR64)
2420 {
2421 if (pos + 12U > segment_len)
2422 return -1;
2423 address =
2424 (u64)read_le32(&current[pos + 4U]) | ((u64)read_le32(&current[pos + 8U]) << 32);
2425 }
2426 else
2427 {
2428 address = read_le32(&current[pos + 4U]);
2429 }
2430 if (chain_len < 8U || chain_len > sizeof(segment))
2431 return -1;
2432 if (!dma_read(address, segment, chain_len))
2433 return -1;
2434 current = segment;
2435 segment_len = chain_len;
2436 pos = 0;
2437 chain_pos = next_offset * 4U;
2438 continue;
2439 }
2440
2441 if (type != LSI_SGEF_TYPE_SIMPLE)
2442 return -1;
2443
2444 if (flags & LSI_SGEF_ADDR64)
2445 {
2446 if (pos + 12U > segment_len)
2447 return -1;
2448 address =
2449 (u64)read_le32(&current[pos + 4U]) | ((u64)read_le32(&current[pos + 8U]) << 32);
2450 pos += 12U;
2451 }
2452 else
2453 {
2454 address = read_le32(&current[pos + 4U]);
2455 pos += 8U;
2456 }
2457
2458 const u32 length = flags_len & LSI_SGE_LENGTH_MASK;
2459 if (length)
2460 {
2461 if ((u32)count >= max_entries)
2462 return -1;
2463 out[count].addr = address;
2464 out[count].len = length;
2465 count++;
2466 }
2467
2468 if (flags & LSI_SGEF_END_OF_LIST)
2469 break;
2470 if (flags & LSI_SGEF_LAST_ELEMENT)
2471 {
2472 if (chain_pos == 0)
2473 break;
2474 pos = chain_pos;
2475 }
2476 }
2477 return count;
2478}

2580 }

2586 return false;
2587 }

2595 if (!bytes)
2596 break;

2598 scsi_xfer_done(0);

2601 if (scsi_get_phase(0) != SCSI_PHASE_FREE)
2602 release_scsi_target(target);
