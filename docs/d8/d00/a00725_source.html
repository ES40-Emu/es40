339 // Shared by JIT stores and LL/SC. 64-byte spacing avoids false sharing on
340 // 64-byte host cache lines. JIT code keeps these addresses until destruction.
341 struct alignas(64) LLBucket {
342 std::atomic<u64> sequence;
343 std::atomic<u8> lock;
344 };
345 static_assert(std::is_standard_layout<LLBucket>::value, "JIT LL bucket layout");
346 static_assert(sizeof(LLBucket) == 64, "JIT LL bucket stride");
347 static_assert(offsetof(LLBucket, sequence) == 0, "JIT LL sequence offset");
348 static_assert(offsetof(LLBucket, lock) == 8, "JIT LL lock offset");
349 static const u32 kLLBuckets = 65536; // 64B-line hash; power of two
350 static_assert(kLLBuckets == 65536, "JIT LL bucket hash mask");
351 LLBucket* ll_buckets() { return m_ll_buckets; }
352 static_assert(sizeof(std::atomic<u8>) == 1, "JIT LL watch byte layout");
353 std::atomic<u8>* ll_watch_data() { return m_ll_watch.get(); }
354 bool ll_page_watched(u64 phys) const {
355 return m_ll_watch_pages[(size_t)(phys >> 13)].load(std::memory_order_acquire) != 0;

393 // Detected-idle naps on SMP guests. A CPU arms the physical 64-byte lines
394 // its idle loop reads; a store to one changes that line's LL bucket
395 // sequence. Other CPUs compare sequences at each JIT chain boundary and wake
396 // the napper, so a dispatch by plain store (Tru64 thread_setrun) is seen
397 // without an IPI. Bucket aliases only cause early wakes.

403 if (m_idle_watchers.load(std::memory_order_acquire) & ~(1u << cpuid))
404 idle_watch_wake(cpuid);
405 }
