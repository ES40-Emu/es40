104void CDMA::ResetPCI()
105{
106 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
107 // While guest master clear, system reset also resets channel programming,
108 // request lines and any accept/pending demand/block/cascade service.
109 memset(&state, 0, sizeof(state));
110 for (int controller = 0; controller < 2; controller++)
111 {
112 state.controller[controller].mask = 0x0f;
113 state.controller[controller].lobyte = true;
114 }
115}

173u64 CDMA::dma_address(u16 pagebase, u32 current,
174 const SDMA_transfer_properties& properties)
175{
176 if (!(isa_bridge.config_read(0, 0x42, 8) & 0x40))
177 pagebase &= 0x00ff;
178
179 // Shifted addressing drives A16 from current, not low-page bit 0.
180 pagebase &= (u16)(0xffffU << properties.address_shift);
181
182 return ((u64)pagebase << 16) | ((u64)current << properties.address_shift);
183}

185u32 CDMA::step_address(u32 current, size_t units, bool decrement,
186 const SDMA_transfer_properties& properties)
187{
188 u32 delta = (u32)(units * properties.address_step);
189 return (decrement ? current - delta : current + delta) & properties.address_mask;
190}

192size_t CDMA::increment_chunk_size(u32 current, size_t length,
193 const SDMA_transfer_properties& properties)
194{
195 // Current ISA geometry is contiguous and wraps only the current register.
196 size_t units_to_wrap = ((size_t)properties.address_mask + 1 - current) /
197 properties.address_step;
198 size_t bytes_to_wrap = units_to_wrap * properties.data_width;
199 return length < bytes_to_wrap ? length : bytes_to_wrap;
200}

216u64 CDMA::ReadMem(int index, u64 address, int dsize)
217{
218 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
219 u64 ret;
220 u8 data = 0;
221 int ctrlr;
222 int num;
223 //printf("dma: Readmem %s, %" PRIx64 ", %x\n",DMA_INDEX(index),address, dsize);
224 switch (dsize)
225 {
226 case 32:
227 ret = ReadMem(index, address, 8);
228 ret |= ReadMem(index, address + 1, 8) << 8;
229 ret |= ReadMem(index, address + 2, 8) << 16;
230 ret |= ReadMem(index, address + 3, 8) << 24;
231 return ret;
232
233 case 16:
234 ret = ReadMem(index, address, 8);
235 ret |= ReadMem(index, address + 1, 8) << 8;
236 return ret;
237
238 case 8:
239
240
241
242
243 if (index == DMA1_IO_CHANNEL || index == DMA1_IO_MAIN)
244 address >>= 1;
245
246
247
248 switch (index)
249 {
250 case DMA0_IO_CHANNEL:
251 case DMA1_IO_CHANNEL:
252 ctrlr = (index == DMA1_IO_CHANNEL) ? 1 : 0;
253 num = ((address & 0x0e) >> 1) + (ctrlr * 4);
254 if (address & 1)
255 {
256 // word count registers
257 data = (state.channel[num].count >>
258 (state.controller[ctrlr].lobyte ? 0 : 8)) & 0xff;
259 }
260 else
261 {
262 // current address
263 data = (state.channel[num].current >>
264 (state.controller[ctrlr].lobyte ? 0 : 8)) & 0xff;
265 }
266 state.controller[ctrlr].lobyte =
267 !state.controller[ctrlr].lobyte;
268 break;
269
270 case DMA0_IO_MAIN:
271 case DMA1_IO_MAIN:
272 ctrlr = (index == DMA1_IO_MAIN) ? 1 : 0;
273 if (address == 0)
274 {
275 // Pending requests are visible even when masked or disabled.
276 data = state.controller[ctrlr].status | (get_requests(ctrlr) << 4);
277 state.controller[ctrlr].status = 0;
278 }
279 else if (address == 7)
280 {
281 data = state.controller[ctrlr].mask & 0x0f;
282 }
283 break;
284
285 case DMA_IO_LPAGE:
286 case DMA_IO_HPAGE:
287 num = dma_page_channel(address);
288 if (num < 0)
289 {
290 data = 0xff;
291 break;
292 }
293 if (index == DMA_IO_LPAGE)
294 data = state.channel[num].pagebase & 0xff;
295 else
296 data = (state.channel[num].pagebase >> 8) & 0xff;
297 break;
298
299 case DMA0_IO_EXT:
300 case DMA1_IO_EXT:
301 // M1543C extended-mode ports are write-only. Use the same all-ones
302 // value as an unmapped I/O read, without changing controller state.
303 data = 0xff;
304 break;
305
306 default:
307 FAILURE(InvalidArgument, "dma: ReadMem index out of range");
308 }
309
310#if defined(DEBUG_DMA)
311 printf("dma: read %s,%02" PRIx64 ": %02" PRIx8 ". \n", DMA_INDEX(index), address, data);
312#endif
313 }
314 return data;
315}

317void CDMA::WriteMem(int index, u64 address, int dsize, u64 data)
318{
319 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
320 int num = 0;
321 switch (dsize)
322 {
323 case 32:
324 WriteMem(index, address + 0, 8, (data >> 0) & 0xff);
325 WriteMem(index, address + 1, 8, (data >> 8) & 0xff);
326 WriteMem(index, address + 2, 8, (data >> 16) & 0xff);
327 WriteMem(index, address + 3, 8, (data >> 24) & 0xff);
328 return;
329
330 case 16:
331 WriteMem(index, address + 0, 8, (data >> 0) & 0xff);
332 WriteMem(index, address + 1, 8, (data >> 8) & 0xff);
333 return;
334
335 case 8:
336 data &= 0xff;
337 if (index == DMA1_IO_CHANNEL || index == DMA1_IO_MAIN)
338 address >>= 1;
339
340#if defined(DEBUG_DMA)
341 //printf("dma: write %s, %02x: %02x. \n", DMA_INDEX(index), (u32)address, data);
342#endif
343 switch (index)
344 {
345 case DMA0_IO_CHANNEL:
346 case DMA1_IO_CHANNEL:
347 {
348 int ctrlr = (index == DMA1_IO_CHANNEL) ? 1 : 0;
349 num = ((address & 0x0e) >> 1) + (ctrlr * 4);
350 // Reprogramming a channel releases any held service.
351 release_service(num);
352 if (address & 1)
353 {
354 if (state.controller[ctrlr].lobyte)
355 state.channel[num].base_count =
356 (state.channel[num].base_count & 0xff00) | data;
357 else
358 state.channel[num].base_count =
359 (state.channel[num].base_count & 0x00ff) | (data << 8);
360 state.channel[num].count = state.channel[num].base_count;
361#if defined(DEBUG_DMA)
362 printf("dma channel %d count: %04x\n", num, state.channel[num].count);
363#endif
364 }
365 else {
366 if (state.controller[ctrlr].lobyte)
367 state.channel[num].base = (state.channel[num].base & 0xff00) | data;
368 else
369 state.channel[num].base = (state.channel[num].base & 0x00ff) | (data << 8);
370 state.channel[num].current = state.channel[num].base;
371#if defined(DEBUG_DMA)
372 printf("dma channel %d base: %04x\n", num, state.channel[num].base);
373#endif
374 }
375 state.controller[ctrlr].lobyte =
376 !state.controller[ctrlr].lobyte;
377 break;
378 }
379
380 case DMA1_IO_MAIN:
381 case DMA0_IO_MAIN:
382 {
383 int ctrlr = (index == DMA1_IO_MAIN) ? 1 : 0;
384 num = ctrlr;
385 switch (address) {
386 case 0: // command
387 if (DMA_TRACE_CONTROLLER(num))
388 printf("dma: command register %d written with %" PRIx64 "\n", num, data);
389 if (data & 0x04)
390 {
391 // Release under the old priority mode before changing command.
392 for (int channel = 0; channel < 4; channel++)
393 release_service(num * 4 + channel);
394 }
395 state.controller[num].command = data;
396 break;
397
398 case 1: // request
399 set_request(num, data & 0x03, (data & 0x04) >> 2);
400 break;
401
402 case 2: // single mask
403 if (DMA_TRACE_CHANNEL((num * 4) + (data & 0x03)))
404 printf("dma: mask single on %d : %" PRId64 " %s\n", num, data & 0x03, data & 0x4 ? "Masked" : "Unmasked");
405 state.controller[num].mask = (state.controller[num].mask & ~(1 << (data & 0x03))) | (((data & 0x04) >> 2) << (data & 0x03));
406 // Explicit mask-set writes cancel accepted service between calls.
407 if (data & 0x04)
408 release_service(num * 4 + (data & 0x03));
409 if (DMA_TRACE_CHANNEL((num * 4) + (data & 0x03)))
410 printf(" Mask status: %x\n", state.controller[num].mask);
411 do_dma();
412 break;
413
414 case 3: // mode register
415 if (DMA_TRACE_CHANNEL((num * 4) + (data & 0x03)))
416 {
417 printf("dma: mode register %d for channel %" PRId64 " written with %" PRIx64 "\n", num, (num * 4) + (data & 0x03), data);
418 printf(" Mode: %s, Address %s, Autoinit %s, Command: %s\n",
419 (data & 0x80 ? (data & 0x40 ? "Cascade" : "Block") : (data & 0x40 ? "Single" : "Demand")),
420 (data & 0x20 ? "Decrement" : "Increment"),
421 (data & 0x10 ? "Enable" : "Disable"),
422 (data & 0x08 ? (data & 0x04 ? "Illegal" : "Read") : (data & 0x04 ? "Write" : "Verify")));
423 }
424
425
426 release_service(num * 4 + (data & 0x03));
427 state.channel[(num * 4) + (data & 0x03)].mode = data;
428 break;
429
430 case 4: // clear flipflop(s)
431 if (DMA_TRACE_CONTROLLER(num))
432 printf("dma: flipflops cleared for dma %d\n", num);
433 state.controller[num].lobyte = true;
434 break;
435
436 case 5: // master reset
437#if defined(DEBUG_DMA)
438 printf("DMA-I-RESET: DMA %d reset.", index - DMA_IO_BASE);
439#endif
440 for (int channel = 0; channel < 4; channel++)
441 release_service(num * 4 + channel);
442 state.controller[num].lobyte = true;
443 state.controller[num].command = 0;
444 state.controller[num].status = 0;
445 state.controller[num].request = 0;
446 state.controller[num].next_priority = 0;
447 // Reset the controller, not the request lines driven by devices.
448 state.controller[num].mask = 0x0f;
449 break;
450
451 case 6: // master enable
452 state.controller[num].mask = 0x00;
453 do_dma();
454 break;
455
456
457 case 7: // master mask
458 state.controller[num].mask = data & 0x0f;
459 for (int channel = 0; channel < 4; channel++)
460 {
461 if (data & (1 << channel))
462 release_service(num * 4 + channel);
463 }
464 do_dma();
465 break;
466 }
467 break;
468 }
469
470 case DMA_IO_LPAGE:
471 case DMA_IO_HPAGE:
472 num = dma_page_channel(address);
473 if (num < 0)
474 return;
475 release_service(num);
476 if (index == DMA_IO_LPAGE)
477 state.channel[num].pagebase = (state.channel[num].pagebase & 0xff00) | data;
478 else
479 state.channel[num].pagebase = (state.channel[num].pagebase & 0xff) | (data << 8);
480
481#if defined(DEBUG_DMA)
482 printf("dma channel %d pagebase: %04x\n", num, state.channel[num].pagebase);
483#endif
484
485 break;
486
487
488 case DMA0_IO_EXT:
489 case DMA1_IO_EXT:
490 // Each extended register occupies one byte; wider accesses are split
491 // above, and the following bytes must not alias this register.
492 if (address != 0)
493 return;
494 // M1543C 4.2.1: channel select 1:0, timing 5:4; all other bits reserved.
495 data &= 0x33;
496 num = (index == DMA1_IO_EXT ? 4 : 0) + (int)(data & 0x03);
497 state.ali.cycle_timing[num] = (u8)(data & 0x30);
498 if (DMA_TRACE_CONTROLLER(index - DMA0_IO_EXT))
499 printf("dma: ALi extended mode channel %d written: %02" PRIx64 " (%s timing)\n",
500 num, data, state.ali.cycle_timing[num] == 0x30 ? "Type F" : "compatible");
501 break;
502
503
504 default:
505 FAILURE(InvalidArgument, "dma: WriteMem index out of range");
506 }
507
508 return;
509 }
510}

518int CDMA::SaveState(FILE* f)
519{
520 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
521 long ss = sizeof(state);
522

526 fwrite(&dma_magic2, sizeof(u32), 1, f) != 1)
527 {
528 printf("dma: error writing state file!\n");
529 return -1;
530 }
531
532 printf("dma: %ld bytes saved.\n", ss);
533 return 0;
534}

539int CDMA::RestoreState(FILE* f)
540{
541 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
542 long ss = 0;
543 u32 m1 = 0;
544 u32 m2 = 0;
545 size_t r;
546 SDMA_state restored = {};
547
548 r = fread(&m1, sizeof(u32), 1, f);
549 if (r != 1)
550 {
551 printf("dma: unexpected end of file!\n");
552 return -1;
553 }
554
555 if (m1 != dma_magic1)
556 {
557 printf("dma: MAGIC 1 does not match!\n");
558 return -1;
559 }
560
561 r = fread(&ss, sizeof(long), 1, f);
562 if (r != 1)
563 {
564 printf("dma: unexpected end of file!\n");
565 return -1;
566 }
567
568 if (ss != sizeof(restored))
569 {
570 printf("dma: STRUCT SIZE does not match!\n");
571 return -1;
572 }
573
574 r = fread(&restored, sizeof(restored), 1, f);
575 if (r != 1)
576 {
577 printf("dma: unexpected end of file!\n");
578 return -1;
579 }
580
581 r = fread(&m2, sizeof(u32), 1, f);
582 if (r != 1)
583 {
584 printf("dma: unexpected end of file!\n");
585 return -1;
586 }
587
588 if (m2 != dma_magic2)
589 {
590 printf("dma: MAGIC 2 does not match!\n");
591 return -1;
592 }
593
594 state = restored;
595 printf("dma: %ld bytes restored.\n", ss);
596 return 0;
597}

615void CDMA::set_drq(int channel, bool asserted)
616{
617 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
618 // Apply the same device-channel validation as the transfer interfaces.
619 validate_device_channel(channel);
620 int ctrlr = channel < 4 ? 0 : 1;
621 int local_channel = channel & 0x03;
622 if (asserted)
623 state.controller[ctrlr].drq |= (1 << local_channel);
624 else
625 {
626 state.controller[ctrlr].drq &= ~(1 << local_channel);
627 // Even a drop followed by reassertion before the next unit releases demand.
628 if (state.controller[ctrlr].demand_active & (1 << local_channel))
629 release_service(channel);
630 }
631 do_dma();
632}

637u8 CDMA::get_requests(int ctrlr)
638{
639 u8 requests = state.controller[ctrlr].request | state.controller[ctrlr].drq;
640 // Controller 0's HRQ drives controller 1's channel-4 request input.
641 // Software and retained block requests bypass the mask; incoming DRQ does not.
642 if (ctrlr == 1 && !(state.controller[0].command & 0x04) &&
643 ((state.controller[0].request | state.controller[0].block_active |
644 (state.controller[0].drq & ~state.controller[0].mask)) & 0x0f))
645 requests |= 0x01;
646 return requests & 0x0f;
647}

649bool CDMA::cascade_enabled() const
650{
651 // These gate the grant, not the request input seen in controller 1's status.
652 return !(state.controller[1].command & 0x04) &&
653 !(state.controller[1].mask & 0x01) &&
654 (state.channel[4].mode & 0xc0) == 0xc0;
655}

660bool CDMA::service_requested(int channel) const
661{
662 validate_device_channel(channel);
663 int ctrlr = channel < 4 ? 0 : 1;
664 u8 bit = 1 << (channel & 0x03);
665 u8 mode = state.channel[channel].mode & 0xc0;
666 // A block keeps its accepted request after either request source drops.
667 if (mode == 0x80 && (state.controller[ctrlr].block_active & bit))
668 return true;
669 // Single and block service accept software requests even while masked.
670 if ((mode == 0x40 || mode == 0x80) && (state.controller[ctrlr].request & bit))
671 return true;
672 if (mode == 0xc0) // Cascade does not service device data.
673 return false;
674 // Demand follows hardware DRQ; a software request cannot keep it running.
675 return (state.controller[ctrlr].drq & ~state.controller[ctrlr].mask & bit) != 0;
676}

681int CDMA::select_service_channel(int ctrlr) const
682{
683 if (state.controller[ctrlr].command & 0x04)
684 return -1;
685
686 u8 requests = 0;
687 int cascade_channel = -1;
688 for (int local_channel = 0; local_channel < 4; local_channel++)
689 {
690 int channel = ctrlr * 4 + local_channel;
691 if (channel == 4)
692 {
693 // Raw status can include requests for unsupported services. Only an
694 // eligible lower winner competes here, not the raw channel-4 input.
695 if (cascade_enabled())
696 cascade_channel = select_service_channel(0);
697 if (cascade_channel >= 0)
698 requests |= 0x01;
699 }
700 else if ((state.channel[channel].mode & 0x0c) != 0x0c &&
701 service_requested(channel))
702 requests |= 1 << local_channel;
703 }
704
705 // Demand/block service is not preempted by a newly arriving request.
706 // In particular, an upper owner must win before a new cascade request.
707 u8 held = state.controller[ctrlr].block_active | state.controller[ctrlr].demand_active;
708 if (ctrlr == 1 && state.cascade_active)
709 held |= 0x01;
710 held &= requests;
711 if (held)
712 requests = held;
713
714 int first = state.controller[ctrlr].command & 0x10 ?
715 state.controller[ctrlr].next_priority : 0;
716 for (int offset = 0; offset < 4; offset++)
717 {
718 int local_channel = (first + offset) & 0x03;
719 if (requests & (1 << local_channel))
720 return ctrlr == 1 && local_channel == 0 ?
721 cascade_channel : ctrlr * 4 + local_channel;
722 }
723 return -1;
724}

726bool CDMA::service_selected(int channel) const

731bool CDMA::service_ready(int channel, u8 transfer_type) const
732{
733 if (!service_selected(channel) ||
734 !(isa_bridge.config_read(0, 0x04, 16) & 0x04))
735 return false;
736
737 // Direction does not affect arbitration against other pending channels.
738 // Verify doesn't move payload bytes.
739 u8 type = state.channel[channel].mode & 0x0c;
740 return type == 0x00 || type == transfer_type;
741}

756void CDMA::release_service(int channel)
757{
758 int ctrlr = channel < 4 ? 0 : 1;
759 u8 bit = 1 << (channel & 0x03);
760 bool held = ((state.controller[ctrlr].block_active |
761 state.controller[ctrlr].demand_active) & bit) != 0;
762 state.controller[ctrlr].block_active &= ~bit;
763 state.controller[ctrlr].demand_active &= ~bit;
764 if (channel == 4)
765 {
766 held = held || state.cascade_active;
767 state.cascade_active = false;
768 }
769 if (!held)
770 return;
771 rotate_priority(channel);
772 if (channel < 4 && state.cascade_active)
773 {
774 state.cascade_active = false;
775 rotate_priority(4);
776 }
777}

782void CDMA::finish_service(int channel, const SDMA_result& result)
783{
784 // Verify services a unit despite transferring zero bytes.
785 if (result.blocked)
786 return;
787 int ctrlr = channel < 4 ? 0 : 1;
788 u8 bit = 1 << (channel & 0x03);
789 u8 mode = state.channel[channel].mode & 0xc0;
790 if (result.terminal_count || result.external_eop || mode == 0x40 ||
791 (mode == 0x00 && !(state.controller[ctrlr].drq & bit)))
792 {
793 release_service(channel);
794 // Completion may already have recorded a previously held service.
795 // These idempotent assignments also cover a first unit ending at TC/EOP
796 // and a lower service that has just reacquired its upper cascade grant.
797 rotate_priority(channel);
798 if (channel < 4)
799 rotate_priority(4);
800 return;
801 }
802 if (mode == 0x80)
803 state.controller[ctrlr].block_active |= bit;
804 else if (mode == 0x00)
805 state.controller[ctrlr].demand_active |= bit;
806 if (channel < 4)
807 state.cascade_active = true;
808}

874bool CDMA::advance_transfer(int channel, size_t length,
875 const SDMA_transfer_properties& properties, bool eop)
876{
877 size_t units = length / properties.data_width;
878 size_t count_units = length / properties.count_unit;
879 // Check before narrowing: a full transfer can contain 65536 units.
880 bool terminal_count = count_units == (size_t)state.channel[channel].count + 1;
881
882 state.channel[channel].current = (u16)step_address(state.channel[channel].current,
883 units, (state.channel[channel].mode & 0x20) != 0, properties);
884 state.channel[channel].count -= (u16)count_units;
885
886 if (terminal_count || eop)
887 complete_transfer(channel);
888 return terminal_count;
889}

894void CDMA::complete_transfer(int channel)
895{
896 int ctrlr = channel < 4 ? 0 : 1;
897 int local_channel = channel & 0x03;
898 state.controller[ctrlr].status |= 1 << local_channel;
899 // TC and EOP clear the software request; the device owns its DRQ level.
900 state.controller[ctrlr].request &= ~(1 << local_channel);
901 release_service(channel);
902 if (state.channel[channel].mode & 0x10)
903 {
904 state.channel[channel].current = state.channel[channel].base;
905 state.channel[channel].count = state.channel[channel].base_count;
906 }
907 else
908 state.controller[ctrlr].mask |= 1 << local_channel;
909}

914CDMA::SDMA_result CDMA::send_data(int channel, void* data, size_t length, bool eop)
915{
916 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
917 const SDMA_transfer_properties properties = transfer_properties(channel);
918 int ctrlr = channel < 4 ? 0 : 1;
919 int local_channel = channel & 0x03;
920 SDMA_result result = { 0, true, false, false };
921 // Cascade passes bus control to another controller, not buffer data.
922 if ((state.channel[channel].mode & 0xc0) == 0xc0)
923 {
924 if (DMA_TRACE_CHANNEL(channel))
925 printf("dma: send on channel %d blocked by cascade mode.\n", channel);
926 return result;
927 }
928 if (channel < 4 && !cascade_enabled())
929 {
930 if (DMA_TRACE_CHANNEL(channel))
931 printf("dma: send on channel %d blocked by channel 4: command %02x, mask %02x, mode %02x.\n",
932 channel, state.controller[1].command, state.controller[1].mask,
933 state.channel[4].mode);
934 return result;
935 }
936
937 // The PCI helpers otherwise silently skip DMA when bus mastering is disabled.
938 if (!(isa_bridge.config_read(0, 0x04, 16) & 0x04))
939 return result;
940
941 if ((state.controller[ctrlr].command & 0x04) == 0)
942 {
943 // The mask inhibits incoming DRQ, not software or retained block requests.
944 if ((state.controller[ctrlr].mask & (1 << local_channel)) == 0 ||
945 ((state.controller[ctrlr].request | state.controller[ctrlr].block_active) &
946 (1 << local_channel)))
947 {
948 u8 transfer_type = state.channel[channel].mode & 0x0c;
949 // 8237 Write means device to memory; Verify has no data direction.
950 if (transfer_type != 0x00 && transfer_type != 0x04)
951 {
952 if (DMA_TRACE_CHANNEL(channel))
953 printf("dma: send on channel %d blocked by transfer type %02x.\n",
954 channel, transfer_type);
955 return result;
956 }
957 size_t count = get_transfer_size(channel);
958 if (length > 0 && length < count) count = length;
959 if (count % properties.data_width)
960 FAILURE(InvalidArgument, "dma: word-channel transfer length must be even");
961
962 if (transfer_type == 0x00)
963 {
964 if (DMA_TRACE_CHANNEL(channel))
965 printf("dma: verify on channel %d: %zx bytes.\n", channel, count);
966 result.terminal_count = advance_transfer(channel, count, properties, eop);
967 result.external_eop = eop;
968 result.blocked = false;
969 return result;
970 }
971
972 u64 addr = dma_address(state.channel[channel].pagebase,
973 state.channel[channel].current, properties);
974 if (DMA_TRACE_CHANNEL(channel))
975 {
976 printf("DMA send_data: %zx @ %16" PRIx64 "\n ", count, addr);
977 for (size_t i = 0; i < count; i++)
978 {
979 printf("%02x ", *((char*)data + i) & 0xff);
980 if (i % 16 == 15)
981 printf("\n ");
982 }
983 printf("\n");
984 }
985
986 // Device buffers are byte streams, even for word channels.
987 if (state.channel[channel].mode & 0x20)
988 {
989 u32 current = state.channel[channel].current;
990 // Decrement by units, preserving the byte order within each word.
991 for (size_t offset = 0; offset < count; offset += properties.data_width)
992 {
993 addr = dma_address(state.channel[channel].pagebase, current, properties);
994 isa_bridge.do_pci_write((u32)addr, (u8*)data + offset, 1, properties.data_width);
995 current = step_address(current, 1, true, properties);
996 }
997 }
998 else
999 {
1000 size_t first_count = increment_chunk_size(state.channel[channel].current,
1001 count, properties);
1002 isa_bridge.do_pci_write((u32)addr, data, 1, first_count);
1003 // A programmed transfer contains at most 65536 units, so only one wrap is possible.
1004 if (first_count < count)
1005 {
1006 u64 wrap_addr = dma_address(state.channel[channel].pagebase, 0, properties);
1007 isa_bridge.do_pci_write((u32)wrap_addr, (u8*)data + first_count, 1,
1008 count - first_count);
1009 }
1010 }
1011 result.terminal_count = advance_transfer(channel, count, properties, eop);
1012 result.external_eop = eop;
1013 result.transferred = count;
1014 result.blocked = false;
1015 }
1016 else if (DMA_TRACE_CHANNEL(channel))
1017 {
1018 printf("dma: dma requested by device on channel %d, but it is masked.\n", channel);
1019 }
1020 }
1021 else if (DMA_TRACE_CONTROLLER(ctrlr))
1022 {
1023 printf("dma: dma requested by device, but controller %d is disabled.\n", ctrlr);
1024 }
1025 return result;
1026}

1171CDMA::SDMA_result CDMA::service_eop(int channel)
1172{
1173 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
1174 validate_device_channel(channel);
1175 SDMA_result result = { 0, true, false, false };
1176 int ctrlr = channel < 4 ? 0 : 1;
1177 u8 bit = 1 << (channel & 0x03);
1178 u8 mode = state.channel[channel].mode & 0xc0;
1179 // EOP cannot acquire service or terminate a channel while it is idle.
1180 if (state.controller[ctrlr].command & 0x04)
1181 return result;
1182 if (!((mode == 0x00 && (state.controller[ctrlr].demand_active & bit)) ||
1183 (mode == 0x80 && (state.controller[ctrlr].block_active & bit))))
1184 return result;
1185 // A retained lower request may have lost its accepted upper grant.
1186 if (channel < 4 && (!state.cascade_active || !cascade_enabled()))
1187 return result;
1188
1189 // No new memory cycle or arbitration: complete only the accepted service.
1190 complete_transfer(channel);
1191 result.blocked = false;
1192 result.external_eop = true;
1193 return result;
