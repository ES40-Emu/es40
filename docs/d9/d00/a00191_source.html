197 /* Register format constants */
198#define QNAN U64(0x0008000000000000) /* quiet NaN flag */
199#define CQNAN U64(0xFFF8000000000000) /* canonical quiet NaN */
200#define FPZERO U64(0x0000000000000000) /* plus zero (fp) */
201#define FMZERO U64(0x8000000000000000) /* minus zero (fp) */
202#define FPINF U64(0x7FF0000000000000) /* plus infinity (fp) */
203#define FMINF U64(0xFFF0000000000000) /* minus infinity (fp) */
204#define FPMAX_T U64(0x7FEFFFFFFFFFFFFF) /* plus MAX (T) */
205#define FPMAX_S U64(0x47EFFFFFE0000000) /* plus MAX (S, register format) */
206#define IPMAX U64(0x7FFFFFFFFFFFFFFF) /* plus MAX (int) */
207#define IMMAX U64(0x8000000000000000) /* minus MAX (int) */
208#define FPSLOW U64(0x000000001FFFFFFF) /* S lower bits <28:0> */
209
210/* Unpacked rounding constants */
211#define UF_SRND U64(0x0000008000000000) /* S normal round */
212#define UF_SINF U64(0x000000FFFFFFFFFF) /* S infinity round */
213#define UF_TRND U64(0x0000000000000400) /* T normal round */
214#define UF_TINF U64(0x00000000000007FF) /* T infinity round */

225static inline u64 ieee_quiet_nan_result(u64 op, u32 dp)
226{
227 op |= QNAN;
228 if (dp == DT_S) /* AAR 4.7.10.1 */
229 op &= ~FPSLOW;
230 return op;

241u64 CAlphaCPU::ieee_lds(u32 op)
242{
243 u32 exp = S_GETEXP(op); /* get exponent */
244
245 if (exp == S_NAN)
246 exp = FPR_NAN; /* inf or NaN? */
247 else if (exp != 0)
248 exp = exp + T_BIAS - S_BIAS; /* zero or denorm? */
249 return(((u64)(op & S_SIGN)) ? FPR_SIGN : 0) | /* reg format */
250 (((u64)exp) << FPR_V_EXP) | (((u64)(op & ~(S_SIGN | S_EXP))) << S_V_FRAC);
251}

262{
263 return (u32)((((op >> 62) & 0x3) << 30) | ((op >> 29) & 0x3FFFFFFF));
264}

287u64 CAlphaCPU::ieee_cvtst(u64 op, u32 ins)
288{
289 UFP b;
290
291 /* S-denormal conversion requires normalization and an exponent rebias
292 to T format (ARM 4.10.14). This describes the numerical conversion;
293 denormal-operand trap handling is a separate requirement. */
294 if (FPR_GETEXP(op) == 0 && FPR_GETFRAC(op) != 0
295 && !(state.fpcr & FPCR_DNZ))
296 {
297 b.sign = FPR_GETSIGN(op);
298 b.exp = 1; /* encoded exponent zero means 1 - S_BIAS (ARM 2.2.6.1) */
299 b.frac = FPR_GETFRAC(op) << FPR_GUARD;
300 ieee_norm(&b); /* renormalize denormal */
301 b.exp = b.exp + T_BIAS - S_BIAS; /* rebias S-exp to T-exp */
302 return ieee_rpack(&b, ins, DT_T);
303 }
304
305 /* For zero / normal / infinity / NaN the S-format bit pattern is
306 already a valid T value. ieee_unpack raises INV on sNaN; we must
307 also silence the NaN to qNaN on the result (HRM 4.8.4 / IEEE-754
308 7.2: an invalid-op result that returns a NaN must set the QNAN bit).
309 Mirrors the sibling helper ieee_cvtts. */
310 u32 ftpb = ieee_unpack(op, &b, ins);
311 if (ftpb == UFT_NAN)
312 return op | QNAN;
313 if (ftpb == UFT_ZERO)
314 return op & FPR_SIGN; /* DNZ flushed a denormal to its signed zero (AAR 4.7.7.11) */
315 return op;
316}

326u64 CAlphaCPU::ieee_cvtts(u64 op, u32 ins)
327{
328 UFP b;
329 u32 ftpb;
330
331 ftpb = ieee_unpack(op, &b, ins); /* unpack */
332 if (Q_FINITE(ftpb))
333 return ieee_rpack(&b, ins, DT_S); /* finite? round, pack */
334 if (ftpb == UFT_NAN)
335 return (op | QNAN) & ~FPSLOW; /* nan? cvt to quiet */
336 if (ftpb == UFT_INF)
337 return op; /* inf? unchanged */
338 return 0; /* denorm? 0 */
339}

367s32 CAlphaCPU::ieee_fcmp(u64 s1, u64 s2, u32 ins, u32 trap_nan)
368{
369 UFP a;
370
371 UFP b;
372 u32 ftpa;
373 u32 ftpb;
374
375 ftpa = ieee_unpack(s1, &a, ins);
376 ftpb = ieee_unpack(s2, &b, ins);
377 if ((ftpa == UFT_NAN) || (ftpb == UFT_NAN))
378 { /* NaN involved? */
379 if (trap_nan)
380 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins);
381 return 2; // FIXED: Return 2 for unordered, not +1
382 } /* force failure */
383
384 if (ftpa == UFT_ZERO)
385 a.sign = 0; /* only +0 allowed */
386 if (ftpb == UFT_ZERO)
387 b.sign = 0;
388 if (a.sign != b.sign)
389 return(a.sign ? -1 : +1); /* unequal signs? */
390 if (a.exp != b.exp)
391 return((a.sign ^ (a.exp < b.exp)) ? -1 : +1);
392 if (a.frac != b.frac)
393 return((a.sign ^ (a.frac < b.frac)) ? -1 : +1);
394 return 0;
395}

406u64 CAlphaCPU::ieee_cvtif(u64 val, u32 ins, u32 dp)
407{
408 UFP a;
409
410 if (val == 0)
411 return 0; /* 0? return +0 */
412 if (((s64)val) < 0)
413 { /* < 0? */
414 a.sign = 1; /* set sign */
415 val = NEG_Q(val);
416 } /* |val| */
417 else
418 a.sign = 0;
419 a.exp = 63 + T_BIAS; /* set exp */
420 a.frac = val; /* set frac */
421 ieee_norm(&a); /* normalize */
422 return ieee_rpack(&a, ins, dp); /* round and pack */
423}

439u64 CAlphaCPU::ieee_cvtfi(u64 op, u32 ins)
440{
441 UFP a;
442 u64 sticky;
443 u32 rndm;
444 u32 ftpa;
445 u32 ovf = 0;
446 s32 ubexp;
447
448 ftpa = ieee_unpack(op, &a, ins); /* unpack */
449 if (!Q_FINITE(ftpa)) /* inf, NaN, dnorm? */
450 {
451 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* inv operation */
452 return 0;
453 }
454
455 if (ftpa == UFT_ZERO)
456 return 0; /* zero? */
457 ubexp = a.exp - T_BIAS; /* unbiased exp */
458 if (ubexp < 0) /* < 1? */
459 {
460 if (ubexp == -1)
461 sticky = a.frac; /* [.5,1)? */
462 else
463 sticky = 1; /* (0,.5) */
464 a.frac = 0;
465 }
466 else if (ubexp <= UF_V_NM) /* in range? */
467 {
468 /* When ubexp == UF_V_NM (63), the shift amount (UF_V_NM - ubexp) is
469 zero and (64 - 0) is 64, which is undefined behavior for `<< 64`
470 on a 64-bit value. The architectural meaning at the boundary is
471 "no bits shifted out" — the value is exact in integer form. */
472 s32 sh = UF_V_NM - ubexp;
473 sticky = sh ? ((a.frac << (64 - sh)) & X64_QUAD) : 0;
474 a.frac = a.frac >> sh; /* result */
475 }
476 else
477 {
478 if ((ubexp - UF_V_NM) > 63)
479 a.frac = 0; /* out of range */
480 else
481 a.frac = (a.frac << (ubexp - UF_V_NM)) & X64_QUAD;
482 ovf = 1; /* overflow */
483 sticky = 0; /* no rounding */
484 }
485
486 rndm = I_GETFRND(ins); /* get round mode */
487 if (rndm == I_FRND_D)
488 rndm = FPCR_GETFRND(state.fpcr); /* dynamic? use FPCR<DYN> */
489 if (((rndm == I_FRND_N) && (sticky & Q_SIGN)) /* nearest? */
490 || ((rndm == I_FRND_P) && !a.sign && sticky) /* +inf and +? */
491 || ((rndm == I_FRND_M) && a.sign && sticky)) /* -inf and -? */
492 {
493 a.frac = (a.frac + 1) & X64_QUAD;
494 if (a.frac == 0)
495 ovf = 1; /* overflow? */
496 if ((rndm == I_FRND_N) && (sticky == Q_SIGN)) /* round nearest hack */
497 a.frac = a.frac & ~1;
498 }
499
500 if (a.frac > (a.sign ? IMMAX : IPMAX))
501 ovf = 1; /* overflow? */
502
503 if (ovf)
504 /* Pass `ins` (not 0) so I_GETRC(ins) reports the real destination
505 register and ins & I_FTRP_S sets TRAP_SWC correctly on /S. */
506 ieee_trap(TRAP_IOV, ins & I_FTRP_V, 0, ins); /* overflow trap */
507 if (ovf || sticky) /* ovflo or round? */
508 ieee_trap(TRAP_INE, Q_SUI(ins), FPCR_INED, ins);
509 return(a.sign ? NEG_Q(a.frac) : a.frac);
510}

540u64 CAlphaCPU::ieee_fadd(u64 s1, u64 s2, u32 ins, u32 dp, bool sub)
541{
542 UFP a;
543
544 UFP b;
545
546 UFP t;
547 u32 ftpa;
548 u32 ftpb;
549 u32 sticky;
550 s32 ediff;
551
552 ftpa = ieee_unpack(s1, &a, ins); /* unpack operands */
553 ftpb = ieee_unpack(s2, &b, ins);
554 if (ftpb == UFT_NAN)
555 return ieee_quiet_nan_result(s2, dp); /* B = NaN? quiet B */
556 if (ftpa == UFT_NAN)
557 return ieee_quiet_nan_result(s1, dp); /* A = NaN? quiet A */
558 if (sub)
559 b.sign = b.sign ^ 1; /* sign of B */
560 if (ftpb == UFT_INF)
561 { /* B = inf? */
562 if ((ftpa == UFT_INF) && (a.sign ^ b.sign))
563 { /* eff sub of inf? */
564 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* inv op trap */
565 return CQNAN;
566 } /* canonical NaN */
567
568 return(sub ? (s2 ^ FPR_SIGN) : s2);
569 } /* return B */
570
571 if (ftpa == UFT_INF)
572 return s1; /* A = inf? ret A */
573 if (ftpa == UFT_ZERO)
574 {
575 if (ftpb == UFT_ZERO && (a.sign != b.sign))
576 {
577 /* Both operands zero, opposite effective signs (e.g. +0 + -0,
578 or +0 - +0). IEEE-754 6.3: sign is '+' except under
579 roundTowardNegative where it is '-'. */
580 u32 rndm = I_GETFRND(ins);
581 if (rndm == I_FRND_D) rndm = FPCR_GETFRND(state.fpcr);
582 a.sign = (rndm == I_FRND_M) ? 1 : 0;
583 a.exp = 0; a.frac = 0;
584 }
585 else
586 {
587 a = b; /* s1 = 0, result is b */
588 }
589 }
590 else if (ftpb != UFT_ZERO)
591 { /* s2 != 0? */
592 if ((a.exp < b.exp) /* s1 < s2? swap */
593 || ((a.exp == b.exp) && (a.frac < b.frac)))
594 {
595 t = a;
596 a = b;
597 b = t;
598 }
599
600 ediff = a.exp - b.exp; /* exp diff */
601 if (ediff > 63)
602 b.frac = 1; /* >63? retain sticky */
603 else if (ediff)
604 { /* [1,63]? shift */
605 sticky = ((b.frac << (64 - ediff)) & X64_QUAD) ? 1 : 0; /* lost bits */
606 b.frac = ((b.frac >> ediff) & X64_QUAD) | sticky;
607 }
608
609 if (a.sign ^ b.sign)
610 { /* eff sub? */
611 a.frac = (a.frac - b.frac) & X64_QUAD; /* subtract fractions */
612 ieee_norm(&a);
613 if (a.frac == 0)
614 {
615 /* Exact cancellation. IEEE-754 6.3: sign is '+' except
616 under roundTowardNegative where it is '-'. */
617 u32 rndm = I_GETFRND(ins);
618 if (rndm == I_FRND_D) rndm = FPCR_GETFRND(state.fpcr);
619 a.sign = (rndm == I_FRND_M) ? 1 : 0;
620 }
621 } /* normalize */
622 else
623 { /* eff add */
624 a.frac = (a.frac + b.frac) & X64_QUAD; /* add frac */
625 if (a.frac < b.frac)
626 { /* chk for carry */
627 a.frac = UF_NM | (a.frac >> 1) | /* shift in carry */
628 (a.frac & 1); /* retain sticky */
629 a.exp = a.exp + 1;
630 }
631 } /* skip norm */
632 } /* end else if */
633
634 return ieee_rpack(&a, ins, dp); /* round and pack */
635}

661u64 CAlphaCPU::ieee_fmul(u64 s1, u64 s2, u32 ins, u32 dp)
662{
663 UFP a;
664
665 UFP b;
666 u32 ftpa;
667 u32 ftpb;
668 u64 resl;
669
670 ftpa = ieee_unpack(s1, &a, ins); /* unpack operands */
671 ftpb = ieee_unpack(s2, &b, ins);
672 if (ftpb == UFT_NAN)
673 return ieee_quiet_nan_result(s2, dp); /* B = NaN? quiet B */
674 if (ftpa == UFT_NAN)
675 return ieee_quiet_nan_result(s1, dp); /* A = NaN? quiet A */
676 a.sign = a.sign ^ b.sign; /* sign of result */
677 if ((ftpa == UFT_ZERO) || (ftpb == UFT_ZERO))
678 { /* zero operand? */
679 if ((ftpa == UFT_INF) || (ftpb == UFT_INF))
680 { /* 0 * inf? */
681 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* inv op trap */
682 return CQNAN;
683 } /* canonical NaN */
684
685 return(a.sign ? FMZERO : FPZERO);
686 } /* return signed 0 */
687
688 if (ftpb == UFT_INF)
689 return(a.sign ? FMINF : FPINF); /* B = inf? */
690 if (ftpa == UFT_INF)
691 return(a.sign ? FMINF : FPINF); /* A = inf? */
692 a.exp = a.exp + b.exp + 1 - T_BIAS; /* add exponents */
693 resl = uemul64(a.frac, b.frac, &a.frac); /* multiply fracs */
694 ieee_norm(&a); /* normalize */
695 a.frac = a.frac | (resl ? 1 : 0); /* sticky bit */
696 return ieee_rpack(&a, ins, dp); /* round and pack */
697}

720u64 CAlphaCPU::ieee_fdiv(u64 s1, u64 s2, u32 ins, u32 dp)
721{
722 UFP a;
723
724 UFP b;
725 u32 ftpa;
726 u32 ftpb;
727 u32 sticky;
728
729 ftpa = ieee_unpack(s1, &a, ins);
730 ftpb = ieee_unpack(s2, &b, ins);
731 if (ftpb == UFT_NAN)
732 return ieee_quiet_nan_result(s2, dp); /* B = NaN? quiet B */
733 if (ftpa == UFT_NAN)
734 return ieee_quiet_nan_result(s1, dp); /* A = NaN? quiet A */
735 a.sign = a.sign ^ b.sign; /* sign of result */
736 if (ftpb == UFT_INF)
737 { /* B = inf? */
738 if (ftpa == UFT_INF)
739 { /* inf/inf? */
740 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* inv op trap */
741 return CQNAN;
742 } /* canonical NaN */
743
744 return(a.sign ? FMZERO : FPZERO);
745 } /* !inf/inf, ret 0 */
746
747 if (ftpa == UFT_INF)
748 { /* A = inf? */
749 /* inf/0 returns +/-inf with NO exception per IEEE-754 7.3:
750 DZE is signaled only when the dividend is finite non-zero.
751 Previously raised DZE here, breaking inf-arithmetic in libm. */
752 return(a.sign ? FMINF : FPINF);
753 } /* return inf */
754
755 if (ftpb == UFT_ZERO)
756 { /* B = 0? */
757 if (ftpa == UFT_ZERO)
758 { /* 0/0? */
759 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* inv op trap */
760 return CQNAN;
761 } /* canonical NaN */
762
763 ieee_trap(TRAP_DZE, 1, FPCR_DZED, ins); /* div by 0 trap */
764 return(a.sign ? FMINF : FPINF);
765 } /* return inf */
766
767 if (ftpa == UFT_ZERO)
768 return(a.sign ? FMZERO : FPZERO); /* A = 0? */
769 a.exp = a.exp - b.exp + T_BIAS; /* unbiased exp */
770 a.frac = a.frac >> 1; /* allow 1 bit left */
771 b.frac = b.frac >> 1;
772 a.frac = ufdiv64(a.frac, b.frac, 55, &sticky); /* divide */
773 ieee_norm(&a); /* normalize */
774 a.frac = a.frac | sticky; /* insert sticky */
775 return ieee_rpack(&a, ins, dp); /* round and pack */
776}

794u64 CAlphaCPU::ieee_sqrt(u64 op, u32 ins, u32 dp)
795{
796 u32 ftpb;
797 UFP b;
798
799 ftpb = ieee_unpack(op, &b, ins); /* unpack */
800 if (ftpb == UFT_NAN)
801 return ieee_quiet_nan_result(op, dp); /* NaN? */
802 if ((ftpb == UFT_ZERO) || /* zero? */ ((ftpb == UFT_INF) && !b.sign))
803 return op; /* +infinity? */
804 if (b.sign)
805 { /* minus? */
806 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* signal inv op */
807 return CQNAN;
808 }
809
810 const s32 in_exp = b.exp; /* fsqrt64 keys on INPUT exp parity */
811 b.exp = ((b.exp - T_BIAS) >> 1) + T_BIAS; /* result exponent */
812 b.frac = fsqrt64(b.frac, in_exp); /* result fraction */
813 return ieee_rpack(&b, ins, dp); /* round and pack */
814}

840int CAlphaCPU::ieee_unpack(u64 op, UFP* r, u32 ins)
841{
842 r->sign = FPR_GETSIGN(op); /* get sign */
843 r->exp = FPR_GETEXP(op); /* get exponent */
844 r->frac = FPR_GETFRAC(op); /* get fraction */
845 if (r->exp == 0) /* exponent = 0? */
846 {
847 if (r->frac == 0)
848 return UFT_ZERO; /* frac = 0? then true 0 */
849 if (state.fpcr & FPCR_DNZ) /* denorms to 0? */
850 {
851 r->frac = 0; /* clear fraction */
852 return UFT_ZERO;
853 }
854
855 r->exp = 1; /* encoded exponent zero means 1 - T_BIAS (ARM 2.2.6.2) */
856 r->frac = r->frac << FPR_GUARD; /* guard fraction */
857 ieee_norm(r); /* normalize dnorm */
858 /* Denormal input operands take an *unmaskable* trap when
859 FPCR[DNZ] is clear (HRM A.11: "denormal input operands for
860 arithmetic operations produce an unmaskable denormal trap").
861 Pass fpcrdsb = 0 so neither the /S qualifier nor FPCR[INVD]
862 can suppress it -- same idiom as the unmaskable integer-
863 overflow trap in ieee_cvtfi(). */
864 ieee_trap(TRAP_INV, 1, 0, ins); /* signal denormal-operand trap */
865 return UFT_DENORM;
866 }
867
868 if (r->exp == FPR_NAN) /* exponent = max? */
869 {
870 if (r->frac == 0)
871 return UFT_INF; /* frac = 0? then inf */
872 if (!(r->frac & QNAN)) /* signaling NaN? */
873 ieee_trap(TRAP_INV, 1, FPCR_INVD, ins); /* signal inv op */
874 return UFT_NAN;
875 }
876
877 r->frac = (r->frac | FPR_HB) << FPR_GUARD; /* ins hidden bit, guard */
878 return UFT_FIN; /* finite */
879}

890void CAlphaCPU::ieee_norm(UFP* r)
891{
892 s32 i;
893 static u64 normmask[5] = {
894 U64(0xc000000000000000), U64(0xf000000000000000), U64(0xff00000000000000),
895 U64(0xffff000000000000), U64(0xffffffff00000000) };
896 static s32 normtab[6] = { 1, 2, 4, 8, 16, 32 };
897
898 r->frac = r->frac & X64_QUAD;
899 if (r->frac == 0)
900 { /* if fraction = 0 */
901 r->exp = 0; /* result is signed 0 */
902 return;
903 }
904
905 while ((r->frac & UF_NM) == 0)
906 { /* normalized? */
907 for (i = 0; i < 5; i++)
908 { /* find first 1 */
909 if (r->frac & normmask[i])
910 break;
911 }
912
913 r->frac = r->frac << normtab[i]; /* shift frac */
914 r->exp = r->exp - normtab[i];
915 } /* decr exp */
916
917 return;
918}

952u64 CAlphaCPU::ieee_rpack(UFP* r, u32 ins, u32 dp)
953{
954 static const u64 stdrnd[2] = { UF_SRND, UF_TRND };
955 static const u64 infrnd[2] = { UF_SINF, UF_TINF };
956 static const s32 expmax[2] = { T_BIAS - S_BIAS + S_M_EXP - 1, T_M_EXP - 1 };
957 static const s32 expmin[2] = { T_BIAS - S_BIAS, 0 };
958 u64 rndadd;
959 u64 rndbits;
960 u64 res;
961 u64 rndm;
962
963 if (r->frac == 0)
964 return(((u64)r->sign) << FPR_V_SIGN); /* result 0? */
965 rndm = I_GETFRND(ins); /* inst round mode */
966 if (rndm == I_FRND_D)
967 rndm = FPCR_GETFRND(state.fpcr); /* dynamic? use FPCR */
968 rndbits = r->frac & infrnd[dp]; /* isolate round bits */
969 if (rndm == I_FRND_N)
970 rndadd = stdrnd[dp]; /* round to nearest? */
971 else if (((rndm == I_FRND_P) && !r->sign) /* round to inf and */
972 || ((rndm == I_FRND_M) && r->sign)) /* right sign? */
973 rndadd = infrnd[dp];
974 else
975 rndadd = 0;
976 r->frac = (r->frac + rndadd) & X64_QUAD; /* round */
977 if ((r->frac & UF_NM) == 0)
978 { /* carry out? */
979 r->frac = (r->frac >> 1) | UF_NM; /* renormalize */
980 r->exp = r->exp + 1;
981 }
982
983 if (rndbits) /* inexact? */
984 ieee_trap(TRAP_INE, Q_SUI(ins), FPCR_INED, ins); /* set inexact */
985 if (r->exp > expmax[dp])
986 { /* ovflo? */
987 ieee_trap(TRAP_OVF, 1, FPCR_OVFD, ins); /* set overflow trap */
988 ieee_trap(TRAP_INE, Q_SUI(ins), FPCR_INED, ins); /* set inexact */
989 if (rndadd) /* did we round? */
990 return(r->sign ? FMINF : FPINF); /* return infinity */
991 /* ARM 4.7.7.6: use the destination format's largest finite value. */
992 return (((u64)r->sign) << FPR_V_SIGN) | (dp == DT_S ? FPMAX_S : FPMAX_T);
993 } /* no, return max */
994
995 if (r->exp <= expmin[dp])
996 { /* underflow? */
997 /* UNFD (with /S) disables the underflow trap; UNDZ separately controls
998 flush-to-zero result. The two bits are independent per HRM 4.7.7.1. */
999 ieee_trap(TRAP_UNF, ins & I_FTRP_U, FPCR_UNFD, ins);
1000 ieee_trap(TRAP_INE, Q_SUI(ins), FPCR_INED, ins); /* set inexact */
1001 /* Preserve sign of zero on underflow per IEEE-754 / HRM 4.7.7.3. */
1002 return ((u64)r->sign) << FPR_V_SIGN;
1003 } /* underflow to signed zero */
1004
1005 res = (((u64)r->sign) << FPR_V_SIGN) | /* form result */
1006 (((u64)r->exp) << FPR_V_EXP) | ((r->frac >> FPR_GUARD) & FPR_FRAC);
1007 if ((rndm == I_FRND_N) && (rndbits == stdrnd[dp])) /* halfway -> round to even (S LSB=bit29, T LSB=bit0) */
1008 res &= ~(dp == DT_S ? U64(0x0000000020000000) : U64(1));
1009 if (dp == DT_S) /* S_floating canonical: fraction <28:0> zero (ARM Fig 2-12) */
1010 res &= ~U64(0x000000001FFFFFFF);
1011 return res;
1012}

1027void CAlphaCPU::ieee_trap(u64 trap, u32 instenb, u64 fpcrdsb, u32 ins)
1028{
1029 u64 real_trap = U64(0x0);
1030
1031 /* HRM 4.7.6: the FPCR sticky bit is set whenever the corresponding
1032 exception is detected, regardless of whether a trap is delivered. */
1033 if (~state.fpcr & (trap << 51)) // trap bit not set in FPCR
1034 {
1035 state.fpcr |= trap << 51; // set FPCR sticky bit (newly detected)
1036 real_trap |= trap << 41; // SET trap bit in EXC_SUM
1037 }
1038 if ((instenb != 0) /* not enabled in inst? ignore */
1039 && !((ins & I_FTRP_S) && (state.fpcr & fpcrdsb))) /* /S and disabled? ignore */real_trap |= trap; // trap bit in EXC_SUM
1040 if (real_trap)
1041 ARITH_TRAP(real_trap | ((ins & I_FTRP_S) ? TRAP_SWC : 0), I_GETRC(ins));
1042 return;
1043}
