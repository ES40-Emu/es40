34constexpr AlphaHWStoreDescriptor decode(std::uint32_t instruction)
35{
36 AlphaHWStoreDescriptor store{};
37 if ((instruction >> 26) != 0x1f)
38 return store;
39
40 // EV68 HW_ST TYPE[15:12]: bit 0 selects quadword; only 0..5/12/13 exist.
41 const unsigned form = (instruction >> 12) & 0xf;
42 store.size_bits = (form & 1) ? 64 : 32;
43 store.displacement = static_cast<std::int32_t>(instruction & 0xfff);
44 if (store.displacement & 0x800)
45 store.displacement -= 0x1000;
46 if (form > 5 && form != 12 && form != 13)
47 return store;
48
49 store.address = form < 4 ? AlphaHWStoreAddress::Physical
50 : AlphaHWStoreAddress::Virtual;
51 store.alternate = form == 12 || form == 13;
52 store.conditional = form == 2 || form == 3;
53 // Virtual and conditional forms retain their interpreter fallback.
54 store.compilable = form == 0 || form == 1;
55 return store;
56}
57}
