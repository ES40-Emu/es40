78// Decoded PAL return rules, outside saved CPU state. Read the selected source
79// at execution time; clear the reservation after selecting the target.
80struct AlphaPALReturnDescriptor
81{
82 bool valid = false;
83 std::uint64_t target_mask = ~std::uint64_t(0);
84 std::uint64_t target_bits = 0;
85 bool clear_lock = false;
86 bool compilable = false;
87 AlphaPALReturnSource source = AlphaPALReturnSource::RegisterB;
88#if defined(IDB)
89 const char* mnemonic = nullptr; // Use the opcode name by default.
90#endif
91};
92
93constexpr bool alpha_pal_return_valid(const AlphaPALReturnDescriptor& entry)
94{
95 return entry.valid && (entry.source == AlphaPALReturnSource::RegisterB
96 || entry.source == AlphaPALReturnSource::ExceptionAddress);
97}
98
99constexpr std::uint64_t alpha_pal_return_target(
100 const AlphaPALReturnDescriptor& entry, std::uint64_t register_b,
101 std::uint64_t exception_address)
102{
103 const std::uint64_t source = entry.source == AlphaPALReturnSource::ExceptionAddress
104 ? exception_address : register_b;
105 return (source & entry.target_mask) | entry.target_bits;
106}
107
108constexpr bool alpha_pal_return_compilable(const AlphaPALReturnDescriptor& entry)
109{
110 return alpha_pal_return_valid(entry) && entry.compilable;
111}

113// Family access rules for the PAL-reserved hardware instructions.
114enum class AlphaPALInstructionAccess
115{
116 Denied,
117 PalOnly,
118 PalOrKernelHwe
119};
120
121constexpr bool alpha_pal_instruction_allowed(AlphaPALInstructionAccess access,
122 bool pal_mode, std::uint32_t mode,
123 bool hwe)
124{
125 switch (access)
126 {
127 case AlphaPALInstructionAccess::Denied: return false;
128 case AlphaPALInstructionAccess::PalOnly: return pal_mode;
129 case AlphaPALInstructionAccess::PalOrKernelHwe:
130 return pal_mode || (mode == 0 && hwe);
131 }
132 return false;
133}
134
135// These rules guarantee access throughout a PAL block.
136// Native HWE execution stays interpreted.
137constexpr bool alpha_pal_instruction_compilable(AlphaPALInstructionAccess access)
138{
139 return access == AlphaPALInstructionAccess::PalOnly
140 || access == AlphaPALInstructionAccess::PalOrKernelHwe;
141}

149// True consumes the call; false continues normal dispatch.
150using AlphaPALCallHandler = bool (*)(CAlphaCPU& cpu);

152// Family PAL rules, copied with the CPU profile.
153struct AlphaPALPolicy
154{
155 AlphaPALExceptionEntry (*exception_entry)(AlphaPALException reason,
156 std::uint64_t current_pc);
157 bool (*call_pal_valid)(std::uint32_t function, std::uint32_t mode);
158 AlphaPALCallEntry (*call_pal_entry)(std::uint32_t function,
159 std::uint64_t current_pc,
160 std::uint64_t next_pc);
161 // Pass the complete instruction so restrictions on individual forms apply.
162 // An absent access rule denies all.
163 AlphaPALInstructionAccess (*hardware_instruction_access)(std::uint32_t instruction) = nullptr;
164 AlphaPALReturnDescriptor (*hardware_return)(std::uint32_t instruction) = nullptr;
165 // Optional compatibility handling after the base write and HLE reset.
166 void (*base_changed)(CAlphaCPU& cpu, std::uint64_t old_base, bool old_hle) = nullptr;
167 // An absent context is unprivileged, with hardware access disabled.
168 AlphaPALAccessContext (*access_context)(const CAlphaCPU& cpu) = nullptr;
169 // Pure selection; handlers run after validation and keep calls interpreted.
170 AlphaPALCallHandler (*call_pal_override)(std::uint32_t function) = nullptr;
171 // Interpreter PAL replacements; JIT calls use native PAL. False falls through.
172 bool (*call_pal_compatibility)(CAlphaCPU& cpu, std::uint32_t function) = nullptr;
173 // Host instruction-count throttle only; null leaves it disabled.
174 bool (*instruction_pacing)(const CAlphaCPU& cpu) = nullptr;
175 // Standalone interpreter compatibility before interrupt service; null does nothing.
176 void (*interpreter_step)(CAlphaCPU& cpu) = nullptr;
177 // Legacy trace addresses; absent callbacks return zero.
178 std::uint64_t (*trace_prbr)(CAlphaCPU& cpu) = nullptr;
179 std::uint64_t (*trace_hwpcb)(CAlphaCPU& cpu) = nullptr;
180};
181
182constexpr std::uint32_t alpha_pal_call_allowed_modes(
183 const AlphaPALPolicy& policy, std::uint32_t function)
184{
185 std::uint32_t modes = 0;
186 if (policy.call_pal_valid)
187 for (std::uint32_t mode = 0; mode < 4; ++mode)
188 if (policy.call_pal_valid(function, mode)) modes |= 1u << mode;
189 return modes;
190}
191
192constexpr AlphaPALCallHandler alpha_pal_call_override(
193 const AlphaPALPolicy& policy, std::uint32_t function)
194{
195 return policy.call_pal_override ? policy.call_pal_override(function) : nullptr;
196}
197
198constexpr bool alpha_pal_call_compilable(
199 const AlphaPALPolicy& policy, std::uint32_t function)
200{
201 return !alpha_pal_call_override(policy, function)
202 && alpha_pal_call_allowed_modes(policy, function) != 0;
203}
204
205constexpr AlphaPALReturnDescriptor alpha_pal_return_descriptor(
206 const AlphaPALPolicy& policy, std::uint32_t instruction)
207{
208 return policy.hardware_return ? policy.hardware_return(instruction)
209 : AlphaPALReturnDescriptor{};
210}
