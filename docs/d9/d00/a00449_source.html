311// X11 split-image offsets must fit signed 16-bit coordinates.
312static void validate_x11_dimensions(unsigned width, unsigned height)
313{
314 if(!width || !height || width > 32767 || height > 32767 ||
315 (size_t)width > (std::numeric_limits<size_t>::max)() / sizeof(u32) / height)
316 FAILURE(Configuration, "Invalid X11 display dimensions");
317}
318
319static XImage* create_x11_image(unsigned width, unsigned height)
320{
321 validate_x11_dimensions(width, height);
322 XImage* image = XCreateImage(bx_x_display, default_visual,
323 DefaultDepth(bx_x_display, bx_x_screen_num), ZPixmap, 0, nullptr,
324 width, height, 32, 0);
325 if(!image)
326 FAILURE(OutOfMemory, "Unable to create X11 image");
327 if(image->bytes_per_line <= 0 || image->bits_per_pixel < image->depth ||
328 (size_t)image->bytes_per_line > (std::numeric_limits<size_t>::max)() / height)
329 {
330 XDestroyImage(image);
331 FAILURE(Configuration, "Invalid X11 image layout");
332 }
333 image->data = (char*)calloc(height, (size_t)image->bytes_per_line);
334 if(!image->data)
335 {
336 XDestroyImage(image);
337 FAILURE(OutOfMemory, "Unable to allocate X11 image pixels");
338 }
339 return image;
340}

342static unsigned long x11_color_component(unsigned value, unsigned long mask)
343{
344 if(!mask) return 0;
345 unsigned shift = 0;
346 while(!(mask & 1)) { mask >>= 1; ++shift; }
347 return (unsigned long)(((u64)value * mask + 127) / 255) << shift;
348}

350static unsigned long x11_rgb(unsigned red, unsigned green, unsigned blue)
351{
352 return x11_color_component(red, default_visual->red_mask) |
353 x11_color_component(green, default_visual->green_mask) |
354 x11_color_component(blue, default_visual->blue_mask);
355}

618void bx_x11_gui_c::specific_init_impl(unsigned tilewidth, unsigned tileheight)
619try
620{
621 if(x_init_done) { new_gfx_api = 1; return; }
622 validate_x11_dimensions(tilewidth, tileheight);
623 load_video_settings();
624 load_mouse_settings();
625 int x;
626 int y; /* window position */
627 unsigned int border_width = 4; /* four pixels */
628 const char* icon_name = "ES40";
629 XSizeHints size_hints = {};
630 char* display_name = NULL;
631
632 /* create GC for frame presentation */
633 unsigned long valuemask = 0; /* ignore XGCvalues and use defaults */
634 XGCValues values = {};
635 XSetWindowAttributes win_attr = {};
636
637 /* connect to X server */
638 if((bx_x_display = XOpenDisplay(display_name)) == NULL)
639 {
640 FAILURE(Runtime, "Unable to connect to the X11 display");
641 }
642 error_display = bx_x_display;
643 x_error = 0;
644 previous_x_error_handler = XSetErrorHandler(record_x11_error);
645
646 /* get screen size from display structure macro */
647 bx_x_screen_num = DefaultScreen(bx_x_display);
648
649 /* Note that in a real application, x and y would default to 0
650 * but would be settable from the command line or resource database.
651 */
652 x = y = 0;
653
654 // Temporary values so we can create the window
655 dimension_x = 640;
656 dimension_y = 400;
657 client_width = dimension_x;
658 client_height = dimension_y;
659
660 /* create opaque window */
661 win = XCreateSimpleWindow(bx_x_display,
662 RootWindow(bx_x_display, bx_x_screen_num), x, y,
663 dimension_x, dimension_y, border_width,
664 BlackPixel(bx_x_display, bx_x_screen_num),
665 BlackPixel(bx_x_display, bx_x_screen_num));
666 if(!win) FAILURE(Runtime, "Unable to create the X11 window");
667
668 // (attempt to) enable backing store
669 win_attr.save_under = 1;
670 win_attr.backing_store = Always;
671 XChangeWindowAttributes(bx_x_display, win, CWSaveUnder | CWBackingStore,
672 &win_attr);
673
674 default_visual = DefaultVisual(bx_x_display, bx_x_screen_num);
675 if(default_visual->c_class != TrueColor)
676 FAILURE(Configuration, "X11 full-frame display requires a TrueColor visual");
677 black_pixel = BlackPixel(bx_x_display, bx_x_screen_num);
678 white_pixel = WhitePixel(bx_x_display, bx_x_screen_num);
679 XSetWindowBackground(bx_x_display, win, black_pixel);
680
681 /* Set size hints for window manager. The window manager may
682 * override these settings. Note that in a real
683 * application if size or position were set by the user
684 * the flags would be UPosition and USize, and these would
685 * override the window manager's preferences for this window.
686 */
687
688 /* x, y, width, and height hints are now taken from
689 * the actual settings of the window when mapped. Note
690 * that PPosition and PSize must be specified anyway.
691 */
692 size_hints.flags = PPosition | PSize | PMinSize | PMaxSize;
693 size_hints.max_width = size_hints.min_width = dimension_x;
694 size_hints.max_height = size_hints.min_height = dimension_y;
695 {
696 XWMHints wm_hints = {};
697 XClassHint class_hints = {};
698 wm_hints.flags = StateHint | InputHint;
699 wm_hints.initial_state = NormalState;

756{
757 if(!val)
758 {
759 const bool was_captured = mouse_captured;
760 mouse_captured = mouse_warp_pending = mouse_position_valid = false;
761 mouse_buttons = ignored_mouse_buttons = 0;
762 mouse_remainder_x = mouse_remainder_y = 0.0;
763 mouse_sync_pending = true;
764 if(bx_x_display)
765 {
766 if(keyboard_grabbed) XUngrabKeyboard(bx_x_display, CurrentTime);
767 mouse_input_serial = XNextRequest(bx_x_display);
768 XUngrabPointer(bx_x_display, CurrentTime);
769 if(mouse_absolute && was_captured && win && !window_destroyed)
770 XUndefineCursor(bx_x_display, win);
771 XFlush(bx_x_display);
772 }
773 keyboard_grabbed = false;
774 if(was_captured) update_mouse_title();
775 return;
776 }
777 if(mouse_captured || !bx_x_display || !win || window_destroyed || retired ||
778 input_paused || input_detached || !guest_keyboard || !keyboard_focused) return;
779 mouse_remainder_x = mouse_remainder_y = 0.0;
780 Window focus;
781 int revert;
782 XGetInputFocus(bx_x_display, &focus, &revert);
783 XWindowAttributes attributes = {};
784 if((focus != win && focus != PointerRoot) ||
785 !XGetWindowAttributes(bx_x_display, win, &attributes) || attributes.map_state != IsViewable)
786 return;

788 {
789 const char empty = 0;

792 {
793 XColor color = {};
794 hidden_cursor = XCreatePixmapCursor(bx_x_display, bitmap, bitmap, &color, &color, 0, 0);
795 XFreePixmap(bx_x_display, bitmap);

799 printf("%%X11-W-MOUSE: Unable to create the capture cursor.\n");

804 {
805 const int result = XGrabPointer(bx_x_display, win, False,
806 ButtonPressMask | ButtonReleaseMask | PointerMotionMask,
807 GrabModeAsync, GrabModeAsync, win, hidden_cursor, CurrentTime);
808 if(result != GrabSuccess)
809 {
810 printf("%%X11-W-MOUSE: Unable to capture the mouse (X grab status %d).\n", result);
811 return;
812 }
813 }
814 mouse_captured = true;
815 mouse_sync_pending = true;
816 try
817 {

819 {

828 keyboard_result);

831 unsigned buttons;
832 const unsigned long pointer_serial = XNextRequest(bx_x_display);
833 if(!XQueryPointer(bx_x_display, win, &root, &child, &root_x, &root_y,

838 return;

841 ((buttons & Button3Mask) ? 2 : 0) |
842 ((buttons & Button2Mask) ? 4 : 0);
843 mouse_buttons = 0;
844 mouse_position_valid = mouse_warp_pending = false;
845 recenter_mouse();
846 mouse_input_serial = mouse_motion_serial = mouse_absolute ? pointer_serial : mouse_warp_serial;
847 update_mouse_title();
848 }
849 catch(...)

851 try { mouse_enabled_changed_impl(false); } catch(...) {}

853 }

902{
903 if(!mouse_captured || input_paused || input_detached ||
904 !guest_keyboard || !keyboard_focused || retired || XPending(bx_x_display)) return;
905 Window root, child;
906 int root_x, root_y, x, y;
907 unsigned state;

912 ((state & Button2Mask) ? 4 : 0);

923void bx_x11_gui_c::mouse_button(XButtonEvent* event, bool released)
924{
925 const unsigned bit = event->button == Button1 ? 1 : event->button == Button3 ? 2 :
926 event->button == Button2 ? 4 : 0;
927 if(event->serial < mouse_input_serial)
928 {
929 if(released) ignored_mouse_buttons &= ~bit;
930 return;
931 }
932 if(input_paused || input_detached || !guest_keyboard || !keyboard_focused || retired) return;
933 if(mouse_absolute && !released && (event->x < 0 || event->y < 0 ||
934 (unsigned)event->x >= client_width || (unsigned)event->y >= client_height))
935 {
936 XUngrabPointer(bx_x_display, CurrentTime);
937 return;
938 }
939 if(!mouse_captured)
940 {
941 if(!released && event->button == Button1)
942 {
943 mouse_enabled_changed_impl(true);
944 if(mouse_captured) ignored_mouse_buttons |= bit;
945 }
946 return;
947 }
948 // Cancel X11's implicit button grab so absolute input can leave the window.
949 if(mouse_absolute && !released) XUngrabPointer(bx_x_display, CurrentTime);
950 sync_mouse_capture();
951 if(ignored_mouse_buttons & bit)
952 {
953 if(released) ignored_mouse_buttons &= ~bit;
954 return;
955 }
956 const unsigned buttons = released ? mouse_buttons & ~bit : mouse_buttons | bit;
957 const int wheel = released ? 0 : event->button == Button4 ? 1 : event->button == Button5 ? -1 : 0;
958 if(buttons != mouse_buttons || wheel)
959 guest_keyboard->mouse_motion(0, 0, wheel, buttons);
960 mouse_buttons = buttons;
961}

963void bx_x11_gui_c::mouse_motion(XMotionEvent* event)
964{
965 if(!mouse_captured || event->serial < mouse_input_serial ||
966 event->serial < mouse_motion_serial || input_paused ||
967 input_detached || !guest_keyboard || !keyboard_focused || retired) return;
968 if(mouse_absolute && (event->x < 0 || event->y < 0 ||
969 (unsigned)event->x >= client_width || (unsigned)event->y >= client_height))
970 {
971 mouse_position_valid = false;
972 return;
973 }
974 if(mouse_warp_pending && event->serial >= mouse_warp_serial)
975 {
976 // Rebase even if the synthetic center event was coalesced.
977 mouse_x = mouse_warp_x;
978 mouse_y = mouse_warp_y;
979 mouse_position_valid = true;
980 mouse_warp_pending = false;
981 }
982 if(mouse_position_valid)
983 send_mouse_motion((double)event->x - mouse_x, (double)mouse_y - event->y);
984 mouse_x = event->x;
985 mouse_y = event->y;
986 mouse_position_valid = true;
987}

989void bx_x11_gui_c::send_mouse_motion(double dx, double dy)
990{
991 if(!mouse_captured || input_paused || input_detached || !guest_keyboard ||
992 !keyboard_focused || retired) return;
993 mouse_remainder_x += dx * (mouse_invert_x ? -mouse_speed : mouse_speed);
994 mouse_remainder_y += dy * (mouse_invert_y ? -mouse_speed : mouse_speed);
995 const double minimum = (std::numeric_limits<int>::min)();
996 const double maximum = (std::numeric_limits<int>::max)();
997 mouse_remainder_x = (std::max)(minimum, (std::min)(maximum, mouse_remainder_x));
998 mouse_remainder_y = (std::max)(minimum, (std::min)(maximum, mouse_remainder_y));
999 const int scaled_x = (int)mouse_remainder_x, scaled_y = (int)mouse_remainder_y;
1000 mouse_remainder_x -= scaled_x;
1001 mouse_remainder_y -= scaled_y;
1002 sync_mouse_capture();
1003 if(scaled_x || scaled_y) guest_keyboard->mouse_motion(scaled_x, scaled_y, 0, mouse_buttons);
1004}

1007static Bool x11_lifecycle_event(Display*, XEvent* event, XPointer)
1008{
1009 switch(event->type)
1010 {
1011 case KeyPress: case KeyRelease:
1012 case ButtonPress: case ButtonRelease:
1013 case MotionNotify: case EnterNotify: case LeaveNotify:
1014 case FocusIn: case FocusOut: case MappingNotify:
1015 return False;
1016 default:

1021void bx_x11_gui_c::poll_events(bool process_input)
1022{
1023 if(!bx_x_display) return;
1024 check_x11_error();
1025 check_mouse_capture();
1026 if(process_input) sync_mouse_capture();
1027 XEvent report;
1028
1029 // Bound polling so worker calls and barriers cannot starve.
1030 for(unsigned count = 0; count < 256; ++count)
1031 {
1032 if(!process_input || input_paused || input_detached)
1033 {
1034 if(!XCheckIfEvent(bx_x_display, &report, x11_lifecycle_event, nullptr)) break;
1035 }
1036 else
1037 {
1038 if(!XPending(bx_x_display)) break;
1039 XNextEvent(bx_x_display, &report);
1040 }
1041 if(report.type == MappingNotify)
1042 {
1043 XRefreshKeyboardMapping(&report.xmapping);
1044 load_hotkeys();
1045 continue;
1046 }
1047 if(report.xany.window != win) continue;
1048 switch(report.type)
1049 {
1050 case Expose:
1051 if(report.xexpose.count == 0) present_frame();
1052 break;
1053
1054 case ConfigureNotify:
1055 configure_window(report.xconfigure.width, report.xconfigure.height);
1056 reset_absolute_mouse_position();
1057 if(mouse_captured && !mouse_absolute)
1058 {
1059 mouse_position_valid = mouse_warp_pending = false;
1060 recenter_mouse();
1061 mouse_motion_serial = mouse_warp_serial;
1062 }
1063 if(reset_window_pending && reset_stage == reset_phase::unmaximize &&
1064 report.xconfigure.serial >= reset_configure_serial)
1065 reset_configure_seen = true;
1066 window_state_dirty = true;
1067 break;
1068
1069 case PropertyNotify:
1070 if(report.xproperty.atom == wm_state_atom || report.xproperty.atom == net_wm_state_atom)
1071 window_state_dirty = true;
1072 break;
1073
1074 case UnmapNotify:
1075 mouse_enabled_changed_impl(false);
1076 window_state_dirty = true;
1077 break;
1078
1079 case ButtonPress:
1080 case ButtonRelease:
1081 mouse_button(&report.xbutton, report.type == ButtonRelease);
1082 break;
1083
1084 case KeyPress:
1085 dispatch_keypress(&report.xkey);
1086 break;
1087
1088 case KeyRelease:
1089 if(!detectable_repeat && XEventsQueued(bx_x_display, QueuedAfterReading))
1090 {
1091 XEvent next;
1092 XPeekEvent(bx_x_display, &next);
1093 if(next.type == KeyPress && next.xkey.window == report.xkey.window &&
1094 next.xkey.keycode == report.xkey.keycode && next.xkey.time == report.xkey.time)
1095 break;
1096 }
1097 keypress(&report.xkey, true);
1098 break;
1099
1100 case FocusOut:
1101 if(report.xfocus.mode != NotifyGrab && report.xfocus.mode != NotifyUngrab &&
1102 report.xfocus.detail != NotifyInferior)
1103 {
1104 keyboard_focused = false;
1105 mouse_enabled_changed_impl(false);
1106 release_keys();
1107 }
1108 break;
1109
1110 case FocusIn:
1111 if(report.xfocus.mode != NotifyGrab && report.xfocus.mode != NotifyUngrab)
1112 keyboard_focused = true;
1113 break;
1114
1115 case MotionNotify:
1116 mouse_motion(&report.xmotion);
1117 break;
1118
1119 case EnterNotify:
1120 case LeaveNotify:
1121 if(mouse_absolute && report.xcrossing.mode == NotifyNormal)
1122 mouse_position_valid = false;
1123 break;
1124
1125 case MapNotify:
1126 window_state_dirty = true;
1127 // Repaint the retained frame without calling back into a guest device.
1128 present_frame();
1129 break;
1130
1131 case ClientMessage:
1132 if(report.xclient.message_type == wm_protocols && report.xclient.format == 32 &&
1133 (Atom)report.xclient.data.l[0] == wm_delete_window)
1134 {
1135 mouse_enabled_changed_impl(false);
1136 if(guest_system) guest_system->RequestShutdown();
1137 }
1138 break;
1139
1140 case DestroyNotify:
1141 if(report.xdestroywindow.window == win)
1142 {
1143 window_destroyed = true;
1144 mouse_enabled_changed_impl(false);
1145 if(guest_system) guest_system->RequestShutdown();
1146 }
1147 break;
1148
1149 default:
1150
1151 // (mch) Ignore...
1152 BX_DEBUG(("XXX: default Xevent type"));
1153
1154 /* all events selected by StructureNotifyMask are thrown away here,
1155 * since nothing is done with them */
1156 break;
1157 } /* end switch */
1158 } /* end while */
1159
1160 recenter_mouse();
1161 reconcile_keyboard_releases(process_input);
1162 if(window_state_dirty || pending_guest_resize || pending_window_hints ||
1163 fullscreen_transition_pending || reset_window_pending) update_window_state();
1164 if(process_input)
1165 {
1166 sync_mouse_capture();
1167 reconcile_mouse_buttons();
1168 }
1169 check_x11_error();
1170}

1237static x11_hotkey_key resolve_x11_hotkey(const std::string& configured)
1238{
1239 std::string name = configured;
1240 const struct { const char* alias; const char* name; } aliases[] = {
1241 { "Enter", "Return" }, { "Esc", "Escape" }, { "Del", "Delete" },
1242 { "Ins", "Insert" }, { "PgUp", "PageUp" }, { "PgDn", "PageDown" },
1243 { "PageDn", "PageDown" }, { "KpPlus", "Keypad +" },
1244 { "KeypadPlus", "Keypad +" }, { "KpMinus", "Keypad -" },
1245 { "KeypadMinus", "Keypad -" }
1246 };
1247 for(const auto& alias : aliases)
1248 if(x11_hotkey_equal(name, alias.alias)) { name = alias.name; break; }
1249 const struct { KeySym key; const char* name; } names[] = {
1250 { XK_Return, "Return" }, { XK_Escape, "Escape" }, { XK_BackSpace, "Backspace" },
1251 { XK_Tab, "Tab" }, { XK_space, "Space" }, { XK_Caps_Lock, "CapsLock" },
1252 { XK_Print, "PrintScreen" }, { XK_Scroll_Lock, "ScrollLock" }, { XK_Pause, "Pause" },
1253 { XK_Insert, "Insert" }, { XK_Home, "Home" }, { XK_Page_Up, "PageUp" },
1254 { XK_Delete, "Delete" }, { XK_End, "End" }, { XK_Page_Down, "PageDown" },
1255 { XK_Right, "Right" }, { XK_Left, "Left" }, { XK_Down, "Down" }, { XK_Up, "Up" },
1256 { XK_Num_Lock, "Numlock" }, { XK_KP_Divide, "Keypad /" },
1257 { XK_KP_Multiply, "Keypad *" }, { XK_KP_Subtract, "Keypad -" },
1258 { XK_KP_Add, "Keypad +" }, { XK_KP_Enter, "Keypad Enter" },
1259 { XK_KP_Decimal, "Keypad ." }, { XK_KP_Equal, "Keypad =" }
1260 };
1261 for(const auto& named : names)
1262 if(x11_hotkey_equal(name, named.name)) return { (u32)named.key, named.name };
1263 for(unsigned i = 0; i < 24; ++i)
1264 {
1265 const std::string label = "F" + std::to_string(i + 1);
1266 if(x11_hotkey_equal(name, label.c_str())) return { XK_F1 + i, label };
1267 }
1268 for(unsigned i = 0; i < 10; ++i)
1269 {
1270 const std::string label = "Keypad " + std::to_string(i);
1271 if(x11_hotkey_equal(name, label.c_str())) return { XK_KP_0 + i, label };
1272 }
1273 if(name.size() != 1 || (unsigned char)name[0] < 33 || (unsigned char)name[0] > 126)
1274 return {};
1275 KeySym symbol = (unsigned char)std::tolower((unsigned char)name[0]);

1277 if(code)

1292void bx_x11_gui_c::refresh_hotkey_modifiers()
1293{
1294 XModifierKeymap* mapping = XGetModifierMapping(bx_x_display);
1295 if(!mapping) FAILURE(Runtime, "Unable to read X11 keyboard modifiers");
1296 for(unsigned index = 0; index < 8; ++index)
1297 {
1298 unsigned modifiers = 0;
1299 bool lock = index == 1;
1300 for(int slot = 0; slot < mapping->max_keypermod; ++slot)
1301 {
1302 const KeyCode code = mapping->modifiermap[index * mapping->max_keypermod + slot];
1303 if(!code) continue;
1304 for(unsigned level = 0; level < 4; ++level)
1305 {
1306 const KeySym symbol = x11_key_symbol(code, 0, level);
1307 modifiers |= x11_host_modifier(symbol);
1308 lock = lock || symbol == XK_Caps_Lock || symbol == XK_Shift_Lock ||
1309 symbol == XK_Num_Lock || symbol == XK_Scroll_Lock;
1310 }
1311 }
1312 if(modifiers & X11_HOTKEY_ALT) modifiers &= ~X11_HOTKEY_GUI;
1313 hotkey_modifier_masks[index] = modifiers ? modifiers : lock ? 0 : X11_HOTKEY_UNSUPPORTED;
1314 }
1315 hotkey_modifier_masks[0] = X11_HOTKEY_SHIFT;
1316 hotkey_modifier_masks[2] = X11_HOTKEY_CTRL;
1317 for(auto& modifiers : hotkey_key_modifiers) modifiers = 0;
1318 for(unsigned index = 0; index < 8; ++index)
1319 for(int slot = 0; slot < mapping->max_keypermod; ++slot)
1320 {
1321 const KeyCode code = mapping->modifiermap[index * mapping->max_keypermod + slot];
1322 if(code) hotkey_key_modifiers[code] |= hotkey_modifier_masks[index];
1323 }
1324 XFreeModifiermap(mapping);
1325}

1449{

1451 const bool live = process_input && !input_paused && guest_keyboard;
1452 bool pending = false;
1453 for(const auto& key : pressed_keys)
1454 pending = pending || (key.down && (live || key.suppressed));
1455 if(!pending || XPending(bx_x_display)) return;

1458 if(!XQueryKeymap(bx_x_display, physical) || x_error || XPending(bx_x_display)) return;

1460 {

1466 if(!live) continue;
1467 if(!guest_key_held(key.guest, code))
1468 guest_keyboard->gen_scancode(key.guest | BX_KEY_RELEASED);
1469 }
1470 key = {};
1471 key_release_serial[code] = serial;
1472 }
1473 CTRL_pressed = false;
1474 for(const auto& key : pressed_keys)
1475 if(key.down && key.control) CTRL_pressed = true;

1513void bx_x11_gui_c::keypress(XKeyEvent* event, bool released)
1514{
1515 if(!guest_keyboard || input_paused || input_detached ||
1516 event->keycode < 8 || event->keycode >= 256) return;
1517 if(event->serial < key_release_serial[event->keycode]) return;
1518 auto& key = pressed_keys[event->keycode];
1519 if(released)
1520 {
1521 if(!key.down) return;
1522 if(key.guest != BX_KEY_UNHANDLED && !guest_key_held(key.guest, event->keycode))
1523 guest_keyboard->gen_scancode(key.guest | BX_KEY_RELEASED);
1524 key = {};
1525 CTRL_pressed = false;
1526 for(const auto& held : pressed_keys)
1527 if(held.down && held.control) CTRL_pressed = true;
1528 return;
1529 }
1530 if(!keyboard_focused) return;
1531 if(key.down)
1532 {
1533 if(key.scale_delta && scale_change_enable)
1534 {
1535 const auto& binding = key.scale_delta > 0 ? hotkey_scale_up : hotkey_scale_down;
1536 if(binding.matches((u32)x11_hotkey_symbol(event->keycode, XkbGroupForCoreState(event->state)),
1537 hotkey_modifiers(event->state)))
1538 adjust_window_scale(key.scale_delta);
1539 }
1540 if(key.guest != BX_KEY_UNHANDLED && !key.modifier)
1541 guest_keyboard->gen_scancode(key.guest);
1542 return;
1543 }
1544
1545 const u32 host_key = (u32)x11_hotkey_symbol(event->keycode, XkbGroupForCoreState(event->state));
1546 const unsigned modifiers = hotkey_modifiers(event->state);
1547 if(hotkey_ctrl_alt_delete.matches(host_key, modifiers))
1548 {
1549 send_ctrl_alt_delete(event->keycode);
1550 return;
1551 }
1552 if(hotkey_reset_window.matches(host_key, modifiers))
1553 {
1554 key.down = key.suppressed = true;
1555 reset_window_size();
1556 return;
1557 }
1558 if(scale_change_enable && (hotkey_scale_up.matches(host_key, modifiers) ||
1559 hotkey_scale_down.matches(host_key, modifiers)))
1560 {
1561 key.down = key.suppressed = true;
1562 key.scale_delta = hotkey_scale_up.matches(host_key, modifiers) ? 1 : -1;
1563 adjust_window_scale(key.scale_delta);
1564 return;
1565 }
1566 if(hotkey_mouse_capture.matches(host_key, modifiers))
1567 {
1568 key.down = key.suppressed = true;
1569 for(unsigned code = 8; code < 256; ++code)
1570 {
1571 auto& held = pressed_keys[code];
1572 if(!held.down || !(held.host_modifiers & hotkey_mouse_capture.modifiers)) continue;
1573 if(held.guest != BX_KEY_UNHANDLED && !guest_key_held(held.guest, code))
1574 guest_keyboard->gen_scancode(held.guest | BX_KEY_RELEASED);
1575 held.guest = BX_KEY_UNHANDLED;
1576 held.suppressed = true;
1577 }
1578 mouse_enabled_changed_impl(!mouse_captured);
1579 return;
1580 }
1581 if((host_key == XK_Return || host_key == XK_KP_Enter) && (modifiers & X11_HOTKEY_ALT))
1582 {
1583 key.down = key.suppressed = true;
1584 toggle_fullscreen();
1585 return;
1586 }
1587
1588 KeySym symbol = NoSymbol;
1589 char buffer[MAX_MAPPED_STRING_LENGTH];
1590 XLookupString(event, buffer, sizeof(buffer), &symbol, nullptr);
1591 const u32 guest = translate_key(symbol, host_key);
1592 if(guest != BX_KEY_UNHANDLED && !guest_key_held(guest))
1593 guest_keyboard->gen_scancode(guest);
1594 key.guest = guest;
1595 key.down = true;
1596 key.host_modifiers = hotkey_key_modifiers[event->keycode];
1597 key.modifier = key.host_modifiers != 0 || x11_modifier_key(symbol, guest);
1598 key.control = symbol == XK_Control_L || symbol == XK_Control_R;
1599 CTRL_pressed = CTRL_pressed || key.control;
1600}
