37// FNV-1a/64 over a block's source words -- record() uses it to revalidate kept code after a flush
38// instead of recompiling, and to catch self-modifying code (changed bytes -> different hash).
39static inline uint64_t src_hash(const uint8_t* p, uint32_t n_instr)
40{
41 uint64_t h = 1469598103934665603ULL;
42 for (uint32_t i = 0; i < n_instr; ++i) {
43 const auto* w = p + static_cast<std::size_t>(i) * 4;
44 const uint32_t word = uint32_t(w[0]) | (uint32_t(w[1]) << 8)
45 | (uint32_t(w[2]) << 16) | (uint32_t(w[3]) << 24);
46 h ^= word;
47 h *= 1099511628211ULL;
48 }
49 return h;
50}

69static inline bool jit_matches_counted_scan(const uint32_t* words) {
70 return std::memcmp(words, kJitCountedScanHead, sizeof(kJitCountedScanHead)) == 0
71 && std::memcmp(words + 32, kJitCountedScanRestore, sizeof(kJitCountedScanRestore)) == 0;
72}

74static inline bool jit_matches_word_scan(const uint32_t* words) {
75 return std::memcmp(words, kJitWordScanLoop, sizeof(kJitWordScanLoop)) == 0;

78// Check all fusion words, including lookahead.
79static inline bool jit_scan_span_supported(const AlphaCPUProfile& profile, bool shadow, uint64_t pc,
80 const uint32_t* words, uint32_t count) {
81 // Internal branches share the head's execute check and bypass dispatch.
82 const auto& fetch = profile.instruction_fetch;
83 if (shadow || (pc & 1) || !fetch.jit_same_page_access || !fetch.jit_fuse_nonpal_blocks
84 || (pc >> 13) != ((pc + 4 * (count - 1)) >> 13)) return false;
85 for (uint32_t i = 0; i < count; ++i) {
86 if (!profile.supports_instruction(words[i]) || profile.retires_instruction_at_decode(words[i])
87 || profile.requires_fen_at_decode(words[i]) || profile.fen_rounding_modes(words[i]))
88 return false;
89 }
90 return true;
91}
