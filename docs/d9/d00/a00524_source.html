137int CPort80::SaveState(FILE* f)

140

145 printf("%s: %ld bytes saved.\n", devid_string, ss);

152int CPort80::RestoreState(FILE* f)
153{
154 long ss;
155 u32 m1;
156 u32 m2;
157 size_t r;
158
159 r = fread(&m1, sizeof(u32), 1, f);
160 if (r != 1)
161 {
162 printf("%s: unexpected end of file!\n", devid_string);
163 return -1;
164 }
165
166 if (m1 != p80_magic1)
167 {
168 printf("%s: MAGIC 1 does not match!\n", devid_string);
169 return -1;
170 }
171
172 r = fread(&ss, sizeof(long), 1, f);
173 if (r != 1)
174 {
175 printf("%s: unexpected end of file!\n", devid_string);
176 return -1;
177 }
178
179 if (ss != sizeof(state))
180 {
181 printf("%s: STRUCT SIZE does not match!\n", devid_string);
182 return -1;
183 }
184
185 r = fread(&state, sizeof(state), 1, f);
186 if (r != 1)
187 {
188 printf("%s: unexpected end of file!\n", devid_string);
189 return -1;
190 }
191
192 r = fread(&m2, sizeof(u32), 1, f);
193 if (r != 1)
194 {
195 printf("%s: unexpected end of file!\n", devid_string);
196 return -1;
197 }
198
199 if (m2 != p80_magic2)
200 {
201 printf("%s: MAGIC 1 does not match!\n", devid_string);
202 return -1;
203 }
204
205 printf("%s: %ld bytes restored.\n", devid_string, ss);
206 return 0;
