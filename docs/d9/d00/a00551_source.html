46/*
47 * Register description for the Digital Semiconductor ``Tulip'' (21x4x)
48 * Ethernet controller family, and a variety of clone chips, including:
49 *
50 * - Macronix 98713, 98713A, 98715, 98715A, 98725 (PMAC):
51 *
52 * These chips are fairly straight-forward Tulip clones.
53 * The 98713 is a very close 21140A clone. It has GPR
54 * and MII media, and a GPIO facility, and uses the ISV
55 * SROM format (or, at least, should, because of the GPIO
56 * facility). The 98713A has MII, no GPIO facility, and
57 * an internal NWay block. The 98715, 98715A, and 98725
58 * have only GPR media and the NWay block. The 98715,
59 * 98715A, and 98725 support power management.
60 *
61 * The 98715AEC adds 802.3x flow Frame based Flow Control to the
62 * 98715A.
63 *
64 * - Lite-On 82C115 (PNIC II):
65 *
66 * A clone of the Macronix MX98725, with the following differences:
67 *
68 * - Wake-On-LAN support
69 * - 128-bit multicast hash table rather than the
70 * standard 512-bit hash table
71 * - 802.3x flow control
72 *
73 * - Lite-On 82C168, 82C169 (PNIC):
74 *
75 * Pretty close, with only a few minor differences:
76 *
77 * - EEPROM is accessed completely differently.
78 * - MII is accessed completely differently.
79 * - No SIO facility (due to the above two differences).
80 * - GPIO interface is different than the 21140's.
81 * - Boards that lack PHYs use the internal NWay block
82 * and transceiver.
83 *
84 * - Winbond 89C840F
85 *
86 * Less similar, but still roughly compatible (enough so
87 * that the driver can be adapted, at least):
88 *
89 * - Registers lack the pad word between them.
90 * - Instead of a setup frame, there are two station
91 * address registers and two multicast hash table
92 * registers (64-bit multicast hash table).
93 * - Only supported media interface is MII-over-SIO.
94 * - Different OPMODE register bits for various things
95 * (mostly media related).
96 *
97 * - ADMtek AL981
98 *
99 * Another pretty-close clone:
100 *
101 * - Wake-On-LAN support
102 * - Instead of a setup frame, there are two station
103 * address registers and two multicast hash table
104 * registers (64-bit multicast hash table).
105 * - 802.3x flow control
106 * - Only supported media interface is built-in PHY
107 * which is accessed through a set of special registers.
108 * - Not all registers have the pad word between them,
109 * but luckily, there are all AL981-specific registers,
110 * so this is easy to deal with.
111 *
112 * - ADMtek AN983 and AN985
113 *
114 * Similar to the ADMtek AL981, but with a few differences.
115 *
116 * - Xircom X3201-3
117 *
118 * CardBus 21143 clone, with a few differences:
119 *
120 * - No MicroWire SROM; Ethernet address must come
121 * from CIS.
122 * - Transmit buffers must also be 32-bit aligned.
123 * - The BUSMODE_SWR bit is not self-clearing.
124 * - Must include FS|LS in setup packet descriptor.
125 * - SIA is not 21143-like, and all media attachments
126 * are MII-on-SIO.
127 *
128 * - Davicom DM9102 and DM9102A
129 *
130 * Pretty similar to the 21140A, with a few differences:
131 *
132 * - Wake-On-LAN support
133 * - DM9102 has built-in 10/100 PHY on MII interface.
134 * - DM9102A has built-in 10/100 PHY on MII interface,
135 * as well as a HomePNA 1 PHY on an alternate MII
136 * interface (selected by clearing OPMODE_PS).
137 * - The chip has a bug in the transmit DMA logic,
138 * requiring that the packet be comprised of only
139 * one DMA segment.
140 * - The bus interface is buggy, and the BUSMODE register
141 * must be initialized to 0.
142 * - There seems to be an interrupt logic bug, requiring
143 * that interrupts be disabled on the chip during the
144 * interrupt handler.
145 *
146 * - ASIX AX88140
147 *
148 * 21143 clone with a few differences:
149 *
150 * - Specific broadcast bit in the OPMODE register.
151 * - Transmit buffer must be 32-bit aligned.
152 * - The BUSMODE_SWR bit is not self-clearing.
153 * - External 10BaseT PHY or 10/100 MII.
154 *
155 * Some of the clone chips have different registers, and some have
156 * different bits in the same registers. These will be denoted by
157 * PMAC, PNICII, PNIC, DM, WINB, ADM and AX in the register/bit names.
158 */

248/*
249 * The Tulip filter is programmed by "transmitting" a Setup Packet
250 * (indicated by TDCTL_Tx_SET). The filtering type is indicated
251 * as follows:
252 *
253 * FT1 FT0 Description
254 * --- --- -----------
255 * 0 0 Perfect Filtering: The Tulip interprets the
256 * descriptor buffer as a table of 16 MAC addresses
257 * that the Tulip should receive.
258 *
259 * 0 1 Hash Filtering: The Tulip interprets the
260 * descriptor buffer as a 512-bit hash table
261 * plus one perfect address. If the incoming
262 * address is Multicast, the hash table filters
263 * the address, else the address is filtered by
264 * the perfect address.
265 *
266 * 1 0 Inverse Filtering: Like Perfect Filtering, except
267 * the table is addresses that the Tulip does NOT
268 * receive.
269 *
270 * 1 1 Hash-only Filtering: Like Hash Filtering, but
271 * physical addresses are matched by the hash table
272 * as well, and not by matching a single perfect
273 * address.
274 *
275 * A Setup Packet must always be 192 bytes long. The Tulip can store
276 * 16 MAC addresses. If not all 16 are specified in Perfect Filtering
277 * or Inverse Filtering mode, then unused entries should duplicate
278 * one of the valid entries.
279 */
280#define TDCTL_Tx_FT_PERFECT 0
281#define TDCTL_Tx_FT_HASH TDCTL_Tx_FT0
282#define TDCTL_Tx_FT_INVERSE TDCTL_Tx_FT1
283#define TDCTL_Tx_FT_HASHONLY (TDCTL_Tx_FT1|TDCTL_Tx_FT0)
