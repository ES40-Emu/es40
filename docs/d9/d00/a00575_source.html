135bool CNetworkPcap::init(const char *devid_string, CConfigurator *cfg)
136{
137 pcap_if_t *alldevs;
138 pcap_if_t *d;
139 unsigned int inum;
140 unsigned int i = 0;
141 char errbuf[PCAP_ERRBUF_SIZE];
142
143 char *adapter = cfg->get_text_value("adapter");
144 bool from_menu = (adapter == NULL);
145 if (!adapter) {
146 printf("\n%s: Choose a network adapter to connect to:\n", devid_string);
147 if (pcap_findalldevs(&alldevs, errbuf) == -1) {
148 printf("%s: Error in pcap_findalldevs: %s\n", devid_string, errbuf);
149 return false;
150 }
151
152 for (d = alldevs; d; d = d->next) {
153 printf("%d. %s\n ", ++i, d->name);
154 if (d->description)
155 printf(" (%s)", d->description);
156 else
157 printf(" (No description available)");
158 /* Loopback and other IP-only sources carry no ethernet header. */
159 if (d->flags & PCAP_IF_LOOPBACK)
160 printf(" [loopback - no ethernet frames]");
161 printf("\n");
162 }
163
164 if (i == 0) {
165 printf("%s: No network interfaces found\n", devid_string);
166 return false;
167 }
168
169 if (i == 1)
170 inum = 1;
171 else {
172 for (;;) {
173 char input_buf[64];
174 int parsed;
175
176 printf("%%NIC-Q-NICNO: Enter the interface number (1-%d): ", i);
177 fflush(stdout);
178
179 if (fgets(input_buf, sizeof(input_buf), stdin) == NULL) {
180 printf("%s: Unexpected end of input while selecting network "
181 "interface\n", devid_string);
182 return false;
183 }
184
185 if (sscanf(input_buf, "%d", &parsed) != 1
186 || parsed < 1
187 || (unsigned int)parsed > i) {
188 printf("%%NIC-W-BADSEL: Invalid selection. Please enter a number "
189 "between 1 and %d.\n", i);
190 continue;
191 }
192
193 inum = (unsigned int)parsed;
194 break;
195 }
196 }
197
198 for (d = alldevs, i = 0; i < inum - 1; d = d->next, i++)
199 ;
200
201 adapter = d->name;
202 }
203
204#if defined(WIN32)
205 // Opening with pcap_open on Windows allows specification of
206 // PCAP_OPENFLAG_NOCAPTURE_LOCAL, which stops the pcap device from seeing
207 // it's own transmitted packets.
208 //
209 // This is important because:
210 // 1. Real ethernet cards don't reflect packets except while in
211 // loopback mode(s).
212 // 2. Reflecting all packets increases inbound packet processing and
213 // host load.
214 // 3. DECNET Phase IV will think a reflected packet is from another
215 // node that has the same DECNET Phase IV address
216 // (AA-xx-xx-xx-xx-xx), and will panic on startup and abort.
217 // 4. Libpcap doesn't reflect packets, and we want winpcap/libpcap
218 // processing to be identical.
219 // Loopback packets are handled via direct entry in the receive queue.
220 if ((fp = (pcap_t *)pcap_open(
221 adapter, 65536 /*snaplen: capture entire packets */,
222 PCAP_OPENFLAG_PROMISCUOUS |
223 PCAP_OPENFLAG_NOCAPTURE_LOCAL /*promiscuous */,
224 10 /*packet buffer timeout: [ms] */, 0 /* auth structure */,
225 errbuf)) == NULL) // connect to pcap...
226#else
227 if ((fp = pcap_open_live(adapter, 65536 /*snaplen: capture entire packets */,
228 1 /*promiscuous */,
229 10 /*packet buffer timeout: [ms] */,
230 errbuf)) == NULL) // connect to pcap...
231#endif
232 {
233 printf("%s: Error opening adapter %s:\n %s\n", devid_string, adapter, errbuf);
234 return false;
235 }
236
237 if (pcap_setnonblock(fp, 1, errbuf) == PCAP_ERROR) {
238 printf("%s: Error setting adapter %s non-blocking:\n %s\n", devid_string, adapter, errbuf);
239 pcap_close(fp);
240 fp = nullptr;
241 return false;
242 }
243
244 /* Loopback, VPN and other IP-only capture sources carry no ethernet
245 header, so no ether filter can be compiled. */
246 linktype = pcap_datalink(fp);
247 if (linktype != DLT_EN10MB) {
248 const char *lt = pcap_datalink_val_to_name(linktype);
249 printf("%s: WARNING: adapter %s is not an ethernet capture source "
250 "(link type %s [%d], not EN10MB).\n", devid_string,
251 adapter, lt ? lt : "unknown", linktype);
252 printf("%s: WARNING: captured packets have no ethernet header and "
253 "receive filtering is disabled. Testing use only; pick a "
254 "physical or bridged ethernet adapter for real networking."
255 "\n", devid_string);
256
257 /* Warn on non-ethernet adapter */
258 if (from_menu) {
259 for (;;) {
260 char answer[16];
261 printf("%%NIC-Q-NONETH: Use this adapter anyway? (y/n): ");
262 fflush(stdout);
263 if (fgets(answer, sizeof(answer), stdin) == NULL) {
264 printf("%s: Unexpected end of input; not using %s\n",
265 devid_string, adapter);
266 pcap_close(fp);
267 fp = nullptr;
268 return false;
269 }
270 if (answer[0] == 'y' || answer[0] == 'Y')
271 break;
272 if (answer[0] == 'n' || answer[0] == 'N') {
273 printf("%s: Not using %s\n", devid_string, adapter);
274 pcap_close(fp);
275 fp = nullptr;
276 return false;
277 }
278 }
279 }
280 }
281
282 opened = true;
283 printf("%s: Using pcap adapter %s\n", devid_string, adapter);
284 return true;
285}

298int CNetworkPcap::receive(const u8 **data, int *len)
299{
300 if (!fp)
301 return -1;
302 struct pcap_pkthdr *packet_header;
303 const unsigned char *packet_data = NULL;
304 int res = pcap_next_ex(fp, &packet_header, &packet_data);
305 if (res > 0) {
306 *data = packet_data;
307 *len = packet_header->caplen;
308 return 1;
309 }
310 if (res == 0)
311 return 0; /* timeout, no packet */
312 return -1; /* error */
313}

315void CNetworkPcap::set_filter(u8 mac_list[][6], int num_macs,
316 bool promiscuous, bool receive_all,
317 bool pass_multicast, bool inverse)
318{
319 if (!fp)
320 return;
321
322 /* Packet filtering only works on ethernet... */
323 if (linktype != DLT_EN10MB) {
324 if (!nonether_filter_warned) {
325 printf("Packet filtering disabled: link type %d is not ethernet.\n",
326 linktype);
327 nonether_filter_warned = true;
328 }
329 return;
330 }
331
332 char mac_txt[16][20];
333 char filter[1000];
334 int numUnique = 0;
335 int unique[16];
336 int count = num_macs > 16 ? 16 : num_macs;
337
338 for (int i = 0; i < count; i++) {
339 bool zero = (mac_list[i][0] | mac_list[i][1] | mac_list[i][2] |
340 mac_list[i][3] | mac_list[i][4] | mac_list[i][5]) == 0;
341 if (zero)
342 continue;
343 sprintf(mac_txt[i], "%02x:%02x:%02x:%02x:%02x:%02x",
344 mac_list[i][0], mac_list[i][1], mac_list[i][2],
345 mac_list[i][3], mac_list[i][4], mac_list[i][5]);
346
347 bool u = true;
348 for (int j = 0; j < numUnique; j++) {
349 if (memcmp(mac_list[i], mac_list[unique[j]], 6) == 0) {
350 u = false;
351 break;
352 }
353 }
354 if (u) {
355 unique[numUnique] = i;
356 numUnique++;
357 }
358 }
359
360 filter[0] = '\0';
361 strcpy(filter, "ether broadcast");
362
363 if (!(promiscuous || receive_all)) {
364 char list[800];
365
366 if (pass_multicast)
367 strcat(filter, " or ether multicast");
368
369 list[0] = '\0';
370 if (numUnique == 0) {
371 /* No addresses learned yet; there's no "our own MAC"
372 to fall back to at this layer (the caller owns
373 that), so just leave the list empty and rely on the
374 broadcast/multicast terms above. */
375 } else {
376 for (int i = 0; i < numUnique; i++) {
377 strcat(list, (i == 0) ? "ether dst " : " or ether dst ");
378 strcat(list, mac_txt[unique[i]]);
379 }
380 }
381
382 if (list[0]) {
383 if (inverse) {
384 strcat(filter, " or (not (");
385 strcat(filter, list);
386 strcat(filter, "))");
387 } else {
388 strcat(filter, " or (");
389 strcat(filter, list);
390 strcat(filter, ")");
391 }
392 }
393 } /* else PR/RA: leave filter as match-all broadcasts + everything */
394
395 /* In promiscuous/receive-all we still compile an empty/very permissive filter. */
396 if (promiscuous || receive_all) {
397 /* Let the OS accept everything: empty expr compiles to "match all". */
398 filter[0] = '\0';
399 }
400
401 if (filter_active) {
402 pcap_freecode(&fcode);
403 filter_active = false;
404 }
405
406 if (pcap_compile(fp, &fcode, filter, 1, 0xffffffff) < 0) {
407 printf("Unable to compile the packet filter (%s): %s\n", filter,
408 pcap_geterr(fp));
409 return;
410 }
411
412 filter_active = true;
413
414 if (pcap_setfilter(fp, &fcode) < 0)
415 printf("Error setting the filter: %s\n", pcap_geterr(fp));
416
417 return;
418}

420void CNetworkPcap::close()
421{
422 if (fp && filter_active) {
423 pcap_freecode(&fcode);
424 filter_active = false;
425 }
426 if (fp) {
427 pcap_close(fp);
428 fp = nullptr;
429 }
430 opened = false;
431}
