191{
192 host_audio::Format as;
193
194 as.freq = 44100;
195 as.channels = 2;
196 as.format = host_audio::SampleFormat::s16_le;
197 memset((void*)&state, 0, sizeof(state));
198 const u64 latency = cfg->get_num_value("latency_ms", true, 40);
199 m_latency_ms = (unsigned)(std::min)((std::max)(latency, (u64)10), (u64)500);
200 m_audio_backend = host_audio::backend_for_gui(cfg->get_gui_name());
201 host_audio::SetupGuard setup(m_audio_backend, {&state.audio_be_in, &state.audio_be_out},
202 {&state.adc_voice, &state.dac_voice[0], &state.dac_voice[1]});
203 if (!host_audio::initialize(m_audio_backend)) {
204 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio");
205 }
206 // Request ~10ms host periods; SDL environment overrides still apply.
207 host_audio::set_sample_frames_hint("512", m_audio_backend);
208 state.audio_be_in = host_audio::open_input_device(m_audio_backend);
209 if (!state.audio_be_in) {
210 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio input");
211 }
212 state.audio_be_out = host_audio::open_output_device(m_audio_backend);
213 if (!state.audio_be_out) {
214 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio output");
215 }
216 {
217 int out_freq = 0;
218 int out_frames = 0;
219 if (!host_audio::output_device_format(state.audio_be_out, &out_freq, &out_frames))
220 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to query %s audio output");
221 m_out_freq = out_freq > 0 ? out_freq : 48000;
222 m_out_period_frames = out_frames > 0 ? out_frames : 512;
223 }
224 host_audio::create_stream(state.adc_voice, nullptr, &as, m_audio_backend);
225 host_audio::create_stream(state.dac_voice[0], &as, nullptr, m_audio_backend);
226 host_audio::create_stream(state.dac_voice[1], &as, nullptr, m_audio_backend);
227
228 if (!state.adc_voice || !state.dac_voice[0] || !state.dac_voice[1])
229 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize ES1371 audio streams");
230
231 host_audio::set_input_callback(state.adc_voice, es1371_dac_callback_adc, this);
232 host_audio::set_output_callback(state.dac_voice[0], es1371_dac_callback_dac1, this);
233 host_audio::set_output_callback(state.dac_voice[1], es1371_dac_callback_dac2, this);
234 setup.complete();
235}

303{

315void CES1371::stop_threads()
316{
317 m_audio_stop.store(true);
318 m_audio_wait.notify_one();
319 {
320 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
321 audio_running = false;
322 for (size_t i = 0; i < NB_CHANNELS; ++i)
323 reset_audio_request(i, false, false);
324 }
325 // The worker never waits for the bus, including when our caller already
326 // owns it. Joining here therefore cannot leave it blocked behind us.
327 if (m_audio_thread)
328 {
329 m_audio_thread->join();
330 delete m_audio_thread;
331 m_audio_thread = nullptr;
332 }
333 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
334 unbind_voices();
335}

337void CES1371::start_threads()
338{
339 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
340 if (audio_running)
341 return;
342
343 for (size_t i = 0; i < NB_CHANNELS; ++i)
344 reset_audio_request(i, false, true);
345 m_audio_failed.store(false);
346 m_audio_wake.store(false);

348 {
349 es1371_update_voices(&state, state.ctl, state.sctl, true);
350 m_audio_stop.store(false);
351 audio_running = true;
352 m_audio_thread = new CThread("es1371");
353 m_audio_thread->start(*this);
354 }
355 catch (...)
356 {
357 audio_running = false;
358 m_audio_stop.store(true);
359 delete m_audio_thread;
360 m_audio_thread = nullptr;
361 for (size_t i = 0; i < NB_CHANNELS; ++i)
362 reset_audio_request(i, false, false);
363 unbind_voices();
364 throw;
365 }

374// Guest-format bytes per second at the channel's programmed rate.
375uint32_t CES1371::chan_rate_bps(size_t channel) const
376{
377 return channel_rate(channel) << state.chan[channel].shift;

379// Guest-format bytes one host device pull removes from the stream.
380int CES1371::dac_pull_bytes(uint32_t rate_bps) const
381{
382 return (int)(std::min)((u64)rate_bps * m_out_period_frames / m_out_freq,
383 (u64)(1 << 20));
384}

386// Lead to keep queued: configured latency, never under three host pulls.
387int CES1371::dac_lead_bytes(size_t channel, uint32_t rate_bps) const
388{
389 const u64 by_latency = (u64)rate_bps * m_latency_ms / 1000;
390 const u64 by_period = (u64)dac_pull_bytes(rate_bps) * 3;
391 return (int)(std::min)((std::max)(by_latency, by_period), (u64)(1 << 20));
392}

394// One worker tick: playback is paced at the sample rate against a fixed SDL lead.
395CES1371::ServiceResult CES1371::service_audio_channel(size_t channel,
396 std::chrono::steady_clock::time_point now)
397{
398 std::unique_lock<std::recursive_mutex> bus_lock(
399 cSystem->get_device_bus_mutex(), std::try_to_lock);
400 if (!bus_lock.owns_lock())
401 return SERVICE_BUSY;
402 if (!audio_running || m_audio_stop.load())
403 return SERVICE_IDLE;
404 if (!channel_enabled(channel, state.ctl, state.sctl))
405 return SERVICE_IDLE;
406
407 struct chan* d = &state.chan[channel];
408 host_audio::Stream& voice = channel == ADC_CHANNEL
409 ? state.adc_voice : state.dac_voice[channel];
410 CES1371StreamLock stream_lock(voice);
411
412 if (channel == ADC_CHANNEL)
413 {
414 const int avail = host_audio::available(voice);
415 if (avail < 0)
416 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to query audio stream");
417 if (avail <= 0)
418 return SERVICE_IDLE;
419 const int moved = es1371_run_channel(&state, channel, (std::min)(avail, 4096));
420 // Release the stream lock before unbinding to preserve SDL's lock order.
421 stream_lock.unlock();
422 if (d->halted) host_audio::unbind_stream(voice);
423 return moved > 0 && moved < avail ? SERVICE_MORE : SERVICE_IDLE;
424 }
425
426 const uint32_t rate = chan_rate_bps(channel);
427 if (!rate)
428 return SERVICE_IDLE;
429 const int queued_output = host_audio::queued(voice);
430 if (queued_output < 0)
431 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to query audio stream");
432 // Convert queued stereo-S16 bytes back to guest bytes for pacing.
433 const int queued = (queued_output / 4) << d->shift;
434 const int lead = dac_lead_bytes(channel, rate);
435 const int deficit = lead - queued;
436
437 auto elapsed = now - d->pace_last;
438 d->pace_last = now;
439 if (deficit <= 0)
440 {
441 d->credit_q16 = 0;
442 return SERVICE_IDLE;
443 }
444 const int64_t dt_ns = (std::min)((int64_t)
445 std::chrono::duration_cast<std::chrono::nanoseconds>(elapsed).count(),
446 (int64_t)50000000);
447 // 2x when more than a pull behind (start-up/stall), else 2% over-rate.
448 const int pull = dac_pull_bytes(rate);
449 const int rate_q8 = deficit > pull + pull / 4 ? 512 : 261;
450 if (dt_ns > 0)
451 d->credit_q16 += ((int64_t)rate * dt_ns * rate_q8 << 8) / 1000000000;
452
453 // Step cap of half a period (2..8ms) so a tick crosses at most one IRQ boundary.
454 const int sc_bytes = ((d->scount & 0xffff) + 1) << d->shift;
455 const int ms2 = (int)((u64)rate * 2 / 1000);
456 const int ms8 = (int)((u64)rate * 8 / 1000);
457 const int step_cap = (std::min)((std::max)(sc_bytes / 2, ms2), ms8);
458 if (d->credit_q16 > ((int64_t)step_cap << 16))
459 d->credit_q16 = (int64_t)step_cap << 16;
460
461 const int budget = (std::min)((int)(d->credit_q16 >> 16), deficit);
462 if (budget <= 0)
463 return SERVICE_IDLE;
464 const int moved = es1371_run_channel(&state, channel, budget);
465 d->credit_q16 -= (int64_t)moved << 16;
466 if (d->credit_q16 < 0)
467 d->credit_q16 = 0;
468 return SERVICE_IDLE;
469}

471void CES1371::run()
472{
473 try
474 {
475 int busy_spins = 0;
476 while (!m_audio_stop.load())
477 {
478 const auto now = std::chrono::steady_clock::now();
479 bool more = false, busy = false;
480 for (size_t i = 0; i < NB_CHANNELS; ++i)
481 {
482 const ServiceResult r = service_audio_channel(i, now);
483 more = more || r == SERVICE_MORE;
484 busy = busy || r == SERVICE_BUSY;
485 }
486 // Bus collisions are short; retry a few times before sleeping.
487 if (more || (busy && ++busy_spins < 16))
488 {
489 std::this_thread::yield();
490 continue;
491 }
492 busy_spins = 0;
493 std::unique_lock<std::mutex> wait_lock(m_audio_wait_mutex);
494 m_audio_wait.wait_for(wait_lock, std::chrono::milliseconds(1), [this] {
495 return m_audio_stop.load() || m_audio_wake.exchange(false);
496 });
497 }
498 }
499 catch (const CException& e)
500 {
501 printf("%s: Exception in audio refill thread: %s.\n",
502 devid_string, e.displayText().c_str());
503 m_audio_failed.store(true);
504 }
505 catch (const std::exception& e)
506 {
507 printf("%s: Exception in audio refill thread: %s.\n",
508 devid_string, e.what());
509 m_audio_failed.store(true);
510 }
511 catch (...)
512 {
513 printf("%s: Unknown exception in audio refill thread.\n", devid_string);
514 m_audio_failed.store(true);
515 }
516}

565{
566 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
567 if (!f || audio_running)
568 return -1;
569 const int res = CPCIDevice::RestoreState(f);
570 if (res)
571 return res;
572
573 ES1371SavedState saved{};
574 u32 magic = 0, size = 0;
575 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != es1371_magic1 ||
576 fread(&size, sizeof(size), 1, f) != 1 || size != sizeof(saved) ||
577 fread(&saved, sizeof(saved), 1, f) != 1 ||
578 fread(&magic, sizeof(magic), 1, f) != 1 || magic != es1371_magic2 ||
579 saved.mempage > 0xf || saved.src_phase > 1)
580 return -1;
581 for (size_t i = 0; i < NB_CHANNELS; ++i)
582 if (saved.channel[i].leftover > 3 || saved.channel[i].halted > 1)
583 return -1;
584
585 state.ctl = saved.ctl;
586 state.status = saved.status;
587 state.mempage = saved.mempage;
588 state.codec = saved.codec;
589 state.sctl = saved.sctl;
590 state.src = saved.src;
591 state.src_phase = saved.src_phase;
592 state.legacy = saved.legacy;
593 state.uart_control = saved.uart_control;
594 state.uart_test = saved.uart_test;
595 state.phantom_addr = saved.phantom_addr;
596 state.phantom_count = saved.phantom_count;
597 memcpy(state.src_ram, saved.src_ram, sizeof(state.src_ram));
598 memcpy(state.ac97, saved.ac97, sizeof(state.ac97));
599 for (size_t i = 0; i < NB_CHANNELS; ++i)
600 {
601 const auto& b = es1371_chan_bits[i];
602 const u32 fmt = (state.sctl & b.sctl_fmt) >> b.sctl_sh_fmt;
603 state.chan[i].shift = (fmt & 1) + (fmt >> 1);
604 state.chan[i].leftover = saved.channel[i].leftover;
605 state.chan[i].scount = saved.channel[i].scount;
606 state.chan[i].frame_addr = saved.channel[i].frame_addr;
607 state.chan[i].frame_cnt = saved.channel[i].frame_cnt;
608 state.chan[i].halted = saved.channel[i].halted;
609 state.chan[i].freq = channel_rate(i);
610 }
611 es1371_update_status(&state, state.status);
612 return 0;
613}

615void CES1371::es1371_update_status(ES1371State* s, uint32_t new_status)
616{
617 new_status &= ~STAT_UART;
618 if ((s->ctl & CTRL_UART_EN) &&
619 (s->uart_control & UCTRL_TXINTEN) == UCTRL_ENA_TXINT)
620 new_status |= STAT_UART;
621 new_status &= ~STAT_MPWR;
622 if ((endian_32(pci_state.config_data[0][0xe0 / 4]) & 3) != ((s->ctl >> 8) & 3))
623 new_status |= STAT_MPWR;
624 uint32_t level = (new_status & (STAT_DAC1 | STAT_DAC2 | STAT_ADC | STAT_UART)) |
625 (s->legacy & 0x01000000); // FIRQ is independent of the x86 trap engine.
626 if (s->ctl & CTRL_PWR_INTRM) level |= new_status & STAT_MPWR;
627
628 if (level) {
629 s->status = new_status | STAT_INTR;
630 }
631 else {
632 s->status = new_status & ~STAT_INTR;
633 }
634 do_pci_interrupt(0, !!level);
635}

639 size_t i;
640
641 s->ctl = 0xfcff0000;
642 s->status = 0x7ffffec0;
643 s->mempage = 0;
644 s->codec = 0;

667}

699uint32_t CES1371::channel_rate(size_t channel) const
700{
701 const unsigned base = 0x70 + (unsigned)channel * 4;
702 const uint32_t frequency = ((state.src_ram[base + 1] & 0xfc00U) << 5) |
703 (state.src_ram[base + 3] & 0x7fffU);
704 if (!frequency)
705 return 0;
706 uint64_t rate;
707 if (channel == ADC_CHANNEL) {
708 const uint32_t n = (state.src_ram[0x78] >> 4) & 0x1f;
709 rate = ((uint64_t)48000 * 32768 * n + frequency / 2) / frequency;
710 } else {
711 rate = ((uint64_t)frequency * 3000 + 16384) / 32768;
712 }
713 // Clamp invalid rates to the 48 kHz AC-link clock.
714 return (uint32_t)(std::min)(rate, (uint64_t)48000);
715}

717bool CES1371::channel_enabled(size_t channel, uint32_t ctl, uint32_t sctl) const
718{
719 const auto& b = es1371_chan_bits[channel];
720 const uint32_t disable = SRC_DIS_DAC1 >> channel;
721 return (ctl & b.ctl_en) && !(sctl & b.sctl_pause) &&
722 !(state.src & (SRC_DISABLE | disable)) &&
723 !(channel == ADC_CHANNEL && (ctl & CTRL_ADC_STOP)) &&
724 !state.chan[channel].halted && channel_rate(channel) != 0 &&
725 (endian_32(pci_state.config_data[0][0x04 / 4]) & 4) &&
726 !(endian_32(pci_state.config_data[0][0xe0 / 4]) & 3);
727}

729void CES1371::es1371_update_voices(ES1371State* s, uint32_t ctl, uint32_t sctl,
730 bool force)
731{
732 for (size_t i = 0; i < NB_CHANNELS; ++i) {
733 auto& d = s->chan[i];
734 const auto& b = es1371_chan_bits[i];
735 if ((ctl & b.ctl_en) && !(s->ctl & b.ctl_en))
736 d.halted = 0;
737 const uint32_t fmt = (sctl & b.sctl_fmt) >> b.sctl_sh_fmt;
738 const uint32_t old_fmt = (s->sctl & b.sctl_fmt) >> b.sctl_sh_fmt;
739 const uint32_t freq = channel_rate(i);
740 const bool format_changed = force || fmt != old_fmt || freq != d.freq;
741 const bool enabled = channel_enabled(i, ctl, sctl);
742 const bool changed = format_changed || enabled != m_audio_enabled[i].load() ||
743 ((ctl ^ s->ctl) & b.ctl_en) || ((sctl ^ s->sctl) & b.sctl_pause);
744 d.shift = (fmt & 1) + (fmt >> 1);
745 d.freq = freq;
746 if (!(audio_running || force) || !changed)
747 continue;
748 reset_audio_request(i, false, format_changed);
749 host_audio::Stream& voice = i == ADC_CHANNEL ? s->adc_voice : s->dac_voice[i];
750 // SDL rejects rebinding; unbind first without holding the stream lock.
751 host_audio::unbind_stream(voice);
752 if (format_changed && freq) {
753 host_audio::Format spec;
754 spec.freq = freq;
755 spec.channels = i == ADC_CHANNEL ? 1 << (fmt & 1) : 2;
756 spec.format = i == ADC_CHANNEL && !(fmt & 2) ? host_audio::SampleFormat::u8 : host_audio::SampleFormat::s16_le;
757 // Preserve SDL's host end of the stream when changing guest format.
758 if (!host_audio::set_format(voice,
759 i == ADC_CHANNEL ? nullptr : &spec,
760 i == ADC_CHANNEL ? &spec : nullptr))
761 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to set ES1371 audio format");
762 }
763 if (enabled) {
764 if (!host_audio::bind_stream(i == ADC_CHANNEL ? s->audio_be_in : s->audio_be_out, voice))
765 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to bind ES1371 audio stream");
766 } else {
767 host_audio::unbind_stream(voice);
768 }
769 reset_audio_request(i, enabled, false);
770 }
771 s->ctl = ctl;
772 s->sctl = sctl;
773}

775void CES1371::reset_codec()
776{
777 // TriTech TR28023 / Creative CT1297, a fixed-rate AC'97 1.0 codec.
778 memset(state.ac97, 0, sizeof(state.ac97));
779 state.ac97[0x02 / 2] = 0x8000;
780 state.ac97[0x06 / 2] = 0x8000;
781 state.ac97[0x0a / 2] = 0x0000;
782 state.ac97[0x0c / 2] = 0x8008;
783 state.ac97[0x0e / 2] = 0x8008;
784 for (unsigned reg = 0x10; reg <= 0x18; reg += 2)
785 state.ac97[reg / 2] = 0x8808;
786 state.ac97[0x1c / 2] = 0x8000;
787 state.ac97[0x26 / 2] = 0x000f;
788 state.ac97[0x7c / 2] = 0x5452;
789 state.ac97[0x7e / 2] = 0x4103;
790}

792void CES1371::write_codec(uint32_t command)
793{
794 const unsigned reg = (command >> 16) & 0x7f;
795 if (command & CODEC_READ) {
796 state.codec = (command & 0x00ff0000) | CODEC_READY |
797 ((reg & 1) ? 0xffffU : state.ac97[reg / 2]);
798 return;
799 }
800 state.codec = command & 0x007fffff;
801 if (reg & 1)
802 return;
803 uint16_t mask = 0;
804 switch (reg) {
805 case 0x00: reset_codec(); return;
806 case 0x02: mask = 0xbf3f; break;
807 case 0x06: mask = 0x803f; break;
808 case 0x0a: mask = 0x801e; break;
809 case 0x0c: mask = 0x801f; break;
810 case 0x0e: mask = 0x805f; break;
811 case 0x10: case 0x12: case 0x14: case 0x16: case 0x18:
812 mask = 0x9f1f; break;
813 case 0x1a: mask = 0x0707; break;
814 case 0x1c: mask = 0x8f0f; break;
815 case 0x20: mask = 0x0200; break;
816 case 0x26: {
817 // Derive ready flags from power state; PR4/PR5 are write-only.
818 const uint16_t power = command & 0x3f00;
819 uint16_t ready = (~(power >> 8)) & 0x0f;
820 if (power & 0x0800) ready &= ~0x0c;
821 state.ac97[reg / 2] = (power & 0x0f00) | ready;
822 return;
823 }
824 default: return; // Unsupported features and identity registers are read-only.
825 }
826 state.ac97[reg / 2] = command & mask;
827}

829int CES1371::apply_volume(size_t channel, uint8_t* data, int bytes)
830{
831 const auto& b = es1371_chan_bits[channel];
832 const uint32_t fmt = (state.sctl & b.sctl_fmt) >> b.sctl_sh_fmt;
833 const bool stereo = (fmt & 1) != 0;
834 const int sample_bytes = (fmt & 2) ? 2 : 1;
835 double gain[2];
836 for (unsigned side = 0; side < 2; ++side) {
837 const unsigned shift = side ? 0 : 8;
838 if (channel == ADC_CHANNEL) {
839 const uint16_t record = state.ac97[0x1c / 2];
840 const unsigned n = (state.src_ram[0x78] >> 4) & 0x1f;
841 gain[side] = (record & 0x8000) || (state.ac97[0x26 / 2] & 0x0100) || !n
842 ? 0.0 : std::pow(10.0, ((record >> shift) & 15) * 1.5 / 20.0) *
843 (int16_t)state.src_ram[0x6c + side] / (n * 256.0);
844 } else {
845 const uint16_t master = state.ac97[0x02 / 2];
846 const uint16_t pcm = state.ac97[0x18 / 2];
847 const double db = 12.0 - (((master >> shift) & 63) +
848 ((pcm >> shift) & 31)) * 1.5;
849 gain[side] = ((master | pcm) & 0x8000) || (state.ac97[0x26 / 2] & 0x0e00)
850 ? 0.0 : std::pow(10.0, db / 20.0) *
851 (int16_t)state.src_ram[0x7c + channel * 2 + side] / 4096.0;
852 }
853 }
854 if (channel != ADC_CHANNEL) {
855 const int frame_bytes = sample_bytes * (stereo ? 2 : 1);
856 const int frames = bytes / frame_bytes;
857 // Expand backwards to stereo S16 without overwriting unread samples.
858 for (int frame = frames - 1; frame >= 0; --frame) {
859 int samples[2];
860 for (unsigned side = 0; side < 2; ++side) {
861 const int pos = frame * frame_bytes + (stereo ? side * sample_bytes : 0);
862 samples[side] = sample_bytes == 2
863 ? (int16_t)((uint16_t)data[pos] | ((uint16_t)data[pos + 1] << 8))
864 : ((int)data[pos] - 128) * 256;
865 }
866 for (unsigned side = 0; side < 2; ++side) {
867 const double scaled = samples[side] * gain[side];
868 const int sample = (int)(std::max)(-32768.0, (std::min)(32767.0, scaled));
869 const int pos = frame * 4 + side * 2;
870 data[pos] = (uint8_t)sample;
871 data[pos + 1] = (uint8_t)((uint16_t)sample >> 8);
872 }
873 }
874 return frames * 4;
875 }
876 for (int pos = 0; pos + sample_bytes <= bytes; pos += sample_bytes) {
877 const unsigned side = stereo ? (pos / sample_bytes) & 1 : 0;
878 int sample = sample_bytes == 2
879 ? (int16_t)((uint16_t)data[pos] | ((uint16_t)data[pos + 1] << 8))
880 : ((int)data[pos] - 128) * 256;
881 const double scaled = sample * gain[side];
882 sample = (int)(std::max)(-32768.0, (std::min)(32767.0, scaled));
883 if (sample_bytes == 2) {
884 data[pos] = (uint8_t)sample;
885 data[pos + 1] = (uint8_t)((uint16_t)sample >> 8);
886 } else {
887 data[pos] = (uint8_t)((sample + 32768) >> 8);
888 }
889 }
890 return bytes;
891}
892uint32_t CES1371::es1371_fixup(ES1371State* s, uint32_t addr)
893{
894 addr &= 0xff;
895 if (addr >= 0x30 && addr <= 0x3f) {
896 addr |= s->mempage << 8;
897 }
898 return addr;
899}

901void CES1371::es1371_write(void* opaque, u64 addr, uint32_t val, uint32_t mask)
902{
903 ES1371State* s = (ES1371State*)opaque;
904 addr = es1371_fixup(s, (uint32_t)addr);
905 const auto merge = [val, mask](uint32_t old) { return (old & ~mask) | (val & mask); };
906 switch (addr) {
907 case ES1371_REG_CONTROL: {
908 // GPIO inputs and reserved high bits cannot be changed by the guest.
909 const uint32_t ctl = (merge(s->ctl) & 0x030fffff) | 0xfcf00000;
910 es1371_update_voices(s, ctl, s->sctl);
911 es1371_update_status(s, s->status);
912 break;
913 }
914 case ES1371_REG_MEMPAGE: s->mempage = merge(s->mempage) & 0xf; break;
915 case ES1371_REG_UART_DATA:
916 if (mask & 0xff00) {
917 s->uart_control = (val >> 8) & 0xe3;
918 if ((s->uart_control & UCTRL_CNTRL) == UCTRL_CNTRL_SWR)
919 s->uart_control = 0;
920 }
921 if (mask & 0xff0000) s->uart_test = (val >> 16) & 1;
922 // The transmitter consumes bytes immediately; there is no host MIDI sink.
923 es1371_update_status(s, s->status);
924 break;
925 case ES1371_REG_SRC: {
926 s->src = merge(s->src) & 0xff790000;
927 s->src_phase = 0;
928 if (s->src & SRC_WRITE) {
929 const unsigned reg = (s->src >> 25) & 0x7f;
930 const uint32_t data_mask = mask & 0xffff;
931 s->src_ram[reg] = (uint16_t)((s->src_ram[reg] & ~data_mask) | (val & data_mask));
932 }
933 es1371_update_voices(s, s->ctl, s->sctl);
934 break;
935 }
936 case ES1371_REG_CODEC:
937 // Command writes normally use a dword, but retain unwritten byte lanes.
938 write_codec(merge(s->codec));
939 break;
940 case ES1371_REG_LEGACY:
941 // No x86 legacy event trap is connected on the Alpha PCI bus.
942 s->legacy = (merge(s->legacy) & 0xffff0000) | 0x0000f801;
943 es1371_update_status(s, s->status);
944 break;
945 case ES1371_REG_SERIAL_CONTROL: {
946 const uint32_t serial = (merge(s->sctl) & 0x007fffff) | 0xff800000;
947 if ((serial & SCTRL_P1SCTRLD) && !(s->sctl & SCTRL_P1SCTRLD)) {
948 auto& d = s->chan[DAC1_CHANNEL];
949 d.scount = (d.scount & 0xffff) * 0x10001U;
950 }
951 es1371_maybe_lower_irq(s, serial);
952 es1371_update_voices(s, s->ctl, serial);
953 break;
954 }
955 case ES1371_REG_DAC1_SCOUNT:
956 case ES1371_REG_DAC2_SCOUNT:
957 case ES1371_REG_ADC_SCOUNT: {
958 if (!(mask & 0xffff)) break;
959 auto& d = s->chan[(addr - ES1371_REG_DAC1_SCOUNT) >> 2];
960 const uint32_t reload = merge(d.scount) & 0xffff;
961 d.scount = reload * 0x10001U;
962 break;
963 }
964 case ES1371_REG_DAC1_FRAMEADR:
965 case ES1371_REG_DAC2_FRAMEADR:
966 case ES1371_REG_ADC_FRAMEADR: {
967 const unsigned channel = addr == ES1371_REG_ADC_FRAMEADR ? ADC_CHANNEL :
968 (unsigned)(addr - ES1371_REG_DAC1_FRAMEADR) >> 3;
969 s->chan[channel].frame_addr = merge(s->chan[channel].frame_addr) & ~3U;
970 break;
971 }
972 case ES1371_REG_DAC1_FRAMECNT:
973 case ES1371_REG_DAC2_FRAMECNT:
974 case ES1371_REG_ADC_FRAMECNT: {
975 const unsigned channel = addr == ES1371_REG_ADC_FRAMECNT ? ADC_CHANNEL :
976 (unsigned)(addr - ES1371_REG_DAC1_FRAMECNT) >> 3;
977 s->chan[channel].frame_cnt = merge(s->chan[channel].frame_cnt);
978 s->chan[channel].leftover = 0;
979 break;
980 }
981 case ES1371_REG_PHANTOM_FRAMEADR:
982 s->phantom_addr = merge(s->phantom_addr) & ~3U; break;
983 case ES1371_REG_PHANTOM_FRAMECNT:
984 s->phantom_count = merge(s->phantom_count); break;
985 default: break;
986 }
987}

989uint64_t CES1371::es1371_read(void* opaque, u64 addr, unsigned size)
990{
991 ES1371State* s = (ES1371State*)opaque;
992 addr = es1371_fixup(s, (uint32_t)addr);
993 switch (addr) {
994 case ES1371_REG_CONTROL: return s->ctl;
995 case ES1371_REG_STATUS: return s->status;
996 case ES1371_REG_UART_DATA:
997 return (s->uart_test << 16) | (USTAT_TXRDY << 8) |
998 ((s->status & STAT_UART) ? USTAT_TXINT << 8 : 0);
999 case ES1371_REG_MEMPAGE: return s->mempage;
1000 case ES1371_REG_SRC: {
1001 uint32_t value = (s->src & ~SRC_STATE) | s->src_ram[(s->src >> 25) & 0x7f];
1002 // Report idle, then safe, to satisfy both driver polling sequences.
1003 if ((s->src & SRC_STATE) && s->src_phase++) value |= SRC_STATE;
1004 if (s->src_phase > 1) s->src_phase = 1;
1005 return value;
1006 }
1007 case ES1371_REG_CODEC: return s->codec;
1008 case ES1371_REG_LEGACY: return s->legacy;
1009 case ES1371_REG_SERIAL_CONTROL: return s->sctl;
1010 case ES1371_REG_DAC1_SCOUNT:
1011 case ES1371_REG_DAC2_SCOUNT:
1012 case ES1371_REG_ADC_SCOUNT:
1013 return s->chan[(addr - ES1371_REG_DAC1_SCOUNT) >> 2].scount;
1014 case ES1371_REG_DAC1_FRAMEADR: return s->chan[0].frame_addr;
1015 case ES1371_REG_DAC2_FRAMEADR: return s->chan[1].frame_addr;
1016 case ES1371_REG_ADC_FRAMEADR: return s->chan[2].frame_addr;
1017 case ES1371_REG_DAC1_FRAMECNT: return s->chan[0].frame_cnt;
1018 case ES1371_REG_DAC2_FRAMECNT: return s->chan[1].frame_cnt;
1019 case ES1371_REG_ADC_FRAMECNT: return s->chan[2].frame_cnt;
1020 case ES1371_REG_PHANTOM_FRAMEADR: return s->phantom_addr;
1021 case ES1371_REG_PHANTOM_FRAMECNT: return s->phantom_count;
1022 case 0xe30: return 0; // Empty UART receive FIFO (VALID clear).
1023 default: return ~0U;
1024 }
1025}

1027u32 CES1371::ReadMem_Bar(int func, int bar, u32 address, int dsize)
1028{
1029 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
1030 if (func != 0 || bar != 0 || address >= 0x40 ||
1031 (dsize != 8 && dsize != 16 && dsize != 32) ||
1032 (address & (dsize / 8 - 1)))
1033 return ~0U;
1034 const uint32_t value = (uint32_t)es1371_read(&state, address & ~3U, 32);
1035 const uint32_t mask = dsize == 32 ? ~0U : (1U << dsize) - 1;
1036 return (value >> ((address & 3) * 8)) & mask;
1037}

1039void CES1371::WriteMem_Bar(int func, int bar, u32 address, int dsize, u32 data)
1040{

1042 if (func != 0 || bar != 0 || address >= 0x40 ||

1044 (address & (dsize / 8 - 1)))
1045 return;
1046 const unsigned shift = (address & 3) * 8;

1048 es1371_write(&state, address & ~3U, data << shift, mask);

1050int CES1371::es1371_transfer_audio(ES1371State* s, struct chan* d, int loop_sel,
1051 int maxb, bool* irq)
1052{
1053 uint8_t tmpbuf[4096 * 4]; // Maximum mono-U8 to stereo-S16 expansion.
1054 const int index = (int)(d - s->chan);
1055 const int sc = d->scount & 0xffff;
1056 int csc_bytes = ((d->scount >> 16) + 1) << d->shift;
1057 const bool nonloop = (s->sctl & loop_sel) != 0;
1058 int remaining = maxb;
1059 bool period_done = false;
1060
1061 // Loop only to wrap the frame buffer; one sample-count period (IRQ) per call.
1062 while (remaining > 0 && !period_done) {
1063 int cnt = d->frame_cnt >> 16;
1064 const int size = d->frame_cnt & 0xffff;
1065 if (size < cnt) break;
1066
1067 // leftover bytes in the current dword have already been consumed.
1068 const int left = ((size - cnt + 1) << 2) - (int)d->leftover;
1069 if (left <= 0) break;
1070 const int target = MIN(remaining, MIN(left, csc_bytes));
1071 uint32_t addr = d->frame_addr + (cnt << 2) + d->leftover;
1072 int transferred = 0;
1073 while (transferred < target) {
1074 const int to_copy = MIN(target - transferred, 4096);
1075 int copied;
1076 if (index == ADC_CHANNEL) {
1077 copied = host_audio::read(s->adc_voice, tmpbuf, to_copy);
1078 if (copied <= 0) break;
1079 apply_volume(index, tmpbuf, copied);
1080 do_pci_write(addr, tmpbuf, 1, copied);
1081 } else {
1082 do_pci_read(addr, tmpbuf, 1, to_copy);
1083 const int output_bytes = apply_volume(index, tmpbuf, to_copy);
1084 copied = host_audio::write(s->dac_voice[index], tmpbuf, output_bytes) ? to_copy : 0;
1085 if (!copied) break;
1086 }
1087 addr += copied;
1088 transferred += copied;
1089 }
1090 remaining -= transferred;
1091 csc_bytes -= transferred;
1092 if (!csc_bytes) {
1093 *irq = true;
1094 period_done = true;
1095 csc_bytes = (sc + 1) << d->shift;
1096 if (nonloop) d->halted = 1;
1097 }
1098 d->scount = sc | (((csc_bytes - 1) >> d->shift) << 16);
1099
1100 cnt += (transferred + d->leftover) >> 2;
1101 d->frame_cnt = size;
1102 if (cnt <= size) d->frame_cnt |= cnt << 16;
1103 d->leftover = (transferred + d->leftover) & 3;
1104
1105 // Stop on an SDL failure/empty capture queue.
1106 if (transferred < target) break;
1107 }
1108 return maxb - remaining;
1109}
