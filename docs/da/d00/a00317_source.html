57// Log/error any asmjit emit failure (badly formed instruction / bad operand) that the Assembler
58// would otherwise accept and then ship as a silently truncated block/trace. Strict instruction
59// validation and formatted disassembly remain optional; rejecting a failed image is not.
60class JitErrorHandler : public asmjit::ErrorHandler {
61public:
62 FILE* fp = nullptr; // disassembly trace file (falls back to stderr if unopened)
63 int cpu_id = -1;
64 bool failed = false;
65 void handle_error(asmjit::Error err, const char* message, asmjit::BaseEmitter*) override {
66 (void) err;
67 failed = true;
68 fprintf(fp ? fp : stderr, "[JIT][CPU%d][EMIT-ERROR] %s\n", cpu_id, message);
69 }
70};

72enum SafeOp {
73 OP_NONE,
74 OP_ADDQ, OP_SUBQ, OP_ADDL, OP_SUBL,
75 OP_S4ADDQ, OP_S8ADDQ, OP_S4SUBQ, OP_S8SUBQ,
76 OP_S4ADDL, OP_S8ADDL, OP_S4SUBL, OP_S8SUBL, // INTA (0x10) scaled longword: sext32((Ra*scale) +/- Rb)
77 OP_CMPBGE, // INTA (0x10) per-byte unsigned compare -> 8-bit mask
78 OP_SEXTB, OP_SEXTW, // FPTI (0x1c) CIX/BWX: Rc = sign-extend byte/word of op2
79 OP_CTPOP, OP_CTLZ, OP_CTTZ, // FPTI (0x1c) CIX bit-count of op2: popcount / leading / trailing zeros
80 OP_ITOFS, OP_ITOFF, OP_ITOFT, // ITFP (0x14) int->FP reg moves: f[Fc] = fmt(Ra)
81 OP_FTOIS, OP_FTOIT, // FPTI (0x1c) FP->int reg moves: Rc = fmt(f[Fa])
82 OP_FLTL, // FLTL (0x17) non-arithmetic: FPCR moves, CPYSx, FCMOVx, CVTLQ/QL
83 OP_CVTQT, OP_CVTQS, // FLTI (0x16) int->IEEE convert: f[Fc] = (T/S)(s64)f[Fb] via SSE
84 OP_ADDT, OP_SUBT, OP_MULT, OP_DIVT, // FLTI (0x16) IEEE T-float arith: f[Fc] = f[Fa] op f[Fb] via SSE
85 OP_CMPTUN, OP_CMPTEQ, OP_CMPTLT, OP_CMPTLE, // FLTI (0x16) IEEE T-float compares: f[Fc] = (cmp) ? 2.0 : 0.0
86 OP_ADDS, OP_SUBS, OP_MULS, OP_DIVS, // FLTI (0x16) IEEE S-float arith: f[Fc] = round_single(f[Fa] op f[Fb])
87 OP_CVTST, OP_CVTTS, OP_CVTTQ, // FLTI (0x16) IEEE converts: S->T widen, T->S narrow, T->Q to int
88 OP_SQRTS, OP_SQRTT, // ITFP (0x14) IEEE sqrt: f[Fc] = sqrt(f[Fb]) via sqrtss/sqrtsd
89 OP_FLTV, // FLTV (0x15) VAX arith/convert/compare: f[Fc] via jit_fltv helper
90 OP_AND, OP_BIS, OP_XOR, OP_BIC, OP_ORNOT, OP_EQV,
91 OP_CMOV, // INTL (0x11) conditional moves CMOVxx: Rc = cond(Ra) ? op2 : Rc
92 OP_AMASK, OP_IMPLVER, // INTL (0x11) CPU feature probes from the selected profile
93 OP_CMPEQ, OP_CMPLT, OP_CMPLE, OP_CMPULT, OP_CMPULE,
94 OP_SLL, OP_SRL, OP_SRA, OP_MULQ,
95 OP_MULL, OP_UMULH, // INTM (0x13): MULL = sext32(Ra*op2); UMULH = hi64 of unsigned Ra*op2
96 OP_EXTL, OP_EXTH, OP_INSL, OP_INSH, OP_MSKL, OP_MSKH, OP_ZAP, // INTS (0x12) byte-manip (Rb&7 keyed)
97 OP_NOP, OP_MFENCE, // MISC (0x18): prefetch/cache hints (no-op), barriers (mfence)
98 OP_RPCC, OP_RC, OP_RS, // MISC (0x18) state reads (Ra dest): cycle counter / read-and-clear,set intr flag
99 OP_LDQ, OP_LDL, // memory-format loads: Ra = MEM[Rb + disp16]
100 OP_LDBU, OP_LDWU, // BWX byte/word loads (0x0a/0x0c): Ra = zero-extend MEM[Rb+disp16].{b,w}
101 OP_LDL_L, OP_LDQ_L, // load-locked (0x2a/0x2b): Ra = MEM[Rb+disp16] + establish LL/SC monitor
102 OP_STL_C, OP_STQ_C, // store-conditional (0x2e/0x2f): cond store Ra -> MEM; Ra = success(1)/fail(0)
103 OP_STQ, OP_STL, // memory-format stores: MEM[Rb + disp16] = Ra
104 OP_STB, OP_STW, // BWX byte/word stores (0x0e/0x0d): MEM[Rb + disp16].{b,w} = Ra
105 OP_LDQ_U, OP_STQ_U, // unaligned quad (0x0b/0x0f): MEM[(Rb+disp16) & ~7] load/store
106 OP_LDT, OP_LDS, OP_STT, OP_STS, // FP memory (0x23/0x22/0x27/0x26): f[Fa] <-> MEM, LDT/STT raw, LDS/STS ieee conv
107 OP_LDF, OP_LDG, OP_STF, OP_STG, // VAX FP memory (0x20/0x21/0x24/0x25): f[Fa] <-> MEM, F/G format conversion (helper)
108 OP_LDA, OP_LDAH, // load-address: Ra = Rb + disp16 (<<16 for LDAH); pure ALU
109 OP_HW_MFPR, // HW_MFPR (0x19), PALmode only: family-selected IPR read into Ra
110 OP_HW_LDL, OP_HW_LDQ, // HW_LD (0x1b): normalized physical longword/quadword loads
111 OP_HW_LD_VIRT, // HW_LD: normalized virtual loads with family-selected access flags
112 OP_HW_MTPR, // HW_MTPR (0x1d): family-approved helper writes, PALmode only
113 OP_HW_MTPR_TERM, // HW_MTPR: a family write changes compiled assumptions -> redispatch
114 OP_HW_STL, OP_HW_STQ, // HW_ST (0x1f): normalized physical longword/quadword stores
115 OP_JMP, // JMP/JSR/RET (0x1a): Ra = PC+4; PC = Rb & ~3 (computed target)
116 OP_HW_RET, // HW_RET (0x1e), PALmode: family-selected pure return-target transform
117 OP_CALL_PAL, // CALL_PAL (0x00): save exc_addr/optional link; PC = pal_base | entry offset
118 // Branch-format terminators (contiguous; see is_branch). Conditional on Ra, plus BR/BSR.
119 OP_BEQ, OP_BNE, OP_BLT, OP_BLE, OP_BGT, OP_BGE, OP_BLBC, OP_BLBS, OP_BR, OP_BSR,
120 OP_FBEQ, OP_FBNE, OP_FBLT, OP_FBLE, OP_FBGT, OP_FBGE // FP branches (0x31-0x37): branch on f[Fa] vs 0.0
121};

123static inline bool is_branch(SafeOp op) { return op >= OP_BEQ && op <= OP_FBGE; }
124static inline bool is_fp_branch(SafeOp op) { return op >= OP_FBEQ && op <= OP_FBGE; }
125static inline bool is_store(SafeOp op) { return op == OP_STQ || op == OP_STL; }
126// A terminator ends the block and writes its own next PC (branches + the computed jump).
127static inline bool is_terminator(SafeOp op) { return op == OP_JMP || op == OP_HW_RET || op == OP_CALL_PAL || op == OP_HW_MTPR_TERM || is_branch(op); }

141// Safe = goto-free, register-only operate-format ops (no trap, memory, or branch).
142// pal_block enables PALmode-only ops (HW_MFPR): outside PALmode they'd OPCDEC, so only
143// compile them when the block is PALmode (the dispatcher keys blocks by PC bit 0).
144SafeOp classify(uint32_t ins, bool pal_block, const AlphaIPRReadPolicy& ipr_read,
145 const AlphaIPRWritePolicy& ipr_write, const AlphaPALPolicy& pal,
146 const AlphaHWLoadPolicy& hw_load, const AlphaHWStorePolicy& hw_store)
147{
148 uint32_t opcode = ins >> 26;
149 uint32_t func = (ins >> 5) & 0x7F;
150 switch (opcode) {
151 case 0x10: // INTA
152 switch (func) {
153 case 0x20: return OP_ADDQ; case 0x29: return OP_SUBQ;
154 case 0x00: return OP_ADDL; case 0x09: return OP_SUBL;
155 case 0x22: return OP_S4ADDQ; case 0x32: return OP_S8ADDQ;
156 case 0x2b: return OP_S4SUBQ; case 0x3b: return OP_S8SUBQ;
157 case 0x02: return OP_S4ADDL; case 0x12: return OP_S8ADDL;
158 case 0x0b: return OP_S4SUBL; case 0x1b: return OP_S8SUBL;
159 case 0x0f: return OP_CMPBGE;
160 case 0x2d: return OP_CMPEQ; case 0x4d: return OP_CMPLT;
161 case 0x6d: return OP_CMPLE; case 0x1d: return OP_CMPULT;
162 case 0x3d: return OP_CMPULE;
163 // ADDL/V (0x40), SUBL/V (0x49), ADDQ/V (0x60), SUBQ/V (0x69): overflow-trapping -> interpret
164 }
165 break;
166 case 0x11: // INTL
167 switch (func) {
168 case 0x00: return OP_AND; case 0x20: return OP_BIS;
169 case 0x40: return OP_XOR; case 0x08: return OP_BIC;
170 case 0x28: return OP_ORNOT; case 0x48: return OP_EQV;
171 case 0x14: case 0x16: case 0x24: case 0x26: // CMOVLBS/LBC/EQ/NE
172 case 0x44: case 0x46: case 0x64: case 0x66: // CMOVLT/GE/LE/GT
173 return OP_CMOV;
174 case 0x61: // AMASK: Ra must be R31 (else the interp's
175 // DO_AMASK traps OPCDEC) -- compile only the architecturally valid form.
176 if (((ins >> 21) & 0x1f) != 31) return OP_NONE;
177 return OP_AMASK;
178 case 0x6c: return OP_IMPLVER; // IMPLVER: Rc = implementation version constant
179 }
180 break;
181 case 0x12: // INTS: shifts + byte-manipulation (extract / insert / mask / zap), Rb&7 keyed
182 switch (func) {
183 case 0x39: return OP_SLL; case 0x34: return OP_SRL; case 0x3c: return OP_SRA;
184 case 0x06: case 0x16: case 0x26: case 0x36: return OP_EXTL; // EXTBL/WL/LL/QL
185 case 0x5a: case 0x6a: case 0x7a: return OP_EXTH; // EXTWH/LH/QH
186 case 0x0b: case 0x1b: case 0x2b: case 0x3b: return OP_INSL; // INSBL/WL/LL/QL
187 case 0x57: case 0x67: case 0x77: return OP_INSH; // INSWH/LH/QH
188 case 0x02: case 0x12: case 0x22: case 0x32: return OP_MSKL; // MSKBL/WL/LL/QL
189 case 0x52: case 0x62: case 0x72: return OP_MSKH; // MSKWH/LH/QH
190 case 0x30: case 0x31: return OP_ZAP; // ZAP / ZAPNOT
191 }
192 break;
193 case 0x13: // INTM (integer multiply). MULL/V (0x40) + MULQ/V (0x60) overflow-trap -> interpret.
194 if (func == 0x20) return OP_MULQ;
195 if (func == 0x00) return OP_MULL; // 32-bit multiply, low 32 sign-extended
196 if (func == 0x30) return OP_UMULH; // high 64 bits of the unsigned 128-bit product
197 break;
198 case 0x14: { // ITFP: compile the int->FP register moves + IEEE SQRT*. Fc==31 is interpreted
199 // (the interp zeroes f[31] per instruction -- a compiled write would desync).
200 const uint32_t f14 = (ins >> 5) & 0x7ff;
201 if ((ins & 0x1f) == 31) return OP_NONE; // Fc==31 -> interp
202 const uint32_t sb = f14 & 0x3f; // IEEE sqrt (source Fb): SQRTS 0x0b / SQRTT 0x2b
203 if (sb == 0x0b || sb == 0x2b) {
204 if (((ins >> 21) & 0x1f) != 31) return OP_NONE; // Ra must be R31 (else DO_SQRT* traps OPCDEC)
205 if (((f14 & 0x600) == 0x200) || ((f14 & 0x500) == 0x400)) return OP_NONE; // invalid qualifier -> OPCDEC
206 const uint32_t r = (f14 >> 6) & 3;
207 if (r == 0 || r == 1) return OP_NONE; // /C, /M: SSE rounds nearest -> interpret
208 if (((f14 >> 8) & 7) == 7) return OP_NONE; // /SUI -> interpret
209 return (sb == 0x2b) ? OP_SQRTT : OP_SQRTS;
210 }
211 if (((ins >> 16) & 0x1f) != 31) return OP_NONE; // ITOFx: Rb must be 31 (Ra is the int source)
212 if (f14 == 0x004) return OP_ITOFS;
213 if (f14 == 0x014) return OP_ITOFF;
214 if (f14 == 0x024) return OP_ITOFT;
215 return OP_NONE;
216 }
217 case 0x17: { // FLTL non-arithmetic: sign-copies and FCMOVs inline, FPCR moves and CVTLQ/QL
218 // via jit_fltl; only the /V trap variants of CVTQL stay interpreted.
219 const uint32_t f17 = (ins >> 5) & 0x7ff;
220 const bool ok = f17 == 0x010 || (f17 >= 0x020 && f17 <= 0x022) || f17 == 0x024
221 || f17 == 0x025 || (f17 >= 0x02a && f17 <= 0x02f) || f17 == 0x030;
222 if (!ok) return OP_NONE;
223 // f31-dest gate (interp zeroes f[31] per instr): MF_FPCR writes f[Fa]; MT_FPCR has no FP dest
224 if (f17 == 0x025) { if (((ins >> 21) & 0x1f) == 31) return OP_NONE; }
225 else if (f17 != 0x024) { if ((ins & 0x1f) == 31) return OP_NONE; }
226 return OP_FLTL;
227 }
228 case 0x15: { // FLTV (VAX): helper-dispatched arith/convert/compare; bail to the interp on trap.
229 if ((ins & 0x1f) == 31) return OP_NONE; // Fc==31: interp zeroes f[31] per instr
230 const uint32_t f = (ins >> 5) & 0x7ff;
231 if (f == 0x0a5 || f == 0x4a5 || f == 0x0a6 || f == 0x4a6 || f == 0x0a7 || f == 0x4a7 // CMPGEQ/LT/LE
232 || f == 0x03c || f == 0x0bc || f == 0x03e || f == 0x0be) return OP_FLTV; // CVTQF/CVTQG
233 if (f & 0x200) return OP_NONE; // invalid qualifier -> interp OPCDECs (UNKNOWN2)
234 switch (f & 0x7f) { // base-op arith / G<->F<->D / G->Q convert
235 case 0x000: case 0x001: case 0x002: case 0x003: // ADDF/SUBF/MULF/DIVF
236 case 0x01e: // CVTDG
237 case 0x020: case 0x021: case 0x022: case 0x023: // ADDG/SUBG/MULG/DIVG
238 case 0x02c: case 0x02d: case 0x02f: // CVTGF/CVTGD/CVTGQ
239 return OP_FLTV;
240 }
241 return OP_NONE; // other 0x15 funcs: interp OPCDECs
242 }
243 case 0x16: { // FLTI (IEEE): SSE-inline the int->float converts; bail to the interp on traps/edges.
244 const uint32_t f = (ins >> 5) & 0x7ff;
245 if ((ins & 0x1f) == 31) return OP_NONE; // Fc==31: interp zeroes f[31] per instr (all 0x16)
246 if (f == 0x2ac || f == 0x6ac) return OP_CVTST; // CVTST (S->T): before the invalid gate (SRC bit set);
247 // S-denormal renormalizes / NaN quiets -> bail, else valid-T copy
248 if ((f & 0x3f) == 0x2f) { // CVTTQ (T->Q int): /C chop is valid -> handle before round gate
249 if (((f >> 6) & 3) == 1) return OP_NONE; // /M -> interp
250 if (((f >> 8) & 7) == 7) return OP_NONE; // /SUI -> interp
251 if (((f & 0x600) == 0x200) || ((f & 0x500) == 0x400)) return OP_NONE; // invalid qualifier
252 return OP_CVTTQ;
253 }
254 if (((f & 0x600) == 0x200) || ((f & 0x500) == 0x400)) return OP_NONE; // invalid qualifier -> OPCDEC
255 const uint32_t rnd = (f >> 6) & 3; // 0=/C 1=/M 2=/N 3=/D (dynamic, runtime-checked)
256 if (rnd == 0 || rnd == 1) return OP_NONE; // chopped / minus-inf: SSE rounds to nearest -> interpret
257 if (((f >> 8) & 7) == 7) return OP_NONE; // /SUI: traps on every inexact -> interpret
258 if (f == 0x0a4 || f == 0x5a4) return OP_CMPTUN; // IEEE compares (None + /SU): f[Fc] = (cmp) ? 2.0 : 0.0
259 if (f == 0x0a5 || f == 0x5a5) return OP_CMPTEQ; // full-function; a NaN operand bails to the interp
260 if (f == 0x0a6 || f == 0x5a6) return OP_CMPTLT;
261 if (f == 0x0a7 || f == 0x5a7) return OP_CMPTLE;
262 const uint32_t baseop = f & 0x3f;
263 if (baseop == 0x00) return OP_ADDS; // ADDS \ IEEE S-float (single) arith via SSE
264 if (baseop == 0x01) return OP_SUBS; // SUBS >
265 if (baseop == 0x02) return OP_MULS; // MULS |
266 if (baseop == 0x03) return OP_DIVS; // DIVS /
267 if (baseop == 0x20) return OP_ADDT; // ADDT \ IEEE T-float (double) arith via SSE:
268 if (baseop == 0x21) return OP_SUBT; // SUBT > f[Fc] = f[Fa] op f[Fb]
269 if (baseop == 0x22) return OP_MULT; // MULT |
270 if (baseop == 0x23) return OP_DIVT; // DIVT /
271 if (baseop == 0x2c) return OP_CVTTS; // CVTTS (T->S narrow): f[Fc] = round_single(f[Fb])
272 if (baseop == 0x3e && (f & 0x300) != 0x100) return OP_CVTQT; // CVTQT: f[Fc] = (double)(s64) f[Fb]
273 if (baseop == 0x3c && (f & 0x300) != 0x100) return OP_CVTQS; // CVTQS: f[Fc] = (single)(s64) f[Fb]
274 return OP_NONE; // all other 0x16 funcs interpreted
275 }
276 case 0x1c: // FPTI (CIX/BWX/MVI/FP-moves): the sign-extends, bit-counts, and FP->int register
277 // moves compile here (CTPOP needs host POPCNT). MVI packed media (PERR/MIN/MAX/PK/UNPK)
278 // stays interpreted for now.
279 if (func == 0x00) return OP_SEXTB; // sign-extend byte of op2
280 if (func == 0x01) return OP_SEXTW; // sign-extend word of op2
281 if (func == 0x32) return OP_CTLZ; // count leading zeros of op2 (BSR; op2==0 -> 64)
282 if (func == 0x33) return OP_CTTZ; // count trailing zeros of op2 (BSF; op2==0 -> 64)
283 if (func == 0x30 && host_has_popcnt()) return OP_CTPOP; // popcount; interpret when host lacks POPCNT
284 if (func == 0x70 || func == 0x78) { // FTOIT / FTOIS: Rb must be 31; Rc==31 -> interpret
285 if (((ins >> 16) & 0x1f) != 31 || (ins & 0x1f) == 31) return OP_NONE;
286 return (func == 0x70) ? OP_FTOIT : OP_FTOIS;
287 }
288 break;
289 case 0x18: // MISC: MB -> full fence; TRAPB/EXCB/WMB + prefetch/cache hints -> no code
290 switch (ins & 0xFFFF) {
291 case 0x4000: return OP_MFENCE; // MB: StoreLoad fence
292 case 0x0000: case 0x0400: case 0x4400: // TRAPB, EXCB, WMB: no code under precise traps and TSO
293 case 0x8000: case 0xA000: case 0xE800: // FETCH, FETCH_M, ECB
294 case 0xF800: case 0xFC00: return OP_NOP; // WH64, WH64EN
295 // RPCC/RC/RS read time-varying / consumed state into Ra; the verify log+replays the read so the
296 // two passes agree. RC/RS also side-effect the soft-interrupt flag, so the Ra==31 (no GPR dest,
297 // flag-only) forms are compiled too -- the helper still runs, the store is skipped. RPCC Ra==31
298 // is a pure no-op read (nothing to replay), so it stays interpreted.
299 case 0xC000: if (((ins >> 21) & 0x1f) != 31) return OP_RPCC; break; // RPCC (cycle counter)
300 case 0xE000: return OP_RC; // RC (read & clear soft-intr flag); Ra==31 -> clear-only
301 case 0xF000: return OP_RS; // RS (read & set soft-intr flag); Ra==31 -> set-only
302 }
303 break;
304 case 0x00: { // CALL_PAL: compile valid standard funcs (priv 0x00-0x3f, unpriv 0x80-0xbf)
305 const uint32_t fn = ins & 0x1FFFFFFF;
306 if (fn == 0x3E) break; // WTINT: interpret -> DO_CALL_PAL (native idle completion when enabled)
307 if (fn <= 0x3F || (fn >= 0x80 && fn <= 0xBF)) return OP_CALL_PAL;
308 break; // SRM specials (0x1234xx) / invalid ranges -> interpret
309 }
310 case 0x08: return OP_LDA; // load address (Ra = Rb + disp16) -- pure ALU, no memory
311 case 0x09: return OP_LDAH; // load address high (Ra = Rb + (disp16 << 16))
312 case 0x19: { // HW_MFPR: read IPR -> Ra. PALmode-only (else OPCDEC).
313 if (!pal_block || !pal.hardware_instruction_access
314 || !alpha_pal_instruction_compilable(pal.hardware_instruction_access(ins)))
315 return OP_NONE;
316 const auto descriptor = ipr_read.describe(ins);
317 if (descriptor.kind == AlphaIPRReadKind::Unsupported) return OP_NONE;
318 if (((ins >> 21) & 0x1f) == 31 && descriptor.discard_safe) return OP_NOP;
319 return OP_HW_MFPR;
320 }
321 case 0x1b: { // HW_LD: the family declares the normalized access and eligibility.
322 if (!pal_block || !pal.hardware_instruction_access
323 || !alpha_pal_instruction_compilable(pal.hardware_instruction_access(ins)))
324 return OP_NONE;
325 const auto descriptor = alpha_decode_hw_load(hw_load, ins);
326 if (!alpha_hw_load_compilable(descriptor)) return OP_NONE;
327 if (descriptor.address == AlphaHWLoadAddress::Physical) {
328 if (((ins >> 21) & 0x1f) == 31 && !descriptor.discard_safe) return OP_NONE;
329 return descriptor.size_bits == 32 ? OP_HW_LDL : OP_HW_LDQ;
330 }
331 if (descriptor.address == AlphaHWLoadAddress::Virtual) return OP_HW_LD_VIRT;
332 return OP_NONE;
333 }
334 case 0x1d: { // HW_MTPR (PALmode): the family declares compiled-safe writes.
335 if (!pal_block || !pal.hardware_instruction_access
336 || !alpha_pal_instruction_compilable(pal.hardware_instruction_access(ins)))
337 return OP_NONE;
338 switch (ipr_write.describe(ins)) {
339 case AlphaIPRWriteKind::Helper: return OP_HW_MTPR;
340 case AlphaIPRWriteKind::HelperTerminator: return OP_HW_MTPR_TERM;
341 case AlphaIPRWriteKind::Noop: return OP_NOP;
342 case AlphaIPRWriteKind::Interpreter: return OP_NONE;
343 }
344 return OP_NONE;
345 }
346 case 0x1f: { // HW_ST: the family declares normalized compiled-safe physical writes.
347 if (!pal_block || !pal.hardware_instruction_access
348 || !alpha_pal_instruction_compilable(pal.hardware_instruction_access(ins)))
349 return OP_NONE;
350 const auto descriptor = alpha_decode_hw_store(hw_store, ins);
351 if (!alpha_hw_store_compilable(descriptor)) return OP_NONE;
352 return descriptor.size_bits == 32 ? OP_HW_STL : OP_HW_STQ;
353 }
354 case 0x1a: return OP_JMP; // JMP/JSR/RET: computed jump (target = Rb & ~3). Now compiled --
355 // chained in-frame via a block-cache lookup (jit_indirect), which
356 // handles varying targets without the old single-slot thrash.
357 case 0x1e: // HW_RET (HWREI): the family selects a compilable pure target transform.
358 return pal_block && pal.hardware_instruction_access
359 && alpha_pal_instruction_compilable(pal.hardware_instruction_access(ins))
360 && alpha_pal_return_compilable(alpha_pal_return_descriptor(pal, ins))
361 ? OP_HW_RET : OP_NONE;
362 case 0x28: return OP_LDL; // memory-format loads (Ra = MEM[Rb+disp16])
363 case 0x29: return OP_LDQ;
364 case 0x0a: return OP_LDBU; // BWX byte/word loads (Ra = zero-extend MEM[Rb+disp16].{b,w})
365 case 0x0c: return OP_LDWU;
366 case 0x0b: return OP_LDQ_U; // unaligned quad load: Ra = MEM[(Rb+disp16) & ~7]
367 case 0x2a: // LDL_L / LDQ_L: the load-locked half of LL/SC. Compile the value-
368 case 0x2b: // returning forms only -- Ra==31 (lock without consuming the value)
369 // leaves the loaded value out of the GPRs, so the verify can't replay it; interpret those.
370 if (((ins >> 21) & 0x1f) == 31) return OP_NONE;
371 return (opcode == 0x2a) ? OP_LDL_L : OP_LDQ_L;
372 case 0x2c: return OP_STL; // memory-format stores (MEM[Rb+disp16] = Ra)
373 case 0x2d: return OP_STQ;
374 case 0x0e: return OP_STB; // BWX byte/word stores (MEM[Rb+disp16].{b,w} = Ra low bits)
375 case 0x0d: return OP_STW;
376 case 0x0f: return OP_STQ_U; // unaligned quad store: MEM[(Rb+disp16) & ~7] = Ra
377 case 0x23: return OP_LDT; // FP loads (f[Fa] = MEM[Rb+disp16]): LDT raw 8B
378 case 0x22: return OP_LDS; // LDS ieee_lds(4B)
379 case 0x27: return OP_STT; // FP stores (MEM[Rb+disp16] = f[Fa]): STT raw 8B
380 case 0x26: return OP_STS; // STS ieee_sts(4B)
381 case 0x20: return OP_LDF; // VAX FP loads: LDF vax_ldf(4B)
382 case 0x21: return OP_LDG; // LDG vax_ldg(8B)
383 case 0x24: return OP_STF; // VAX FP stores: STF vax_stf(4B)
384 case 0x25: return OP_STG; // STG vax_stg(8B)
385 case 0x2e: // STL_C / STQ_C: the store-conditional half of LL/SC. Ra is both the
386 case 0x2f: // value AND the success/fail dest. Compile Ra!=31 (Ra==31 discards the
387 // result into R31, so the verify can't capture it from a GPR); interpret those.
388 if (((ins >> 21) & 0x1f) == 31) return OP_NONE;
389 return (opcode == 0x2e) ? OP_STL_C : OP_STQ_C;
390 // Branch format: opcode | Ra | disp21. Conditional on Ra, plus BR/BSR.
391 case 0x30: return OP_BR; case 0x34: return OP_BSR;
392 case 0x38: return OP_BLBC; case 0x39: return OP_BEQ;
393 case 0x3a: return OP_BLT; case 0x3b: return OP_BLE;
394 case 0x3c: return OP_BLBS; case 0x3d: return OP_BNE;
395 case 0x3e: return OP_BGE; case 0x3f: return OP_BGT;
396 case 0x31: return OP_FBEQ; case 0x32: return OP_FBLT; // FP branches: FPSTART, then f[Fa] vs 0.0
397 case 0x33: return OP_FBLE; case 0x35: return OP_FBNE;
398 case 0x36: return OP_FBGE; case 0x37: return OP_FBGT;
399 }
400 return OP_NONE;
401}

446#ifdef JIT_REGPROF
447// Bitmask of the Alpha integer GPRs a block's prefix touches (read or written), for pin selection.
448// Format-aware over the ops that drive the store-forward chains -- integer operate, memory, branch,
449// JMP, and the MISC state reads; FP / HW / CALL_PAL are skipped (not the GPR forwarding path).
450static uint32_t regprof_mask(const uint32_t* w, uint32_t n)
451{
452 uint32_t mask = 0;
453 auto touch = [&](int r) { if (r != 31) mask |= (1u << r); }; // R31 is hardwired 0, never a pin
454 for (uint32_t i = 0; i < n; ++i) {
455 const uint32_t ins = w[i];
456 const uint32_t op = ins >> 26;
457 const int ra = (ins >> 21) & 0x1f, rb = (ins >> 16) & 0x1f, rc = ins & 0x1f;
458 const bool islit = ((ins >> 12) & 1) != 0;
459 switch (op) {
460 case 0x10: case 0x11: case 0x12: case 0x13: case 0x1c: // integer operate: Ra, Rc, Rb(if reg)
461 touch(ra); touch(rc); if (!islit) touch(rb); break;
462 case 0x08: case 0x09: case 0x0a: case 0x0b: case 0x0c: // LDA/LDAH + BWX ld/st: Ra, Rb(base)
463 case 0x0d: case 0x0e: case 0x0f:
464 case 0x28: case 0x29: case 0x2a: case 0x2b: // int LDL/LDQ/LDx_L
465 case 0x2c: case 0x2d: case 0x2e: case 0x2f: // int STL/STQ/STx_C
466 case 0x1a: // JMP/JSR/RET: Ra(link), Rb(target)
467 touch(ra); touch(rb); break;
468 case 0x30: case 0x34: case 0x38: case 0x39: case 0x3a: // integer branches (incl BR/BSR link): Ra
469 case 0x3b: case 0x3c: case 0x3d: case 0x3e: case 0x3f:
470 touch(ra); break;
471 case 0x18: { // MISC: RPCC/RC/RS write Ra
472 const uint32_t f = ins & 0xffff;
473 if (f == 0xc000 || f == 0xe000 || f == 0xf000) touch(ra);
474 break;
475 }
476 default: break; // FP / HW / CALL_PAL: not the GPR forwarding path
477 }
478 }
479 return mask;
480}

599// Lock the LL/SC bucket for an aligned RAM store.
600// Preserve RDX (address), R10 (bias/base), R11 (DPC slot) and guest pins.
601// Wached stores leave RCX = bucket.
602// Unwatched stores jump to raw_store if supplied, else leave RCX = 0.
603// First LL sets the watch byte after draining active CPUs.
604static void emit_x64_ram_store_lock(asmjit::x86::Assembler& a,
605 const CJitEngine::JitOffsets& off, uintptr_t ll_watch, uintptr_t ll_buckets,
606 const asmjit::Label& slow, bool physical,
607 const asmjit::Label* raw_store = nullptr)
608{
609 using namespace asmjit;
610 const Label watched = a.new_label(), ready = a.new_label();
611 if (physical) a.mov(x86::rax, x86::rdx);
612 else {
613 a.lea(x86::rax, x86::ptr(x86::r10, x86::rdx));
614 a.sub(x86::rax, x86::qword_ptr(x86::rbp, off.dram_ptr));
615 }
616 a.shr(x86::rax, imm(6)); // per-line watch index
617 a.mov(x86::rcx, imm(ll_watch));
618 a.cmp(x86::byte_ptr(x86::rcx, x86::rax), imm(0)); // acquire under x86 TSO
619 if (raw_store) {
620 a.je(*raw_store); // unmonitored line on a monitored page
621 } else {
622 a.jne(watched);
623 a.xor_(x86::ecx, x86::ecx); // no bucket held
624 a.jmp(ready);
625 }
626 a.bind(watched);
627 // CSystem::LLBucket: 64-byte stride, sequence at 0, byte lock at 8.
628 a.and_(x86::eax, imm(65535));
629 a.shl(x86::rax, imm(6)); // bucket stride
630 a.mov(x86::rcx, imm(ll_buckets));
631 a.add(x86::rcx, x86::rax);
632 a.mov(x86::eax, imm(1));
633 a.xchg(x86::byte_ptr(x86::rcx, 8), x86::al); // acquire; XCHG is implicitly locked
634 a.test(x86::al, x86::al);
635 a.jnz(slow);
636 a.bind(ready);
637}

639static void emit_x64_ram_store_unlock(asmjit::x86::Assembler& a,
640 bool bucket_held = false)
641{
642 using namespace asmjit;
643 const Label ready = a.new_label();
644 if (!bucket_held) {
645 a.test(x86::rcx, x86::rcx);
646 a.jz(ready);
647 }
648 // Update the sequence while locked, then release after the RAM write.
649 a.inc(x86::qword_ptr(x86::rcx));
650 a.mov(x86::byte_ptr(x86::rcx, 8), imm(0)); // release under x86 TSO
651 a.bind(ready);
652}

654static void emit_x64_integer_store(asmjit::x86::Assembler& a,
655 int size_bits, int ra, int pin, int slot)
656{
657 using namespace asmjit;
658 const x86::Gp value = pin >= 0 ? x86::gpq((uint32_t)pin) : x86::rax;
659 if (pin < 0) {
660 if (ra == 31) a.xor_(x86::eax, x86::eax);
661 else a.mov(x86::rax, x86::qword_ptr(x86::rbx, slot * 8));
662 }
663 if (size_bits == 64) a.mov(x86::qword_ptr(x86::r10, x86::rdx), value);
664 else if (size_bits == 32) a.mov(x86::dword_ptr(x86::r10, x86::rdx), value.r32());
665 else if (size_bits == 16) a.mov(x86::word_ptr(x86::r10, x86::rdx), value.r16());
666 else a.mov(x86::byte_ptr(x86::r10, x86::rdx), value.r8());
667}

669// Full-probe miss: R10 = expected tag (with alignment bits), R11 = DPC slot,
670// RDX = VA. A watched hit preserves the slot/bias for the next memory op.
671static void emit_watched_store_stub(asmjit::x86::Assembler& a,
672 const CJitEngine::JitOffsets& off, uintptr_t ll_watch, uintptr_t ll_buckets,
673 const ColdMemStub& s)
674{
675 if (!s.watched_store) return;
676 using namespace asmjit;
677 a.bind(s.watched_probe);
678 a.or_(x86::r10, imm(off.dpc_ll_watch_tag));
679 a.cmp(x86::qword_ptr(x86::rbp, x86::r11, 0,
680 off.dpc_write_row + off.dpc_virt_page), x86::r10);
681 a.jne(s.slow); // miss or unaligned store; RAM untouched
682 a.mov(x86::r10, x86::qword_ptr(x86::rbp, x86::r11, 0,
683 off.dpc_write_row + off.dpc_host_bias));
684 if (s.watched_reuse.is_valid()) a.bind(s.watched_reuse);
685 emit_x64_ram_store_lock(a, off, ll_watch, ll_buckets, s.slow, false,
686 &s.raw_store);
687 if (s.kind == ColdMemStub::STORE) {
688 emit_x64_integer_store(a, s.size_bits, s.ra, s.pin, s.slot);
689 } else { // FPMEM is raw STT; FPSTART/alignment checks already passed
690 a.mov(x86::rax, x86::qword_ptr(x86::rbp, off.f_base + s.ra * 8));
691 a.mov(x86::qword_ptr(x86::r10, x86::rdx), x86::rax);
692 }
693 emit_x64_ram_store_unlock(a, true);
694 a.jmp(s.join);
695}

697// Emit one cold stub. Arg marshalling mirrors emit_call: non-immediate sources are placed before
698// immediates so a size/selector immediate can't overwrite RDX
699#ifdef JIT_STATS
700static uint64_t* g_cold_stub_counters = nullptr; // [0] = helper bail, [1] = fault delivered (set per compile_block)
701#endif
702static void emit_cold_mem_stub(asmjit::x86::Assembler& a, const uint8_t* gpa,
703 const CJitEngine::JitOffsets& off, const ColdMemStub& s)
704{
705 using namespace asmjit;
706 auto aq = [&](int k) { return x86::gpq(gpa[k]); };
707 auto ad = [&](int k) { return x86::gpd(gpa[k]); };
708 a.bind(s.slow);
709 // the region-cached GPR's host reg is an ABI arg reg so we spill before using, and read
710 // the store value from regs[] when it IS that reg (placement overwrites it). restore after.
711 if (s.vol_bind >= 0) a.mov(x86::qword_ptr(x86::rbx, s.vol_bind * 8), x86::gpq((uint32_t) s.vol_host));
712 if (s.vol_bind2 >= 0) a.mov(x86::qword_ptr(x86::rbx, s.vol_bind2 * 8), x86::gpq((uint32_t) s.vol_host2));
713 if (s.kind != ColdMemStub::FPMEM)
714 a.mov(x86::qword_ptr(x86::rsp, 48), x86::rdx); // preserve VA for DPC reuse
715 a.mov(x86::r10, imm(s.fault_pc));
716 a.mov(x86::qword_ptr(x86::rbp, off.state_current_pc), x86::r10);
717 const bool val_in_vol = (s.vol_bind >= 0 && s.pin == s.vol_host)
718 || (s.vol_bind2 >= 0 && s.pin == s.vol_host2);
719 a.mov(aq(0), x86::rbp); // cpu
720 if (aq(1).id() != x86::rdx.id()) a.mov(aq(1), x86::rdx); // va (precomputed in RDX)
721 switch (s.kind) {
722 case ColdMemStub::LOAD: // jit_read(cpu, va, descr, &out)
723 a.lea(aq(3), x86::qword_ptr(x86::rsp, 32)); // &out slot
724 a.mov(aq(2), imm(((uint64_t)s.ins << 32) | (uint32_t)s.size_bits));
725 break;
726 case ColdMemStub::STORE: // jit_write(cpu, va, descr, value)
727 if (s.ra == 31) a.xor_(ad(3), ad(3)); // value (R31 == 0)
728 else if (s.pin >= 0 && !val_in_vol) a.mov(aq(3), x86::gpq((uint32_t) s.pin));
729 else a.mov(aq(3), x86::qword_ptr(x86::rbx, s.slot * 8));
730 a.mov(aq(2), imm(((uint64_t)s.ins << 32) | (uint32_t)s.size_bits));
731 break;
732 case ColdMemStub::FPMEM: // jit_fp_read/write(cpu, va, fa, descr)
733 a.mov(ad(2), imm((uint32_t) s.ra));
734 a.mov(aq(3), imm(((uint64_t)s.ins << 32) | s.descr));
735 break;
736 }
737 if (s.hidx >= 0 && off.helpers)
738 a.call(x86::qword_ptr(x86::rbp, (int32_t) (off.helpers + s.hidx * 8))); // via the CPU-resident table
739 else { a.mov(x86::rax, imm((uint64_t) s.helper)); a.call(x86::rax); }
740 if (s.vol_bind >= 0) a.mov(x86::gpq((uint32_t) s.vol_host), x86::qword_ptr(x86::rbx, s.vol_bind * 8));
741 if (s.vol_bind2 >= 0) a.mov(x86::gpq((uint32_t) s.vol_host2), x86::qword_ptr(x86::rbx, s.vol_bind2 * 8));
742 Label ok = a.new_label(), trapped = a.new_label();
743 a.test(x86::eax, x86::eax);
744 a.jz(ok);
745 a.cmp(x86::eax, imm(2));
746 a.je(trapped);
747#ifdef JIT_STATS
748 if (g_cold_stub_counters) { a.mov(x86::rax, imm((uint64_t) &g_cold_stub_counters[0])); a.inc(x86::qword_ptr(x86::rax)); }
749#endif
750 a.mov(x86::r10, imm(s.fault_pc)); // fault: resume at this op
751 a.mov(x86::qword_ptr(x86::rbp, off.state_pc), x86::r10);
752 a.mov(x86::eax, imm(s.i)); // this iteration: i instrs done
753 a.add(x86::eax, x86::r13d); // + earlier chained iterations
754 a.jmp(s.done);
755 a.bind(trapped); // fault delivered: keep PAL entry PC
756#ifdef JIT_STATS
757 if (g_cold_stub_counters) { a.mov(x86::rax, imm((uint64_t) &g_cold_stub_counters[1])); a.inc(x86::qword_ptr(x86::rax)); }
758#endif
759 a.mov(x86::eax, imm(s.i + 1)); // count the faulting instruction once
760 a.add(x86::eax, x86::r13d);
761 a.jmp(s.done);
762 a.bind(ok);
763 if (s.kind != ColdMemStub::FPMEM) {
764 // Recreate the hot path's {VA, slot, host bias} contract before joining the next
765 // instruction. jit_read/jit_write filled the matching DRAM DPC entry on success.
766 a.mov(x86::rdx, x86::qword_ptr(x86::rsp, 48));
767 uint32_t shift = 0;
768 while ((1u << shift) < off.dpc_stride) ++shift;
769 if ((1u << shift) == off.dpc_stride && shift <= 13) {
770 if (host_has_bmi2())
771 a.rorx(x86::r11, x86::rdx, imm(13 - shift));
772 else {
773 a.mov(x86::r11, x86::rdx);
774 if (shift != 13) a.shr(x86::r11, imm(13 - shift));
775 }
776 a.and_(x86::r11, imm((uint64_t)off.dpc_mask << shift));
777 } else {
778 a.mov(x86::r11, x86::rdx);
779 a.shr(x86::r11, imm(13)); a.and_(x86::r11, imm(off.dpc_mask));
780 a.imul(x86::r11, x86::r11, imm(off.dpc_stride));
781 }
782 a.add(x86::r11d, x86::dword_ptr(x86::rbp, (int32_t) off.dpc_bank)); // current mode's bank
783 const int row = (s.kind == ColdMemStub::STORE) ? (int)off.dpc_write_row : 0;
784 a.mov(x86::r10, x86::qword_ptr(x86::rbp, x86::r11, 0, row + (int)off.dpc_host_bias));
785
786 }
787 if (s.kind == ColdMemStub::LOAD) {
788 // Result extraction + dest write must leave RAX = r[ra] to uphold the fast path's
789 // value-forward contract at `join` (the next op may reuse RAX as Ra).
790 if (s.size_bits == 64) a.mov(x86::rax, x86::qword_ptr(x86::rsp, 32));
791 else if (s.size_bits == 32) a.movsxd(x86::rax, x86::dword_ptr(x86::rsp, 32));
792 else if (s.size_bits == 16) a.movzx(x86::eax, x86::word_ptr(x86::rsp, 32));
793 else a.movzx(x86::eax, x86::byte_ptr(x86::rsp, 32));
794 if (s.pin >= 0) a.mov(x86::gpq((uint32_t) s.pin), x86::rax);
795 else a.mov(x86::qword_ptr(x86::rbx, s.slot * 8), x86::rax);
796 }
797 a.jmp(s.join);
798}

1132 auto emit_dpc_offset = [&]() {
1133 uint32_t shift = 0;
1134 while ((1u << shift) < m_off.dpc_stride) ++shift;
1135 if ((1u << shift) == m_off.dpc_stride && shift <= 13) {
1136 // ((va >> 13) & mask) << shift == (va >> (13-shift)) & (mask << shift).
1137 // Doing this with the extraction saves two instructions on every
1138 // inline memory access.
1139 if (host_has_bmi2())
1140 a.rorx(x86::r11, x86::rdx, imm(13 - shift));
1141 else {
1142 a.mov(x86::r11, x86::rdx);
1143 if (shift != 13) a.shr(x86::r11, imm(13 - shift));
1144 }

1757 // CVTQT/CVTQS (0x16): f[Fc] = (double|single)(s64) f[Fb] via SSE. Inline the steady-state
1758 // common case; bail to the interpreter (it owns rounding/traps/FPCR) on any edge that would
1759 // trap or need non-nearest rounding. The verify carries fpcr/exc_sum across its two passes, so
1760 // the inline path leaves them untouched -- correctness rests on the bails, not on FPCR upkeep.
1761 if (op == OP_CVTQT || op == OP_CVTQS) {
1762 const bool dyn = (((ins >> 11) & 3) == 3); // /D: rounding is FPCR<59:58>, checked at runtime
1763 Label bail = a.new_label(), fdone = a.new_label(), cont = a.new_label();

1765 a.je(bail);

1768 a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.fpcr));

1770 a.cmp(x86::eax, imm(2)); a.jne(bail); // FPCR<59:58> != round-to-nearest -> bail
1771 }

1773 else a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * rb));
1774 if (op == OP_CVTQT) { a.cvtsi2sd(x86::xmm0, x86::rax); a.cvttsd2si(x86::rcx, x86::xmm0); }
1775 else { a.cvtsi2ss(x86::xmm0, x86::rax); a.cvttss2si(x86::rcx, x86::xmm0); }

1777 a.je(fdone);
1778 a.bt(x86::qword_ptr(x86::rbp, m_off.fpcr), imm(56)); // inexact: FPCR.INE clear -> first one traps
1779 a.jnc(bail);
1780 a.bind(fdone);
1781 if (op == OP_CVTQS) a.cvtss2sd(x86::xmm0, x86::xmm0); // widen single -> register (T-format) bits
1782 a.movq(x86::qword_ptr(x86::rbp, m_off.f_base + 8u * rc), x86::xmm0);
1783 a.jmp(cont);
1784 a.bind(bail);
1785 set_pc(b->tag + 4 * (uint64_t)i); // resume this instruction in the interpreter
1786 JIT_STAT_EXIT(5); a.mov(x86::eax, imm(i)); a.add(x86::eax, x86::r13d); a.jmp(done);
1787 a.bind(cont);
1788 continue;
1789 }

1791 // IEEE T-float arithmetic (0x16 ADDT/SUBT/MULT/DIVT): inline SSE on the double f-registers.
1792 // FPCR.INE already sticky-set (so an inexact result won't trap). Every edge bails to the interpreter:
1793 // FP-off, /D-not-nearest, denormal operand, INE-clear (a first inexact would trap), or an Inf/NaN/
1794 // denormal result (overflow/invalid/div-zero/underflow).
1795 if (op == OP_ADDT || op == OP_SUBT || op == OP_MULT || op == OP_DIVT) {
1796 const bool dyn = (((ins >> 11) & 3) == 3); // /D: rounding is FPCR<59:58>, checked at runtime
1797 Label bail = a.new_label(), cont = a.new_label();
1798 auto class_bail = [&](const x86::Vec& x, bool chk_inf_nan) { // bail if denormal (always) / Inf|NaN (if asked)
1799 Label ok = a.new_label();
1800 a.movq(x86::rax, x);
1801 a.mov(x86::rcx, x86::rax);
1802 a.shr(x86::rcx, imm(52)); a.and_(x86::ecx, imm(0x7ff)); // biased exponent
1803 if (chk_inf_nan) { a.cmp(x86::ecx, imm(0x7ff)); a.je(bail); }
1804 a.test(x86::ecx, x86::ecx); a.jnz(ok); // exp != 0 -> normal (or Inf/NaN)
1805 a.shl(x86::rax, imm(12)); a.jnz(bail); // exp == 0, mantissa != 0 -> denormal
1806 a.bind(ok);
1807 };
1808 if (!prev_fp_t_guarded) {
1809 a.cmp(x86::byte_ptr(x86::rbp, m_off.fpen), imm(0)); a.je(bail); // FPSTART: FP disabled -> FEN trap
1810 }
1811 if (!prev_fp_t_exc_zero)
1812 a.mov(x86::qword_ptr(x86::rbp, m_off.exc_sum), imm(0));
1813 if (!prev_fp_t_guarded) {
1814 a.bt(x86::qword_ptr(x86::rbp, m_off.fpcr), imm(56)); a.jnc(bail); // INE clear -> first inexact traps
1815 }
1816 if (dyn && !prev_fp_t_nearest) { // dynamic rounding: SSE gives nearest
1817 a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.fpcr));
1818 a.shr(x86::rax, imm(58)); a.and_(x86::eax, imm(3));
1819 a.cmp(x86::eax, imm(2)); a.jne(bail);
1820 }
1821 a.movq(x86::xmm0, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)ra)); // f[Fa]
1822 a.movq(x86::xmm1, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)rb)); // f[Fb]
1823 if (!(prev_fp_t_non_denormal & (uint32_t{1} << ra)))
1824 class_bail(x86::xmm0, false); // denormal operand -> interp (DNZ/trap)
1825 if (!(prev_fp_t_non_denormal & (uint32_t{1} << rb)))
1826 class_bail(x86::xmm1, false);
1827 switch (op) {
1828 case OP_ADDT: a.addsd(x86::xmm0, x86::xmm1); break;
1829 case OP_SUBT: a.subsd(x86::xmm0, x86::xmm1); break;
1830 case OP_MULT: a.mulsd(x86::xmm0, x86::xmm1); break;
1831 default: a.divsd(x86::xmm0, x86::xmm1); break; // OP_DIVT
1832 }
1833 class_bail(x86::xmm0, true); // Inf/NaN/denormal result -> interp
1834 if (op == OP_MULT || op == OP_DIVT) {
1835 // Complete underflow rounds to +-0 with no denormal to catch
1836 // ieee_rpack still sets FPCR.UNF (+ /U //SU trap) so a zero result bails to the
1837 // interp unless it is provably an operand of +-0, or (DIV) a +-Inf divisor
1838 // finite/Inf is an exact signed zero with no exception (ieee_fdiv), committed inline.
1839 // ADD/SUB are exempt entirely
1840 Label nz = a.new_label();
1841 a.movq(x86::rax, x86::xmm0);
1842 a.add(x86::rax, x86::rax); // drop the sign bit
1843 a.jnz(nz);
1844 a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)ra));
1845 a.add(x86::rax, x86::rax);
1846 a.jz(nz); // Fa == +-0 -> exact zero
1847 if (op == OP_MULT) {
1848 a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)rb));
1849 a.add(x86::rax, x86::rax);
1850 a.jz(nz); // Fb == +-0 -> exact zero
1851 } else {
1852 a.mov(x86::rax, x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)rb));
1853 a.add(x86::rax, x86::rax); // +-Inf -> 0xFFE0...0
1854 a.mov(x86::rcx, imm(0xFFE0000000000000ull));
1855 a.cmp(x86::rax, x86::rcx);
1856 a.je(nz); // finite/+-Inf -> exact zero
1857 }
1858 a.jmp(bail); // underflowed to zero
1859 a.bind(nz);
1860 }
1861 a.movq(x86::qword_ptr(x86::rbp, m_off.f_base + 8u * (uint32_t)rc), x86::xmm0); // f[Fc]
1862 a.jmp(cont);
1863 a.bind(bail);
1864 set_pc(b->tag + 4 * (uint64_t)i);
1865 JIT_STAT_EXIT(5); a.mov(x86::eax, imm(i)); a.add(x86::eax, x86::r13d); a.jmp(done);
1866 a.bind(cont);
1867 // Only successful execution reaches the following instruction. Result
1868 // and complete-underflow checks above remain mandatory for every op.
1869 regalloc.fp_t_guarded = true;
1870 regalloc.fp_t_exc_zero = true;
1871 regalloc.fp_t_nearest = prev_fp_t_nearest || dyn;
1872 regalloc.fp_t_non_denormal = prev_fp_t_non_denormal | (uint32_t{1} << rc);
1873 continue;
1874 }

1876 // IEEE T-float compares (0x16 CMPTUN/EQ/LT/LE): f[Fc] = (Fa cmp Fb) ? 2.0 : 0.0. No rounding or
1877 // inexact. A NaN operand (INV / quiet-NaN / SWC rules) or a denormal operand (unmaskable denormal
1878 // trap -- the interp's ieee_fcmp traps via ieee_unpack) goes to the interp; +/-Inf, zero and
1879 // normals compare inline via ucomisd. Two ordered operands make CMPTUN always false.
1880 if (op == OP_CMPTUN || op == OP_CMPTEQ || op == OP_CMPTLT || op == OP_CMPTLE) {
1881 Label bail = a.new_label(), cont = a.new_label();
1882 auto cmp_bail = [&](const x86::Vec& x) { // bail on NaN or denormal; Inf/zero/normal -> ok
1883 Label ok = a.new_label(), special = a.new_label();
1884 a.movq(x86::rax, x);
1885 a.mov(x86::rcx, x86::rax);
1886 a.shr(x86::rcx, imm(52)); a.and_(x86::ecx, imm(0x7ff)); // biased exponent
1887 a.cmp(x86::ecx, imm(0x7ff)); a.je(special); // exp all-ones -> Inf or NaN
1888 a.test(x86::ecx, x86::ecx); a.jnz(ok); // exp != 0 -> normal
1889 a.bind(special); // exp == 0 (zero/denormal) or all-ones (Inf/NaN)
1890 a.shl(x86::rax, imm(12)); a.jnz(bail); // mantissa != 0 -> denormal or NaN -> bail
1891 a.bind(ok); // mantissa == 0 -> zero or Inf -> compare inline
1892 };

2622 case OP_AMASK: // Rc = op2 & ~selected feature mask
2623 op2_rcx(); // classify enforced Ra==31 (the Ra!=31 form traps OPCDEC in the interpreter)
2624 a.mov(x86::rax, imm(~m_profile.amask));
2625 a.and_(x86::rax, x86::rcx);
2626 break;
2627 case OP_IMPLVER: // Rc = selected implementation version
2628 a.mov(x86::eax, imm(m_profile.implver));
2629 break;
2630
2631 case OP_CMPEQ: emit_compare(x86::Inst::kIdSete); break;
2632 case OP_CMPLT: emit_compare(x86::Inst::kIdSetl); break;
2633 case OP_CMPLE: emit_compare(x86::Inst::kIdSetle); break;
2634 case OP_CMPULT: emit_compare(x86::Inst::kIdSetb); break;
2635 case OP_CMPULE: emit_compare(x86::Inst::kIdSetbe); break;
2636
2637 case OP_CMPBGE: // 8 parallel unsigned byte compares (Ra.byte[i] >= op2.byte[i]) -> bit i; bits 63:8 = 0
2638 op1_rax(); op2_rcx();
2639 a.movq(x86::xmm0, x86::rax); // Ra (8 bytes)
2640 a.movq(x86::xmm1, x86::rcx); // op2 (8 bytes; literal in byte 0, 0 elsewhere)
2641 a.movdqa(x86::xmm2, x86::xmm0); // copy of Ra
2642 a.pmaxub(x86::xmm2, x86::xmm1); // per-byte unsigned max(Ra, op2)
2643 a.pcmpeqb(x86::xmm2, x86::xmm0); // 0xff where max == Ra, i.e. Ra >= op2
2644 a.pmovmskb(x86::eax, x86::xmm2); // gather byte-sign bits -> low 8 bits = the mask
2645 a.movzx(x86::eax, x86::al); // discard the high (zero-padded) lane bits
2646 break;
2647
2648 case OP_ADDL: case OP_SUBL: // 32-bit, result sign-extended to 64
2649 case OP_S4ADDL: case OP_S8ADDL: case OP_S4SUBL: case OP_S8SUBL: // scaled longword: sext32((Ra*scale) +/- Rb)
2650 {
2651 const bool issub = (op == OP_SUBL || op == OP_S4SUBL || op == OP_S8SUBL);
2652 const int sh = (op == OP_S4ADDL || op == OP_S4SUBL) ? 2 // Ra*4
2653 : (op == OP_S8ADDL || op == OP_S8SUBL) ? 3 // Ra*8
2654 : 0;
2655 {
2656 const int dp = pin_id(rc);
2657 const bool op2_is_rc = !islit && rb != 31 && rb == rc;
2658 if (rc != 31 && dp >= 0 && sh == 0 && !op2_is_rc) {
2659 const x86::Gp dd = x86::gpd((uint32_t) dp);
2660 if (ra == 31) a.xor_(dd, dd);
2661 else if (pin_id(ra) != dp) mov_from_reg32(dd, ra);
2662 if (islit) { if (issub) a.sub(dd, imm(lit)); else a.add(dd, imm(lit)); }
2663 else if (rb != 31) {
2664 const int bp = pin_id(rb);
2665 const Operand o2 = (bp >= 0) ? Operand(x86::gpd((uint32_t) bp)) : Operand(reg32(rb));
2666 a.emit(issub ? x86::Inst::kIdSub : x86::Inst::kIdAdd, dd, o2);
2667 }
2668 a.movsxd(x86::gpq((uint32_t) dp), dd);
2669 result_in_dest = true;
2670 break;
2671 }
2672 }
2673 if (ra == 31) a.xor_(x86::eax, x86::eax);
2674 else mov_from_reg32(x86::eax, ra); // shadow-remapped (was a raw rbx read)
2675 if (sh) a.shl(x86::eax, imm(sh)); // scale in 32-bit: (Ra<<sh)[31:0] == ((RAV<<sh)+..)[31:0]
2676 if (islit) {
2677 if (issub) a.sub(x86::eax, imm(lit));
2678 else a.add(x86::eax, imm(lit));
2679 } else if (rb != 31) {
2680 mov_from_reg32(x86::ecx, rb);
2681 if (issub) a.sub(x86::eax, x86::ecx);
2682 else a.add(x86::eax, x86::ecx);
2683 }
2684 a.movsxd(x86::rax, x86::eax);
2685 break;
2686 }
2687 default: break;
2688 }
2689
2690 if (rc != 31 && !result_in_dest) mov_to_reg(rc, x86::rax);
2691 // These integer emitters leave the cached VA, page slot and host bias in
2692 // RDX/R11/R10 intact. Keep them available to the next memory operation
2693 // unless this instruction changed its address base.
2694 //
2695 // That next operation still checks the page tag and alignment,
2696 // including after a slow helper.
2697 //
2698 // UMULH uses RDX and CTTZ uses R10, so neither belongs in this list.
2699 if (rc != prev_dpc_base) {
2700 switch (op) {
2701 case OP_ADDL: case OP_SUBL: case OP_ADDQ: case OP_SUBQ:
2702 case OP_S4ADDL: case OP_S8ADDL: case OP_S4SUBL: case OP_S8SUBL:
2703 case OP_S4ADDQ: case OP_S8ADDQ: case OP_S4SUBQ: case OP_S8SUBQ:
2704 case OP_AND: case OP_BIS: case OP_XOR: case OP_BIC: case OP_ORNOT: case OP_EQV:
2705 case OP_MULQ: case OP_MULL:
2706 case OP_SLL: case OP_SRL: case OP_SRA: case OP_SEXTB: case OP_SEXTW:
2707 case OP_CTPOP: case OP_CTLZ: case OP_AMASK: case OP_IMPLVER:
2708 case OP_CMPEQ: case OP_CMPLT: case OP_CMPLE: case OP_CMPULT: case OP_CMPULE:
2709 case OP_CMPBGE:
2710 preserve_dpc();
2711 break;
2712 default: break;
2713 }
2714 }
2715 } while (0);
2716}

2718// Shared entry: set up the frame, load the pins, jump to the block body. Shared exit: sync the pins
2719// back to regs[], tear the frame down, return EAX (instructions completed). Rebuilt with the runtime.
2720void CJitEngine::build_trampolines()
2721{
2722 using namespace asmjit;
2723 JitRuntime* rt = (JitRuntime*) m_rt;
2724 CallConv cc;
2725 if (cc.init(CallConvId::kCDecl, rt->environment()) != Error::kOk) return;
2726 const uint8_t* gpa = cc.passed_order(RegGroup::kGp);
2727 auto aq = [&](int i) { return x86::gpq(gpa[i]); };
2728 {
2729 CodeHolder code; code.init(rt->environment());
2730 x86::Assembler a(&code);
2731 a.push(x86::rbx);
2732 a.push(x86::rbp);
2733 a.push(x86::r14); // callee-saved pin for Alpha R30 (SP)
2734 a.push(x86::r12); // callee-saved guest-register pins
2735 a.push(x86::r13); // chain instruction count
2736 a.push(x86::r15);
2737#ifdef _WIN32
2738 a.push(x86::rsi); // callee-saved on Win64: pin for Alpha R27 (PV)
2739 a.push(x86::rdi); // callee-saved on Win64: pin for Alpha R0 (v0)
2740#endif
2741 a.sub(x86::rsp, imm(56)); // 32 shadow + helper scratch/out slots; 6/8 pushes -> 16-byte alignment
2742 a.mov(x86::rax, aq(2)); // body (arg 2), before the pins claim r8/r9
2743 a.mov(x86::rbp, aq(0)); // cpu (arg 0)
2744 a.mov(x86::rbx, aq(1)); // regs (arg 1)
2745 a.xor_(x86::r13d, x86::r13d); // chain instruction count := 0
2746 a.mov(x86::r12, x86::qword_ptr(x86::rbx, 1 * 8)); // R1
2747 a.mov(x86::r15, x86::qword_ptr(x86::rbx, 16 * 8)); // R16 (a0)
2748 a.mov(x86::r14, x86::qword_ptr(x86::rbx, 30 * 8)); // R30 (SP)
2749 a.mov(x86::r8, x86::qword_ptr(x86::rbx, kVolPins[0] * 8)); // R22, caller-saved global pin
2750 a.mov(x86::r9, x86::qword_ptr(x86::rbx, kVolPins[1] * 8)); // R23, caller-saved global pin
2751#ifdef _WIN32
2752 a.mov(x86::rsi, x86::qword_ptr(x86::rbx, 27 * 8)); // R27 (PV)
2753 a.mov(x86::rdi, x86::qword_ptr(x86::rbx, 0 * 8)); // R0 (v0)
2754#endif
2755 a.jmp(x86::rax);
2756 JitEnterFn fn = nullptr;
2757 if (rt->add(&fn, &code) != Error::kOk) { fprintf(stderr, "[JIT][CPU%d] FATAL: entry trampoline\n", m_cpu_id); abort(); }
2758 m_enter = fn;
2759 }
2760 {
2761 CodeHolder code; code.init(rt->environment());
2762 x86::Assembler a(&code);
2763 a.mov(x86::qword_ptr(x86::rbx, 1 * 8), x86::r12); // R1
2764 a.mov(x86::qword_ptr(x86::rbx, 16 * 8), x86::r15); // R16 (a0)
2765 a.mov(x86::qword_ptr(x86::rbx, 30 * 8), x86::r14); // R30 (SP)
2766 a.mov(x86::qword_ptr(x86::rbx, kVolPins[0] * 8), x86::r8); // caller-saved global pins
2767 a.mov(x86::qword_ptr(x86::rbx, kVolPins[1] * 8), x86::r9);
2768#ifdef _WIN32
2769 a.mov(x86::qword_ptr(x86::rbx, 27 * 8), x86::rsi); // R27 (PV)
2770 a.mov(x86::qword_ptr(x86::rbx, 0 * 8), x86::rdi); // R0 (v0)
2771#endif
2772 a.add(x86::rsp, imm(56));
2773#ifdef _WIN32
2774 a.pop(x86::rdi);
2775 a.pop(x86::rsi);
2776#endif
2777 a.pop(x86::r15);
2778 a.pop(x86::r13);
2779 a.pop(x86::r12);
2780 a.pop(x86::r14);
2781 a.pop(x86::rbp);
2782 a.pop(x86::rbx);
2783 a.ret();
2784 void* fn = nullptr;
2785 if (rt->add(&fn, &code) != Error::kOk) { fprintf(stderr, "[JIT][CPU%d] FATAL: exit trampoline\n", m_cpu_id); abort(); }
2786 m_exit = fn;
2787 if (m_exit_slot) *m_exit_slot = m_exit;
2788 }
2789}

2791void CJitEngine::compile_block(JitBlock* b, const AlphaInstructionSource& sources, void* read_helper, void* write_helper, void* opcdec_helper, void* hw_mfpr_helper, void* hw_ld_helper, void* hw_mtpr_helper, void* hw_st_helper, void* indirect_helper, void* read_locked_helper, void* stc_helper, void* misc_helper, void* read_vpte_helper, void* read_wchk_helper, void* itof_helper, void* ftoi_helper, void* fltl_helper, void* fp_read_helper, void* fp_write_helper, void* fltv_helper)
2792{
2793 // Only the interpreted miss path calls this for an uncompiled block. Keep the
2794 // first encounter interpreted; compiled dispatch and chaining pay no counter cost.
2795 if (b->compile_encounters < 2) ++b->compile_encounters;
2796 if (b->compile_encounters < 2) return;
2797
2798 using namespace asmjit;
2799 // Reclaim must self-trigger here, NOT only in flush(): flush() runs when the guest executes
2800 // IMB/IC_FLUSH, and a compute-heavy phase can go minutes without one while recompiles keep
2801 // allocating -- code memory grew unbounded (multi-GB). Safe: we're in this CPU's cold path.
2802 if (m_rt && m_code_bytes >= kReclaimBytes) {
2803 reclaim_code();
2804 b->valid = true; // b was just (re)validated by record(); restore it after the wipe
2805 }
2806 b->compiled = true;
2807 b->jit_body_same_page = nullptr;
2808
2809 if (b->n_instr == 0) return;
2810 const auto source = sources.view(b->tag);
2811 if (!source.contains(1) || source.identity != b->phys
2812 || source.asm_global != b->asm_global) {
2813 b->compiled = false;
2814 return;
2815 }
2816
2817 // PALmode blocks (PC bit 0) use the register-bank mapping captured in b->pal_shadow
2818 const bool pal_block = (b->tag & 1) != 0;
2819
2820 // The family bounds a contiguous instruction source. Copy it before another
2821 // query can invalidate the borrowed view, including optional fusion lookahead.
2822 const uint32_t scan_len = (std::min)((std::min)(b->n_instr, uint32_t{64}),
2823 source.word_count);
2824 const uint32_t snapshot_len = (std::min)(source.word_count,
2825 (std::max)(scan_len, uint32_t{39}));
2826 std::vector<uint32_t> source_words(snapshot_len);
2827 if (!source.copy_words(source_words.data(), snapshot_len)) {
2828 b->compiled = false;
2829 return;
2830 }
2831 const uint32_t* const words = source_words.data();
2832 uint32_t plen = 0;
2833 bool terminator_branch = false; // last instruction is a compiled terminator (sets its own PC)
2834 bool terminator_jmp = false; // ...and it's a computed jump (don't chain: targets vary)
2835 while (plen < scan_len)
2836 {
2837 SafeOp sop = classify(words[plen], pal_block, m_profile.ipr_read, m_profile.ipr_write,
2838 m_profile.pal, m_profile.hw_load, m_profile.hw_store);
2839 if (sop == OP_NONE) { // uncompilable op ends the straight-line prefix
2840#ifdef JIT_STATS
2841 const uint32_t bop = words[plen] >> 26;
2842 m_term_op[bop]++; // tally what cut this block (the coverage gap to chase)
2843 if (bop == 0x00) // CALL_PAL: also tally the function code (low 8 bits)
2844 m_pal_func[words[plen] & 0xFF]++;
2845 else if (bop == 0x1d) // HW_MTPR: diagnostic raw instruction[15:8], not a family selector.
2846 m_mtpr_encoding_hi[(words[plen] >> 8) & 0xFF]++;
2847 else if (bop == 0x1b) // HW_LD: tally the form (phys/virt/lock/vpte/chk, ins[15:12])
2848 m_hwld_func[(words[plen] >> 12) & 0xF]++;
2849 else if (bop == 0x18) // MISC: tally the Ra==31 form (ins[15:12]: 0xc RPCC / 0xe RC / 0xf RS)
2850 m_misc_func[(words[plen] >> 12) & 0xF]++;
2851 // Punch list: one-shot print of the first ACTIONABLE breaker -- skip the opcodes whose
2852 // compilable subset is already settled, so it points at the next NEW target rather than a
2853 // decided one: 0x00 CALL_PAL (terminator), 0x1b HW_LD / 0x1f HW_ST (physical done;
2854 // conditional/virtual forms side-effecting), 0x1d HW_MTPR (pure-store IPRs done; rest
2855 // side-effecting), 0x10 INTA + 0x13 INTM (non-trapping ops done; only /V overflow-trap variants left),
2856 // 0x18 MISC (barriers/hints + RPCC/RC/RS via log/replay done; only the rare Ra==31 RC/RS forms interpret),
2857 // 0x14 ITFP (ITOF* moves done; SQRT* = the deferred FP-math class: FPCR/rounding/traps),
2858 // 0x17 FLTL (non-arithmetic done; only CVTQL/V trap variants left).
2859 // JMP (0x1a) + HW_RET (0x1e) are now compiled+chained. Stats count all.
2860 if (!m_first_breaker_logged && bop != 0x00 && bop != 0x1b && bop != 0x1d && bop != 0x1f && bop != 0x10 && bop != 0x18 && bop != 0x13 && bop != 0x14 && bop != 0x17) {
2861 m_first_breaker_logged = true;
2862 printf("[JIT][PUNCH][CPU%d] first unhandled breaker: %s(0x%02x) ins=%08x at pc=%016llx%s\n",
2863 m_cpu_id, jit_opcode_name(bop), bop, words[plen],
2864 (unsigned long long) ((b->tag & ~(uint64_t) 1) + (uint64_t) plen * 4),
2865 pal_block ? " [PALmode]" : "");
2866 }
2867#endif
2868 break;
2869 }
2870 plen++;
2871 if (is_terminator(sop)) { // branch or computed jump ends the block
2872 terminator_branch = true;
2873 if (sop == OP_JMP || sop == OP_HW_RET) terminator_jmp = true;
2874 break;
2875 }
2876 }
2877
2878 if (plen == 0) return;
2879
2880 // counted-substring outer loop.
2881 // Most candidates fail on byte zero. If we're lucky. That's a good outcome.
2882 // Recognize the instruction sequence and scan those first bytes 16 at a time.
2883 // Potential 22 guest instruction shortcut.
2884 bool counted_scan_fused = false;
2885 // Full-word portion of OpenVMS byte-search routine.
2886 bool word_scan_fused = false;
2887 bool byte_scan_fused = false;
2888 bool tail_scan_fused = false;
2889#if defined(_WIN32) && !defined(JIT_VERIFY)
2890 if (!pal_block && plen == 6 && snapshot_len >= 39) {
2891 static const uint32_t head[15] = {
2892 0x40020536, 0x47e0341c, 0x400203b7, 0x46df04dc, 0x4797041c, 0xe3800022,
2893 0x2ffe0000, 0x2ffe0000, 0x2ae10000, 0x2b830000, 0x40003120, 0x20630001,
2894 0x20210001, 0x42fc0536, 0xf6c00011
2895 };
2896 static const uint32_t mismatch[7] = {
2897 0x40c03126, 0x47e40400, 0x47e50401, 0x20e70001,
2898 0x47e60402, 0x47e70403, 0xc3ffffd9
2899 };
2900 counted_scan_fused = memcmp(words, head, sizeof(head)) == 0
2901 && memcmp(words + 32, mismatch, sizeof(mismatch)) == 0;
2902 }
2903 if (!pal_block && plen == 5 && snapshot_len >= 10) {
2904 static const uint32_t word_scan[10] = {
2905 0xa6e10000, 0x42fc01f6, 0x439701f8, 0x46d80016, 0xf6c00028,
2906 0x40011120, 0x40211401, 0xf81ffff8, 0x40011400, 0xe4000007
2907 };
2908 word_scan_fused = memcmp(words, word_scan, sizeof(word_scan)) == 0;
2909 }
2910 // Inner byte-search loop.
2911 // R23 contains the aligned source word
2912 // Compare from R1&7 until R27 matches, R0 reaches zero, or R1 reaches the next word.
2913 if (!pal_block && plen == 3 && snapshot_len >= 8) {
2914 static const uint32_t byte_scan[8] = {
2915 0x4ae100d8, 0x43780536, 0xe6c0001e,
2916 0x40003520, 0x40203401, 0xe400001b,
2917 0x4420f016, 0xf6dffff8
2918 };
2919 byte_scan_fused = memcmp(words, byte_scan, sizeof(byte_scan)) == 0;
2920 }
2921 // End of the inner loop search.
2922 // R23 remains the loaded word and R0 is the remaining byte count.
2923 // Each mismatch executes the 3 op update block.
2924 if (!pal_block && plen == 3 && snapshot_len >= 6) {
2925 static const uint32_t tail_scan[6] = {
2926 0x4ae100d8, 0x431b0538, 0xe7000003,
2927 0x40003120, 0x40203401, 0xf41ffffa
2928 };
2929 tail_scan_fused = memcmp(words, tail_scan, sizeof(tail_scan)) == 0;
2930 }
2931#endif
2932 const uint32_t hash_len = counted_scan_fused ? 39u : byte_scan_fused ? 8u
2933 : tail_scan_fused ? 6u : word_scan_fused ? 10u : plen;
2934 assert(hash_len <= source_words.size());
2935 const uint64_t source_hash = src_hash(
2936 reinterpret_cast<const uint8_t*>(source_words.data()), hash_len);
2937#ifdef JIT_REGPROF
2938 b->rp_mask = regprof_mask(words, plen); // GPR-access fingerprint; exec-weighted at report time
2939#endif
2940
2941 // Emit uint32_t fn(CAlphaCPU* cpu, uint64_t* regs) (Win64: cpu=RCX, regs=RDX).
2942 // Keep cpu in RBP and regs in RBX (callee-saved, so they survive helper calls);
2943 // reserve a 40-byte frame (32 shadow + 8 load-out slot) that keeps RSP 16-aligned
2944 // for calls. RAX = op1/result, RCX = operand2 (CL for variable shifts).
2945 JitErrorHandler eh;
2946 eh.cpu_id = m_cpu_id;
2947#ifdef JIT_DISASM
2948 eh.fp = m_disasm_fp;
2949 StringLogger logger;
2950#endif
2951 CodeHolder code;
2952 if (code.init(((JitRuntime*) m_rt)->environment()) != Error::kOk) {
2953 m_reclaim_pending = true; // init only fails on OOM here: free code at the next
2954 b->compiled = false; // safe point and retry, don't latch the block interpreted
2955 return;
2956 }
2957 code.set_error_handler(&eh);
2958#ifdef JIT_DISASM
2959 // Dev: capture this block's disassembly, validate each emitted instruction, and trap any
2960 // emit failure (dumped + bailed near rt->add() below). Logging formats every instruction.
2961 code.set_logger(&logger);
2962#endif
2963 x86::Assembler a(&code);
2964#ifdef JIT_DISASM
2965 a.add_diagnostic_options(DiagnosticOptions::kValidateAssembler);
2966#endif
2967
2968 // Host integer-argument registers:
2969 // Win64 {rcx,rdx,r8,r9}; System V {rdi,rsi,rdx,rcx}. aq(i)/ad(i) = the i argument as a
2970 // 64-/32-bit register; emit_call (below) marshals into them, replacing out usage of
2971 // #ifdef _WIN32. Helpers limited to <= 4 integer arguments.
2972 CallConv cc;
2973 if (cc.init(CallConvId::kCDecl, ((JitRuntime*) m_rt)->environment()) != Error::kOk) return;
2974 const uint8_t* gpa = cc.passed_order(RegGroup::kGp);
2975 auto aq = [&](int i) { return x86::gpq(gpa[i]); };
2976 auto ad = [&](int i) { return x86::gpd(gpa[i]); };
2977 // cpu (RBP), regs (RBX), chain counter (R13), and callee-saved guest pins
2978 // all hold live values across helper calls, so they must be callee-saved under the host ABI.
2979 // Verified at dev time (compiled out in NDEBUG builds).
2980 [[maybe_unused]] const uint32_t kPinnedGp =
2981 (1u << x86::rbp.id()) | (1u << x86::rbx.id()) | (1u << x86::r14.id())
2982 | (1u << x86::r12.id()) | (1u << x86::r13.id()) | (1u << x86::r15.id())
2983#ifdef _WIN32
2984 | (1u << x86::rsi.id()) | (1u << x86::rdi.id())
2985#endif
2986 ;
2987 assert((((uint32_t) cc.preserved_regs(RegGroup::kGp)) & kPinnedGp) == kPinnedGp);
2988
2989 // The frame (pushes, 56-byte scratch, pins in r12/r15/r14/r8/r9[/rsi/rdi]) is set up by the
2990 // engine's shared entry trampoline (build_trampolines); every exit jumps to the shared epilogue.
2991 Label done = a.new_label(); // shared exit: `jmp [cpu + jit_exit]` (EAX preset by caller)
2992 Label body = a.new_label(); // chained re-entry
2993 const Label body_same_page = a.new_label();
2994 const Label exec_proof = a.new_label(), exec_denied = a.new_label();
2995 size_t exec_mask_imm_off = 0;
2996 a.bind(body);
2997 const size_t body_off = code.code_size(); // byte offset of the chained entry from fn
2998 if (!pal_block) {
2999 // Check instruction access for CM. Static same-page branches use body_same_page;
3000 // their source checked the same 8 KiB page. Authorization patches the imm8.
3001 // jit_mode_bit is refreshed on C entry and compiled CM writes.
3002 if (a.test(x86::byte_ptr(x86::rbp, m_off.jit_mode_bit), imm(0)) != Error::kOk)
3003 return;
3004 exec_mask_imm_off = a.offset() - 1; // TEST r/m8, imm8 ends with its one-byte immediate
3005 a.jz(exec_denied);
3006 const bool had_optimized_align = a.has_encoding_option(EncodingOptions::kOptimizedAlign);
3007 a.add_encoding_options(EncodingOptions::kOptimizedAlign);
3008 const Error align_err = a.align(AlignMode::kCode, 16);
3009 if (!had_optimized_align)
3010 a.clear_encoding_options(EncodingOptions::kOptimizedAlign);
3011 if (align_err != Error::kOk)
3012 return;
3013 }
3014 a.bind(body_same_page);
3015 const size_t body_same_page_off = code.code_size();
3016#ifdef JIT_REGPROF
3017 a.mov(x86::rax, imm((uint64_t) &b->rp_hits)); // REGPROF: count every execution (cold entry + chained re-entry)
3018 a.inc(x86::qword_ptr(x86::rax)); // RAX is dead at body entry -- the first op reloads it
3019#endif
3020

3757// is the exact one the block path already verifies). Blocks are fused with a GUARD between them: after a
3758// block's terminator (which left R10 = its next PC), check R10 == the next fused block's tag; on a hit fall
3759// through in-trace, on a miss SIDE-EXIT to the dispatcher at the real next PC. n_blocks==1 = the single-
3760// block trace. Fills the caller-provided slot t (code + per-segment source descriptors + coherence epoch).
