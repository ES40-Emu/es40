107// Configuration latches used by the RV200 ROM. These set board-level drive
108// strengths, DRAM timings, bus timeouts and the empty TV output's clocks.
109// The synchronous model does not simulate analog signalling or clock cycles.
110static bool configuration_latch(uint32_t a)
111{
112 switch (a)
113 {
114 case CRV200::BUS_CNTL1:
115 case CRV200::RBBM_CNTL:
116 case CRV200::TEST_DEBUG_MUX:
117 case CRV200::MEM_CNTL:
118 case CRV200::MEM_TIMING_CNTL:
119 case CRV200::MC_AGP_LOCATION:
120 case CRV200::MEM_INIT_LAT_TIMER:
121 case CRV200::PAD_CTLR_STRENGTH:
122 case CRV200::AGP_CNTL:
123 case CRV200::MEM_IO_CNTL_A0:
124 case CRV200::MEM_IO_CNTL_A1:
125 case CRV200::MC_DEBUG:
126 case CRV200::MEM_IO_OE_CNTL:
127 case CRV200::TMDS_PLL_CNTL:
128 case CRV200::DAC_MACRO_CNTL:
129 case CRV200::TV_MASTER_CNTL:
130 return true;
131 default:
132 return false;
133 }
134}

136// RV200 uses DAC_CNTL2 for CRTC selection. DISP_OUTPUT_CNTL selects the
137// subsequent merger/RMX/transform path, unlike R200/R300's CRTC mux here.
138// The hit RMX path is a bypass while both stretch enables are clear.
139static bool dac_route_supported(uint32_t output, uint32_t horz, uint32_t vert)
140{
141 const uint32_t source = output & 3u;
142 return source == 0 ||
143 (source == 2 && !(output & (1u << 8)) &&
144 !((horz | vert) & (1u << 25)));
145}

147// Only forced-level load detection has reached the TV DAC. Inactive guns
148// have no input to evaluate; each active gun must select FORCE_DAC_DATA.
149static bool tv_dac_forced(uint32_t mux)
150{
151 for (unsigned channel = 0; channel < 3; ++channel)
152 if ((mux & (1u << channel)) &&
153 ((mux >> (4 + 4 * channel)) & 7u) != 6)
154 return false;
155 return true;
156}

165static bool source_rejected(
166 uint32_t control, uint32_t key, uint32_t mask, uint32_t source,
167 uint32_t bytes)
168{
169 if (!control)
170 return false;
171 if (bytes < 4)
172 mask &= (1u << (8 * bytes)) - 1;
173 const bool equal = ((source ^ key) & mask) == 0;
174 return (control & 7) == 4 ? equal : !equal;
175}

177// Select the bit order within each byte; byte order itself is unchanged.
178static bool mono_bit(uint32_t word, unsigned pixel, bool lsb)
179{
180 const unsigned bit =
181 (pixel & ~7u) + (lsb ? (pixel & 7u) : 7u - (pixel & 7u));
182 return ((word >> bit) & 1u) != 0;
183}

195uint32_t CRV200::ReadMem(uint32_t a, int dsize)
196{
197 m_counters.reads++;
198 run_cp();
199 const unsigned bits = unsigned(dsize);
200 if (!valid_access(a, bits, RegisterSpaceSize))
201 {
202 fail("invalid MMIO read " + hex(a));
203 return 0xffffffffu;
204 }
205 if (a >= 0x3b0 && a < 0x3e0)
206 {
207 if (!m_vga_read)
208 {
209 fail("VGA register frontend is not connected");
210 return field_mask(bits);
211 }
212 uint32_t value = 0;
213 for (unsigned i = 0; i < bits / 8; ++i)
214 value |= uint32_t(m_vga_read(a + i)) << (8 * i);
215 return value;
216 }
217 if ((a & ~3u) == MM_DATA)
218 {
219 const uint32_t idx = r(MM_INDEX);
220 uint32_t target;
221 if (idx & 0x80000000u)
222 return uint32_t(mem_read((idx & 0x7ffffffcu) + (a & 3u), dsize));
223 return indexed_target(a & 3u, target) ? ReadMem(target, dsize) : 0;
224 }
225 // Side-effect ports cannot be split implicitly into multiple reads.
226 const uint32_t base = alias(a & ~3u), shift = (a & 3u) * 8;
227 if (base == PALETTE_DATA && bits != 32)
228 {
229 fail("palette data requires DWORD access");
230 return 0;
231 }
232 return (read32(base) >> shift) & field_mask(bits);
233}

235void CRV200::WriteMem(uint32_t a, int dsize, uint32_t value)
236{
237 m_counters.writes++;
238 const unsigned bits = unsigned(dsize);
239 if (!valid_access(a, bits, RegisterSpaceSize))
240 {
241 fail("invalid MMIO write " + hex(a));
242 return;
243 }
244 if (a >= 0x3b0 && a < 0x3e0)
245 {
246 if (!m_vga_write)
247 {
248 fail("VGA register frontend is not connected");
249 return;
250 }
251 for (unsigned i = 0; i < bits / 8; ++i)
252 m_vga_write(a + i, uint8_t(value >> (8 * i)));
253 return;
254 }
255 if ((a & ~3u) == MM_DATA)
256 {
257 const uint32_t idx = r(MM_INDEX);
258 uint32_t target;
259 if (idx & 0x80000000u)
260 mem_write((idx & 0x7ffffffcu) + (a & 3u), dsize, value);
261 else if (indexed_target(a & 3u, target))
262 WriteMem(target, dsize, value);
263 return;
264 }
265 const uint32_t base = alias(a & ~3u), shift = (a & 3u) * 8,
266 mask = field_mask(bits) << shift;
267 if ((base == PALETTE_DATA || base == DST_HEIGHT_WIDTH ||
268 base == DST_WIDTH_HEIGHT || base == DST_LINE_END ||
269 (base >= HOST_DATA0 && base <= HOST_DATA_LAST)) &&
270 bits != 32)
271 {
272 fail("command/data port requires DWORD access at " + hex(a));
273 return;
274 }
275 write32(base, (value << shift) & mask, mask);
276 run_cp();
277}

279// MM_DATA is a window onto the register MM_INDEX selects; its byte lanes
280// address the matching bytes of that register, as the x86 BIOS uses them.
281bool CRV200::indexed_target(uint32_t lane, uint32_t& target)
282{
283 target = (r(MM_INDEX) & ~3u) + lane;
284 if (target >= RegisterSpaceSize)
285 {
286 fail("MM_DATA register index outside MMIO aperture");
287 return false;
288 }
289 if (target < 8)
290 {
291 fail("recursive MM_DATA index");
292 return false;
293 }
294 return true;
295}

297uint32_t CRV200::read32(uint32_t a)
298{
299 if (a == MC_IND_DATA)
300 return m_mc[r(MC_IND_INDEX) & 31];
301 if (a == CLOCK_CNTL_DATA)
302 {
303 const uint32_t i = r(CLOCK_CNTL_INDEX) & 63, v = m_pll[i];
304 // PLL_TEST_CNTL[31:24] counts 27 MHz reference clocks; the x86 BIOS
305 // times its delays with it. Each read observes one more microsecond.
306 if (i == PLL_TEST_CNTL)
307 m_pll[i] += 27u << 24;
308 return v;
309 }
310 if (a == PALETTE_DATA)
311 {
312 const uint32_t idx = (r(PALETTE_INDEX) >> 16) & 255;
313 const uint32_t val = palette_color(idx, palette_bank());
314 r(PALETTE_INDEX) =
315 (r(PALETTE_INDEX) & ~0xff0000u) | (((idx + 1) & 255) << 16);
316 return val;
317 }
318 if (a == SRC_PITCH || a == DST_PITCH)
319 return (r(a == SRC_PITCH ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET) >> 16) &
320 0x3fc0u;
321 if (a == SRC_OFFSET || a == DST_OFFSET)
322 return (r(a == SRC_OFFSET ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET) &
323 0x3fffffu)
324 << 10;
325 if (a == SRC_X || a == DST_X)
326 return r(a == SRC_X ? SRC_Y_X : DST_Y_X) & 0xffffu;
327 if (a == SRC_Y || a == DST_Y)
328 return r(a == SRC_Y ? SRC_Y_X : DST_Y_X) >> 16;
329 if (a == DP_CNTL_XDIR_YDIR_YMAJOR)
330 return ((r(DP_CNTL) & 1u) << 31) | ((r(DP_CNTL) & 2u) << 14) |
331 ((r(DP_CNTL) & 4u) >> 2);
332 // Synchronous PIO availability, NOT a cycle model.
333 if (a == RBBM_STATUS)
334 return (busy() ? GUI_ACTIVE : 0u) | (halted() ? 0u : 64u);
335 // Both channels powered up (bits 1:0, polled by the BIOS) and idle; no
336 // DRAM timing.
337 if (a == MC_STATUS)
338 return 7u;
339 if (a == DAC_CNTL)
340 return (r(a) & ~0xf0u) | dac_comparators(false);
341 if (a == DAC_CNTL2)
342 return (r(a) & ~DAC2_CMP_OUTPUT_MASK) | dac_comparators(true);
343 if (a == TV_DAC_CNTL)
344 return r(a) & ~((1u << 5) | (7u << 29));
345 if (a == FP_GEN_CNTL || a == FP2_GEN_CNTL)
346 return r(a) & ~FP_DETECT_SENSE;
347 if (a == CRTC_STATUS || a == CRTC2_STATUS)
348 {
349 // VBLANK_SAVE latches at vertical blank start until written with 1.
350 const unsigned controller = a == CRTC2_STATUS;
351 const bool enabled =
352 r(controller ? CRTC2_GEN_CNTL : CRTC_GEN_CNTL) & CRTC_EN;
353 const bool vblank =
354 enabled && scanline(controller) >= vertical_display(controller);
355 return (vblank ? CRTC_VBLANK_CUR : 0u) |
356 (r(a) & CRTC_VBLANK_SAVE);
357 }
358 if (a == CRTC_VLINE_CRNT_VLINE || a == CRTC2_VLINE_CRNT_VLINE)
359 return (scanline(a == CRTC2_VLINE_CRNT_VLINE) & 0x7ff) << 16;
360 if (a == CRTC_CRNT_FRAME || a == CRTC2_CRNT_FRAME)
361 return a == CRTC2_CRNT_FRAME ? m_frame2 : m_frame;
362 if (a == DSTCACHE_CTLSTAT || a == RB2D_DSTCACHE_CTLSTAT)
363 return busy() ? 0x80000000u : 0;
364 if (a == CP_RB_RPTR)
365 return r(a);
366 if (a == CP_ME_RAM_DATAH)
367 return r(a) & 63u;
368 if (a == CP_ME_RAM_DATAL)
369 {
370 const uint32_t value = r(a);
371 if (m_cp_ram_read)
372 {
373 const auto& entry = m_cp_microcode[r(CP_ME_RAM_ADDR) & 255];
374 r(CP_ME_RAM_DATAH) = entry[0];
375 r(CP_ME_RAM_DATAL) = entry[1];
376 r(CP_ME_RAM_ADDR) = (r(CP_ME_RAM_ADDR) + 1) & 255;
377 }
378 return value;
379 }
380 if (a == CP_ME_RAM_RADDR)
381 return 0; // Write-only read-address selector.
382 if (a == RB3D_ZCACHE_CTLSTAT)
383 return 0; // No queued 3D rendering or dirty Z cache.
384 if (a >= 0xf00 && a < 0x1000)
385 return m_config_read ? m_config_read(a - 0xf00) : 0xffffffffu;
386 if (a >= GPIO_VGA_DDC && a <= GPIO_CRT2_DDC)
387 {
388 // Disconnected DDC: open-drain pull-ups, no EDID responder or ACK.
389 uint32_t v = r(a) & ~0x300u;
390 if (!(v & (1u << 16)) || (v & 1u))
391 v |= 1u << 8;
392 if (!(v & (1u << 17)) || (v & 2u))
393 v |= 1u << 9;
394 return v;
395 }
396 return r(a);
397}

399uint32_t CRV200::dac_comparators(bool secondary) const
400{
401 const uint32_t control = r(secondary ? DAC_CNTL2 : DAC_CNTL);
402 const uint32_t force = r(DAC_EXT_CNTL);
403 if (!(control & (secondary ? 0x80u : 8u)) ||
404 !(force & (secondary ? 2u : 32u)))
405 return 0;
406 // PS/2 range: 0.7 V full-scale with a 75-ohm monitor attached, twice
407 // that voltage on the unterminated secondary connector. The comparator
408 // reference is 0.440 V and its result is high below that reference.
409 // Only the forced-level test is modeled; active-video samples are not.
410 if ((r(DAC_CNTL) & 3u) != 2u)
411 return 0;
412 const uint32_t data = (force >> 8) & 1023, selection = (force >> 6) & 3;
413 uint32_t result = 0;
414 for (unsigned channel = 0; channel < 3; ++channel)
415 {
416 const uint32_t level = selection == 3 || selection == channel ? data : 0;
417 if (level * (secondary ? 1400u : 700u) < 440u * 1023u)
418 result |= 1u << channel;
419 }
420 if (result == 7)
421 result |= 8;
422 return result << (secondary ? 8 : 4);
423}

425void CRV200::write32(uint32_t a, uint32_t val, uint32_t lanes)
426{
427 const uint32_t old = r(a), v = (old & ~lanes) | (val & lanes);
428 if ((a >= BIOS_SCRATCH0 && a <= BIOS_SCRATCH7) || configuration_latch(a))
429 {
430 r(a) = v;
431 return;
432 }
433 if (a == MC_IND_INDEX)
434 {
435 r(a) = v & 0x11fu;
436 return;
437 }
438 if (a == MC_IND_DATA)
439 {
440 if (r(MC_IND_INDEX) & 0x100)
441 {
442 uint32_t& mc = m_mc[r(MC_IND_INDEX) & 31];
443 mc = (mc & ~lanes) | (val & lanes);
444 }
445 return;
446 }
447 if (a == MEM_SDRAM_MODE_REG)
448 {
449 // MEM_CFG_TYPE is a read-only board strap. Mode-register programming
450 // completes synchronously, so MC_STATUS continues to report ready.
451 r(a) = (v & ~(1u << 30)) | (old & (1u << 30));
452 return;
453 }
454 if (a == DISPLAY_BASE_ADDR)
455 {
456 r(a) = v & 0xffc00000u;
457 return;
458 }
459 if (a == DISP_MISC_CNTL)
460 {
461 r(a) = v;
462 return;
463 }
464 if (a == DISP_PWR_MAN)
465 {
466 // DPMS gates connector sync, not internal timing or rendering to VRAM.
467 // D-state transition policies are retained; PCI power transitions have
468 // not been exercised by this board's guest driver.
469 r(a) = v;
470 return;
471 }
472 if (a == DISP_OUTPUT_CNTL &&
473 dac_route_supported(v, r(FP_HORZ_STRETCH), r(FP_VERT_STRETCH)))
474 {
475 r(a) = v;
476 return;
477 }
478 if (a == DISP_TEST_DEBUG_CNTL && !(v & 0x33fu))
479 {
480 // Underflow flags are read-only/W1C. A synchronous framebuffer has
481 // no fetch underflows; retain the subpicture priority control only.
482 r(a) = v & (1u << 28);
483 return;
484 }
485 if ((a == OV0_SCALE_CNTL && !(v & 0x40000000u)) ||
486 ((a == FP_HORZ_STRETCH || a == FP_VERT_STRETCH) &&
487 !(v & (1u << 25))) ||
488 (a == LVDS_GEN_CNTL && !(v & 0x80u)) ||
489 (a == OV0_FLAG_CNTL && !(r(OV0_SCALE_CNTL) & 0x40000000u)) ||
490 (a == TV_PRE_DAC_MUX_CNTL &&
491 (!(r(TV_MASTER_CNTL) & 0x80000000u) || tv_dac_forced(v))) ||
492 (a == SEPROM_CNTL1 && !(v & 0xffffu)) ||
493 // FCP0_SRC_GND: the capture clock input is tied low.
494 (a == FCP_CNTL && v == 4) ||
495 ((a == CAP0_TRIG_CNTL || a == VIPH_CONTROL || a == 0x94 ||
496 a == CRTC_MORE_CNTL || a == OVR_WID_LEFT_RIGHT ||
497 a == OVR_WID_TOP_BOTTOM) && v == 0))
498 {
499 // Observed reset/disabled states and forced TV-DAC load probes. The TV
500 // connector is empty; enabling its clock or forcing its DAC cannot
501 // drive the primary CRT or create a monitor load. Video through the TV
502 // mux, active capture/VIP/scaler/RMX/LVDS and EEPROM commands still warn.
503 r(a) = v;
504 return;
505 }
506 if (a >= 0xf00 && a < 0x1000)
507 {
508 note("read-only PCI configuration mirror write " + hex(a));
509 return;
510 }
511 if (a == CLOCK_CNTL_INDEX)
512 {
513 // Keep PLL_DIV_SEL [9:8], not only the index/write flag.
514 r(a) = v & 0x3bfu;
515 return;
516 }
517 if (a == CLOCK_CNTL_DATA)
518 {
519 // Byte lanes update only their part of the selected PLL register.
520 uint32_t& pll = m_pll[r(CLOCK_CNTL_INDEX) & 63];
521 if (r(CLOCK_CNTL_INDEX) & 0x80)
522 pll = (pll & ~lanes) | (val & lanes);
523 return;
524 }
525 if (a == CRTC_STATUS || a == CRTC2_STATUS)
526 {
527 r(a) &= ~(val & lanes & CRTC_VBLANK_SAVE);
528 return;
529 }
530 if (a == DAC_CNTL)
531 {
532 // The board's VGA frontend consumes DAC_8BIT_EN; other fields are stored.
533 r(a) = v & ~0xf0u;
534 return;
535 }
536 if (a == DAC_CNTL2 || a == TV_DAC_CNTL ||
537 a == FP_GEN_CNTL || a == FP2_GEN_CNTL)
538 {
539 // Only the primary analog connector has a monitor. Sense outputs are
540 // read-only; enabling a controller or comparator cannot attach one.
541 const uint32_t sense = a == DAC_CNTL2 ? DAC2_CMP_OUTPUT_MASK :
542 a == TV_DAC_CNTL ? ((1u << 5) | (7u << 29)) : FP_DETECT_SENSE;
543 r(a) = v & ~sense;
544 return;
545 }
546 if (a == PALETTE_INDEX)
547 {
548 r(a) = v & 0x00ff00ffu;
549 if (m_palette_index)
550 m_palette_index();
551 return;
552 }
553 if (a == PALETTE_DATA)
554 {
555 const uint32_t idx = r(PALETTE_INDEX) & 255;
556 set_palette_color(idx, v, palette_bank());
557 r(PALETTE_INDEX) = (r(PALETTE_INDEX) & ~255u) | ((idx + 1) & 255);
558 return;
559 }
560 if (a == GEN_INT_STATUS)
561 {
562 r(a) &= ~(val & lanes & (1u | CRTC2_VBLANK_INT));
563 update_irq();
564 return;
565 }
566 if (a == GEN_INT_CNTL)
567 {
568 r(a) = v;
569 update_irq();
570 return;
571 }
572 if (a == RBBM_SOFT_RESET)
573 {
574 if ((v & ~old) & (1u | 32u))
575 {
576 m_pending = Pending{};
577 if ((v & ~old) & 1u)
578 {
579 m_cp_packet = {};
580 m_cp_ib_cursor = m_cp_ib_left = m_cp_since_update = 0;
581 r(CP_RB_RPTR) = r(CP_RB_WPTR) = 0;
582 }
583 m_inferred_clear_armed = false;
584 m_fault.clear();
585 }
586 r(a) = v;
587 return;
588 }
589 if (a == RBBM_STATUS || a == MC_STATUS || a == CONFIG_MEMSIZE ||
590 a == CONFIG_APER_SIZE || a == CONFIG_APER_0_BASE ||
591 a == CONFIG_APER_1_BASE || a == CONFIG_REG_1_BASE ||
592 a == CONFIG_REG_APER_SIZE || a == CRTC_CRNT_FRAME ||
593 a == CRTC2_CRNT_FRAME || a == CP_RB_RPTR)
594 return;
595 if (a == AGP_BASE || a == CP_RB_WPTR_DELAY || a == AIC_CNTL)
596 {
597 r(a) = v;
598 if ((a == CP_RB_WPTR_DELAY || a == AIC_CNTL) && v)
599 fail("unimplemented CP delay or PCI GART mode at " + hex(a));
600 return;
601 }
602 if (a == CP_RB_BASE || a == CP_IB_BASE || a == CP_RB_RPTR_ADDR ||
603 a == CP_RB_RPTR_WR || a == SCRATCH_ADDR || a == SCRATCH_UMSK)
604 {
605 const uint32_t mask = a == CP_RB_RPTR_WR ? 0x7fffffu :
606 a == SCRATCH_ADDR ? 0xffffffe0u :
607 a == SCRATCH_UMSK ? 0x3003fu :

871 return;
872 }
873 if (a == SRC_X || a == SRC_Y || a == DST_X || a == DST_Y)

881 }

884 r(a) = v;
885 return;
886 }

888 {

891 return;
892 }
893 if (a == SC_LEFT || a == SC_TOP)
894 {

896 r(SC_TOP_LEFT) = a == SC_LEFT
897 ? ((r(SC_TOP_LEFT) & 0xffff0000u) | (v & 0xffff))
898 : ((r(SC_TOP_LEFT) & 0xffff) | (v << 16));
899 return;
900 }
901 if (a == SC_RIGHT || a == SC_BOTTOM)
902 {
903 r(a) = v;
904 r(SC_BOTTOM_RIGHT) = a == SC_RIGHT
905 ? ((r(SC_BOTTOM_RIGHT) & 0xffff0000u) | (v & 0xffff))
906 : ((r(SC_BOTTOM_RIGHT) & 0xffff) | (v << 16));
907 return;
908 }
909 if (a == DST_HEIGHT_WIDTH || a == DST_WIDTH_HEIGHT)
910 {
911 r(a) = v;
912 if (halted())
913 return;

915 a == DST_HEIGHT_WIDTH ? v & 0x3fff : (v >> 16) & 0x3fff,
916 a == DST_HEIGHT_WIDTH ? (v >> 16) & 0x3fff : v & 0x3fff);
917 return;
918 }

920 {
921 r(a) = v;

924 }
925 if (!shadow_allowed(a) && m_options.strict_mmio)
926 {
927 fail("unknown MMIO write " + hex(a));
928 return;
929 }
930 r(a) = v;
931 m_counters.shadow_writes++;
932 // Stored without a modeled effect: report the register once.
933 if (!m_warned[a / 4])
934 {
935 m_warned[a / 4] = true;
936 if (m_unimplemented)
937 m_unimplemented("RV200 register", a, v, true);
938 }
939}

943// Aperture 1 mirrors aperture 0 with its own NONSURF byte swapping, unless
944// HOST_PATH_CNTL.HDP_APER_CNTL maps it to the following VRAM instead.
945bool CRV200::aperture_index(uint64_t a, size_t& index)
946{
947 if (a < 2ull * ApertureSize)
948 {
949 const bool ap1 = a >= ApertureSize;
950 const uint32_t flags = (r(SURFACE_CNTL) >> (ap1 ? 22 : 20)) & 3u;
951 const uint32_t sw = (flags & 1u) ^ ((flags & 2u) ? 3u : 0u);
952 uint64_t offset = a;
953 if (ap1 && !(r(HOST_PATH_CNTL) & HDP_APER_CNTL))
954 offset -= ApertureSize;
955 offset ^= sw;
956 if (offset < m_vram.size())
957 {
958 index = size_t(offset);
959 return true;
960 }
961 }
962 if (!m_aperture_warned)
963 {
964 m_aperture_warned = true;
965 note("aperture access beyond implemented VRAM " + hex(uint32_t(a)));
966 }
967 return false;
968}

1004bool CRV200::translate(uint64_t gpu, size_t count, size_t& index) const
1005{
1006 const uint64_t base = uint64_t(r(MC_FB_LOCATION) & 0xffff) << 16;
1007 const uint64_t end = (uint64_t(r(MC_FB_LOCATION) >> 16) + 1) << 16;
1008 if (gpu < base || gpu > end || count > end - gpu ||
1009 gpu - base > m_vram.size() || count > m_vram.size() - (gpu - base))
1010 return false;
1011 index = size_t(gpu - base);
1012 return true;
1013}

1049// Scissor edges are 14-bit sign-magnitude; right/bottom are exclusive (XAA
1050// writes x2 + 1).
1051static int32_t scissor_edge(uint32_t v)
1052{
1053 return (v & 0x8000) ? -int32_t(v & 0x3fff) : int32_t(v & 0x3fff);
1054}

1061
1062void CRV200::draw(uint32_t w, uint32_t h, bool positive_host)
1063{
1064 const bool clear_candidate = m_inferred_clear_armed;
1065 m_inferred_clear_armed = false;
1066 if (m_pending.active)
1067 {
1068 fail("new primitive while host upload is pending");
1069 return;
1070 }
1071 if (!w || !h)
1072 return;
1073 const uint32_t g = r(DP_GUI_MASTER_CNTL), type = r(DP_DATATYPE);
1074 uint32_t bpp = bytes_per_pixel(type & 15);
1075 const uint8_t op = uint8_t(r(DP_MIX) >> 16);
1076 bool uses_s = rop_uses(op, 1);
1077 const bool uses_p = rop_uses(op, 2);
1078 const uint32_t source = (r(DP_MIX) >> 8) & 7, brush = (type >> 8) & 15;
1079 // Inferred from behavior and analysis:
1080 // OpenVMS RV200ClearMemory saves a configured 32-bit datatype, writes zero,
1081 // submits a black PATCOPY rectangle, then restores the datatype.
1082 // Neither that sequence nor the available ATI references establishes
1083 // datatype 0's physical byte enables or address scaling.
1084 // This is best guess for the usage now to bring up OpenVMS
1085 // We'll use the preceding width for that one black clear, with ordinary
1086 // linear row overlap. We don't actually know what really happens in
1087 // real silicon and must not become a generic datatype-0 pixel format.
1088 const bool inferred_clear = clear_candidate && type == 0 && op == 0xf0 &&
1089 source == 2 && r(DP_BRUSH_FRGD_CLR) == 0 &&
1090 r(DP_BRUSH_BKGD_CLR) == 0 && r(DP_WRITE_MSK) == 0xffffffffu &&
1091 r(CLR_CMP_CNTL) == 0 && r(DP_SRC_ENDIAN) == 0 &&
1092 !(g & ((1u << 27) | GMC_SRC_CLIP | 0x80000000u)) &&
1093 !(r(DST_PITCH_OFFSET) & 0xc0000000u);
1094 if (inferred_clear)
1095 bpp = 4;
1096 if (!bpp)
1097 {
1098 fail("unsupported pixel format " + hex(type & 15) +
1099 " (DP_DATATYPE=" + hex(type) + ", rectangle=" +
1100 std::to_string(w) + "x" + std::to_string(h) + ", DST_PITCH_OFFSET=" +
1101 hex(r(DST_PITCH_OFFSET)) + ")");
1102 return;
1103 }
1104 if ((g & (1u << 27)) || !compare_supported(r(CLR_CMP_CNTL)))
1105 {
1106 fail("3D or unsupported color comparison in 2D operation");
1107 return;
1108 }
1109 const uint32_t compare = r(CLR_CMP_CNTL);
1110 const uint32_t key = r(CLR_CMP_CLR_SRC), key_mask = r(CLR_CMP_MASK);
1111 // A compare can require source data even when the selected ROP does not;
1112 // host input is consumed even for a source-independent ROP.
1113 if (compare || source == 3)
1114 uses_s = true;
1115 if (g & GMC_SRC_CLIP)
1116 {
1117 fail("source clipping unsupported");
1118 return;
1119 }
1120 if (uses_p && brush != 13 && brush != 0 && brush != 1)
1121 {
1122 fail("unsupported pattern brush datatype");
1123 return;
1124 }
1125 if (uses_p && brush <= 1 && (g & 0x80000000u))
1126 {
1127 fail("8x8 brush origin auto-load not implemented");
1128 return;
1129 }
1130 if (uses_s && source != 2 && source != 3)
1131 {
1132 fail("unsupported 2D source selector");
1133 return;
1134 }
1135 const uint32_t source_format = (type >> 16) & 3;
1136 const bool mono = source_format < 2;
1137 if (uses_s && source_format != 3 && !(source == 3 && mono))
1138 {
1139 fail("unsupported source datatype/transport");
1140 return;
1141 }
1142 if (uses_s && source == 3 && uses_p && brush != 13)
1143 {
1144 fail("combined host transfer and patterned brush unsupported");
1145 return;
1146 }
1147 if ((type & 0x20000000u) || r(DP_SRC_ENDIAN))
1148 {
1149 fail("host/source endianness mode unsupported");
1150 return;
1151 }
1152 const uint32_t dst = r(DST_PITCH_OFFSET), src = r(SRC_PITCH_OFFSET);
1153 if ((dst & 0xc0000000u) || (uses_s && source == 2 && (src & 0xc0000000u)))
1154 {
1155 fail("tiled engine surfaces unsupported");
1156 return;
1157 }
1158 const uint32_t db = (dst & 0x3fffffu) << 10, dp = (dst >> 16) & 0x3fc0u;
1159 const uint32_t sb = (src & 0x3fffffu) << 10, sp = (src >> 16) & 0x3fc0u;
1160 const int32_t x = engine_coordinate(r(DST_Y_X)),
1161 y = engine_coordinate(r(DST_Y_X) >> 16);
1162 const int32_t sx = engine_coordinate(r(SRC_Y_X)),
1163 sy = engine_coordinate(r(SRC_Y_X) >> 16);
1164 const int32_t xd = positive_host || (r(DP_CNTL) & 1) ? 1 : -1,
1165 yd = positive_host || (r(DP_CNTL) & 2) ? 1 : -1;
1166 const uint32_t tl = r(SC_TOP_LEFT), br = r(SC_BOTTOM_RIGHT),
1167 mask = r(DP_WRITE_MSK), p = r(DP_BRUSH_FRGD_CLR);
1168 const uint32_t brush_x = r(BRUSH_Y_X) & 7,
1169 brush_y = (r(BRUSH_Y_X) >> 8) & 7;
1170 const auto destination_address = [&](int32_t xx, int32_t yy, size_t& di) {
1171 if (!inferred_clear)
1172 return pixel_address(db, dp, xx, yy, bpp, di);
1173 // The inferred 2048-pixel clear has a 4096-byte pitch: rows overlap
1174 // at 32 bits/pixel. Do not clip to pitch or change the guest's stride.
1175 return xx >= 0 && yy >= 0 && translate(
1176 uint64_t(db) + uint64_t(yy) * dp + uint64_t(xx) * bpp, bpp, di);
1177 };
1178 // Validate the entire destination/source footprint before modifying VRAM.
1179 for (uint32_t j = 0; j < h; j++)
1180 for (uint32_t i = 0; i < w; i++)
1181 {
1182 const int32_t xx = x + xd * int32_t(i), yy = y + yd * int32_t(j);
1183 size_t di = 0, si = 0;
1184 if (clipped(xx, yy, tl, br))
1185 continue;
1186 if (!destination_address(xx, yy, di) ||
1187 (uses_s && source == 2 &&
1188 !pixel_address(
1189 sb, sp, sx + xd * int32_t(i), sy + yd * int32_t(j), bpp,
1190 si)))
1191 {
1192 fail("2D address outside GPU framebuffer window/pitch");
1193 return;
1194 }
1195 }
1196 if (inferred_clear)
1197 note("Inferred datatype-0 black clear using preceding 32-bit format; "
1198 "physical VRAM footprint unverified");
1199 m_counters.draws++;
1200 if (uses_s && source == 3)
1201 {
1202 if (bpp == 3 && !mono)
1203 {
1204 fail("24-bit HOST_DATA packing unsupported");
1205 return;
1206 }
1207 Pending& q = m_pending;
1208 q = Pending{};
1209 q.active = true;
1210 q.packed = true;
1211 q.width = w;
1212 q.height = h;
1213 q.words_per_row = mono ? (w + 31) / 32 : (w * bpp + 3) / 4;
1214 q.x = x;
1215 q.y = y;
1216 q.dx = xd;
1217 q.dy = yd;
1218 q.base = db;
1219 q.pitch = dp;
1220 q.bpp = bpp;
1221 q.mask = mask;
1222 q.pattern = p;
1223 q.rop = op;
1224 q.sc_tl = tl;
1225 q.sc_br = br;
1226 q.source_format = source_format;
1227 q.mono_lsb = (type >> 30) & 1u;
1228 q.fg = r(DP_SRC_FRGD_CLR);
1229 q.bg = r(DP_SRC_BKGD_CLR);
1230 q.compare = compare;
1231 q.compare_color = key;
1232 q.compare_mask = key_mask;
1233 return;
1234 }
1235 for (uint32_t j = 0; j < h; j++)
1236 for (uint32_t i = 0; i < w; i++)
1237 {
1238 const int32_t xx = x + xd * int32_t(i), yy = y + yd * int32_t(j);
1239 size_t di = 0, si = 0;
1240 if (clipped(xx, yy, tl, br))
1241 continue;
1242 destination_address(xx, yy, di);
1243 uint32_t s = 0;
1244 if (uses_s && source == 2)
1245 {
1246 pixel_address(
1247 sb, sp, sx + xd * int32_t(i), sy + yd * int32_t(j), bpp,
1248 si);
1249 s = pixel_read(si, bpp);
1250 }
1251 if (source_rejected(compare, key, key_mask, s, bpp))
1252 continue;
1253 uint32_t pattern = p;
1254 if (uses_p && brush <= 1)
1255 {
1256 // Screen-aligned pattern; BRUSH_Y_X offsets its origin.
1257 const unsigned row = unsigned(yy + int32_t(brush_y)) & 7u;
1258 const unsigned col = unsigned(xx + int32_t(brush_x)) & 7u;
1259 const uint32_t data = r(row < 4 ? BRUSH_DATA0 : BRUSH_DATA1);
1260 const bool ink = mono_bit(
1261 data, (row & 3u) * 8u + col, (type & (1u << 30)) != 0);
1262 if (!ink && brush == 1)
1263 continue;
1264 pattern = ink ? p : r(DP_BRUSH_BKGD_CLR);
1265 }
1266 const uint32_t d = pixel_read(di, bpp);
1267 pixel_write(
1268 di, bpp, (rop3(op, pattern, s, d) & mask) | (d & ~mask));
1269 }
1270}

1272// Zero-width Bresenham line from DST_LINE_START, excluding the end pixel (XAA
1273// draws it separately). Solid and 32x1 mono brushes; the brush phase is
1274// DST_LINE_PATCOUNT[4:0].
1275void CRV200::draw_line(uint32_t start, uint32_t end)
1276{
1277 m_inferred_clear_armed = false;
1278 if (m_pending.active)
1279 {
1280 fail("new primitive while host upload is pending");
1281 return;
1282 }
1283 const uint32_t g = r(DP_GUI_MASTER_CNTL), type = r(DP_DATATYPE),
1284 bpp = bytes_per_pixel(type & 15), brush = (type >> 8) & 15;
1285 const uint8_t op = uint8_t(r(DP_MIX) >> 16);
1286 const bool uses_p = rop_uses(op, 2);
1287 if (!bpp)
1288 {
1289 fail("unsupported pixel format " + hex(type & 15) +
1290 " (DP_DATATYPE=" + hex(type) + ", line)");
1291 return;
1292 }
1293 if ((g & (1u << 27)) || r(CLR_CMP_CNTL))
1294 {
1295 fail("3D or color comparison in line operation");
1296 return;
1297 }
1298 if (rop_uses(op, 1))
1299 {
1300 fail("source-dependent line ROP unsupported");
1301 return;
1302 }
1303 if (uses_p && brush != 13 && brush != 6 && brush != 7)
1304 {
1305 fail("unsupported line brush datatype");
1306 return;
1307 }
1308 const uint32_t dst = r(DST_PITCH_OFFSET);
1309 if (dst & 0xc0000000u)
1310 {
1311 fail("tiled engine surfaces unsupported");
1312 return;
1313 }
1314 const uint32_t db = (dst & 0x3fffffu) << 10, dp = (dst >> 16) & 0x3fc0u;
1315 const int32_t x0 = engine_coordinate(start),
1316 y0 = engine_coordinate(start >> 16);
1317 const int32_t dx = engine_coordinate(end) - x0,
1318 dy = engine_coordinate(end >> 16) - y0;
1319 const int32_t adx = dx < 0 ? -dx : dx, ady = dy < 0 ? -dy : dy;
1320 const int32_t sx = dx < 0 ? -1 : 1, sy = dy < 0 ? -1 : 1;
1321 const bool xmajor = adx >= ady;
1322 const int32_t major = xmajor ? adx : ady, minor = xmajor ? ady : adx;
1323 const uint32_t tl = r(SC_TOP_LEFT), br = r(SC_BOTTOM_RIGHT),
1324 mask = r(DP_WRITE_MSK), fg = r(DP_BRUSH_FRGD_CLR),
1325 bg = r(DP_BRUSH_BKGD_CLR), pattern = r(BRUSH_DATA0),
1326 phase = r(DST_LINE_PATCOUNT) & 31;
1327 const bool lsb = (type & (1u << 30)) != 0;
1328 // Validate the unclipped footprint before modifying VRAM.
1329 for (int pass = 0; pass < 2; ++pass)
1330 {
1331 int32_t x = x0, y = y0, err = 2 * minor - major;
1332 for (int32_t i = 0; i < major; ++i)
1333 {
1334 size_t di = 0;
1335 if (!clipped(x, y, tl, br))
1336 {
1337 if (!pixel_address(db, dp, x, y, bpp, di))
1338 {
1339 fail("line address outside GPU framebuffer window/pitch");
1340 return;
1341 }
1342 const bool ink =
1343 brush == 13 || mono_bit(pattern, (phase + i) & 31, lsb);
1344 if (pass && (ink || brush != 7 || !uses_p))
1345 {
1346 const uint32_t d = pixel_read(di, bpp);
1347 const uint32_t p = ink ? fg : bg;
1348 pixel_write(
1349 di, bpp, (rop3(op, p, 0, d) & mask) | (d & ~mask));
1350 }
1351 }
1352 if (err > 0)
1353 {
1354 if (xmajor)
1355 y += sy;
1356 else
1357 x += sx;
1358 err -= 2 * major;
1359 }
1360 err += 2 * minor;
1361 if (xmajor)
1362 x += sx;
1363 else
1364 y += sy;
1365 }
1366 }
1367 // Clipped and transparent points still consume positions in the pattern.
1368 r(DST_LINE_PATCOUNT) =
1369 (r(DST_LINE_PATCOUNT) & ~31u) | ((phase + uint32_t(major)) & 31u);
1370 m_counters.draws++;
1371}

1373void CRV200::host_word(uint32_t value, bool last)
1374{
1375 if (halted())
1376 return;
1377 if (!m_pending.active)
1378 {
1379 fail("HOST_DATA without pending upload");
1380 return;
1381 }
1382 auto& p = m_pending;
1383 const bool mono = p.source_format < 2;
1384 const uint32_t pixels_per_word = mono ? 32u : 4u / p.bpp;
1385 // Source 3 is a continuous pixel/bit stream.
1386 const uint32_t total_words = p.packed
1387 ? (p.width * p.height + pixels_per_word - 1) / pixels_per_word
1388 : p.words_per_row * p.height;
1389 if (last && p.word + 1 != total_words)
1390 {
1391 fail("premature HOST_DATA_LAST");
1392 return;
1393 }
1394 for (uint32_t n = 0; n < pixels_per_word; n++)
1395 {
1396 const uint32_t pixel = p.word * pixels_per_word + n;
1397 const uint32_t row = p.packed ? pixel / p.width : p.word / p.words_per_row;
1398 const uint32_t col = p.packed ? pixel % p.width :
1399 (p.word % p.words_per_row) * pixels_per_word + n;
1400 if (row >= p.height || col >= p.width)
1401 continue;
1402 const int32_t x = p.x + p.dx * int32_t(col),
1403 y = p.y + p.dy * int32_t(row);
1404 size_t idx = 0;
1405 if (clipped(x, y, p.sc_tl, p.sc_br))
1406 continue;
1407 if (!pixel_address(p.base, p.pitch, x, y, p.bpp, idx))
1408 {
1409 fail("host upload address invalidated");
1410 return;
1411 }
1412 uint32_t source;
1413 if (mono)
1414 {
1415 const bool ink = mono_bit(value, n, p.mono_lsb != 0);
1416 if (!ink && p.source_format == 1)
1417 continue;
1418 source = ink ? p.fg : p.bg;
1419 }
1420 else
1421 source = value >> (8 * p.bpp * n);
1422 if (source_rejected(
1423 p.compare, p.compare_color, p.compare_mask, source, p.bpp))
1424 continue;
1425 const uint32_t d = pixel_read(idx, p.bpp);
1426 pixel_write(
1427 idx, p.bpp,
1428 (rop3(uint8_t(p.rop), p.pattern, source, d) & p.mask) |
1429 (d & ~p.mask));
1430 }
1431 if (++p.word == total_words)
1432 p.active = false;
1433}

1437void CRV200::advance_scanlines(uint32_t lines)
1438{
1439 run_cp();
1440 // Register-derived vertical timing; PLL frequency is not evaluated.
1441 if (lines > 1000000u)
1442 {
1443 fail("scanline budget exceeded");
1444 return;
1445 }
1446 // An atomic PLL update completes on the next model step.
1447 m_pll[PPLL_REF_DIV] &= ~(1u << 15);
1448 m_pll[P2PLL_REF_DIV] &= ~(1u << 15);
1449 for (unsigned controller = 0; controller < 2; ++controller)
1450 {
1451 if (!(r(controller ? CRTC2_GEN_CNTL : CRTC_GEN_CNTL) & CRTC_EN))
1452 continue;
1453 const uint32_t total = vertical_total(controller),
1454 display = vertical_display(controller);
1455 if (display >= total)
1456 continue;
1457 uint32_t& line = controller ? m_scanline2 : m_scanline;
1458 uint32_t& frame = controller ? m_frame2 : m_frame;
1459 for (uint32_t n = 0; n < lines; n++)
1460 {
1461 if (++line >= total)
1462 {
1463 line = 0;
1464 frame++;
1465 }
1466 if (line == display)
1467 {
1468 r(GEN_INT_STATUS) |= controller ? CRTC2_VBLANK_INT : 1u;
1469 r(controller ? CRTC2_STATUS : CRTC_STATUS) |= CRTC_VBLANK_SAVE;
1470 }
1471 }
1472 }
1473 update_irq();
1474}

1484CRV200::Frame CRV200::scanout(std::string* error) const
1485{
1486 auto reject = [error](const char* why) {
1487 if (error)
1488 *error = why;
1489 return Frame{};
1490 };
1491 const unsigned controller = display_controller();
1492 if (r(DISP_MISC_CNTL) & (controller ? 0x9000u : 0x11u))
1493 return reject("display pixel pipeline held in reset");
1494 if (!display_output_enabled())
1495 return reject("display output powered down or unsupported DAC route");
1496 const uint32_t gen = r(controller ? CRTC2_GEN_CNTL : CRTC_GEN_CNTL);
1497 const bool blank = controller
1498 ? (gen & (CRTC2_DISPLAY_DIS | (1u << 26) | (3u << 28) | 0x70u))
1499 : (r(CRTC_EXT_CNTL) & DISPLAY_DIS);
1500 if (!(gen & CRTC_EN) || blank)
1501 return reject("CRTC disabled/blanked");
1502 if (gen & 3u)
1503 return reject("interlaced/doublescan scanout unsupported");
1504 if (controller && (gen & (1u << 15)))
1505 return reject("CRTC2 icon scanout unsupported");
1506 const uint32_t cursor_mode = (gen >> 20) & 7;
1507 if ((gen & CRTC_CUR_EN) && cursor_mode != 0 && cursor_mode != 2)
1508 return reject("unsupported cursor format");
1509 if (r(controller ? CRTC2_OFFSET_CNTL : CRTC_OFFSET_CNTL))
1510 return reject(
1511 "nonzero CRTC_OFFSET_CNTL requires an unimplemented display "
1512 "layout");
1513 const uint32_t fmt = (gen >> 8) & 15, bpp = bytes_per_pixel(fmt);
1514 const uint32_t htotal =
1515 r(controller ? CRTC2_H_TOTAL_DISP : CRTC_H_TOTAL_DISP);
1516 const uint32_t w = (((htotal >> 16) & 0x1ff) + 1) * 8,
1517 h = vertical_display(controller);
1518 const uint32_t pitch =
1519 (r(controller ? CRTC2_PITCH : CRTC_PITCH) & 0x7ff) * 8 * bpp;
1520 const uint64_t base =
1521 uint64_t(controller ? 0 : r(DISPLAY_BASE_ADDR)) +
1522 (r(controller ? CRTC2_OFFSET : CRTC_OFFSET) & ~7u);
1523 if (!bpp || w > 4096 || h > 4096 || uint64_t(w) * bpp > pitch)
1524 return reject("unsupported/invalid display format or pitch");
1525 size_t first = 0, last = 0;
1526 if (!translate(base, bpp, first) ||
1527 !translate(
1528 uint64_t(base) + uint64_t(h - 1) * pitch + uint64_t(w - 1) * bpp,
1529 bpp, last))
1530 return reject("display address outside GPU framebuffer window");
1531 Frame f;
1532 f.width = w;
1533 f.height = h;
1534 f.argb.resize(size_t(w) * h);
1535 for (uint32_t y = 0; y < h; y++)
1536 for (uint32_t x = 0; x < w; x++)
1537 {
1538 uint32_t v = pixel_read(
1539 first + size_t(y) * pitch + size_t(x) * bpp, bpp),
1540 rr = 0, gg = 0, bb = 0;
1541 if (fmt == 2)
1542 {
1543 v = palette_color(v, controller);
1544 rr = (v >> 16) & 255;
1545 gg = (v >> 8) & 255;
1546 bb = v & 255;
1547 }
1548 else if (fmt == 3 || fmt == 4)
1549 {
1550 rr = (v >> (fmt == 3 ? 10 : 11)) & 31;
1551 gg = (v >> 5) & (fmt == 3 ? 31 : 63);
1552 bb = v & 31;
1553 rr = (rr << 3) | (rr >> 2);
1554 bb = (bb << 3) | (bb >> 2);
1555 gg = fmt == 3 ? ((gg << 3) | (gg >> 2))
1556 : ((gg << 2) | (gg >> 4));
1557 }
1558 else
1559 {
1560 rr = (v >> 16) & 255;
1561 gg = (v >> 8) & 255;
1562 bb = v & 255;
1563 }
1564 if (fmt != 2)
1565 {
1566 rr = (palette_color(rr, controller) >> 16) & 255;
1567 gg = (palette_color(gg, controller) >> 8) & 255;
1568 bb = palette_color(bb, controller) & 255;
1569 }
1570 f.argb[size_t(y) * w + x] =
1571 0xff000000u | (rr << 16) | (gg << 8) | bb;
1572 }
1573 composite_cursor(f, controller);
1574 return f;
1575}

1577// 64x64 cursor at CUR_OFFSET. Mono rows: 8 AND then 8 XOR bytes, MSB first
1578// (XFree86 R128/Radeon); ARGB rows: 64 premultiplied pixels. HORZ_VERT_OFF
1579// skips left columns and shortens the top.
1580void CRV200::composite_cursor(Frame& f, unsigned controller) const
1581{
1582 const uint32_t gen = r(controller ? CRTC2_GEN_CNTL : CRTC_GEN_CNTL),
1583 mode = (gen >> 20) & 7, delta = controller ? 0x100 : 0;
1584 if (!(gen & CRTC_CUR_EN) || (mode != 0 && mode != 2))
1585 return;
1586 const uint32_t posn = r(CUR_HORZ_VERT_POSN + delta),
1587 off = r(CUR_HORZ_VERT_OFF + delta);
1588 const uint32_t px = (posn >> 16) & 0x3fff, py = posn & 0xfff;
1589 const uint32_t xoff = (off >> 16) & 0x3f, yoff = off & 0x3f;
1590 const uint32_t stride = mode ? 256 : 16;
1591 const uint32_t clr0 = r(CUR_CLR0 + delta) & 0xffffff,
1592 clr1 = r(CUR_CLR1 + delta) & 0xffffff;
1593 const uint64_t base =
1594 uint64_t(r(controller ? CRTC2_DISPLAY_BASE_ADDR : DISPLAY_BASE_ADDR)) +
1595 (r(CUR_OFFSET + delta) & 0x07fffff0u);
1596 for (uint32_t row = 0; row < 64 - yoff && py + row < f.height; ++row)
1597 {
1598 size_t line = 0;
1599 if (!translate(
1600 base + uint64_t(row) * stride, stride, line))
1601 return;
1602 for (uint32_t col = xoff; col < 64 && px + col - xoff < f.width; ++col)
1603 {
1604 uint32_t& d = f.argb[size_t(py + row) * f.width + px + col - xoff];
1605 if (!mode)
1606 {
1607 const unsigned bit = 7 - (col & 7);
1608 const bool and_bit = (m_vram[line + col / 8] >> bit) & 1;
1609 const bool xor_bit = (m_vram[line + 8 + col / 8] >> bit) & 1;
1610 if (!and_bit)
1611 d = 0xff000000u | (xor_bit ? clr1 : clr0);
1612 else if (xor_bit)
1613 d ^= 0x00ffffffu;
1614 continue;
1615 }
1616 const uint32_t c = pixel_read(line + col * 4, 4), a = c >> 24;
1617 uint32_t out = 0xff000000u;
1618 for (unsigned shift = 0; shift < 24; shift += 8)
1619 {
1620 const uint32_t v = ((c >> shift) & 255) +
1621 (((d >> shift) & 255) * (255 - a) + 127) / 255;
1622 out |= std::min(v, 255u) << shift;
1623 }
1624 d = out;
1625 }
1626 }
1627}

1736// Command processor:
1737// the reached firmware exposes type-0 register writes, type-2 padding and type-3 NOP.
1738// So far only one indirect stream.
1739static uint32_t cp_swap(uint32_t value, unsigned mode)
1740{
1741 if (mode == 1)
1742 return ((value & 0x00ff00ffu) << 8) | ((value & 0xff00ff00u) >> 8);
1743 if (mode == 2)
1744 return ((value & 255u) << 24) | ((value & 0xff00u) << 8) |
1745 ((value >> 8) & 0xff00u) | (value >> 24);
1746 if (mode == 3)
1747 return (value << 16) | (value >> 16);
1748 return value;
1749}
1750
1751bool CRV200::cp_memory(uint64_t address, uint32_t& value, bool write,
1752 unsigned swap)
1753{
1754 if (address > 0xfffffffcu || (address & 3))
1755 return false;
1756 const uint64_t fb_start = uint64_t(r(MC_FB_LOCATION) & 0xffffu) << 16;
1757 const uint64_t fb_end = (uint64_t(r(MC_FB_LOCATION) >> 16) + 1) << 16;
1758 if (address >= fb_start && address < fb_end)
1759 {
1760 size_t index;
1761 if (!translate(address, 4, index))
1762 return false;
1763 // BUF_SWAP applies only to system-memory command fetches.
1764 if (write)
1765 {
1766 const uint32_t data = cp_swap(value, swap);
1767 for (unsigned n = 0; n < 4; ++n)
1768 m_vram[index + n] = uint8_t(data >> (n * 8));
1769 }
1770 else
1771 value = pixel_read(index, 4);
1772 return true;
1773 }
1774 if ((r(AIC_CNTL) & 1) || (r(BUS_CNTL) & (1u << 6)))
1775 return false; // Unreached PCI GART, or disabled bus mastering.
1776 const uint64_t agp_start = uint64_t(r(MC_AGP_LOCATION) & 0xffffu) << 16;
1777 const uint64_t agp_end = (uint64_t(r(MC_AGP_LOCATION) >> 16) + 1) << 16;
1778 if (address >= agp_start && address < agp_end)
1779 address = address - agp_start + r(AGP_BASE);
1780 if (address > 0xfffffffcu)
1781 return false;
1782 uint8_t bytes[4];
1783 if (write)
1784 {
1785 const uint32_t data = cp_swap(value, swap);
1786 for (unsigned n = 0; n < 4; ++n)
1787 bytes[n] = uint8_t(data >> (8 * n));
1788 return m_dma_write && m_dma_write(uint32_t(address), bytes, 4);
1789 }
1790 if (!m_dma_read || !m_dma_read(uint32_t(address), bytes, 4))
1791 return false;
1792 value = uint32_t(bytes[0]) | (uint32_t(bytes[1]) << 8) |
1793 (uint32_t(bytes[2]) << 16) | (uint32_t(bytes[3]) << 24);
1794 value = cp_swap(value, swap);
1795 return true;
1796}

1822void CRV200::run_cp()
1823{
1824 const unsigned mode = r(CP_CSQ_CNTL) >> 28;
1825 if (m_cp_running || halted() || !mode ||
1826 (r(RBBM_SOFT_RESET) & 1) || !cp_busy())
1827 return;
1828 if (mode != 2 && mode != 4 && mode != 6 && mode != 8)
1829 {
1830 fail("unimplemented CP queue mode " + hex(mode));
1831 return;
1832 }
1833 const uint32_t control = r(CP_RB_CNTL);
1834 const unsigned size = control & 63, block = (control >> 8) & 63;
1835 // The pointers are 23 bits. Larger exponents cannot be represented.
1836 if (size > 22 || ((control >> 18) & 3) == 3)
1837 {
1838 fail("invalid/unimplemented CP ring size or fetch mode " + hex(control));
1839 return;
1840 }
1841 const uint32_t ring_words = 2u << size;
1842 if (r(CP_RB_RPTR) >= ring_words || r(CP_RB_WPTR) >= ring_words)
1843 {
1844 fail("CP ring pointer outside configured buffer");
1845 return;
1846 }
1847 m_cp_running = true;
1848 // A finite scheduling quantum also permits large rings to be snapshotted
1849 // between packets. Further MMIO ticks resume execution.
1850 for (unsigned budget = 0; budget < 65536 && !halted(); ++budget)
1851 {
1852 const unsigned stream = m_cp_ib_left ? 1 : 0;
1853 if (stream && mode == 2)
1854 break; // Indirect fetching is disabled, not completed.
1855 if (!stream && m_cp_packet[1].remaining)
1856 {
1857 fail("CP indirect buffer ends inside a packet");
1858 break;
1859 }
1860 if (!stream && r(CP_RB_RPTR) == r(CP_RB_WPTR))
1861 break;
1862 CPPacket& packet = m_cp_packet[stream];
1863 const uint64_t address = stream ? uint64_t(m_cp_ib_cursor) :
1864 uint64_t(r(CP_RB_BASE)) + uint64_t(r(CP_RB_RPTR)) * 4;
1865 uint32_t word;
1866 if (!cp_memory(address, word, false, (control >> 16) & 3))
1867 {
1868 fail("CP command fetch failed at " + hex(uint32_t(address)));
1869 break;
1870 }
1871 if (packet.remaining && (packet.header >> 30) == 0)
1872 {
1873 const uint32_t target = alias(packet.reg);
1874 if (target >= RegisterSpaceSize || target < 8 ||
1875 (target >= 0x1000 && target < 0x1400))
1876 {
1877 fail("unsupported CP register target " + hex(target));
1878 break;
1879 }
1880 if (target == CP_CSQ_CNTL || target == CP_RB_BASE ||
1881 target == CP_RB_CNTL || target == CP_RB_WPTR ||
1882 target == RBBM_SOFT_RESET)
1883 {
1884 fail("CP self-control register writes are unsupported at " + hex(target));
1885 break;
1886 }
1887 if (target == WAIT_UNTIL && (word & ~0x0007c200u))
1888 {
1889 fail("unsupported CP wait conditions " + hex(word));
1890 break;
1891 }
1892 if (m_pending.active &&
1893 ((target == WAIT_UNTIL &&
1894 ((word & 0x00054000u) || (r(ISYNC_CNTL) & 16))) ||
1895 ((r(ISYNC_CNTL) & 2) &&
1896 (target == 0x26c0 || target == 0x1c44 ||
1897 target == RB3D_ZCACHE_CTLSTAT)) ||
1898 ((r(ISYNC_CNTL) & 32) && target >= SCRATCH_REG0 &&
1899 target <= SCRATCH_REG5)))
1900 break; // Leave this DWORD pending until the upload completes.
1901 if (stream && target == CP_IB_BUFSZ && (word & 0x7fffffu))
1902 {
1903 fail("nested CP indirect buffers are unsupported");
1904 break;
1905 }
1906 }
1907 // Fetch/consume before executing the register side effect.
1908 // An IB initiator can then suspend the primary parser at the following word.
1909 if (stream)
1910 {
1911 --m_cp_ib_left;
1912 if (m_cp_ib_left && m_cp_ib_cursor > 0xfffffffbu)
1913 {
1914 fail("CP indirect buffer crosses the 32-bit address limit");
1915 break;
1916 }
1917 m_cp_ib_cursor += 4;
1918 }
1919 else
1920 {
1921 r(CP_RB_RPTR) = (r(CP_RB_RPTR) + 1) & (ring_words - 1);
1922 ++m_cp_since_update;
1923 }
1924 if (!packet.remaining)
1925 {
1926 packet.header = word;
1927 const unsigned kind = word >> 30;
1928 if (kind == 0 || (kind == 3 &&
1929 (((word >> 8) & 255) == 0x10 || ((word >> 8) & 255) == 0x94)))
1930 {
1931 packet.remaining = ((word >> 16) & 0x3fff) + 1;
1932 packet.reg = (word & 0x1fff) * 4;
1933 packet.body.clear();
1934 if (kind == 3 && ((word >> 8) & 255) == 0x94)
1935 packet.body.reserve(packet.remaining);
1936 }
1937 else if (kind != 2)
1938 fail("unsupported CP packet " + hex(word));
1939 }
1940 else
1941 {
1942 --packet.remaining;
1943 if ((packet.header >> 30) == 0)
1944 {
1945 const uint32_t target = alias(packet.reg);
1946 if (!(packet.header & (1u << 15)))
1947 packet.reg += 4;
1948 write32(target, word, 0xffffffffu);
1949 }
1950 else if (((packet.header >> 8) & 255) == 0x94)
1951 {
1952 packet.body.push_back(word);
1953 if (!packet.remaining)
1954 {
1955 cp_hostdata(packet.body);
1956 packet.body.clear();
1957 }
1958 }
1959 }
1960 if (!stream && !halted() && block <= 22 &&
1961 m_cp_since_update >= (2u << block) && !cp_writeback())
1962 break;
1963 }
1964 // The CP publishes RPTR at an empty ring as well as at block intervals.
1965 if (!halted() && m_cp_since_update && !m_cp_ib_left &&
1966 r(CP_RB_RPTR) == r(CP_RB_WPTR))
1967 cp_writeback();
1968 m_cp_running = false;
1969}

1972void CRV200::cp_hostdata(const std::vector<uint32_t>& body)
1973{
1974 if (body.empty() || m_pending.active)
1975 {
1976 fail("CP HOSTDATA_BLT missing settings or host upload already pending");
1977 return;
1978 }
1979 size_t cursor = 0;
1980 const uint32_t gui = body[cursor++];
1981 std::vector<std::pair<uint32_t, uint32_t>> settings;
1982 settings.emplace_back(DP_GUI_MASTER_CNTL, gui);
1983 const auto take = [&](uint32_t target) {
1984 if (cursor == body.size())
1985 return false;
1986 settings.emplace_back(target, body[cursor++]);
1987 return true;
1988 };
1989 bool valid = (!(gui & GMC_SRC_PITCH) || take(SRC_PITCH_OFFSET)) &&
1990 (!(gui & GMC_DST_PITCH) || take(DST_PITCH_OFFSET)) &&
1991 (!(gui & GMC_SRC_CLIP) || take(SRC_SC_BOTTOM_RIGHT)) &&
1992 (!(gui & GMC_DST_CLIP) || (take(SC_TOP_LEFT) && take(SC_BOTTOM_RIGHT)));
1993 if (valid)
1994 {
1995 switch ((gui >> 4) & 15)
1996 {
1997 case 0:
1998 valid = take(DP_BRUSH_BKGD_CLR) && take(DP_BRUSH_FRGD_CLR) &&
1999 take(BRUSH_DATA0) && take(BRUSH_DATA1);
2000 break;
2001 case 1:
2002 valid = take(DP_BRUSH_FRGD_CLR) && take(BRUSH_DATA0) && take(BRUSH_DATA1);
2003 break;
2004 case 13:
2005 valid = take(DP_BRUSH_FRGD_CLR);
2006 break;
2007 case 15:
2008 break;
2009 default:
2010 fail("unsupported CP HOSTDATA_BLT brush setup " + hex((gui >> 4) & 15));
2011 return;
2012 }
2013 }
2014 valid = valid && (!(gui & 0x80000000u) || take(BRUSH_Y_X)) &&
2015 take(DP_SRC_FRGD_CLR) && take(DP_SRC_BKGD_CLR);
2016 if (!valid || cursor == body.size())
2017 {
2018 fail("truncated CP HOSTDATA_BLT setup or missing bitmap");
2019 return;
2020 }
2021 const unsigned format = (gui >> 8) & 15, source = (gui >> 12) & 3;
2022 const uint32_t bpp = bytes_per_pixel(format);
2023 const bool mono = source < 2;
2024 if (!bpp || source == 2 || (!mono && bpp == 3) ||
2025 ((gui >> 24) & 7) != 3)
2026 {
2027 fail("unsupported CP HOSTDATA_BLT datatype or transport " + hex(gui));
2028 return;
2029 }
2030 struct Bitmap
2031 {
2032 uint32_t origin, extent, words;
2033 size_t data;
2034 };
2035 std::vector<Bitmap> bitmaps;
2036 while (cursor < body.size())
2037 {
2038 if (body.size() - cursor < 3)
2039 {
2040 fail("truncated CP HOSTDATA_BLT bitmap header");
2041 return;
2042 }
2043 const uint32_t origin = body[cursor++], extent = body[cursor++];
2044 const uint32_t words = body[cursor++] & 0x3fffu;
2045 const uint32_t width = extent & 0x3fff, height = (extent >> 16) & 0x3fff;
2046 const uint64_t bits = uint64_t(width) * height * (mono ? 1u : 8 * bpp);
2047 if (words != (bits + 31) / 32 || words > body.size() - cursor)
2048 {
2049 fail("CP HOSTDATA_BLT bitmap length disagrees with geometry or packet");
2050 return;
2051 }
2052 bitmaps.push_back({origin, extent, words, cursor});
2053 cursor += words;
2054 }
2055 for (const auto& setting : settings)
2056 write32(setting.first, setting.second, 0xffffffffu);
2057 for (const auto& bitmap : bitmaps)
2058 {
2059 if (halted())
2060 return;
2061 write32(DST_Y_X, bitmap.origin, 0xffffffffu);
2062 r(DST_HEIGHT_WIDTH) = bitmap.extent;
2063 // Packet geometry specifies a top-left origin.
2064 draw(bitmap.extent & 0x3fff, (bitmap.extent >> 16) & 0x3fff, true);
2065 for (uint32_t n = 0; n < bitmap.words && !halted(); ++n)
2066 host_word(body[bitmap.data + n], n + 1 == bitmap.words);
2067 if (!halted() && m_pending.active)
2068 {
2069 fail("CP HOSTDATA_BLT left an incomplete host upload");
2070 return;
2071 }
2072 }
2073}
