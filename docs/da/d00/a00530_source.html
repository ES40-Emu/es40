195 // =============================================================================
196 // Design:
197 //
198 // socket --> recv() --> raw buf --> cook IAC --> staging buffer --> FIFO
199 // | |
200 // | v
201 // every execute() guest reads
202 // (~20 ms) via ReadMem
203 //
204 // - recv() and IAC cooking happen every RECV_TICKS cycles (~200ms).
205 // Cooked data goes into the staging buffer.
206 // - drain_staging() runs every execute() cycle (~20ms) but only
207 // delivers one character when the guest has consumed the previous
208 // one (FIFO empty). This self-clocks to the emulated CPU speed,
209 // matching how a real UART works — the next character doesn't
210 // arrive until the line shifts it in, and the CPU is always fast
211 // enough to read it before then.
212 // - If staging still has data, recv() is skipped — TCP provides
213 // backpressure at the OS level.
214 // =============================================================================

288 // Port is claimed in the address map but presents itself as a missing UART:
289 // scratchpad writes don't stick, MSR/LSR read 0xff, IIR=0xff. KDCOM and
290 // SERIAL.SYS conclude "no UART here" and skip further probing/polling.

300 // healthy idle 16550 — TX shift register always empty (THRE/TSRE), modem

302 // are silently discarded; nothing ever arrives at RX. No socket is opened,

565u64 CSerial::ReadMem(int index, u64 address, int dsize)
566{
567 u8 d;
568
569 if (disabled)
570 return 0xffu; // missing-UART signature
571
572 std::lock_guard<std::recursive_mutex> lock(m_register_mutex);
573
574 switch (address)
575 {
576 case 0: // data buffer
577 if (state.bLCR & 0x80)
578 {
579 return state.bBRB_LSB;
580 }
581 else
582 {
583 if (state.rcvR != state.rcvW)
584 {
585 state.bRDR = state.rcvBuffer[state.rcvR];
586 state.rcvR++;
587 if (state.rcvR == FIFO_SIZE)
588 state.rcvR = 0;
589 TRC_DEV4("Read character %02x (%c) on serial port %d\n", state.bRDR,
590 printable(state.bRDR), state.iNumber);
591#if defined(DEBUG_SERIAL)
592 printf("Read character %02x (%c) on serial port %d\n", state.bRDR,
593 printable(state.bRDR), state.iNumber);
594#endif
595 }
596 else
597 {
598 TRC_DEV2("Read past FIFO on serial port %d\n", state.iNumber);
599#if defined(DEBUG_SERIAL)
600 printf("Read past FIFO on serial port %d\n", state.iNumber);
601#endif
602 }
603
604 // Received-data interrupt follows the FIFO — it may have just emptied.
605 eval_interrupts();
606 return state.bRDR;
607 }
608
609 case 1:
610 if (state.bLCR & 0x80)
611 {
612 return state.bBRB_MSB;
613 }
614 else
615 {
616 return state.bIER;
617 }
618
619 case 2: //interrupt cause
620 d = state.bIIR;
621 if (d == 0x02)
622 state.thre_pending = false; // reading IIR acknowledges THRE
623 eval_interrupts(); // recompute IIR and re-drive the IRQ line
624 return d;
625
626 case 3:
627 return state.bLCR;
628
629 case 4:
630 return state.bMCR;
631
632 case 5: //serialization state
633 if (state.rcvR != state.rcvW)
634 state.bLSR = 0x61; // THRE, TSRE, RxRD
635 else
636 state.bLSR = 0x60; // THRE, TSRE
637 return state.bLSR;
638
639 case 6:
640 return state.bMSR;
641
642 default:
643 return state.bSPR;
644 }
645}

647void CSerial::WriteMem(int index, u64 address, int dsize, u64 data)
648{
649 u8 d;
650 d = (u8)data;
651
652 if (disabled)
653 return; // ignore guest writes to a disabled port
654
655 std::unique_lock<std::recursive_mutex> lock(m_register_mutex);
656
657 switch (address)
658 {
659 case 0:
660 if (state.bLCR & 0x80) // divisor latch access bit set
661 {
662
663 // LSB of divisor latch
664 state.bBRB_LSB = d;
665 }
666 else
667 {
668 if (state.bMCR & 0x10)
669 {
670 // MCR.LOOP set: deliver TX byte to our own RX FIFO instead of the wire.
671 // Windows 2000 UART detection writes a sentinel here and expects to read it back via LSR.DR.
672 int nextW = state.rcvW + 1;
673 if (nextW == FIFO_SIZE)
674 nextW = 0;
675 if (nextW != state.rcvR)
676 {
677 state.rcvBuffer[state.rcvW] = (char)d;
678 state.rcvW = nextW;
679 }
680 }
681 else
682 {
683 // Socket output can block. Let host RX progress while the device
684 // bus still serializes guest accesses, then complete TX below.
685 lock.unlock();
686 write((const char*)&d, 1);
687 lock.lock();
688 }
689 TRC_DEV4("Write character %02x (%c) on serial port %d\n", d, printable(d),
690 state.iNumber);
691#if defined(DEBUG_SERIAL)
692 printf("Write character %02x (%c) on serial port %d\n", d, printable(d),
693 state.iNumber);
694#endif
695 // TX is instantaneous: the holding register is empty again, so
696 // re-arm THRE (delivered only if the THRE interrupt is enabled).
697 state.thre_pending = true;
698 eval_interrupts();
699 }
700 break;
701
702 case 1:
703 if (state.bLCR & 0x80) // divisor latch access bit set
704 {
705
706 // MSB of divisor latch
707 state.bBRB_MSB = d;
708 }
709 else
710 {
711 // Interrupt Enable Register. Enabling the THRE interrupt while the
712 // (instantly-drained) holding register is empty arms one THRE edge.
713 u8 prev_ier = state.bIER;
714 state.bIER = d;
715 if ((d & 0x02) && !(prev_ier & 0x02))
716 state.thre_pending = true;
717 eval_interrupts();
718 }
719 break;
720
721 case 2:
722 state.bFCR = d;
723 break;
724
725 case 3:
726 state.bLCR = d;
727 break;
728
729 case 4:
730 {
731 u8 prev_mcr = state.bMCR;
732 state.bMCR = d;
733 if (d & 0x10)
734 {
735 // MCR.LOOP set: mirror modem-control output bits into MSR status bits.
736 // PC16550D loopback: DTR->DSR, RTS->CTS, OUT1->RI, OUT2->DCD.
737 // Windows 2000 SERIAL.SYS uses this to confirm a 16550 by toggling MCR.OUT1
738 // and watching MSR.RI track it.
739 u8 status = 0;
740 if (d & 0x01) status |= 0x20; // DTR -> DSR
741 if (d & 0x02) status |= 0x10; // RTS -> CTS
742 if (d & 0x04) status |= 0x40; // OUT1 -> RI
743 if (d & 0x08) status |= 0x80; // OUT2 -> DCD
744 state.bMSR = status | (state.bMSR & 0x0f);
745 }
746 else
747 {
748 // Loopback released: present the default "modem connected" status the rest of the code assumes.
749 state.bMSR = (state.bMSR & 0x0f) | 0x30;
750 }
751 if (prev_mcr != d)
752 printf("SRL%d: MCR %02x -> %02x (LOOP=%d DTR=%d RTS=%d OUT1=%d OUT2=%d), MSR now %02x\n",
753 state.iNumber, prev_mcr, d,
754 (d >> 4) & 1, d & 1, (d >> 1) & 1, (d >> 2) & 1, (d >> 3) & 1,
755 state.bMSR);
756 break;
757 }
758
759 default:
760 state.bSPR = d;
761 }
762}

764void CSerial::eval_interrupts()
765{
766 if (disabled)
767 return;
768
769 std::lock_guard<std::recursive_mutex> lock(m_register_mutex);
770
771 // RX-available is a level (follows the receive FIFO); THR-empty is edge-
772 // latched in thre_pending (set on THR write / THRE enable, cleared on IIR
773 // read) because es40's instant TX would make a plain THRE level storm.
774 u8 iir = 0x01; // no interrupt
775 if ((state.bIER & 0x01) && (state.rcvR != state.rcvW))
776 iir = 0x04; // received data available
777 else if ((state.bIER & 0x02) && state.thre_pending)
778 iir = 0x02; // transmitter holding register empty
779 state.bIIR = iir;
780
781 // Drive IRQ4 (serial0) / IRQ3 (serial1) as a level following the cause;
782 // pic_set_line() makes one edge per rising transition, retracting on fall.
783 // Keep the UART lock until publication so a newer cause cannot be
784 // followed by a stale IRQ level from another thread.
785 theAli->pic_set_line(0, 4 - state.iNumber, (iir & 0x01) == 0);
786}

805int CSerial::receive(const char* data, int dsize)
806{
807 std::lock_guard<std::recursive_mutex> lock(m_register_mutex);
808 int consumed = 0;
809
810 while (dsize)
811 {
812 int nextW = state.rcvW + 1;
813 if (nextW == FIFO_SIZE)
814 nextW = 0;
815
816 if (nextW == state.rcvR)
817 break; // FIFO full
818
819 state.rcvBuffer[state.rcvW] = *data;
820 state.rcvW = nextW;
821 data++;
822 dsize--;
823 consumed++;
824 eval_interrupts();
825 }
826
827 return consumed;
828}

834void CSerial::drain_staging()
835{
836 if (stageLen == 0)
837 return;
838
839 std::lock_guard<std::recursive_mutex> lock(m_register_mutex);
840
841 // Only deliver when the guest has consumed the previous character.
842 // This self-clocks to the emulated CPU speed, just like a real
843 // UART where the next character doesn't arrive until the baud
844 // rate shifts it in — and the CPU is always fast enough to read
845 // it before then.
846 if (state.rcvR != state.rcvW)
847 return; // previous character not yet read
848
849 // Deliver exactly one character
850 receive(stageBuf, 1);
851 stageLen--;
852 if (stageLen > 0)
853 memmove(stageBuf, stageBuf + 1, stageLen);
854}

899void CSerial::serial_menu()
900{
901 fd_set readset;
902 unsigned char buffer[FIFO_SIZE + 1];
903 int size;
904 struct timeval tv;
905 bool exitLoop = false;
906 bool waitForChoice = false;
907
908 cSystem->stop_threads();
909
910 write_cstr("\r\n<BREAK> received. What do you want to do?\r\n");

912#if defined(IDB)
913 write_cstr(" 1. End run\r\n");

1023// execute — Main loop body, called every ~20ms from the serial thread.

1025void CSerial::execute()
1026{
1027 // Resume an interrupted reconnect.
1028 if (connectSocket == INVALID_SOCKET)
1029 {
1030 WaitForConnection();
1031 return;
1032 }
1033 fd_set readset;
1034 unsigned char raw[FIFO_SIZE + 16];
1035 unsigned char cbuffer[FIFO_SIZE + 1];
1036 unsigned char* b;
1037 unsigned char* c;
1038 unsigned char* end;
1039 int size;
1040 struct timeval tv;
1041
1042 drain_staging();
1043
1044 // ------------------------------------------------------------------
1045 // Every RECV_TICKS cycles: read from the socket and cook telnet out
1046 // ------------------------------------------------------------------
1047 state.serial_cycles++;
1048 if (state.serial_cycles >= RECV_TICKS)
1049 {
1050 // If staging buffer still has data, don't pull more off the socket.
1051 // TCP backpressure holds the rest at the OS level.
1052 if (stageLen > 0)
1053 {
1054 state.serial_cycles = 0;
1055 eval_interrupts();
1056 return;
1057 }
1058
1059 FD_ZERO(&readset);
1060 FD_SET(connectSocket, &readset);
1061 tv.tv_sec = 0;
1062 tv.tv_usec = 0;
1063#if defined(_WIN32)
1064 if (select(0, &readset, NULL, NULL, &tv) > 0)
1065#else
1066 if (select((int)connectSocket + 1, &readset, NULL, NULL, &tv) > 0)
1067#endif
1068 {
1069 unsigned char* recv_buf = raw + iac_carry_len;
1070
1071#if defined(_WIN32) || defined(__VMS)
1072 size = recv(connectSocket, (char*)recv_buf, FIFO_SIZE, 0);
1073#else
1074 size = read(connectSocket, recv_buf, FIFO_SIZE);
1075#endif
1076
1077 extern int got_sigint;
1078 if (size <= 0 && !got_sigint)
1079 {
1080 printf("%%SRL-W-DISCONNECT: Write socket closed on other end for serial port %d.\n",
1081 state.iNumber);
1082 printf("-SRL-I-WAITFOR: Waiting for a new connection on port %d.\n",
1083 listenPort);
1084 WaitForConnection();
1085 return;
1086 }
1087
1088 if (size <= 0)
1089 {
1090 state.serial_cycles = 0;
1091 eval_interrupts();
1092 return;
1093 }
1094
1095 // Prepend carryover from partial telnet sequence
1096 if (iac_carry_len > 0)
1097 {
1098 memcpy(raw, iac_carry, iac_carry_len);
1099 size += iac_carry_len;
1100 iac_carry_len = 0;
1101 }
1102
1103 // raw_mode: skip telnet IAC cooking AND the staging buffer's baud-rate
1104 // drip-feed. The byte stream must stay 8-bit clean and flow at line
1105 // rate for windbg/kgdb, which sends arbitrary binary (including 0xff)
1106 // in KD packets and times out if delivery is throttled.
1107 if (raw_mode)
1108 {
1109 int delivered = receive((const char*)raw, (int)size);
1110 if (delivered < (int)size)
1111 printf("%%SRL-W-RAWFULL: raw_mode FIFO full on port %d, dropped %d byte(s)\n",
1112 state.iNumber, (int)size - delivered);
1113 state.serial_cycles = 0;
1114 eval_interrupts();
1115 return;
1116 }
1117
1118 b = raw;
1119 c = cbuffer;
1120 end = raw + size;
1121
1122 while (b < end)
1123 {
1124 // ----- Subnegotiation scan (SB ... IAC SE) -----
1125 if (in_subneg)
1126 {
1127 while (b < end)
1128 {
1129 if (*b == IAC)
1130 {
1131 if (b + 1 >= end)
1132 {
1133 iac_carry[0] = IAC;
1134 iac_carry_len = 1;
1135 b = end;
1136 break;
1137 }
1138 if (*(b + 1) == SE)
1139 {
1140 b += 2;
1141 in_subneg = false;
1142 break;
1143 }
1144 else if (*(b + 1) == IAC)
1145 {
1146 b += 2;
1147 }
1148 else
1149 {
1150 b += 2;
1151 }
1152 }
1153 else
1154 {
1155 b++;
1156 }
1157 }
1158 continue;
1159 }
1160
1161 // ----- Telnet IAC with bounds-safe lookahead -----
1162 if (*b == IAC)
1163 {
1164 ssize_t remaining = end - b;
1165
1166 if (remaining < 2)
1167 {
1168 iac_carry[0] = IAC;
1169 iac_carry_len = 1;
1170 b = end;
1171 break;
1172 }
1173
1174 unsigned char cmd = *(b + 1);
1175
1176 if (cmd == IAC)
1177 {
1178 *c++ = 0xFF;
1179 b += 2;
1180 continue;
1181 }
1182 else if (cmd >= WILL && cmd <= DONT)
1183 {
1184 if (remaining < 3)
1185 {
1186 for (ssize_t i = 0; i < remaining; i++)
1187 iac_carry[i] = b[i];
1188 iac_carry_len = (int)remaining;
1189 b = end;
1190 break;
1191 }
1192 b += 3;
1193 continue;
1194 }
1195 else if (cmd == SB)
1196 {
1197 in_subneg = true;
1198 b += 2;
1199 continue;
1200 }
1201 else if (cmd == BREAK)
1202 {
1203 breakHit = true;
1204 b += 2;
1205 continue;
1206 }
1207 else if (cmd == AYT)
1208 {
1209 b += 2;
1210 continue;
1211 }
1212 else
1213 {
1214 b += 2;
1215 continue;
1216 }
1217 }
1218
1219 // ----- Normal data byte -----
1220 *c = *b;
1221 c++;
1222 b++;
1223 }
1224
1225 // Move cooked data into the staging buffer.
1226 // drain_staging() will drip-feed it into the FIFO at baud rate
1227 // on subsequent execute() cycles.
1228 if (c > cbuffer)
1229 {
1230 int cooked_len = (int)(c - cbuffer);
1231
1232 if (cooked_len <= (int)(STAGE_SIZE - stageLen))
1233 {
1234 memcpy(stageBuf + stageLen, cbuffer, cooked_len);
1235 stageLen += cooked_len;
1236 }
1237 else
1238 {
1239 // Staging buffer full — this means the guest is severely
1240 // behind. Copy what fits; the rest is lost, but this
1241 // should only happen under extreme conditions.
1242 int space = STAGE_SIZE - stageLen;
1243 if (space > 0)
1244 {
1245 memcpy(stageBuf + stageLen, cbuffer, space);
1246 stageLen += space;
1247 }
1248#if defined(DEBUG_SERIAL)
1249 printf("%%SRL-W-STAGEDROP: staging buffer full on port %d, "
1250 "dropped %d bytes\n", state.iNumber, cooked_len - space);
1251#endif
1252 }
1253 }
1254 }
1255
1256 state.serial_cycles = 0;
1257 }
1258
1259 eval_interrupts();
1260}

1342{
1343 close_connection();

1345 return;
1346 const char* telnet_options = "%c%c%c";
1347 char buffer[8];

1368 nargv = strchr(nargv + 1, '\"');
1369 if (nargv)
1370 nargv = strchr(nargv, ' ');
1371 if (!nargv)
1372 break;
1373 *nargv++ = '\0';
1374 i++;
1375 argv[i] = NULL;
1376 }
1377
1378 argv[i + 1] = NULL;
1379 strcpy(s2, argv[0]);
1380 nargv = s2;

1382 {
1383 nargv++;

1387 //printf("%s: Starting %s\n", devid_string,nargv);
1388#if defined(_WIN32)

1393 int status;

1395 {
1396 execvp(argv[0], argv);
1397 FAILURE_1(Runtime, "Exec of '%s' failed.\n", argv[0]);
1398 }
1399 else
1400 {
1401 sleep(1); // give it a chance to start up.
1402 waitpid(child, &status, WNOHANG); // reap it, if needed.
1403 if (kill(child, 0) < 0)
1404 { // uh oh, no kiddo.
1405 FAILURE_1(Runtime, "Exec of '%s' has failed.\n", argv[0]);
1406 }
1407 }
1408#endif
1409 }
1410#endif
1411 // Poll so stop_threads() can interrupt the wait.
1412 fd_set readset;
1413 struct timeval tv;
1414 while (connectSocket == INVALID_SOCKET)
1415 {
1416 if (StopThread)
1417 return;
1418 FD_ZERO(&readset);
1419#if !defined(_WIN32)
1420 int highest_socket = 0;
1421#endif
1422 for (int64_t listener : listenSockets)
1423 {
1424 FD_SET(listener, &readset);
1425#if !defined(_WIN32)
1426 if (listener > highest_socket)
1427 highest_socket = (int)listener;
1428#endif
1429 }
1430 tv.tv_sec = 0;
1431 tv.tv_usec = 100000;
1432#if defined(_WIN32)
1433 const int ready = select(0, &readset, NULL, NULL, &tv);
1434#else
1435 const int ready = select(highest_socket + 1, &readset, NULL, NULL, &tv);
1436#endif
1437 if (ready < 0)
1438 {
1439#if defined(_WIN32)
1440 const int error = WSAGetLastError();
1441 if (error == WSAEINTR)
1442#else
1443 const int error = errno;
1444 if (error == EINTR)
1445#endif
1446 continue;
1447 FAILURE_2(Runtime, "%s: serial listener select failed (error %d)", devid_string, error);
1448 }
1449 if (ready == 0 || StopThread)
1450 continue;
1451 for (size_t i = 0; i < listenSockets.size(); ++i)
1452 {
1453 if (!FD_ISSET(listenSockets[i], &readset))
1454 continue;
1455 connectSocket = accept(listenSockets[i], NULL, NULL);
1456 if (connectSocket == INVALID_SOCKET)
1457 {
1458#if defined(_WIN32)
1459 const int error = WSAGetLastError();
1460 if (error == WSAEINTR || error == WSAEWOULDBLOCK ||
1461 error == WSAECONNRESET || error == WSAECONNABORTED)
1462 continue;
1463#else
1464 const int error = errno;
1465 if (error == EINTR || error == EAGAIN || error == EWOULDBLOCK || error == ECONNABORTED)
1466 continue;
1467#if defined(__linux__)
1468 if (error == ENETDOWN || error == EPROTO || error == ENOPROTOOPT ||
1469 error == EHOSTDOWN || error == ENONET || error == EHOSTUNREACH ||
1470 error == EOPNOTSUPP || error == ENETUNREACH)
1471 continue;
1472#endif
1473#endif
1474 FAILURE(Runtime, std::string(devid_string) + ": serial address " + listenAddresses[i] +
1475 " accept failed (error " + std::to_string(error) + ")");
1476 }
1477#if !defined(_WIN32)
1478 if (connectSocket >= FD_SETSIZE)
1479 {
1480 close_connection();
1481 FAILURE(Runtime, "Accepted serial socket exceeds FD_SETSIZE");
1482 }
1483#endif
1484#if defined(_WIN32)
1485 u_long nonblocking = 0;
1486 const int blocking_error = ioctlsocket(connectSocket, FIONBIO, &nonblocking);
1487#else
1488 int nonblocking = 0;
1489 const int blocking_error = ioctl(connectSocket, FIONBIO, &nonblocking);
1490#endif
1491 if (blocking_error != 0)
1492 {
1493 close_connection();
1494 FAILURE(Runtime, "Cannot set blocking mode on serial connection");
1495 }
1496 break;
1497 }
1498 }
1499
1500 iac_carry_len = 0;
1501 in_subneg = false;
1502 stageLen = 0;
1503
1504 state.serial_cycles = 0;
1505
1506
1507 if (state.iNumber != 1 && !raw_mode) // skip for kgdb-default port and any raw-mode port
1508 {
1509 // Send some control characters to the telnet client to handle
1510 // character-at-a-time mode.
1511 sprintf(buffer, telnet_options, IAC, DO, TELOPT_ECHO);
1512 write_cstr(buffer);

1515 write_cstr(buffer);
1516

1521 write_cstr(buffer);
1522

1524 write_cstr(buffer);

1527 unsigned char opt_do_binary[3] = { IAC, DO, TELOPT_BINARY }; // tell client to send raw
1528 unsigned char opt_will_binary[3] = { IAC, WILL, TELOPT_BINARY }; // tell client to receive raw
1529 write((const char*)opt_do_binary, 3);
1530 write((const char*)opt_will_binary, 3);
