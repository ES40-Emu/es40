58static uint32_t powerstorm_setting(
59 CConfigurator* cfg, const char* key, uint32_t def,
60 uint32_t max = 0xffffffff)
61{
62 const char* text = cfg->get_text_value(key, nullptr);
63 if (!text || !*text)
64 return def;
65 char* end = nullptr;
66 errno = 0;
67 const unsigned long long n = std::strtoull(text, &end, 0);
68 if (errno || end == text || *end || text[0] == '-' || n > max)
69 throw std::runtime_error(
70 std::string("Invalid powerstorm setting: ") + key);
71 return static_cast<uint32_t>(n);
72}

74static void powerstorm_write_u32(FILE* f, uint32_t v)
75{
76 uint8_t b[4];
77 for (unsigned i = 0; i < 4; ++i)
78 b[i] = uint8_t(v >> (8 * i));
79 if (fwrite(b, 1, 4, f) != 4)
80 throw std::runtime_error("Snapshot write failed");
81}

83static uint32_t powerstorm_read_u32(FILE* f)
84{
85 uint8_t b[4];
86 if (fread(b, 1, 4, f) != 4)
87 throw std::runtime_error("Snapshot read failed");
88 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
89 (uint32_t(b[3]) << 24);
90}

92static bool powerstorm_vga_window_offset(u32& address, unsigned selection)
93{
94 switch (selection & 3)
95 {
96 case 1:
97 return address < 0x10000;
98 case 2:
99 if (address < 0x10000 || address >= 0x18000)
100 return false;
101 address -= 0x10000;
102 break;
103 case 3:
104 if (address < 0x18000 || address >= 0x20000)
105 return false;
106 address -= 0x18000;
107 break;
108 default:
109 return address < 0x20000;
110 }
111 return true;
112}

116void CPowerStorm3xx::crtc_map(address_map& map)
117{
118 unimplemented_map(map, "VGA CRTC register");
119 vga_device::crtc_map(map);
120 // ROM 0x3641..0x3649 stores its I/O base here; 0x3657..0x366d reads it back.
121 map(0x1c, 0x1d)
122 .lrw8(
123 NAME([this](offs_t offset) {
124 return u8(m_crtc_io_base >> (offset * 8));
125 }),
126 NAME([this](offs_t offset, u8 data) {
127 const unsigned shift = offset * 8;
128 m_crtc_io_base =
129 u16((m_crtc_io_base & ~(0xffu << shift)) |
130 (u32(data) << shift));
131 }));
132 // Reject S3 identification/unlock probes without extension state.
133 map(0x30, 0x30).lr8(NAME([](offs_t) { return u8(0xff); }));
134 map(0x38, 0x39).lrw8(
135 NAME([](offs_t) { return u8(0xff); }),
136 NAME([](offs_t, u8) {}));
137}

139void CPowerStorm3xx::sequencer_map(address_map& map)
140{
141 unimplemented_map(map, "VGA sequencer register");
142 // These frontends mask stored bytes, including direct snapshot writes.
143 // Limit the canonical map to SR00-04 so unknown indices keep diagnostics.
144 auto standard = std::make_shared<address_map>(5);
145 vga_device::sequencer_map(*standard);
146 map(0x00, 0x04).lrw8(
147 NAME([standard](offs_t index) {
148 return standard->read_byte(index);
149 }),
150 NAME(([this, standard](offs_t index, u8 data) {
151 static constexpr u8 masks[] = {0x03, 0x3f, 0x0f, 0x3f, 0x0e};
152 vga.sequencer.data[index] = data & masks[index];
153 standard->write_byte(index, data);
154 })));
155}

163void CPowerStorm3xx::attribute_map(address_map& map)
164{
165 unimplemented_map(map, "VGA attribute register");
166 vga_device::attribute_map(map);
167 // Palette reads remain available when PAS disables CPU writes.
168 map(0x20, 0x2f).lr8(NAME([this](offs_t offset) {
169 return vga.attribute.data[offset];
170 }));
171 // These frontends retain mode-control bits 6/7 and mask reserved bit 4.
172 map(0x10, 0x10).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
173 vga.attribute.data[0x10] = data & 0xef;

175 // Color select does not change the pending horizontal panning value.
176 map(0x14, 0x14).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
177 vga.attribute.data[0x14] = data & 0xf;
178 }));

198u8 CPowerStorm3xx::mem_r(offs_t address)
199{
200 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
201 return 0xff;
202 unsigned selected = vga.gc.read_map_sel & 3;
203 if (vga.sequencer.data[4] & 8)
204 {
205 // Keep packed chain-four bytes in the layout consumed by vga_device scanout.
206 for (unsigned plane = 0; plane < 4; ++plane)
207 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
208 selected = address & 3;
209 }
210 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
211 {
212 // vga_device text stores character/attribute pairs together; fonts stay planar.
213 vga.gc.latch[0] = vga.memory[address & ~1u];
214 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
215 for (unsigned plane = 2; plane < 4; ++plane)
216 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
217 selected = (vga.gc.read_map_sel & 2) | (address & 1);
218 }
219 else
220 {
221 for (unsigned plane = 0; plane < 4; ++plane)
222 vga.gc.latch[plane] =
223 vga.memory[(address & 0xffff) + plane * 0x10000];
224 }
225 if (!vga.gc.read_mode)
226 return vga.gc.latch[selected];
227 u8 result = 0xff;
228 for (unsigned plane = 0; plane < 4; ++plane)
229 if (vga.gc.color_dont_care & (1 << plane))
230 result &=
231 u8(vga.gc.latch[plane] ^
232 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
233 return result;
234}

236void CPowerStorm3xx::mem_w(offs_t address, u8 data)
237{
238 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
239 return;
240 if (vga.sequencer.data[4] & 8)
241 {
242 if (vga.sequencer.map_mask & (1 << (address & 3)))
243 vga.memory[address] = vga_latch_write(address & 3, data);
244 return;
245 }
246 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
247 {
248 const unsigned parity = address & 1;
249 if (vga.sequencer.map_mask & (1 << parity))
250 vga.memory[address] = vga_latch_write(parity, data);
251 if (vga.sequencer.map_mask & (1 << (parity + 2)))
252 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
253 vga_latch_write(parity + 2, data);
254 return;
255 }
256 for (unsigned plane = 0; plane < 4; ++plane)
257 if (vga.sequencer.map_mask & (1 << plane))
258 vga.memory[(address & 0xffff) + plane * 0x10000] =
259 vga_latch_write(plane, data);
260}

281CRealImage2100::Frame CPowerStorm3xx::render_frame(CRealImage2100::Frame frame)
282{
283 std::lock_guard<std::recursive_mutex> guard(
284 cSystem->get_device_bus_mutex());
285 if (m_pause.load())
286 return {};
287 // The native display path remains active when the driver blanks legacy VGA.
288 if (m_realimage.native_display())
289 return m_realimage.scanout(std::move(frame));
290 if (!m_board_vga_enabled || !m_vga_enable ||
291 !vga.crtc.sync_en || (vga.sequencer.data[0] & 3) != 3 ||
292 (vga.sequencer.data[1] & 0x20) || !(vga.attribute.index & 0x20) ||
293 !vga.crtc.maximum_scan_line)
294 return {};
295 frame.width = (vga.crtc.horz_disp_end + 1) *
296 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
297 frame.height = vga.crtc.vert_disp_end + 1;
298 if (!frame.width || frame.width > 4096 || frame.height > 1024)
299 return {};
300 // Allow vga_device double-scan and one extra character row below the visible area.
301 m_render_bitmap.resize(frame.width, frame.height * 2 + 64);
302 screen().set_visible_area(frame.width, frame.height);
303 screen().tick_frame();
304 vga.crtc.start_addr = latch_start_addr();
305 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
306 palette_update();
307 svga_device::screen_update(
308 screen(), m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
309 frame.argb.assign(
310 m_render_bitmap.raw(),
311 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
312 return frame;
313}

337void CPowerStorm3xx::run()
338{
339 try
340 {
341 m_output.initialize(X_TILESIZE, Y_TILESIZE);
342 // Only this display thread owns the reusable frame.
343 CRealImage2100::Frame frame;
344 while (!m_stop.load())
345 {
346 {
347 CPowerStormDisplayLock guard(m_output.display());
348 m_output.display().handle_events();
349 }
350 if (m_pause.load())
351 {
352 m_pause_ack.store(true);
353 CThread::sleep(10);
354 continue;
355 }
356 m_pause_ack.store(false);
357 update(frame);
358 CThread::sleep(20);
359 }
360 }
361 catch (const CException& e)
362 {
363 printf(
364 "%s: Exception in display thread: %s.\n",
365 devid_string,
366 e.displayText().c_str());
367 m_worker_failed.store(true);

370 {
371 printf(
372 "%s: Exception in display thread: %s.\n", devid_string, e.what());
373 m_worker_failed.store(true);

379std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_data() const
380{
381 std::array<u32, 64> data{};
382 data[0] = u32(vendor) | (u32(device) << 16);
383 data[2] = (class_code << 8) | revision;
384 data[6] = 0x00000001; // BAR2: native index/data I/O pair
385 data[11] = u32(subsystem_vendor) | (u32(subsystem_device) << 16);
386 data[15] = 0x000001ff; // INTA#, interrupt line unassigned
387 return data;
388}

390std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_mask(u32 rom_size) const
391{
392 std::array<u32, 64> mask{};
393 mask[1] = 0x00000147; // I/O, memory, bus master, parity response, SERR
394 mask[3] = 0x0000ffff; // cache-line size and latency timer
395 mask[4] = ~(bar0_size - 1);
396 mask[5] = ~(bar1_size - 1);
397 mask[6] = ~(IOBARSize - 1);
398 if (rom_size)
399 mask[12] = (~(rom_size - 1) & 0xfffff800u) | 1; // ROM decode and enable
400 mask[15] = 0xff; // interrupt line only; pin fixed
401 return mask;
402}

404CPowerStorm3xx::OptionROM CPowerStorm3xx::OptionROM::parse(
405 const std::vector<u8>& raw, const std::string& layout)
406{
407 auto fail = [](const char* message) {
408 throw std::invalid_argument(message);
409 };
410 const size_t n = raw.size();
411 if (n < 2048 || n > 1024 * 1024)
412 fail("ROM must be 2 KiB to 1 MiB");
413 if (layout != "exact" && layout != "pad_ff")
414 fail("rom_layout must be exact or pad_ff");
415 if (layout == "exact" && (n & (n - 1)))
416 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
417 auto word = [&](size_t p) -> u16 {
418 if (p > n - 2)
419 fail("ROM field out of bounds");
420 return u16(raw[p] | (u16(raw[p + 1]) << 8));
421 };
422 size_t at = 0;
423 bool last = false;
424 OptionROM rom;
425 for (unsigned images = 0; images < 32 && !last; ++images)
426 {
427 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
428 fail("ROM missing 55 AA signature");
429 const size_t p = at + word(at + 0x18);
430 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
431 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
432 fail("ROM missing/bounded PCIR structure");
433 if (word(p + 4) != 0x10ba || word(p + 6) != 0x0304)
434 fail("ROM PCIR identity is not 10BA:0304 (PowerStorm 300/350)");
435 const size_t length = size_t(word(p + 0x10)) * 512;
436 if (!length || length > n - at || p + 0x18 > at + length)
437 fail("Invalid option-ROM image length");
438 if (word(p + 0x0a) < 0x18 || word(p + 0x0a) > at + length - p)
439 fail("Invalid PCIR structure length");
440 if (raw[p + 0x14] != 0)
441 fail("Only an x86 option-ROM image is supported");
442 if (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length)
443 fail("Invalid x86 initialization image length");
444 unsigned sum = 0;
445 for (size_t i = at; i < at + length; ++i)
446 sum += raw[i];
447 if (sum & 255)
448 fail("Option-ROM checksum failed");
449 rom.pcir_class = u32(raw[p + 13]) | (u32(raw[p + 14]) << 8) |
450 (u32(raw[p + 15]) << 16);
451 last = (raw[p + 0x15] & 0x80) != 0;
452 at += length;
453 }
454 if (!last)
455 fail("Unterminated option-ROM image chain");
456 // Keep the supplied image unmodified; only an erased tail may follow it.
457 for (size_t i = at; i < n; ++i)
458 if (raw[i] != 0xff)
459 fail("Unexpected data after last option-ROM image");
460 rom.bytes = raw;
461 rom.file_size = u32(n);
462 rom.image_bytes = u32(at);
463 if (layout == "pad_ff")
464 {
465 size_t decoded = 2048;
466 while (decoded < n)
467 decoded *= 2;
468 rom.bytes.resize(decoded, 0xff);
469 }
470 return rom;
471}

501{

503 return;
504 CConfigurator* cfg = myCfg;
505 init_maps();
506 add_legacy_io(LegacyIO, 0x3b0, 0x30);
507 add_legacy_mem(LegacyMemory, 0xa0000, 0x20000);

510 throw std::runtime_error("powerstorm model must be 300 or 350");

512 m_realimage.configure_texture_memory(m_model == 350 ?

519 m_profile.subsystem_vendor = u16(powerstorm_setting(
520 cfg, "subsystem_vendor", compaq ? 0x0e11 : 0x1011, 65535));
521 m_profile.subsystem_device = u16(powerstorm_setting(
522 cfg, "subsystem_device", compaq ? 0x4d31 : 0x4d35, 65535));

524 powerstorm_setting(cfg, "bar0_size", m_profile.bar0_size, 0x8000000);

526 powerstorm_setting(cfg, "bar1_size", 0x2000000, 0x8000000);
527 for (const u32 size : {m_profile.bar0_size, m_profile.bar1_size})
528 if (size < 0x1000000 || (size & (size - 1)))
529 throw std::runtime_error(
530 "powerstorm BAR sizes must be powers of two from 16 to 128 "
531 "MiB");
532 m_frame_file = cfg->get_text_value("frame_file", "");

539 throw std::runtime_error("Cannot open powerstorm ROM: " + romname);

542 throw std::runtime_error("powerstorm ROM must be 2 KiB to 1 MiB");

546 reinterpret_cast<char*>(raw.data()),
547 static_cast<std::streamsize>(raw.size()));
548 if (!f)
549 throw std::runtime_error("Cannot read full powerstorm ROM");

552#ifdef DEBUG_VGA

554 "%s: ROM %u bytes, image %u bytes, PCIR class %06x (PCI class "

563 throw std::runtime_error("rom_layout must be exact or pad_ff");

566 {
567 if (std::filesystem::exists(tracename))
568 throw std::runtime_error(
569 "Refusing to overwrite existing powerstorm trace: " +
570 tracename);
571 m_trace.open(tracename);
572 if (!m_trace)
573 throw std::runtime_error("Cannot create powerstorm trace");
574 m_trace << "# PowerStorm PCI/BAR/VGA accesses; widths are bits. No "
575 "guest-PC hook.\n";
576 }
577 const auto profile_data = m_profile.config_data(),
578 profile_mask = m_profile.config_mask(u32(m_rom.size()));
579 u32 data[64]{}, mask[64]{};
580 for (unsigned i = 0; i < 64; ++i)
581 {
582 data[i] = profile_data[i];
583 mask[i] = profile_mask[i];
584 }
585 add_function(0, data, mask);
586 m_realimage.set_irq_callback([this](bool level) {

588 do_pci_interrupt(0, level);
589 });
590 m_realimage.set_diagnostic_callback(
591 [this](const CRealImage2100::Diagnostic& d) {

708{
709 if (!m_initialized || !m_board_vga_enabled)
710 return false;
711 const u32 command = endian_32(pci_state.config_data[0][1]);

717bool CPowerStorm3xx::vga_port_enabled(u32 port) const noexcept
718{
719 if (!m_board_vga_enabled)
720 return false;
721 // The VGA enable register stays addressable while the frontend is off.
722 if (port == 0x3c3)
723 return true;
724 if (!m_vga_enable)
725 return false;
726 if ((port == 0x3b4 || port == 0x3b5 || port == 0x3ba) && !m_ioas)
727 return true;
728 if ((port == 0x3d4 || port == 0x3d5 || port == 0x3da) && m_ioas)
729 return true;
730 return (port >= 0x3c0 && port <= 0x3ca) || port == 0x3cc || port == 0x3ce ||
731 port == 0x3cf;
732}
733
734bool CPowerStorm3xx::decodes_memory_access(
735 int index, u64 address, int dsize, bool write) const noexcept
736{
737 if (index >= PCI_RANGE_BASE)
738 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
739 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
740 return false;
741 const unsigned bytes = unsigned(dsize) / 8;
742 if (index == LegacyIO)
743 {
744 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
745 return false;
746 // Claim only accesses whose every byte decodes to this device.
747 for (unsigned i = 0; i < bytes; ++i)
748 if (!vga_port_enabled(0x3b0 + u32(address) + i))
749 return false;
750 return true;
751 }
752 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
753 bytes > 0x20000 - address)
754 return false;
755 switch (vga.gc.memory_map_sel & 3)
756 {
757 case 1:
758 return address + bytes <= 0x10000;
759 case 2:
760 return address >= 0x10000 && address + bytes <= 0x18000;
761 case 3:
762 return address >= 0x18000;
763 default:
764 return true;
765 }
766}

859bool CPowerStorm3xx::dma_ram_range(u32 start, size_t length)
860{
861 if (length > 0x100000000ull - start)
862 return false;
863 for (size_t done = 0; done < length;)
864 {
865 const u64 physical = cSystem->PCI_Phys(myPCIBus, start + u32(done));
866 const size_t chunk = std::min(length - done, size_t(8192 - (physical & 8191)));
867 if (!cSystem->PtrToMem(physical) || !cSystem->PtrToMem(physical + chunk - 1))
868 return false;
869 done += chunk;
870 }
871 return true;
872}

874bool CPowerStorm3xx::dma_read(
875 u32 address, uint8_t* destination, size_t count, u32 completion)
876{
877 if (!(config_read(0, 4, 16) & 4) || !destination || !count ||
878 count > CRealImage2100::DMACommandListMaxBytes ||
879 ((address | completion | count) & 3) ||
880 !dma_ram_range(address, count) || !dma_ram_range(completion, 4))
881 return false;
882 do_pci_read(address, destination, 1, count);
883 if (m_trace)
884 {
885 m_trace << "# DMA read " << std::hex << address << " bytes=" << count
886 << " completion=" << completion << '\n';
887 for (size_t offset = 0; offset < count; offset += 64)
888 {
889 m_trace << "# DMA payload " << offset;
890 for (size_t i = offset; i < std::min(offset + 64, count); i += 4)
891 {
892 const uint32_t value = uint32_t(destination[i]) |
893 (uint32_t(destination[i + 1]) << 8) |
894 (uint32_t(destination[i + 2]) << 16) |
895 (uint32_t(destination[i + 3]) << 24);
896 m_trace << ' ' << value;
897 }
898 m_trace << '\n';
899 }
900 m_trace << std::dec;
901 }
902 return true;
903}

906{

908 !dma_ram_range(completion, 4))
909 return false;
910 uint8_t done[4]{};
911 do_pci_write(completion, done, 1, sizeof(done));
912 if (m_trace)
913 m_trace << "# DMA complete " << std::hex << completion << std::dec << '\n';
914 return true;
915}

919{

1061std::vector<u8> CPowerStorm3xx::save_vga_state() const
1062{
1063 std::vector<u8> bytes{1};
1064 for (unsigned i = 0; i < 25; ++i)
1065 bytes.push_back(m_crtc_map.read_byte(i));
1066 for (unsigned i = 0; i < 5; ++i)
1067 bytes.push_back(m_seq_map.read_byte(i));
1068 for (unsigned i = 0; i < 9; ++i)
1069 bytes.push_back(m_gc_map.read_byte(i));
1070 for (unsigned i = 0; i < 21; ++i)
1071 bytes.push_back(m_atc_map.read_byte(i));
1072 bytes.insert(
1073 bytes.end(),
1074 {vga.crtc.index,
1075 vga.sequencer.index,
1076 vga.gc.index,
1077 vga.attribute.index,
1078 vga.miscellaneous_output,
1079 vga.feature_control,
1080 m_vga_enable,
1081 vga.dac.mask,
1082 u8(vga.attribute.state),
1083 u8(vga.dac.read),
1084 u8(vga.dac.state)});
1085 for (const u8 value : vga.dac.loading)
1086 bytes.push_back(value);
1087 for (const u8 value : vga.gc.latch)
1088 bytes.push_back(value);
1089 for (unsigned i = 0; i < 4; ++i)
1090 bytes.push_back(u8(vga.crtc.start_addr >> (8 * i)));
1091 bytes.push_back(vga.attribute.pel_shift);
1092 bytes.push_back(vga.cursor.visible);
1093 bytes.push_back(vga.crtc.irq_latch);
1094 for (unsigned i = 0; i < 4; ++i)
1095 bytes.push_back(u8(m_vga_scanline >> (8 * i)));
1096 for (unsigned i = 0; i < 4; ++i)
1097 bytes.push_back(u8(m_vga_status_phase >> (8 * i)));
1098 bytes.push_back(u8(m_crtc_io_base));
1099 bytes.push_back(u8(m_crtc_io_base >> 8));
1100 return bytes;
1101}
1102
1103void CPowerStorm3xx::restore_vga_state(const std::vector<u8>& bytes)
1104{
1105 auto dword = [&](size_t offset) {
1106 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1107 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1108 };
1109 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1110 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1111 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1112 dword(90) > 31)
1113 throw std::runtime_error("Invalid VGA snapshot fields");
1114 reset_vga();
1115 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1116 for (unsigned i = 0; i < 25; ++i)
1117 if (i != 0x11)
1118 m_crtc_map.write_byte(i, bytes[1 + i]);
1119 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1120 for (unsigned i = 0; i < 5; ++i)
1121 m_seq_map.write_byte(i, bytes[26 + i]);
1122 for (unsigned i = 0; i < 9; ++i)
1123 m_gc_map.write_byte(i, bytes[31 + i]);
1124 for (unsigned i = 0; i < 21; ++i)
1125 m_atc_map.write_byte(i, bytes[40 + i]);
1126 // Every register must read back exactly as saved.
1127 for (unsigned i = 0; i < 25; ++i)
1128 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1129 throw std::runtime_error("Invalid CRTC snapshot encoding");
1130 for (unsigned i = 0; i < 5; ++i)
1131 if (m_seq_map.read_byte(i) != bytes[26 + i])
1132 throw std::runtime_error("Invalid sequencer snapshot encoding");
1133 for (unsigned i = 0; i < 9; ++i)
1134 if (m_gc_map.read_byte(i) != bytes[31 + i])
1135 throw std::runtime_error(
1136 "Invalid graphics-controller snapshot encoding");
1137 for (unsigned i = 0; i < 21; ++i)
1138 if (m_atc_map.read_byte(i) != bytes[40 + i])
1139 throw std::runtime_error("Invalid attribute snapshot encoding");
1140 vga.crtc.index = bytes[61];
1141 vga.sequencer.index = bytes[62];
1142 vga.gc.index = bytes[63];
1143 vga.attribute.index = bytes[64];
1144 vga.miscellaneous_output = bytes[65];
1145 m_ioas = (bytes[65] & 1) != 0;
1146 vga.feature_control = bytes[66];
1147 m_vga_enable = bytes[67];
1148 vga.dac.mask = bytes[68];
1149 vga.attribute.state = bytes[69];
1150 vga.dac.read = bytes[70];
1151 vga.dac.state = bytes[71];
1152 for (unsigned i = 0; i < 3; ++i)
1153 vga.dac.loading[i] = bytes[72 + i];
1154 for (unsigned i = 0; i < 4; ++i)
1155 vga.gc.latch[i] = bytes[75 + i];
1156 vga.crtc.start_addr = dword(79);
1157 vga.attribute.pel_shift = bytes[83];
1158 vga.cursor.visible = bytes[84];
1159 vga.crtc.irq_latch = bytes[85];
1160 m_vga_scanline = dword(86);
1161 m_vga_status_phase = dword(90);
1162 m_crtc_io_base = u16(bytes[94] | (u16(bytes[95]) << 8));
1163 vga.dac.dirty = 1;
1164}

1206
1207int CPowerStorm3xx::RestoreState(FILE* f)
1208{
1209 if (!f)
1210 return -1;
1211 std::lock_guard<std::recursive_mutex> guard(
1212 cSystem->get_device_bus_mutex());
1213 try
1214 {
1215 // Restore PCI first; any restore failure must abort the VM restore.
1216 struct ReplayGuard
1217 {
1218 bool& flag;
1219
1220 ReplayGuard(bool& value) : flag(value) { flag = true; }
1221
1222 ~ReplayGuard() { flag = false; }
1223 } replay(m_replaying_pci);
1224
1225 if (CPCIDevice::RestoreState(f) || powerstorm_read_u32(f) != 0x31475350)
1226 return -1;
1227 const u32 identity_size = powerstorm_read_u32(f);
1228 if (identity_size > 8192)
