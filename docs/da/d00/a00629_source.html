259void CDisk::scsi_reset()
260{
261 const bool media_removed = state.scsi.media_changed == SCSI_MEDIA_STATE_REMOVED ||
262 state.scsi.media_changed == SCSI_MEDIA_STATE_RESET_REMOVED;
263 const bool media_changed = state.scsi.media_changed == SCSI_MEDIA_STATE_CHANGED ||
264 state.scsi.media_changed == SCSI_MEDIA_STATE_RESET_CHANGED;
265 memset(&state.scsi, 0, sizeof(state.scsi));
266 state.scsi.media_changed = media_removed ? SCSI_MEDIA_STATE_RESET_REMOVED :
267 (media_changed ? SCSI_MEDIA_STATE_RESET_CHANGED : SCSI_MEDIA_STATE_RESET);
268 media_lock_changed(false);
269}

279 const u8 offline = is_offline() ? 1 : 0;
280 long ss = sizeof(state) + sizeof(offline);
281
282 if (fwrite(&disk_magic1, sizeof(u32), 1, f) != 1 ||
283 fwrite(&ss, sizeof(long), 1, f) != 1 ||
284 fwrite(&state, sizeof(state), 1, f) != 1 ||
285 fwrite(&offline, sizeof(offline), 1, f) != 1 ||
286 fwrite(&disk_magic2, sizeof(u32), 1, f) != 1)
287 return -1;
288 printf("%s: %d bytes saved.\n", devid_string, (int)ss);
289 return 0;

295int CDisk::RestoreState(FILE* f)
296{
297 long ss;
298 u32 m1;
299 u32 m2;
300 size_t r;
301
302 r = fread(&m1, sizeof(u32), 1, f);
303 if (r != 1)
304 {
305 printf("%s: unexpected end of file!\n", devid_string);
306 return -1;
307 }
308
309 if (m1 != disk_magic1)
310 {
311 printf("%s: MAGIC 1 does not match!\n", devid_string);
312 return -1;
313 }
314
315 r = fread(&ss, sizeof(long), 1, f);
316 if (r != 1)
317 {
318 printf("%s: unexpected end of file!\n", devid_string);
319 return -1;
320 }
321
322 if (ss != sizeof(state) && ss != sizeof(state) + sizeof(u8))
323 {
324 printf("%s: STRUCT SIZE does not match!\n", devid_string);
325 return -1;
326 }
327
328 // Do not replace the live protocol state unless the entire record is valid.
329 std::unique_ptr<SDisk_state> restored_state(new SDisk_state);
330 r = fread(restored_state.get(), sizeof(state), 1, f);
331 if (r != 1)
332 {
333 printf("%s: unexpected end of file!\n", devid_string);
334 return -1;
335 }
336
337 // Older disk records predate failure injection and restore as online.
338 u8 offline = 0;
339 if (ss != sizeof(state) &&
340 (fread(&offline, sizeof(offline), 1, f) != 1 || offline > 1))
341 {
342 printf("%s: Invalid disk offline state.\n", devid_string);
343 return -1;
344 }
345 r = fread(&m2, sizeof(u32), 1, f);
346 if (r != 1)
347 {
348 printf("%s: unexpected end of file!\n", devid_string);
349 return -1;
350 }
351
352 if (m2 != disk_magic2)
353 {
354 printf("%s: MAGIC 2 does not match!\n", devid_string);
355 return -1;
356 }
357
358 const auto& s = *restored_state;
359 if (s.byte_pos < 0 || s.byte_pos > byte_size ||
360 (byte_size > 0 && (!s.block_size || s.block_size > (u64)byte_size)) ||
361 s.scsi.msgi.available > sizeof(s.scsi.msgi.data) ||
362 s.scsi.msgi.read > s.scsi.msgi.available ||
363 s.scsi.msgo.written > sizeof(s.scsi.msgo.data) ||
364 s.scsi.cmd.written > sizeof(s.scsi.cmd.data) ||
365 s.scsi.dati.available > sizeof(s.scsi.dati.data) ||
366 s.scsi.dati.read > s.scsi.dati.available ||
367 s.scsi.dato.expected > sizeof(s.scsi.dato.data) ||
368 s.scsi.dato.written > s.scsi.dato.expected ||
369 s.scsi.stat.available > sizeof(s.scsi.stat.data) ||
370 s.scsi.stat.read > s.scsi.stat.available ||
371 s.scsi.sense.available > sizeof(s.scsi.sense.data))
372 {
373 printf("%s: Invalid disk position or protocol buffer bounds.\n", devid_string);
374 return -1;
375 }
376 memcpy(&state, restored_state.get(), sizeof(state));
377 set_offline(offline != 0);
378 //calc_cylinders(); // state.block_size may have changed.
379 determine_layout();
380 media_lock_changed(state.scsi.locked);
381
382 printf("%s: %d bytes restored.\n", devid_string, (int)ss);
383 return 0;
384}

396size_t CDisk::scsi_expected_xfer_me(int bus)
397{
398 switch (scsi_get_phase(0))
399 {
400 case SCSI_PHASE_DATA_OUT:
401 return state.scsi.dato.expected - state.scsi.dato.written;
402
403 case SCSI_PHASE_DATA_IN:
404 return state.scsi.dati.available - state.scsi.dati.read;
405
406 case SCSI_PHASE_COMMAND:
407 return 256 - state.scsi.cmd.written;
408
409 case SCSI_PHASE_STATUS:
410 return state.scsi.stat.available - state.scsi.stat.read;
411
412 case SCSI_PHASE_MSG_OUT:
413 return 256 - state.scsi.msgo.written;
414
415 case SCSI_PHASE_MSG_IN:
416 return state.scsi.msgi.available - state.scsi.msgi.read;
417
418 default:
419 FAILURE_2(IllegalState, "%s: transfer requested in phase %d\n",
420 devid_string, scsi_get_phase(0));
421 }

705void CDisk::do_scsi_error(int errcode, int info)
706{
707 state.scsi.stat.available = 1;
708 state.scsi.stat.data[0] = 0;
709 state.scsi.stat.read = 0;
710 state.scsi.msgi.available = 1;
711 state.scsi.msgi.data[0] = 0;
712 state.scsi.msgi.read = 0;
713
714 if (errcode == SCSI_OK)
715 {
716#if defined(DEBUG_SCSI)
717 printf("%s: Command returns OK status.\n", devid_string);
718#endif
719 return;
720 }
721
722 state.scsi.stat.data[0] = 0x02; // check sense
723 state.scsi.sense.data[0] = 0xf0; // error code
724 state.scsi.sense.data[1] = 0x00; // segment number
725 state.scsi.sense.data[3] = info >> 24; // info
726 state.scsi.sense.data[4] = info >> 16;
727 state.scsi.sense.data[5] = info >> 8;
728 state.scsi.sense.data[6] = info & 0xFF;
729 state.scsi.sense.data[7] = 10; // additional sense length
730 state.scsi.sense.data[8] = 0x00; // command specific
731 state.scsi.sense.data[9] = 0x00;
732 state.scsi.sense.data[10] = 0x00;
733 state.scsi.sense.data[11] = 0x00;
734 state.scsi.sense.data[14] = 0x00; // FRU code
735 state.scsi.sense.data[15] = 0x00; // sense key specific
736 state.scsi.sense.data[16] = 0x00;
737 state.scsi.sense.data[17] = 0x00;
738 state.scsi.sense.available = 18;
739
740 switch (errcode)
741 {
742 case SCSI_ILL_CMD:
743 state.scsi.sense.data[2] = 0x05; // illegal request
744 state.scsi.sense.data[12] = 0x20; // invalid command
745 state.scsi.sense.data[13] = 0x00;
746#if defined(DEBUG_SCSI)
747 printf("%s: Command returns check sense status (sense: ILLEGAL COMMAND).\n",
748 devid_string);
749#endif
750 break;
751
752 case SCSI_INVALID_FIELD:
753 state.scsi.sense.data[2] = 0x05; // ILLEGAL REQUEST
754 state.scsi.sense.data[12] = 0x24; // INVALID FIELD IN CDB
755 state.scsi.sense.data[13] = 0x00;
756#if defined(DEBUG_SCSI)
757 printf("%s: Check sense: INVALID FIELD IN CDB.\n", devid_string);
758#endif
759 break;
760
761 case SCSI_INVALID_LUN:
762 state.scsi.sense.data[2] = 0x05; // ILLEGAL REQUEST
763 state.scsi.sense.data[12] = 0x25; // INVALID LUN
764 state.scsi.sense.data[13] = 0x00;
765#if defined(DEBUG_SCSI)
766 printf("%s: Check sense: INVALID LUN.\n", devid_string);
767#endif
768 break;
769
770 case SCSI_LBA_RANGE:
771 state.scsi.sense.data[2] = 0x05; // illegal request
772 state.scsi.sense.data[12] = 0x21; // LBA out of range
773 state.scsi.sense.data[13] = 0x00;
774#if defined(DEBUG_SCSI)
775 printf("%s: Command returns check sense status (sense: LBA OUT OF RANGE).\n",
776 devid_string);
777#endif
778 break;
779
780 case SCSI_MEDIA_CHANGE:
781 state.scsi.sense.data[2] = 0x06; // unit attention
782 state.scsi.sense.data[12] = 0x28; // media changed
783 state.scsi.sense.data[13] = 0x00;
784 break;
785
786 case SCSI_MEDIA_REMOVED:
787 state.scsi.sense.data[2] = 0x02; // not ready
788 state.scsi.sense.data[12] = 0x3a; // media not present
789 state.scsi.sense.data[13] = 0x00;
790 break;
791
792 case SCSI_OFFLINE:
793 state.scsi.sense.data[2] = 0x02; // NOT READY
794 state.scsi.sense.data[12] = 0x04; // logical unit not ready
795 state.scsi.sense.data[13] = 0x03; // manual intervention required
796 break;
797
798 case SCSI_MEDIA_LOCKED:

803
804 case SCSI_HARDWARE_ERROR:
805 state.scsi.sense.data[2] = 0x04; // hardware error
806 state.scsi.sense.data[12] = 0x44; // internal target failure
807 state.scsi.sense.data[13] = 0x00;
808 break;
809
810 case SCSI_WRITE_PROTECTED:
811 state.scsi.sense.data[2] = 0x07; // data protect
812 state.scsi.sense.data[12] = 0x27; // write protected
813 state.scsi.sense.data[13] = 0x00;
814 break;
815
816 case SCSI_TOO_BIG:
817 state.scsi.sense.data[2] = 0x05; // illegal request
818 state.scsi.sense.data[12] = 0x55; // system resource failure
819 state.scsi.sense.data[13] = 0x00;

826 case SCSI_RESET:

830#if defined(DEBUG_SCSI)

832#endif
833 break;
834 }

840static u32 lba2msf(off_t_large lba)
841{
842#define PREGAP_SECTORS 150
843#define CD_FRAMES_PER_SEC 75
844#define CD_MAX_LSN 450150
845
846#define bin2bcd(x) ((x / 10) << 4) | (x % 10)
847 int m;
848
849 int s;
850
851 int f;
852 lba -= PREGAP_SECTORS;
853 if (lba >= -PREGAP_SECTORS)
854 {
855 m = (lba + PREGAP_SECTORS) / (CD_FRAMES_PER_SEC * 60);
856 lba -= m * (CD_FRAMES_PER_SEC * 60);
857 s = (lba + PREGAP_SECTORS) / CD_FRAMES_PER_SEC;
858 lba -= s * CD_FRAMES_PER_SEC;
859 f = lba + PREGAP_SECTORS;
860 }
861 else
862 {
863 m = (lba + CD_MAX_LSN) / (CD_FRAMES_PER_SEC * 60);
864 lba -= m * (CD_FRAMES_PER_SEC * 60);
865 s = (lba + CD_MAX_LSN) / CD_FRAMES_PER_SEC;
866 lba -= s * CD_FRAMES_PER_SEC;
867 f = lba + CD_MAX_LSN;
868 }
869
870 if (m > 99)
871 m = 99;
872
873 //printf("m=%d, s=%d, f=%d == m=%x, s=%x, f=%x\n", m,s,f,bin2bcd(m),bin2bcd(s),bin2bcd(f));
874 return bin2bcd(m) << 16 | bin2bcd(s) << 8 | bin2bcd(f);
875}

1014int CDisk::mode_sense_page(int page, bool changeable, int q)
1015{
1016 u8* d = state.scsi.dati.data;
1017 int len;
1018
1019 switch (page)
1020 {
1021 case SCSIMP_READ_WRITE_ERRREC: len = 10; break;
1022 case SCSIMP_FORMAT_PARAMS: len = 22; break;
1023 case SCSIMP_RIGID_GEOMETRY: len = 22; break;
1024 case SCSIMP_CACHING: len = 0x12; break;
1025 case SCSIMP_INFO_EXCEPTIONS: len = 0x0a; break;
1026 case SCSIMP_CDROM_CAP: len = 0x14; break;

1033 d[q + 1] = (u8)len;

1101#endif

1113 state.scsi.media_changed = SCSI_MEDIA_STATE_REMOVED;
1114 else if (state.scsi.media_changed == SCSI_MEDIA_STATE_RESET_CHANGED)
1115 state.scsi.media_changed = SCSI_MEDIA_STATE_CHANGED;

1117 state.scsi.media_changed = SCSI_MEDIA_STATE_STABLE;

1122 if (cdrom() && state.scsi.media_changed == SCSI_MEDIA_STATE_CHANGED &&
1123 state.scsi.cmd.data[0] != SCSICMD_INQUIRY &&
1124 state.scsi.cmd.data[0] != SCSICMD_REQUEST_SENSE)

1126 state.scsi.media_changed = SCSI_MEDIA_STATE_STABLE;

1150 state.scsi.cmd.data[0] != SCSICMD_REQUEST_SENSE)

1162#endif
