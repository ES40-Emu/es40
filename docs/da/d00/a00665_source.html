253 u8 get_status_byte()
254 {
255
256 // top bit is 0 , bit 6 is 1 if remote mode.
257 u8 ret = (u8)((mode == MOUSE_MODE_REMOTE) ? 0x40 : 0);
258 ret |= (enable << 5);
259 ret |= (scaling == 1) ? 0 : (1 << 4);
260 ret |= ((button_status & 0x1) << 2);
261 ret |= ((button_status & 0x2) << 0);
262 return ret;
263 }
264
265 u8 get_resolution_byte()
266 {
267 u8 ret = 0;
268
269 switch (resolution_cpmm)
270 {
271 case 1: ret = 0; break;
272 case 2: ret = 1; break;
273 case 4: ret = 2; break;
274 case 8: ret = 3; break;
275 default: FAILURE(NotImplemented, "mouse: invalid resolution_cpmm");
276 };
277 return ret;
278 }
