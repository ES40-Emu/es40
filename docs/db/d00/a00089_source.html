143u64 CAliM1543C_usb::usb_hci_read(u64 address, int dsize)
144{
145 u64 data = 0;
146 if (dsize != 32)
147 printf("%%USB-W-HCIREAD: Non dword read, returning 32 bits anyway.\n");
148 switch (address)
149 {
150 case 0: // HcRevision
151 data = 0x00000110;
152 break;
153
154 case 4: // HcControl
155 case 8: // HcCommandStatus
156 case 0x0c: // HcInterruptStatus
157 case 0x10: // HcInterrupt Enable
158 case 0x14: // HcInterruptDisable
159 case 0x18: // HcHCCA (datasheet says 0x17, but that's wrong)
160 case 0x1c: // HcPeriodCurrentED
161 case 0x20: // HcControlHeadED
162 case 0x24: // HcControlCurrentED
163 case 0x28: // HcBulkHeadED
164 case 0x2c: // HcBulkCurrentED
165 case 0x30: // HcDoneHead
166 case 0x34: // HcFmInterval
167 case 0x38: // HcFrameRemaining
168 case 0x3c: // HcFmNumber
169 case 0x40: // HcPeriodicStart
170 case 0x44: // HcLSThreshold
171 case 0x48: // HcRhDescriptorA
172 case 0x4c: // HcRhDescriptorB
173 case 0x50: // HcRhStatus
174 case 0x54: // HcRhPortStatus1
175 case 0x58: // HcRhPortStatus2
176 case 0x5c: // HcRhPortStatus3
177 case 0x100: // HceControlRegister
178 case 0x104: // HceInputRegister
179 case 0x108: // HceOutputRegister
180 case 0x10c: // HceStatusRegister
181 data = state.usb_data[address / 4];
182 break;
183
184 default:
185 printf("%%USB-W-HCIREAD: Reading from unknown address %x. Ignoring.\n",
186 (int)address);
187 }
188
189 return data;
190}

192void CAliM1543C_usb::usb_hci_write(u64 address, int dsize, u64 data)
193{
194 if (dsize != 32)
195 printf("%%USB-W-HCIWRITE: Non dword write, writing 32 bits anyway.\n");
196
197 switch (address)
198 {
199 case 0x04: // HcControl
200 state.usb_data[address / 4] = data;
201 break;
202
203 case 0x08: // HcCommandStatus
204 // Bit 0 = HCR (Host Controller Reset). Make it self-clear and reset regs.
205 if (data & 0x1) {
206 // Reset to a benign, quiescent state
207 memset(state.usb_data, 0, sizeof(state.usb_data));
208 state.usb_data[0x34 / 4] = 0x2edf; // HcFmInterval
209 state.usb_data[0x48 / 4] = 0x01000203; // 3 ports, NPS (no power switching)
210 state.usb_data[0x54 / 4] = 0x00000100; // ports always powered
211 state.usb_data[0x58 / 4] = 0x00000100;
212 state.usb_data[0x5c / 4] = 0x00000100;
213 // No pending or enabled interrupts after reset
214 // HcInterruptStatus = 0, HcInterruptEnable = 0
215 // Leave HcControl in USBReset/Suspend equivalent (0)
216 }
217 // HCR auto-clears; store everything else with bit0 cleared
218 state.usb_data[0x08 / 4] = (data & ~0x1);
219 break;
220
221 case 0x0c: // HcInterruptStatus
222 state.usb_data[0x0c / 4] &= ~data;
223 break;
224
225 case 0x10: // HcInterrupt Enable
226 state.usb_data[0x10 / 4] |= data;
227 break;
228
229 case 0x14: // HcInterruptDisable
230 state.usb_data[0x10 / 4] &= ~data;
231 break;
232
233 case 0x50: // HcRhStatus: write-1 command bits (OHCI 7.4.3)
234 if (data & 0x00008000) state.usb_data[0x50 / 4] |= 0x00008000; // SetRemoteWakeupEnable
235 if (data & 0x80000000) state.usb_data[0x50 / 4] &= ~0x00008000; // ClearRemoteWakeupEnable
236 if (data & 0x00020000) state.usb_data[0x50 / 4] &= ~0x00020000; // clear OCIC
237 // Set/ClearGlobalPower ignored: NPS, ports always powered
238 break;
239
240 case 0x54: // HcRhPortStatus1..3: write-1 command bits (OHCI 7.4.4)
241 case 0x58:
242 case 0x5c:
243 {
244 u32 ps = state.usb_data[address / 4];
245 if (data & 0x00000001) ps &= ~0x00000002; // ClearPortEnable
246 if (data & 0x00000016) // Set{Enable,Suspend,Reset} with no device -> CSC
247 ps |= 0x00010000;
248 ps &= ~(data & 0x001f0000); // write-1-to-clear change bits
249 ps |= 0x00000100; // PPS always set with NPS
250 state.usb_data[address / 4] = ps;
251 if (ps & 0x001f0000)
252 state.usb_data[0x0c / 4] |= 0x40; // reflect pending change as RHSC
253 break;
254 }
255
256 case 0x18: // HcHCCA (datasheet says 0x17, but that's wrong)
257 case 0x1c: // HcPeriodCurrentED
258 case 0x20: // HcControlHeadED
259 case 0x24: // HcControlCurrentED
260 case 0x28: // HcBulkHeadED
261 case 0x2c: // HcBulkCurrentED
262 case 0x30: // HcDoneHead
263 case 0x34: // HcFmInterval
264 case 0x38: // HcFrameRemaining
265 case 0x3c: // HcFmNumber
266 case 0x40: // HcPeriodicStart
267 case 0x44: // HcLSThreshold
268 case 0x48: // HcRhDescriptorA
269 case 0x4c: // HcRhDescriptorB
270 case 0x100: // HceControlRegister
271 case 0x104: // HceInputRegister
272 case 0x108: // HceOutputRegister
273 case 0x10c: // HceStatusRegister
274 state.usb_data[address / 4] = data;
275 break;
276
277 default:
278 printf("%%USB-W-HCIWRITE: Writing to unknown address %x. Ignoring.\n",
279 (int)address);
280 }
281}

308int CAliM1543C_usb::RestoreState(FILE* f)
309{
310 long ss;
311 u32 m1;
312 u32 m2;
313 int res;
314 size_t r;
315
316 if ((res = CPCIDevice::RestoreState(f)))
317 return res;
318
319 r = fread(&m1, sizeof(u32), 1, f);
320 if (r != 1)
321 {
322 printf("%s: unexpected end of file!\n", devid_string);
323 return -1;
324 }
325
326 if (m1 != usb_magic1)
327 {
328 printf("%s: MAGIC 1 does not match!\n", devid_string);
329 return -1;
330 }
331
332 r = fread(&ss, sizeof(long), 1, f);
333 if (r != 1)
334 {
335 printf("%s: unexpected end of file!\n", devid_string);
336 return -1;
337 }
338
339 if (ss != sizeof(state))
340 {
341 printf("%s: STRUCT SIZE does not match!\n", devid_string);
342 return -1;
343 }
344
345 r = fread(&state, sizeof(state), 1, f);
346 if (r != 1)
347 {
348 printf("%s: unexpected end of file!\n", devid_string);
349 return -1;
350 }
351
352 r = fread(&m2, sizeof(u32), 1, f);
353 if (r != 1)
354 {
355 printf("%s: unexpected end of file!\n", devid_string);
356 return -1;
357 }
358
359 if (m2 != usb_magic2)
360 {
361 printf("%s: MAGIC 1 does not match!\n", devid_string);
362 return -1;
363 }
364
365 printf("%s: %d bytes restored.\n", devid_string, (int)ss);
366 return 0;
