44 static constexpr AlphaIPRReadDescriptor describe(std::uint32_t instruction)
45 {
46 const auto function = selector(instruction);
47 if ((function & 0xc0) == 0x40)
48 return {AlphaIPRReadKind::Helper, false, true}; // PCTX group.
49
50 switch (function)
51 {
52 case 0x06: return {AlphaIPRReadKind::ExceptionAddress, false, true};
53 case 0x0f: return {AlphaIPRReadKind::ExceptionSummary, false, true};
54 case 0x10: return {AlphaIPRReadKind::PalBase, false, true};
55 case 0x27: return {AlphaIPRReadKind::MemoryFaultStatus, false, true};
56 case 0xc2: return {AlphaIPRReadKind::FaultAddress, false, true};
57 case 0x0d: return {AlphaIPRReadKind::Helper, true, true}; // ISUM is asynchronous.
58 case 0x05: case 0x07:
59 case 0x08: case 0x09: case 0x0a: case 0x0b: case 0x0c:
60 case 0x11: case 0x14: case 0x16: case 0x2a: case 0x2b:
61 case 0xc0: case 0xc3:
62 return {AlphaIPRReadKind::Helper, false, true};
63 default:
64 return {AlphaIPRReadKind::Unsupported, false, true};
65 }
66 }
67
68 static constexpr AlphaIPRWriteKind describe_write(std::uint32_t instruction)
69 {
70 const auto function = selector(instruction);
71 // ASN changes require interpreter dispatch; the other PCTX fields can compile.
72 if ((function & 0xc0) == 0x40)
73 return (function & 1) ? AlphaIPRWriteKind::Interpreter : AlphaIPRWriteKind::Helper;
74
75 switch (function)
76 {
77 case 0x00: case 0x14: case 0x20: case 0x26:
78 case 0x29: case 0xa0: case 0xc0: // Field stores.
79 case 0x01: case 0x21: case 0xa1: // Translation-buffer fills.
80 case 0x02: case 0x03: case 0x04: // ITB invalidation.
81 case 0x13: // Lazy instruction-cache flush.
82 case 0x09: case 0x0a: case 0x0b: case 0x0c: // Mode/interrupt fields.
83 case 0x24: case 0xa4: // Single-entry DTB invalidation.
84 return AlphaIPRWriteKind::Helper;
85 case 0x11: // I_CTL changes compiled assumptions, including SDE.
86 return AlphaIPRWriteKind::HelperTerminator;
87 case 0x15: case 0x17: case 0x27:
88 case 0x2b: case 0x2c: case 0x2d:
89 return AlphaIPRWriteKind::Noop;
90 default:
91 return AlphaIPRWriteKind::Interpreter;
92 }
93 }
94};
