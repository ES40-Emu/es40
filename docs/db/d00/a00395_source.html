58static uint32_t powerstorm_setting(
59 CConfigurator* cfg, const char* key, uint32_t def,
60 uint32_t max = 0xffffffff)
61{
62 const char* text = cfg->get_text_value(key, nullptr);
63 if (!text || !*text)
64 return def;
65 char* end = nullptr;
66 errno = 0;
67 const unsigned long long n = std::strtoull(text, &end, 0);
68 if (errno || end == text || *end || text[0] == '-' || n > max)
69 throw std::runtime_error(
70 std::string("Invalid powerstorm setting: ") + key);
71 return static_cast<uint32_t>(n);
72}

74static void powerstorm_write_u32(FILE* f, uint32_t v)
75{
76 uint8_t b[4];
77 for (unsigned i = 0; i < 4; ++i)
78 b[i] = uint8_t(v >> (8 * i));
79 if (fwrite(b, 1, 4, f) != 4)
80 throw std::runtime_error("Snapshot write failed");
81}

83static uint32_t powerstorm_read_u32(FILE* f)
84{
85 uint8_t b[4];
86 if (fread(b, 1, 4, f) != 4)
87 throw std::runtime_error("Snapshot read failed");
88 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
89 (uint32_t(b[3]) << 24);
90}

92static bool powerstorm_vga_window_offset(u32& address, unsigned selection)
93{
94 switch (selection & 3)
95 {
96 case 1:
97 return address < 0x10000;
98 case 2:
99 if (address < 0x10000 || address >= 0x18000)
100 return false;
101 address -= 0x10000;
102 break;
103 case 3:
104 if (address < 0x18000 || address >= 0x20000)
105 return false;
106 address -= 0x18000;
107 break;
108 default:
109 return address < 0x20000;
110 }
111 return true;
112}

116void CPowerStorm3xx::crtc_map(address_map& map)
117{
118 unimplemented_map(map, "VGA CRTC register");
119 vga_device::crtc_map(map);
120 // ROM 0x3641..0x3649 stores its I/O base here; 0x3657..0x366d reads it back.
121 map(0x1c, 0x1d)
122 .lrw8(
123 NAME([this](offs_t offset) {
124 return u8(m_crtc_io_base >> (offset * 8));
125 }),
126 NAME([this](offs_t offset, u8 data) {
127 const unsigned shift = offset * 8;
128 m_crtc_io_base =
129 u16((m_crtc_io_base & ~(0xffu << shift)) |
130 (u32(data) << shift));
131 }));
132 // Reject S3 identification/unlock probes without extension state.
133 map(0x30, 0x30).lr8(NAME([](offs_t) { return u8(0xff); }));
134 map(0x38, 0x39).lrw8(
135 NAME([](offs_t) { return u8(0xff); }),
136 NAME([](offs_t, u8) {}));
137}

139void CPowerStorm3xx::sequencer_map(address_map& map)
140{
141 unimplemented_map(map, "VGA sequencer register");
142 // These frontends mask stored bytes, including direct snapshot writes.
143 // Limit the canonical map to SR00-04 so unknown indices keep diagnostics.
144 auto standard = std::make_shared<address_map>(5);
145 vga_device::sequencer_map(*standard);
146 map(0x00, 0x04).lrw8(
147 NAME([standard](offs_t index) {
148 return standard->read_byte(index);
149 }),
150 NAME(([this, standard](offs_t index, u8 data) {
151 static constexpr u8 masks[] = {0x03, 0x3f, 0x0f, 0x3f, 0x0e};
152 vga.sequencer.data[index] = data & masks[index];
153 standard->write_byte(index, data);
154 })));
155}

163void CPowerStorm3xx::attribute_map(address_map& map)
164{
165 unimplemented_map(map, "VGA attribute register");
166 vga_device::attribute_map(map);
167 // Palette reads remain available when PAS disables CPU writes.
168 map(0x20, 0x2f).lr8(NAME([this](offs_t offset) {
169 return vga.attribute.data[offset];
170 }));
171 // These frontends retain mode-control bits 6/7 and mask reserved bit 4.
172 map(0x10, 0x10).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
173 vga.attribute.data[0x10] = data & 0xef;
174 }));
175 // Color select does not change the pending horizontal panning value.
176 map(0x14, 0x14).mirror(0x20).lw8(NAME([this](offs_t, u8 data) {
177 vga.attribute.data[0x14] = data & 0xf;
178 }));

198u8 CPowerStorm3xx::mem_r(offs_t address)
199{
200 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
201 return 0xff;
202 unsigned selected = vga.gc.read_map_sel & 3;
203 if (vga.sequencer.data[4] & 8)
204 {
205 // Keep packed chain-four bytes in the layout consumed by vga_device scanout.
206 for (unsigned plane = 0; plane < 4; ++plane)
207 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
208 selected = address & 3;
209 }
210 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
211 {
212 // vga_device text stores character/attribute pairs together; fonts stay planar.
213 vga.gc.latch[0] = vga.memory[address & ~1u];
214 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
215 for (unsigned plane = 2; plane < 4; ++plane)
216 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
217 selected = (vga.gc.read_map_sel & 2) | (address & 1);
218 }
219 else
220 {
221 for (unsigned plane = 0; plane < 4; ++plane)
222 vga.gc.latch[plane] =
223 vga.memory[(address & 0xffff) + plane * 0x10000];
224 }
225 if (!vga.gc.read_mode)
226 return vga.gc.latch[selected];
227 u8 result = 0xff;
228 for (unsigned plane = 0; plane < 4; ++plane)
229 if (vga.gc.color_dont_care & (1 << plane))
230 result &=
231 u8(vga.gc.latch[plane] ^
232 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
233 return result;
234}

236void CPowerStorm3xx::mem_w(offs_t address, u8 data)
237{
238 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
239 return;
240 if (vga.sequencer.data[4] & 8)
241 {
242 if (vga.sequencer.map_mask & (1 << (address & 3)))
243 vga.memory[address] = vga_latch_write(address & 3, data);
244 return;
245 }
246 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
247 {
248 const unsigned parity = address & 1;
249 if (vga.sequencer.map_mask & (1 << parity))
250 vga.memory[address] = vga_latch_write(parity, data);
251 if (vga.sequencer.map_mask & (1 << (parity + 2)))
252 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
253 vga_latch_write(parity + 2, data);
254 return;
255 }
256 for (unsigned plane = 0; plane < 4; ++plane)
257 if (vga.sequencer.map_mask & (1 << plane))
258 vga.memory[(address & 0xffff) + plane * 0x10000] =
259 vga_latch_write(plane, data);
260}

293CRealImage2100::Frame CPowerStorm3xx::render_frame(CRealImage2100::Frame frame)
294{
295 std::lock_guard<std::recursive_mutex> guard(
296 cSystem->get_device_bus_mutex());
297 if (m_pause.load())
298 return {};
299 // The native display path remains active when the driver blanks legacy VGA.
300 if (m_realimage.native_display())
301 return m_realimage.scanout(std::move(frame));
302 if (!m_board_vga_enabled || !m_vga_enable ||
303 !vga.crtc.sync_en || (vga.sequencer.data[0] & 3) != 3 ||
304 (vga.sequencer.data[1] & 0x20) || !(vga.attribute.index & 0x20) ||
305 !vga.crtc.maximum_scan_line)
306 return {};
307 frame.width = (vga.crtc.horz_disp_end + 1) *
308 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
309 frame.height = vga.crtc.vert_disp_end + 1;
310 if (!frame.width || frame.width > 4096 || frame.height > 1024)
311 return {};
312 // Allow vga_device double-scan and one extra character row below the visible area.
313 m_render_bitmap.resize(frame.width, frame.height * 2 + 64);
314 screen().set_visible_area(frame.width, frame.height);
315 screen().tick_frame();
316 vga.crtc.start_addr = latch_start_addr();
317 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
318 palette_update();
319 svga_device::screen_update(
320 screen(), m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
321 frame.argb.assign(
322 m_render_bitmap.raw(),
323 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
324 return frame;
325}

357void CPowerStorm3xx::run()
358{
359 try
360 {
361 if (!m_gui_initialized)
362 {
363 m_output.display().init(X_TILESIZE, Y_TILESIZE);
364 m_gui_initialized = true;
365 }
366 // Only this display thread owns the reusable frame.
367 CRealImage2100::Frame frame;
368 while (!m_stop.load())
369 {
370 {
371 CPowerStormDisplayLock guard(m_output.display());
372 m_output.display().handle_events();
373 }
374 if (m_pause.load())
375 {
376 m_pause_ack.store(true);
377 CThread::sleep(10);
378 continue;
379 }
380 m_pause_ack.store(false);
381 update(frame);
382 CThread::sleep(20);
383 }
384 }
385 catch (const CException& e)
386 {
387 printf(
388 "%s: Exception in display thread: %s.\n",
389 devid_string,
390 e.displayText().c_str());

394 {
395 printf(
396 "%s: Exception in display thread: %s.\n", devid_string, e.what());
397 m_worker_failed.store(true);

403std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_data() const
404{
405 std::array<u32, 64> data{};
406 data[0] = u32(vendor) | (u32(device) << 16);
407 data[2] = (class_code << 8) | revision;
408 data[6] = 0x00000001; // BAR2: native index/data I/O pair
409 data[11] = u32(subsystem_vendor) | (u32(subsystem_device) << 16);
410 data[15] = 0x000001ff; // INTA#, interrupt line unassigned
411 return data;
412}

414std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_mask(u32 rom_size) const
415{
416 std::array<u32, 64> mask{};
417 mask[1] = 0x00000147; // I/O, memory, bus master, parity response, SERR
418 mask[3] = 0x0000ffff; // cache-line size and latency timer
419 mask[4] = ~(bar0_size - 1);
420 mask[5] = ~(bar1_size - 1);
421 mask[6] = ~(IOBARSize - 1);
422 if (rom_size)
423 mask[12] = (~(rom_size - 1) & 0xfffff800u) | 1; // ROM decode and enable
424 mask[15] = 0xff; // interrupt line only; pin fixed
425 return mask;
426}

428CPowerStorm3xx::OptionROM CPowerStorm3xx::OptionROM::parse(
429 const std::vector<u8>& raw, const std::string& layout)
430{
431 auto fail = [](const char* message) {
432 throw std::invalid_argument(message);
433 };
434 const size_t n = raw.size();
435 if (n < 2048 || n > 1024 * 1024)
436 fail("ROM must be 2 KiB to 1 MiB");
437 if (layout != "exact" && layout != "pad_ff")
438 fail("rom_layout must be exact or pad_ff");
439 if (layout == "exact" && (n & (n - 1)))
440 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
441 auto word = [&](size_t p) -> u16 {
442 if (p > n - 2)
443 fail("ROM field out of bounds");
444 return u16(raw[p] | (u16(raw[p + 1]) << 8));
445 };
446 size_t at = 0;
447 bool last = false;
448 OptionROM rom;
449 for (unsigned images = 0; images < 32 && !last; ++images)
450 {
451 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
452 fail("ROM missing 55 AA signature");
453 const size_t p = at + word(at + 0x18);
454 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
455 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
456 fail("ROM missing/bounded PCIR structure");
457 if (word(p + 4) != 0x10ba || word(p + 6) != 0x0304)
458 fail("ROM PCIR identity is not 10BA:0304 (PowerStorm 300/350)");
459 const size_t length = size_t(word(p + 0x10)) * 512;
460 if (!length || length > n - at || p + 0x18 > at + length)
461 fail("Invalid option-ROM image length");
462 if (word(p + 0x0a) < 0x18 || word(p + 0x0a) > at + length - p)
463 fail("Invalid PCIR structure length");
464 if (raw[p + 0x14] != 0)
465 fail("Only an x86 option-ROM image is supported");
466 if (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length)
467 fail("Invalid x86 initialization image length");
468 unsigned sum = 0;
469 for (size_t i = at; i < at + length; ++i)
470 sum += raw[i];
471 if (sum & 255)
472 fail("Option-ROM checksum failed");
473 rom.pcir_class = u32(raw[p + 13]) | (u32(raw[p + 14]) << 8) |
474 (u32(raw[p + 15]) << 16);
475 last = (raw[p + 0x15] & 0x80) != 0;
476 at += length;
477 }
478 if (!last)
479 fail("Unterminated option-ROM image chain");
480 // Keep the supplied image unmodified; only an erased tail may follow it.
481 for (size_t i = at; i < n; ++i)
482 if (raw[i] != 0xff)
483 fail("Unexpected data after last option-ROM image");
484 rom.bytes = raw;
485 rom.file_size = u32(n);
486 rom.image_bytes = u32(at);
487 if (layout == "pad_ff")
488 {
489 size_t decoded = 2048;
490 while (decoded < n)
491 decoded *= 2;
492 rom.bytes.resize(decoded, 0xff);
493 }
494 return rom;
495}

523void CPowerStorm3xx::init()
524try
525{
526 if (m_initialized)
527 return;
528 CConfigurator* cfg = myCfg;
529 init_maps();
530 add_legacy_io(LegacyIO, 0x3b0, 0x30);
531 add_legacy_mem(LegacyMemory, 0xa0000, 0x20000);
532 m_model = powerstorm_setting(cfg, "model", 300, 350);
533 if (m_model != 300 && m_model != 350)
534 throw std::runtime_error("powerstorm model must be 300 or 350");
535 m_realimage.configure_framebuffer(m_model == 350 ? 24u : 12u);
536 m_realimage.configure_texture_memory(m_model == 350 ?
537 CRealImage2100::MaxTextureSize : CRealImage2100::MinTextureSize);
538 m_board_vga_enabled = cfg->get_bool_value("vga_enabled", true);
539 m_trace_apertures = cfg->get_bool_value("trace_apertures", false);
540 m_profile.revision = u8(powerstorm_setting(cfg, "revision", 0, 255));
541 // SRM (PROBE_IO.C) identifies boards by subsystem; NT distinguishes 4D35.
542 const bool compaq = m_model == 300;
543 m_profile.subsystem_vendor = u16(powerstorm_setting(
544 cfg, "subsystem_vendor", compaq ? 0x0e11 : 0x1011, 65535));
545 m_profile.subsystem_device = u16(powerstorm_setting(
546 cfg, "subsystem_device", compaq ? 0x4d31 : 0x4d35, 65535));
547 m_profile.bar0_size =
548 powerstorm_setting(cfg, "bar0_size", m_profile.bar0_size, 0x8000000);
549 m_profile.bar1_size =
550 powerstorm_setting(cfg, "bar1_size", 0x2000000, 0x8000000);
551 for (const u32 size : {m_profile.bar0_size, m_profile.bar1_size})
552 if (size < 0x1000000 || (size & (size - 1)))
553 throw std::runtime_error(
554 "powerstorm BAR sizes must be powers of two from 16 to 128 "
555 "MiB");
556 m_frame_file = cfg->get_text_value("frame_file", "");
557 const std::string romname = cfg->get_text_value("rom", "");
558 m_rom_layout = cfg->get_text_value("rom_layout", "exact");
559 if (!romname.empty())
560 {
561 std::ifstream f(romname, std::ios::binary | std::ios::ate);
562 if (!f)
563 throw std::runtime_error("Cannot open powerstorm ROM: " + romname);
564 const auto size = f.tellg();
565 if (size < 2048 || size > 1024 * 1024)
566 throw std::runtime_error("powerstorm ROM must be 2 KiB to 1 MiB");
567 f.seekg(0);
568 std::vector<uint8_t> raw(static_cast<size_t>(size));
569 f.read(
570 reinterpret_cast<char*>(raw.data()),
571 static_cast<std::streamsize>(raw.size()));
572 if (!f)
573 throw std::runtime_error("Cannot read full powerstorm ROM");
574 const auto parsed = OptionROM::parse(raw, m_rom_layout);
575 m_rom = parsed.bytes;
576#ifdef DEBUG_VGA
577 printf(
578 "%s: ROM %u bytes, image %u bytes, PCIR class %06x (PCI class "
579 "030000).\n",
580 devid_string,
581 parsed.file_size,
582 parsed.image_bytes,
583 parsed.pcir_class);
584#endif
585 }
586 else if (m_rom_layout != "exact" && m_rom_layout != "pad_ff")
587 throw std::runtime_error("rom_layout must be exact or pad_ff");
588 const std::string tracename = cfg->get_text_value("trace", "");
589 if (!tracename.empty())
590 {
591 if (std::filesystem::exists(tracename))
592 throw std::runtime_error(
593 "Refusing to overwrite existing powerstorm trace: " +
594 tracename);
595 m_trace.open(tracename);
596 if (!m_trace)
597 throw std::runtime_error("Cannot create powerstorm trace");
598 m_trace << "# PowerStorm PCI/BAR/VGA accesses; widths are bits. No "
599 "guest-PC hook.\n";
600 }
601 const auto profile_data = m_profile.config_data(),
602 profile_mask = m_profile.config_mask(u32(m_rom.size()));
603 u32 data[64]{}, mask[64]{};
604 for (unsigned i = 0; i < 64; ++i)
605 {
606 data[i] = profile_data[i];
607 mask[i] = profile_mask[i];
608 }

741bool CPowerStorm3xx::vga_port_enabled(u32 port) const noexcept
742{
743 if (!m_board_vga_enabled)
744 return false;
745 // The VGA enable register stays addressable while the frontend is off.
746 if (port == 0x3c3)
747 return true;
748 if (!m_vga_enable)
749 return false;
750 if ((port == 0x3b4 || port == 0x3b5 || port == 0x3ba) && !m_ioas)
751 return true;
752 if ((port == 0x3d4 || port == 0x3d5 || port == 0x3da) && m_ioas)
753 return true;
754 return (port >= 0x3c0 && port <= 0x3ca) || port == 0x3cc || port == 0x3ce ||
755 port == 0x3cf;
756}

758bool CPowerStorm3xx::decodes_memory_access(
759 int index, u64 address, int dsize, bool write) const noexcept
760{
761 if (index >= PCI_RANGE_BASE)
762 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
763 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
764 return false;
765 const unsigned bytes = unsigned(dsize) / 8;
766 if (index == LegacyIO)
767 {
768 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
769 return false;
770 // Claim only accesses whose every byte decodes to this device.
771 for (unsigned i = 0; i < bytes; ++i)
772 if (!vga_port_enabled(0x3b0 + u32(address) + i))
773 return false;
774 return true;
775 }
776 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
777 bytes > 0x20000 - address)
778 return false;
779 switch (vga.gc.memory_map_sel & 3)
780 {
781 case 1:
782 return address + bytes <= 0x10000;
783 case 2:
784 return address >= 0x10000 && address + bytes <= 0x18000;
785 case 3:
786 return address >= 0x18000;
787 default:
788 return true;
789 }
790}

883bool CPowerStorm3xx::dma_ram_range(u32 start, size_t length)
884{
885 if (length > 0x100000000ull - start)
886 return false;
887 for (size_t done = 0; done < length;)
888 {
889 const u64 physical = cSystem->PCI_Phys(myPCIBus, start + u32(done));
890 const size_t chunk = std::min(length - done, size_t(8192 - (physical & 8191)));
891 if (!cSystem->PtrToMem(physical) || !cSystem->PtrToMem(physical + chunk - 1))
892 return false;
893 done += chunk;
894 }
895 return true;
896}

898bool CPowerStorm3xx::dma_read(
899 u32 address, uint8_t* destination, size_t count, u32 completion)
900{
901 if (!(config_read(0, 4, 16) & 4) || !destination || !count ||
902 count > CRealImage2100::DMACommandListMaxBytes ||
903 ((address | completion | count) & 3) ||
904 !dma_ram_range(address, count) || !dma_ram_range(completion, 4))
905 return false;
906 do_pci_read(address, destination, 1, count);
907 if (m_trace)
908 {
909 m_trace << "# DMA read " << std::hex << address << " bytes=" << count
910 << " completion=" << completion << '\n';
911 for (size_t offset = 0; offset < count; offset += 64)
912 {
913 m_trace << "# DMA payload " << offset;
914 for (size_t i = offset; i < std::min(offset + 64, count); i += 4)
915 {
916 const uint32_t value = uint32_t(destination[i]) |
917 (uint32_t(destination[i + 1]) << 8) |
918 (uint32_t(destination[i + 2]) << 16) |
919 (uint32_t(destination[i + 3]) << 24);
920 m_trace << ' ' << value;
921 }
922 m_trace << '\n';
923 }
924 m_trace << std::dec;
925 }
926 return true;
927}

934 uint8_t done[4]{};

1086{

1088 for (unsigned i = 0; i < 25; ++i)
1089 bytes.push_back(m_crtc_map.read_byte(i));
1090 for (unsigned i = 0; i < 5; ++i)
1091 bytes.push_back(m_seq_map.read_byte(i));
1092 for (unsigned i = 0; i < 9; ++i)
1093 bytes.push_back(m_gc_map.read_byte(i));
1094 for (unsigned i = 0; i < 21; ++i)
1095 bytes.push_back(m_atc_map.read_byte(i));
1096 bytes.insert(
1097 bytes.end(),
1098 {vga.crtc.index,
1099 vga.sequencer.index,
1100 vga.gc.index,
1101 vga.attribute.index,
1102 vga.miscellaneous_output,
1103 vga.feature_control,

1105 vga.dac.mask,
1106 u8(vga.attribute.state),
1107 u8(vga.dac.read),
1108 u8(vga.dac.state)});
1109 for (const u8 value : vga.dac.loading)
1110 bytes.push_back(value);
1111 for (const u8 value : vga.gc.latch)
1112 bytes.push_back(value);
1113 for (unsigned i = 0; i < 4; ++i)
1114 bytes.push_back(u8(vga.crtc.start_addr >> (8 * i)));
1115 bytes.push_back(vga.attribute.pel_shift);
1116 bytes.push_back(vga.cursor.visible);
1117 bytes.push_back(vga.crtc.irq_latch);
1118 for (unsigned i = 0; i < 4; ++i)
1119 bytes.push_back(u8(m_vga_scanline >> (8 * i)));
1120 for (unsigned i = 0; i < 4; ++i)
1121 bytes.push_back(u8(m_vga_status_phase >> (8 * i)));
1122 bytes.push_back(u8(m_crtc_io_base));
1123 bytes.push_back(u8(m_crtc_io_base >> 8));
1124 return bytes;

1127void CPowerStorm3xx::restore_vga_state(const std::vector<u8>& bytes)
1128{
1129 auto dword = [&](size_t offset) {
1130 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1131 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1132 };
1133 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1134 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1135 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1136 dword(90) > 31)
1137 throw std::runtime_error("Invalid VGA snapshot fields");
1138 reset_vga();
1139 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1140 for (unsigned i = 0; i < 25; ++i)
1141 if (i != 0x11)
1142 m_crtc_map.write_byte(i, bytes[1 + i]);
1143 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1144 for (unsigned i = 0; i < 5; ++i)
1145 m_seq_map.write_byte(i, bytes[26 + i]);
1146 for (unsigned i = 0; i < 9; ++i)
1147 m_gc_map.write_byte(i, bytes[31 + i]);
1148 for (unsigned i = 0; i < 21; ++i)
1149 m_atc_map.write_byte(i, bytes[40 + i]);
1150 // Every register must read back exactly as saved.
1151 for (unsigned i = 0; i < 25; ++i)
1152 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1153 throw std::runtime_error("Invalid CRTC snapshot encoding");
1154 for (unsigned i = 0; i < 5; ++i)
1155 if (m_seq_map.read_byte(i) != bytes[26 + i])
1156 throw std::runtime_error("Invalid sequencer snapshot encoding");
1157 for (unsigned i = 0; i < 9; ++i)
1158 if (m_gc_map.read_byte(i) != bytes[31 + i])
1159 throw std::runtime_error(
1160 "Invalid graphics-controller snapshot encoding");
1161 for (unsigned i = 0; i < 21; ++i)
1162 if (m_atc_map.read_byte(i) != bytes[40 + i])
1163 throw std::runtime_error("Invalid attribute snapshot encoding");
1164 vga.crtc.index = bytes[61];
1165 vga.sequencer.index = bytes[62];
1166 vga.gc.index = bytes[63];
1167 vga.attribute.index = bytes[64];
1168 vga.miscellaneous_output = bytes[65];
1169 m_ioas = (bytes[65] & 1) != 0;
1170 vga.feature_control = bytes[66];
1171 m_vga_enable = bytes[67];
1172 vga.dac.mask = bytes[68];
1173 vga.attribute.state = bytes[69];
1174 vga.dac.read = bytes[70];
1175 vga.dac.state = bytes[71];
1176 for (unsigned i = 0; i < 3; ++i)
1177 vga.dac.loading[i] = bytes[72 + i];
1178 for (unsigned i = 0; i < 4; ++i)
1179 vga.gc.latch[i] = bytes[75 + i];
1180 vga.crtc.start_addr = dword(79);
1181 vga.attribute.pel_shift = bytes[83];
1182 vga.cursor.visible = bytes[84];
1183 vga.crtc.irq_latch = bytes[85];
1184 m_vga_scanline = dword(86);
1185 m_vga_status_phase = dword(90);
1186 m_crtc_io_base = u16(bytes[94] | (u16(bytes[95]) << 8));
1187 vga.dac.dirty = 1;
1188}

1190int CPowerStorm3xx::SaveState(FILE* f)
1191{
1192 if (!f)
1193 return -1;
1194 std::lock_guard<std::recursive_mutex> guard(
1195 cSystem->get_device_bus_mutex());
1196 try
1197 {
1198 std::ostringstream stream(std::ios::binary);
1199 m_realimage.SaveState(stream);
1200 const std::string bytes = stream.str();
1201 const std::vector<u8> vga_bytes = save_vga_state();
1202 const std::string identity = snapshot_identity();
1203 if (identity.size() > 8192 || vga_bytes.size() != VGAStateSize)
1204 return -1;
1205 if (CPCIDevice::SaveState(f))
1206 return -1;
1207 powerstorm_write_u32(
1208 f, 0x31475350); // PSG1: identity, VGA frontend, core
1209 powerstorm_write_u32(f, u32(identity.size()));
1210 if (fwrite(identity.data(), 1, identity.size(), f) != identity.size())
1211 return -1;
1212 powerstorm_write_u32(f, u32(vga_bytes.size()));
1213 if (fwrite(vga_bytes.data(), 1, vga_bytes.size(), f) !=
1214 vga_bytes.size())
1215 return -1;
1216 powerstorm_write_u32(f, u32(m_access_count));
1217 powerstorm_write_u32(f, u32(m_access_count >> 32));
1218 powerstorm_write_u32(f, u32(bytes.size()));

1220 return -1;
1221 return ferror(f) ? -1 : 0;
1222 }
1223 catch (const std::exception& e)
