33// EV68 cache predicates used by the family fetch and source-observation policy.
34bool Ev68InstructionFetch::icache_line_matches(const CAlphaCPU& cpu, int i, u64 address)
35{
36 const auto& line = cpu.state.icache[i];
37 return line.valid && (line.asn == cpu.state.asn || line.asm_bit)
38 && line.address == (address & ICACHE_MATCH_MASK);
39}

41bool Ev68InstructionFetch::icache_hit(const CAlphaCPU& cpu, int i, u64 address)
42{
43 return cpu.icache_enabled && icache_line_matches(cpu, i, address)
44 && ((address & 1) || (cpu.icache_valid_modes[i] & (1u << cpu.state.cm)));
45}

47void Ev68InstructionFetch::invalidate(CAlphaCPU& cpu, bool non_global)
48{
49 auto& state = cpu.state;
50 if (cpu.icache_enabled)
51 {
52 if (non_global)
53 {
54 int i;
55 for (i = 0; i < ICACHE_ENTRIES; i++)
56 if (!state.icache[i].asm_bit)
57 state.icache[i].valid = false;
58 }
59 else
60 {
61 for (int i = 0; i < ICACHE_ENTRIES; i++) {
62 state.icache[i].valid = false;
63 }
64 state.next_icache = 0; // old version, may be relied on elsewhere
65 state.last_found_icache = 0;
66 }
67 }
68 cpu.seq_remaining = 0;
69}

71void Ev68InstructionFetch::lifecycle(CAlphaCPU& cpu, AlphaInstructionFetchEvent event)
72{
73 switch (event)
74 {
75 case AlphaInstructionFetchEvent::Initialize:
76 cpu.seq_line_ptr = nullptr;
77 cpu.seq_offset = 0;
78 cpu.seq_remaining = 0;
79 cpu.seq_next_pc = 0;
80 break;
81 case AlphaInstructionFetchEvent::AdvancePC:
82 cpu.state.pc_phys += 4;
83 if (cpu.state.rem_ins_in_page)
84 cpu.state.rem_ins_in_page--;
85 break;
86 case AlphaInstructionFetchEvent::RedirectPC:
87 cpu.state.rem_ins_in_page = 0;
88 cpu.seq_remaining = 0;
89 break;
90 case AlphaInstructionFetchEvent::CompiledExit:
91 cpu.seq_remaining = 0;
92 break;
93 case AlphaInstructionFetchEvent::TranslationChanged:
94 case AlphaInstructionFetchEvent::Restore:
95 memset(cpu.icache_valid_modes, 0, sizeof(cpu.icache_valid_modes));
96 cpu.seq_remaining = 0;
97 cpu.state.rem_ins_in_page = 0;
98 break;
99 }
100}

102void Ev68InstructionFetch::prepare_dispatch(CAlphaCPU& cpu)
103{
104 // Preserve the ES40 firmware entry behavior after an in-place image rewrite.
105 if (cpu.state.pc == (cpu.state.pal_base | 1))
106 cpu.flush_icache();

109AlphaInstructionMapping Ev68InstructionFetch::probe(CAlphaCPU& cpu, u64 pc)
110{
111 AlphaInstructionMapping mapping;
112 if (pc & 1)
113 {
114 // PAL instruction addresses are physical; retain the dispatcher's alignment.
115 mapping.physical = pc & ~U64(1);
116 mapping.asm_global = true;
117 }
118 else
119 {
120 const int line = (int)(((pc & ~U64(3)) >> 11) & (ICACHE_ENTRIES - 1));
121 if (icache_hit(cpu, line, pc))
122 {
123 mapping.physical = cpu.state.icache[line].p_address + (pc & ICACHE_BYTE_MASK);
124 mapping.asm_global = cpu.state.icache[line].asm_bit;
125 }
126 else if (cpu.virt2phys(pc, &mapping.physical, ACCESS_EXEC | FAKE,
127 &mapping.asm_global, 0) != 0)
128 return mapping;
129 }
130 mapping.valid = true;
131 return mapping;
132}

134bool Ev68InstructionFetch::peek_cached(const CAlphaCPU& cpu, u32* instruction)
135{
136 if (cpu.state.pc == cpu.seq_next_pc && cpu.seq_remaining > 0 && cpu.seq_cm == cpu.state.cm)
137 {
138 *instruction = endian_32(cpu.seq_line_ptr[cpu.seq_offset]);
139 return true;
140 }
141 if (cpu.icache_enabled)
142 {
143 const int line = (int)((cpu.state.pc >> 11) & (ICACHE_ENTRIES - 1));
144 if (icache_hit(cpu, line, cpu.state.pc))
145 {
146 *instruction = endian_32(cpu.state.icache[line].data[(cpu.state.pc >> 2) & ICACHE_INDEX_MASK]);
147 return true;
148 }
149 }
150 return false;
151}

153bool Ev68InstructionFetch::matches_interpreted(const CAlphaCPU& cpu, u64 start_pc, u32 count)
154{
155 if (cpu.icache_enabled)
156 {
157 const u64 vs = start_pc & ~U64(3);
158 const u64 ve = vs + 4 * (u64)count;
159 for (u64 v = vs & ~U64(0x7ff); v < ve; v += 0x800)
160 {
161 const int li = (int)((v >> 11) & (ICACHE_ENTRIES - 1));
162 const auto& line = cpu.state.icache[li];
163 // Compare cached bytes even if the current mode has no cached permission.
164 // This deliberately differs from the execution/peek hit predicate.
165 if (line.valid && (line.asn == cpu.state.asn || line.asm_bit)
166 && line.address == ((v | (start_pc & 1)) & ICACHE_MATCH_MASK))
167 {
168 const u64 lo = (v > vs) ? v : vs;
169 const u64 hi = (v + 0x800 < ve) ? v + 0x800 : ve;
170 const u64 poff = line.p_address + (lo - v);
171 if (poff + (hi - lo) <= cpu.dram_size
172 && memcmp((const uint8_t*)cpu.dram_ptr + poff,
173 (const uint8_t*)line.data + (lo - v), (size_t)(hi - lo)))
174 return false;
175 }
176 }
177 }
178 return true;
179}

187AlphaInstructionSourceView Ev68InstructionFetch::source(CAlphaCPU& cpu, u64 pc)
188{
189 const auto mapping = probe(cpu, pc);
190 if (!mapping.valid || !cpu.dram_ptr || (mapping.physical & 3)
191 || mapping.physical >= cpu.dram_size)
192 return {};
193
194 // The EV68 JIT uses backing RAM so its interpreted-span check separately rejects
195 // cached insn that differ. Never cross the current 8 KiB translation.
196 const u64 ram_words = (cpu.dram_size - mapping.physical) / sizeof(u32);
197 const u64 page_words = (U64(0x2000) - (mapping.physical & U64(0x1fff))) / sizeof(u32);
198 const u32 count = (u32)((ram_words < page_words) ? ram_words : page_words);
199 if (!count) return {};
200 return {reinterpret_cast<const u8*>(cpu.dram_ptr) + mapping.physical,
201 mapping.physical, count, mapping.asm_global};
202}

204int Ev68InstructionFetch::fetch(CAlphaCPU& cpu, u32* instruction)
205{
206 if (cpu.state.pc == cpu.seq_next_pc && cpu.seq_remaining > 0 && cpu.seq_cm == cpu.state.cm)
207 {
208 *instruction = endian_32(cpu.seq_line_ptr[cpu.seq_offset]);
209#if defined(DEBUG_ARC) && !defined(ES40_JIT)
210 if (cpu.state.current_pc < 0x10000)
211 printf("PC=%016" PRIx64 " ins=%08x\n", cpu.state.current_pc, *instruction);
212#endif
213 cpu.seq_offset++;
214 cpu.seq_remaining--;
215 cpu.seq_next_pc += 4;
216 }
217 else
218 {
219 prepare_dispatch(cpu);
220 const int result = read_cache(cpu, cpu.state.pc, instruction);
221 if (result)
222 return result;
223
224 if (!(cpu.state.pc & 1) && icache_hit(cpu, cpu.state.last_found_icache, cpu.state.pc))
225 {
226 const int line = cpu.state.last_found_icache;
227 cpu.seq_line_ptr = cpu.state.icache[line].data;
228 cpu.seq_offset = (int)((cpu.state.pc >> 2) & ICACHE_INDEX_MASK) + 1;
229 cpu.seq_remaining = ICACHE_LINE_SIZE - cpu.seq_offset;
230 cpu.seq_next_pc = (cpu.state.pc & ~U64(0x3)) + 4;
231 cpu.seq_cm = cpu.state.cm;
232 }
233 else
234 cpu.seq_remaining = 0;
235 }
236
237#if defined(IDB)
238 cpu.current_pc_physical = cpu.state.pc_phys;
239#endif
240 return 0;
241}

263int Ev68InstructionFetch::read_cache(CAlphaCPU& cpu, u64 address, u32* data)
264{
265 // Direct-map the icache: 2 KiB lines (ICACHE_LINE_SIZE * 4 == 2048 bytes).
266 // Use VA[...:11] as the set index. The PAL bit (VA<0>) remains part of the tag.
267 const u64 kLineShift = 11;
268 const u64 v_aligned = address & ~U64(0x3);
269 const int i = (int)((v_aligned >> kLineShift) & (ICACHE_ENTRIES - 1));
270 u64 v_a;
271 u64 p_a;
272 int result;
273 bool asm_bit;
274
275 if (cpu.icache_enabled)
276 {
277 bool hit = icache_line_matches(cpu, i, address);
278 if (hit && !(address & 1) && !(cpu.icache_valid_modes[i] & (1u << cpu.state.cm))) {
279 // A cached instruction is still subject to the current mode's access
280 // checks, including for ASM lines. Keep its bytes until IMB when the
281 // translation is unchanged; SRM can depend on the cached instruction.
282 result = cpu.virt2phys(address, &p_a, ACCESS_EXEC, &asm_bit, 0);
283 if (result) return result;
284 hit = p_a == cpu.state.icache[i].p_address + (address & ICACHE_BYTE_MASK);
285 if (hit) {
286 if (asm_bit != cpu.state.icache[i].asm_bit) {
287 // A newly checked process-specific mapping must not grant access
288 // to other ASNs through an older global cache line (or vice versa).
289 cpu.icache_valid_modes[i] = 0;
290 cpu.state.icache[i].asn = cpu.state.asn;
291 cpu.state.icache[i].asm_bit = asm_bit;
292 }
293 cpu.icache_valid_modes[i] |= (u8)(1u << cpu.state.cm);
294 }
295 }
296
297 // ---- Fast hit probe
298 if (hit) {
299
300 *data = endian_32(cpu.state.icache[i].data[(address >> 2) & ICACHE_INDEX_MASK]);
301
302 // keep debug/pc_phys coherent even when icache is enabled
303 cpu.state.pc_phys = cpu.state.icache[i].p_address + (address & ICACHE_BYTE_MASK);
304
305#ifdef IDB
306 cpu.current_pc_physical = cpu.state.pc_phys;
307#endif
308 cpu.state.last_found_icache = i;
309 return 0;
310 }
311
312 // ---- Miss: translate + fill
313 v_a = address & ICACHE_MATCH_MASK;
314 if (address & 1) {
315 // PALmode: VA<0> is PAL marker; physical is VA with bit0 cleared
316 p_a = v_a & ~U64(0x1);
317 asm_bit = true;
318 }
319 else {
320 result = cpu.virt2phys(v_a, &p_a, ACCESS_EXEC, &asm_bit, 0);
321 if (result) return result;
322 }
323
324 // Attempt to get a pointer into DRAM. If this is PIO (e.g., TIG flash),
325 // PtrToMem returns null and we must *not* try to memcpy from it.
326 char* mem = cpu.cSystem->PtrToMem(p_a);
327 if (mem && cpu.cSystem->PtrToMem(p_a + ((ICACHE_LINE_SIZE * 4) - 1))) {
328 // DRAM-backed: fill the direct-mapped icache line.
329 memcpy(cpu.state.icache[i].data, mem, ICACHE_LINE_SIZE * 4);
330 cpu.state.icache[i].valid = true;
331 cpu.state.icache[i].asn = cpu.state.asn;
332 cpu.state.icache[i].asm_bit = asm_bit;
333 cpu.state.icache[i].address = address & ICACHE_MATCH_MASK;
334 cpu.state.icache[i].p_address = p_a;
335 cpu.icache_valid_modes[i] = (u8)(1u << cpu.state.cm);
336
337 *data = endian_32(cpu.state.icache[i].data[(address >> 2) & ICACHE_INDEX_MASK]);
338
339 // same pc_phys update on fill
340 cpu.state.pc_phys = p_a + (address & ICACHE_BYTE_MASK);
341#ifdef IDB
342 cpu.current_pc_physical = cpu.state.pc_phys;
343#endif
344 cpu.state.last_found_icache = i;
345 return 0;
346 }
347 else {
348 // PIO/TIG-backed: cannot fill icache lines.
349 // Read exactly the requested instruction as 4 byte reads via the system bus.
350 const u64 p_instr = p_a + (address & ICACHE_BYTE_MASK);
351 u32 ins = 0;
352 ins |= (u8)cpu.cSystem->ReadMem(p_instr + 0, 8, &cpu);
353 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 1, 8, &cpu)) << 8;
354 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 2, 8, &cpu)) << 16;
355 ins |= ((u8)cpu.cSystem->ReadMem(p_instr + 3, 8, &cpu)) << 24;
356 *data = ins; // already in target little-endian form
357
358 cpu.state.pc_phys = p_instr;
359#ifdef IDB
360 cpu.current_pc_physical = cpu.state.pc_phys;
361#endif
362 return 0;
363 }
364 }
365
366 // ---- Icache disabled (unchanged)
367 if (address & 1) {
368 cpu.state.pc_phys = address & ~U64(0x3);
369 cpu.state.rem_ins_in_page = 1;
370 }
371 else {
372 if (!cpu.state.rem_ins_in_page || cpu.ins_page_cm != cpu.state.cm) {
373 result = cpu.virt2phys(address, &cpu.state.pc_phys, ACCESS_EXEC, &asm_bit, 0);
374 if (result) return result;
375 cpu.state.rem_ins_in_page = 2048 - ((((u32)address) >> 2) & 2047);
376 cpu.ins_page_cm = cpu.state.cm;
377 }
378 }
379
380 *data = (u32)cpu.cSystem->ReadMem(cpu.state.pc_phys, 32, &cpu);
381 return 0;
382}
