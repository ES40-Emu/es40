70 FAILURE_1(Configuration, "%s: Disk has no device attached!\n", devid_string);
71 }
72
73 if (read_only)
74 {
75#if defined(_WIN32)
76 buffer = (char*)malloc(2048);
77 buffer_size = 2048;
78 handle = CreateFile(filename, GENERIC_READ, FILE_SHARE_READ, NULL,
79 OPEN_EXISTING,
80 FILE_ATTRIBUTE_NORMAL | FILE_FLAG_RANDOM_ACCESS, NULL);

105 {
106 FAILURE_2(Runtime, "%s: Could not open device %s.", devid_string, filename);

110 // determine size...
111#if defined(_WIN32)

122 sectors = x.SectorsPerTrack;
123 heads = x.TracksPerCylinder;
124 byte_size = x.Cylinders.QuadPart *

133#else
134 fseek_large(handle, 0, SEEK_END);
135 byte_size = ftell_large(handle);
136 fseek_large(handle, 0, SEEK_SET);
137 state.byte_pos = ftell_large(handle);

139 sectors = 32;
140 heads = 8;
141#endif
142
143 //calc_cylinders();
144 determine_layout();
145
146 model_number = myCfg->get_text_value("model_number", filename);
147
148 printf("%s: Mounted device %s, %" PRId64 " %zu-byte blocks, %" PRId64 "/%ld/%ld.\n",
149 devid_string, filename, byte_size / state.block_size, state.block_size,
150 cylinders, heads, sectors);

203size_t CDiskDevice::read_bytes(void* dest, size_t bytes)
204{
205 if (is_offline())
206 return 0;
207
208 // printf("%s: read %d bytes @ %" PRId64 ".\n",devid_string,bytes,state.byte_pos);
209#if defined(_WIN32)
210 off_t_large byte_from = (state.byte_pos / dev_block_size) * dev_block_size;
211 off_t_large byte_to =
212 (
213 ((state.byte_pos + bytes - 1) / dev_block_size) +
214 1
215 ) *
216 dev_block_size;
217 DWORD byte_len = (DWORD)(byte_to - byte_from);
218 DWORD byte_off = (DWORD)(state.byte_pos - byte_from);
219 LARGE_INTEGER a;
220 DWORD r;
221
222 if (byte_len > buffer_size)
223 {
224 buffer_size = byte_len;
225 CHECK_REALLOCATION(buffer, realloc(buffer, buffer_size), char);
226
227 // printf("%s: buffer enlarged to %d bytes.\n",devid_string,buffer_size);
228 }
229
230 a.QuadPart = byte_from;
231 SetFilePointerEx(handle, a, NULL, FILE_BEGIN);
232
233 ReadFile(handle, buffer, byte_len, &r, NULL);
234
235 if (r != (byte_len))
236 {
237 printf("%s: Tried to read %lu bytes from pos %" PRId64
238 "d, but could only read %lu bytes!\n", devid_string, byte_len,
239 byte_from, r);
240 printf("%s: Error %ld.\n", devid_string, GetLastError());
241 }
242
243 memcpy(dest, buffer + byte_off, bytes);
244 state.byte_pos += bytes;
245 return bytes;
246#else
247 size_t r;
248 r = fread(dest, 1, bytes, handle);
249 state.byte_pos = ftell_large(handle);
250 return r;
251#endif
252}

254size_t CDiskDevice::write_bytes(void* src, size_t bytes)
255{
256 if (is_offline())
257 return 0;
258 if (read_only)
259 return 0;
260
261#if defined(_WIN32)
262 off_t_large byte_from = (state.byte_pos / dev_block_size) * dev_block_size;
263 off_t_large byte_to =
264 (
265 ((state.byte_pos + bytes - 1) / dev_block_size) +
266 1
267 ) *
268 dev_block_size;
269 DWORD byte_len = (DWORD)(byte_to - byte_from);
270 DWORD byte_off = (DWORD)(state.byte_pos - byte_from);
271 LARGE_INTEGER a;
272 DWORD r;
273
274 if (byte_len > buffer_size)
275 {
276 buffer_size = byte_len;
277 CHECK_REALLOCATION(buffer, realloc(buffer, buffer_size), char);
278 }
279
280 if (byte_from != state.byte_pos)
281 {
282
283 // we don't write the entire first block, so we read it
284 // from disk first so we don't corrupt the disk
285 a.QuadPart = byte_from;
286 SetFilePointerEx(handle, a, NULL, FILE_BEGIN);
287 ReadFile(handle, buffer, (DWORD)dev_block_size, &r, NULL);
288 if (r != (dev_block_size))
289 {
290 printf("%s: Tried to read %zu bytes from pos %" PRId64
291 "d, but could only read %lu bytes!\n", devid_string,
292 dev_block_size, byte_from, r);
293 FAILURE(InvalidArgument,
294 "Error during device write operation. Terminating to avoid disk corruption.");
295 }
296 }
297
298 // Skip the trailing read only when the first read already preserved this sector.
299 if ((byte_to != state.byte_pos + bytes) &&
300 ((byte_from == state.byte_pos) || (byte_to - byte_from > dev_block_size)))
301 {
302
303 // we don't write the entire last block, so we read it
304 // from disk first so we don't corrupt the disk
305 a.QuadPart = byte_to - dev_block_size;
306 SetFilePointerEx(handle, a, NULL, FILE_BEGIN);
307 ReadFile(handle, buffer + byte_len - dev_block_size, (DWORD)dev_block_size,
308 &r, NULL);
309 if (r != (dev_block_size))
310 {
311 printf("%s: Tried to read %zu bytes from pos %" PRId64
312 "d, but could only read %lu bytes!\n", devid_string,
313 dev_block_size, byte_to - dev_block_size, r);
314 FAILURE(InvalidArgument,
315 "Error during device write operation. Terminating to avoid disk corruption.");
316 }
317 }
318
319 // add the data we're writing to the buffer
320 memcpy(buffer + byte_off, src, bytes);
321
322 a.QuadPart = byte_from;
323 SetFilePointerEx(handle, a, NULL, FILE_BEGIN);
324
325 // and write the buffer to disk
326 WriteFile(handle, buffer, byte_len, &r, NULL);
327
328 if (r != byte_len)
329 {
330 printf("%s: Tried to write %lu bytes to pos %" PRId64
331 "d, but could only write %lu bytes!\n", devid_string, byte_len,
332 byte_from, r);
333 FAILURE(InvalidArgument,
334 "Error during device write operation. Terminating to avoid disk corruption.");
335 }
336
337 state.byte_pos += bytes;
338 return bytes;
339#else
340 size_t r;
341 r = fwrite(src, 1, bytes, handle);
342 state.byte_pos = ftell_large(handle);
343 return r;
344#endif
345}
