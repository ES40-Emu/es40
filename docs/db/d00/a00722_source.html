514void CSystem::allocate_ll_watch()
515{
516 const size_t lines = (size_t)((U64(1) << iNumMemoryBits) >> 6);
517 const size_t pages = (size_t)((U64(1) << iNumMemoryBits) >> 13);
518 try
519 {
520 auto watched = std::make_unique<std::atomic<u8>[]>(lines);
521 auto watched_pages = std::make_unique<std::atomic<u8>[]>(pages);
522 for (size_t line = 0; line < lines; ++line)
523 watched[line].store(0, std::memory_order_relaxed);
524 for (size_t page = 0; page < pages; ++page)
525 watched_pages[page].store(0, std::memory_order_relaxed);
526 m_ll_watch = std::move(watched);
527 m_ll_watch_pages = std::move(watched_pages);
528 // Drop reservations tied to the old RAM allocation.
529 state.cpu_lock_flags.store(0, std::memory_order_relaxed);
530 clear_ll_reservations();
531 }
532 catch (const std::bad_alloc&)
533 {
534 FAILURE(OutOfMemory, "Unable to allocate the RAM load-locked watch map");
535 }
536}

603{

606 for (int i = 0; i < iNumComponents; ++i)
607 {

610 if (!provider)

612
613 std::unordered_set<unsigned> output_ids;
614 const std::size_t count = provider->output_count();
615 for (std::size_t ordinal = 0; ordinal < count; ++ordinal)
616 {
617 const CDisplayOutput* output = provider->output_at(ordinal);
618 if (!output)
619 FAILURE_2(Runtime, "Missing display output %zu for %s", ordinal,
620 component->devid_string);

622 FAILURE_2(Runtime, "Duplicate display output %u for %s", output->id(),

624 const SDisplayOutput binding = { component, output };
625 const auto inserted = displays.emplace(&output->display(), binding);
626 if (!inserted.second)
627 {
628 const auto& previous = inserted.first->second;
629 FAILURE_4(Runtime, "Display outputs %s/%u and %s/%u share a GUI",
630 previous.component->devid_string, previous.output->id(),
631 component->devid_string, output->id());
632 }
633 outputs.push_back(binding);
634 }

742/*
743 * AlphaBIOS does not create an EBDA. Reserve one KiB at the conventional
744 * memory boundary before an x86 option ROM reads BAR6.
745 */
746void CSystem::PrepareX86OptionROMMemory()
747{
748 static const u64 BDA_EBDA_SEGMENT = U64(0x0000040e);
749 static const u64 BDA_MEMORY_SIZE = U64(0x00000413);
750 static const u64 BDA_TIMER_TICKS = U64(0x0000046c);
751 static const u32 ALPHABIOS_IVT_ENTRY = 0xfdec0008U;
752 static const u32 ALPHABIOS_IVT_08 = 0xfdec0020U;
753 static const u32 MAXIMUM_CONVENTIONAL_MEMORY_KB = 576U;
754 static const u32 MINIMUM_CONVENTIONAL_MEMORY_KB = 128U;
755 static const u32 EBDA_BLOCK_SIZE = 1024U;
756 static const u32 EBDA_SIZE_KB = 1U;
757
758 if (iNumMemoryBits < 20 ||
759 ReadMem(0x0000, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
760 ReadMem(0x0004, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
761 ReadMem(0x0020, 32, nullptr) != ALPHABIOS_IVT_08)
762 return;
763
764 u32 conventional_memory_kb =
765 (u32)ReadMem(BDA_MEMORY_SIZE, 16, nullptr);
766 if (conventional_memory_kb < MINIMUM_CONVENTIONAL_MEMORY_KB ||
767 conventional_memory_kb > MAXIMUM_CONVENTIONAL_MEMORY_KB)
768 return;
769
770 if (ReadMem(BDA_EBDA_SEGMENT, 16, nullptr) == 0)
771 {
772 const u64 ebda_address = (u64)conventional_memory_kb * 1024U;
773 for (u64 address = ebda_address;
774 address < ebda_address + EBDA_BLOCK_SIZE;
775 ++address)
776 {
777 if (ReadMem(address, 8, nullptr) != 0)
778 return;
779 }
780
781 WriteMem(BDA_EBDA_SEGMENT, 16, ebda_address >> 4, nullptr);
782 WriteMem(ebda_address, 8, EBDA_SIZE_KB, nullptr);
783 }
784
785 if (!m_x86_bios_clock_active.load(std::memory_order_acquire))
786 {
787 m_x86_bios_clock_initial_ticks =
788 (u32)ReadMem(BDA_TIMER_TICKS, 32, nullptr);
789 m_x86_bios_clock_elapsed_ticks = 0;
790 m_x86_bios_clock_epoch = std::chrono::steady_clock::now();
791 m_x86_bios_clock_active.store(true, std::memory_order_release);
792 }
793}

795/* AlphaBIOS's x86 interpreter does not advance the 18.2 Hz BDA clock. */
796void CSystem::UpdateX86BIOSClock()
797{
798 static const u64 BDA_TIMER_TICKS = U64(0x0000046c);
799 static const u64 BDA_TIMER_ROLLOVER = U64(0x00000470);
800 static const u32 ALPHABIOS_IVT_ENTRY = 0xfdec0008U;
801 static const u32 BIOS_TICKS_PER_DAY = 0x001800b0U;
802
803 if (!m_x86_bios_clock_active.load(std::memory_order_acquire))
804 return;
805
806 const u64 elapsed_ms = (u64)std::chrono::duration_cast<
807 std::chrono::milliseconds>(std::chrono::steady_clock::now() -
808 m_x86_bios_clock_epoch).count();
809 const u64 elapsed_ticks = elapsed_ms * 182U / 10000U;
810 if (elapsed_ticks == m_x86_bios_clock_elapsed_ticks)
811 return;
812
813 if (ReadMem(0x0000, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
814 ReadMem(0x0004, 32, nullptr) != ALPHABIOS_IVT_ENTRY)
815 {
816 m_x86_bios_clock_active.store(false, std::memory_order_release);
817 return;
818 }
819
820 const u64 previous_ticks = m_x86_bios_clock_initial_ticks +
821 m_x86_bios_clock_elapsed_ticks;
822 const u64 current_ticks = m_x86_bios_clock_initial_ticks + elapsed_ticks;
823 WriteMem(BDA_TIMER_TICKS, 32, current_ticks % BIOS_TICKS_PER_DAY, nullptr);
824
825 const u64 rollovers = current_ticks / BIOS_TICKS_PER_DAY -
826 previous_ticks / BIOS_TICKS_PER_DAY;
827 if (rollovers != 0)
828 {
829 const u32 rollover = (u32)ReadMem(BDA_TIMER_ROLLOVER, 8, nullptr);
830 WriteMem(BDA_TIMER_ROLLOVER, 8, rollover + rollovers, nullptr);
831 }
832 m_x86_bios_clock_elapsed_ticks = elapsed_ticks;
833}

1104void CSystem::watch_ll_line(int cpuid, u64 phys)
1105{
1106#ifndef ES40_JIT
1107 // Interpreter SMP stores already take the bucket lock.
1108 (void)cpuid; (void)phys;
1109 return;
1110#else
1111 if (iNumCPUs <= 1) return;
1112 const size_t line = (size_t)(phys >> 6);
1113 if (m_ll_watch[line].load(std::memory_order_acquire))
1114 return;
1115
1116 // Clear active before waiting: the mutex owner may be waiting for this CPU.
1117 const bool resume = m_ll_jit_cpu[cpuid].depth != 0;
1118 if (resume)
1119 m_ll_jit_cpu[cpuid].active.store(false, std::memory_order_seq_cst);
1120 try
1121 {
1122 std::lock_guard<std::mutex> setup(m_ll_watch_setup);
1123 if (!m_ll_watch[line].load(std::memory_order_acquire))
1124 {
1125 struct FreezeGuard
1126 {
1127 std::atomic<bool>& frozen;
1128 explicit FreezeGuard(std::atomic<bool>& flag) : frozen(flag)
1129 { frozen.store(true, std::memory_order_seq_cst); }
1130 ~FreezeGuard() { frozen.store(false, std::memory_order_seq_cst); }
1131 } freeze(m_ll_jit_freeze);
1132
1133 // Wait for stores already in flight. No bucket or device lock is held.
1134 for (int cpu = 0; cpu < iNumCPUs; ++cpu)
1135 while (m_ll_jit_cpu[cpu].active.load(std::memory_order_seq_cst))
1136 std::this_thread::yield();
1137 m_ll_watch[line].store(1, std::memory_order_release);
1138 auto& watched = m_ll_watch_pages[(size_t)(phys >> 13)];
1139 if (!watched.load(std::memory_order_relaxed))
1140 {
1141 watched.store(1, std::memory_order_release);
1142 // Mark every cached write alias before letting CPUs run again.
1143 for (int cpu = 0; cpu < iNumCPUs; ++cpu)
1144 acCPUs[cpu]->promote_write_dpc_page(phys);
1145 }
1146 }
1147 }
1148 catch (...)
1149 {
1150 // Re-enter execution before propagating the exception.
1151 if (resume)
1152 enter_ll_jit(cpuid);
1153 throw;
1154 }
1155 if (resume)
1156 enter_ll_jit(cpuid);
1157#endif
1158}

1161// invalidates the reservation. Serialize the RAM access and its generation
1162// update with both LDx_L's read/publication and STx_C's check/store.
1163void CSystem::cpu_lock_bucket(u32 bucket)
1164{
1165 for (unsigned spins = 0; m_ll_buckets[bucket].lock.exchange(1, std::memory_order_acquire); ++spins)
1166 if ((spins & 63) == 63)

1168}

1257bool CSystem::cpu_take_lock(int cpuid, u64 address, u64* expected, bool* same_address)
1258{
1259 auto& reservation = m_ll_reservations[cpuid];
1260 // I/O-space conditional stores have no cache line to watch; treat as held.
1261 bool held = (address & CPU_LOCK_IO_MASK) ||
1262 (reservation.valid && cpu_lock_matches(reservation.address, address));
1263
1264 // STx_C always consumes this CPU's lock, success or fail.
1265 reservation.valid = false;
1266 if (held)
1267 {
1268 *expected = reservation.value;
1269 *same_address = (reservation.address == address);
1270 }
1271 return held;

1277u64 CSystem::cpu_stx_c(int cpuid, u64 phys, int size_bits, u64 value,
1278 char* dram, u64 dram_sz, CSystemComponent* source)
1279{
1280 phys &= U64(0x00000807ffffffff);
1281 u64 expected = 0;
1282 bool same_address = false;
1283 if (!cpu_take_lock(cpuid, phys, &expected, &same_address))
1284 return 0;
1285
1286 if (phys >= dram_sz)
1287 {
1288 WriteMem(phys, size_bits, value, source); // I/O-space conditional store
1289 return 1;
1290 }
1291
1292 const u32 b = (u32)((phys >> 6) & (kLLBuckets - 1));
1293 CLLRAMGuard guard(this, phys);
1294 u64 ok;
1295 if (m_ll_buckets[b].sequence.load(std::memory_order_relaxed) != m_ll_reservations[cpuid].sequence)
1296 ok = 0; // a store hit this reservation bucket since our LDx_L
1297 else if (same_address)
1298 ok = dram_cas(dram, phys, expected, value, size_bits) ? 1 : 0;
1299 else
1300 {
1301 // STx_C to a different quadword in the locked line: no value to compare.
1302 dram_write(dram, phys, size_bits, value);
1303 ok = 1;
1304 }
1305 if (ok)
1306 guard.invalidate();
1307 return ok;
1308}

1319bool CSystem::idle_watch_arm(int cpuid, const u64* lines, int n)
1320{
1321 idle_watch_disarm(cpuid);
1322 if (iNumCPUs <= 1 || n <= 0 || n > kIdleWatchLines)
1323 return false;
1324 const u64 ram_lines = (U64(1) << iNumMemoryBits) >> 6;
1325 for (int i = 0; i < n; ++i)
1326 if (lines[i] >= ram_lines)
1327 return false;
1328 // JIT stores to an unwatched line skip the bucket, and its sequence.
1329 for (int i = 0; i < n; ++i)
1330 watch_ll_line(cpuid, lines[i] << 6);
1331 IdleWatch& w = m_idle_watch[cpuid];
1332 for (int i = 0; i < n; ++i)
1333 {
1334 const u32 b = (u32)(lines[i] & (kLLBuckets - 1));
1335 w.bucket[i].store(b, std::memory_order_relaxed);
1336 w.sequence[i].store(m_ll_buckets[b].sequence.load(std::memory_order_acquire),
1337 std::memory_order_relaxed);
1338 }
1339 w.n.store(n, std::memory_order_relaxed);
1340 m_idle_watchers.fetch_or(1u << cpuid, std::memory_order_seq_cst);
1341 return true;
1342}

1349// Also true once another CPU has disarmed this one to wake it.
1350bool CSystem::idle_watch_fired(int cpuid) const
1351{
1352 if (!(m_idle_watchers.load(std::memory_order_seq_cst) & (1u << cpuid)))
1353 return true;
1354 const IdleWatch& w = m_idle_watch[cpuid];
1355 const int n = w.n.load(std::memory_order_relaxed);
1356 for (int i = 0; i < n && i < kIdleWatchLines; ++i)
1357 if (m_ll_buckets[w.bucket[i].load(std::memory_order_relaxed)].sequence.load(std::memory_order_acquire)
1358 != w.sequence[i].load(std::memory_order_relaxed))
1359 return true;
1360 return false;
1361}

1363void CSystem::idle_watch_wake(int cpuid)
1364{
1365 u32 others = m_idle_watchers.load(std::memory_order_acquire) & ~(1u << cpuid);
1366 for (int cpu = 0; others; ++cpu, others >>= 1)
1367 {
1368 if (!(others & 1) || !idle_watch_fired(cpu))
1369 continue;
1370 // Disarm first so later chains don't notify again before it runs.
1371 if (m_idle_watchers.fetch_and(~(1u << cpu), std::memory_order_seq_cst) & (1u << cpu))
1372 acCPUs[cpu]->wake_idle();
1373 }
1374}

1379void CSystem::recompute_range_overlaps()
1380{

1382 return fallback->component->memory_decode_fallback(fallback->index) &&
1383 fallback->component->memory_decode_owner() ==
1384 specific->component->memory_decode_owner();
1385 };

1388 for (int i = 0; i < iNumMemories; ++i)

1400 if (!yields_to(b, a))

1402 }
1403 }

1406// Called with device_bus_mutex held, before any handler runs.
1407// Positive decode precedes subtractive decode.
1408// A bridge forwarding claim yields to that bridge's specific handler.
1409// Distinct remaining components are claimants.
1410void CSystem::collect_decode_claims(u64 address, int dsize, bool write,
1411 SDecodeClaims& claims, const CSystemComponent* source) const
1412{
1413 claims.count = 0;
1414 claims.subtractive = false;
1415 auto eligible = [&](const SMemoryUser* r) {
1416 return address >= r->base && address < r->base + r->length &&
1417 r->component->decodes_memory_access(r->index, address - r->base, dsize, write);
1418 };
1419
1420 int first = 0;
1421 while (first < iNumMemories && !eligible(asMemories[first].get()))
1422 ++first;
1423 if (first == iNumMemories)
1424 return;
1425 if (!asMemories[first]->may_overlap)
1426 {
1427 claims.range[claims.count++] = *asMemories[first];
1428 return;
1429 }
1430
1431 int positive[SDecodeClaims::kMax * 4];
1432 int subtractive[SDecodeClaims::kMax * 4];
1433 int npos = 0, nsub = 0;
1434 for (int i = first; i < iNumMemories; ++i)
1435 {
1436 const SMemoryUser* r = asMemories[i].get();
1437 if (!eligible(r))
1438 continue;
1439 const bool sub = r->component->uses_subtractive_decode(r->index,
1440 address - r->base, dsize, write);
1441 int* list = sub ? subtractive : positive;
1442 int& n = sub ? nsub : npos;
1443 if (n == SDecodeClaims::kMax * 4)
1444 {
1445 SDecodeClaims partial;
1446 partial.subtractive = sub;
1447 for (int k = 0; k < SDecodeClaims::kMax; ++k)
1448 partial.range[partial.count++] = *asMemories[list[k]];
1449 report_decode_overlap(address, dsize, write, partial, source,
1450 "too many eligible ranges to resolve");
1451 }
1452 list[n++] = i;
1453 }
1454 const int* list = npos ? positive : subtractive;
1455 const int n = npos ? npos : nsub;
1456 claims.subtractive = npos == 0;
1457
1458 for (int k = 0; k < n; ++k)
1459 {
1460 const SMemoryUser* r = asMemories[list[k]].get();
1461 if (r->component->memory_decode_fallback(r->index))
1462 {
1463 bool yields = false;
1464 for (int m = 0; m < n && !yields; ++m)
1465 {
1466 const SMemoryUser* o = asMemories[list[m]].get();
1467 yields = o->component->memory_decode_owner() == r->component->memory_decode_owner() &&
1468 !o->component->memory_decode_fallback(o->index);
1469 }
1470 if (yields)
1471 continue;
1472 }
1473 bool seen = false;
1474 for (int m = 0; m < claims.count && !seen; ++m)
1475 seen = claims.range[m].component == r->component;
1476 if (seen)
1477 continue; // same-component alias
1478 if (claims.count == SDecodeClaims::kMax)
1479 report_decode_overlap(address, dsize, write, claims, source,
1480 "too many claimants to resolve");
1481 claims.range[claims.count++] = *r;
1482 }
1483}

1561// every claimant, reads must agree. This does not model electrical timing.
1562void CSystem::prepare_shared_access(u64 address, int dsize, bool write,
1563 const SDecodeClaims& claims, const CSystemComponent* source)
1564{
1565 if (stop_on_decode_conflict)
1566 report_decode_overlap(address, dsize, write, claims, source,
1567 "debug.stop_on_decode_conflict");
1568 if (claims.subtractive)
1569 report_decode_overlap(address, dsize, write, claims, source,
1570 "several subtractive responders");
1571 const bool valid_width = dsize == 8 || dsize == 16 || dsize == 32 || dsize == 64;
1572 auto within = [&](u64 base, u64 length) {
1573 return valid_width && address >= base && address < base + length &&
1574 (u64)(dsize / 8) <= base + length - address;

1595// Every claimant performs the read before one value is chosen.
1596// Disagreement is undefined contention on real hardware.
1597u64 CSystem::resolve_shared_read(u64 address, int dsize,
1598 const SDecodeClaims& claims, const CSystemComponent* source)
1599{
1600 const u64 mask = dsize == 64 ? ~U64(0) : (U64(1) << dsize) - 1;
1601 u64 values[SDecodeClaims::kMax];
1602 std::string contexts[SDecodeClaims::kMax];
1603 bool agree = true;
1604 for (int k = 0; k < claims.count; ++k)
1605 {
1606 const SMemoryUser* r = &claims.range[k];
1607 // Capture register context before the read's own side effects.
1608 contexts[k] = r->component->describe_access_context(r->index,
1609 address - r->base);
1610 values[k] = r->component->ReadMem(r->index, address - r->base, dsize) & mask;
1611 agree &= values[k] == values[0];
1612 }
1613 if (agree)
1614 return values[0];
1615
1616 u64 anded = mask, ored = 0, differ = 0;
1617 for (int k = 0; k < claims.count; ++k)
1618 {
1619 anded &= values[k];
1620 ored |= values[k];
1621 differ |= values[k] ^ values[0];
1622 }
1623 std::ostringstream key, detail;
1624 key << std::hex << address << ':' << dsize << ':' << differ;
1625 detail << "claimant values differ, mask 0x" << std::hex << differ
1626 << ", policy ";
1627 switch (shared_read_policy)
1628 {
1629 case SharedReadPolicy::Stop: detail << "stop"; break;
1630 case SharedReadPolicy::Claimant: detail << "claimant " << std::dec
1631 << shared_read_hose << ':' << shared_read_device; break;
1632 case SharedReadPolicy::And: detail << "and"; break;
1633 case SharedReadPolicy::Or: detail << "or"; break;
1634 }
1635 for (int k = 0; k < claims.count; ++k)
1636 {
1637 detail << "\n " << claims.range[k].component->devid_string
1638 << " returned 0x" << std::hex << values[k];
1639 if (!contexts[k].empty())
1640 detail << " [" << contexts[k] << ']';
1641 key << ':' << contexts[k];
1642 }
1643 if (shared_read_policy == SharedReadPolicy::Stop)
1644 report_decode_overlap(address, dsize, false, claims, source,
1645 "claimants returned different read data", detail.str());
1646
1647 std::ostringstream line;
1648 line << "read " << std::dec << dsize << "-bit 0x" << std::hex << address
1649 << " from " << describe_source(source) << ": " << detail.str();
1650 log_shared_event("contention:" + key.str(), line.str());
1651
1652 switch (shared_read_policy)
1653 {
1654 case SharedReadPolicy::And: return anded;
1655 case SharedReadPolicy::Or: return ored;
1656 default: break;
1657 }
1658 for (int k = 0; k < claims.count; ++k)
1659 {
1660 const auto* pci = dynamic_cast<const CPCIDevice*>(
1661 claims.range[k].component);
1662 if (pci && pci->pci_bus() == shared_read_hose &&
1663 pci->pci_dev() == shared_read_device)
1664 return values[k];
1665 }
1666 report_decode_overlap(address, dsize, false, claims, source,
1667 "debug.shared_read_claimant is not among the claimants", detail.str());
1668}

1713void CSystem::WriteMem(u64 address, int dsize, u64 data, CSystemComponent* source)
1714{
1715 u64 a;
1716 u8* p;
1717#if defined(ALIGN_MEM_ACCESS)
1718 u64 t64;
1719 u32 t32;
1720 u16 t16;
1721#endif //defined(ALIGN_MEM_ACCESS)
1722 a = address & U64(0x00000807ffffffff);
1723
1724 if (a >> iNumMemoryBits) // non-memory: chipset PIO decode
1725 {
1726 pio_write(a, dsize, data, source);
1727 return;
1728 }
1729
1730 // PAL replacements and byte-split CPU accesses also reach this entry point.
1731 // Keep device DMA on its existing path.
1732 // Only CPU-originated RAM writes join the CPU reservation protocol here.
1733 // Preserve WriteMem's endian conversion.
1734 for (int cpu = 0; source && cpu < iNumCPUs; ++cpu)
1735 if (source == acCPUs[cpu])
1736 {
1737 switch (dsize)
1738 {
1739 case 8: break;
1740 case 16: data = endian_16((u16)data); break;
1741 case 32: data = endian_32((u32)data); break;
1742 default: data = endian_64(data); dsize = 64; break;
1743 }
1744 cpu_write_ram(a, dsize, data);
1745 return;
1746 }
1747
1748 p = (u8*)memory + a;
1749
1750 switch (dsize)
1751 {
1752 case 8: *((u8*)p) = (u8)data; break;
1753 case 16: *((u16*)p) = endian_16((u16)data); break;
1754 case 32: *((u32*)p) = endian_32((u32)data); break;
1755 default: *((u64*)p) = endian_64((u64)data);
1756 }
1757}
