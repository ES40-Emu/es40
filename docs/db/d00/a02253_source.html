222 inline uint8_t vga_logical_op(uint8_t data, uint8_t plane, uint8_t mask)
223 {
224 uint8_t res = 0;
225 switch (vga.gc.logical_op & 3)
226 {
227 case 0: /* NONE */
228 res = (data & mask) | (vga.gc.latch[plane] & ~mask);
229 break;
230 case 1: /* AND */
231 res = (data | ~mask) & (vga.gc.latch[plane]);
232 break;
233 case 2: /* OR */
234 res = (data & mask) | (vga.gc.latch[plane]);
235 break;
236 case 3: /* XOR */
237 res = (data & mask) ^ (vga.gc.latch[plane]);
238 break;
239 }
240 return res;
241 }
