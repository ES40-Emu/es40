38SDL_AudioSpec native_format(const Format& format)
39{
40 SDL_AudioSpec spec;
41 spec.freq = format.freq;
42 spec.channels = format.channels;
43 spec.format = format.format == SampleFormat::u8 ? SDL_AUDIO_U8 : SDL_AUDIO_S16LE;
44 return spec;
45}

47void SDLCALL stream_callback(void* userdata, SDL_AudioStream*, int, int)
48{
49 auto& stream = *static_cast<Stream*>(userdata);
50 // Forward on SDL's thread under its existing stream lock.
51 stream.callback(stream.userdata);
52}

73bool output_device_format(std::uintptr_t device, int* frequency, int* sample_frames)
74{
75 SDL_AudioSpec spec;
76 const bool result = SDL_GetAudioDeviceFormat(static_cast<SDL_AudioDeviceID>(device), &spec, sample_frames);
77 if (result) *frequency = spec.freq;
78 return result;
79}

91void create_stream(Stream& stream, const Format* source, const Format* target)
92{
93 SDL_AudioSpec src, dst;
94 if (source) src = native_format(*source);
95 if (target) dst = native_format(*target);
96 stream.native = SDL_CreateAudioStream(source ? &src : nullptr, target ? &dst : nullptr);
97}
