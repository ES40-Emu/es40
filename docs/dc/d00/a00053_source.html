51class CMutex
52{
53public:
54 using ScopedLock = std::lock_guard<CMutex>;
55
56 CMutex() : lockName("?") {}
57 explicit CMutex(const char* lName) : lockName(lName) {}
58 ~CMutex() = default;
59
60 void lock()
61 {
62#if defined(DEBUG_LOCKS)
63 printf(" LOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
64#endif
65 _mutex.lock();
66#if defined(DEBUG_LOCKS)
67 printf(" LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
68#endif
69 }
70
71 void lock(long milliseconds)
72 {
73#if defined(DEBUG_LOCKS)
74 printf(" TIMED LOCK mutex %s from thread %s (%ld ms).\n", lockName, CURRENT_THREAD_NAME, milliseconds);
75#endif
76 if (!tryLock(milliseconds))
77 {
78 printf("TIMEOUT locking mutex %s from thread %s after %ld ms.\n",
79 lockName, CURRENT_THREAD_NAME, milliseconds);
80 throw std::runtime_error("Mutex lock timeout");
81 }
82#if defined(DEBUG_LOCKS)
83 printf(" LOCKED mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
84#endif
85 }
86
87 bool tryLock() { return _mutex.try_lock(); }
88
89 bool tryLock(long milliseconds)
90 {
91 auto deadline = std::chrono::steady_clock::now()
92 + std::chrono::milliseconds(milliseconds);
93 do
94 {
95 if (_mutex.try_lock())
96 return true;
97 std::this_thread::sleep_for(std::chrono::milliseconds(5));
98 } while (std::chrono::steady_clock::now() < deadline);
99 return false;
100 }
101
102 void unlock()
103 {
104#if defined(DEBUG_LOCKS)
105 printf(" UNLOCK mutex %s from thread %s.\n", lockName, CURRENT_THREAD_NAME);
106#endif
107 _mutex.unlock();
108 }
109
110 const char* lockName;
111
112private:
113 std::recursive_mutex _mutex;
114 CMutex(const CMutex&) = delete;
115 CMutex& operator=(const CMutex&) = delete;
116};
