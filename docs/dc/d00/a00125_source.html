57static std::wstring wide(const std::string& text)
58{
59 if (text.empty()) return {};
60 const int count = MultiByteToWideChar(CP_ACP, 0, text.data(), (int)text.size(), nullptr, 0);
61 if (!count) throw std::runtime_error("Unable to decode configuration text.");
62 std::wstring result(count, L'\0');
63 MultiByteToWideChar(CP_ACP, 0, text.data(), (int)text.size(), result.data(), count);
64 return result;
65}
66
67static std::string narrow(const std::wstring& text)
68{
69 if (text.empty()) return {};
70 const bool utf8 = GetACP() == CP_UTF8;
71 BOOL substituted = FALSE;
72 const DWORD flags = utf8 ? 0 : WC_NO_BEST_FIT_CHARS;
73 const int count = WideCharToMultiByte(CP_ACP, flags, text.data(), (int)text.size(), nullptr, 0,
74 nullptr, utf8 ? nullptr : &substituted);
75 if (!count || substituted) throw std::runtime_error("This value cannot be represented in the active Windows code page.");
76 std::string result(count, '\0');
77 if (!WideCharToMultiByte(CP_ACP, flags, text.data(), (int)text.size(), result.data(), count,
78 nullptr, utf8 ? nullptr : &substituted) || substituted)
79 throw std::runtime_error("This value cannot be represented in the active Windows code page.");
80 return result;
81}

83static std::wstring control_text(const char* source)
84{
85 const std::wstring text = wide(source ? source : "");
86 std::wstring result;
87 for (size_t i = 0; i < text.size(); ++i)
88 {
89 if (text[i] == L'\r')
90 { if (i + 1 < text.size() && text[i + 1] == L'\n') ++i; result += L"\r\n"; }
91 else if (text[i] == L'\n') result += L"\r\n";
92 else result += text[i];
93 }
94 return result;
95}

97static std::wstring window_text(HWND control)
98{
99 std::wstring text(GetWindowTextLengthW(control) + 1, L'\0');
100 GetWindowTextW(control, text.data(), (int)text.size());
101 text.resize(wcslen(text.c_str()));
102 return text;
103}

176 void form_layout()
177 {
178 if (!active || !viewport || active->type != kind::form) return;
179 RECT bounds; GetClientRect(viewport, &bounds);
180 const int row = px(76), total = (int)active->inputs.size() * row + px(12);
181 const int maximum = std::max(0, total - (int)bounds.bottom);
182 active->scroll = std::max(0, std::min(active->scroll, maximum));
183 SCROLLINFO scroll = { sizeof(scroll), SIF_RANGE | SIF_PAGE | SIF_POS };
184 scroll.nMin = 0; scroll.nMax = std::max(0, total - 1);
185 scroll.nPage = (UINT)bounds.bottom; scroll.nPos = active->scroll;
186 SetScrollInfo(viewport, SB_VERT, &scroll, TRUE);
187 for (size_t i = 0; i < active->inputs.size(); ++i)
188 {
189 const int top = px(10) + (int)i * row - active->scroll;
190 MoveWindow(active->labels[i], px(12), top, std::max(1, (int)bounds.right - px(28)), px(22), TRUE);
191 const bool choice = (*active->fields)[i].type == EditorField::Type::Choice;
192 MoveWindow(active->inputs[i], px(12), top + px(26), std::max(1, (int)bounds.right - px(28)), choice ? px(280) : px(28), TRUE);
193 }

238 void build_page(page& current)
239 {
240 building = true;
241 current.inputs.clear(); current.labels.clear();
242 heading = add_control(window, L"STATIC", L"Machine settings", SS_LEFT, 101);
243 SendMessageW(heading, WM_SETFONT, (WPARAM)title_font, TRUE);
244 subtitle = add_control(window, L"STATIC", wide(current.title), SS_LEFT, 102);
245 if (current.type == kind::menu)
246 {
247 list = add_control(window, L"LISTBOX", L"", LBS_NOTIFY | LBS_NOINTEGRALHEIGHT | WS_VSCROLL | WS_TABSTOP,
248 id_menu, WS_EX_CLIENTEDGE);
249 for (int index : current.menu_indices)
250 {
251 const auto& item = current.menu_entries[index];
252 std::wstring label = wide(item.text);
253 if (item.select_callback) label += L" >";
254 SendMessageW(list, LB_ADDSTRING, 0, (LPARAM)label.c_str());
255 }
256 SendMessageW(list, LB_SETCURSEL, current.current, 0);
257 }
258 else if (current.type == kind::form)
259 {
260 viewport = CreateWindowExW(WS_EX_CONTROLPARENT | WS_EX_CLIENTEDGE, form_window_class, L"",
261 WS_CHILD | WS_VISIBLE | WS_VSCROLL | WS_CLIPCHILDREN, 0, 0, 1, 1, window, nullptr, GetModuleHandleW(nullptr), this);
262 if (!viewport) throw std::runtime_error("Unable to create the settings form.");
263 controls.push_back(viewport);
264 for (size_t i = 0; i < current.fields->size(); ++i)
265 {
266 const auto& field = (*current.fields)[i];
267 std::wstring label = wide(current.form_entries[i].label ? current.form_entries[i].label : "");
268 if (field.type == EditorField::Type::Integer || field.type == EditorField::Type::Number)
269 {
270 std::wostringstream range;
271 range << L" (" << field.minimum << L" to " << field.maximum << L")";
272 label += range.str();
273 }
274 current.labels.push_back(add_control(viewport, L"STATIC", label, SS_LEFT, 3000 + (int)i));
275 const bool choice = field.type == EditorField::Type::Choice;
276 HWND input = add_control(viewport, choice ? L"COMBOBOX" : L"EDIT", choice ? L"" : current.drafts[i],
277 WS_TABSTOP | (choice ? CBS_DROPDOWNLIST | CBS_HASSTRINGS | WS_VSCROLL : ES_AUTOHSCROLL), id_field + (int)i,
278 choice ? 0 : WS_EX_CLIENTEDGE);
279 current.inputs.push_back(input);
280 if (choice)
281 {
282 for (const auto& option : field.choices)
283 { const std::wstring text = wide(option); SendMessageW(input, CB_ADDSTRING, 0, (LPARAM)text.c_str()); }
284 LRESULT selected = SendMessageW(input, CB_FINDSTRINGEXACT, (WPARAM)-1, (LPARAM)current.drafts[i].c_str());
285 if (selected == CB_ERR)
286 selected = SendMessageW(input, CB_ADDSTRING, 0, (LPARAM)current.drafts[i].c_str());
287 SendMessageW(input, CB_SETCURSEL, selected, 0);
288 SendMessageW(input, CB_SETMINVISIBLE, 12, 0);
289 }
290 else SendMessageW(input, EM_SETLIMITTEXT, 32767, 0);
291 }
292 }
293 description = add_control(window, L"EDIT", current.type == kind::text ? control_text(current.body.c_str()) : L"",
294 ES_MULTILINE | ES_READONLY | ES_AUTOVSCROLL | WS_VSCROLL | WS_TABSTOP, 103, WS_EX_CLIENTEDGE);
295 SendMessageW(description, EM_SETLIMITTEXT, 1024 * 1024, 0);
296 add_control(window, L"BUTTON", current.type == kind::menu ? L"&Open" : current.type == kind::form ? L"&Apply" : L"&Close",
297 WS_TABSTOP | BS_DEFPUSHBUTTON, IDOK);
298 if (current.type != kind::text)
299 {
300 add_control(window, L"BUTTON", current.type == kind::form ? L"&Cancel" : current.main_menu ? L"&Save and close" : L"&Back",
301 WS_TABSTOP | BS_PUSHBUTTON, IDCANCEL);
302 add_control(window, L"BUTTON", L"&Help (F1)", WS_TABSTOP | BS_PUSHBUTTON, id_help);
303 add_control(window, L"BUTTON", L"Cancel &all", WS_TABSTOP | BS_PUSHBUTTON, id_quit);
304 }
305 building = false;
306 layout(); update_description();
307 InvalidateRect(window, nullptr, TRUE);
308 if (current.type == kind::menu) SetFocus(list);
309 else if (current.type == kind::form && !current.inputs.empty())
310 { reveal_field(current.current); SetFocus(current.inputs[current.current]); }
311 else SetFocus(GetDlgItem(window, IDOK));
312 UpdateWindow(window);
313 }

315 void apply_form()
316 {
317 if (!active || !active->fields) return;
318 auto proposal = *active->fields;
319 for (size_t i = 0; i < proposal.size(); ++i)
320 {
321 try
322 {
323 proposal[i].value = narrow(active->drafts[i]);
324 if (!validate_editor_field(proposal[i]))
325 throw std::runtime_error(std::string("Invalid value for ") + active->form_entries[i].label + ". Check the field help and permitted values.");
326 }
327 catch (const std::exception& error)
328 {
329 reveal_field((int)i); SetFocus(active->inputs[i]);
330 MessageBoxW(window, wide(error.what()).c_str(), L"Invalid setting", MB_OK | MB_ICONWARNING);
331 SetFocus(active->inputs[i]); return;
332 }
333 }
334 *active->fields = std::move(proposal); active->action = IDOK;
335 }

337 LRESULT event(HWND hwnd, UINT message, WPARAM wparam, LPARAM lparam)
338 {
339 switch (message)
340 {
341 case WM_CLOSE: closed = true; return 0;
342 case WM_SIZE: if (hwnd == window) layout(); else if (hwnd == viewport) form_layout(); return 0;
343 case WM_GETMINMAXINFO:
344 if (hwnd == window) reinterpret_cast<MINMAXINFO*>(lparam)->ptMinTrackSize = { px(780), px(600) };
345 return 0;
346 case WM_COMMAND:
347 {
348 if (!active || building) return 0;
349 const int id = LOWORD(wparam), notification = HIWORD(wparam);
350 if (id >= id_field && active->type == kind::form && id < id_field + (int)active->drafts.size())
351 {
352 const int index = id - id_field;
353 const bool choice = (*active->fields)[index].type == EditorField::Type::Choice;
354 if (notification == (choice ? CBN_SELCHANGE : EN_CHANGE))
355 {
356 if (choice)
357 {
358 const LRESULT selected = SendMessageW((HWND)lparam, CB_GETCURSEL, 0, 0);
359 const LRESULT length = SendMessageW((HWND)lparam, CB_GETLBTEXTLEN, selected, 0);
360 if (selected != CB_ERR && length != CB_ERR)
361 {
362 std::wstring value((size_t)length + 1, L'\0');
363 SendMessageW((HWND)lparam, CB_GETLBTEXT, selected, (LPARAM)value.data());
364 value.resize((size_t)length); active->drafts[index] = std::move(value);
365 }
366 }
367 else active->drafts[index] = window_text((HWND)lparam);
368 }
369 if (notification == (choice ? CBN_SETFOCUS : EN_SETFOCUS)) reveal_field(index);
370 return 0;
371 }
372 if (id == id_menu)
373 {
374 const int selected = (int)SendMessageW(list, LB_GETCURSEL, 0, 0);
375 if (selected != LB_ERR) active->current = selected;
376 update_description();
377 if (notification == LBN_DBLCLK) active->action = IDOK;
378 return 0;
379 }
380 if (id == id_quit) { closed = true; return 0; }
381 if (id == id_help) { active->action = help_action; return 0; }
382 if (id == IDCANCEL) { active->action = MENU_SEL_EXIT; return 0; }
383 if (id == IDOK)
384 {
385 if (active->type == kind::form) apply_form(); else active->action = IDOK;
386 return 0;
387 }
388 break;
389 }
390 case WM_VSCROLL:
391 if (hwnd == viewport && active)
392 {
393 RECT bounds; GetClientRect(viewport, &bounds);
394 switch (LOWORD(wparam))
395 {
396 case SB_LINEUP: scroll_form(-px(38)); break;
397 case SB_LINEDOWN: scroll_form(px(38)); break;
398 case SB_PAGEUP: scroll_form(-bounds.bottom); break;
399 case SB_PAGEDOWN: scroll_form(bounds.bottom); break;
400 case SB_TOP: active->scroll = 0; form_layout(); break;
401 case SB_BOTTOM: active->scroll = INT_MAX / 2; form_layout(); break;
402 case SB_THUMBTRACK: case SB_THUMBPOSITION:
403 { SCROLLINFO scroll = {sizeof(scroll), SIF_TRACKPOS}; GetScrollInfo(viewport, SB_VERT, &scroll); active->scroll = scroll.nTrackPos; form_layout(); break; }
404 }
405 return 0;
406 }
407 break;
408 case WM_MOUSEWHEEL:
409 if (active && active->type == kind::form)
410 { scroll_form(-GET_WHEEL_DELTA_WPARAM(wparam) * px(76) / WHEEL_DELTA); return 0; }
411 break;
412 case WM_CTLCOLORSTATIC:
413 if ((HWND)lparam == heading || (HWND)lparam == subtitle)
414 { SetTextColor((HDC)wparam, RGB(35,48,66)); SetBkColor((HDC)wparam, RGB(245,247,250)); return (LRESULT)background; }
415 break;
416 case WM_ERASEBKGND:
417 { RECT bounds; GetClientRect(hwnd, &bounds); FillRect((HDC)wparam, &bounds, hwnd == viewport ? (HBRUSH)(COLOR_WINDOW + 1) : background); return 1; }
418 }
419 return DefWindowProcW(hwnd, message, wparam, lparam);
420 }

422 int run_page(page& current)
423 {
424 current.action = pending_action; active = &current;
425 try
426 {
427 build_page(current);
428 while (!closed && !failure && current.action == pending_action)
429 {
430 MSG message;
431 const int result = GetMessageW(&message, nullptr, 0, 0);
432 if (result <= 0) { closed = true; break; }
433 if (message.message == WM_KEYDOWN && (message.hwnd == window || IsChild(window, message.hwnd)))
434 {
435 const bool dropdown = current.type == kind::form && current.current < (int)current.inputs.size() &&
436 (*current.fields)[current.current].type == EditorField::Type::Choice &&
437 SendMessageW(current.inputs[current.current], CB_GETDROPPEDSTATE, 0, 0);
438 if (message.wParam == VK_F1 && current.type != kind::text) { current.action = help_action; continue; }
439 if (message.wParam == VK_F2) { SendMessageW(window, WM_COMMAND, current.type == kind::menu ? IDCANCEL : IDOK, 0); continue; }
440 if (message.wParam == VK_F3) { closed = true; continue; }
441 if (message.wParam == VK_ESCAPE && !dropdown)
442 { current.action = current.type == kind::menu && current.main_menu ? MENU_SEL_QUIT : MENU_SEL_EXIT; continue; }
443 if (message.wParam == VK_RETURN && !dropdown && GetFocus() == list)
444 { current.action = IDOK; continue; }
445 if (message.wParam == VK_RETURN && !dropdown && current.type == kind::form &&
446 std::find(current.inputs.begin(), current.inputs.end(), GetFocus()) != current.inputs.end())
447 { SendMessageW(window, WM_COMMAND, IDOK, 0); continue; }
448 }
449 if (!IsDialogMessageW(window, &message)) { TranslateMessage(&message); DispatchMessageW(&message); }
450 }
451 }
452 catch (...) { active = nullptr; destroy_controls(); throw; }
453 active = nullptr; destroy_controls();
454 if (failure) std::rethrow_exception(failure);
455 if (closed) throw EditorClosed{};
456 return current.action;
457 }
