35int regcomp(regex_t *preg, const char *pattern, int cflags)
36{
37 std::regex::flag_type flags = (cflags & REG_EXTENDED) ? std::regex::extended : std::regex::basic;
38 if (cflags & REG_ICASE)
39 flags |= std::regex::icase;
40 if (cflags & REG_NOSUB)
41 flags |= std::regex::nosubs;
42
43 preg->re_impl = nullptr;
44 preg->re_nsub = 0;
45 try
46 {
47 std::regex *re = new std::regex(pattern, flags);
48 preg->re_impl = re;
49 preg->re_nsub = re->mark_count();
50 return 0;
51 }
52 catch (const std::regex_error &)
53 {
54 return REG_BADPAT;
55 }
56 catch (const std::bad_alloc &)
57 {
58 return REG_ESPACE;
59 }
60}

62int regexec(const regex_t *preg, const char *string, size_t nmatch, regmatch_t pmatch[], int eflags)
63{
64 const std::regex *re = static_cast<const std::regex *>(preg->re_impl);
65 if (re == nullptr)
66 return REG_BADPAT;
67
68 std::regex_constants::match_flag_type mflags = std::regex_constants::match_default;
69 if (eflags & REG_NOTBOL)
70 mflags |= std::regex_constants::match_not_bol;
71 if (eflags & REG_NOTEOL)
72 mflags |= std::regex_constants::match_not_eol;
73
74 try
75 {
76 std::cmatch m;
77 if (!std::regex_search(string, m, *re, mflags))
78 return REG_NOMATCH;
79
80 for (size_t i = 0; i < nmatch; ++i)
81 {
82 if (i < m.size() && m[i].matched)
83 {
84 pmatch[i].rm_so = m.position(i);
85 pmatch[i].rm_eo = m.position(i) + m.length(i);
86 }
87 else
88 pmatch[i].rm_so = pmatch[i].rm_eo = -1;
89 }
90 return 0;
91 }
92 catch (const std::regex_error &)
93 {
94 return REG_ESPACE;
95 }
96}

99{
100 delete static_cast<std::regex *>(preg->re_impl);
101 preg->re_impl = nullptr;
