192int CAlphaCPU::vax_fcmp(u64 s1, u64 s2, u32 ins)
193{
194 UFP a;
195
196 UFP b;
197
198 vax_unpack(s1, &a, ins);
199 vax_unpack(s2, &b, ins);
200 if (s1 == s2)
201 return 0; /* equal? */
202 if (a.sign != b.sign)
203 return(a.sign ? -1 : +1); /* opp signs? */
204 return(((s1 < s2) ^ a.sign) ? -1 : +1); /* like signs */
205}

216u64 CAlphaCPU::vax_cvtif(u64 val, u32 ins, u32 dp)
217{
218 s64 num = (s64)val;
219 UFP a;
220
221 if (num == 0)
222 return 0; /* 0? return +0 */
223 if (num < 0)
224 { /* < 0? */
225 a.sign = 1; /* set sign */
226 val = NEG_Q(val);
227 } /* |val| */
228 else
229 a.sign = 0;
230 a.exp = 64 + G_BIAS; /* set exp */
231 a.frac = val; /* set frac */
232 vax_norm(&a); /* normalize */
233 return vax_rpack(&a, ins, dp); /* round and pack */
234}

247u64 CAlphaCPU::vax_cvtfi(u64 op, u32 ins)
248{
249 UFP a;
250 u32 rndm = I_GETFRND(ins);
251 s32 ubexp;
252
253 vax_unpack(op, &a, ins);
254 ubexp = a.exp - G_BIAS; /* unbiased exp */
255 if (ubexp < 0)
256 return 0; /* zero or too small? */
257 if (ubexp <= UF_V_NM)
258 { /* in range? */
259 a.frac = a.frac >> (UF_V_NM - ubexp); /* leave rnd bit */
260 if (rndm)
261 a.frac = a.frac + 1; /* not chopped, round */
262 a.frac = a.frac >> 1; /* now justified */
263 if ((a.frac > (a.sign ? IMMAX : IPMAX)) /* out of range? */
264 && (ins & I_FTRP_V)) /* trap enabled? */vax_trap(TRAP_IOV, ins);
265 } /* set overflow */
266 else
267 {
268 if (ubexp > (UF_V_NM + 64))
269 a.frac = 0; /* out of range */
270 else
271 a.frac = (a.frac << (ubexp - UF_V_NM - 1)) & X64_QUAD; /* no rnd bit */
272 if (ins & I_FTRP_V) /* trap enabled? */
273 vax_trap(TRAP_IOV, ins);
274 } /* set overflow */
275
276 return(a.sign ? NEG_Q(a.frac) : a.frac);
277}

290u64 CAlphaCPU::vax_fadd(u64 s1, u64 s2, u32 ins, u32 dp, bool sub)
291{
292 UFP a;
293
294 UFP b;
295
296 UFP t;
297 u32 sticky;
298 s32 ediff;
299
300 vax_unpack(s1, &a, ins);
301 vax_unpack(s2, &b, ins);
302 if (sub)
303 b.sign = b.sign ^ 1; /* sub? invert b sign */
304 if (a.exp == 0)
305 a = b; /* s1 = 0? */
306 else if (b.exp) /* s2 != 0? */
307 {
308 if ((a.exp < b.exp) /* |s1| < |s2|? swap */
309 || ((a.exp == b.exp) && (a.frac < b.frac)))
310 {
311 t = a;
312 a = b;
313 b = t;
314 }
315
316 ediff = a.exp - b.exp; /* exp diff */
317 if (a.sign ^ b.sign) /* eff sub? */
318 {
319 if (ediff > 63)
320 b.frac = 1; /* >63? retain sticky */
321 else if (ediff) /* [1,63]? shift */
322 {
323 sticky = ((b.frac << (64 - ediff)) & X64_QUAD) ? 1 : 0; /* lost bits */
324 b.frac = (b.frac >> ediff) | sticky;
325 }
326
327 a.frac = (a.frac - b.frac) & X64_QUAD; /* subtract fractions */
328 vax_norm(&a); /* normalize */
329 }
330 else /* eff add */
331 {
332 if (ediff > 63)
333 b.frac = 0; /* >63? b disappears */
334 else if (ediff)
335 b.frac = b.frac >> ediff; /* denormalize */
336 a.frac = (a.frac + b.frac) & X64_QUAD; /* add frac */
337 if (a.frac < b.frac) /* chk for carry */
338 {
339 a.frac = UF_NM | (a.frac >> 1); /* shift in carry */
340 a.exp = a.exp + 1; /* skip norm */
341 }
342 }
343 } /* end else if */
344
345 return vax_rpack(&a, ins, dp); /* round and pack */
346}

358u64 CAlphaCPU::vax_fmul(u64 s1, u64 s2, u32 ins, u32 dp)
359{
360 UFP a;
361
362 UFP b;
363
364 vax_unpack(s1, &a, ins);
365 vax_unpack(s2, &b, ins);
366 if ((a.exp == 0) || (b.exp == 0))
367 return 0; /* zero argument? */
368 a.sign = a.sign ^ b.sign; /* sign of result */
369 a.exp = a.exp + b.exp - G_BIAS; /* add exponents */
370 uemul64(a.frac, b.frac, &a.frac); /* mpy fractions */
371 vax_norm(&a); /* normalize */
372 return vax_rpack(&a, ins, dp); /* round and pack */
373}

389u64 CAlphaCPU::vax_fdiv(u64 s1, u64 s2, u32 ins, u32 dp)
390{
391 UFP a;
392
393 UFP b;
394
395 vax_unpack(s1, &a, ins);
396 vax_unpack(s2, &b, ins);
397 if (b.exp == 0)
398 { /* divr = 0? */
399 vax_trap(TRAP_DZE, ins); /* dze trap */
400 return 0;
401 }
402
403 if (a.exp == 0)
404 return 0; /* divd = 0? */
405 a.sign = a.sign ^ b.sign; /* result sign */
406 a.exp = a.exp - b.exp + G_BIAS + 1; /* unbiased exp */
407 a.frac = a.frac >> 1; /* allow 1 bit left */
408 b.frac = b.frac >> 1;
409 a.frac = ufdiv64(a.frac, b.frac, 55, NULL); /* divide */
410 vax_norm(&a); /* normalize */
411 return vax_rpack(&a, ins, dp); /* round and pack */
412}

423u64 CAlphaCPU::vax_sqrt(u64 op, u32 ins, u32 dp)
424{
425 UFP b;
426
427 vax_unpack(op, &b, ins);
428 if (b.exp == 0)
429 return 0; /* zero? */
430 if (b.sign)
431 { /* minus? */
432 vax_trap(TRAP_INV, ins); /* invalid operand */
433 return 0;
434 }
435
436 const s32 in_exp = b.exp; /* fsqrt64 keys on INPUT exp parity */
437 b.exp = ((b.exp + 1 - G_BIAS) >> 1) + G_BIAS; /* result exponent */
438 b.frac = fsqrt64(b.frac, in_exp); /* result fraction */
439 return vax_rpack(&b, ins, dp); /* round and pack */
440}

480void CAlphaCPU::vax_unpack(u64 op, UFP* r, u32 ins)
481{
482 r->sign = FPR_GETSIGN(op); /* get sign */
483 r->exp = FPR_GETEXP(op); /* get exponent */
484 r->frac = FPR_GETFRAC(op); /* get fraction */
485 if (r->exp == 0) /* exp = 0? */
486 {
487 if (r->sign != 0) /* rsvd op? */
488 vax_trap(TRAP_INV, ins);
489
490 // zero
491 r->frac = r->sign = 0;
492 return;
493 }
494
495 r->frac = (r->frac | FPR_HB) << FPR_GUARD; /* ins hidden bit, guard */
496 return;
497}

512{
513 r->sign = FDR_GETSIGN(op); /* get sign */
514 r->exp = FDR_GETEXP(op); /* get exponent */
515 r->frac = FDR_GETFRAC(op); /* get fraction */
516 if (r->exp == 0) /* exp = 0? */
517 {
518 if (r->sign != 0) /* rsvd op? */
519 vax_trap(TRAP_INV, ins);
520 r->frac = r->sign = 0;
521 return;
522 }
523

525 r->frac = (r->frac | FDR_HB) << FDR_GUARD; /* ins hidden bit, guard */

538void CAlphaCPU::vax_norm(UFP* r)
539{
540 s32 i;
541 static u64 normmask[5] = {
542 U64(0xc000000000000000), U64(0xf000000000000000), U64(0xff00000000000000),
543 U64(0xffff000000000000), U64(0xffffffff00000000) };
544 static s32 normtab[6] = { 1, 2, 4, 8, 16, 32 };
545
546 r->frac = r->frac & X64_QUAD;
547 if (r->frac == 0)
548 { /* if fraction = 0 */
549 r->sign = r->exp = 0; /* result is 0 */
550 return;
551 }
552
553 while ((r->frac & UF_NM) == 0)
554 { /* normalized? */
555 for (i = 0; i < 5; i++)
556 { /* find first 1 */
557 if (r->frac & normmask[i])
558 break;
559 }
560
561 r->frac = r->frac << normtab[i]; /* shift frac */
562 r->exp = r->exp - normtab[i];
563 } /* decr exp */
564
565 return;
566}

581u64 CAlphaCPU::vax_rpack(UFP* r, u32 ins, u32 dp)
582{
583 u32 rndm = I_GETFRND(ins);
584 static const u64 roundbit[2] = { UF_FRND, UF_GRND };
585 static const s32 expmax[2] = { G_BIAS - F_BIAS + F_M_EXP, G_M_EXP };
586 static const s32 expmin[2] = { G_BIAS - F_BIAS, 0 };
587
588 if (r->frac == 0)
589 return 0; /* result 0? */
590 if (rndm)
591 { /* round? */
592 r->frac = (r->frac + roundbit[dp]) & X64_QUAD; /* add round bit */
593 if ((r->frac & UF_NM) == 0)
594 { /* carry out? */
595 r->frac = (r->frac >> 1) | UF_NM; /* renormalize */
596 r->exp = r->exp + 1;
597 }
598 }
599
600 if (r->exp > expmax[dp])
601 { /* ovflo? */
602 vax_trap(TRAP_OVF, ins); /* set trap */
603 r->exp = expmax[dp];
604 } /* return max */
605
606 if (r->exp <= expmin[dp])
607 { /* underflow? */
608 if (ins & I_FTRP_V)
609 vax_trap(TRAP_UNF, ins); /* enabled? set trap */
610 return 0;
611 } /* underflow to 0 */
612
613 return(((u64)r->sign) << FPR_V_SIGN) | (((u64)r->exp) << FPR_V_EXP) | ((r->frac >> FPR_GUARD) & FPR_FRAC);
614}

628u64 CAlphaCPU::vax_rpack_d(UFP* r, u32 ins)
629{
630 if (r->frac == 0)
631 return 0; /* result 0? */
632 r->exp = r->exp + D_BIAS - G_BIAS; /* rebias */
633 if (r->exp > FDR_M_EXP)
634 { /* ovflo? */
635 vax_trap(TRAP_OVF, ins); /* set trap */
636 r->exp = FDR_M_EXP;
637 } /* return max */
638
639 if (r->exp <= 0)
640 { /* underflow? */
641 if (ins & I_FTRP_V)
642 vax_trap(TRAP_UNF, ins); /* enabled? set trap */
643 return 0;
644 } /* underflow to 0 */
645
646 return(((u64)r->sign) << FDR_V_SIGN) | (((u64)r->exp) << FDR_V_EXP) | ((r->frac >> FDR_GUARD) & FDR_FRAC);
647}
