75// Decoded PAL return rules, outside saved CPU state. Read the selected source
76// at execution time; EXC_ADDR returns and forms with side effects stay interpreted.
77struct AlphaPALReturnDescriptor
78{
79 bool valid = false;
80 std::uint64_t target_mask = ~std::uint64_t(0);
81 std::uint64_t target_bits = 0;
82 bool clear_lock = false;
83 bool compilable = false;
84 AlphaPALReturnSource source = AlphaPALReturnSource::RegisterB;
85};
86
87constexpr bool alpha_pal_return_valid(const AlphaPALReturnDescriptor& entry)
88{
89 return entry.valid && (entry.source == AlphaPALReturnSource::RegisterB
90 || entry.source == AlphaPALReturnSource::ExceptionAddress);
91}
92
93constexpr std::uint64_t alpha_pal_return_target(
94 const AlphaPALReturnDescriptor& entry, std::uint64_t register_b,
95 std::uint64_t exception_address)
96{
97 const std::uint64_t source = entry.source == AlphaPALReturnSource::ExceptionAddress
98 ? exception_address : register_b;
99 return (source & entry.target_mask) | entry.target_bits;
100}
101
102constexpr bool alpha_pal_return_compilable(const AlphaPALReturnDescriptor& entry)
103{
104 return alpha_pal_return_valid(entry) && entry.compilable && !entry.clear_lock
105 && entry.source == AlphaPALReturnSource::RegisterB;
106}

108// Family access rules for the PAL-reserved hardware instructions.
109enum class AlphaPALInstructionAccess
110{
111 Denied,
112 PalOnly,
113 PalOrKernelHwe
114};
115
116constexpr bool alpha_pal_instruction_allowed(AlphaPALInstructionAccess access,
117 bool pal_mode, std::uint32_t mode,
118 bool hwe)
119{
120 switch (access)
121 {
122 case AlphaPALInstructionAccess::Denied: return false;
123 case AlphaPALInstructionAccess::PalOnly: return pal_mode;
124 case AlphaPALInstructionAccess::PalOrKernelHwe:
125 return pal_mode || (mode == 0 && hwe);
126 }
127 return false;
128}
129
130// These rules guarantee access throughout a PAL block.
131// Native HWE execution stays interpreted.
132constexpr bool alpha_pal_instruction_compilable(AlphaPALInstructionAccess access)
133{
134 return access == AlphaPALInstructionAccess::PalOnly
135 || access == AlphaPALInstructionAccess::PalOrKernelHwe;
136}

138// Stateless family PAL rules, copied with the CPU profile.
139struct AlphaPALPolicy
140{
141 AlphaPALExceptionEntry (*exception_entry)(AlphaPALException reason,
142 std::uint64_t current_pc);
143 bool (*call_pal_valid)(std::uint32_t function, std::uint32_t mode);
144 AlphaPALCallEntry (*call_pal_entry)(std::uint32_t function,
145 std::uint64_t current_pc,
146 std::uint64_t next_pc);
147 // Pass the complete instruction so restrictions on individual forms apply.
148 // An absent access rule denies all.
149 AlphaPALInstructionAccess (*hardware_instruction_access)(std::uint32_t instruction) = nullptr;
150 AlphaPALReturnDescriptor (*hardware_return)(std::uint32_t instruction) = nullptr;
151};
152
153constexpr AlphaPALReturnDescriptor alpha_pal_return_descriptor(
154 const AlphaPALPolicy& policy, std::uint32_t instruction)
155{
156 return policy.hardware_return ? policy.hardware_return(instruction)
157 : AlphaPALReturnDescriptor{};
158}
