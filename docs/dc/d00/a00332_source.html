77// MAME addrmap.h binds member handlers through device_t and rw_delegate.
78// ES40 byte maps have no device_t owner, so FUNC binds the current object.
79// Typed invocation retains virtual dispatch and base adjustments.
80template <typename Owner, typename Member>
81auto bind_map_member(Owner* owner, Member member)
82{
83 static_assert(std::is_member_function_pointer_v<Member>,
84 "FUNC requires a member function");
85 return [owner, member](auto&&... args) -> decltype(auto) {
86 return std::invoke(member, owner, std::forward<decltype(args)>(args)...);
87 };
88}
89
90// Map compatibility: use inside the owning object's methods or constructors.
91// Upstream emucore.h: #define FUNC(x) &x, #x
92#ifndef FUNC
93#define FUNC(x) ::bind_map_member(this, &x), #x
94#endif

99// address_map_entry — returned by map(lo, hi), chains handler bindings/.mirror
100
101class address_map_entry
102{
103public:
104 address_map_entry(address_map& parent, unsigned lo, unsigned hi)
105 : m_map(parent), m_lo(lo), m_hi(hi), m_mirror(0), m_select(0) {
106 }
107
108 // Byte member handlers use the same dispatch and offset rules as lambdas.
109 template <typename T>
110 address_map_entry& r(T&& read_fn, const char* name)
111 {
112 return lr8(std::forward<T>(read_fn), name);
113 }
114
115 template <typename T>
116 address_map_entry& w(T&& write_fn, const char* name)
117 {
118 return lw8(std::forward<T>(write_fn), name);
119 }
120
121 template <typename T, typename U>
122 address_map_entry& rw(T&& read_fn, const char* read_name,
123 U&& write_fn, const char* write_name)
124 {
125 return lrw8(std::forward<T>(read_fn), read_name,
126 std::forward<U>(write_fn), write_name);
127 }
128
130 template <typename T, typename U>
131 address_map_entry& lrw8(T&& read_fn, const char* read_name,
132 U&& write_fn, const char* write_name);
133
135 template <typename T>
136 address_map_entry& lr8(T&& read_fn, const char* name);
137
139 template <typename T>
140 address_map_entry& lw8(T&& write_fn, const char* name);
141
143 address_map_entry& noprw();
144
146 address_map_entry& mirror(unsigned bits)
147 {
148 m_mirror = bits;
149 return *this;
150 }
151
152 address_map_entry& select(unsigned bits)
153 {
154 m_select = bits;
155 return *this;
156 }
157
158private:
159 offs_t handler_mask() const
160 {
161 offs_t changing_bits = m_lo ^ m_hi;
162 changing_bits |= changing_bits >> 1;
163 changing_bits |= changing_bits >> 2;
164 changing_bits |= changing_bits >> 4;
165 changing_bits |= changing_bits >> 8;
166 changing_bits |= changing_bits >> 16;
167 return changing_bits | m_select;
168 }
169
171 void for_each_addr(std::function<void(unsigned)> fn) const;
172
173 address_map& m_map;
174 unsigned m_lo;
175 unsigned m_hi;
176 unsigned m_mirror;
177 unsigned m_select;
178};

183class address_map
184{
185public:
186 explicit address_map(unsigned size = 256)
187 : m_size(size), m_slots(size), m_unmapval(0xFF), m_globalmask(0)
188 {
189 }
190
191 address_map_entry operator()(unsigned lo, unsigned hi)
192 {
193 return address_map_entry(*this, lo, hi);
194 }
195
196 // map-level
197
198 void unmap_value_high() { m_unmapval = 0xFF; }
199 void unmap_value_low() { m_unmapval = 0x00; }
200 void global_mask(offs_t mask) { m_globalmask = mask; }
201
202 // Dispatch
203
204 u8 read_byte(unsigned addr) const
205 {
206 if (m_globalmask) addr &= m_globalmask;
207 if (addr < m_size && m_slots[addr].read_fn)
208 return m_slots[addr].read_fn(static_cast<offs_t>(addr - m_slots[addr].read_base));
209 return m_unmapval;
210 }
211
212 void write_byte(unsigned addr, u8 data)
213 {
214 if (m_globalmask) addr &= m_globalmask;
215 if (addr < m_size && m_slots[addr].write_fn)
216 m_slots[addr].write_fn(static_cast<offs_t>(addr - m_slots[addr].write_base), data);
217 }
218
219 bool has_handler(unsigned addr) const
220 {
221 if (m_globalmask) addr &= m_globalmask;
222 if (addr >= m_size) return false;
223 return m_slots[addr].read_fn || m_slots[addr].write_fn;
224 }
225
226 bool has_read_handler(unsigned addr) const
227 {
228 if (m_globalmask) addr &= m_globalmask;
229 return addr < m_size && bool(m_slots[addr].read_fn);
230 }
231
232 bool has_write_handler(unsigned addr) const
233 {
234 if (m_globalmask) addr &= m_globalmask;
235 return addr < m_size && bool(m_slots[addr].write_fn);
236 }
237
238 const char* debug_name(unsigned addr) const
239 {
240 if (m_globalmask) addr &= m_globalmask;
241 return (addr < m_size) ? m_slots[addr].name : nullptr;
242 }
243
244 u8 unmapval() const { return m_unmapval; }
245
246 // called by address_map_entry
247
248 void install_read(unsigned addr, std::function<u8(offs_t)> fn, unsigned base, const char* name)
249 {
250 if (addr < m_size)
251 {
252 m_slots[addr].read_fn = std::move(fn);
253 m_slots[addr].read_base = base;
254 if (name) m_slots[addr].name = name;
255 }
256 }
257
258 void install_write(unsigned addr, std::function<void(offs_t, u8)> fn, unsigned base, const char* name)
259 {
260 if (addr < m_size)
261 {
262 m_slots[addr].write_fn = std::move(fn);
263 m_slots[addr].write_base = base;
264 if (name) m_slots[addr].name = name;
265 }
266 }
267
268 void install_nop(unsigned addr, unsigned base, const char* name)
269 {
270 if (addr < m_size)
271 {
272 m_slots[addr].read_fn = [](offs_t) -> u8 { return 0; };
273 m_slots[addr].write_fn = [](offs_t, u8) {};
274 m_slots[addr].read_base = base;
275 m_slots[addr].write_base = base;
276 if (name) m_slots[addr].name = name;
277 }
278 }
279
280 unsigned size() const { return m_size; }

282private:
283 struct Slot
284 {
285 std::function<u8(offs_t)> read_fn;
286 std::function<void(offs_t, u8)> write_fn;
287 unsigned read_base = 0;
288 unsigned write_base = 0;
289 const char* name = nullptr;
290 };
291
292 unsigned m_size;
293 std::vector<Slot> m_slots;
294 u8 m_unmapval;
295 offs_t m_globalmask;
296};

299// mirror expansion helper

301inline void address_map_entry::for_each_addr(std::function<void(unsigned)> fn) const
302{
303 const unsigned expanded_bits = m_mirror | m_select;
304 if (expanded_bits == 0)
305 {
306 for (unsigned a = m_lo; a <= m_hi; ++a)
307 fn(a);
308 }
309 else
310 {
311 // combinations of mirror bits
312 // any mirror bit can be 0 or 1 independently... i think
313 std::vector<unsigned> mirror_bits;
314 for (unsigned b = 0; b < 32; ++b)
315 if (expanded_bits & (1u << b))
316 mirror_bits.push_back(1u << b);
317
318 unsigned combos = 1u << mirror_bits.size();
319 for (unsigned c = 0; c < combos; ++c)
320 {
321 unsigned mirror_val = 0;
322 for (unsigned i = 0; i < mirror_bits.size(); ++i)
323 if (c & (1u << i))
324 mirror_val |= mirror_bits[i];
325
326 for (unsigned a = m_lo; a <= m_hi; ++a)
327 fn(a | mirror_val);
328 }
329 }
330}

335template <typename T, typename U>
336address_map_entry& address_map_entry::lrw8(T&& read_fn, const char* read_name,
337 U&& write_fn, const char* write_name)
338{
339 auto rfn = std::make_shared<std::function<u8(offs_t)>>(std::forward<T>(read_fn));
340 auto wfn = std::make_shared<std::function<void(offs_t, u8)>>(std::forward<U>(write_fn));
341 unsigned base = m_lo;
342 const offs_t mask = handler_mask();
343
344 for_each_addr([&](unsigned addr) {
345 m_map.install_read(addr, [rfn, mask](offs_t off) { return (*rfn)(off & mask); }, base, read_name);
346 m_map.install_write(addr, [wfn, mask](offs_t off, u8 d) { (*wfn)(off & mask, d); }, base, write_name);
347 });
348 return *this;
349}
