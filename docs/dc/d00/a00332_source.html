30constexpr AlphaHWLoadDescriptor ev4_decode_hw_load(std::uint32_t instruction)
31{
32 AlphaHWLoadDescriptor load{};
33 if ((instruction >> 26) != 0x1b)
34 return load;
35
36 // HW_LD bits 15:12 are PHY, ALT, RWC, QW (HRM Table 4-8).
37 const unsigned form = (instruction >> 12) & 0xf;
38 const bool physical = (form & 8) != 0;
39 load.size_bits = (form & 1) ? 64 : 32;
40 load.displacement = static_cast<std::int32_t>(instruction & 0xfff);
41 if (load.displacement & 0x800)
42 load.displacement -= 0x1000;
43 load.address = physical ? AlphaHWLoadAddress::Physical : AlphaHWLoadAddress::Virtual;
44 load.alternate = !physical && (form & 4) != 0;
45 load.write_check = !physical && (form & 2) != 0;
46 load.locked = physical && (form & 4) != 0;
47 load.compilable = !load.locked;
48 return load;
49}
