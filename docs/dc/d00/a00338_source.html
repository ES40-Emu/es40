127template <typename T, typename U, typename... V> constexpr T bitswap(T val, U b, V... c) noexcept
128{
129 if constexpr (sizeof...(c) > 0U)
130 return (BIT(val, b) << sizeof...(c)) | bitswap(val, c...);
131 else
132 return BIT(val, b);
133}
