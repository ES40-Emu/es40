30constexpr bool ev4_ieee_divide(std::uint32_t instruction)
31{
32 const auto function = (instruction >> 5) & 0x3f;
33 return (instruction >> 26) == 0x16 && (function == 0x03 || function == 0x23);
34}
35
36constexpr AlphaIEEETrapResult ev4_ieee_evaluate(std::uint64_t fpcr, std::uint64_t trap,
37 std::uint32_t instruction_enable, AlphaIEEETrapMasking, std::uint32_t instruction)
38{
39 const auto flags = trap & 0x7e;
40 // Original 21064 DIV never sets FPCR.INE (HRM 2.6/2.7).
41 const auto sticky = ev4_ieee_divide(instruction) ? flags & ~std::uint64_t(0x20) : flags;
42 fpcr |= sticky << 51;
43 if (instruction_enable && flags)
44 return {true, flags | ((instruction & 0x8000) ? 1u : 0u), fpcr};
45 return {false, 0, fpcr};
46}

48constexpr AlphaIEEEInexactRule ev4_ieee_describe_inexact(std::uint32_t instruction)
49{
50 return ev4_ieee_divide(instruction) ? AlphaIEEEInexactRule{AlphaIEEEInexactKind::Noop, 0}
51 : AlphaIEEEInexactRule{AlphaIEEEInexactKind::FpcrBitSet, 56};
52}

54constexpr std::uint64_t ev4_ieee_underflow_result(bool)
55{
56 return 0; // Underflow produces positive zero.
57}
