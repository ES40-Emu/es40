90{
91 /* Preserve sign of zero on underflow per IEEE-754 / HRM 4.7.7.3. */

95AlphaIEEETrapResult ev68_ieee_evaluate(std::uint64_t fpcr, std::uint64_t trap,
96 std::uint32_t instenb, AlphaIEEETrapMasking masking, std::uint32_t ins)
97{
98 u64 real_trap = U64(0x0);
99 u64 fpcrdsb = 0;
100 if (masking == AlphaIEEETrapMasking::SoftwareCompletion)
101 {
102 if (trap & TRAP_INV) fpcrdsb |= FPCR_INVD;
103 if (trap & TRAP_DZE) fpcrdsb |= FPCR_DZED;
104 if (trap & TRAP_OVF) fpcrdsb |= FPCR_OVFD;
105 if (trap & TRAP_UNF) fpcrdsb |= FPCR_UNFD;
106 if (trap & TRAP_INE) fpcrdsb |= FPCR_INED;
107 }
108
109 /* HRM 4.7.6: the FPCR sticky bit is set whenever the corresponding
110 exception is detected, regardless of whether a trap is delivered. */
111 if (~fpcr & (trap << 51)) // trap bit not set in FPCR
112 {
113 fpcr |= trap << 51; // set FPCR sticky bit (newly detected)
114 real_trap |= trap << 41; // SET trap bit in EXC_SUM
115 }
116 if ((instenb != 0) /* not enabled in inst? ignore */
117 && !((ins & I_FTRP_S) && (fpcr & fpcrdsb))) /* /S and disabled? ignore */real_trap |= trap; // trap bit in EXC_SUM
118 if (real_trap)
119 return {true, real_trap | ((ins & I_FTRP_S) ? TRAP_SWC : 0), fpcr};
120 return {false, 0, fpcr};
121}
