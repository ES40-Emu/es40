82void svga_device::svga_vh_rgb8(bitmap_rgb32& bitmap, const rectangle& cliprect)
83{
84 const int height = vga.crtc.maximum_scan_line * (vga.crtc.scan_doubling + 1);
85
86 const uint16_t mask_comp = line_compare_mask();
87 int curr_addr = 0;
88
89 // clear the backbuffer for 320x240 linear modes, avoid transitional artifacts
90 // bitmap.fill(0, cliprect);
91
92 // uint16_t line_length;
93 // if(vga.crtc.dw)
94 // line_length = vga.crtc.offset << 3; // doubleword mode
95 // else
96 // {
97 // line_length = vga.crtc.offset << 4;
98 // }
99
100 const int VGA_COLUMNS = vga.crtc.horz_disp_end + 1;
101 const int LINES = (vga.crtc.vert_disp_end + 1) * (get_interlace_mode() + 1);
102
103 uint8_t start_shift = (!(vga.sequencer.data[4] & 0x08) || svga.ignore_chain4) ? 2 : 0;
104 for (int addr = vga.crtc.start_addr << start_shift, line = 0; line < LINES; line += height, addr += offset(), curr_addr += offset())
105 {
106 for (int yi = 0; yi < height; yi++)
107 {
108 if ((line + yi) < (vga.crtc.line_compare & mask_comp))
109 curr_addr = addr;
110 if ((line + yi) == (vga.crtc.line_compare & mask_comp))
111 curr_addr = 0;
112 uint32_t* const bitmapline = &bitmap.pix(line + yi);
113 addr %= vga.svga_intf.vram_size;
114 for (int pos = curr_addr, c = 0, column = 0; column < VGA_COLUMNS; column++, c += 8, pos += 0x8)
115 {
116 if (pos + 0x08 >= vga.svga_intf.vram_size)
117 return;
118
119 for (int xi = 0; xi < 8; xi++)
120 {
121 const int dest_x = c + xi;
122
123 if (!screen().visible_area().contains(dest_x, line + yi))
124 continue;
125 bitmapline[dest_x] = pen(vga.memory[pos + xi]);
126 }
127 }
128 }
129 }

132void svga_device::svga_vh_rgb15(bitmap_rgb32& bitmap, const rectangle& cliprect)
133{
134 constexpr uint32_t IV = 0xff000000;
135 const unsigned height = vga.crtc.maximum_scan_line * (vga.crtc.scan_doubling + 1);
136
137 const unsigned TLINES = (vga.crtc.vert_disp_end + 1) * (get_interlace_mode() + 1);
138 const unsigned TGA_COLUMNS = vga.crtc.horz_disp_end + 1;
139 /* line compare is screen sensitive */
140// uint16_t mask_comp = 0xff | (TLINES & 0x300);
141 int curr_addr = 0;
142 // (void)curr_addr;
143 int yi = 0;
144
145 // bitmap.fill(0, cliprect);
146
147 for (unsigned addr = vga.crtc.start_addr << 2, line = 0; line < TLINES; line += height, addr += offset(), curr_addr += offset())
148 {
149 uint32_t* const bitmapline = &bitmap.pix(line);
150 addr %= vga.svga_intf.vram_size;
151 for (unsigned pos = addr, c = 0, column = 0; column < TGA_COLUMNS; column++, c += 8, pos += 0x10)
152 {
153 if (pos + 0x10 >= vga.svga_intf.vram_size)
154 return;
155 for (int xi = 0, xm = 0; xi < 8; xi++, xm += 2)
156 {
157 const int dest_x = c + xi;
158
159 if (!screen().visible_area().contains(dest_x, line + yi))
160 continue;
161
162 const u16 MV = (vga.memory[pos + xm] + (vga.memory[pos + xm + 1] << 8));
163 int r = (MV & 0x7c00) >> 10;
164 int g = (MV & 0x03e0) >> 5;
165 int b = (MV & 0x001f) >> 0;
166 r = (r << 3) | (r & 0x7);
167 g = (g << 3) | (g & 0x7);
168 b = (b << 3) | (b & 0x7);
169 bitmapline[dest_x] = IV | (r << 16) | (g << 8) | (b << 0);
170 }
171 }
172 }

300uint8_t svga_device::pc_vga_choosevideomode()
301{
302 if (vga.crtc.sync_en)
303 {
304 if (vga.dac.dirty)
305 {
306 palette_update();
307 vga.dac.dirty = 0;
308 }
309
310 if (vga.attribute.data[0x10] & 0x80)
311 {
312 for (int i = 0; i < 16; i++)
313 {
314 vga.pens[i] = pen((vga.attribute.data[i] & 0x0f)
315 | ((vga.attribute.data[0x14] & 0xf) << 4));
316 }
317 }
318 else
319 {
320 for (int i = 0; i < 16; i++)
321 {
322 vga.pens[i] = pen((vga.attribute.data[i] & 0x3f)
323 | ((vga.attribute.data[0x14] & 0xc) << 4));
324 }
325 }
326
327 if (svga.rgb32_en)
328 {
329 return RGB32_MODE;
330 }
331 else if (svga.rgb24_en)
332 {
333 return RGB24_MODE;
334 }
335 else if (svga.rgb16_en)
336 {
337 return RGB16_MODE;
338 }
339 else if (svga.rgb15_en)
340 {
341 return RGB15_MODE;
342 }
343 else if (svga.rgb8_en)
344 {
345 return RGB8_MODE;
346 }
347 else if (!vga.gc.alpha_dis) // !GRAPHIC_MODE
348 {
349 return TEXT_MODE;
350 }
351 else if (vga.gc.shift256)
352 {
353 return VGA_MODE;
354 }
355 else if (vga.gc.shift_reg)
356 {
357 return CGA_MODE;
358 }
359 else if (vga.gc.memory_map_sel == 0x03)
360 {
361 return MONO_MODE;
362 }
363 else
364 {
365 return EGA_MODE;
366 }
367 }
368
369 return SCREEN_OFF;
370}

391{

401 case MONO_MODE: vga_vh_mono(bitmap, cliprect); break;
402 case RGB8_MODE: svga_vh_rgb8(bitmap, cliprect); break;
403 case RGB15_MODE: svga_vh_rgb15(bitmap, cliprect); break;
404 case RGB16_MODE: svga_vh_rgb16(bitmap, cliprect); break;
405 case RGB24_MODE: svga_vh_rgb24(bitmap, cliprect); break;
