78std::uint64_t ev68_trace_address_fallback(std::uint64_t address, bool instruction)
79{
80 const auto region = address >> 28;
81 if (region == 2)
82 return address - 0x1fe00000; // This trace alias preserves the low bits.
83 if (region == 1)
84 address -= 0x0fffe000;
85 return instruction ? address & ~U64(3) : address;
86}

88void Ev68PALCompatibility::trace_process_name(const CAlphaCPU& cpu, std::uint64_t prbr,
89 char* name, std::size_t capacity)
90{
91 if (!capacity)
92 return;
93 name[0] = '\0';
94 if (cpu.state.pal_base != kVmsPalBase || !prbr)
95 return;
96
97 constexpr std::uint64_t offset = 0x154;
98 constexpr std::size_t field_size = 20;
99 const auto ram_size = U64(1) << cpu.cSystem->get_memory_bits();
100 if (prbr >= ram_size || ram_size - prbr < offset + field_size)
101 return;
102
103 const auto length = capacity > field_size ? field_size : capacity - 1;
104 strncpy(name, cpu.cSystem->PtrToMem(prbr + offset), length);
105 name[length] = '\0';
106}

119std::uint64_t Ev68PALCompatibility::trace_prbr(CAlphaCPU& cpu)
120{
121 u64 v_prbr; // virtual
122 u64 p_prbr; // physical
123 bool b;
124 if (cpu.state.r[21 + 32] && ((u64)(cpu.state.r[21 + 32] + 0xaf) < (u64)
125 ((U64(0x1) << cpu.cSystem->get_memory_bits()))))
126 v_prbr = cpu.cSystem->ReadMem(cpu.state.r[21 + 32] + 0xa8, 64, &cpu);
127 else
128 v_prbr = cpu.cSystem->ReadMem(0x70a8 + (0x200 * cpu.get_cpuid()), 64, &cpu);
129 if (cpu.virt2phys(v_prbr, &p_prbr, ACCESS_READ | FAKE | NO_CHECK, &b, 0))
130 p_prbr = v_prbr;
131 if ((u64)p_prbr > (u64)(U64(0x1) << cpu.cSystem->get_memory_bits()))
132 p_prbr = 0;
133 return p_prbr;
134}

153void Ev68PALCompatibility::interpreter_step(CAlphaCPU& cpu)
154{
155 //--------------------------------------------------------------------------------
156 // Interpreter-only SRM memory-test shortcuts.
157 //--------------------------------------------------------------------------------
158#if defined(SKIP_SRM_MEMTEST) && !defined(ES40_JIT)
159 auto& state = cpu.state;
160 // All five SRM mem-test patch points are in page 0x8b000 and only fire
161 // during early SRM boot. Gate on the page so every other instruction pays
162 // one compare instead of five.
163 // JIT builds must NOT use these hooks - a block-cache miss interprets a few
164 // iterations, thus only one hook like 0x8bb90 fires alone, and aborts the
165 // fill while a still-compiled verify loop then runs, warm inits show mem errors
166 // JIT is fast enough, even 16GB configs don't take long.
167 if ((state.current_pc & ~U64(0xFFF)) == U64(0x8b000))
168 {
169 if (state.current_pc == U64(0x8bb90))
170 {
171 if (state.r[5] != U64(0xaaaaaaaaaaaaaaaa))
172 {
173 printf("wrong memory check skip!\n");
174 }
175 else
176 {
177 state.r[0] = state.r[4];
178 }
179 }
180
181 if (state.current_pc == U64(0x8bbe0))
182 {
183 if (state.r[5] != U64(0xaaaaaaaaaaaaaaaa))
184 {
185 printf("wrong memory check skip!\n");
186 }
187 else
188 {
189 state.r[16] = 0;
190 }
191 }
192
193 if (state.current_pc == U64(0x8bc28))
194 {
195 if (state.r[5] != U64(0xaaaaaaaaaaaaaaaa))
196 {
197 printf("wrong memory check skip!\n");
198 }
199 else
200 {
201 state.r[8] = state.r[4];
202 }
203 }
204
205 if (state.current_pc == U64(0x8bc70))
206 {
207 if (state.r[7] != U64(0x5555555555555555))
208 {
209 printf("wrong memory check skip1!\n");
210 }
211 else
212 {
213 state.r[0] = 0;
214 }
215 }
216
217 if (state.current_pc == U64(0x8bcb0))
218 {
219 if (state.r[7] != U64(0x5555555555555555))
220 {
221 printf("wrong memory check skip2!\n");
222 }
223 else
224 {
225 state.r[3] = state.r[4];
226 }
227 }
228 }
229#else
230 (void)cpu;
231#endif
232 //--------------------------------------------------------------------------------
233 // end of skip memory test section
234 //--------------------------------------------------------------------------------
235}

237bool Ev68PALCompatibility::instruction_pacing(const CAlphaCPU& cpu)
238{
239 // Gated off for the firmware/VMS PALcode (base 0x8000, and the pre-PAL reset state):
240 // SRM runs under it, and SRM's per-CPU speed calibration counts cycles
241 // per tick in a tight spin. NT and OSF-style PALcodes load at other bases and are paced.
242 return cpu.state.pal_base && cpu.state.pal_base != kVmsPalBase;
243}

245bool Ev68PALCompatibility::wait_for_interrupt(CAlphaCPU& cpu)
246{
247 if (cpu.idle_nap_enabled && cpu.state.pal_base == kVmsPalBase)
248 {
249 /* WTINT before the VMS vs native PAL decision, to accommodate
250 * the WTINT usage scenario under JIT */
251 cpu.vmspal_call_wtint();
252 return true;
253 }
254 if (!cpu.state.pal_vms)
255 {
256 /* Non-VMS WTINT implementation */
257 cpu.idle_nap();
258 return true;
259 }
260 return false;

472void Ev68PALCompatibility::base_changed(CAlphaCPU& cpu, std::uint64_t old_base, bool old_hle)
473{
474 auto& state = cpu.state;
475 const u64 pb = state.pal_base;
476
477 // VMS PALcode uses base 0x8000
478 state.pal_vms = (pb == kVmsPalBase) && !cpu.vmspal_lle_enabled;
479 //state.pal_vms = false;
480
481#ifdef DEBUG_PAL
482 if (pb != old_base || old_hle != state.pal_vms)
483 {
484 // Only the VMS PALcode (base 0x8000) is a type the emulator knows about.
485 const char* type = (pb == kVmsPalBase) ? "VMS PALcode" : "unknown PALcode type";
486 printf("%%CPU-I-PALSWITCH: cpu%d PAL_BASE %016" PRIx64 " -> %016" PRIx64 ": %s, %s\n",
487 cpu.get_cpuid(), old_base, pb, type,
488 state.pal_vms ? "vmspal replacement routines engaged" :
489 cpu.vmspal_lle_enabled ? "running native (replacements disabled)" :
490 "running native (no replacements for this type)");
491 }
492#endif
493
494#ifdef DEBUG_PAL
495 printf("%%CPU-I-PALSWITCH: PAL=%016" PRIx64 " p21=%016" PRIx64 " p22=%016" PRIx64 " r22=%016" PRIx64 "\n", pb, state.r[53], state.r[54], state.r[22]);
496 // Dump PAL scratch area contents for non-VMS PAL
497 if (!state.pal_vms && state.r[53] != 0) {
498
499 u64 scratch = U64(0x7cf420);
500
501 printf("%%CPU-I-PALSCR: Scratch area at %016" PRIx64 ":\n", scratch);
502 printf("%%CPU-I-PALSCR: +0x00 VPTB = %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x00, 64, &cpu));
503 printf("%%CPU-I-PALSCR: +0x08 PTBR = %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x08, 64, &cpu));
504 printf("%%CPU-I-PALSCR: +0x10 PCBB = %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x10, 64, &cpu));
505 printf("%%CPU-I-PALSCR: +0x18 KSP = %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x18, 64, &cpu));
506 printf("%%CPU-I-PALSCR: +0x98 WHAMI= %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x98, 64, &cpu));
507 printf("%%CPU-I-PALSCR: +0x170 SCBB= %016" PRIx64 "\n", cpu.cSystem->ReadMem(scratch + 0x170, 64, &cpu));
508 }
509 else if (!state.pal_vms && state.r[53] == 0) {
510 printf("%%CPU-W-NOP21: PAL switched but p21=0! Scratch area not available.\n");
511 }
512#endif
