41inline std::uint32_t mix(std::uint32_t first, std::uint32_t second, unsigned fraction)
42{
43 const unsigned inverse = 256 - fraction;
44 const std::uint32_t rb = (((first & 0x00ff00ff) * inverse +
45 (second & 0x00ff00ff) * fraction + 0x00800080) >> 8) & 0x00ff00ff;
46 const std::uint32_t g = (((first & 0x0000ff00) * inverse +
47 (second & 0x0000ff00) * fraction + 0x00008000) >> 8) & 0x0000ff00;
48 return rb | g;
49}
