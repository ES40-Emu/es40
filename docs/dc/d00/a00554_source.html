65static bool check_socket()
66{
67 struct sockaddr_storage address;
68 socklen_t address_len = sizeof(address);
69 int type = 0;
70 socklen_t type_len = sizeof(type);
71
72 if (getsockopt(socket_fd, SOL_SOCKET, SO_TYPE, &type, &type_len) != 0 ||
73 type != SOCK_DGRAM)
74 return false;
75 if (getpeername(socket_fd, (struct sockaddr *)&address,
76 &address_len) != 0)
77 return false;
78 return address.ss_family == AF_UNIX;
79}

81static bool send_reply(uint32_t status, uint32_t max_packet,
82 int system_error)
83{
84 struct es40_vmnet_open_reply reply;
85
86 memset(&reply, 0, sizeof(reply));
87 reply.msg.type = ES40_VMNET_MSG_OPEN_REPLY;
88 reply.magic = ES40_VMNET_PROTO_MAGIC;
89 reply.version = ES40_VMNET_PROTO_VERSION;
90 reply.status = status;
91 reply.helper_euid = (uint32_t)initial_euid;
92 reply.max_packet_size = max_packet;
93 reply.system_error = system_error;
94 ssize_t sent;
95 do {
96 sent = send(socket_fd, &reply, sizeof(reply), MSG_DONTWAIT);
97 } while (sent < 0 && errno == EINTR);
98 return sent == (ssize_t)sizeof(reply);
99}

114static vmnet_return_t start_vmnet(const char *adapter)
115{
116 dispatch_semaphore_t started;
117 xpc_object_t description;
118 __block vmnet_return_t status = VMNET_FAILURE;
119
120 dispatch_q = dispatch_queue_create("es40.vmnet",
121 DISPATCH_QUEUE_SERIAL);
122 if (dispatch_q == nullptr)
123 return VMNET_MEM_FAILURE;
124 started = dispatch_semaphore_create(0);
125 if (started == nullptr)
126 return VMNET_MEM_FAILURE;
127 description = xpc_dictionary_create(NULL, NULL, 0);
128 if (description == nullptr) {
129 dispatch_release(started);
130 return VMNET_MEM_FAILURE;
131 }
132
133 xpc_dictionary_set_uint64(description, vmnet_operation_mode_key,
134 VMNET_BRIDGED_MODE);
135 xpc_dictionary_set_string(description, vmnet_shared_interface_name_key,
136 adapter);
137 xpc_dictionary_set_bool(description, vmnet_allocate_mac_address_key,
138 false);
139 vmnet_if = vmnet_start_interface(description, dispatch_q,
140 ^(vmnet_return_t result, xpc_object_t __nullable parameters) {
141 status = result;
142 if (result == VMNET_SUCCESS) {
143 max_packet_size = xpc_dictionary_get_uint64(parameters,
144 vmnet_max_packet_size_key);
145 }
146 dispatch_semaphore_signal(started);
147 });
148 xpc_release(description);
149
150 /* A null interface does not run the completion block. */
151 if (vmnet_if != nullptr)
152 dispatch_semaphore_wait(started, DISPATCH_TIME_FOREVER);
153 dispatch_release(started);
154 return status;
155}

157static bool stop_vmnet()
158{
159 bool stopped_cleanly = true;
160
161 if (dispatch_q != nullptr) {
162 dispatch_sync(dispatch_q, ^{
163 forwarding = false;
164 });
165 }
166 if (vmnet_if != nullptr) {
167 dispatch_semaphore_t stopped = dispatch_semaphore_create(0);
168 vmnet_interface_set_event_callback(vmnet_if, 0, NULL, NULL);
169 dispatch_sync(dispatch_q, ^{});
170 if (stopped == nullptr)
171 return false;
172 vmnet_return_t status = vmnet_stop_interface(vmnet_if, dispatch_q,
173 ^(vmnet_return_t result) {
174 (void)result;
175 dispatch_semaphore_signal(stopped);
176 });
177 if (status == VMNET_SUCCESS) {
178 dispatch_semaphore_wait(stopped, DISPATCH_TIME_FOREVER);
179 vmnet_if = nullptr;
180 } else {
181 stopped_cleanly = false;
182 }
183 dispatch_release(stopped);
184 }
185 if (dispatch_q != nullptr && stopped_cleanly) {
186 dispatch_release(dispatch_q);
187 dispatch_q = nullptr;
188 }
189 return stopped_cleanly;
190}

192static bool drop_privileges(int *saved_error)
193{
194 uid_t uid = getuid();
195 gid_t gid = getgid();
196
197 if (setgid(gid) != 0 || setuid(uid) != 0) {
198 *saved_error = errno;
199 return false;
200 }
201 if (geteuid() != uid || getegid() != gid) {
202 *saved_error = EPERM;
203 return false;
204 }
205 return true;
206}

208static bool send_packet(const void *data, size_t len)
209{
210 struct es40_vmnet_msg header = { ES40_VMNET_MSG_PACKET };
211 struct iovec iov[2] = {
212 { &header, sizeof(header) },
213 { (void *)data, len }
214 };
215 struct msghdr message;
216 memset(&message, 0, sizeof(message));
217 message.msg_iov = iov;
218 message.msg_iovlen = 2;
219
220 ssize_t sent;
221 do {
222 sent = sendmsg(socket_fd, &message, MSG_DONTWAIT);
223 } while (sent < 0 && errno == EINTR);
224 if (sent == (ssize_t)(sizeof(header) + len))
225 return true;
226 return sent < 0 && (errno == EAGAIN || errno == EWOULDBLOCK ||
227 errno == ENOBUFS);
228}

230static void forward_packets(uint64_t estimated)
231{
232 if (!forwarding)
233 return;
234 (void)estimated;
235 uint64_t budget = 256;
236
237 while (budget-- > 0) {
238 struct iovec iov = { rx_buf, (size_t)max_packet_size };
239 struct vmpktdesc packet;
240 int count = 1;
241
242 packet.vm_pkt_size = max_packet_size;
243 packet.vm_pkt_iov = &iov;
244 packet.vm_pkt_iovcnt = 1;
245 packet.vm_flags = 0;
246 vmnet_return_t status = vmnet_read(vmnet_if, &packet, &count);
247 if (status != VMNET_SUCCESS) {
248 send_status(ES40_VMNET_OP_READ, status);
249 return;
250 }
251 if (count == 0)
252 return;
253 if (packet.vm_pkt_size > max_packet_size ||
254 !send_packet(rx_buf, (size_t)packet.vm_pkt_size))
255 return;
256 }
257
258 /* Yield to queued control work, then continue draining the interface. */
259 dispatch_async(dispatch_q, ^{
260 forward_packets(0);
261 });
262}

264static void pump_packets(void *tx_buf, pid_t parent_pid)
265{
266 struct pollfd poll_fd;
267 poll_fd.fd = socket_fd;
268 poll_fd.events = POLLIN;
269
270 for (;;) {
271 int ready = poll(&poll_fd, 1, 1000);
272 if (ready < 0) {
273 if (errno == EINTR)
274 continue;
275 return;
276 }
277 if (ready == 0) {
278 if (getppid() != parent_pid)
279 return;
280 continue;
281 }
282 if (poll_fd.revents & (POLLERR | POLLHUP | POLLNVAL))
283 return;
284
285 for (;;) {
286 struct es40_vmnet_msg header;
287 struct iovec iov[2] = {
288 { &header, sizeof(header) },
289 { tx_buf, (size_t)max_packet_size }
290 };
291 struct msghdr message;
292 memset(&message, 0, sizeof(message));
293 message.msg_iov = iov;
294 message.msg_iovlen = 2;
295
296 ssize_t received = recvmsg(socket_fd, &message, MSG_DONTWAIT);
297 if (received < 0) {
298 if (errno == EINTR)
299 continue;
300 if (errno == EAGAIN || errno == EWOULDBLOCK)
301 break;
302 return;
303 }
304 if (received < (ssize_t)sizeof(header))
305 return;
306 if (header.type == ES40_VMNET_MSG_SHUTDOWN &&
307 received == (ssize_t)sizeof(header))
308 return;
309 if ((message.msg_flags & MSG_TRUNC) ||
310 header.type != ES40_VMNET_MSG_PACKET ||
311 received <= (ssize_t)sizeof(header))
312 return;
313
314 size_t len = (size_t)received - sizeof(header);
315 struct iovec packet_iov = { tx_buf, len };
316 struct vmpktdesc packet;
317 int count = 1;
318
319 packet.vm_pkt_size = len;
320 packet.vm_pkt_iov = &packet_iov;
321 packet.vm_pkt_iovcnt = 1;
322 packet.vm_flags = 0;
323 vmnet_return_t status = vmnet_write(vmnet_if, &packet,
324 &count);
325 if (status != VMNET_SUCCESS)
326 send_status(ES40_VMNET_OP_WRITE, status);
327 else if (count == 0)
328 send_status(ES40_VMNET_OP_WRITE,
329 VMNET_BUFFER_EXHAUSTED);
330 }
331 }
332}

334int main()
335{
336 struct es40_vmnet_open_req request;
337 vmnet_return_t status;
338 void *tx_buf = nullptr;
339 int saved_error = 0;
340 pid_t parent_pid = getppid();
341
342 initial_euid = geteuid();
343 if (!check_socket()) {
344 fprintf(stderr, ES40_VMNET_HELPER_NAME
345 ": must be started by the es40 emulator\n");
346 return 2;
347 }
348
349 long max_fd = sysconf(_SC_OPEN_MAX);
350 for (int fd = ES40_VMNET_HELPER_FD + 1;
351 max_fd > 0 && fd < max_fd; fd++)
352 close(fd);
353 signal(SIGPIPE, SIG_IGN);
354
355 struct timeval timeout = { 10, 0 };
356 setsockopt(socket_fd, SOL_SOCKET, SO_RCVTIMEO, &timeout,
357 sizeof(timeout));
358 struct iovec request_iov = { &request, sizeof(request) };
359 struct msghdr request_message;
360 memset(&request_message, 0, sizeof(request_message));
361 request_message.msg_iov = &request_iov;
362 request_message.msg_iovlen = 1;
363 ssize_t received;
364 do {
365 received = recvmsg(socket_fd, &request_message, 0);
366 } while (received < 0 && errno == EINTR);
367 if (received != (ssize_t)sizeof(request) ||
368 (request_message.msg_flags & MSG_TRUNC))
369 return 1;
370 if (request.msg.type != ES40_VMNET_MSG_OPEN ||
371 request.magic != ES40_VMNET_PROTO_MAGIC ||
372 request.version != ES40_VMNET_PROTO_VERSION) {
373 send_reply(ES40_VMNET_STATUS_HELPER_ERROR, 0, EPROTO);
374 return 1;
375 }
376 request.adapter[IFNAMSIZ - 1] = '\0';
377 if (request.adapter[0] == '\0') {
378 send_reply(ES40_VMNET_STATUS_HELPER_ERROR, 0, EINVAL);
379 return 1;
380 }
381
382 status = start_vmnet(request.adapter);
383 if (status != VMNET_SUCCESS) {
384 send_reply((uint32_t)status, 0, 0);
385 stop_vmnet();
386 return 1;
387 }
388 if (max_packet_size == 0 ||
389 max_packet_size > ES40_VMNET_MAX_FRAME) {
390 send_reply(ES40_VMNET_STATUS_HELPER_ERROR, 0, EOVERFLOW);
391 stop_vmnet();
392 return 1;
393 }
394 rx_buf = malloc((size_t)max_packet_size);
395 tx_buf = malloc((size_t)max_packet_size);
396 if (rx_buf == nullptr || tx_buf == nullptr) {
397 send_reply(ES40_VMNET_STATUS_HELPER_ERROR, 0, ENOMEM);
398 stop_vmnet();
399 free(rx_buf);
400 free(tx_buf);
401 return 1;
402 }
403
404 status = vmnet_interface_set_event_callback(vmnet_if,
405 VMNET_INTERFACE_PACKETS_AVAILABLE, dispatch_q,
406 ^(interface_event_t events, xpc_object_t event) {
407 if (events & VMNET_INTERFACE_PACKETS_AVAILABLE) {
408 forward_packets(xpc_dictionary_get_uint64(event,
409 vmnet_estimated_packets_available_key));
410 }
411 });
412 if (status != VMNET_SUCCESS) {
413 send_reply((uint32_t)status, 0, 0);
414 stop_vmnet();
415 free(rx_buf);
416 free(tx_buf);
417 return 1;
418 }
419
420 /* Complete all privileged vmnet setup before permanently dropping root. */
421 if (!drop_privileges(&saved_error)) {
422 send_reply(ES40_VMNET_STATUS_HELPER_ERROR, 0, saved_error);
423 if (stop_vmnet()) {
424 free(rx_buf);
425 free(tx_buf);
426 }
427 return 1;
428 }
429
430 if (send_reply(ES40_VMNET_STATUS_OK,
431 (uint32_t)max_packet_size, 0)) {
432 dispatch_sync(dispatch_q, ^{
433 forwarding = true;
434 forward_packets(1);
435 });
436 pump_packets(tx_buf, parent_pid);
437 }
438
439 if (stop_vmnet()) {
440 free(rx_buf);
441 free(tx_buf);
442 }
443 return 0;
444}
