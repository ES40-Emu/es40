42constexpr std::uint32_t magic = 0x31424649; // IFB1, little endian
43constexpr std::size_t header_bytes = 64;
44constexpr unsigned max_dimension = 4096;
45// Fixed capacity preserves reader mappings across mode changes.
46constexpr std::size_t capacity = header_bytes + std::size_t(max_dimension) * max_dimension * 4;

64void store_sequence(std::uint64_t* address, std::uint64_t value)
65{
66#if defined(_WIN32)
67 InterlockedExchange64(reinterpret_cast<volatile LONG64*>(address), static_cast<LONG64>(value));
68#elif defined(ES40_SHM_POSIX)
69 static_assert(__atomic_always_lock_free(8, nullptr), "IFB1 requires lock-free 64-bit atomics");
70 // Order sequence updates around pixel writes.
71 __atomic_exchange_n(address, value, __ATOMIC_SEQ_CST);
72#endif
73}

99 void open(const char* path)
100 {
101 const std::uint32_t endian = 1;
102 if (*reinterpret_cast<const unsigned char*>(&endian) != 1)
103 fail("export requires a little-endian host");
104 if (!path || !*path) fail("empty path");
105 std::uint64_t size = 0;
106#if defined(_WIN32)
107 // Allow readers while excluding other writers.
108 file = CreateFileA(path, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ,
109 nullptr, OPEN_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
110 if (file == INVALID_HANDLE_VALUE) system_fail("open (path must be writable and have no other writer)");
111 LARGE_INTEGER length;
112 if (!GetFileSizeEx(file, &length)) system_fail("get size");
113 size = static_cast<std::uint64_t>(length.QuadPart);
114#elif defined(ES40_SHM_POSIX)
115 file = ::open(path, O_RDWR | O_CREAT | O_CLOEXEC, 0600);
116 if (file < 0) system_fail("open");
117 if (flock(file, LOCK_EX | LOCK_NB) != 0) system_fail("lock (another writer may be using this path)");
118 struct stat info;
119 if (fstat(file, &info) != 0) system_fail("stat");
120 if (!S_ISREG(info.st_mode)) fail("path must name a regular file");
121 size = static_cast<std::uint64_t>(info.st_size);
122#else
123 fail("export is not supported on this host");
124#endif
125 if (size != 0 && size < capacity)
126 fail("existing file is too small; use a new path or remove it after stopping its readers");
127 if (size == 0)
128 {
129#if defined(_WIN32)
130 LARGE_INTEGER length;
131 length.QuadPart = capacity;
132 if (!SetFilePointerEx(file, length, nullptr, FILE_BEGIN) || !SetEndOfFile(file))
133 system_fail("allocate file");
134#elif defined(ES40_SHM_POSIX)
135 if (ftruncate(file, static_cast<off_t>(capacity)) != 0) system_fail("allocate file");
136#endif
137 }
138#if defined(_WIN32)
139 mapping = CreateFileMappingA(file, nullptr, PAGE_READWRITE, 0, 0, nullptr);
140 if (!mapping) system_fail("create mapping");
141 view = static_cast<unsigned char*>(MapViewOfFile(mapping, FILE_MAP_WRITE, 0, 0, capacity));
142 if (!view) system_fail("map file");
143#elif defined(ES40_SHM_POSIX)
144 void* mapped = mmap(nullptr, capacity, PROT_READ | PROT_WRITE, MAP_SHARED, file, 0);
145 if (mapped == MAP_FAILED) system_fail("map file");
146 view = static_cast<unsigned char*>(mapped);
147#endif
148 auto* header = reinterpret_cast<std::uint32_t*>(view);
149 if (size != 0 && (header[0] != magic || header[1] != 1 || header[5] != 32))
150 fail("existing file is not an IFB1 v1 framebuffer");
151 // Preserve the sequence across producer restarts.
152 std::memcpy(&sequence, view + 24, sizeof(sequence));
153 sequence &= ~std::uint64_t(1);
154 store_sequence(reinterpret_cast<std::uint64_t*>(view + 24), sequence + 1);
155 header[1] = 1;
156 header[5] = 32;
157 // Publish header constants before magic for existing IFB1 readers.
158 std::memset(view + 48, 0, 16);
159#if defined(_WIN32)
160 InterlockedExchange(reinterpret_cast<volatile LONG*>(view), magic);
161#elif defined(ES40_SHM_POSIX)
162 __atomic_store_n(header, magic, __ATOMIC_RELEASE);
163#endif
164 }
165};

174void CShmFramebuffer::publish(const std::uint32_t* pixels, unsigned width, unsigned height)
175{
176 if (!width || !height || width > max_dimension || height > max_dimension)
177 fail("frame dimensions must be in 1..4096");
178 auto* view = m_impl->view;
179 auto* sequence = reinterpret_cast<std::uint64_t*>(view + 24);
180 store_sequence(sequence, m_impl->sequence + 1);
181 auto* header = reinterpret_cast<std::uint32_t*>(view);
182 header[2] = width;
183 header[3] = height;
184 header[4] = width * 4;
185 header[8] = header[9] = 0;
186 header[10] = width;
187 header[11] = height;
188 const std::size_t bytes = std::size_t(width) * height * 4;
189 if (pixels) std::memcpy(view + header_bytes, pixels, bytes);
190 else std::memset(view + header_bytes, 0, bytes);
191 m_impl->sequence += 2;
192 store_sequence(sequence, m_impl->sequence);
193}
