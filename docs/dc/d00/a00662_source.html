226CKeyboard::InputResult CKeyboard::queue_input(const InputEvent* events, size_t count,
227 u64* batch_id, const unsigned* expected_buttons, size_t source)
228{
229 CFastMutex::ScopedLock guard(kbdLock);
230 if (source >= INPUT_SOURCES || (count && !events))
231 return InputResult::Invalid;
232 if (count > INPUT_CAPACITY)
233 return InputResult::Full;
234 for (size_t i = 0; i < count; ++i)
235 {
236 const InputEvent& event = events[i];
237 if (event.type == InputEvent::Type::Key)
238 {
239 if ((event.key & ~BX_KEY_RELEASED) >= BX_KEY_NBKEYS)
240 return InputResult::Invalid;
241 }
242 else if (event.type == InputEvent::Type::Mouse)
243 {
244 if (event.buttons > 7 || event.dx < -32767 || event.dx > 32767
245 || event.dy < -32767 || event.dy > 32767
246 || event.dz < -32767 || event.dz > 32767)
247 return InputResult::Invalid;
248 }
249 else
250 return InputResult::Invalid;
251 }
252 if (StopThread || input_release_pending[source])
253 return InputResult::Unavailable;
254 const bool buttons_changed = expected_buttons && *expected_buttons != input_buttons_locked(source);
255 for (size_t i = 0; i < count; ++i)
256 if (!input_available(events[i])
257 || (buttons_changed && events[i].type == InputEvent::Type::Mouse))
258 return InputResult::Unavailable;
259 if (count > INPUT_CAPACITY - input_count)
260 return InputResult::Full;
261 size_t tag = 0;
262 if (batch_id)
263 {
264 if (last_input_batch == (u64)-1)
265 return InputResult::Full;
266 size_t slot = INPUT_CAPACITY;
267 for (size_t i = 0; i < INPUT_CAPACITY; ++i)
268 {
269 if (!input_history[i].id)
270 {
271 slot = i;
272 break;
273 }
274 if (!input_history[i].pending && (slot == INPUT_CAPACITY
275 || input_history[i].id < input_history[slot].id))
276 slot = i;
277 }
278 if (slot == INPUT_CAPACITY)
279 return InputResult::Full;
280 input_history[slot] = { ++last_input_batch, count, 0, 0 };
281 *batch_id = last_input_batch;
282 tag = slot + 1;
283 }
284 for (size_t i = 0; i < count; ++i)
285 {
286 const size_t index = (input_head + input_count++) % INPUT_CAPACITY;
287 input_queue[index] = events[i];
288 input_batch_tags[index] = tag;
289 input_sources[index] = source;
290 }
291 return InputResult::Accepted;
292}

347void CKeyboard::release_input_locked(size_t source)
348{
349 size_t kept = 0;
350 for (size_t i = 0; i < input_count; ++i)
351 {
352 const size_t index = (input_head + i) % INPUT_CAPACITY;
353 if (input_sources[index] == source)
354 finish_input_event(index, true);
355 else
356 {
357 const size_t target = (input_head + kept++) % INPUT_CAPACITY;
358 input_queue[target] = input_queue[index];
359 input_batch_tags[target] = input_batch_tags[index];
360 input_sources[target] = input_sources[index];
361 }
362 }
363 input_count = kept;
364 input_release_pending[source] = external_buttons[source] != 0;
365 for (bool down : external_keys[source])
366 input_release_pending[source] = input_release_pending[source] || down;
367}

463void CKeyboard::service_input_release()
464{
465 for (size_t source = 0; source < INPUT_SOURCES; ++source)
466 {
467 if (!input_release_pending[source])
468 continue;
469 bool waiting = false;
470 for (u32 key = 0; key < BX_KEY_NBKEYS; ++key)
471 {
472 if (!external_keys[source][key])
473 continue;
474 if (other_key_held(key, source))
475 external_keys[source][key] = false;
476 else if (input_busy() || !gen_scancode_locked(key | BX_KEY_RELEASED))
477 waiting = true;
478 else
479 external_keys[source][key] = false;
480 }
481 if (external_buttons[source])
482 {
483 const unsigned remaining = combined_buttons(source);
484 bool released = (external_buttons[source] | remaining) == remaining;
485 if (!released)
486 {
487 InputEvent event;
488 event.type = InputEvent::Type::Mouse;
489 released = !input_busy() && input_available(event)
490 && mouse_enQ_packet((u8)(0x08 | remaining), 0, 0, 0);
491 }
492 if (released)
493 {
494 external_buttons[source] = 0;
495 state.mouse.button_status = (u8)remaining;
496 }
497 else
498 waiting = true;
499 }
500 input_release_pending[source] = waiting;
501 }
502}

504void CKeyboard::service_input()
505{
506 if (StopThread)
507 return;
508 // A disabled device may defer one source's release without blocking other input.
509 service_input_release();
510 while (input_count && !input_available(input_queue[input_head]))
511 {
512 finish_input_event(input_head, true);
513 input_head = (input_head + 1) % INPUT_CAPACITY;
514 --input_count;
515 }
516 if (!input_count)
517 return;
518 // Finish earlier input before admitting another key sequence or mouse packet.
519 if (input_busy())
520 return;
521 InputEvent& event = input_queue[input_head];
522 const size_t source = input_sources[input_head];
523 if (event.type == InputEvent::Type::Key)
524 {
525 if (!apply_input_key(event.key, source))
526 return;
527 }
528 else
529 {
530 const int dx = event.dx > 254 ? 254 : event.dx < -254 ? -254 : event.dx;
531 const int dy = event.dy > 254 ? 254 : event.dy < -254 ? -254 : event.dy;
532 const int dz = event.dz > 127 ? 127 : event.dz < -127 ? -127 : event.dz;
533 const unsigned buttons = event.buttons | combined_buttons(source);
534 const u8 header = (u8)(0x08 | buttons | (dx < 0 ? 0x10 : 0)
535 | (dy < 0 ? 0x20 : 0));
536 if (!mouse_enQ_packet(header, (u8)dx, (u8)dy, (u8)-dz))
537 return;
538 external_buttons[source] = event.buttons;
539 state.mouse.button_status = (u8)buttons;
540 event.dx -= dx;
541 event.dy -= dy;
542 event.dz -= dz;
543 if (event.dx || event.dy || event.dz)
544 return;
545 }
546 finish_input_event(input_head, false);
547 input_head = (input_head + 1) % INPUT_CAPACITY;
548 --input_count;
549}

583bool CKeyboard::apply_input_key(u32 key, size_t source)
584{
585 const u32 code = key & ~BX_KEY_RELEASED;
586 if (code >= BX_KEY_NBKEYS || source > INPUT_SOURCES)
587 return false;
588 const bool external = source != INPUT_SOURCES;
589 bool& held = external ? external_keys[source][code] : local_keys[code];
590 const bool other = other_key_held(code, source);
591 const bool down = (key & BX_KEY_RELEASED) == 0;
592 if (!external && down
593 && (!state.kbd_clock_enabled || !state.kbd_internal_buffer.scanning_enabled))
594 return false;
595 if (!external && !down)
596 held = false;
597 if ((!other || (down && held)) && !gen_scancode_locked(key))
598 return false;
599 held = down;
600 return true;
601}

603bool CKeyboard::gen_scancode_locked(u32 key)
604{
605 if ((key & ~BX_KEY_RELEASED) >= BX_KEY_NBKEYS || state.current_scancodes_set >= 3)
606 return false;
607 unsigned char* scancode;
608 u8 i;
609
610#ifdef DEBUG_KBD
611 printf("gen_scancode(): %s %s \n", bx_keymap->getBXKeyName(key),
612 (key >> 31) ? "released" : "pressed");
613 if (!state.scancodes_translate)
614 BX_DEBUG(("keyboard: gen_scancode with scancode_translate cleared"));
615#endif
616
617 // Ignore scancode if keyboard clock is driven low
618 if (state.kbd_clock_enabled == 0)
619 return false;
620
621 // Ignore scancode if scanning is disabled
622 if (state.kbd_internal_buffer.scanning_enabled == 0)
623 return false;
624
625 // Source: http://www.win.tue.nl/~aeb/linux/kbd/scancodes-10.html
626 //
627 // Three scancode sets
628 //
629 // The usual PC keyboards are capable of producing three sets
630 // of scancodes. Writing 0xf0 followed by 1, 2 or 3 to port
631 // 0x60 will put the keyboard in scancode mode 1, 2 or 3.
632 // Writing 0xf0 followed by 0 queries the mode, resulting in
633 // a scancode byte 43, 41 or 3f from the keyboard.
634 //
635 // Set 1 contains the values that the XT keyboard (with only
636 // one set of scancodes) produced, with extensions for new
637 // keys. Someone decided that another numbering was more
638 // logical and invented scancode Set 2. However, it was
639 // realized that new scancodes would break old programs, so
640 // the keyboard output was fed to a 8042 microprocessor on
641 // the motherboard that could translate Set 2 back into Set
642 // 1. Indeed a smart construction. This is the default today.
643 // Finally there is the PS/2 version, Set 3, more regular,
644 // but used by almost nobody.
645 //
646 // Sets 2 and 3 are designed to be translated by the 8042.
647 // Set 1 should not be translated.
648 //
649 // Make and Break Codes
650 //
651 // The key press / key release is coded as follows:
652 //
653 // For Set 1, if the make code of a key is c, the break
654 // code will be c+0x80. If the make code is e0 c, the
655 // break code will be e0 c+0x80. The Pause key has make
656 // code e1 1d 45 e1 9d c5 and does not generate a break code.
657 //
658 // For Set 2, if the make code of a key is c, the break code
659 // will be f0 c. If the make code is e0 c, the break code
660 // will be e0 f0 c. The Pause key has the 8-byte make code
661 // e1 14 77 e1 f0 14 f0 77.
662 //
663 // For Set 3, by default most keys do not generate a break
664 // code - only CapsLock, LShift, RShift, LCtrl and LAlt do.
665 // However, by default all non-traditional keys do generate
666 // a break code - thus, LWin, RWin, Menu do, and for example
667 // on the Microsoft Internet keyboard, so do Back, Forward,
668 // Stop, Mail, Search, Favorites, Web/Home, MyComputer,
669 // Calculator, Sleep. On my BTC keyboard, also the Macro key
670 // does.
671 //
672 // In Scancode Mode 3 it is possible to enable or disable
673 // key repeat and the production of break codes either on a
674 // key-by-key basis or for all keys at once. And just like
675 // for Set 2, key release is indicated by a f0 prefix in
676 // those cases where it is indicated. There is nothing
677 // special with the Pause key in scancode mode 3.
678 if (key & BX_KEY_RELEASED)
679 scancode = (unsigned char*)scancodes[(key & 0xFF)][state.current_scancodes_set].brek;
680 else
681 scancode = (unsigned char*)scancodes[(key & 0xFF)][state.current_scancodes_set].make;
682 unsigned bytes = 0;
683 for (const unsigned char* p = scancode; *p; ++p)
684 if (!state.scancodes_translate || *p != 0xf0)
685 ++bytes;
686 if (bytes > (unsigned)(BX_KBD_ELEMENTS - state.kbd_internal_buffer.num_elements))
687 return false;
688
689 // Translation
690 //
691 // The 8042 microprocessor translates the incoming byte stream
692 // produced by the keyboard, and turns an f0 prefix into an OR
693 // with 80 for the next byte.
694 //
695 // Unless told not to translate, the keyboard controller translates
696 // keyboard scancodes into the scancodes it returns to the CPU using
697 // the following table (in hex):
698 //
699 // +----+-------------------------------------------------+
700 // | | 00 01 02 03 04 05 06 07 08 09 0a 0b 0c 0d 0e 0f |
701 // +----+-------------------------------------------------+
702 // | 00 | ff 43 41 3f 3d 3b 3c 58 64 44 42 40 3e 0f 29 59 |
703 // | 10 | 65 38 2a 70 1d 10 02 5a 66 71 2c 1f 1e 11 03 5b |
704 // | 20 | 67 2e 2d 20 12 05 04 5c 68 39 2f 21 14 13 06 5d |
705 // | 30 | 69 31 30 23 22 15 07 5e 6a 72 32 24 16 08 09 5f |
706 // | 40 | 6b 33 25 17 18 0b 0a 60 6c 34 35 26 27 19 0c 61 |
707 // | 50 | 6d 73 28 74 1a 0d 62 6e 3a 36 1c 1b 75 2b 63 76 |
708 // | 60 | 55 56 77 78 79 7a 0e 7b 7c 4f 7d 4b 47 7e 7f 6f |
709 // | 70 | 52 53 50 4c 4d 48 01 45 57 4e 51 4a 37 49 46 54 |
710 // | 80 | 80 81 82 41 54 85 86 87 88 89 8a 8b 8c 8d 8e 8f |
711 // | 90 | 90 91 92 93 94 95 96 97 98 99 9a 9b 9c 9d 9e 9f |
712 // | a0 | a0 a1 a2 a3 a4 a5 a6 a7 a8 a9 aa ab ac ad ae af |
713 // | b0 | b0 b1 b2 b3 b4 b5 b6 b7 b8 b9 ba bb bc bd be bf |
714 // | c0 | c0 c1 c2 c3 c4 c5 c6 c7 c8 c9 ca cb cc cd ce cf |
715 // | d0 | d0 d1 d2 d3 d4 d5 d6 d7 d8 d9 da db dc dd de df |
716 // | e0 | e0 e1 e2 e3 e4 e5 e6 e7 e8 e9 ea eb ec ed ee ef |
717 // | f0 | - f1 f2 f3 f4 f5 f6 f7 f8 f9 fa fb fc fd fe ff |
718 // +----+-------------------------------------------------+
719 if (state.scancodes_translate)
720 {
721
722 // Translate before send
723 u8 escaped = 0x00;
724
725 for (i = 0; i < strlen((const char*)scancode); i++)
726 {
727 if (scancode[i] == 0xF0)
728 {
729 escaped = 0x80;
730 }
731 else
732 {
733#ifdef DEBUG_KBD
734 printf("gen_scancode(): writing translated %02x \n",
735 translation8042[scancode[i]] | escaped);
736#endif
737 enQ(translation8042[scancode[i]] | escaped);
738 escaped = 0x00;
739 }
740 }
741 }
742 else
743 {
744
745 // Send raw data
746 for (i = 0; i < strlen((const char*)scancode); i++)
747 {
748#ifdef DEBUG_KBD
749 printf("gen_scancode(): writing raw %02x \n", scancode[i]);
750#endif
751 enQ(scancode[i]);
752 }
753 }
754 return true;
755}

790{
791 int tail;
792
793#ifdef DEBUG_KBD
794 printf("enQ(0x%02x)", (unsigned)scancode);
795#endif
796 if (state.kbd_internal_buffer.num_elements >= BX_KBD_ELEMENTS)
797 {
798 printf("internal keyboard buffer full, ignoring scancode.(%02x) \n",
799 (unsigned)scancode);
800 return;
801 }
802
803 /* enqueue scancode in multibyte internal keyboard buffer */
804#ifdef DEBUG_KBD
805 BX_DEBUG(("enQ: putting scancode 0x%02x in internal buffer", (unsigned)scancode));
806#endif
807 tail =
808 (
809 state.kbd_internal_buffer.head +
810 state.kbd_internal_buffer.num_elements
811 ) %
812 BX_KBD_ELEMENTS;
813 state.kbd_internal_buffer.buffer[tail] = scancode;
814 state.kbd_internal_buffer.num_elements++;
815
816 // Event-driven: deliver to the output buffer now (if free) and drive the
817 // IRQ, instead of waiting for the 20ms poll.
818 kbd_service();

940void CKeyboard::write_60(u8 value)
941{
942#ifdef DEBUG_KBD
943 printf("kbd: port 60 write: %02x. \n", value);
944#endif
945
946 // Device/controller commands take priority over asynchronous host input
947 // in the shared 8042 output path.
948 prepare_command_response();
949
950 // data byte written last to 0x60
951 state.status.c_d = 0;
952
953 // if expecting data byte from command last sent to port 64h
954 if (state.expecting_port60h)
955 {
956 state.expecting_port60h = 0;
957#ifdef DEBUG_KBD
958 if (state.status.inpb)
959 printf("write to port 60h, not ready for write \n");
960#endif
961 switch (state.last_comm)
962 {
963 case 0x60: // write command byte
964 {
965
966 //The keyboard controller is provided with some RAM, for example
967 // 32 bytes, that can be accessed by the CPU. The most important
968 // part of this RAM is byte 0, the Controller Command Byte (CCB).
969 // It can be read/written by writing 0x20/0x60 to port 0x64 and
970 // then reading/writing a data byte from/to port 0x60.
971 //
972 // This byte has the following layout.
973 //
974 // +---+-------+----+----+---+------+-----+-----+
975 // | 0 | XLATE | ME | KE | 0 | SYSF | MIE | KIE |
976 // +---+-------+----+----+---+------+-----+-----+
977 //
978 // Bit 6: Translate
979 // 0: No translation.
980 // 1: Translate keyboard scancodes, using the translation table
981 // given above. MCA type 2 controllers cannot set this bit
982 // to 1. In this case scan code conversion is set using
983 // keyboard command 0xf0 to port 0x60.
984 //
985 // Bit 5: Mouse enable
986 // 0: Enable mouse.
987 // 1: Disable mouse by driving the clock line low.
988 //
989 // Bit 4: Keyboard enable
990 // 0: Enable keyboard.
991 // 1: Disable keyboard by driving the clock line low.
992 //
993 // Bit 2: System flag
994 // This bit is shown in bit 2 of the status register. A
995 // "cold reboot" is one with this bit set to zero. A
996 // "warm reboot" is one with this bit set to one (BAT
997 // already completed). This will influence the tests and
998 // initializations done by the POST.
999 //
1000 // Bit 1: Mouse interrupt enable
1001 // 0: Do not use mouse interrupts.
1002 // 1: Send interrupt request IRQ12 when the mouse output
1003 // buffer is full.
1004 //
1005 // Bit 0: Keyboard interrupt enable
1006 // 0: Do not use keyboard interrupts.
1007 // 1: Send interrupt request IRQ1 when the keyboard output
1008 // buffer is full.
1009 //
1010 // When no interrupts are used, the CPU has to poll bits 0
1011 // (and 5) of the status register.
1012 bool scan_convert;
1013
1014 //The keyboard controller is provided with some RAM, for example
1015 bool disable_keyboard;
1016
1017 //The keyboard controller is provided with some RAM, for example
1018 bool disable_aux;
1019
1020 scan_convert = (value >> 6) & 0x01;
1021 disable_aux = (value >> 5) & 0x01;
1022 disable_keyboard = (value >> 4) & 0x01;
1023 state.status.sysf = (value >> 2) & 0x01;
1024 state.allow_irq1 = (value >> 0) & 0x01;
1025 state.allow_irq12 = (value >> 1) & 0x01;
1026 set_kbd_clock_enable(!disable_keyboard);
1027 set_aux_clock_enable(!disable_aux);
1028 if (state.allow_irq12 && state.status.auxb)
1029 state.irq12_requested = 1;
1030 else if (state.allow_irq1 && state.status.outb)
1031 state.irq1_requested = 1;
1032
1033#ifdef DEBUG_KBD
1034 BX_DEBUG((" allow_irq12 set to %u", (unsigned)state.allow_irq12));
1035 if (!scan_convert)
1036 BX_INFO(("keyboard: scan convert turned off"));
1037#endif
1038
1039 // (mch) NT needs this
1040 state.scancodes_translate = scan_convert;
1041 }
1042 break;
1043
1044 case 0xd1: // write output port
1045#ifdef DEBUG_KBD
1046 BX_DEBUG(("write output port with value %02xh", (unsigned)value));
1047#endif
1048 break;
1049
1050 case 0xd4: // Write to mouse
1051 // I don't think this enables the AUX clock
1052 //set_aux_clock_enable(1); // enable aux clock line
1053 ctrl_to_mouse(value);
1054
1055 // ??? should I reset to previous value of aux enable?
1056 break;
1057
1058 case 0xd3: // write mouse output buffer
1059 // Queue in mouse output buffer
1060 controller_enQ(value, 1);
1061 break;
1062
1063 case 0xd2:
1064
1065 // Queue in keyboard output buffer
1066 controller_enQ(value, 0);
1067 break;
1068
1069 default:
1070 printf("=== unsupported write to port 60h(lastcomm=%02x): %02x \n",
1071 (unsigned)state.last_comm, (unsigned)value);
1072 }
1073 }
1074 else
1075 {
1076
1077 // data byte written last to 0x60
1078 state.status.c_d = 0;
1079 state.expecting_port60h = 0;
1080
1081 /* pass byte to keyboard */
1082
1083 /* ??? should conditionally pass to mouse device here ??? */
1084 if (state.kbd_clock_enabled == 0)
1085 set_kbd_clock_enable(1);
1086 ctrl_to_kbd(value);
1087 }
1088
1089 // Guest I/O services bytes already produced by the emulated devices.
1090 // Host mouse movement is sampled only by the keyboard worker so a write
1091 // cannot manufacture a stream packet behind its own command response.
1092 kbd_service();
1093}

1099{
1100#ifdef DEBUG_KBD
1101 printf("kbd: port 64 write: %02x. \n", value);
1102#endif
1103
1104 static int kbd_initialized = 0;
1105 u8 command_byte;
1106
1107 // Controller commands take priority over asynchronous host input.
1108 prepare_command_response();
1109
1110 // command byte written last to 0x64
1111 state.status.c_d = 1;
1112 state.last_comm = value;
1113
1114 // most commands NOT expecting port60 write next
1115 state.expecting_port60h = 0;
1116
1117 switch (value)
1118 {
1119 case 0x20: // get keyboard command byte
1120#ifdef DEBUG_KBD
1121 BX_DEBUG(("get keyboard command byte"));
1122#endif
1123
1124 command_byte = (state.scancodes_translate << 6) |
1125 ((!state.aux_clock_enabled) << 5) | ((!state.kbd_clock_enabled) << 4) |
1126 (0 << 3) | (state.status.sysf << 2) | (state.allow_irq12 << 1) |
1127 (state.allow_irq1 << 0);
1128 // A controller response waits behind an occupied OBF; it must not be
1129 // silently lost just because device data arrived first.
1130 controller_enQ(command_byte, 0);
1131 break;
1132
1133 case 0x60: // write command byte
1134#ifdef DEBUG_KBD
1135 printf("kbd_ctrl: command 60: write command byte. \n");
1136#endif
1137
1138 // following byte written to port 60h is command byte
1139 state.expecting_port60h = 1;
1140 break;
1141
1142 case 0xa0:
1143#ifdef DEBUG_KBD
1144 printf("kbd_ctrl: command a0: BIOS name (not supported). \n");
1145#endif
1146 break;
1147
1148 case 0xa1:
1149#ifdef DEBUG_KBD
1150 printf("kbd_ctrl: command a0: BIOS version (not supported). \n");
1151#endif
1152 break;
1153
1154 case 0xa7: // disable the aux device
1155 set_aux_clock_enable(0);
1156#ifdef DEBUG_KBD
1157 printf("kbd_ctrl: command a7: aux i/f disable. \n");
1158#endif
1159 break;
1160

1162 set_aux_clock_enable(1);
1163#ifdef DEBUG_KBD
1164 printf("kbd_ctrl: command a7: aux i/f enable. \n");
1165#endif
1166 break;
1167
1168 case 0xa9: // Test Mouse Port
1169 // controller output buffer must be empty
1170#ifdef DEBUG_KBD
1171 printf("kbd_ctrl: command a9: aux i/f test. \n");
1172#endif

1174 {
1175 printf("kbd: OUTB set and command 0x%02x encountered", value);
1176 break;
1177 }
1178
1179 controller_enQ(0x00, 0); // no errors detected
1180 break;

1183#ifdef DEBUG_KBD
1184 printf("kbd_ctrl: command aa: self test. \n");
1185#endif

1187 discard_input_type(InputEvent::Type::Key);
1188 state.kbd_internal_buffer.head = 0;
1189 state.kbd_internal_buffer.num_elements = 0;
1190 state.deferred_scancode = 0;
1191 state.deferred_scancode_valid = false;
1192
1193 if (kbd_initialized == 0)
1194 {
1195 state.kbd_controller_Qsize = 0;
1196 state.kbd_controller_Qsource = 0;
1197 state.status.outb = 0;
1198 kbd_initialized = 1;
1199 }
1200
1201 // controller output buffer must be empty
1202 if (state.status.outb)
1203 {
1204 printf("kbd: OUTB set and command 0x%02x encountered", value);
1205
1206 //break;
1207 // drain the queue?
1208 state.kbd_internal_buffer.head = 0;
1209 state.kbd_internal_buffer.num_elements = 0;
1210 state.status.outb = 0;
1211 }
1212
1213 state.status.sysf = 1; // self test complete
1214 controller_enQ(0x55, 0); // controller OK
1215 break;
1216
1217 case 0xab: // Interface Test
1218#ifdef DEBUG_KBD
1219 printf("kbd_ctrl: command ab: kbd i/f test. \n");
1220#endif
1221
1222 // controller output buffer must be empty
1223 if (state.status.outb)
1224 {
1225 printf("kbd: OUTB set and command 0x%02x encountered", value);
1226 break;
1227 }
1228
1229 controller_enQ(0x00, 0);
1230 break;
1231
1232 case 0xad: // disable keyboard
1233 set_kbd_clock_enable(0);
1234#ifdef DEBUG_KBD
1235 printf("kbd_ctrl: command ad: kbd i/f disable. \n");
1236#endif
1237 break;
1238
1239 case 0xae: // enable keyboard
1240 set_kbd_clock_enable(1);
1241#ifdef DEBUG_KBD
1242 printf("kbd_ctrl: command ae: kbd i/f enable. \n");
1243#endif
1244 break;

1247#ifdef DEBUG_KBD
1248 printf("kbd_ctrl: command af: controller version (not supported). \n");
1249#endif
1250 break;
1251
1252 case 0xc0: // read input port
1253#ifdef DEBUG_KBD
1254 printf("kbd_ctrl: command c0: read input port. \n");
1255#endif
1256
1257 // controller output buffer must be empty
1258 if (state.status.outb)

1260 BX_PANIC(("kbd: OUTB set and command 0x%02x encountered", value));

1262 }

1265 controller_enQ(0x80, 0);
1266 break;

1269#ifdef DEBUG_KBD
1270 printf("kbd_ctrl: command d0: read output port. (partial) \n");

1286#ifdef DEBUG_KBD
1287 printf("kbd_ctrl: command d1: write output port. \n");

1296 printf("kbd_ctrl: command d3: write aux output buffer. \n");

1300 state.expecting_port60h = 1;
1301 break;

1304#ifdef DEBUG_KBD

1309 state.expecting_port60h = 1;

1321 case 0xe0: // Read Test Inputs

1360void CKeyboard::controller_enQ(u8 data, unsigned source)
1361{
1362
1363 // source is 0 for keyboard, 1 for mouse
1364#ifdef DEBUG_KBD
1365 BX_DEBUG(("controller_enQ(%02x) source=%02x", (unsigned)data, source));
1366#endif
1367
1368 // see if we need to Q this byte from the controller
1369 // remember this includes mouse bytes.
1370 if (state.status.outb)
1371 {
1372 if (state.kbd_controller_Qsize >= BX_KBD_CONTROLLER_QSIZE)
1373 {
1374 BX_ERROR(("controller_enq(): controller_Q full, dropping %02x "
1375 "from source %u", (unsigned)data, source));
1376 return;
1377 }
1378
1379 const unsigned queue_index = state.kbd_controller_Qsize++;
1380 state.kbd_controller_Q[queue_index] = data;
1381 if (source)
1382 state.kbd_controller_Qsource |= (1u << queue_index);
1383 else
1384 state.kbd_controller_Qsource &= ~(1u << queue_index);
1385 return;
1386 }
1387
1388 // the Q is empty
1389 if (source == 0)
1390 { // keyboard
1391 state.kbd_output_buffer = data;
1392 state.status.outb = 1;
1393 state.status.auxb = 0;
1394 state.kbd_output_is_scancode = false;
1395 state.aux_output_is_stream = false;
1396 state.status.inpb = 0;
1397 if (state.allow_irq1)
1398 state.irq1_requested = 1;
1399 }
1400 else
1401 { // mouse
1402 state.aux_output_buffer = data;
1403 state.status.outb = 1;
1404 state.status.auxb = 1;
1405 state.kbd_output_is_scancode = false;
1406 state.aux_output_is_stream = false;
1407 state.status.inpb = 0;
1408 if (state.allow_irq12)
1409 state.irq12_requested = 1;
1410 }

1418bool CKeyboard::controller_deQ()
1419{
1420 if (state.status.outb || !state.kbd_controller_Qsize)
1421 return false;
1422
1423 const u8 data = state.kbd_controller_Q[0];
1424 const bool source = (state.kbd_controller_Qsource & 1u) != 0;
1425
1426 for (unsigned i = 1; i < state.kbd_controller_Qsize; i++)
1427 state.kbd_controller_Q[i - 1] = state.kbd_controller_Q[i];
1428
1429 state.kbd_controller_Qsize--;
1430 state.kbd_controller_Qsource >>= 1;
1431 state.status.outb = 1;
1432 state.status.auxb = source;
1433 state.kbd_output_is_scancode = false;
1434 state.aux_output_is_stream = false;
1435 state.status.inpb = 0;
1436
1437 if (source)
1438 {
1439 state.aux_output_buffer = data;
1440 if (state.allow_irq12)
1441 state.irq12_requested = 1;
1442 }
1443 else
1444 {
1445 state.kbd_output_buffer = data;
1446 if (state.allow_irq1)
1447 state.irq1_requested = 1;
1448 }
1449
1450 return true;
1451}

1840 state.mouse.im_request = 2;

1848 state.mouse.im_request = 0;
1849 }
1850 else
1851 {
1852 state.mouse.im_request = 0;
1853 }
1854
1855 controller_enQ(0xFA, 1); // ack
1856 break;
1857
1858 case 0xe8: // Set Mouse Resolution

1860 {
1861 case 0: state.mouse.resolution_cpmm = 1; break;
1862 case 1: state.mouse.resolution_cpmm = 2; break;
1863 case 2: state.mouse.resolution_cpmm = 4; break;

1871#endif

1873 break;
1874
1875 default:
1876 BX_PANIC(("MOUSE: unknown last command (%02xh)",
1877 (unsigned)state.last_mouse_command));
1878 }
1879 }

1886 if (state.mouse.mode == MOUSE_MODE_WRAP)

1893
1894 // if (bx_dbg.mouse)

1920#endif

1926 break;
1927

1929 // if (bx_dbg.mouse)
1930#ifdef DEBUG_KBD
1931 BX_INFO(("[mouse] Mouse stream mode on."));
1932#endif

1934 controller_enQ(0xFA, 1); // ACK
1935 break;
1936

1941
1942 // if (bx_dbg.mouse)
1943#ifdef DEBUG_KBD
1944 BX_INFO(("[mouse] Mouse wrap mode off."));
1945#endif

1948 // ### TODO disabling reporting in stream mode

1956 // ### TODO disable interrupts if in stream mode.
1957 // if (bx_dbg.mouse)
1958#ifdef DEBUG_KBD

1964 break;

1967 // if (bx_dbg.mouse)
1968#ifdef DEBUG_KBD
1969 BX_INFO(("[mouse] Mouse remote mode on."));
1970#endif
1971
1972 // ### TODO should we flush/discard/ignore any already queued packets?
1973 state.mouse.mode = MOUSE_MODE_REMOTE;
1974 controller_enQ(0xFA, 1); // ACK
1975 break;
1976
1977 case 0xf2: // Read Device Type
1978 controller_enQ(0xFA, 1); // ACK
1979 if (state.mouse.im_mode)
1980 controller_enQ(0x03, 1); // Device ID (wheel z-mouse)

1991 break;
1992
1993 case 0xf4: // Enable (in stream mode)

1995 state.mouse.mode = MOUSE_MODE_STREAM;

2082#endif
2083 break;
2084

2186void CKeyboard::create_mouse_packet(bool force_enq)
2187{
2188 u8 b1;
2189
2190 u8 b2;
2191
2192 u8 b3;
2193
2194 u8 b4;
2195
2196 // A reporting-disabled, non-stream, or controller-inhibited mouse must
2197 // never create an unsolicited PS/2 movement packet.
2198 if (!state.mouse.enable || state.mouse.mode != MOUSE_MODE_STREAM
2199 || !state.aux_clock_enabled)
2200 return;
2201
2202 if (!force_enq)
2203 {
2204 // Do not queue unsolicited stream data behind a byte already exposed
2205 // in the controller's single output buffer.
2206 if (state.status.outb)
2207 return;
2208
2209 // Keep device command/parameter exchanges contiguous. Besides the
2210 // controller's two-write commands, this covers the interval between a
2211 // keyboard or mouse command ACK and its required parameter.
2212 if (input_transaction_active())
2213 return;
2214 }
2215
2216 if (state.mouse_internal_buffer.num_elements && !force_enq)
2217 {
