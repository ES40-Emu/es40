90CDiskRam::CDiskRam(CConfigurator* cfg, CSystem* sys, CDiskController* c,
91 int idebus, int idedev) : CDisk(cfg, sys, c, idebus, idedev)
92{
93 byte_size = myCfg->get_num_value("size", false, 512 * 1024 * 1024);
94
95 CHECK_ALLOCATION(ramdisk = calloc((size_t)byte_size, 1));
96 auto filename = myCfg->get_text_value("file");
97 model_number = nullptr;
98 if (filename)
99 {
100 FILE* handle = NULL;
101 if (read_only)
102 handle = fopen(filename, "rb");
103 else
104 handle = fopen_large(filename, "rb+");
105 if (handle)
106 {
107 fseek_large(handle, 0, SEEK_END);
108 auto file_byte_size = ftell_large(handle);
109 fseek_large(handle, 0, SEEK_SET);
110 auto original_ramdisk = ramdisk;
111 CHECK_REALLOCATION(ramdisk, realloc(ramdisk, (size_t)file_byte_size), char);
112 byte_size = file_byte_size;
113 auto r = fread(ramdisk, 1, (size_t)file_byte_size, handle);
114 if (r != (size_t)file_byte_size)
115 FAILURE_1(IO, "%s: unexpected end of file!\n", filename);
116 fclose(handle);
117
118 model_number = myCfg->get_text_value("model_number", filename);
119
120 // skip to the filename portion of the path.
121 char* p = model_number;
122#if defined(_WIN32)
123 char x = '\\';
124#elif defined(__VMS)
125 char x = ']';
126#else
127 char x = '/';
128#endif
129 while (*p)
130 {
131 if (*p == x)
132 model_number = p + 1;
133 p++;
134 }
135 }
136 }
137
138 state.byte_pos = 0;
139
140 sectors = 32;
141 heads = 8;
142
143 //calc_cylinders();
144 determine_layout();
145
146 if (!model_number)
147 model_number = myCfg->get_text_value("model_number", "ES40RAMDISK");
148
149 printf("%s: Mounted RAMDISK, %" PRId64 " %zu-byte blocks, %" PRId64 "/%ld/%ld.\n",
150 devid_string, byte_size / state.block_size, state.block_size,
151 cylinders, heads, sectors);
152}

171int CDiskRam::SaveState(FILE* f)
172{
173 if (byte_size < 0 || (u64)byte_size > SIZE_MAX || (byte_size && !ramdisk))
174 return -1;
175 const u64 size = (u64)byte_size;
176 // payload before record so restorestate can verify first
177 if (fwrite(&ramdisk_magic1, sizeof(ramdisk_magic1), 1, f) != 1 ||
178 fwrite(&size, sizeof(size), 1, f) != 1 ||
179 (size && fwrite(ramdisk, 1, (size_t)size, f) != (size_t)size) ||
180 fwrite(&ramdisk_magic2, sizeof(ramdisk_magic2), 1, f) != 1)
181 return -1;
182 return CDisk::SaveState(f);
183}

185int CDiskRam::RestoreState(FILE* f)
186{
187 u32 magic = 0;
188 u64 size = 0;
189 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != ramdisk_magic1 ||
190 fread(&size, sizeof(size), 1, f) != 1 || byte_size < 0 ||
191 size != (u64)byte_size || size > SIZE_MAX || (size && !ramdisk))
192 return -1;
193
194 // the file cannot enlarge the live allocation.
195 // if either this payload or the following disk protocol record is invalid.
196 std::unique_ptr<void, decltype(&free)> restored(size ? malloc((size_t)size) : nullptr, &free);
197 if (size && (!restored || fread(restored.get(), 1, (size_t)size, f) != (size_t)size))
198 return -1;
199 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != ramdisk_magic2 ||
200 CDisk::RestoreState(f))
201 return -1;
202 if (size)
203 memcpy(ramdisk, restored.get(), (size_t)size);
204 return 0;
205}

220size_t CDiskRam::read_bytes(void* dest, size_t bytes)
221{
222 if (is_offline())
223 return 0;
224 if (state.byte_pos >= byte_size)
225 return 0;
226
227 off_t_large available = byte_size - state.byte_pos;
228 if ((off_t_large)bytes > available)
229 bytes = (size_t)available;
230
231 memcpy(dest, &(((char*)ramdisk)[state.byte_pos]), bytes);
232 state.byte_pos += (unsigned long)bytes;
233 return bytes;
234}
