103CES1370::CES1370(CConfigurator* cfg, class CSystem* c, int pcibus, int pcidev) : CPCIDevice(cfg, c, pcibus, pcidev)
104{
105 host_audio::Format as;
106
107 as.freq = 44100;
108 as.channels = 2;
109 as.format = host_audio::SampleFormat::s16_le;
110 memset((void*)&state, 0, sizeof(state));
111 const u64 latency = cfg->get_num_value("latency_ms", true, 40);
112 m_latency_ms = (unsigned)(std::min)((std::max)(latency, (u64)10), (u64)500);
113 m_audio_backend = host_audio::backend_for_gui(cfg->get_gui_name());
114 host_audio::SetupGuard setup(m_audio_backend, {&state.audio_be_in, &state.audio_be_out},
115 {&state.adc_voice, &state.dac_voice[0], &state.dac_voice[1]});
116 if (!host_audio::initialize(m_audio_backend)) {
117 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio");
118 }
119 // Request ~10ms host periods; SDL environment overrides still apply.
120 host_audio::set_sample_frames_hint("512", m_audio_backend);
121 state.audio_be_in = host_audio::open_input_device(m_audio_backend);
122 if (!state.audio_be_in) {
123 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio input");
124 }
125 state.audio_be_out = host_audio::open_output_device(m_audio_backend);
126 if (!state.audio_be_out) {
127 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to initialize %s audio output");
128 }
129 {
130 int out_freq = 0;
131 int out_frames = 0;
132 if (!host_audio::output_device_format(state.audio_be_out, &out_freq, &out_frames))
133 HOST_AUDIO_FAILURE(m_audio_backend, "Failed to query %s audio output");
134 m_out_freq = out_freq > 0 ? out_freq : 48000;
135 m_out_period_frames = out_frames > 0 ? out_frames : 512;
136 }
137 host_audio::create_stream(state.adc_voice, nullptr, &as, m_audio_backend);
138 host_audio::create_stream(state.dac_voice[0], &as, nullptr, m_audio_backend);
139 host_audio::create_stream(state.dac_voice[1], &as, nullptr, m_audio_backend);
140
141 host_audio::set_input_callback(state.adc_voice, es1370_dac_callback_adc, this);
142 host_audio::set_output_callback(state.dac_voice[0], es1370_dac_callback_dac1, this);
143 host_audio::set_output_callback(state.dac_voice[1], es1370_dac_callback_dac2, this);
144 setup.complete();
145}

187void CES1370::reset_audio_request(size_t channel, bool enabled, bool clear)
188{
189 host_audio::Stream& voice = channel == ADC_CHANNEL
190 ? state.adc_voice : state.dac_voice[channel];
191 CES1370StreamLock stream_lock(voice);
192 m_audio_enabled[channel].store(false);
193 state.chan[channel].credit_q16 = 0;
194 state.chan[channel].pace_last = std::chrono::steady_clock::now();
195 if (clear && !host_audio::clear_stream(voice))
196 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to clear audio stream");
197 m_audio_enabled[channel].store(enabled);

200void CES1370::stop_threads()
201{
202 m_audio_stop.store(true);
203 m_audio_wait.notify_one();
204 {
205 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
206 audio_running = false;
207 for (size_t i = 0; i < NB_CHANNELS; ++i)
208 reset_audio_request(i, false, false);
209 }
210 // The worker never waits for the bus, including when our caller already
211 // owns it. Joining here therefore cannot leave it blocked behind us.
212 if (m_audio_thread)
213 {
214 m_audio_thread->join();
215 delete m_audio_thread;
216 m_audio_thread = nullptr;
217 }
218 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
219 unbind_voices();
220}

222void CES1370::start_threads()
223{
224 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
225 if (audio_running)
226 return;
227
228 for (size_t i = 0; i < NB_CHANNELS; ++i)
229 reset_audio_request(i, false, true);
230 m_audio_failed.store(false);
231 m_audio_wake.store(false);
232 try
233 {
234 es1370_update_voices(&state, state.ctl, state.sctl, true);
235 m_audio_stop.store(false);
236 audio_running = true;
237 m_audio_thread = new CThread("es1370");
238 m_audio_thread->start(*this);
239 }
240 catch (...)
241 {
242 audio_running = false;
243 m_audio_stop.store(true);
244 delete m_audio_thread;
245 m_audio_thread = nullptr;
246 for (size_t i = 0; i < NB_CHANNELS; ++i)
247 reset_audio_request(i, false, false);
248 unbind_voices();
249 throw;
250 }
251}

259// Guest-format bytes per second at the channel's programmed rate.
260uint32_t CES1370::chan_rate_bps(size_t channel) const
261{
262 uint32_t freq = 0, unused = 0;
263 es1370_chan_bits[channel].calc_freq(const_cast<ES1370State*>(&state),
264 state.ctl, &unused, &freq);
265 return freq << state.chan[channel].shift;
266}

268// Guest-format bytes one host device pull removes from the stream.
269int CES1370::dac_pull_bytes(uint32_t rate_bps) const
270{
271 return (int)(std::min)((u64)rate_bps * m_out_period_frames / m_out_freq,
272 (u64)(1 << 20));
273}

275// Lead to keep queued: configured latency, never under three host pulls.
276int CES1370::dac_lead_bytes(size_t channel, uint32_t rate_bps) const
277{
278 const u64 by_latency = (u64)rate_bps * m_latency_ms / 1000;
279 const u64 by_period = (u64)dac_pull_bytes(rate_bps) * 3;
280 return (int)(std::min)((std::max)(by_latency, by_period), (u64)(1 << 20));
281}

283// One worker tick: playback is paced at the sample rate against a fixed SDL lead.
284CES1370::ServiceResult CES1370::service_audio_channel(size_t channel,
285 std::chrono::steady_clock::time_point now)
286{
287 std::unique_lock<std::recursive_mutex> bus_lock(
288 cSystem->get_device_bus_mutex(), std::try_to_lock);
289 if (!bus_lock.owns_lock())
290 return SERVICE_BUSY;
291 if (!audio_running || m_audio_stop.load())
292 return SERVICE_IDLE;
293 const chan_bits& bits = es1370_chan_bits[channel];
294 if (!(state.ctl & bits.ctl_en) || (state.sctl & bits.sctl_pause))
295 return SERVICE_IDLE;
296
297 struct chan* d = &state.chan[channel];
298 host_audio::Stream& voice = channel == ADC_CHANNEL
299 ? state.adc_voice : state.dac_voice[channel];
300 CES1370StreamLock stream_lock(voice);
301
302 if (channel == ADC_CHANNEL)
303 {
304 const int avail = host_audio::available(voice);
305 if (avail < 0)
306 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to query audio stream");
307 if (avail <= 0)
308 return SERVICE_IDLE;
309 const int moved = es1370_run_channel(&state, channel, (std::min)(avail, 4096));
310 return moved > 0 && moved < avail ? SERVICE_MORE : SERVICE_IDLE;
311 }
312
313 const uint32_t rate = chan_rate_bps(channel);
314 if (!rate)
315 return SERVICE_IDLE;
316 const int queued = host_audio::queued(voice);
317 if (queued < 0)
318 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to query audio stream");
319 const int lead = dac_lead_bytes(channel, rate);
320 const int deficit = lead - queued;
321
322 auto elapsed = now - d->pace_last;
323 d->pace_last = now;
324 if (deficit <= 0)
325 {
326 d->credit_q16 = 0;
327 return SERVICE_IDLE;
328 }
329 const int64_t dt_ns = (std::min)((int64_t)
330 std::chrono::duration_cast<std::chrono::nanoseconds>(elapsed).count(),
331 (int64_t)50000000);
332 // 2x when more than a pull behind (start-up/stall), else 2% over-rate.
333 const int pull = dac_pull_bytes(rate);
334 const int rate_q8 = deficit > pull + pull / 4 ? 512 : 261;
335 if (dt_ns > 0)
336 d->credit_q16 += ((int64_t)rate * dt_ns * rate_q8 << 8) / 1000000000;
337
338 // Step cap of half a period (2..8ms) so a tick crosses at most one IRQ boundary.
339 const int sc_bytes = ((d->scount & 0xffff) + 1) << d->shift;
340 const int ms2 = (int)((u64)rate * 2 / 1000);
341 const int ms8 = (int)((u64)rate * 8 / 1000);
342 const int step_cap = (std::min)((std::max)(sc_bytes / 2, ms2), ms8);
343 if (d->credit_q16 > ((int64_t)step_cap << 16))
344 d->credit_q16 = (int64_t)step_cap << 16;
345
346 const int budget = (std::min)((int)(d->credit_q16 >> 16), deficit);
347 if (budget <= 0)
348 return SERVICE_IDLE;
349 const int moved = es1370_run_channel(&state, channel, budget);
350 d->credit_q16 -= (int64_t)moved << 16;
351 if (d->credit_q16 < 0)
352 d->credit_q16 = 0;
353 return SERVICE_IDLE;
354}

356void CES1370::run()
357{
358 try
359 {
360 int busy_spins = 0;
361 while (!m_audio_stop.load())
362 {
363 const auto now = std::chrono::steady_clock::now();
364 bool more = false, busy = false;
365 for (size_t i = 0; i < NB_CHANNELS; ++i)
366 {
367 const ServiceResult r = service_audio_channel(i, now);
368 more = more || r == SERVICE_MORE;
369 busy = busy || r == SERVICE_BUSY;
370 }
371 // Bus collisions are short; retry a few times before sleeping.
372 if (more || (busy && ++busy_spins < 16))
373 {
374 std::this_thread::yield();
375 continue;
376 }
377 busy_spins = 0;
378 std::unique_lock<std::mutex> wait_lock(m_audio_wait_mutex);
379 m_audio_wait.wait_for(wait_lock, std::chrono::milliseconds(1), [this] {
380 return m_audio_stop.load() || m_audio_wake.exchange(false);
381 });
382 }
383 }
384 catch (const CException& e)
385 {
386 printf("%s: Exception in audio refill thread: %s.\n",
387 devid_string, e.displayText().c_str());
388 m_audio_failed.store(true);
389 }
390 catch (const std::exception& e)
391 {
392 printf("%s: Exception in audio refill thread: %s.\n",
393 devid_string, e.what());
394 m_audio_failed.store(true);
395 }
396 catch (...)
397 {
398 printf("%s: Unknown exception in audio refill thread.\n", devid_string);
399 m_audio_failed.store(true);
400 }
401}

406int CES1370::SaveState(FILE* f)
407{
408 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
409 if (!f || audio_running)
410 return -1;
411 const int res = CPCIDevice::SaveState(f);
412 if (res)
413 return res;
414
415 ES1370SavedState saved{};
416 saved.ctl = state.ctl;
417 saved.status = state.status;
418 saved.mempage = state.mempage;
419 saved.codec = state.codec;
420 saved.sctl = state.sctl;
421 for (size_t i = 0; i < NB_CHANNELS; ++i)
422 {
423 saved.channel[i].leftover = state.chan[i].leftover;
424 saved.channel[i].scount = state.chan[i].scount;
425 saved.channel[i].frame_addr = state.chan[i].frame_addr;
426 saved.channel[i].frame_cnt = state.chan[i].frame_cnt;
427 }
428 static_assert(sizeof(ES1370SavedState) == 17 * sizeof(u32),
429 "Audio snapshot contains only fixed-width guest fields");
430 const u32 size = sizeof(saved);
431 if (fwrite(&es1370_magic1, sizeof(es1370_magic1), 1, f) != 1 ||
432 fwrite(&size, sizeof(size), 1, f) != 1 ||
433 fwrite(&saved, sizeof(saved), 1, f) != 1 ||
434 fwrite(&es1370_magic2, sizeof(es1370_magic2), 1, f) != 1)
435 return -1;
436 return 0;
437}

439int CES1370::RestoreState(FILE* f)
440{
441 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
442 if (!f || audio_running)
443 return -1;
444 const int res = CPCIDevice::RestoreState(f);
445 if (res)
446 return res;
447
448 ES1370SavedState saved{};
449 u32 magic = 0, size = 0;
450 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != es1370_magic1 ||
451 fread(&size, sizeof(size), 1, f) != 1 || size != sizeof(saved) ||
452 fread(&saved, sizeof(saved), 1, f) != 1 ||
453 fread(&magic, sizeof(magic), 1, f) != 1 || magic != es1370_magic2 ||
454 saved.mempage > 0xf)
455 return -1;
456 for (size_t i = 0; i < NB_CHANNELS; ++i)
457 if (saved.channel[i].leftover > 3)
458 return -1;
459
460 state.ctl = saved.ctl;
461 state.status = saved.status;
462 state.mempage = saved.mempage;
463 state.codec = saved.codec;
464 state.sctl = saved.sctl;
465 for (size_t i = 0; i < NB_CHANNELS; ++i)
466 {
467 const auto& b = es1370_chan_bits[i];
468 const u32 fmt = (state.sctl & b.sctl_fmt) >> b.sctl_sh_fmt;
469 state.chan[i].shift = (fmt & 1) + (fmt >> 1);
470 state.chan[i].leftover = saved.channel[i].leftover;
471 state.chan[i].scount = saved.channel[i].scount;
472 state.chan[i].frame_addr = saved.channel[i].frame_addr;
473 state.chan[i].frame_cnt = saved.channel[i].frame_cnt;
474 }
475 return 0;
476}

478void CES1370::es1370_update_status(ES1370State* s, uint32_t new_status)
479{
480 uint32_t level = new_status & (STAT_DAC1 | STAT_DAC2 | STAT_ADC);
481
482 if (level) {
483 s->status = new_status | STAT_INTR;
484 }
485 else {
486 s->status = new_status & ~STAT_INTR;
487 }
488 do_pci_interrupt(0, !!level);

491void CES1370::es1370_reset(ES1370State* s)
492{
493 size_t i;
494
495 s->ctl = 1;
496 s->status = 0x60;
497 s->mempage = 0;
498 s->codec = 0;
499 s->sctl = 0;
500
501 for (i = 0; i < NB_CHANNELS; ++i) {
502 struct chan* d = &s->chan[i];
503 d->scount = 0;
504 d->leftover = 0;
505 if (i == ADC_CHANNEL) {
506 host_audio::unbind_stream(s->adc_voice);
507 }
508 else {
509 host_audio::unbind_stream(s->dac_voice[i]);
510 }
511 }
512 do_pci_interrupt(0, 0);
513 es1370_update_voices(s, s->ctl, s->sctl);
514}

516void CES1370::es1370_maybe_lower_irq(ES1370State* s, uint32_t sctl)
517{
518 uint32_t new_status = s->status;
519
520 if (!(sctl & SCTRL_P1INTEN) && (s->sctl & SCTRL_P1INTEN)) {
521 new_status &= ~STAT_DAC1;
522 }
523
524 if (!(sctl & SCTRL_P2INTEN) && (s->sctl & SCTRL_P2INTEN)) {
525 new_status &= ~STAT_DAC2;
526 }
527
528 if (!(sctl & SCTRL_R1INTEN) && (s->sctl & SCTRL_R1INTEN)) {
529 new_status &= ~STAT_ADC;
530 }
531
532 if (new_status != s->status) {
533 es1370_update_status(s, new_status);
534 }
535}

556void CES1370::es1370_update_voices(ES1370State* s, uint32_t ctl, uint32_t sctl,
557 bool force)
558{
559 size_t i;
560 uint32_t old_freq, new_freq, old_fmt, new_fmt;
561
562 for (i = 0; i < NB_CHANNELS; ++i) {
563 struct chan* d = &s->chan[i];
564 const struct chan_bits* b = &es1370_chan_bits[i];
565
566 new_fmt = (sctl & b->sctl_fmt) >> b->sctl_sh_fmt;
567 old_fmt = (s->sctl & b->sctl_fmt) >> b->sctl_sh_fmt;
568
569 b->calc_freq(s, ctl, &old_freq, &new_freq);
570 const bool format_changed = force || old_fmt != new_fmt || old_freq != new_freq;
571 const bool changed = format_changed || ((ctl ^ s->ctl) & b->ctl_en) ||
572 ((sctl ^ s->sctl) & b->sctl_pause);
573 if (changed && (audio_running || force))
574 reset_audio_request(i, false, format_changed);
575 if (force || (old_fmt != new_fmt) || (old_freq != new_freq)) {
576 d->shift = (new_fmt & 1) + (new_fmt >> 1);
577 if (new_freq && (audio_running || force)) {
578 //struct audsettings as;
579 host_audio::Format as;
580
581 as.freq = new_freq;
582 as.channels = 1 << (new_fmt & 1);
583 as.format = (new_fmt & 2) ? host_audio::SampleFormat::s16_le : host_audio::SampleFormat::u8;
584
585 host_audio::Stream& voice = i == ADC_CHANNEL ? s->adc_voice : s->dac_voice[i];

588 }

591 if ((audio_running || force) &&
592 (force || ((ctl ^ s->ctl) & b->ctl_en) || ((sctl ^ s->sctl) & b->sctl_pause))) {

598 HOST_AUDIO_FAILURE(m_audio_backend, "Unable to resume audio capture");
599 }

602 }

607 }
608 else {
609 host_audio::unbind_stream(s->dac_voice[i]);
610 }
611 }

613 if (changed && (audio_running || force))
614 reset_audio_request(i,
615 (ctl & b->ctl_en) && !(sctl & b->sctl_pause), format_changed);
616 }

618 s->ctl = ctl;

769u32 CES1370::ReadMem_Bar(int func, int bar, u32 address, int dsize)
770{
771 if (bar != 0) return ~0U;
772 if (dsize < 32) {
773 auto val = CES1370::ReadMem_Bar(func, bar, address & ~3, 32);
774 switch (dsize)
775 {
776 case 8:
777 return (val >> ((address & 3) * 8)) & 0xff;
778 case 16:
779 if (address & 2) {
780 return (val >> 16) & 0xffff;
781 }
782 else {
783 return val & 0xffff;
784 }
785 }
786 }
787 if (dsize == 32) {
788 return es1370_read(&state, address, dsize);
789 }
790 if (dsize == 64) {
791 uint64_t low = es1370_read(&state, address, 32);
792 uint64_t high = es1370_read(&state, address + 4, 32);
793 return low | (high << 32);
794 }
795 return ~0U;
796}

798void CES1370::WriteMem_Bar(int func, int bar, u32 address, int dsize, u32 data)
799{
800 if (bar != 0) return;
801
802 if (dsize < 32) {
803 auto val = CES1370::ReadMem_Bar(func, bar, address & ~3, 32);
804 switch (dsize)
805 {
806 case 8:
807 {
808 val = (val & ~(0xff << ((address & 3) * 8))) | ((data & 0xff) << ((address & 3) * 8));
809 CES1370::WriteMem_Bar(func, bar, address & ~3, 32, val);
810 return;
811 }
812 case 16:
813 {
814 if (address & 2) {
815 val = (val & 0xffff) | (data << 16);
816 }
817 else {
818 val = (val & 0xffff0000) | data;
819 }
820 CES1370::WriteMem_Bar(func, bar, address & ~3, 32, val);
821 return;
822 }
823 }
824 }
825
826 if (dsize == 32) {
827 es1370_write(&state, address, data, dsize);
828 return;
829 }

832int CES1370::es1370_transfer_audio(ES1370State* s, struct chan* d, int loop_sel,
833 int maxb, bool* irq)
834{
835 uint8_t tmpbuf[4096];
836 const int index = d - s->chan;
837 const int sc = d->scount & 0xffff;
838 int csc_bytes = ((d->scount >> 16) + 1) << d->shift;
839 const bool nonloop = (s->sctl & loop_sel) != 0;
840 int remaining = maxb;
841 bool period_done = false;
842
843 // Loop only to wrap the frame buffer; one sample-count period (IRQ) per call.
844 while (remaining > 0 && !period_done) {
845 int cnt = d->frame_cnt >> 16;
846 const int size = d->frame_cnt & 0xffff;
847 if (size < cnt) break;
848
849 // leftover bytes in the current dword have already been consumed.
850 const int left = ((size - cnt + 1) << 2) - (int)d->leftover;
851 if (left <= 0) break;
852 const int target = MIN(remaining, MIN(left, csc_bytes));
853 uint32_t addr = d->frame_addr + (cnt << 2) + d->leftover;
854 int transferred = 0;
855 while (transferred < target) {
856 const int to_copy = MIN(target - transferred, (int)sizeof(tmpbuf));
857 int copied;
858 if (index == ADC_CHANNEL) {
859 copied = host_audio::read(s->adc_voice, tmpbuf, to_copy);
860 if (copied <= 0) break;
861 do_pci_write(addr, tmpbuf, 1, copied);
862 } else {
863 do_pci_read(addr, tmpbuf, 1, to_copy);
864 copied = host_audio::write(s->dac_voice[index], tmpbuf, to_copy) ? to_copy : 0;
865 if (!copied) break;
866 }
867 addr += copied;
868 transferred += copied;
869 }
870 remaining -= transferred;
871 csc_bytes -= transferred;
872 if (!csc_bytes) {
873 *irq = true;
874 period_done = true;
875 csc_bytes = (sc + 1) << d->shift;
876 }
877 d->scount = sc | (((csc_bytes - 1) >> d->shift) << 16);
878
879 cnt += (transferred + d->leftover) >> 2;
880 if (!nonloop) {
881 d->frame_cnt = size;
882 if (cnt <= size) d->frame_cnt |= cnt << 16;
883 }
884 d->leftover = (transferred + d->leftover) & 3;
885
886 // Stop on an SDL failure/empty capture queue.
887 if (transferred < target || nonloop) break;
888 }
889 return maxb - remaining;
890}

892int CES1370::es1370_run_channel(ES1370State* s, size_t chan, int free_or_avail)
893{
894 uint32_t new_status = s->status;
895 int max_bytes;
896 bool irq;
897 struct chan* d = &s->chan[chan];
898 const struct chan_bits* b = &es1370_chan_bits[chan];
899
900 if (!(s->ctl & b->ctl_en) || (s->sctl & b->sctl_pause)) {
901 return 0;
902 }
903
904 max_bytes = free_or_avail;
905 max_bytes &= ~((1 << d->shift) - 1);
906 if (max_bytes <= 0) {
907 return 0;
908 }
909
910 irq = s->sctl & b->sctl_inten && s->status & b->stat_int;
911
912 const int moved = es1370_transfer_audio(s, d, b->sctl_loopsel, max_bytes, &irq);
913
914 if (irq) {
915 if (s->sctl & b->sctl_inten) {
916 new_status |= b->stat_int;
917 }
918 }
919
920 if (new_status != s->status) {
921 es1370_update_status(s, new_status);
922 }
923 return moved;
924}
