41static inline void banner_border(void)
42{
43 int i;

50static inline void banner_line(const char* text, bool centered)
51{
52 int len = (int)strlen(text);
53 if (len > BANNER_INNER_WIDTH - 4)
54 len = BANNER_INNER_WIDTH - 4; // truncate rather than break the border
55
56 int lpad = centered ? (BANNER_INNER_WIDTH - len + 1) / 2 : 2;
57 int rpad = BANNER_INNER_WIDTH - len - lpad;
58 printf(" ||%*s%.*s%*s||\n", lpad, "", len, text, rpad, "");
59}
