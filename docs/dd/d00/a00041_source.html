102u64 CWSS::ReadMem(int index, u64 address, int dsize)
103{
104 service_audio(cSystem->device_access_time());
105 u64 result = 0;
106 for (int shift = 0; shift < dsize; shift += 8, ++address) {
107 u8 value = 0xff;
108 if (decodes_memory_access(index, address, 8, false))
109 value = address < 4 ? 0x04 | (state.config & 0x40) : state.codec.read((unsigned)address - 4);
110 result |= (u64)value << shift;
111 }
112 return result;
113}

115void CWSS::WriteMem(int index, u64 address, int dsize, u64 data)
116{
117 const auto now = cSystem->device_access_time();
118 service_audio(now);
119 for (int shift = 0; shift < dsize; shift += 8, ++address) {
120 if (!decodes_memory_access(index, address, 8, true))
121 continue;
122 const u8 value = (u8)(data >> shift);
123 if (address < 4) {
124 if (dma_channels[value & 3] != dma_channels[state.config & 3]) {
125 set_drq(false);
126 state.position = 0;
127 }
128 if ((value ^ state.config) & 0x38)
129 set_irq(false);
130 state.config = value;
131 } else {
132 const bool playing = state.codec.playback_enabled();
133 const bool recording = state.codec.capture_enabled();
134 const bool format_write = address == 5 && (state.codec.read(0) & 0x4f) == 0x48;
135 state.codec.write((unsigned)address - 4, value);
136 if (format_write || playing != state.codec.playback_enabled() ||
137 recording != state.codec.capture_enabled()) {
138 set_drq(false);
139 state.position = 0;
140 frame_credit = 0;
141 last_tick = now;
142 if (running.load()) update_streams();
143 }
144 }
145 set_irq(state.codec.irq_asserted());
146 }
147}

158void CWSS::update_streams()
159{
160 const unsigned rate = state.codec.sample_rate();
161 const auto format = host_format(rate);
162 if (host_rate != rate) {
163 if (!host_audio::set_format(playback, &format, nullptr))
164 HOST_AUDIO_FAILURE(backend, "WSS: unable to set playback rate");
165 if (capture && !host_audio::set_format(capture, nullptr, &format))
166 HOST_AUDIO_FAILURE(backend, "WSS: unable to set capture rate");
167 host_rate = rate;
168 }
169 if (!state.codec.capture_enabled()) {
170 close_capture();
171 } else if (!capture_requested) {
172 input = host_audio::open_input_device(backend);
173 if (input) {
174 host_audio::create_stream(capture, nullptr, &format, backend);
175 if (!capture || !host_audio::bind_stream(input, capture))
176 close_capture();
177 }
178 if (!capture)
179 printf("%s: capture input unavailable, recording silence.\n", devid_string);
180 capture_requested = true;
181 }
182}

184void CWSS::start_threads()
185{
186 std::lock_guard<std::recursive_mutex> lock(cSystem->get_device_bus_mutex());
187 if (running.load()) return;
188 if (!host_audio::bind_stream(output, playback))
189 HOST_AUDIO_FAILURE(backend, "WSS: unable to bind playback stream");
190 try {
191 update_streams();
192 frame_credit = 0;
193 last_tick = cSystem->device_access_time();
194 failed.store(false);
195 running.store(true);
196 worker.start(*this);
197 } catch (...) {
198 stop_threads();
199 throw;
200 }

237void CWSS::service_audio(std::chrono::steady_clock::time_point now)
238{
239 if (!running.load() || now <= last_tick) return;
240 const auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(now - last_tick).count();
241 last_tick = now;
242 const bool playing = state.codec.playback_enabled();
243 if (!playing && !state.codec.capture_enabled()) return;
244 // Limit catch-up after host stalls while retaining the fractional sample clock.
245 constexpr s64 second = 1000000000;
246 frame_credit += (std::min<s64>)(ns, second / 50) * state.codec.sample_rate();
247 const int frames = int((std::min<s64>)(frame_credit / second, batch_frames));
248 if (!frames) return;
249 frame_credit %= second;
250 set_drq(true);
251
252 u8 pcm[batch_frames * 4];
253 int received = 0;
254 if (!playing && capture) {
255 received = host_audio::read(capture, pcm, frames * 4);
256 if (received < 0)
257 HOST_AUDIO_FAILURE(backend, "WSS: unable to read capture stream");
258 }
259 const int channel = dma_channels[state.config & 3];
260 const unsigned frame_bytes = state.codec.frame_bytes();
261 int complete = 0;
262 while (complete < frames) {
263 u8* sample = &pcm[complete * 4];
264 if (!playing && !state.position) {
265 s16 left = 0, right = 0;
266 if (complete * 4 + 4 <= received) {
267 left = (s16)(sample[0] | (sample[1] << 8));
268 right = (s16)(sample[2] | (sample[3] << 8));
269 }
270 state.codec.encode_frame(left, right, state.frame);
271 }
272 while (state.position < frame_bytes) {
273 u16 unit = state.frame[state.position];
274 const auto result = playing ? dma->service_recv_unit(channel, unit)
275 : dma->service_send_unit(channel, unit);
276 if (!result.transferred) break;
277 state.frame[state.position++] = (u8)unit;
278 }
279 if (state.position < frame_bytes) break;
280 state.position = 0;
281 if (playing) {
282 s16 left, right;
283 state.codec.decode_frame(state.frame, left, right);
284 sample[0] = (u8)left;
285 sample[1] = (u8)((u16)left >> 8);
286 sample[2] = (u8)right;
287 sample[3] = (u8)((u16)right >> 8);
288 }
289 ++complete;
290 state.codec.advance_frame();
291 set_irq(state.codec.irq_asserted());
292 if (!state.codec.playback_enabled() && !state.codec.capture_enabled()) break;
293 }
294 const bool held = !state.codec.playback_enabled() && !state.codec.capture_enabled();
295 if (complete == frames || held)
296 set_drq(false);
297 if (!playing && held) close_capture();
298 if (playing && complete) {
299 const int queued = host_audio::queued(playback);
300 // Keep at most a quarter-second of stereo 16-bit samples ahead of the host.
301 if (queued < 0 || (queued < (int)host_rate && !host_audio::write(playback, pcm, complete * 4)))
302 HOST_AUDIO_FAILURE(backend, "WSS: unable to queue playback");
303 }
304}

321 decltype(state) saved{};

323 saved.position >= saved.codec.frame_bytes() || saved.drq > 1 || saved.irq > 1)
