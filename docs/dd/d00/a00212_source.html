35// Model-sized verifier snapshots; zero words require no callbacks.
36struct AlphaJitVerifyPolicy
37{
38 int word_count = 0;
39 void (*capture)(const CAlphaCPU& cpu, std::uint64_t* words) = nullptr;
40 // Restore replay inputs from the pre-interpreter snapshot.
41 void (*prepare_replay)(CAlphaCPU& cpu, const std::uint64_t* words) = nullptr;
42 // Restore the authoritative post-interpreter snapshot.
43 void (*restore)(CAlphaCPU& cpu, const std::uint64_t* words) = nullptr;
44 bool (*differs)(int slot, std::uint64_t left, std::uint64_t right, bool side_exit) = nullptr;
45
46 constexpr bool valid() const
47 {
48 return word_count >= 0 && (!word_count || (capture && prepare_replay && restore && differs));
49 }
50};
