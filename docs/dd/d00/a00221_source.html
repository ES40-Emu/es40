263void CAlphaCPU::vmspal_call_swpctx()
264{
265 p23 = state.pc;
266
267 if (r16 & 0x7f)
268 {
269
270 //context pointer not properly aligned...
271 p4 = 0x430;
272 hw_stq(p21 + 0x150, p23);
273 hw_stq(p21 + 0x158, p4);
274 vmspal_int_initiate_exception();
275 return;
276 }
277
278 hw_ldq(r16 + 0x30, p4);
279 hw_ldq(r16 + 0x38, p5);
280 hw_ldq(r16 + 0x28, p6);
281
282 p20 = state.aster | (state.astrr << 4);
283
284 p6 &= 0xff;
285 state.asn0 = (int)p6;
286 state.asn1 = (int)p6;
287 state.asn = (int)p6;
288 flush_data_page_cache();
289 jit_note_asn_change();
290 state.aster = (int)p4 & 0xf;
291 state.astrr = (int)(p4 >> 4) & 0xf;
292 state.fpen = (int)p5 & 1;
293 state.ppcen = (int)(p5 >> 0x3e) & 1;
294 kick_int_if_pending();
295
296 hw_ldq(r16 + 0x40, p7);
297 hw_ldq(r16 + 0x20, p6);
298
299 p4 = (state.cc & U64(0xffffffff)) + state.cc_offset;
300 state.cc_offset = ((u32)p7 & 0xffffffff) - (state.cc & U64(0xffffffff));
301
302 p6 <<= 0x0d;
303 hw_stq(p21 + 8, p6);
304 hw_ldq(p21 + 0x10, p5);
305 hw_ldq(r16, p6);
306 hw_stl(p5 + 0x40, p4);
307 hw_stq(p5 + 0x30, p20);
308 hw_stq(p5, r30);
309
310 r30 = p6;
311 hw_stq(p21 + 0x10, r16);
312
313 // Revalidate instruction fetch in the new context without discarding ASN-tagged cache lines.
314 set_pc(state.pc);
315}

703{
704 u64 pa;
705
706 p23 = state.pc;
707 p4 = r18 & 3; // alt mode
708 p5 = p22 & 0x18;

765int CAlphaCPU::vmspal_call_rei()
766{
767 u64 phys_address;
768
769 p23 = state.pc;
770
771 // Reading the frame can invoke a TB-miss handler that overwrites p7.
772 const u64 old_cm = p22 & 0x18;
773 hw_stq(p21 + 0x150, p23);
774 if (r30 & 0x3f)
775 {
776
777 // stack not aligned
778 p4 = 0x430;
779 hw_stq(p21 + 0x158, p4);
780 return vmspal_int_initiate_exception();
781 }
782
783 if (old_cm)
784 {
785
786 // what to do if the next instruction results in a TNV exception?
787 ldq(r30 + 0x38, p20);
788 // ARM 10.1.11: REI clears both intr_flag and lock_flag.
789 state.bIntrFlag = false;
790 cSystem->cpu_clear_lock(state.iProcNum);
791 ldq(r30 + 0x30, p23);
792 p4 = p20 & 0x18; // new cm
793 if ((p4 < old_cm) || (p20 & ~U64(0x3f0000000000001b)))
794 {
795 p4 = 0x430;
796 hw_stq(p21 + 0x158, p4);
797 vmspal_int_initiate_exception();
798 return 0;
799 }
800
801 ldq(r30 + 0x10, state.r[4]);
802 ldq(r30 + 0x18, state.r[5]);
803 ldq(r30 + 0x20, state.r[6]);
804 ldq(r30 + 0x28, state.r[7]);
805 ldq(r30, r2);
806 ldq(r30 + 8, r3);
807 hw_ldq(p21 + 0x10, p6);
808 p5 = (p20 >> 56) & 0xff;
809 p7 = old_cm + p6;
810 p6 += p4;
811 p20 &= 0xffff;
812 p22 &= ~U64(0xffff);
813 state.cm = (int)(p4 >> 3) & 3;
814 p22 |= p20;
815 p23 &= ~U64(0x3);
816 p20 = r30 + 0x40;
817 p20 |= p5;
818 hw_stq(p7, p20);
819 hw_ldq(p6, r30);
820 kick_int_if_pending();
821 set_pc(p23);
822 return 0;
823 }
824
825 // A kernel-stack fault must report the REI itself, not its next PC.
826 hw_stq(p21 + 0x150, state.current_pc);
827 kldq(r30 + 0x38, p20);
828 state.bIntrFlag = false;
829 cSystem->cpu_clear_lock(state.iProcNum);
830 p7 = (p20 >> 8) & 0xff;
831 kldq(r30 + 0x10, state.r[4]);
832 kldq(r30 + 0x18, state.r[5]);
833 kldq(r30 + 0x20, state.r[6]);
834 kldq(r30 + 0x28, state.r[7]);
835 kldq(r30, r2);
836 kldq(r30 + 8, r3);
837 p4 = p20 & 0x18; // new cm
838 kldq(r30 + 0x30, p23);
839
840 if (p4)
841 {
842 hw_ldq(p21 + 0x10, p7);
843 p7 += p4;
844 state.cm = (int)(p4 >> 3) & 3;
845 p5 = (p20 >> 56) & 0xff;
846 p20 &= 0xff;
847 p22 &= ~U64(0xffff);
848 p22 |= p20;
849 p23 &= ~U64(0x3);
850 p20 = r30 + 0x40;
851 p20 |= p5;
852 hw_ldq(p7, r30);
853 hw_stq(p21 + 0x18, p20);
854 state.eien = ipl_ier_mask[0][0];
855 state.slen = ipl_ier_mask[0][1];
856 state.cren = ipl_ier_mask[0][2];
857 state.pcen = ipl_ier_mask[0][3];
858 state.sien = ipl_ier_mask[0][4];
859 state.asten = ipl_ier_mask[0][5];
860 kick_int_if_pending();
861 set_pc(p23);
862 return 0;
863 }
864
865 p5 = (p20 >> 56) & 0xff;
866 p20 &= 0xffff;
867 p22 &= ~U64(0xffff);
868 p7 = (p20 >> 8) & 0xff;
869 p22 |= p20;
870 p23 &= ~U64(0x3);
871 p20 = r30 + 0x40;
872 r30 = p20 | p5;
873 state.eien = ipl_ier_mask[p7][0];
874 state.slen = ipl_ier_mask[p7][1];
875 state.cren = ipl_ier_mask[p7][2];
876 state.pcen = ipl_ier_mask[p7][3];
877 state.sien = ipl_ier_mask[p7][4];
