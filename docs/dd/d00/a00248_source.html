69struct AlphaIEEETrapPolicy
70{
71 // Returns FPCR state to apply before reporting an arithmetic trap.
72 AlphaIEEETrapResult (*evaluate)(std::uint64_t fpcr, std::uint64_t trap,
73 std::uint32_t instruction_enable, AlphaIEEETrapMasking masking,
74 std::uint32_t instruction);
75
76 // Pure rule for effect-free INE skips (enable=0); keep paired with evaluate.
77 // Missing rules defer; exact paths need no rule.
78 AlphaIEEEInexactRule (*describe_inexact)(std::uint32_t instruction) = nullptr;
79
80 // Pure underflow result after UNF/INE reporting; null preserves signed zero.
81 std::uint64_t (*underflow_result)(bool negative) = nullptr;
82
83 // Pure rule from live FPCR; null leaves denormal inputs unflushed.
84 bool (*denormal_input_zero)(std::uint64_t fpcr) = nullptr;
85
86 // Input classes requesting INV; delivery remains with evaluate.
87 std::uint32_t invalid_nonfinite_inputs =
88 static_cast<std::uint32_t>(AlphaIEEENonfiniteInput::SignalingNaN);
89
90 bool divide_rounding_inexact = true; // DIV rounding reports INE.
91 bool divide_inexact_completion = false; // /I DIV reports INE after valid inputs.
92 bool stop_on_invalid_input = false; // Stop after an input requests INV.
93};

95constexpr bool alpha_ieee_nonfinite_input_invalid(
96 const AlphaIEEETrapPolicy& policy, AlphaIEEENonfiniteInput input)
97{
98 return (policy.invalid_nonfinite_inputs & static_cast<std::uint32_t>(input)) != 0;
99}
100
101inline AlphaIEEEInexactRule alpha_ieee_inexact_rule(
102 const AlphaIEEETrapPolicy& policy, std::uint32_t instruction)
103{
104 if (!policy.describe_inexact) return {};
105 const auto rule = policy.describe_inexact(instruction);
106 if (rule.kind == AlphaIEEEInexactKind::Noop)
107 return {AlphaIEEEInexactKind::Noop, 0};
108 if (rule.kind == AlphaIEEEInexactKind::FpcrBitSet && rule.fpcr_bit < 64)
109 return rule;
110 return {};
111}
112
113inline AlphaIEEEInexactRule alpha_ieee_divide_inexact_rule(
114 const AlphaIEEETrapPolicy& policy, std::uint32_t instruction)
115{
116 return policy.divide_rounding_inexact ? alpha_ieee_inexact_rule(policy, instruction)
117 : AlphaIEEEInexactRule{AlphaIEEEInexactKind::Noop, 0};
118}
