103// Family decoding feeds one shared load executor. The caller handles faults
104// and unknown forms in the interpreter instruction-exit context.
105#define DO_HW_LD do { \
106 if (!alpha_hw_load_valid(hw_load)) { UNKNOWN2; } \
107 const u64 hw_address = state.r[REG_2] + (u64)hw_load.displacement; \
108 if (hw_load.address == AlphaHWLoadAddress::Virtual) { \
109 const int hw_flags = ACCESS_READ | (hw_load.vpte ? VPTE : 0) \
110 | (hw_load.alternate ? ALT : 0) | (hw_load.write_check ? WRCHK : 0); \
111 if (hw_flags == ACCESS_READ) { \
112 DATA_PHYS_NT(hw_address, ACCESS_READ); \
113 } else if (virt2phys(hw_address, &phys_address, hw_flags, nullptr, ins)) { \
114 ES40_EXECUTE_END(); \
115 } \
116 } else { \
117 phys_address = hw_address; \
118 } \
119 /* Translate before alignment; reserve the address actually read. */ \
120 if (hw_load.locked) { \
121 state.r[REG_1] = cSystem->cpu_read_locked(state.iProcNum, \
122 ALIGN_PHYS(hw_load.size_bits / 8), hw_load.size_bits, this); \
123 LLR; \
124 } else { \
125 state.r[REG_1] = READ_PHYS_NT(hw_load.size_bits); \
126 } \
127 if (hw_load.size_bits == 32) \
128 state.r[REG_1] = sext_u64_32(state.r[REG_1]); \
129} while (0)

134// Family decoding feeds the shared store executor. Faults exit in the caller.
135#define DO_HW_ST do { \
136 if (!alpha_hw_store_valid(hw_store)) { UNKNOWN2; } \
137 const u64 hw_address = state.r[REG_2] + (u64)hw_store.displacement; \
138 if (hw_store.address == AlphaHWStoreAddress::Virtual) { \
139 if (hw_store.alternate) { \
140 DATA_PHYS_NT(hw_address, ACCESS_WRITE | ALT); \
141 } else { \
142 DATA_PHYS_NT(hw_address, ACCESS_WRITE); \
143 } \
144 } else { \
145 phys_address = hw_address; \
146 } \
147 /* Evaluate the source after translation, before any conditional result. */ \
148 if (hw_store.conditional) { \
149 state.r[REG_1] = cSystem->cpu_stx_c(state.iProcNum, \
150 ALIGN_PHYS(hw_store.size_bits / 8), \
151 hw_store.size_bits, state.r[REG_1], dram_ptr, dram_size, this); \
152 } else { \
153 WRITE_PHYS_NT(state.r[REG_1], hw_store.size_bits); \
154 } \
155} while (0)
