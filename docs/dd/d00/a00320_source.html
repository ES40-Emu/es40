197inline u64 real_address(u64 address, CAlphaCPU* c, bool bIBOX)
198{
199 bool b;
200 u64 a;
201
202 if (bIBOX && (address & 1))
203 return address & U64(0xfffffffffffffffc);
204
205 if (!(c->virt2phys(address, &a, ACCESS_READ | NO_CHECK | FAKE, &b, 0)))
206 return a & (bIBOX ? U64(0xfffffffffffffffc) : U64(0xffffffffffffffff));
207
208 return
209 ((address & U64(0xfffffffff0000000)) == U64(0x0000000020000000)) ? address -
210 U64(0x000000001fe00000) :
211 (
212 ((address & U64(0xfffffffff0000000)) == U64(0x0000000010000000)) ? address -
213 U64(0x000000000fffe000) : address
214 ) & (bIBOX ? U64(0xfffffffffffffffc) : U64(0xffffffffffffffff));
215}

256void CTraceEngine::trace(CAlphaCPU* cpu, u64 f, u64 t, bool down, bool up,
257 const char* x, int y)
258{
259 // Not a real jump.
260 if ((t == f + 4) || (t == f))
261 return;
262
263 // p = new process context
264 int p = get_prbr(cpu->get_prbr(), cpu->get_hwpcb());
265
266 // o = old process context
267 int o = asCPUs[cpu->get_cpuid()].last_prbr;
268
269 // Check for process context switch
270 if (p != o)
271 {
272 if (o != -1)
273 {
274 /* Process context switch has happened. Print this in the trace files for
275 both the old and the new process context. */
276 fprintf(asPRBRs[o].f, "\n==> Switch to PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n",
277 asPRBRs[p].prbr, asPRBRs[p].hwpcb, asPRBRs[p].procname);
278 fprintf(asPRBRs[p].f, " This is PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n",
279 asPRBRs[p].prbr, asPRBRs[p].hwpcb, asPRBRs[p].procname);
280 fprintf(asPRBRs[p].f, "<== Switch from PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n\n",
281 asPRBRs[o].prbr, asPRBRs[o].hwpcb, asPRBRs[o].procname);
282 }
283
284 // save process context in cpu record
285 asCPUs[cpu->get_cpuid()].last_prbr = p;
286 }
287
288 // Has tracing been temporarily disabled for this process?
289 if (asPRBRs[p].trc_waitfor)
290 {
291 // Have we reached the point where tracing should be re-enabled?
292 if ((t & ~U64(0x3)) == asPRBRs[p].trc_waitfor)
293 asPRBRs[p].trc_waitfor = 0;
294
295 // Don't trace for now.
296 return;
297 }
298
299 // Get the physical to/from addresses.
300 u64 pc_f = real_address(f, cpu, true);
301 u64 pc_t = real_address(t, cpu, true);
302
303 /* This is where it gets tricky...
304 *
305 * If the up parameter is true, we check if the to address of this trace is equal
306 * to the from address of a previous trace (that was a function call), or equal to
307 * that from address + 4 (next instruction). If this is the case, we consider this
308 * a function return. (decreasing the trace level).
309 *
310 * If this is not the case, we then check if tracing is hidden at this level. If
311 * this is the case, we don't trace this call.
312 *
313 * Next, we check if the down parameter is false, if this is the case, we handle
314 * the trace through trace_br.
315 *
316 * Otherwise, we treat the trace as a function call (increasing the trace level).
317 * We will then check if the to address is a known function address. In either
318 * case we will print the address or the function name and either the formatted
319 * function list or the first few function argument registers.
320 */
321 int oldlvl = asPRBRs[p].trclvl;
322 int i;
323 int j;
324
325 if (up)
326 {
327 // Check if this is a function return
328 for (i = oldlvl - 1; i >= 0; i--)
329 {
330 // Is this a return to an old "from" address?
331 if ((asPRBRs[p].trcadd[i] == (pc_t - 4)) || (asPRBRs[p].trcadd[i] == (pc_t)))
332 {
333 // Yes, return to this address' tracelevel
334 asPRBRs[p].trclvl = i;
335 // Should we stop hiding the trace again?
336 if (asPRBRs[p].trchide > i)
337 asPRBRs[p].trchide = -1;
338
339 // Is tracing hidden at this level?
340 if (asPRBRs[p].trchide != -1)
341 return;
342
343 // Indent to the pevious (higher) trace level
344 for (j = 0; j < oldlvl; j++)
345 fprintf(asPRBRs[p].f, " ");
346
347 // And print the from address, and the value of the r0 register (return value)
348 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ") ($r0 = %" PRIx64 ")\n", f,
349 pc_f, cpu->get_r(0, true));
350
351 // Indent to the new (lower) trace level
352 for (j = 0; j < asPRBRs[p].trclvl; j++)
353 fprintf(asPRBRs[p].f, " ");
354
355 // Print the to address
356 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ") <--\n", t, pc_t);
357 return;
358 }
359 }
360 }
361
362 // Are traces hidden at this level?
363 if (asPRBRs[p].trchide != -1)
364 return;
365
366 if (!down)
367 {
368 // If we're not allowed to consider this a function call, handle the trace through trace_br
369 trace_br(cpu, f, t);
370 return;
371 }
372
373 // Maximum trace level reached?
374 if (oldlvl < 700)
375 asPRBRs[p].trclvl = oldlvl + 1;
376
377 // Set the from address for the old (lower) trace level
378 asPRBRs[p].trcadd[oldlvl] = pc_f;
379
380 // Is there a special message to print?
381 if (x)
382 {
383 // Indent to the old (lower) trace level
384 for (i = 0; i < oldlvl; i++)
385 fprintf(asPRBRs[p].f, " ");
386
387 // Print the special message
388 fprintf(asPRBRs[p].f, x, y);
389 fprintf(asPRBRs[p].f, "\n");
390 }
391
392 // Indent to the old (lower) trace level
393 for (i = 0; i < oldlvl; i++)
394 fprintf(asPRBRs[p].f, " ");
395
396 // Print the from address
397 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ") -->\n", f, pc_f);
398
399 // Indent to the new (higher) trace level
400 for (i = 0; i < asPRBRs[p].trclvl; i++)
401 fprintf(asPRBRs[p].f, " ");
402
403 // Check if this is a known function
404 for (i = 0; i < iNumFunctions; i++)
405 {
406 if (asFunctions[i].address == pc_t)
407 {
408 // Function found
409
410 // Print function name
411 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%s)", t, asFunctions[i].fn_name);
412
413 // And print the argument list
414 write_arglist(cpu, asPRBRs[p].f, asFunctions[i].fn_arglist);
415 fprintf(asPRBRs[p].f, "\n");
416
417 // If this is a "step-over" function, we hide tracing for higher levels.
418 if (asFunctions[i].step_over)
419 asPRBRs[p].trchide = asPRBRs[p].trclvl;
420 return;
421 }
422 }
423
424 // No known function
425
426 // Print the to address
427 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ")", t, pc_t);
428
429 // Print a default argument list
430 write_arglist(cpu, asPRBRs[p].f, "(%s|16%, %s|17%, %s|18%, %s|19%)");
431 fprintf(asPRBRs[p].f, "\n");
432}

443void CTraceEngine::trace_br(CAlphaCPU* cpu, u64 f, u64 t)
444{
445 int p;
446 int o;
447
448 // Get the physical to/from addresses.
449 u64 pc_f = real_address(f, cpu, true);
450 u64 pc_t = real_address(t, cpu, true);
451
452 // p = new process context
453 p = get_prbr(cpu->get_prbr(), cpu->get_hwpcb());
454
455 // o = old process context
456 o = asCPUs[cpu->get_cpuid()].last_prbr;
457
458 // Has tracing been temporarily disabled for this process?
459 if (asPRBRs[p].trc_waitfor)
460 {
461 // Have we reached the point where tracing should be re-enabled?
462 if ((t & ~U64(0x3)) == asPRBRs[p].trc_waitfor)
463 asPRBRs[p].trc_waitfor = 0;
464
465 // Don't trace for now
466 return;
467 }
468
469 // Is tracing hidden at this level?
470 if (asPRBRs[p].trchide != -1)
471 return;
472
473 // Check for process context switch
474 if (p != o)
475 {
476 if (o != -1)
477 {
478 /* Process context switch has happened. Print this in the trace files for
479 both the old and the new process context. */
480 fprintf(asPRBRs[o].f, "\n==> Switch to PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n",
481 asPRBRs[p].prbr, asPRBRs[p].hwpcb, asPRBRs[p].procname);
482 fprintf(asPRBRs[p].f, " This is PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n",
483 asPRBRs[p].prbr, asPRBRs[p].hwpcb, asPRBRs[p].procname);
484 fprintf(asPRBRs[p].f, "<== Switch from PRBR %08" PRIx64 " %08" PRIx64 " (%s)\n\n",
485 asPRBRs[o].prbr, asPRBRs[o].hwpcb, asPRBRs[o].procname);
486 }
487
488 // save process context in cpu record
489 asCPUs[cpu->get_cpuid()].last_prbr = p;
490 }
491
492 int i;
493
494 // Is this a real jump?
495 if ((pc_t > pc_f + 4) || (pc_t < pc_f))
496 {
497 // Indent to the trace level
498 for (i = 0; i < asPRBRs[p].trclvl; i++)
499 fprintf(asPRBRs[p].f, " ");
500
501 // Print from address
502 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ") --+\n", f, pc_f);
503
504 // Indent to the trace level
505 for (i = 0; i < asPRBRs[p].trclvl; i++)
506 fprintf(asPRBRs[p].f, " ");
507
508 // Is this a known function?
509 for (i = 0; i < iNumFunctions; i++)
510 {
511 if (asFunctions[i].address == pc_t)
512 {
513 // Yes, this is a known function
514
515 // Print the function name
516 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%s)", t, asFunctions[i].fn_name);
517
518 // Print the argument list
519 write_arglist(cpu, asPRBRs[p].f, asFunctions[i].fn_arglist);
520 fprintf(asPRBRs[p].f, " <-+\n");
521
522 // If this is a "step-over" function, we hide tracing for higher levels.
523 if (asFunctions[i].step_over)
524 asPRBRs[p].trchide = asPRBRs[p].trclvl;
525 return;
526 }
527 }
528
529 // Print the to address. (No function call, so, no argument list)
530 fprintf(asPRBRs[p].f, "%016" PRIx64 "(%08" PRIx64 ") <-+\n", t, pc_t);
531 }

584int CTraceEngine::get_prbr(u64 prbr, u64 hwpcb)
585{
586 int i;
587 char filename[100];
588
589 for (i = 0; i < iNumPRBRs; i++)
590 {
591 if ((asPRBRs[i].prbr == prbr) && (asPRBRs[i].hwpcb == hwpcb))
592 {
593 if (asPRBRs[i].f == current_trace_file)
594 return i;
595 if (!strncmp(asPRBRs[i].procname, cSystem->PtrToMem(prbr + 0x154), 20))
596 {
597 current_trace_file = asPRBRs[i].f;
598 return i;
599 }
600
601 fclose(asPRBRs[i].f);
602 break;
603 }
604 }
605
606 if (i == iNumPRBRs)
607 {
608 asPRBRs[i].generation = 0;
609 iNumPRBRs++;
610 }
611
612 asPRBRs[i].generation++;
613
614 asPRBRs[i].prbr = prbr;
615 asPRBRs[i].hwpcb = hwpcb;
616 if (prbr > 0 && prbr < (U64(0x1) << cSystem->get_memory_bits()))
617 strncpy(asPRBRs[i].procname, cSystem->PtrToMem(prbr + 0x154), 20);
618 else
619 strcpy(asPRBRs[i].procname, "");
620 sprintf(filename, "trace_%08" PRIx64 "_%08" PRIx64 "_%02d_%s.trc", prbr, hwpcb,
621 asPRBRs[i].generation, asPRBRs[i].procname);
622 asPRBRs[i].f = fopen(filename, "w");
623 if (asPRBRs[i].f == 0)
624 printf("Failed to open file!!\n");
625 asPRBRs[i].trclvl = 0;
626 asPRBRs[i].trchide = -1;
627 asPRBRs[i].trc_waitfor = 0;
628 current_trace_file = asPRBRs[i].f;
629 printf("Add PRBR: %08" PRIx64 "_%08" PRIx64 "\n", prbr, hwpcb);
630 return i;
631}

637void CTraceEngine::write_arglist(CAlphaCPU* c, FILE* fl, const char* a)
638{
639 char o[500];
640 char* op = o;
641 const char* ap = a;
642 char f[20];
643 char* fp;
644 char* rp;
645 int r;
646 u64 value;
647
648 while (*ap)
649 {
650 if (*ap == '%')
651 {
652 ap++;
653 fp = f;
654 *(fp++) = '%';
655 while (*ap != '%')
656 *(fp++) = *(ap++);
657 ap++;
658 *fp = '\0';
659
660 // now we have a formatter in f.
661 rp = strchr(f, '|');
662 *(rp++) = '\0';
663
664 // and the register in rp;
665 r = atoi(rp);
666 value = c->get_r(r, true);
667 if (!strcmp(f, "%s"))
668 {
669 sprintf(op, "%" PRIx64 " (", value);
670 while (*op)
671 op++;
672 value = real_address(value, c, false);
673 if ((value > 0) && (value < (U64(0x1) << cSystem->get_memory_bits())))
674 write_printable_s(op, cSystem->PtrToMem(value));
675 else
676 sprintf(op, "INVPTR");
677 while (*op)
678 op++;
679 *(op++) = ')';
680 *(op) = '\0';
681 }
682 else if (!strcmp(f, "%c"))
683 sprintf(op, "%02" PRIx64 " (%c)", value, printable((char)value));
684 else if (!strcmp(f, "%d"))
685 sprintf(op, "%" PRId64, value);
686 else if (!strcmp(f, "%x"))
687 sprintf(op, "%" PRIx64, value);
688 else if (!strcmp(f, "%0x"))
689 sprintf(op, "%016" PRIx64, value);
690 else if (!strcmp(f, "%016x"))
691 sprintf(op, "%016" PRIx64, value);
692 else if (!strcmp(f, "%08x"))
693 sprintf(op, "%08" PRIx64, value);
694 else if (!strcmp(f, "%04x"))
695 sprintf(op, "%04" PRIx64, value);
696 else if (!strcmp(f, "%02x"))
697 sprintf(op, "%02" PRIx64, value);
698 else
699 sprintf(op, f, value);
700 while (*op)
701 op++;
702 }
703 else
704 {
705 *(op++) = *(ap++);
706 }
707 }
708
709 *op = '\0';
710
711 fprintf(fl, "%s", o);
712}

718void CTraceEngine::read_procfile(const char* filename)
719{
720 FILE* f;
721 u64 address;
722 char linebuffer[1000];
723 char* fn_name;
724 char* fn_args;
725 char* sov;
726 int step_over;
727 int result;
728 int i = 0;
729
730 f = fopen(filename, "r");
731
732 if (f)
733 {
734 while (fscanf(f, "%[^\n] ", linebuffer) != EOF)
735 {
736 address = U64(0x0);
737 fn_name = strchr(linebuffer, ';');
738 if (fn_name)
739 {
740 *fn_name = '\0';
741 fn_name++;
742 result = sscanf(linebuffer, "%" SCNx64, &address);
743 if ((result == 1) && address)
744 {
745 fn_args = strchr(fn_name, ';');
746 if (fn_args)
747 {
748 *fn_args = '\0';
749 fn_args++;
750 sov = strchr(fn_args, ';');
751 if (sov)
752 {
753 *sov = '\0';
754 sov++;
755 result = sscanf(sov, "%d", &step_over);
756 if (result == 1)
757 {
758 add_function(address, fn_name, fn_args, step_over ? true : false);
759 i++;
760 }
761 }
762 }
763 }
764 }
765 }
766
767 fclose(f);
768 printf("%%IDB-I-RDTRC : Read %d entries from trace-file %s\n", i, filename);
769 }

791void CTraceEngine::run_script(const char* filename)
792{
793 char s[100][100];
794 int i;
795
796#if !defined(LS_SLAVE)
797 char c = '\0';
798 int j;
799 FILE* f = NULL;
800

802 {
803 f = fopen(filename, "r");

807 return;
808 }
809 }
810 else
811 {
812 printf("This is the ES40 interactive debugger. To start non-interactively, run es40,\n");
813 printf("Or run this executable (es40_idb) with a last argument of @<script-file>\n");
814 f = stdin;
815 }
816#endif
817 for (;;)
818 {
819#if !defined(LS_SLAVE)
820 if (filename)
821 {
822 if (feof(f))
823 break;
824 }
825 else
826 {
827 printf("IDB %016" PRIx64 " %c>", theSystem->get_cpu(0)->get_clean_pc(),
828 (theSystem->get_cpu(0)->get_pc() & U64(0x1)) ? 'P' : '-');
829 }
830#endif
831 for (i = 0; i < 100;)
832 {
833#if defined(LS_SLAVE)
834 lockstep_receive(s[i], 100);
835 if (!strcmp(s[i], "TERM"))
836 break;
837#else
838 bool u = false;
839 for (j = 0; j < 100;)
840 {
841 fscanf(f, "%c", &c);
842 if (c != '\n' && c != '\r' && c != ' ' && c != '\t')
843 {
844 s[i][j++] = c;
845 u = true;
846 }
847
848 if (c == ' ' || c == '\t' || c == '\n')
849 break;
850 }
851
852 s[i][j] = '\0';
853
854 if (u)
855#endif
856 {
857#if defined(LS_MASTER)
858 lockstep_send(s[i]);
859#endif
860 i++;
861 }
862
863#if !defined(LS_SLAVE)
864 if (c == '\n')
865 {
866#if defined(LS_MASTER)
867 lockstep_send("TERM");
868#endif
869 break;
870 }
871#endif
872 }
873
874 s[i][0] = '\0';
875 if (parse(s))
876 break;
877 }
878
879#if !defined(LS_SLAVE)
880 if (filename)
881 fclose(f);
882#endif
883}

888int CTraceEngine::parse(char command[100][100])
889{
890 int i = 0;
891 int numargs;
892 int result;
893 int RunCycles;
894 u64 iFrom;
895 u64 iTo;
896 u64 iJump;
897
898 for (numargs = 0; command[numargs][0] != '\0'; numargs++);
899
900 if ((numargs > 0) && (command[0][0] == '#' || command[0][0] == ';'
901 || command[0][0] == '!' || (command[0][0] == '/' && command[0][1] == '/')))
902
903 //comment
904 return 0;
905 switch (numargs)
906 {
907 case 0:
908
909 // empty command
910 return 0;
911
912 case 1:
913 if (!strncasecmp(command[0], "EXIT", strlen(command[0]))
914 || !strncasecmp(command[0], "QUIT", strlen(command[0]))) return 1;
915 if (!strncasecmp(command[0], "HELP", strlen(command[0]))
916 || !strncasecmp(command[0], "?", strlen(command[0])))
917 {
918 printf(" \n");
919 printf("Available commands: \n");
920 printf(" HELP | ? \n");
