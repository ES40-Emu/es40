33// ============================================================================
34//
35// EV68 FPCR layout (all meaningful bits are in the upper 32 bits):
36//
37// 63 SUM - Summary Bit (SUM).
38// 62 INED - Inexact Disable (INED).
39// 61 UNFD - Underflow Disable (UNFD)
40// 60 UNDZ - Underflow to Zero (UNDZ)
41// 59:58 DYN - Dynamic Rounding Mode (DYN)
42// 57 IOV - Integer Overflow (IOV)
43// 56 INE - Inexact Result (INE)
44// 55 UNF - Underflow (UNF).
45// 54 OVF - Overflow (OVF)
46// 53 DZE - Division by Zero (DZE)
47// 52 INV - Invalid Operation (INV)
48// 51 OVFD - Overflow Disable (OVFD)
49// 50 DZED - Division by Zero Disable (DZED)
50// 49 INVD - Invalid Operation Disable (INVD)
51// 48 DNZ - Denormal Operands to Zero (DNZ)
52// 47 DNOD - Denormal Operand Exception Disable (DNOD)
53// 46:0 Reserved, must be read as zero
54//
55// Alpha Architecture Reference Manual, 4th ed. [ARM]:
56// Section 4.7.8 - Floating-Point Control Register (FPCR)
57// Section 4.7.8.1 - Accessing the FPCR
58// Section 4.7.8.2 - Default Values of the FPCR
59// Section 4.10.4 - Move from/to Floating-Point Control Register
60// at least according to this one https://download.majix.org/dec/alpha_arch_ref.pdf
61// ============================================================================
62
63std::uint64_t Ev68FPCR::normalize(std::uint64_t value)
64{
65 u64 val = value & ~(FPCR_RAZ | U64(0x00000000FFFFFFFF));
66
67 if (val & FPCR_ERR)
68 val |= FPCR_SUM;
69 else
70 val &= ~FPCR_SUM;
71
72 return val;
73}

75std::uint64_t Ev68FPCR::cvtql_overflow(std::uint64_t stored)
76{
77 return stored | FPCR_IOV;
78}
