37/* ------------ I2C abstract device (open-drain bus) ------------ */
38class I2CDevice {
39public:
40 virtual ~I2CDevice() = default;
41
42 // START/STOP edges (SDA falling/rising while SCL is high)
43 virtual void on_start();
44 virtual void on_stop();
45
46 // Clock edges; sda_line is the current bus level of SDA at this instant.
47 virtual void on_scl_rise(bool sda_line);
48 virtual void on_scl_fall(bool sda_line);
49
50 // Open-drain pulls: return true to pull the line LOW (0), false to release (1).
51 virtual bool pull_sda_low() const; // ACK/data 0-bit
52 virtual bool pull_scl_low() const; // clock stretching (unused by SPD)
53};
