112 void start(CRunnable& target)
113 {
114 if (_thread.joinable())
115 throw std::runtime_error("CThread::start: thread already running");
116
117 _running.store(true, std::memory_order_release);
118
119 auto stableName = std::make_shared<std::string>(_name);
120 CThread* self = this;
121
122 _thread = std::thread([&target, self, stableName]()
123 {
124 tl_threadName = stableName->c_str();
125 tl_currentThread = self;
126
127#if !defined(_WIN32)
128 // Termination and broken-pipe signals are handled on the main thread only.
129 sigset_t blocked;
130 sigemptyset(&blocked);
131 for (int sig : { SIGQUIT, SIGTERM, SIGPIPE })
132 sigaddset(&blocked, sig);
133 pthread_sigmask(SIG_BLOCK, &blocked, nullptr);
134#endif
135
136 try
137 {
138 target.run();
139 }
140 catch (const std::exception& ex)
141 {
142 printf("Thread '%s' terminated with exception: %s\n",
143 stableName->c_str(), ex.what());
144 }
145 catch (...)
146 {
147 printf("Thread '%s' terminated with an unknown exception.\n",
148 stableName->c_str());
149 }
150
151 {
152 std::lock_guard<std::mutex> lk(self->_doneMutex);
153 self->_running.store(false, std::memory_order_release);
154 }
155 self->_doneCond.notify_all();
156 });
157
158 if (_prio != PRIO_NORMAL)
159 applyPriority();
160 }

174 bool tryJoin(long milliseconds)
175 {
176 if (!_thread.joinable())
177 return true;
178
179 {
180 std::unique_lock<std::mutex> lk(_doneMutex);
181 if (!_doneCond.wait_for(lk, std::chrono::milliseconds(milliseconds),
182 [this] { return !_running.load(std::memory_order_acquire); }))
183 {
184 return false; // timeout
185 }
186 }
187
188 if (_thread.joinable())
189 _thread.join();
190 return true;
191 }

227 static int mapPrioWin32(Priority prio)
228 {
229 static const int levels[] = { THREAD_PRIORITY_LOWEST, THREAD_PRIORITY_BELOW_NORMAL,
230 THREAD_PRIORITY_NORMAL, THREAD_PRIORITY_ABOVE_NORMAL, THREAD_PRIORITY_HIGHEST };
231 return (prio >= PRIO_LOWEST && prio <= PRIO_HIGHEST) ? levels[prio]
232 : THREAD_PRIORITY_NORMAL;
233 }
234
235 void applyPriority()
236 {
237 HANDLE h = (HANDLE)(_thread.native_handle());
238 if (!SetThreadPriority(h, mapPrioWin32(_prio)))
239 printf("Warning: SetThreadPriority failed for thread '%s' (error %lu)\n",
240 _name.c_str(), GetLastError());
241 }
242
243#else // POSIX
244
245 static int mapPrioPosix(Priority prio)
246 {
247 // The five levels are spread evenly over the SCHED_OTHER range.
248#if defined(__VMS) || defined(__digital__)
249 const int lo = PRI_OTHER_MIN, hi = PRI_OTHER_MAX;
250#else
251 static const int lo = sched_get_priority_min(SCHED_OTHER);
252 static const int hi = sched_get_priority_max(SCHED_OTHER);
253#endif
254 const int level = (prio >= PRIO_LOWEST && prio <= PRIO_HIGHEST) ? prio : PRIO_NORMAL;
255 return lo + (hi - lo) * level / 4;
256 }
257
258 void applyPriority()
259 {
260 struct sched_param par {};
261 par.sched_priority = mapPrioPosix(_prio);
262 int rc = pthread_setschedparam(
263 _thread.native_handle(), SCHED_OTHER, &par);
264 if (rc != 0)
265 printf("Warning: pthread_setschedparam failed for thread '%s' (rc=%d).\n",
266 _name.c_str(), rc);
267 }
268
269#endif // _WIN32 / POSIX
