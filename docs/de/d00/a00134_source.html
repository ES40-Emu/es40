154 virtual string ask()
155 {
156 AnswerSet::iterator itAnswers;
157 string options;
158
159 /* Iterate through the answer set to create the
160 * options list.
161 */
162
163 for (itAnswers = mAnswerSet.begin(); itAnswers != mAnswerSet.end(); itAnswers++)
164 {
165 if (options != "")
166 options += ",";
167 options += itAnswers->getAnswer();
168 }
169
170 /* Set the options list.
171 */
172 setOptions(options);
173
174 /* Keep repeating the question until a valid
175 * answer is received.
176 */
177 for (;;)
178 {
179 /* Ask the question as a FreeTextQuestion. This
180 * takes care of the explanation and default
181 * value handling.
182 */
183 FreeTextQuestion::ask();
184
185 /* Try to find the answer received in the answer
186 * set.
187 */
188 itAnswers = find(mAnswerSet.begin(), mAnswerSet.end(), mAnswer);
189
190 /* Has a matching answer be found?
191 */
192 if (itAnswers != mAnswerSet.end())
193 {
194 /* Get the value for the received answer.
195 */
196 string value = itAnswers->getValue();
197 haveChosen(mAnswer);
198 mAnswer = value;
199
200 /* Return the value.
201 */
202 return mAnswer;
203 }
204
205 /* No valid, matching answer received.
206 */
207 cout << "\nPlease enter a valid value, or '?' for help.\n\n";
208 }
209 };
