552 // Wall-clock RPCC: state.cc advances by real elapsed time * cpu_hz so it tracks the configured
553 // CPU frequency regardless of how fast/bursty the JIT runs. This is the last sync timestamp;
554 // the delta since it (when cc_ena) is added to state.cc at each sync point, then it's reset to now.
555 std::chrono::steady_clock::time_point cc_last_sync;
556 u64 cc_wall_remainder = 0; // fractional cycle numerator, / 1e9
557
558 // Advance the wall-clock cc to now. Called at batch boundaries AND on every guest RPCC read.
559 void sync_cc_wallclock()
560 {
561 const auto now = std::chrono::steady_clock::now();
562 if (cc_last_sync > now)
563 cc_last_sync = now;
564 auto cc_delta = now - cc_last_sync;
565 cc_last_sync = now;
566 if (state.cc_ena)
567 {
568 if (cc_delta > std::chrono::seconds(1)) // cap (not drop) odd deltas, as at batch top
569 cc_delta = std::chrono::seconds(1);
570 const u64 ns = (u64)std::chrono::duration_cast<std::chrono::nanoseconds>(cc_delta).count();
571 // Carry sub-cycle precision across frequent syncs.
572 const u64 scaled = ns * cpu_hz + cc_wall_remainder;
573 u64 add = scaled / 1000000000ULL;
574 cc_wall_remainder = scaled % 1000000000ULL;
575 // Repay any rpcc_read floor-borrow by WITHHOLDING wall progress, never by subtracting
576 if (cc_borrow)
577 {
578 const u64 repay = cc_borrow < add ? cc_borrow : add;
579 add -= repay;
580 cc_borrow -= repay;
581 }
582 state.cc += add;
583 }
584 }

586 // RPCC read with a forward-progress guarantee. Real EV6 RPCC advances every cycle
587 static constexpr u64 kRpccCollisionStep = 1;
588 u64 cc_last_read = 0;
589 u64 cc_borrow = 0;
590 u64 rpcc_read()
591 {
592 sync_cc_wallclock();
593 if (state.cc_ena && state.cc <= cc_last_read)
594 {
595 cc_borrow += cc_last_read + kRpccCollisionStep - state.cc;
596 state.cc = cc_last_read + kRpccCollisionStep;
597 }
598 cc_last_read = state.cc;
599 return ((u64) state.cc_offset) << 32 | (state.cc & U64(0xffffffff));
600 }

623 // Data page translation cache: direct-mapped by virtual page (kDpcEntries slots/dir) so a
624 // multi-page access pattern doesn't thrash a single slot. The inline load checks one slot.
625 // Cached normal accesses use the current CM/ASN0 context; family control writes that
626 // affect address rules must invalidate it. Privileged VPTE/ALT accesses bypass it.
627 static constexpr int kDpcBits = 7; // 128 slots per mode bank (8KB pages -> 1 MB)
628 // Page-indexed hint into state.tb[]: FindTBEntry and add_tb probe it before anything else.
629 // Entries are validated on use, so a stale hint is harmless and nothing needs clearing.
630 // Hashed by page number: Tru64 text (0x120000000) and data (0x140000000) pages share their
631 // low page bits, so a low-bits index made each alternation fall through to the linear scan.
632 static constexpr int kTbIdxBits = 10;
633 static inline u32 tb_idx_slot(u64 virt) { return ((u32)(virt >> 13) * 0x9E3779B1u) >> (32 - kTbIdxBits); }
634 u16 m_tb_idx[2][1 << kTbIdxBits] = {};
635 inline void tb_index_clear(int t) { memset(m_tb_idx[t], 0, sizeof(m_tb_idx[t])); }
636 static constexpr int kDpcEntries = 1 << kDpcBits;
637 static constexpr u64 kDpcMask = (u64) kDpcEntries - 1;
638 static inline u64 dpc_index(u64 va) { return (va >> 13) & kDpcMask; }
639 struct SDataPageCache {
640 u64 virt_page; // va & ~0x1FFF; write row may include kDpcLLWatchTag
641 u64 phys_base; // pa & ~0x1FFF
642 u64 host_bias; // dram_ptr + phys_base - unmarked VA page; 0 for MMIO (host = bias + va)
643 u8 cm; // current mode (CM) at fill time
644 u8 asn; // data ASN (asn0) at fill time
645 bool valid;
646 } data_page_cache[2][4 * kDpcEntries]; // [rw][cm bank | dpc_index(va)]; [0]=read, [1]=write
647 // Byte offset of the current mode's bank; compiled code adds it to the slot index, so a CM
648 // change selects another bank instead of flushing. Kept in step by dpc_sync_bank().
649 u32 m_dpc_bank = 0;
650 // Derived execution context, outside saved state.
651 u32 m_jit_mode_bit = 1;
652 inline void dpc_sync_bank() {
653 const u32 mode = (u32)(state.cm & 3);
654 m_dpc_bank = mode * kDpcEntries * (u32)sizeof(SDataPageCache);
655 m_jit_mode_bit = 1u << mode;
656 }
657 inline SDataPageCache& dpc_slot(int rw, u64 va) { return data_page_cache[rw][((state.cm & 3) << kDpcBits) | dpc_index(va)]; }
658 static_assert(sizeof(SDataPageCache) == 32, "DPC slots stay power-of-two for JIT shift indexing");

660 inline void flush_data_page_cache() {
661#if defined(JIT_REGPROF) && defined(_WIN32)
662 m_dpc_flushes++;
663#endif
664 // The generated and helper hit paths compare virt_page before using the slot. An all-ones
665 // tag cannot equal an 8 KB-aligned virtual page, so a bulk fill invalidates both rows
666 // and avoids excessive stores on native-PAL translation flushes.
667 memset(data_page_cache, 0xff, sizeof(data_page_cache));
668 }
669 inline void invalidate_data_page_cache(u64 virt, u64 match_mask) {
670 // The common GH=0 mapping occupies exactly one direct-mapped slot. Clearing only its
671 // read/write tags avoids a full cache sweep.
672 constexpr u64 gh0_match = U64(0x0000ffffffffe000);
673 if (match_mask == gh0_match) {
674 const u64 i = dpc_index(virt);
675 for (int bank = 0; bank < 4; ++bank) data_page_cache[0][(bank << kDpcBits) | i].virt_page = ~U64(0);
676 for (int bank = 0; bank < 4; ++bank) data_page_cache[1][(bank << kDpcBits) | i].virt_page = ~U64(0);
677 return;
678 }
679 // A granularity-hint entry can cover multiple cache indices.
680 for (int rw = 0; rw < 2; ++rw)
681 for (int i = 0; i < 4 * kDpcEntries; ++i)
682 if (!((data_page_cache[rw][i].virt_page ^ virt) & match_mask))
683 data_page_cache[rw][i].virt_page = ~U64(0);
684 }

1094inline void CAlphaCPU::irq_h(int number, bool assert, int delay)
1095{
1096 std::unique_lock<std::mutex> irq_lock(m_irq_lock);
1097 const u64 bit = U64(0x1) << number;
1098 bool active = (state.eir & bit) || state.irq_h_timer[number];
1099 if (assert && !active)
1100 {
1101 if (delay)
1102 {
1103 state.irq_h_timer[number] = delay;
1104 state.check_timers = true;
1105 }
1106 else
1107 {
1108 state.eir |= bit;
1109#ifdef JIT_STATS
1110 m_stat_kick[3]++;
1111#endif
1112 state.check_int = true;
1113 }
1114
1115 irq_lock.unlock();
1116 wake_idle();
1117 return;
1118 }
1119
1120 if (assert && (state.eir & bit))
1121 {
1122 // New request on an already-high line - a second device asserting while
1123 // the first is still being serviced.
1124 // The level in eir is already correct, but check_int was consumed when the
1125 // interrupt was dispatched, and after its REI nothing re-polls.
1126 // Re-kick so it is picked up at the next dispatch boundary, matching real hardware.
1127 state.check_int = true;
1128 irq_lock.unlock();
1129 wake_idle();
1130 return;
1131 }
1132
1133 if (!assert && active)
1134 {
1135 state.eir &= ~bit;
1136 state.irq_h_timer[number] = 0;
1137 bool timers_pending = false;
1138 for (int i = 0; i < 6; i++)
1139 {
1140 if (state.irq_h_timer[i])
1141 timers_pending = true;
1142 }
1143 state.check_timers = timers_pending;
1144 }
