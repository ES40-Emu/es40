262void CAlphaCPU::vmspal_call_swpctx()
263{
264 p23 = state.pc;
265
266 if (r16 & 0x7f)
267 {
268
269 //context pointer not properly aligned...
270 p4 = 0x430;
271 hw_stq(p21 + 0x150, p23);
272 hw_stq(p21 + 0x158, p4);
273 vmspal_int_initiate_exception();
274 return;
275 }
276
277 hw_ldq(r16 + 0x30, p4);
278 hw_ldq(r16 + 0x38, p5);
279 hw_ldq(r16 + 0x28, p6);
280
281 p20 = state.aster | (state.astrr << 4);
282
283 p6 &= 0xff;
284 state.asn0 = (int)p6;
285 state.asn1 = (int)p6;
286 state.asn = (int)p6;
287 flush_data_page_cache();
288 jit_note_asn_change();
289 state.aster = (int)p4 & 0xf;
290 state.astrr = (int)(p4 >> 4) & 0xf;
291 state.fpen = (int)p5 & 1;
292 state.ppcen = (int)(p5 >> 0x3e) & 1;
293 kick_int_if_pending();
294
295 hw_ldq(r16 + 0x40, p7);
296 hw_ldq(r16 + 0x20, p6);
297
298 p4 = (state.cc & U64(0xffffffff)) + state.cc_offset;
299 state.cc_offset = ((u32)p7 & 0xffffffff) - (state.cc & U64(0xffffffff));
300
301 p6 <<= 0x0d;
302 hw_stq(p21 + 8, p6);
303 hw_ldq(p21 + 0x10, p5);
304 hw_ldq(r16, p6);
305 hw_stl(p5 + 0x40, p4);
306 hw_stq(p5 + 0x30, p20);
307 hw_stq(p5, r30);
308
309 r30 = p6;
310 hw_stq(p21 + 0x10, r16);
311
312 // Revalidate instruction fetch in the new context without discarding ASN-tagged cache lines.
313 set_pc(state.pc);
314}

702{
703 u64 pa;
704
705 p23 = state.pc;
706 p4 = r18 & 3; // alt mode
707 p5 = p22 & 0x18;
708 p5 >>= 3; // current mode
709 p6 = p5 - p4;

713 hw_stq(p21 + 0x140, state.pc);

764int CAlphaCPU::vmspal_call_rei()
765{
766 u64 phys_address;
767
768 p23 = state.pc;
769
770 // Reading the frame can invoke a TB-miss handler that overwrites p7.
771 const u64 old_cm = p22 & 0x18;
772 hw_stq(p21 + 0x150, p23);
773 if (r30 & 0x3f)
774 {
775
776 // stack not aligned
777 p4 = 0x430;
778 hw_stq(p21 + 0x158, p4);
779 return vmspal_int_initiate_exception();
780 }
781
782 if (old_cm)
783 {
784
785 // what to do if the next instruction results in a TNV exception?
786 ldq(r30 + 0x38, p20);
787 // ARM 10.1.11: REI clears both intr_flag and lock_flag.
788 state.bIntrFlag = false;
789 cSystem->cpu_clear_lock(state.iProcNum);
790 ldq(r30 + 0x30, p23);
791 p4 = p20 & 0x18; // new cm
792 if ((p4 < old_cm) || (p20 & ~U64(0x3f0000000000001b)))
793 {
794 p4 = 0x430;
795 hw_stq(p21 + 0x158, p4);
796 vmspal_int_initiate_exception();
797 return 0;
798 }
799
800 ldq(r30 + 0x10, state.r[4]);
801 ldq(r30 + 0x18, state.r[5]);
802 ldq(r30 + 0x20, state.r[6]);
803 ldq(r30 + 0x28, state.r[7]);
804 ldq(r30, r2);
805 ldq(r30 + 8, r3);
806 hw_ldq(p21 + 0x10, p6);
807 p5 = (p20 >> 56) & 0xff;
808 p7 = old_cm + p6;
809 p6 += p4;
810 p20 &= 0xffff;
811 p22 &= ~U64(0xffff);
812 state.cm = (int)(p4 >> 3) & 3;
813 p22 |= p20;
814 p23 &= ~U64(0x3);
815 p20 = r30 + 0x40;
816 p20 |= p5;
817 hw_stq(p7, p20);
818 hw_ldq(p6, r30);
819 kick_int_if_pending();
820 set_pc(p23);
821 return 0;
822 }
823
824 // A kernel-stack fault must report the REI itself, not its next PC.
825 hw_stq(p21 + 0x150, state.current_pc);
826 kldq(r30 + 0x38, p20);
827 state.bIntrFlag = false;
828 cSystem->cpu_clear_lock(state.iProcNum);
829 p7 = (p20 >> 8) & 0xff;
830 kldq(r30 + 0x10, state.r[4]);
831 kldq(r30 + 0x18, state.r[5]);
832 kldq(r30 + 0x20, state.r[6]);
833 kldq(r30 + 0x28, state.r[7]);
834 kldq(r30, r2);
835 kldq(r30 + 8, r3);
836 p4 = p20 & 0x18; // new cm
837 kldq(r30 + 0x30, p23);
838
839 if (p4)
840 {
841 hw_ldq(p21 + 0x10, p7);
842 p7 += p4;
843 state.cm = (int)(p4 >> 3) & 3;
844 p5 = (p20 >> 56) & 0xff;
845 p20 &= 0xff;
846 p22 &= ~U64(0xffff);
847 p22 |= p20;
848 p23 &= ~U64(0x3);
849 p20 = r30 + 0x40;
850 p20 |= p5;
851 hw_ldq(p7, r30);
852 hw_stq(p21 + 0x18, p20);
853 state.eien = ipl_ier_mask[0][0];
854 state.slen = ipl_ier_mask[0][1];
855 state.cren = ipl_ier_mask[0][2];
856 state.pcen = ipl_ier_mask[0][3];
857 state.sien = ipl_ier_mask[0][4];
858 state.asten = ipl_ier_mask[0][5];
859 kick_int_if_pending();
860 set_pc(p23);
861 return 0;
862 }
863
864 p5 = (p20 >> 56) & 0xff;
865 p20 &= 0xffff;
866 p22 &= ~U64(0xffff);
867 p7 = (p20 >> 8) & 0xff;
868 p22 |= p20;
869 p23 &= ~U64(0x3);
870 p20 = r30 + 0x40;
871 r30 = p20 | p5;
872 state.eien = ipl_ier_mask[p7][0];
873 state.slen = ipl_ier_mask[p7][1];
874 state.cren = ipl_ier_mask[p7][2];
875 state.pcen = ipl_ier_mask[p7][3];
876 state.sien = ipl_ier_mask[p7][4];
877 state.asten = ipl_ier_mask[p7][5];
878 kick_int_if_pending();
879 set_pc(p23);
880 return 0;
881}
