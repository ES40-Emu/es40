39// Decoded hardware store, outside saved CPU state. Plain stores translate before
40// rounding down the physical write. Conditional stores use the original PA.
41struct AlphaHWStoreDescriptor
42{
43 AlphaHWStoreAddress address = AlphaHWStoreAddress::Invalid;
44 std::uint32_t size_bits = 0;
45 std::int32_t displacement = 0;
46 bool alternate = false;
47 bool conditional = false;
48 bool compilable = false;
49};
50
51constexpr bool alpha_hw_store_valid(const AlphaHWStoreDescriptor& store)
52{
53 return (store.address == AlphaHWStoreAddress::Physical
54 || store.address == AlphaHWStoreAddress::Virtual)
55 && (store.size_bits == 32 || store.size_bits == 64);
56}
57
58// The shared JIT currently implements only plain physical hardware stores.
59constexpr bool alpha_hw_store_compilable(const AlphaHWStoreDescriptor& store)
60{
61 return alpha_hw_store_valid(store) && store.compilable
62 && store.address == AlphaHWStoreAddress::Physical && !store.conditional;
63}
64
65struct AlphaHWStorePolicy
66{
67 AlphaHWStoreDescriptor (*decode)(std::uint32_t instruction) = nullptr;
68};
69
70constexpr AlphaHWStoreDescriptor alpha_decode_hw_store(
71 const AlphaHWStorePolicy& policy, std::uint32_t instruction)
72{
73 return policy.decode ? policy.decode(instruction) : AlphaHWStoreDescriptor{};
74}
