67/*
68 * va_debug():
69 *
70 * Used internally by debug() and fatal().
71 */
72static void va_debug(va_list argp, char* fmt)
73{
74 char buf[DEBUG_BUFSIZE + 1];
75 char* s;
76 int i;
77
78 buf[0] = buf[DEBUG_BUFSIZE] = 0;
79
80 //vsnprintf(buf, DEBUG_BUFSIZE, fmt, argp);
81 sprintf(buf, fmt, argp);
82
83 s = buf;
84 while (*s)
85 {
86 if (debug_currently_at_start_of_line)
87 {
88 for (i = 0; i < debug_indent; i++)
89 printf(" ");
90 }
91
92 printf("%c", *s);
93
94 debug_currently_at_start_of_line = 0;
95 if (*s == '\n' || *s == '\r')
96 debug_currently_at_start_of_line = 1;
97 s++;
98 }
99}

113/*
114 * debug():

116 * Debug output (ignored if quiet_mode is set).
117 */
118void debug(char* fmt, ...)
119{
120 va_list argp;
121
122 if (quiet_mode)
123 return;
124
125 va_start(argp, fmt);
126 va_debug(argp, fmt);
127 va_end(argp);
128}
