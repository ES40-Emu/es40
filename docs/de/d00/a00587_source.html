70// ---------------------------------------------------------------------
71// Device open.
72//
73// Linux: a single shared clone device (/dev/net/tun) is opened, and the
74// TUNSETIFF ioctl both selects (attaches to) an existing persistent tap
75// interface and creates a new one if it doesn't exist yet - the kernel
76// decides which based on whether the name is already taken and whether
77// the caller has CAP_NET_ADMIN.
78//
79// FreeBSD/NetBSD: there is no separate clone-and-bind step. A TAP device
80// *is* a character device node, normally /dev/tapN, and using it is just
81// open("/dev/tapN", O_RDWR) - exactly like any other file. If the
82// interface doesn't exist yet (ENOENT/ENXIO), it can be instantiated with
83// the generic BSD "interface cloning" ioctl SIOCIFCREATE, which is the
84// same mechanism "ifconfig tapN create" uses; afterwards the device node
85// can be opened normally.
86// ---------------------------------------------------------------------
87bool CNetworkTap::tap_open(const char *devid_string, const char *name) {
88#if defined(__linux__)
89 struct ifreq ifr;
90
91 tap_fd = open("/dev/net/tun", O_RDWR);
92 if (tap_fd < 0) {
93 printf("%s: Cannot open /dev/net/tun: %s\n", devid_string, strerror(errno));
94 return false;
95 }
96
97 memset(&ifr, 0, sizeof(ifr));
98 ifr.ifr_flags = IFF_TAP | IFF_NO_PI;
99 if (name && name[0])
100 strncpy(ifr.ifr_name, name, IFNAMSIZ - 1);
101
102 if (ioctl(tap_fd, TUNSETIFF, &ifr) < 0) {
103 printf("%s: TUNSETIFF failed for %s: %s\n", devid_string, name, strerror(errno));
104 ::close(tap_fd);
105 tap_fd = -1;
106 return false;
107 }
108
109 return true;
110
111#elif defined(__FreeBSD__) || defined(__NetBSD__)
112 tap_fd = open(name, O_RDWR);
113 if (tap_fd < 0) {
114 printf("%s: Cannot open %s: %s\n", devid_string, name, strerror(errno));
115 return false;
116 }
117
118 return true;
119#else
120#error "HAVE_TAP_NET defined on an unsupported platform"
121#endif
122}

124bool CNetworkTap::init(const char *devid_string, CConfigurator *cfg)
125{
126 devid_for_log = devid_string;
127 char *adapter = cfg->get_text_value("adapter");
128
129 if (!tap_open(devid_string, adapter))
130 return false;
131
132 int flags = fcntl(tap_fd, F_GETFL, 0);
133 if (flags < 0 || fcntl(tap_fd, F_SETFL, flags | O_NONBLOCK) < 0) {
134 printf("%s: Cannot set TAP fd non-blocking: %s\n", devid_string, strerror(errno));
135 close();
136 return false;
137 }
138
139 return true;
140}

142int CNetworkTap::send(const u8 *data, int len) {
143 if (tap_fd < 0)
144 return -1;
145 ssize_t n = write(tap_fd, data, len);
146 if (n < 0) {
147 if (errno == EAGAIN || errno == EWOULDBLOCK)
148 return 0; /* Host-side queue full; drop the packet like a real wire would. */
149 printf("TAP: Error sending packet: %s\n", strerror(errno));
150 return -1;
151 }
152
153 return 0;
154}

156int CNetworkTap::receive(const u8 **data, int *len)
157{
158 if (tap_fd < 0)
159 return -1;
160 ssize_t n = read(tap_fd, rx_buf, sizeof(rx_buf));
161
162 if (n > 0) {
163 *data = rx_buf;
164 *len = (int)n;
165 return 1;
166 }
167
168 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK))
169 return 0; // no packet available
170 if (n == 0)
171 return 0;
172
173 return -1;
174}
