53#define MAKE_COLOUR \
54 ( \
55 red, red_shiftfrom, red_shiftto, red_mask, green, green_shiftfrom, \
56 green_shiftto, green_mask, blue, blue_shiftfrom, blue_shiftto, blue_mask \
57 ) \
58 ( \
59 ( \
60 ( \
61 ((red_shiftto) > (red_shiftfrom)) ? (red) << \
62 ((red_shiftto) - (red_shiftfrom)) : (red) >> \
63 ((red_shiftfrom) - (red_shiftto)) \
64 ) & (red_mask) \
65 ) | \
66 ( \
67 ( \
68 ((green_shiftto) > (green_shiftfrom)) ? (green) << \
69 ((green_shiftto) - (green_shiftfrom)) : (green) >> \
70 ((green_shiftfrom) - (green_shiftto)) \
71 ) & (green_mask) \
72 ) | \
73 ( \
74 ( \
75 ((blue_shiftto) > (blue_shiftfrom)) ? (blue) << \
76 ((blue_shiftto) - (blue_shiftfrom)) : (blue) >> \
77 ((blue_shiftfrom) - (blue_shiftto)) \
78 ) & (blue_mask) \
79 ) \
80 )
