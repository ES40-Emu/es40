76class CSemaphore
77{
78public:
80 CSemaphore(int n, int max)
81 : _n(n), _max(max)
82 {
83 assert(n >= 0 && max > 0 && n <= max);
84 }
85
87 explicit CSemaphore(int n)
88 : _n(n), _max(n)
89 {
90 assert(n > 0);
91 }
92
93 ~CSemaphore() = default;
94
96 void set()
97 {
98 std::lock_guard<std::mutex> lk(_mutex);
99 if (_n < _max)
100 ++_n;
101 _cond.notify_one();
102 }
103
105 void wait()
106 {
107 std::unique_lock<std::mutex> lk(_mutex);
108 _cond.wait(lk, [this] { return _n > 0; });
109 --_n;
110 }
111
113 void wait(long milliseconds)
114 {
115 std::unique_lock<std::mutex> lk(_mutex);
116 if (!_cond.wait_for(lk, std::chrono::milliseconds(milliseconds),
117 [this] { return _n > 0; }))
118 {
119 throw std::runtime_error("CSemaphore::wait: timeout");
120 }
121 --_n;
122 }
123
125 bool tryWait(long milliseconds)
126 {
127 std::unique_lock<std::mutex> lk(_mutex);
128 if (!_cond.wait_for(lk, std::chrono::milliseconds(milliseconds),
129 [this] { return _n > 0; }))
130 {
131 return false;
132 }
133 --_n;
134 return true;
135 }
136
137private:
138 int _n;
139 int _max;
140 std::mutex _mutex;
141 std::condition_variable _cond;
142
143 CSemaphore(const CSemaphore&) = delete;
144 CSemaphore& operator=(const CSemaphore&) = delete;
145};
