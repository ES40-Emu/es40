114EditorField prepare_editor_field(const FormEntry_t& entry, const char *preset)
115{
116 EditorField field;
117 field.value = preset ? preset : (entry.preset ? entry.preset : "");
118 if (entry.validation_callback) entry.validation_callback(&field);
119 if (field.type == EditorField::Type::Choice && field.choices == vector<string>{STR_NO, STR_YES})
120 {
121 // Unify boolean values read from config file
122 if (!strcasecmp(field.value.c_str(), "no") || !strcasecmp(field.value.c_str(), "false") || field.value == "0")
123 field.value = STR_NO;
124 if (!strcasecmp(field.value.c_str(), "yes") || !strcasecmp(field.value.c_str(), "true") || field.value == "1")
125 field.value = STR_YES;
126 }
127 return field;
128}

130bool validate_editor_field(EditorField& field)
131{
132 const string& value = field.value;
133 switch (field.type)
134 {
135 case EditorField::Type::Choice:
136 for (const auto& choice : field.choices)
137 if (!strcasecmp(value.c_str(), choice.c_str())) { field.value = choice; return true; }
138 return false;
139 case EditorField::Type::Integer:
140 case EditorField::Type::Number:
141 {
142 if (value.empty()) return false;
143 char *end;
144 errno = 0;
145 double number = strtod(value.c_str(), &end);
146 if (end == value.c_str() || *end || errno || !isfinite(number)) return false;
147 if (field.type == EditorField::Type::Integer && value.find_first_not_of("+-0123456789") != string::npos) return false;
148 return number >= field.minimum && number <= field.maximum;
149 }
150 case EditorField::Type::Pattern:
151 return regex_match(value, regex(field.pattern, regex::extended));
152 default: return true;
153 }
154}

156FormValues_t show_form(const char *title, FormEntry_t entry[], FormValues_t preset,
157 int count, FormCheckFuncPtr check)
158{
159 existing_form_nodes.clear(); remember_form_nodes(theConfig);
160 vector<EditorField> fields;
161 for (int i = 0; i < count; ++i) fields.push_back(prepare_editor_field(entry[i], preset ? preset[i] : nullptr));
162 const auto initial = fields;
163 while (editor_ui->form(title, entry, fields))
164 {
165 bool valid = true;
166 for (int i = 0; i < count; ++i)
167 if (!validate_editor_field(fields[i]))
168 {
169 string error = string("Invalid value for ") + entry[i].label + ".";
170 show_text("Error", error.c_str()); valid = false; break;
171 }
172 if (!valid) continue;
173 auto values = (FormValues_t)calloc(count, sizeof(char *));
174 for (int i = 0; i < count; ++i) values[i] = strdup(fields[i].value.c_str());
175 changed_fields.assign(count, false);
176 for (int i = 0; i < count; ++i)
177 {
178 bool same = initial[i].value == values[i];
179 if (initial[i].type == EditorField::Type::Integer || initial[i].type == EditorField::Type::Number)
180 same = strtod(initial[i].value.c_str(), nullptr) == strtod(values[i], nullptr);
181 if (initial[i].type == EditorField::Type::Choice)
182 same = !strcasecmp(initial[i].value.c_str(), values[i]);
183 changed_fields[i] = !same;
184 }
185 try
186 {
187 if (!check || check(entry, count, values))
188 {
189 return values;
190 }
191 }
192 catch (...) { for (int i = 0; i < count; ++i) free(values[i]); free(values); throw; }
193 for (int i = 0; i < count; ++i) free(values[i]);
194 free(values);
195 }
196 return nullptr;
197}

232int memory_size2bits(char *mem_size)
233{
234 char unit = mem_size[strlen(mem_size) - 1];
235 u64 amount = atoi(mem_size);
236 int mem_bits = 0;
237 switch (unit)
238 {
239 case 'm':
240 case 'M':
241 amount *= 1024 * 1024;
242 break;
243 case 'g':
244 case 'G':
245 amount *= 1024 * 1024 * 1024;
246 break;
247 default:
248 FAILURE_1(Logic, "Unrecognized memory size: '%s'", mem_size);
249 break;
250 }
251 while ((amount & 1) == 0 && amount != 0)
252 {
253 ++mem_bits;
254 amount >>= 1;
255 }
256 return mem_bits;
257}

264string memory_bits2size(int mem_bits)
265{
266 if (mem_bits < LOW_MEM_BITS || mem_bits > HI_MEM_BITS)
267 FAILURE_1(Logic, "Memory bits out of range: '%d'", mem_bits);
268
269 // from es40-cfg.cpp
270 string unit = "";
271 int j = mem_bits;
272 if (mem_bits < 30)
273 {
274 // Megabyte-range.
275 j -= 20;
276 unit = "M";
277 }
278 else
279 {
280 // Gigabyte range.
281 j -= 30;
282 unit = "G";
283 }
284 return (i2s(1 << j) + unit);
285}

301void write_configuration(const char *filename)
302{
303 const auto text = configuration_document->render(theConfig);
304 if (!strcmp(filename, "-"))
305 {
306#ifdef _WIN32
307 const int mode = _setmode(_fileno(stdout), _O_BINARY);
308#endif
309 bool failed = fwrite(text.data(), 1, text.size(), stdout) != text.size();
310 failed = fflush(stdout) != 0 || failed;
311#ifdef _WIN32
312 _setmode(_fileno(stdout), mode);
313#endif
314 if (failed) FAILURE(Configuration, "failed to write configuration to stdout");
315 return;
316 }
317 auto destination = output_snapshot ? *output_snapshot : read_config_file(filename, true);
318 // Saving back to the loaded file must also detect changes made while editing.
319 if (input_snapshot)
320 {
321 std::error_code ignored;
322 if (input_snapshot->path == destination.path ||
323 (destination.exists && std::filesystem::equivalent(input_snapshot->path, destination.path, ignored)))
324 {
325 destination.exists = input_snapshot->exists;
326 destination.text = input_snapshot->text;
327 }
328 }
329 save_config_file(destination, text);
330}

479void validation_pcislot(EditorField *field)
480{
481 // Only include free PCI slots, plus the field's set slot
482 int mybus = 99, myslot = 99;
483 sscanf(field->value.c_str(), "%d.%d", &mybus, &myslot);
484 vector<char *> choices;
485 for (int bus = 0; bus < MAX_PCI_BUS; ++bus)
486 for (int slot = 1; slot <= MAX_PCI_SLOT[bus]; ++slot)
487 if (is_pcislot_free(bus, slot) || (bus == mybus && slot == myslot))
488 choices.push_back(strdup(pcislot_str(bus, slot).c_str()));
489 choices.push_back(NULL);
490
491 editor_choices(field, choices.data());
492
493 for (size_t i = 0; i < choices.size() - 1; ++i)
494 free(choices[i]);
495}

1593void edit_pcicard(
1594 const char *title,
1595 const char *type,
1596 FormEntry_t entry[],
1597 int num_entries,
1598 bool (*check_values)(FormEntry_t[], int, FormValues_t) = NULL,
1599 void (*set_values)(CConfigurator*, FormEntry_t[], int, FormValues_t) = set_pcicard_values_default,
1600 void (*add_children)(CConfigurator*) = NULL
1601)
1602{
1603 vector<MenuEntry_t> choices = {
1604 {"Add new card", "", NULL}
1605 };
1606 for (auto c : sys0->find_children_myValue(type)) {
1607 choices.push_back({c->get_myName(), "", NULL});
1608 }
1609 int sel = 0;
1610 if (choices.size() > 1)
1611 {
1612 string title = string("Select ") + type + " card in PCI slot to edit";
1613 sel = show_menu(title.c_str(), choices.data(), choices.size(), MENU_2ND_LEVEL);
1614 }
1615 if (sel < 0)
1616 return;
1617
1618 CConfigurator *c;
1619 bool is_new_card = (sel == 0);
1620 int idx = fentry_index(entry, num_entries, "PCI slot");
1621 if (is_new_card)
1622 {
1623 c = new CConfigurator(sys0, (char *)PCI_SLOT_TEMP, (char *)type);
1624 entry[idx].preset = find_first_free_pcislot();
1625 if (entry[idx].preset == NULL)
1626 {
1627 show_text(title, "No free PCI slots");
1628 return;
1629 }
1630 }
1631 else
1632 {
1633 c = sys0->find_child(choices[sel].text);
1634 entry[idx].preset = strdup(c->get_myName() + strlen("pci"));
1635 }
1636
1637 FormValues_t preset = (FormValues_t)calloc(num_entries, sizeof(char *));
1638 for (int i = 0; i < num_entries; ++i)
1639 {
1640 if (i == idx)
1641 preset[i] = (char *)entry[i].preset;
1642 else
1643 {
1644 char *p = !is_new_card && entry[i].name ? c->get_text_value(entry[i].name) : NULL;
1645 preset[i] = p ? p : (char *)entry[i].preset;
1646 }
1647 }
1648
1649 FormValues_t values = show_form(title, entry, preset, num_entries, check_values);
1650 if (values != NULL)
1651 {
1652 set_values(c, entry, num_entries, values);
1653
1654 for (int i = 0; i < num_entries; ++i)
1655 free(values[i]);
1656 free(values);
1657
1658 if (add_children != NULL)
1659 add_children(c);
1660 }
1661 else if (is_new_card)
1662 sys0->remove_child(PCI_SLOT_TEMP);
1663 free(preset);
1664}
