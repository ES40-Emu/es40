40static size_t previous_char(const std::string& value, size_t pos)
41{
42 if (pos) --pos;
43 while (pos && (static_cast<unsigned char>(value[pos]) & 0xc0) == 0x80) --pos;
44 return pos;
45}
46static size_t next_char(const std::string& value, size_t pos)
47{
48 if (pos < value.size()) ++pos;
49 while (pos < value.size() && (static_cast<unsigned char>(value[pos]) & 0xc0) == 0x80) ++pos;
50 return pos;
51}
52static std::vector<std::string> wrap_text(const std::string& body, size_t width)
53{
54 std::vector<std::string> lines;
55 size_t start = 0;
56 do
57 {
58 size_t end = body.find('\n', start);
59 if (end == std::string::npos) end = body.size();
60 std::string line = body.substr(start, end - start);
61 while (line.size() > width)
62 {
63 size_t split = line.rfind(' ', width);
64 if (split == std::string::npos || split == 0) split = width;
65 lines.push_back(line.substr(0, split));
66 line.erase(0, split + (line[split] == ' ' ? 1 : 0));
67 }
68 lines.push_back(line);
69 start = end + 1;
70 } while (start <= body.size());
71 return lines;
72}

271 bool form(const char *title, FormEntry_t entry[], std::vector<EditorField>& fields) override
272 {
273 int current = 0, first = 0;
274 const int count = int(fields.size()), rows = (BOTTOM - TOP) / FORM_ROW;
275 size_t cursor = fields[0].value.size();
276 bool selected = true;
277 struct TextInputScope
278 {
279 SDL_Window *window;
280 ~TextInputScope() { SDL_StopTextInput(window); }
281 } input_scope{window};
282 require(SDL_StartTextInput(window));
283 auto focus = [&](int idx) { current = idx; cursor = fields[current].value.size(); selected = true; };
284 auto cycle = [&](int direction)
285 {
286 auto& field = fields[current];
287 if (field.choices.empty()) return;
288 auto found = std::find(field.choices.begin(), field.choices.end(), field.value);
289 int idx = found == field.choices.end() ? (direction > 0 ? -1 : 0) : int(found - field.choices.begin());
290 field.value = field.choices[(idx + direction + int(field.choices.size())) % field.choices.size()];
291 cursor = field.value.size();
292 };
293 auto insert = [&](const std::string& value)
294 {
295 auto& field = fields[current];
296 if (field.type == EditorField::Type::Choice) return;
297 if (selected) { field.value.clear(); cursor = 0; selected = false; }
298 // Match the terminal frontend's 256-byte field limit without splitting UTF-8.
299 if (field.value.size() + value.size() <= 256 && value.find_first_of("\r\n\t") == std::string::npos)
300 { field.value.insert(cursor, value); cursor += value.size(); }
301 };
302 for (;;)
303 {
304 visible(current, first, count, rows);
305 frame(title, "Up / Down: field Tab / Left / Right: choice F1: help");
306 label(940, 38, std::to_string(current + 1) + " / " + std::to_string(count), MUTED, 13);
307 for (int i = first; i < count && i < first + rows; ++i)
308 {
309 float y = TOP + float(i - first) * FORM_ROW;
310 label(26, y, entry[i].label, i == current ? INK : MUTED);
311 rectangle(26, y + 24, WIDTH - 52, 30, i == current ? ACCENT : INPUT);
312 const auto& field = fields[i];
313 size_t offset = 0;
314 if (i == current)
315 while (cursor - offset > 96) offset = next_char(field.value, offset);
316 label(38, y + 29, field.value.substr(offset), INK, 98);
317 if (field.type == EditorField::Type::Choice) label(WIDTH - 58, y + 29, "<>");
318 else if (i == current)
319 {
320 int column = 0;
321 for (size_t pos = offset; pos < cursor; pos = next_char(field.value, pos)) ++column;
322 rectangle(38 + float(column) * 10, y + 49, 10, 2, INK);
323 if (selected) label(WIDTH - 188, y + 29, "[replace]", MUTED, 9);
324 }
325 }
326 description(entry[current].description);
327 button(26, "F2: Apply", true); button(286, "F3 / Esc: Cancel"); button(546, "F1: Field help");
328 SDL_RenderPresent(renderer);
329 auto event = wait_event();
330 int button = clicked_button(event);
331 if (button == 0) return true;
332 if (button == 1) return false;
333 if (button == 2) text(entry[current].label, entry[current].description);
334 if (event.type == SDL_EVENT_MOUSE_WHEEL)
335 focus(std::max(0, std::min(count - 1, current + (event.wheel.y > 0 ? -1 : 1))));
336 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.button.button == SDL_BUTTON_LEFT &&
337 event.button.x >= 26 && event.button.x < WIDTH - 26 && event.button.y >= TOP &&
338 event.button.y < TOP + rows * FORM_ROW)
339 {
340 int idx = first + int(event.button.y - TOP) / FORM_ROW;
341 if (idx < count)
342 {
343 focus(idx);
344 if (fields[idx].type == EditorField::Type::Choice) cycle(1);
345 }
346 }
347 if (event.type == SDL_EVENT_TEXT_INPUT) insert(event.text.text);
348 if (event.type != SDL_EVENT_KEY_DOWN) continue;
349 auto& field = fields[current];
350 const bool choice = field.type == EditorField::Type::Choice;
351 if (event.key.mod & SDL_KMOD_CTRL)
352 {
353 switch (event.key.key)
354 {
355 case SDLK_A: selected = true; break;
356 case SDLK_C: if (selected) SDL_SetClipboardText(field.value.c_str()); break;
357 case SDLK_X:
358 if (selected && !choice) { SDL_SetClipboardText(field.value.c_str()); field.value.clear(); cursor = 0; }
359 break;
360 case SDLK_V:
361 {
362 char *clipboard = SDL_GetClipboardText();
363 if (clipboard) { insert(clipboard); SDL_free(clipboard); }
364 break;
365 }
366 default: break;
367 }
368 continue;
369 }
370 switch (event.key.key)
371 {
372 case SDLK_F1: text(entry[current].label, entry[current].description); break;
373 case SDLK_F2: return true;
374 case SDLK_F3: case SDLK_ESCAPE: return false;
375 case SDLK_UP: focus((current + count - 1) % count); break;
376 case SDLK_DOWN: case SDLK_RETURN: case SDLK_KP_ENTER: focus((current + 1) % count); break;
377 case SDLK_PAGEUP: focus(std::max(0, current - rows)); break;
378 case SDLK_PAGEDOWN: focus(std::min(count - 1, current + rows)); break;
379 case SDLK_TAB:
380 if (choice) cycle((event.key.mod & SDL_KMOD_SHIFT) ? -1 : 1);
381 else focus((current + ((event.key.mod & SDL_KMOD_SHIFT) ? count - 1 : 1)) % count);
382 break;
383 case SDLK_LEFT:
384 if (choice) cycle(-1); else { cursor = previous_char(field.value, cursor); selected = false; }
385 break;
386 case SDLK_RIGHT:
387 if (choice) cycle(1); else { cursor = next_char(field.value, cursor); selected = false; }
388 break;
389 case SDLK_HOME: cursor = 0; selected = false; break;
390 case SDLK_END: cursor = field.value.size(); selected = false; break;
391 case SDLK_BACKSPACE: case SDLK_DELETE:
392 if (!choice)
393 {
394 if (selected) { field.value.clear(); cursor = 0; selected = false; }
395 else if (event.key.key == SDLK_BACKSPACE && cursor)
396 { size_t prev = previous_char(field.value, cursor); field.value.erase(prev, cursor - prev); cursor = prev; }
397 else if (event.key.key == SDLK_DELETE && cursor < field.value.size())
398 field.value.erase(cursor, next_char(field.value, cursor) - cursor);
399 }
400 break;
401 default: break;
402 }
403 }
404 }
