81// NetBSD libform's REQ_END_LINE is bad, puts cursor one beyond. Fix here.
82inline int es40_form_driver(FORM *form, int c)
83{
84 if (c != REQ_END_LINE && c != REQ_END_FIELD)
85 return form_driver(form, c);
86
87 int status = form_driver(form, REQ_BEG_LINE);
88 for (int i = 0; i < 1024 && form_driver(form, REQ_NEXT_CHAR) == E_OK; ++i)
89 ;
90 return status;
91}

171WINDOW *create_window(
172 int nLines, int nCols,
173 int begin_y, int begin_x,
174 const char *title,
175 const char *helptext)
176{
177 if (title != NULL)
178 set_min_width(&nCols, title);
179 if (helptext != NULL)
180 set_min_width(&nCols, helptext);
181
182 if (nLines > scrh)
183 nLines = scrh - 2;
184 if (nCols > scrw)
185 nCols = scrw - 2;
186
187 if (begin_y == -1 && begin_x == -1)
188 {
189 begin_y = (scrh - nLines - 2) / 2;
190 begin_x = (scrw - nCols - 2) / 2;
191 }
192
193 WINDOW *my_win = newwin(nLines + 2, nCols + 2, begin_y, begin_x);
194 box(my_win, 0, 0);
195 keypad(my_win, TRUE);
196
197 if (title != NULL)
198 mvwprintw_center(my_win, 0, nCols, title);
199 if (helptext != NULL)
200 mvwprintw(my_win, nLines + 1, nCols - strlen(helptext), "%s", helptext);
201
202 return my_win;
203}

211void tui_text(const char *title, const char *text)
212{
213 if (text == NULL)
214 return;
215
216 const char *helptext = "Any key to close";
217
218 int nLines = 1;
219 int nCols = 1;
220 char *linebuffer = strdup(text);
221 char *p, *q;
222
223 if (strchr(linebuffer, '\n') == NULL)
224 set_min_width(&nCols, linebuffer);
225 else
226 for (p = linebuffer; (q = strchr(p, '\n')) != NULL; p = q + 1)
227 {
228 ++nLines;
229 *q = '\0';
230 set_min_width(&nCols, p);
231 }
232
233 WINDOW *my_win = create_window(nLines, nCols, -1, -1, title, helptext);
234 PANEL *my_panel = new_panel(my_win);
235
236 p = linebuffer;
237 for (int i = 1; i <= nLines; ++i)
238 {
239 mvwprintw(my_win, i, 1, "%s", p);
240 p += strlen(p) + 1; // Skip NUL terminating part
241 }
242
243 update_panels();
244 doupdate();
245
246 wgetch(my_win);
247
248 // Clean up
249 del_panel(my_panel);
250 wclear(my_win);
251 wrefresh(my_win);
252 delwin(my_win);
253 update_panels();
254 doupdate();
255
256 free(linebuffer);
257}

274int tui_menu(
275 const char *title,
276 MenuEntry_t entry[], int num_entries,
277 int begin_y, int begin_x)
278{
279 const char *helptext = MYKEY_HELP_STR ": help, " MYKEY_EXIT_STR ": exit";
280 char dashes[80] = "";
281
282 int nLines = num_entries;
283 int nCols = 0;
284
285 set_min_width(&nCols, helptext);
286 for (int i = 0; i < num_entries; ++i)
287 set_min_width(&nCols, entry[i].text);
288
289 WINDOW *my_win = create_window(nLines, nCols, begin_x, begin_y, title, helptext);
290 PANEL *my_panel = new_panel(my_win);
291 update_panels();
292
293 ITEM **my_item = (ITEM **)calloc(num_entries + 1, sizeof(ITEM *));
294 for (int i = 0; i < num_entries; ++i)
295 if (entry[i].text != NULL)
296 my_item[i] = new_item((char *)entry[i].text, NULL); // NetBSD libmenu takes non-const char *
297 else
298 {
299 if (dashes[0] == '\0')
300 {
301 size_t len = nCols;
302 if (len > sizeof(dashes))
303 len = sizeof(dashes);
304 memset(dashes, '-', len);
305 dashes[len - 1] = '\0';
306 }
307 my_item[i] = new_item(dashes, NULL);
308 item_opts_off(my_item[i], O_SELECTABLE);
309 }
310 my_item[num_entries] = (ITEM *)NULL;
311
312 MENU *my_menu = new_menu(my_item);
313 set_menu_grey(my_menu, A_NORMAL); // No multi-select, but unselectable items
314 set_menu_win(my_menu, my_win);
315 set_menu_sub(my_menu, derwin(my_win, nLines, nCols, 1, 1));
316#if defined(_WIN32)
317 // NetBSD libmenu defaults to 16 rows and blank-fills rows past the items; with a shorter
318 // subwindow the failed wmove() makes those blanks overwrite the first items.
319 set_menu_format(my_menu, nLines, 1);
320#endif
321 post_menu(my_menu);
322 update_panels();
323 doupdate();
324
325 bool stay_in_loop = TRUE;
326 int cur_idx = 0;
327 while (stay_in_loop)
328 {
329 ITEM *cur = current_item(my_menu);
330 cur_idx = item_index(cur);
331#if defined(_WIN32)
332 // NetBSD libmenu draws into the derwin() without syncing it; PDCurses only refreshes touched windows.
333 touchwin(my_win);
334 wcursyncup(menu_sub(my_menu));
335#endif
336 switch (wgetch(my_win))
337 {
338 case MYKEY_HELP:
339 show_text(entry[cur_idx].text, entry[cur_idx].description);
340 break;
341 case MYKEY_QUIT:
342 cur_idx = MENU_SEL_QUIT;
343 stay_in_loop = FALSE;
344 break;
345 case MYKEY_EXIT:
346 cur_idx = MENU_SEL_EXIT;
347 stay_in_loop = FALSE;
348 break;
349 case KEY_DOWN:
350 if (cur_idx == num_entries - 1)
351 menu_driver(my_menu, REQ_FIRST_ITEM);
352 else
353 menu_driver(my_menu, REQ_DOWN_ITEM);
354 break;
355 case KEY_UP:
356 if (cur_idx == 0)
357 menu_driver(my_menu, REQ_LAST_ITEM);
358 else
359 menu_driver(my_menu, REQ_UP_ITEM);
360 break;
361 case MYKEY_RETURN:
362 if ((item_opts(cur) & O_SELECTABLE) != 0)
363 {
364 SelectFuncPtr callback = entry[cur_idx].select_callback;
365 if (callback == NULL)
366 stay_in_loop = FALSE;
367 else
368 {
369 callback(item_name(cur));
370 pos_menu_cursor(my_menu);
371 }
372 }
373 break;
374 default:
375 break;
376 }
377 }
378
379 // Clean up
380#if defined(_WIN32)
381 // PDCurses' delwin() refuses a window that still has subwindows (and asserts in Debug builds).
382 WINDOW *my_sub = menu_sub(my_menu);
383#endif
384 unpost_menu(my_menu);
385 free_menu(my_menu);
386 del_panel(my_panel);
387 wclear(my_win);
388 wrefresh(my_win);
389#if defined(_WIN32)
390 delwin(my_sub);
391#endif
392 delwin(my_win);
393 update_panels();
394 doupdate();
395
396 for (int i = 0; i < num_entries; ++i)
397 free_item(my_item[i]);
398 free(my_item);
399
400 return cur_idx;
401}

424bool tui_form(const char *title, FormEntry_t entry[], vector<EditorField>& fields)
425{
426 const int count = (int)fields.size();
427 int nLines = count * 2;
428 int nCols = scrw - 20;
429 if (nLines + 2 > scrh)
430 FAILURE_1(Runtime, "Screen is too small for form, need %d lines", nLines + 2);
431 WINDOW *win = create_window(nLines, nCols + 1, -1, -1, title,
432 "F1: help, F2: apply, F3: cancel");
433 PANEL *panel = new_panel(win);
434 vector<FIELD *> native(count + 1, nullptr);
435 // NetBSD libform retains TYPE_ENUM pointers. Keep them alive through free_field().
436 vector<vector<char *>> choices(count);
437 for (int i = 0; i < count; ++i)
438 {
439 auto& rule = fields[i];
440 FIELD *f = native[i] = new_field(1, nCols - 1, 2 * i + 1, 1, 0, 0);
441 set_field_buffer(f, 0, rule.value.c_str());
442 field_opts_off(f, O_AUTOSKIP | O_STATIC);
443 set_max_field(f, 256);
444 set_field_pad(f, ' ');
445 switch (rule.type)
446 {
447 case EditorField::Type::Choice:
448 for (auto& choice : rule.choices)
449 choices[i].push_back(const_cast<char *>(choice.c_str()));
450 choices[i].push_back(nullptr);
451 set_field_type(f, TYPE_ENUM, choices[i].data(), FALSE, TRUE);
452 break;
453 case EditorField::Type::Integer:
454 set_field_type(f, TYPE_INTEGER, rule.precision, (long)rule.minimum, (long)rule.maximum);
455 field_opts_off(f, O_NULLOK);
456 break;
457 case EditorField::Type::Number:
458 set_field_type(f, TYPE_NUMERIC, rule.precision, rule.minimum, rule.maximum);
459 field_opts_off(f, O_NULLOK);
460 break;
461 case EditorField::Type::Pattern:
462 set_field_type(f, TYPE_REGEXP, rule.pattern.c_str());
463 break;
464 default: break;
465 }
466 set_field_back(f, rule.type == EditorField::Type::Choice ? A_NORMAL : A_UNDERLINE);
467 set_field_status(f, TRUE);
468 }
469 FORM *form = new_form(native.data());
470 WINDOW *sub = derwin(win, nLines, nCols, 1, 1);
471 set_form_win(form, win);
472 set_form_sub(form, sub);
473 post_form(form);
474 for (int i = 0; i < count; ++i)
475 mvwprintw(win, 2 * i + 1, 1, "%s", entry[i].label);
476 form_driver(form, REQ_FIRST_FIELD);
477 form_driver(form, REQ_END_FIELD);
478 bool apply = false, done = false;
479 while (!done)
480 {
481 update_panels();
482 doupdate();
483#ifdef _WIN32
484 touchwin(win);
485 wcursyncup(sub);
486#endif
487 FIELD *f = current_field(form);
488 int idx = field_index(f);
489 const bool choice = fields[idx].type == EditorField::Type::Choice;
490 int key = wgetch(win);
491 switch (key)
492 {
493 case MYKEY_HELP: show_text(entry[idx].label, entry[idx].description); break;
494 case MYKEY_QUIT: done = true; break;
495 case MYKEY_EXIT:
496 if (form_driver(form, REQ_VALIDATION) == E_OK) { apply = true; done = true; }
497 else show_text("Error", "Field validation failed");
498 break;
499 case KEY_DOWN: case MYKEY_RETURN:
500 form_driver(form, idx + 1 == count ? REQ_FIRST_FIELD : REQ_NEXT_FIELD);
501 form_driver(form, REQ_END_FIELD); break;
502 case KEY_UP:
503 form_driver(form, idx == 0 ? REQ_LAST_FIELD : REQ_PREV_FIELD);
504 form_driver(form, REQ_END_FIELD); break;
505 case MYKEY_TAB: case KEY_RIGHT:
506 form_driver(form, choice ? REQ_NEXT_CHOICE : (key == KEY_RIGHT ? REQ_NEXT_CHAR : REQ_END_FIELD)); break;
507 case KEY_STAB: case KEY_LEFT:
508 form_driver(form, choice ? REQ_PREV_CHOICE : (key == KEY_LEFT ? REQ_PREV_CHAR : REQ_BEG_FIELD)); break;
509 case KEY_BACKSPACE: case MYKEY_DEL:
510#ifdef _WIN32
511 case 0x08:
512#endif
513 if (!choice) form_driver(form, REQ_DEL_PREV);
514 break;
515 default:
516 if (!choice && key >= 0 && key <= UCHAR_MAX && isprint(key)) form_driver(form, key);
517 break;
518 }
519 }
520 if (apply)
521 for (int i = 0; i < count; ++i)
522 {
523 fields[i].value = field_buffer(native[i], 0);
524 while (!fields[i].value.empty() && fields[i].value.back() == ' ')
525 fields[i].value.pop_back();
526 }
527 unpost_form(form);
528 free_form(form);
529 del_panel(panel);
530 delwin(sub);
531 delwin(win);
532 for (int i = 0; i < count; ++i) free_field(native[i]);
533 update_panels();
534 doupdate();
535 return apply;
536}

542void es40_banner(const char *title)
543{
544 const char *text[] = {
545 title,
546 "Version " VERSION,
547 "",
548 "Copyright (C) 2007-2025 by the ES40 Emulator Project & Others",
549 "Website: https://github.com/ES40-Emu/es40/",
550 "",
551 "This program is free software; you can redistribute it and/or",
552 "modify it under the terms of the GNU General Public License",
553 "as published by the Free Software Foundation; either version 2",
554 "of the License, or (at your option) any later version."};
555 int num_entries = (int)(sizeof(text) / sizeof(text[0]));
556 const char *helptext = "Any key to continue";
557
558 int nLines = num_entries;
559 int nCols = 0;
560
561 for (int i = 0; i < num_entries; ++i)
562 set_min_width(&nCols, text[i]);
563
564 string helptext_dur = string(helptext) + " (" + to_string(WAITSEC) + ")";
565 WINDOW *my_win = create_window(nLines, nCols, -1, -1, NULL, helptext_dur.c_str());
566 PANEL *my_panel = new_panel(my_win);
567
568 for (int i = 0; i < num_entries; ++i)
569 mvwprintw_center(my_win, i + 1, nCols, text[i]);
570
571 update_panels();
572 doupdate();
573
574 const int factor = 10;
575#if !defined(_WIN32)
576 const timespec req = {0, 1000L * 1000 * 1000 / factor}; // = 100,000,000 ns = 0.1 seconds
577#endif
578 int count = WAITSEC * factor;
579 int maxx, maxy;
580 getmaxyx(my_win, maxy, maxx);
581 nodelay(my_win, TRUE);
582 while (--count > 0 && wgetch(my_win) == ERR)
583 {
584#if defined(_WIN32)
585 Sleep(1000 / factor);
586#else
587 nanosleep(&req, NULL);
588#endif
589 if (count % factor == 0)
590 {
591 // No idea why the coordinates need offsets...
592 mvwprintw(my_win, maxy - 1, maxx - 2 - 3, "(%1d)", count / factor);
593 update_panels();
594 doupdate();
595 }
596 }
597 nodelay(my_win, FALSE);
598
599 // Clean up
600 del_panel(my_panel);
601 wclear(my_win);
602 wrefresh(my_win);
603 delwin(my_win);
604 update_panels();
605 doupdate();
606}
