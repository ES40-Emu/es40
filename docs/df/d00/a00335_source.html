92// core definitions
93constexpr attoseconds_t ATTOSECONDS_PER_SECOND_SQRT = 1'000'000'000;
94constexpr attoseconds_t ATTOSECONDS_PER_SECOND = ATTOSECONDS_PER_SECOND_SQRT * ATTOSECONDS_PER_SECOND_SQRT;
95constexpr attoseconds_t ATTOSECONDS_PER_MILLISECOND = ATTOSECONDS_PER_SECOND / 1'000;
96constexpr attoseconds_t ATTOSECONDS_PER_MICROSECOND = ATTOSECONDS_PER_SECOND / 1'000'000;
97constexpr attoseconds_t ATTOSECONDS_PER_NANOSECOND = ATTOSECONDS_PER_SECOND / 1'000'000'000;

99constexpr seconds_t ATTOTIME_MAX_SECONDS = 1'000'000'000;

107// convert between a double and attoseconds
108inline constexpr double ATTOSECONDS_TO_DOUBLE(attoseconds_t x) { return double(x) * 1e-18; }
109inline constexpr attoseconds_t DOUBLE_TO_ATTOSECONDS(double x) { return attoseconds_t(x * 1e18); }
110
111// convert between hertz (as a double) and attoseconds
112inline constexpr double ATTOSECONDS_TO_HZ(attoseconds_t x) { return double(ATTOSECONDS_PER_SECOND) / double(x); }
113template <typename T> inline constexpr attoseconds_t HZ_TO_ATTOSECONDS(T&& x) { return attoseconds_t(ATTOSECONDS_PER_SECOND / x); }
114inline constexpr attoseconds_t HZ_TO_ATTOSECONDS(const XTAL& x) { return attoseconds_t(ATTOSECONDS_PER_SECOND / x); }
115
116// macros for converting other seconds types to attoseconds
117template <typename T> inline constexpr attoseconds_t ATTOSECONDS_IN_SEC(T&& x) { return attoseconds_t(x) * ATTOSECONDS_PER_SECOND; }
118template <typename T> inline constexpr attoseconds_t ATTOSECONDS_IN_MSEC(T&& x) { return attoseconds_t(x) * ATTOSECONDS_PER_MILLISECOND; }
119template <typename T> inline constexpr attoseconds_t ATTOSECONDS_IN_USEC(T&& x) { return attoseconds_t(x) * ATTOSECONDS_PER_MICROSECOND; }
120template <typename T> inline constexpr attoseconds_t ATTOSECONDS_IN_NSEC(T&& x) { return attoseconds_t(x) * ATTOSECONDS_PER_NANOSECOND; }

133 constexpr attotime() noexcept : m_seconds(0), m_attoseconds(0) {}

139
140 // assignment

142 {
143 this->m_seconds = that.m_seconds;
144 this->m_attoseconds = that.m_attoseconds;
145 return *this;
146 }
147
148 // queries
149 constexpr bool is_zero() const noexcept { return (m_seconds == 0 && m_attoseconds == 0); }
151 constexpr bool is_never() const noexcept { return (m_seconds >= ATTOTIME_MAX_SECONDS); }
152
153 // conversion to other forms

157 double as_khz() const noexcept { assert(!is_zero()); return m_seconds == 0 ? double(ATTOSECONDS_PER_MILLISECOND) / double(m_attoseconds) : is_never() ? 0.0 : 1e-3 / as_double(); }
158 double as_mhz() const noexcept { assert(!is_zero()); return m_seconds == 0 ? double(ATTOSECONDS_PER_MICROSECOND) / double(m_attoseconds) : is_never() ? 0.0 : 1e-6 / as_double(); }
159 u64 as_ticks(u32 frequency) const;
160 u64 as_ticks(const XTAL& xtal) const { return as_ticks(xtal.value()); }
162 const char* as_string(int precision = 9) const;
163
165 std::string to_string() const;
166
168 constexpr attoseconds_t attoseconds() const noexcept { return m_attoseconds; }
170 constexpr seconds_t seconds() const noexcept { return m_seconds; }
171
172 static attotime from_double(double _time);
173 static attotime from_ticks(u64 ticks, u32 frequency);
174 static attotime from_ticks(u64 ticks, const XTAL& xtal) { return from_ticks(ticks, xtal.value()); }
176 static constexpr attotime from_seconds(s32 seconds) { return attotime(seconds, 0); }
178 static constexpr attotime from_msec(s64 msec) { return attotime(msec / 1000, (msec % 1000) * (ATTOSECONDS_PER_SECOND / 1000)); }
180 static constexpr attotime from_usec(s64 usec) { return attotime(usec / 1000000, (usec % 1000000) * (ATTOSECONDS_PER_SECOND / 1000000)); }
182 static constexpr attotime from_nsec(s64 nsec) { return attotime(nsec / 1000000000, (nsec % 1000000000) * (ATTOSECONDS_PER_SECOND / 1000000000)); }
184 static attotime from_hz(u32 frequency) { return (frequency > 1) ? attotime(0, HZ_TO_ATTOSECONDS(frequency)) : (frequency == 1) ? attotime(1, 0) : attotime::never; }
185 static attotime from_hz(int frequency) { return (frequency > 0) ? from_hz(u32(frequency)) : attotime::never; }
186 static attotime from_hz(const XTAL& xtal) { return (xtal.dvalue() > 1.0) ? attotime(0, HZ_TO_ATTOSECONDS(xtal)) : from_hz(xtal.dvalue()); }
187 static attotime from_hz(double frequency)
188 {
189 if (frequency > 1.0)
190 return attotime(0, HZ_TO_ATTOSECONDS(frequency));
191 else if (frequency > 0.0)
192 {
193 double i, f = modf(1.0 / frequency, &i);
194 return attotime(i, f * ATTOSECONDS_PER_SECOND);
195 }
196 else
197 return attotime::never;
198 }

222inline constexpr attotime operator+(const attotime& left, const attotime& right) noexcept
223{
224 attotime result;
225
226 // if one of the items is never, return never
227 if (left.m_seconds >= ATTOTIME_MAX_SECONDS || right.m_seconds >= ATTOTIME_MAX_SECONDS)
228 return attotime::never;
229
230 // add the seconds and attoseconds
231 result.m_attoseconds = left.m_attoseconds + right.m_attoseconds;
232 result.m_seconds = left.m_seconds + right.m_seconds;
233
234 // normalize and return
235 if (result.m_attoseconds >= ATTOSECONDS_PER_SECOND)
236 {
237 result.m_attoseconds -= ATTOSECONDS_PER_SECOND;
238 result.m_seconds++;
239 }
240
241 // overflow
242 if (result.m_seconds >= ATTOTIME_MAX_SECONDS)
243 return attotime::never;
244 return result;
245}

272inline constexpr attotime operator-(const attotime& left, const attotime& right) noexcept
273{
274 attotime result;
275
276 // if time1 is never, return never
277 if (left.m_seconds >= ATTOTIME_MAX_SECONDS)
278 return attotime::never;
279
280 // add the seconds and attoseconds
281 result.m_attoseconds = left.m_attoseconds - right.m_attoseconds;
282 result.m_seconds = left.m_seconds - right.m_seconds;
283
284 // normalize and return
285 if (result.m_attoseconds < 0)
286 {
287 result.m_attoseconds += ATTOSECONDS_PER_SECOND;
288 result.m_seconds--;
289 }
290 return result;
291}

293inline attotime& attotime::operator-=(const attotime& right) noexcept
294{
295 // if time1 is never, return never
296 if (this->m_seconds >= ATTOTIME_MAX_SECONDS)
297 return *this = never;
298
299 // add the seconds and attoseconds
300 m_attoseconds -= right.m_attoseconds;
301 m_seconds -= right.m_seconds;
302
303 // normalize and return
304 if (this->m_attoseconds < 0)
305 {
306 this->m_attoseconds += ATTOSECONDS_PER_SECOND;
307 this->m_seconds--;
308 }
309 return *this;
310}

370inline constexpr attoseconds_t attotime::as_attoseconds() const noexcept
371{
372 return
373 (m_seconds == 0) ? m_attoseconds : // positive values between 0 and 1 second
374 (m_seconds == -1) ? (m_attoseconds - ATTOSECONDS_PER_SECOND) : // negative values between -1 and 0 seconds
