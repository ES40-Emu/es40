30constexpr AlphaHWStoreDescriptor ev4_decode_hw_store(std::uint32_t instruction)
31{
32 AlphaHWStoreDescriptor store{};
33 if ((instruction >> 26) != 0x1f)
34 return store;
35
36 // HW_ST uses PHY[15], ALT[14] and QW[12]; RWC[13] is ignored.
37 const unsigned form = (instruction >> 12) & 0xf;
38 const bool physical = (form & 8) != 0;
39 store.size_bits = (form & 1) ? 64 : 32;
40 store.displacement = static_cast<std::int32_t>(instruction & 0xfff);
41 if (store.displacement & 0x800)
42 store.displacement -= 0x1000;
43 store.address = physical ? AlphaHWStoreAddress::Physical : AlphaHWStoreAddress::Virtual;
44 store.alternate = !physical && (form & 4) != 0;
45 store.conditional = physical && (form & 4) != 0;
46 store.compilable = physical && !store.conditional;
47 return store;
48}
