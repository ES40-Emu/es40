31// EV6/EV68 Dcache blocks are 64 bytes; LDx_L/STx_C monitor that cache line.
32inline constexpr std::uint64_t kEv68LLSCMatchMask = 0x00000807ffffffc0;
33inline constexpr std::uint64_t kEv68LLSCIOMask = 0x0000080000000000;
34
35constexpr bool ev68_llsc_matches(std::uint64_t locked_address, std::uint64_t address)
36{
37 return !((locked_address ^ address) & kEv68LLSCMatchMask);
38}
39
40// I/O-space conditional stores have no cache line to watch; treat as held.
41constexpr bool ev68_llsc_unreserved_store(std::uint64_t address)
42{
43 return (address & kEv68LLSCIOMask) != 0;
44}
45
46inline constexpr AlphaLLSCPolicy kEv68LLSCPolicy{
47 &ev68_llsc_matches, &ev68_llsc_unreserved_store
48};
