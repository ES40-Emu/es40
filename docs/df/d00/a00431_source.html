247void vga_device::device_reset()
248{
249 /* clear out the VGA structure */
250 memset(vga.pens, 0, sizeof(vga.pens));
251 vga.miscellaneous_output = 0;
252 m_ioas = false;
253 vga.feature_control = 0;
254 vga.sequencer.index = 0;
255 memset(vga.sequencer.data, 0, sizeof(vga.sequencer.data));
256 vga.crtc.index = 0;
257 memset(vga.crtc.data, 0, sizeof(vga.crtc.data));
258 vga.gc.index = 0;
259 memset(vga.gc.latch, 0, sizeof(vga.gc.latch));
260 memset(&vga.attribute, 0, sizeof(vga.attribute));
261 memset(&vga.dac, 0, sizeof(vga.dac));
262 memset(&vga.cursor, 0, sizeof(vga.cursor));
263 memset(&vga.oak, 0, sizeof(vga.oak));
264
265 vga.gc.memory_map_sel = 0x3; /* prevent xtbios excepting vga ram as system ram */
266/* amstrad pc1640 bios relies on the position of
267 the video memory area,
268 so I introduced the reset to switch to b8000 area */
269 vga.sequencer.data[4] = 0;
270
271 /* TODO: real defaults */
272 vga.crtc.line_compare = 0x3ff;
273 /* skeleton/indiana.cpp boot PROM doesn't set this and assumes it's 0xff */
274 vga.dac.mask = 0xff;
275}

546 u8 res = 0xFF;
547 if (vga.dac.read)
548 {
549 switch (vga.dac.state++)
550 {
551 case 0:
552 res = vga.dac.color[3 * vga.dac.read_index];
553 break;
554 case 1:
555 res = vga.dac.color[3 * vga.dac.read_index + 1];
556 break;
557 case 2:
558 res = vga.dac.color[3 * vga.dac.read_index + 2];
559 break;
560 }
561
562 if (vga.dac.state == 3)
563 {
564 vga.dac.state = 0;
565 vga.dac.read_index++;
566 }
567 }
568 return res;
569}

1259uint8_t vga_device::vga_latch_write(int offs, uint8_t data)
1260{
1261 uint8_t res = 0;
1262
1263 switch (vga.gc.write_mode & 3)
1264 {
1265 case 0:
1266 data = rotate_right(data);
1267 if (vga.gc.enable_set_reset & 1 << offs)
1268 res = vga_logical_op((vga.gc.set_reset & 1 << offs) ? vga.gc.bit_mask : 0, offs, vga.gc.bit_mask);
1269 else
1270 res = vga_logical_op(data, offs, vga.gc.bit_mask);
1271 break;
1272 case 1:
1273 res = vga.gc.latch[offs];
1274 break;
1275 case 2:
1276 res = vga_logical_op((data & 1 << offs) ? 0xff : 0x00, offs, vga.gc.bit_mask);
1277 break;
1278 case 3:
1279 data = rotate_right(data);
1280 res = vga_logical_op((vga.gc.set_reset & 1 << offs) ? 0xff : 0x00, offs, data & vga.gc.bit_mask);
1281 break;
1282 }
1283
1284 return res;

1287void vga_device::vga_vh_text(bitmap_rgb32& bitmap, const rectangle& cliprect)
1288{
1289 // TODO: make it a subclassable function, cfr. PR2 Video Select in Paradise family
1290#define TEXT_COPY_9COLUMN(ch) (((ch & 0xe0) == 0xc0) && (vga.attribute.data[0x10] & 4))
1291
1292 int width = BIT(vga.sequencer.data[1], 0) ? 8 : 9, height = (vga.crtc.maximum_scan_line) * (vga.crtc.scan_doubling + 1);
1293
1294 const u32 frame_number = (u32)(screen().frame_number());
1295
1296 if (vga.crtc.cursor_enable)
1297 vga.cursor.visible = frame_number & 0x10;
1298 else
1299 vga.cursor.visible = 0;
1300
1301 bool blink_rate = frame_number & 0x20;
1302 const int TEXT_LINES = vga.crtc.vert_disp_end + 1;
1303
1304 unsigned addr = 0;
1305 int line = 0;
1306 for (addr = vga.crtc.start_addr, line = -vga.crtc.preset_row_scan; line < TEXT_LINES;
1307 line += height, addr += (offset() >> 1))
1308 {
1309 for (unsigned pos = addr, column = 0; column < vga.crtc.horz_disp_end + 1; column++, pos++)
1310 {
1311 const uint8_t ch = vga.memory[(pos << 1) + 0];
1312 const uint8_t attr = vga.memory[(pos << 1) + 1];
1313 const uint32_t font_base = vga.sequencer.char_sel.base[BIT(attr, 3)] + (ch << 5);
1314 const bool blink_sel = !!BIT(vga.attribute.data[0x10], 3);
1315 const uint8_t blink_en = (blink_sel && blink_rate) ? BIT(attr, 7) : 0;
1316
1317 uint8_t fore_col = attr & 0xf;
1318 uint8_t back_col = (attr & 0x70) >> 4;
1319 // blink disabled translates MSB of attribute as extra intensity for background color
1320 back_col |= (blink_sel) ? 0 : (BIT(attr, 7) >> 4);
1321
1322 for (int h = std::max(-line, 0); (h < height) && (line + h < std::min(TEXT_LINES, bitmap.height())); h++)
1323 {
1324 uint32_t* const bitmapline = &bitmap.pix(line + h);
1325 uint8_t bits = vga.memory[font_base + (h >> (vga.crtc.scan_doubling))];
1326
1327 int mask, w;
1328 for (mask = 0x80, w = 0; (w < width) && (w < 8); w++, mask >>= 1)
1329 {
1330 pen_t pen;
1331 if (bits & mask)
1332 pen = vga.pens[blink_en ? back_col : fore_col];
1333 else
1334 pen = vga.pens[back_col];
1335
1336 const int dest_x = column * width + w;
1337 if (!screen().visible_area().contains(dest_x, line + h))
1338 continue;
1339 bitmapline[dest_x] = pen;
1340
1341 }
1342 if (w < width)
1343 {
1344 /* 9 column */
1345 pen_t pen;
1346 if (TEXT_COPY_9COLUMN(ch) && (bits & 1))
1347 pen = vga.pens[blink_en ? back_col : fore_col];
1348 else
1349 pen = vga.pens[back_col];
1350
1351 const int dest_x = column * width + w;
1352
1353 if (!screen().visible_area().contains(dest_x, line + h))
1354 continue;
1355 bitmapline[dest_x] = pen;
1356 }
1357 }
1358 if (vga.cursor.visible && (pos == vga.crtc.cursor_addr))
1359 {
1360 for (int h = vga.crtc.cursor_scan_start;
1361 (h <= vga.crtc.cursor_scan_end) && (h < height) && (line + h < TEXT_LINES);
1362 h++)
1363 {
1364 const int dest_x = column * width;
1365
1366 if (!screen().visible_area().contains(dest_x, line + h))
1367 continue;
1368 bitmap.plot_box(dest_x, line + h, width, 1, vga.pens[attr & 0xf]);
1369 }
1370 }
1371 }
1372 }
1373}

1398 vga.memory[(pos & 0xffff) + 0x10000] << 1,

1400 vga.memory[(pos & 0xffff) + 0x30000] << 3 };
1401

1408 data[2] >>= 1;
1409 data[3] >>= 1;

1415 }
1416 }
