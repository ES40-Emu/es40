605void CSym53C895::run()
606{
607 try
608 {
609 for (;;)
610 {
611 mySemaphore.wait();
612 if (StopThread)
613 return;
614 for (;;)
615 {
616 // Peer DMA can enter another device: take the bus before the register lock.
617 std::lock_guard<std::recursive_mutex> bus_lock(cSystem->get_device_bus_mutex());
618 CScopedLock<CMutex> regLock(myRegLock);
619 if (!state.executing)
620 break;
621 execute();
622 }
623 }
624 }
625
626 catch (CException& e)
627 {
628 printf("Exception in SYM thread: %s.\n", e.displayText().c_str());
629
630 // Let the thread die...
631 }
632}

1763void CSym53C895::execute_bm_op()
1764{
1765 bool indirect = (R8(DCMD) >> 5) & 1;
1766 bool table_indirect = (R8(DCMD) >> 4) & 1;
1767 int opcode = (R8(DCMD) >> 3) & 1;
1768 int scsi_phase = (R8(DCMD) >> 0) & 7;
1769
1770#if defined(DEBUG_SYM_SCRIPTS)
1771 printf("SYM: INS = Block Move (i %d, t %d, opc %d, phase %d\n", indirect,
1772 table_indirect, opcode, scsi_phase);
1773#endif
1774 // Compare phase
1775 if (check_phase(scsi_phase) > 0)
1776 {
1777#if defined(DEBUG_SYM_SCRIPTS)
1778 printf("SYM: Ready for transfer.\n");
1779#endif
1780
1781 u32 start;
1782 u32 count;
1783
1784 if (table_indirect)
1785 {
1786 u32 add = R32(DSA) + sext_u32_24(R32(DSPS));
1787
1788 add &= ~0x03; // 810
1789#if defined(DEBUG_SYM_SCRIPTS)
1790 printf("SYM: Reading table at DSA(%08x)+DSPS(%08x) = %08x.\n",
1791 R32(DSA), R32(DSPS), add);
1792#endif
1793 do_pci_read(add, &count, 4, 1);
1794 count &= 0x00ffffff;
1795 do_pci_read(add + 4, &start, 4, 1);
1796 }
1797 else if (indirect)
1798 {
1799 FAILURE(NotImplemented, "SYM: Unsupported: indirect addressing");
1800 }
1801 else
1802 {
1803 start = R32(DSPS);
1804 count = GET_DBC();
1805 }
1806
1807#if defined(DEBUG_SYM_SCRIPTS)
1808 printf("SYM: %08x: MOVE Start/count %x, %x\n", R32(DSP) - 8, start,
1809 count);
1810#endif
1811 R32(DNAD) = start;
1812 SET_DBC(count); // page 5-32
1813 if (count == 0)
1814 {
1815
1816 //printf("SYM: Count equals zero!\n");
1817 RAISE(DSTAT, IID); // page 5-32
1818 return;
1819 }
1820
1821 if ((size_t)count > scsi_expected_xfer(0))
1822 {
1823#if defined(DEBUG_SYM_SCRIPTS)
1824 printf("SYM: xfer %d bytes, max %d expected, in phase %d.\n", count,
1825 scsi_expected_xfer(0), scsi_phase);
1826#endif
1827 count = (u32)scsi_expected_xfer(0);
1828 }
1829
1830 u8* scsi_data_ptr = (u8*)scsi_xfer_ptr(0, count);
1831 u8* org_sdata_ptr = scsi_data_ptr;
1832
1833 switch (scsi_phase)
1834 {
1835 case SCSI_PHASE_COMMAND:
1836 case SCSI_PHASE_DATA_OUT:
1837 case SCSI_PHASE_MSG_OUT:
1838 do_pci_read(R32(DNAD), scsi_data_ptr, 1, count);
1839 R32(DNAD) += count;
1840 break;
1841
1842 case SCSI_PHASE_STATUS:
1843 case SCSI_PHASE_DATA_IN:
1844 case SCSI_PHASE_MSG_IN:
1845 do_pci_write(R32(DNAD), scsi_data_ptr, 1, count);
1846 R32(DNAD) += count;
1847 break;
1848 }
1849
1850 R8(SFBR) = *org_sdata_ptr;
1851 scsi_xfer_done(0);
1852 return;
1853 }

1856/* Execute one SCRIPTS I/O instruction
1857 *
1858 * The I/O instructions perform common SCSI hardware sequences, like
1859 * Selection and reselection. The instruction format in
1860 * the DCMD and DBC register is as follows:
1861 *
1862 * \code
1863 * +---+-----+-+-+-+
1864 * |7 6|5 4 3|2|1|0| DCMD Register
1865 * +---+-----+-+-+-+
1866 * | | | | +- 0: Select with ATN/:
1867 * | | | | Valid only for Select instruction. Assert ATN/ during
1868 * | | | | selection
1869 * | | | +- 1: Table Indirect Mode:
1870 * | | | 0: All information is taken from the instruction,
1871 * | | | and the contents of the SCNTL3 and SXFER registers.
1872 * | | | 1: The DSPS register contains a 24-bit signed offset
1873 * | | | that is added to the DSA register to get a pointer
1874 * | | | to a data structure that contains the destination
1875 * | | | ID, SCNTL3 bits, and SXFER bits. This structure
1876 * | | | is 32 bits long and looks as follows:
1877 * | | | +--------+--------+--------+--------+
1878 * | | | DSA+DSPS: | SCNTL3 | ID | SXFER | |
1879 * | | | +--------+--------+--------+--------+
1880 * | | +- 2: Relative Addrressing:
1881 * | | 0: The value in the DNAD register is an absolute address.
1882 * | | 1: The value in the DNAD register is a 24-bit signed
1883 * | | displacement from the current DSP address.
1884 * | +- 3..5: Op Code
1885 * +- 6..7 Instruction Type: 01 = I/O
1886 *
1887 * +-------+-------++--------+--+-+-++-+-+---+-+-----+
1888 * | |19 16|| |10|9| || |6| |3| | DBC Register
1889 * +-------+-------++--------+--+-+-++-+-+---+-+-----+
1890 * | | | | +- 3: Set/Clear ATN
1891 * | | | +- 6: Set/Clear ACK
1892 * | | +- 9: Set/Clear Target Mode
1893 * | +- 10: Set/Clear Carry
1894 * +- 16..19: Destination ID
1895 * \endcode
1896 *
1897 * The Opcode determines the actual instruction:
1898 * \code
1899 * +-----+-----------------+-------------+
1900 * | OPC | Initiator mode | Target Mode |
1901 * +-----+-----------------+-------------+
1902 * | 000 | Select | Reselect |
1903 * | 001 | Wait Disconnect | Disconnect |
1904 * | 010 | Wait Reselect | Wait Select |
1905 * | 011 | Set | Set |
1906 * | 100 | Clear | Clear |
1907 * +-----+-----------------+-------------+
1908 * \endcode
1909 *
1910 * Select:
1911 * - Arbitrate for the SCSI bus until arbitration is won.
1912 * - If arbitration is won, try to select the destination ID
1913 * - If the controller is selected or reselected before winning
1914 * arbitration, jump to the address in the DNAD register.
1915 * .
1916 *
1917 * Wait Disconnect:
1918 * - Wait for the target to disconnect from the SCSI bus.
1919 * .
1920 *
1921 * Wait Reselect:
1922 * - If the controller is reselected, go to the next instruction.
1923 * - If the controller is selected before being reselected, or if
1924 * the CPU sets the SIGP flag, jump to the address in the DNAD
1925 * register
1926 * .
1927 *
1928 * Set/Clear:
1929 * - Set or Clear the flags whose Set/Clear bits are set in the
1930 * instruction.
1931 * .
1932 **/

1951
1952#if defined(DEBUG_SYM_SCRIPTS)

1954 opcode, relative, table_indirect, atn, destination, sc_carry,

2050
2051 case 4:

2789 // (When an interrupt is signalled, but another interrupt bit is already active,
2790 // the interrupt doesn't go to the interrupt register, but to the interrupt
2791 // stack. The interrupt stack, however, doesn't keep track of the order in which
2792 // interrupts come in, so when the interrupt stack is moved down into the
2793 // interrupt registers, multiple interrupt bits may become active.)
2794 if (!R8(SIST0) && !R8(SIST1) && !R8(DSTAT))
2795 {
2796 R8(SIST0) |= state.sist0_stack;
2797 R8(SIST1) |= state.sist1_stack;
2798 R8(DSTAT) |= state.dstat_stack;
2799 state.sist0_stack = 0;
2800 state.sist1_stack = 0;
2801 state.dstat_stack = 0;
2802 }
2803
2804 // Check for DMA interrupts.
2805 if (R8(DSTAT) & DSTAT_FATAL)
2806 {
2807 // DMA interrupt conditions always halt execution (always fatal)
2808 will_halt = true;
2809
2810 //printf(" will halt(DSTAT).\n");
2811
2812 // Set the DMA interrupt pending bit.
2813 SB_R8(ISTAT, DIP, true);
2814
2815 // If the interrupt is also enabled in the DIEN register, it will
2816 // be signalled to the system.
2817 if (R8(DSTAT) & R8(DIEN) & DSTAT_FATAL)
2818 {
2819 will_assert = true;
2820
2821 //printf(" will assert(DSTAT).\n");
2822 }
2823 }
2824 else
2825 {
2826 // Reset the DMA interrupt pending bit. (It may still be set).
2827 SB_R8(ISTAT, DIP, false);
2828 }
2829
2830 // Check for SCSI engine interrupts
2831 if (R8(SIST0) || R8(SIST1))
2832 {
2833 // Set the SCSI interrupt pending bit.
2834 SB_R8(ISTAT, SIP, true);
2835
2836 //Check if the interrupt is either fatal, or enabled.
2837 if ((R8(SIST0) & (SIST0_FATAL | R8(SIEN0)))
2838 || (R8(SIST1) & (SIST1_FATAL | R8(SIEN1))))
2839 {
2840 // In either case, stop execution.
2841 will_halt = true;
2842
2843 //printf(" will halt(SIST).\n");
2844
2845 // If the interrupt is enabled, signal it to the system.
2846 if ((R8(SIST0) & R8(SIEN0)) || (R8(SIST1) & R8(SIEN1)))
2847 {
2848 will_assert = true;
2849
2850 //printf(" will assert(SIST).\n");
2851 }
2852 }
2853 }
2854 else
2855 {
2856 // Reset the SCSI interrupt pending bit (It may still be set).
2857 SB_R8(ISTAT, SIP, false);
2858 }
2859
2860 // Check the interrupt on the fly bit.
2861 if (TB_R8(ISTAT, INTF))
2862 {
2863 // Signal this to the system
2864 will_assert = true;
2865
2866 //printf(" will assert(INTF).\n");
2867 }
2868
2869 // If interrupts are disabled, don't signal any interrupt to the system.
2870 if (TB_R8(DCNTL, IRQD))
2871 {
2872 will_assert = false;
2873
2874 //printf(" won't assert(IRQD).\n");
2875 }
2876
2877 // Halt execution if will_halt is true.
2878 if (will_halt)
2879 state.executing = false;
2880
2881 // Assert or de-assert the interrupt line as needed
2882 if (will_assert != state.irq_asserted)
2883 {
2884
2885 //printf(" doing...%d\n",will_assert);
2886 do_pci_interrupt(0, will_assert);
2887 state.irq_asserted = will_assert;
2888 }
2889}
