85u8 CSystem::tig_read(u32 a)
86{
87 switch (a)
88 {
89 case 0x30000000: // trr
90 return 0;
91 case 0x30000040: // smir
92 return state.tig.FwWrite;
93 case 0x30000100: // mod_info
94 return state.tig.ModInfo;
95 case 0x300003c0: // ttcr
96 return state.tig.HaltA;
97 case 0x30000440: // clr_irq4: IRQ4 latch clear; our IRQ4 is level-driven
98 case 0x30000480: // clr_pwr_flt_det
99 return 0;
100 case 0x300005c0: // ev6_halt
101 return state.tig.HaltB;
102 case 0x30000a00: // ipcr0-4: PAL MP-restart handshake
103 case 0x30000a40:
104 case 0x30000a80:
105 case 0x30000ac0:
106 case 0x30000b00:
107 return state.tig.ipcr[(a - 0x30000a00) >> 6];
108 case 0x38000180: // Arbiter revision
109 return 0xfe;
110 default: printf("Unknown TIG %08x read attempted.\n", a); return 0;
111 }

129void CSystem::tig_write(u32 a, u8 data)
130{
131 switch (a)
132 {
133 case 0x30000000: // trr
134 return;
135
136 case 0x30000040: // smir
137 state.tig.FwWrite = data;
138 return;
139
140 case 0x30000100: // mod_info
141 state.tig.ModInfo = data;
142 return;
143
144 case 0x300003c0: // ttcr
145 state.tig.HaltA = data;
146 tig_update_halt_lines();
147 return;
148
149 case 0x30000440: // clr_irq4: IRQ4 latch clear; our IRQ4 is level-driven
150 case 0x30000480: // clr_pwr_flt_det
151 return;
152
153 case 0x300005c0: // ev6_halt
154 state.tig.HaltB = data;
155 tig_update_halt_lines();
156 return;
157
158 case 0x30000600: // srcr0
159 case 0x30000640: // srcr1
160 // Empirical: LFU writes 0x30 here when exiting after an update.
161 if (data & 0x30)
162 {
163 printf("%%SYS-I-RESETREQ: TIG SRCR write %07x=%02x\n", a, data);
164 if (theSROM)
165 theSROM->FlushIfDirty();
166 RequestSystemReset();
167 }
168 return;
169
170 case 0x30000a00: // ipcr0-4: PAL MP-restart handshake registers
171 case 0x30000a40:
172 case 0x30000a80:
173 case 0x30000ac0:
174 case 0x30000b00:
175 state.tig.ipcr[(a - 0x30000a00) >> 6] = data;
176 return;
177
178 default:
179 printf("Unknown TIG %07x write with %02x attempted.\n", a, data);
180 }
181}

185void CSystem::init_spd_from_config_mb(uint32_t total_mb)
186{
187 // Sets of 4 identical DIMMs fill one MMB's slot set (J1-4, then J5-8);
188 // each populated MMB is one array (8 DIMMs max = Typhoon ASIZ max 8GB).
189 // DIMM size = total/4 capped at 1GB, so every config is a 4-stick set
190 // minimum and SRM reports 4-way interleave; >4GB spills to more sets.
191 m_dimm_layout.dimm_mb = (total_mb / 4 > 1024) ? 1024 : total_mb / 4;
192 uint32_t n_dimms = total_mb / m_dimm_layout.dimm_mb;
193 m_dimm_layout.n_arrays = (int)((n_dimms + 7) / 8);
194 m_dimm_layout.dimms_per_array = (int)(n_dimms / m_dimm_layout.n_arrays);
195 m_dimm_spd = build_sdram_spd(m_dimm_layout.dimm_mb, /*registered_ecc*/true);
196
197 // The I2C bus can't carry every DIMM (HRM 9.10): attach one
198 // representative EEPROM per 4-DIMM set, at 0x50 + array*2 + set.
199 for (int a = 0; a < m_dimm_layout.n_arrays; a++)
200 for (int s = 0; s < m_dimm_layout.dimms_per_array / 4; s++)
201 m_mpd_bus.attach(std::make_shared<Eeprom24C02>(uint8_t(0x50 + a * 2 + s), m_dimm_spd));
202}
