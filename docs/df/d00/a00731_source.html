121void CSystem::pio_write(u64 a, int dsize, u64 data, CSystemComponent* source)
122{
123 std::lock_guard<std::recursive_mutex> bus_lock(device_bus_mutex);
124 SDeviceAccessScope time_scope(this);
125 SDecodeClaims claims;
126 collect_decode_claims(a, dsize, true, claims, source);
127 const bool mapped = claims.count != 0;
128 if (claims.count > 1)
129 prepare_shared_access(a, dsize, true, claims, source);
130 if (mapped)
131 note_pio_access(a, dsize, true, data, claims.count);
132 const u64 io_base = a & ~U64(0x1ffffff);
133
134 const bool observed_io =
135 (io_base == U64(0x801fc000000) || io_base == U64(0x803fc000000)) &&
136 (dsize == 8 || dsize == 16 || dsize == 32) &&
137 (a & 3) + dsize / 8 <= 4;
138 if (observed_io)
139 dispatch_pci_io_write(a, dsize, data, claims);
140 else
141 for (int k = 0; k < claims.count; ++k)
142 {
143 const SMemoryUser* r = &claims.range[k];
144 r->component->WriteMem(r->index, a - r->base, dsize, data);
145 }
146 if (mapped)
147 return;
148
149 const u64 config_base = a & ~U64(0xffffff);
150 if ((config_base == U64(0x801fe000000) || config_base == U64(0x803fe000000)) &&
151 (dsize == 8 || dsize == 16 || dsize == 32 || dsize == 64) &&
152 (a & (dsize / 8 - 1)) == 0)
153 {
154 // No function claimed this configuration write. Even a burst aborts once.
155 pchip_config_write_abort(config_base == U64(0x803fe000000) ? 1 : 0,
156 (u32)a & 0xffffff);
157 return;
158 }
159
160 if (a >= U64(0x00000801A0000000) && a <= U64(0x00000801AFFFFFFF))
161 {
162 cchip_csr_write((u32)a & 0xFFFFFFF, data, source);
163 return;
164 }
165
166 if (a >= U64(0x0000080180000000) && a <= U64(0x000008018FFFFFFF))
167 {
168 pchip_csr_write(0, (u32)a & 0xFFFFFFF, data);
169 return;
170 }
171
172 if (a >= U64(0x0000080380000000) && a <= U64(0x000008038FFFFFFF))
173 {
174 pchip_csr_write(1, (u32)a & 0xFFFFFFF, data);
175 return;
176 }
177
178 if (a >= U64(0x00000801B0000000) && a <= U64(0x00000801BFFFFFFF))
179 {
180 dchip_csr_write((u32)a & 0xFFFFFFF, (u8)data & 0xff);
181 return;
182 }
183
184 if (a >= U64(0x0000080100000000) && a <= U64(0x000008013FFFFFFF))
185 {
186 tig_write((u32)a & 0x3FFFFFFF, (u8)data);
187 return;
188 }
189
190 if (a >= U64(0x801fc000000) && a < U64(0x801fe000000))
191 {
192 const u64 io_port = a & U64(0x1ffffff);
193
194 if (io_port == U64(0x100))
195 {
196 if (!m_reported_3c509_probe.exchange(true, std::memory_order_acq_rel))
197 printf("%%ISA-I-ELINKPROBE: 3c509 not implemented; ignoring likely BSD probe at IO port 100.\n");
198 return;
199 }
200
201 // OpenVMS's floppy path retains an Intel 82357/EISA high-count-byte
202 // clear even on the M1543C non-EISA machine.
203 // The M1543C does not document port 405h so we can safely swallow
204 if (io_port == U64(0x405) && dsize == 8 && (data & 0xff) == 0)
205 return;
206
207 // VGA BIOS second-adapter probe: cursor-location writes to the CRTC
208 // block its own card is not using; the M1543C sends them to an empty ISA bus.
209 if ((io_port == U64(0x3b4) || io_port == U64(0x3d4)) &&
210 (dsize == 8 || dsize == 16) && ((data & 0xfe) == 0x0e))
211 return;
212
213 // Unused PCI I/O space
214 if (source)
215 printf("Write to unknown IO port %" PRIx64 " (dsize=%d data=%" PRIx64 ") on PCI 0 from %s\n",
216 io_port, dsize, data, source->devid_string);
217 else
218 printf("Write to unknown IO port %" PRIx64 " (dsize=%d data=%" PRIx64 ") on PCI 0\n",
219 io_port, dsize, data);
220 return;
221 }
222
223 if (a >= U64(0x803fc000000) && a < U64(0x803fe000000))
224 {
225
226 // Unused PCI I/O space
227 if (source)
228 {
229 printf("Write to unknown IO port %" PRIx64 " on PCI 1 from %s \n",
230 a & U64(0x1ffffff), source->devid_string);
231 }
232 else
233 printf("Write to unknown IO port %" PRIx64 " on PCI 1 \n",
234 a & U64(0x1ffffff));
235 return;
236 }
237
238 if (a >= U64(0x80000000000) && a < U64(0x80100000000))
239 {
240
241 // Unused PCI memory space
242 u64 paddr = a & U64(0xffffffff);
243 if (paddr > 0xb8fff || paddr < 0xb8000)
244 { // skip legacy video
245#ifdef DEBUG_UNKMEM
246 if (source)
247 {
248 printf("Write to unknown memory %" PRIx64 " on PCI 0 from %s \n",
249 a & U64(0xffffffff), source->devid_string);
250 }
251 else
252 printf("Write to unknown memory %" PRIx64 " on PCI 0 \n",
253 a & U64(0xffffffff));
254#endif
255 }
256 }
257
258 if (a >= U64(0x80200000000) && a < U64(0x80300000000))
259 {
260#ifdef DEBUG_UNKMEM
261 // Unused PCI memory space
262 if (source)
263 {
264 printf("Write to unknown memory %" PRIx64 " on PCI 1 from %s \n",
265 a & U64(0xffffffff), source->devid_string);
266 }
267 else
268 printf("Write to unknown memory %" PRIx64 " on PCI 1 \n",
269 a & U64(0xffffffff));
270#endif
271 return;
272 }
273
274#ifdef DEBUG_UNKMEM
275 if (source)
276 printf("Write to unknown memory %" PRIx64 " from %s \n", a,
277 source->devid_string);
278 else
279 printf("Write to unknown memory %" PRIx64 " \n", a);
280#endif //defined(DEBUG_UNKMEM)
281}

895// Tsunami HRM Table 10-42: bit 9 is reserved, unlike PERRMASK<9>.
896static const u64 pchip_error_flags = U64(0xdff);
897static const u64 pchip_error_info = U64(0xffffffffffff0000);
898
899void CSystem::latch_pchip_error(int num, u64 data)
900{
901 const u64 flags = data & state.pchip[num].perrmask & pchip_error_flags;
902 if (!flags)
903 return;
904 if (state.pchip[num].perr & pchip_error_flags)
905 state.pchip[num].perr |= state.pchip[num].perrmask & U64(1); // LOST
906 else
907 state.pchip[num].perr = (data & pchip_error_info) | flags;
908 update_pchip_error_irqs();
909}

911void CSystem::pchip_config_write_abort(int num, u32 address)
912{
913 u32 pci_address = address & 0xfffffc;
914 if (!(address & 0xff0000))
915 {
916 // HRM 8.5 / Table 10-3: Type 0 uses one-hot IDSEL, not the device number.
917 const unsigned device = (address >> 11) & 31;
918 if (device == 21) // Table 10-3 omits this encoding; leave it unmodeled.
919 return;
920 pci_address = address & 0x7fc;
921 if (device <= 20)
922 pci_address |= 1U << (device + 11);
923 // Devices 22-31 have no IDSEL asserted in Table 10-3.
924 }
925 // HRM 8.8.2.1 / Table 10-42: configuration write CMD=B, NDS, INV=0,
926 // and PCI AD<31:2> in PERROR<47:18>. This is not a dual-address cycle.
927 latch_pchip_error(num, (U64(0xb) << 52) | ((u64)pci_address << 16) | U64(0x100));
928}

930u64 CSystem::pchip_csr_read(int num, u32 a)
931{
932 switch (a)
933 {
934 case 0x000:
935 case 0x040:
936 case 0x080:
937 case 0x0c0:
938 return state.pchip[num].wsba[(a >> 6) & 3];
939
940 case 0x100:
941 case 0x140:
942 case 0x180:
943 case 0x1c0:
944 return state.pchip[num].wsm[(a >> 6) & 3];
945
946 case 0x200:
947 case 0x240:
948 case 0x280:
949 case 0x2c0:
950 return state.pchip[num].tba[(a >> 6) & 3];
951
952 case 0x300:
953 return state.pchip[num].pctl;
954
955 case 0x340:
956 return state.pchip[num].plat;
957
958 case 0x3c0:
959 return state.pchip[num].perr & (pchip_error_info | pchip_error_flags);
960
961 case 0x400: // PERRMASK; bits 63:12 are RAZ (HRM Table 10-43).
962 return state.pchip[num].perrmask & U64(0xfff);
963
964 case 0x480: // TLBIV
965 case 0x4c0: // TLBIA
966 return 0;
967
968 case 0x800: // PCI reset
969 return 0;
970
971 // HRM 10.2.5.8, Table 10-44 (p. 10-52): PERRSET is WO;
972 // read not defined, return 0.
973 case 0x440: // PERRSET
974 return 0;
975
976 // HRM 10.2.5.11 (p. 10-53): PMONCTL is RW.
977 // HRM 10.2.5.12 (p. 10-54): PMONCNT is RO.
978 // Not emulated; return power-on default of 0.
979 case 0x500: // PMONCTL
980 case 0x540: // PMONCNT
981 return 0;
982
983 default:
984 printf("Unknown PCHIP %d CSR %07x read attempted.\n", num, a);
985 return 0;
986 }

994void CSystem::pchip_csr_write(int num, u32 a, u64 data)
995{
996 switch (a)
997 {
998 case 0x000:
999 case 0x040:
1000 case 0x080:
1001 state.pchip[num].wsba[(a >> 6) & 3] = data & U64(0x00000000fff00003);
1002 return;
1003
1004 case 0x0c0:
1005 // HRM 10.2.5.1, Table 10-36 "Window Space Base Address Register (WSBA3)",
1006 // p. 10-46:
1007 // DAC <39> RW "DAC enable"
1008 // ADDR <31:20> RW "Base address if DAC enable = 0"
1009 // SG <1> RO Init=1, "Scatter-gather always enabled"
1010 // ENA <0> RW "Enable"
1011 //
1012 // Bit 1 must not be writable; force it to 1.
1013 state.pchip[num].wsba[3] = (data & U64(0x00000080fff00001)) | U64(0x2);
1014 return;
1015
1016 case 0x100:
1017 case 0x140:
1018 case 0x180:
1019 case 0x1c0:
1020 state.pchip[num].wsm[(a >> 6) & 3] = data & U64(0x00000000fff00000);
1021 return;
1022
1023 case 0x200:
1024 case 0x240:
1025 case 0x280:
1026 case 0x2c0:
1027 state.pchip[num].tba[(a >> 6) & 3] = data & U64(0x00000007fffffc00);
1028 return;
1029
1030 case 0x300:
1031 // HRM 10.2.5.4, Table 10-40 "Pchip Control Register (PCTL)", pp. 10-47..10-49:
1032 //
1033 // RO fields (preserve on write):
1034 // PID <47:46> "Pchip ID. Initialized from PID pins."
1035 // RPP <45> "Remote Pchip present."
1036 // PCLKX <41:40> "PCI clock frequency multiplier."
1037 // REV <31:24> "Indicates the revision of the Pchip (see §8.10)."
1038 //
1039 // RW fields (accept writes):
1040 // PTEVRFY <44> "PTE verify for DMA read."
1041 // FDWDIS <43> "Fast DMA read cache block wrap request disable."
1042 // FDSDIS <42> "Fast DMA start and SGTE request disable."
1043 // PTPMAX <39:36> "Max PTP requests to Cchip, modulo 16."
1044 // CRQMAX <35:32> "Max requests to Cchip until Ack, modulo 16."
1045 // CDQMAX <23:20> "Max data transfers to Dchips until Ack, modulo 16."
1046 // PADM <19> "PADbus mode."
1047 // ECCEN <18> "ECC enable for DMA and SGTE accesses."
1048 // RES <17:16> MBZ,RAZ — written as zero.
1049 // PPRI <15> "Arbiter priority group for the Pchip."
1050 // PRIGRP <14:8> "Arbiter priority group; one bit per PCI slot."
1051 // ARBENA <7> "Internal arbiter enable."
1052 // MWIN <6> "Monster window enable."
1053 // HOLE <5> "512KB-to-1MB window hole enable."
1054 // TGTLAT <4> "Target latency timers enable."
1055 // CHAINDIS <3> "Disable chaining."
1056 // THDIS <2> "Disable antithrash mechanism for TLB."
1057 // FBTB <1> "Fast back-to-back enable."
1058 // FDSC <0> "Fast discard enable."
1059 //
1060 // Preserve mask = bits {63:48(RAZ), 47:45, 41:40, 31:24}
1061 // Writable mask = bits {44:42, 39:32, 23:18, 15:0}
1062
1063 state.pchip[num].pctl &= U64(0xffffe300ff000000); // preserve RO per Table 10-40
1064 state.pchip[num].pctl |= (data & U64(0x00001cff00fcffff)); // write RW per Table 10-40
1065 return;
1066
1067 case 0x340:
1068 state.pchip[num].plat = data;
1069 return;
1070
1071 case 0x3c0: // PERROR; flags are W1C, INFO is read-only.
1072 state.pchip[num].perr &= ~(data & pchip_error_flags);
1073 update_pchip_error_irqs();
1074 return;
1075
1076 case 0x400: // PERRMASK; only MASK<11:0> is writable.
1077 state.pchip[num].perrmask = data & U64(0xfff);
1078 return;
1079
1080 case 0x440: // PERRSET (HRM 10.2.5.8).
1081 // Table 10-44 copies all INFO bits, including software-supplied INV.
1082 latch_pchip_error(num, data);
1083 return;
1084
1085 case 0x480: // TLBIV
1086 case 0x4c0: // TLBIA
1087 return;
1088
1089 case 0x800: // PCI reset
1090 // Tsunami HRM 8.7: each Pchip resets only its own PCI bus.
1091 for (int i = 0; i < iNumComponents; i++)
1092 {
1093 auto* device = dynamic_cast<CPCIDevice*>(acComponents[i]);
1094 if (device && device->pci_bus() == num)
1095 device->ResetPCI();
1096 }
1097 return;
1098
1099 default:
1100 printf("Unknown PCHIP %d CSR %07x write with %016" PRIx64 " attempted.\n", num,
1101 a, data);
1102 }

1105u64 CSystem::cchip_csr_read(u32 a, CSystemComponent* source)
1106{
1107 CAlphaCPU* cpu = (CAlphaCPU*)source;
1108 switch (a)
1109 {
1110 case 0x000:
1111 return state.cchip.csc;
1112
1113 case 0x080:
1114 // printf("MISC: %016" PRIx64 " from CPU %d (@%" PRIx64 ") (other @ %" PRIx64 ").\n",state.cchip.misc | cpu->get_cpuid(),cpu->get_cpuid(), cpu->get_pc()-4, acCPUs[1-cpu->get_cpuid()]->get_pc());
1115 {
1116 // Consistent read of MISC against the concurrent RMW in cchip_csr_write()/clear_clock_int().
1117 std::lock_guard<std::mutex> g(drir_lock);
1118 return state.cchip.misc | ((CAlphaCPU*)source)->get_cpuid();
1119 }
1120
1121 case 0x0c0: { // MPD: bit3=DR (SDA read), bit2=CKR (SCL read), bits1:0 read as 0
1122 uint8_t v = 0;
1123 v |= (m_mpd_bus.scl() ? 1 : 0) << 2; // CKR
1124 v |= (m_mpd_bus.sda() ? 1 : 0) << 3; // DR
1125 return v;
1126 }
1127
1128 case 0x100:
1129 case 0x140:
1130 case 0x180:
1131 case 0x1c0:
1132 {
1133 // AAR0-3: memory presented as up to 4 arrays of at most 8GB each
1134 // (Typhoon ASIZ max, HRM Table 10-15); base address in ADDR<34:24>.
1135 int arr = (int)((a >> 6) & 3);
1136 unsigned int arr_bits = (iNumMemoryBits > 33) ? 33 : iNumMemoryBits;
1137 int n_arr = 1 << (iNumMemoryBits - arr_bits);
1138 if (arr >= n_arr)
1139 return 0; // bank disabled
1140 return ((u64)arr << arr_bits) | ((u64)(arr_bits - 23) << 12);
1141 }
1142
1143 case 0x200:
1144 case 0x240:
1145 case 0x600:
1146 case 0x640:
1147 return state.cchip.dim[((a >> 9) & 2) | ((a >> 6) & 1)];
1148
1149 case 0x280:
1150 case 0x2c0:
1151 case 0x680:
1152 case 0x6c0:
1153 return state.cchip.drir & state.cchip.dim[((a >> 9) & 2) | ((a >> 6) & 1)];
1154
1155 case 0x300:
1156 return state.cchip.drir;
1157
1158 default:
1159 printf("Unknown CCHIP CSR %07x read attempted.\n", a);
1160 return 0;
1161 }

1164void CSystem::cchip_csr_write(u32 a, u64 data, CSystemComponent* source)
1165{
1166 CAlphaCPU* cpu = (CAlphaCPU*)source;
1167 switch (a)
1168 {
1169 case 0x000: // CSC
1170 // HRM 10.2.2.1, Table 10-10 "CSC (Typhoon Only)", p. 10-23:
1171 // AXD <39> RW, Init=0, "Disable memory XOR. (Typhoon only)"
1172 // Bit 39 is in nibble <39:36>; change that nibble from 0x7 to 0xf.
1173 state.cchip.csc &= ~U64(0x077777ffff3f0000);
1174 state.cchip.csc |= (data & U64(0x077777ffff3f0000));
1175 return;
1176
1177 case 0x080: // MISC
1178 {
1179 // Serialize the MISC RMW + IPI delivery against other CPUs and against
1180 // interrupt()/clear_clock_int(); irq_h() may signal the host wait but never
1181 // re-enters here. Host wait locks are released before timer service.
1182 std::lock_guard<std::mutex> g(drir_lock);
1183 state.cchip.misc |= (data & U64(0x00000f0000f00000)); // W1S
1184 state.cchip.misc &= ~(data & U64(0x0000000010000ff0)); // W1C
1185 if (data & U64(0x0000000001000000))
1186 {
1187 state.cchip.misc &= ~U64(0x0000000000ff0000); //Arbitration Clear
1188 printf("Arbitration clear from CPU %d (@%" PRIx64 ").\n", cpu->get_cpuid(),
1189 cpu->get_pc() - 4);
1190 }
1191
1192 if (data & U64(0x00000000000f0000))
1193 {
1194 printf("Arbitration %016" PRIx64 " from CPU %d (@%" PRIx64 ")... ", data,
1195 cpu->get_cpuid(), cpu->get_pc() - 4);
1196 if (!(state.cchip.misc & U64(0x00000000000f0000)))
1197 {
1198 state.cchip.misc |= (data & U64(0x00000000000f0000)); //Arbitration won
1199 printf("won %016" PRIx64 "\n", state.cchip.misc);
1200 }
1201 else
1202 printf("lost %016" PRIx64 "\n", state.cchip.misc);
1203 }
1204
1205 // stop interval timer interrupt
1206 if (data & U64(0x00000000000000f0))
1207 {
1208 for (int i = 0; i < iNumCPUs; i++)
1209 {
1210 if (data & (U64(0x10) << i))
1211 {
1212 acCPUs[i]->irq_h(2, false, 0);
1213
1214 //printf("*** TIMER interrupt cleared for CPU %d\n",i);
1215 }
1216 }
1217 }
1218
1219 // stop inter processor interrupt
1220 if (data & U64(0x0000000000000f00))
1221 {
1222 for (int i = 0; i < iNumCPUs; i++)
1223 {
1224 if (data & (U64(0x100) << i))
1225 {
1226 acCPUs[i]->irq_h(3, false, 0);
1227#ifdef DEBUG_IPI
1228 printf("*** IP interrupt cleared for CPU %d from CPU %d(@ %" PRIx64 ").\n",
1229 i, cpu->get_cpuid(), cpu->get_pc() - 4);
1230#endif
1231 }
1232 }
1233 }
1234
1235 // set inter processor interrupt
1236 if (data & U64(0x000000000000f000))
1237 {
1238 for (int i = 0; i < iNumCPUs; i++)
1239 {
1240 if (data & (U64(0x1000) << i))
1241 {
1242 state.cchip.misc |= U64(0x100) << i;
1243 acCPUs[i]->irq_h(3, true, 0);
1244#ifdef DEBUG_IPI
1245 printf("*** IP interrupt set for CPU %d from CPU %d(@ %" PRIx64 ")\n", i,
1246 cpu->get_cpuid(), cpu->get_pc() - 4);
1247#endif
1248
1249 // CThread::sleep(10);
1250 }
1251 }
1252 }
1253
1254 return;
1255 }
1256
1257 case 0x0c0: { // MPD
1258 // MPD: bit0=CKS (SCL driver: 1=release, 0=pull low)
1259 // bit1=DS (SDA driver: 1=release, 0=pull low)
1260 bool new_cks = (data & 0x1) != 0;
1261 bool new_ds = (data & 0x2) != 0;
1262 m_mpd.cks_out = new_cks;
1263 m_mpd.ds_out = new_ds;
1264 m_mpd_bus.drive_from_host(m_mpd.cks_out, m_mpd.ds_out); // SCL, SDA
1265 return;
1266 }
1267
1268
1269 case 0x200:
1270 case 0x240:
1271 case 0x600:
1272 case 0x640:
1273 {
1274 // A DIM change must re-drive the CPU device-interrupt lines
1275 std::lock_guard<std::mutex> g(drir_lock);
1276 state.cchip.dim[((a >> 9) & 2) | ((a >> 6) & 1)] = data;
1277 recompute_device_irqs();
1278 return;
1279 }
1280
1281 default:
1282 printf("Unknown CCHIP CSR %07x write with %016" PRIx64 " attempted.\n", a, data);
1283 }
1284}
